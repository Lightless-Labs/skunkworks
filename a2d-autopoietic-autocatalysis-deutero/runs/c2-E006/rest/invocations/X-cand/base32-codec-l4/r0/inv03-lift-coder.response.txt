#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    buffered: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            buffered: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut data = self.buffered.clone();
        data.extend_from_slice(input);

        let complete_len = (data.len() / 5) * 5;
        let output = encode_bytes(self.config.alphabet, &data[..complete_len]);

        self.buffered = data[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let output = encode_tail(self.config, &self.buffered);
        self.buffered.clear();
        self.closed = true;
        Ok(output)
    }
}

pub struct Decoder {
    config: Config,
    bit_buffer: u8,
    bit_count: u8,
    data_count: usize,
    last_symbol_value: u8,
    last_symbol_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding_count: usize,
    position: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            bit_buffer: 0,
            bit_count: 0,
            data_count: 0,
            last_symbol_value: 0,
            last_symbol_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding_count: 0,
            position: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;

            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                if next.first_padding_offset.is_none() {
                    let residue = next.data_count % 8;
                    let expected = match residue {
                        2 => 6,
                        4 => 4,
                        5 => 3,
                        7 => 1,
                        _ => return Err(StreamError::Padding { offset }),
                    };
                    next.first_padding_offset = Some(offset);
                    next.expected_padding_count = expected;
                }

                if next.padding_count >= next.expected_padding_count {
                    return Err(StreamError::Padding { offset });
                }

                next.padding_count += 1;
                next.position += 1;
                continue;
            }

            let (value, canonical_byte) = match classify(next.config.alphabet, byte) {
                Some(v) => v,
                None => return Err(StreamError::InvalidByte { offset, byte }),
            };

            if next.config.canonical && byte != canonical_byte {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: canonical_byte,
                });
            }

            if next.first_padding_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count = next.data_count.checked_add(1).ok_or(
                StreamError::InvalidLength { offset },
            )?;
            next.last_symbol_value = value;
            next.last_symbol_offset = Some(offset);

            next.bit_buffer = (next.bit_buffer << 5) | value;
            next.bit_count += 5;

            while next.bit_count >= 8 {
                next.bit_count -= 8;
                output.push((next.bit_buffer >> next.bit_count) & 0xff);
                if next.bit_count == 0 {
                    next.bit_buffer = 0;
                } else {
                    next.bit_buffer &= (1u8 << next.bit_count) - 1;
                }
            }

            next.position += 1;
        }

        self.copy_state_from(&next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };

        if unused_bits > 0 {
            let mask = (1u8 << unused_bits) - 1;
            if self.last_symbol_value & mask != 0 {
                return Err(StreamError::TrailingBits {
                    offset: self.last_symbol_offset.unwrap_or(self.position),
                });
            }
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            data_count: self.data_count,
            last_symbol_value: self.last_symbol_value,
            last_symbol_offset: self.last_symbol_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding_count: self.expected_padding_count,
            position: self.position,
            closed: self.closed,
        }
    }

    fn copy_state_from(&mut self, other: &Self) {
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.data_count = other.data_count;
        self.last_symbol_value = other.last_symbol_value;
        self.last_symbol_offset = other.last_symbol_offset;
        self.first_padding_offset = other.first_padding_offset;
        self.padding_count = other.padding_count;
        self.expected_padding_count = other.expected_padding_count;
        self.position = other.position;
        self.closed = other.closed;
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(alphabet: Alphabet, input: &[u8]) -> String {
    let symbols = alphabet_bytes(alphabet);
    let mut output = String::with_capacity((input.len() / 5) * 8);

    for chunk in input.chunks_exact(5) {
        let bits = ((chunk[0] as u64) << 32)
            | ((chunk[1] as u64) << 24)
            | ((chunk[2] as u64) << 16)
            | ((chunk[3] as u64) << 8)
            | (chunk[4] as u64);

        for shift in (0..8).rev() {
            output.push(symbols[((bits >> (shift * 5)) & 0x1f) as usize] as char);
        }
    }

    output
}

fn encode_tail(config: Config, input: &[u8]) -> String {
    if input.is_empty() {
        return String::new();
    }

    let symbols = alphabet_bytes(config.alphabet);
    let symbol_count = (input.len() * 8 + 4) / 5;
    let mut output = String::with_capacity(8);
    let mut accumulator = 0u32;
    let mut available = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | byte as u32;
        available += 8;
        while available >= 5 {
            available -= 5;
            let value = ((accumulator >> available) & 0x1f) as usize;
            output.push(symbols[value] as char);
        }
    }

    if available > 0 {
        let value = ((accumulator << (5 - available)) & 0x1f) as usize;
        output.push(symbols[value] as char);
    }

    if config.padding == Padding::Required {
        while output.len() < 8 && output.len() < symbol_count.max(8) {
            output.push('=');
        }
    }

    output
}

fn classify(alphabet: Alphabet, byte: u8) -> Option<(u8, u8)> {
    let uppercase = byte.to_ascii_uppercase();
    let value = alphabet_bytes(alphabet)
        .iter()
        .position(|&candidate| candidate == uppercase)?;
    Some((value as u8, uppercase))
}
