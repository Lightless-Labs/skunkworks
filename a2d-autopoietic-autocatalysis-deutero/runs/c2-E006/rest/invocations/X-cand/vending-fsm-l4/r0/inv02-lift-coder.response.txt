#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
    pub maintenance: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
    MaintenanceActive,
    MaintenanceRequired,
    AlreadyMode,
    ReconcileMismatch { coin: u16, expected: u16, found: u16 },
    ValueDrift { before: u64, after: u64 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
    Maintenance { enabled: bool },
    Reconciled { before: [u16; 4], after: [u16; 4] },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Audit {
    pub inserted: u64,
    pub refunded: u64,
    pub sales: u64,
    pub change: u64,
    pub adjustment: i64,
    pub cash: u64,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<usize>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn coin_index(coin: u16) -> Option<usize> {
    match coin {
        1 => Some(0),
        3 => Some(1),
        4 => Some(2),
        10 => Some(3),
        _ => None,
    }
}

fn valid_id(id: &str) -> bool {
    !id.is_empty() && id.len() <= 8 && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn tube_value(tubes: &[u16; 4]) -> u64 {
    tubes.iter().zip(DENOMS).map(|(&count, coin)| u64::from(count) * u64::from(coin)).sum()
}

fn event_push(events: &mut Vec<Event>, kind: EventKind) {
    let seq = events.len();
    events.push(Event { seq, kind });
}

fn change_value(coins: &[u16]) -> u64 {
    coins.iter().map(|&coin| u64::from(coin)).sum()
}

fn search_change(
    denom_pos: usize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if denom_pos == 4 {
        if remaining == 0 {
            let candidate = *counts;
            let candidate_total: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
            let better = match best {
                None => true,
                Some(current) => {
                    let current_total: u32 = current.iter().map(|&n| u32::from(n)).sum();
                    if candidate_total != current_total {
                        candidate_total < current_total
                    } else {
                        // Compare descending denomination counts lexicographically.
                        for i in (0..4).rev() {
                            if candidate[i] != current[i] {
                                return_if_better(candidate[i] > current[i], best, candidate);
                                return;
                            }
                        }
                        false
                    }
                }
            };
            if better {
                *best = Some(candidate);
            }
        }
        return;
    }

    let idx = 3 - denom_pos;
    let coin = DENOMS[idx];
    let max_count = tubes[idx].min(remaining / coin);
    for n in 0..=max_count {
        counts[idx] = n;
        search_change(
            denom_pos + 1,
            remaining - n * coin,
            tubes,
            counts,
            best,
        );
    }
    counts[idx] = 0;
}

fn return_if_better(is_better: bool, best: &mut Option<[u16; 4]>, candidate: [u16; 4]) {
    if is_better {
        *best = Some(candidate);
    }
}

fn make_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for idx in (0..4).rev() {
        let take = tubes[idx].min(remaining / DENOMS[idx]);
        greedy[idx] = take;
        remaining -= take * DENOMS[idx];
    }
    if remaining == 0 {
        return Some(counts_to_descending_coins(&greedy));
    }

    let mut best = None;
    search_change(0, amount, tubes, &mut [0; 4], &mut best);
    best.map(|counts| counts_to_descending_coins(&counts))
}

fn counts_to_descending_coins(counts: &[u16; 4]) -> Vec<u16> {
    let capacity: usize = counts.iter().map(|&n| usize::from(n)).sum();
    let mut coins = Vec::with_capacity(capacity);
    for idx in (0..4).rev() {
        coins.extend(std::iter::repeat_n(DENOMS[idx], usize::from(counts[idx])));
    }
    coins
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for (i, product) in products.iter().enumerate() {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if products[..i].iter().any(|p| p.id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (idx, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[idx] });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.map(|i| self.products[i].id.clone()),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if coin_index(coin).is_none() {
            return Err(Error::BadCoin { coin });
        }
        let new_credit = u32::from(self.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(u64::from(coin));
        event_push(
            &mut self.events,
            EventKind::Inserted {
                coin,
                credit: self.credit,
            },
        );
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected == Some(index) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(index);
        event_push(
            &mut self.events,
            EventKind::Selected {
                id: self.products[index].id.clone(),
            },
        );
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(index) = self.selected else {
            return Err(Error::NoSelection);
        };

        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut {
                id: product.id.clone(),
            });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let id = product.id.clone();
        let price = product.price;
        let stock = product.stock;
        let owed = self.credit - price;
        let change = make_change(owed, &self.tubes).ok_or(Error::NoChange { amount: owed })?;

        let mut proposed = self.tubes;
        for &coin in &change {
            let idx = coin_index(coin).expect("change uses valid denominations");
            proposed[idx] -= 1;
        }
        for &coin in &self.escrow {
            let idx = coin_index(coin).expect("escrow uses accepted denominations");
            proposed[idx] = proposed[idx].saturating_add(1);
        }
        for (idx, &count) in proposed.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[idx] });
            }
        }

        self.tubes = proposed;
        self.products[index].stock = stock - 1;
        self.credit = 0;
        self.escrow.clear();
        self.selected = None;
        self.sales = self.sales.saturating_add(u64::from(price));
        let change_amount = change_value(&change);
        self.change = self.change.saturating_add(change_amount);
        event_push(
            &mut self.events,
            EventKind::Vended {
                id: id.clone(),
                price,
                change: change.clone(),
            },
        );

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let refunded_amount = u64::from(self.credit);

        self.credit = 0;
        self.escrow.clear();
        self.selected = None;
        self.refunded = self.refunded.saturating_add(refunded_amount);
        event_push(
            &mut self.events,
            EventKind::Refunded {
                coins: coins.clone(),
            },
        );
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        event_push(
            &mut self.events,
            EventKind::PriceChanged {
                id: id.to_owned(),
                old,
                new: price,
            },
        );
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let resulting = u32::from(self.products[index].stock) + u32::from(count);
        if resulting > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        let stock = resulting as u16;
        self.products[index].stock = stock;
        event_push(
            &mut self.events,
            EventKind::Restocked {
                id: id.to_owned(),
                added: count,
                stock,
            },
        );
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }

        self.maintenance = enabled;
        self.selected = None;
        event_push(&mut self.events, EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value(&self.tubes).saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(
        &mut self,
        expected: [u16; 4],
        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }

        for idx in 0..4 {
            if expected[idx] != self.tubes[idx] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[idx],
                    expected: expected[idx],
                    found: self.tubes[idx],
                });
            }
        }

        for idx in 0..4 {
            if actual[idx] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[idx] });
            }
        }

        let before = tube_value(&self.tubes);
        let after = tube_value(&actual);
        if before.abs_diff(after) > 20 {
            return Err(Error::ValueDrift { before, after });
        }

        let delta = i128::from(after) - i128::from(before);
        let delta_i64 = delta as i64;
        self.adjustment = self.adjustment.saturating_add(delta_i64);
        let previous = self.tubes;
        self.tubes = actual;
        event_push(
            &mut self.events,
            EventKind::Reconciled {
                before: previous,
                after: actual,
            },
        );
        Ok(())
    }
}
