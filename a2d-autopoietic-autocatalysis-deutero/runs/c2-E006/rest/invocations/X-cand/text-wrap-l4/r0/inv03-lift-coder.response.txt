#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    consumed: usize,
    marker_ended: bool,
    raw: bool,
    gap: usize,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    markers: usize,
    lines: Vec<Line>,
    counts: Vec<usize>,
    strings: Vec<String>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Ending {
    Complete,
    Marker(usize),
    Raw(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.markers != b.markers {
        return a.markers < b.markers;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.counts.iter().zip(&b.counts) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.strings.iter().zip(&b.strings) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn render_pre(line: &Line) -> String {
    let mut s = " ".repeat(line.indent);
    for (i, p) in line.pieces.iter().enumerate() {
        if i != 0 {
            s.push(' ');
        }
        s.push_str(p);
    }
    s
}

fn optimize(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    let width = options.width;
    let mut memo: Vec<Vec<[Option<Layout>; 2]>> = words
        .iter()
        .map(|w| {
            (0..=w.chars.len())
                .map(|_| [None, None])
                .collect()
        })
        .collect();
    let done = Layout {
        cost: 0,
        markers: 0,
        lines: Vec::new(),
        counts: Vec::new(),
        strings: Vec::new(),
    };

    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Vec<[Option<Layout>; 2]>],
        done: &Layout,
    ) -> Layout {
        if wi >= words.len() {
            return done.clone();
        }
        if let Some(v) = &memo[wi][off][usize::from(!first)] {
            return v.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut candidates: Vec<(Vec<String>, usize, usize, Ending, bool)> = Vec::new();
        let mut used = 0usize;
        let mut pieces: Vec<String> = Vec::new();
        let mut idx = wi;
        let mut start = off;
        let mut can_continue = true;

        while can_continue && idx < words.len() {
            let word = &words[idx];
            let remain = word.chars.len() - start;

            if pieces.is_empty() && remain > cap {
                let marker_fits = word.markers.iter().any(|&m| {
                    m > start && m < word.chars.len() && m - start + 1 <= cap
                });
                if !marker_fits {
                    let take = cap;
                    let fragment: String = word.chars[start..start + take].iter().collect();
                    let next = if start + take == word.chars.len() {
                        (idx + 1, 0, false)
                    } else {
                        (idx, start + take, false)
                    };
                    candidates.push((
                        vec![fragment],
                        take,
                        1,
                        Ending::Raw(next.0 * (words.first().map_or(1, |_| 1)) + next.1),
                        true,
                    ));
                    // The packed raw destination is replaced below when constructing the candidate.
                    let _ = next;
                }
            }

            let available = cap.saturating_sub(used + usize::from(!pieces.is_empty()));
            if remain <= available {
                let mut next_pieces = pieces.clone();
                next_pieces.push(word.chars[start..].iter().collect());
                candidates.push((next_pieces, used + usize::from(!pieces.is_empty()) + remain, 0, Ending::Complete, false));
            }

            for &m in &word.markers {
                if m > start && m < word.chars.len() {
                    let prefix_len = m - start;
                    let needed = usize::from(!pieces.is_empty()) + prefix_len + 1;
                    if used + needed <= cap {
                        let mut next_pieces = pieces.clone();
                        let mut prefix: String = word.chars[start..m].iter().collect();
                        prefix.push('-');
                        next_pieces.push(prefix);
                        candidates.push((
                            next_pieces,
                            used + needed,
                            idx - wi + 1,
                            Ending::Marker(m),
                            false,
                        ));
                    }
                }
            }

            if remain > available {
                can_continue = false;
            } else {
                used += usize::from(!pieces.is_empty()) + remain;
                pieces.push(word.chars[start..].iter().collect());
                idx += 1;
                start = 0;
            }
        }

        let mut best: Option<Layout> = None;
        for (parts, display_len, piece_count, ending, is_raw) in candidates {
            let (nwi, noff, nfirst, marker_ended, raw, consumed) = match ending {
                Ending::Complete => {
                    // A complete candidate ends at the last piece included in `parts`.
                    // Reconstruct its destination from its display traversal.
                    let mut left = display_len;
                    let mut dest_wi = wi;
                    let mut dest_off = off;
                    let mut n_pieces = 0usize;
                    while dest_wi < words.len() {
                        if n_pieces >= parts.len() {
                            break;
                        }
                        let len = words[dest_wi].chars.len() - dest_off;
                        if left >= len {
                            left -= len;
                            dest_wi += 1;
                            dest_off = 0;
                            n_pieces += 1;
                            if n_pieces == parts.len() {
                                break;
                            }
                            left = left.saturating_sub(1);
                        } else {
                            break;
                        }
                    }
                    (dest_wi, dest_off, false, false, false, display_len)
                }
                Ending::Marker(m) => (wi, m, false, true, false, display_len.saturating_sub(1)),
                Ending::Raw(_) => {
                    let take = cap;
                    if off + take == words[wi].chars.len() {
                        (wi + 1, 0, false, false, true, take)
                    } else {
                        (wi, off + take, false, false, true, take)
                    }
                }
            };

            let continuation = if nwi >= words.len() {
                done.clone()
            } else {
                solve(nwi, noff, nfirst, words, options, memo, done)
            };
            let line = Line {
                indent,
                pieces: parts,
                consumed,
                marker_ended,
                raw,
                gap: width - indent - display_len,
            };
            let mut layout = continuation;
            let final_line = layout.lines.is_empty();
            let gap_sq = (line.gap as u64) * (line.gap as u64);
            let mut own_cost = if final_line { gap_sq / 4 } else { gap_sq };
            if marker_ended {
                own_cost += options.hyphen_cost as u64;
                layout.markers += 1;
            }
            if raw {
                own_cost += (2 * width + 1) as u64;
            }
            if !final_line && piece_count == 1 {
                own_cost += 3;
            }
            layout.cost += own_cost;
            layout.lines.insert(0, line.clone());
            layout.counts.insert(0, consumed);
            layout.strings.insert(0, render_pre(&line));
            if best.as_ref().is_none_or(|b| better(&layout, b)) {
                best = Some(layout);
            }
            let _ = is_raw;
        }

        let result = best.unwrap_or_else(|| done.clone());
        memo[wi][off][usize::from(!first)] = Some(result.clone());
        result
    }

    if words.is_empty() {
        Vec::new()
    } else {
        solve(0, 0, true, words, options, &mut memo, &done).lines
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_neighbor = |j: usize| {
                j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
            };
            if i == 0 || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut paragraphs: Vec<Vec<Word>> = Vec::new();
    let mut current_segment: Vec<char> = Vec::new();
    let mut boundaries = 0usize;
    let mut segments: Vec<(Vec<char>, usize)> = Vec::new();
    i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                segments.push((std::mem::take(&mut current_segment), j - i - 1));
            } else {
                current_segment.push('\n');
            }
            i = j;
        } else {
            current_segment.push(normalized[i]);
            i += 1;
        }
    }
    segments.push((current_segment, 0));

    let mut output = Vec::new();
    for (segment, boundary_count) in segments {
        let mut words = Vec::new();
        let mut chars_word = Vec::new();
        let mut markers = Vec::new();
        for c in segment {
            if is_separator(c) {
                if !chars_word.is_empty() {
                    words.push(Word { chars: std::mem::take(&mut chars_word), markers: std::mem::take(&mut markers) });
                }
            } else if c == '~' {
                markers.push(chars_word.len());
            } else {
                chars_word.push(c);
            }
        }
        if !chars_word.is_empty() {
            words.push(Word { chars: chars_word, markers });
        }
        let lines = optimize(&words, options);
        let last = lines.len().saturating_sub(1);
        for (line_index, line) in lines.iter().enumerate() {
            let mut s = " ".repeat(line.indent);
            let gaps = line.pieces.len().saturating_sub(1);
            let extra = if options.justify && line_index != last && gaps > 0 {
                line.gap
            } else {
                0
            };
            if gaps == 0 || extra == 0 {
                for (k, p) in line.pieces.iter().enumerate() {
                    if k > 0 {
                        s.push(' ');
                    }
                    s.push_str(p);
                }
            } else {
                let base = extra / gaps;
                let rem = extra % gaps;
                let start = line_index % gaps;
                for (k, p) in line.pieces.iter().enumerate() {
                    if k > 0 {
                        let gap = k - 1;
                        let bonus = usize::from((gap + gaps - start) % gaps < rem);
                        s.push_str(&" ".repeat(1 + base + bonus));
                    }
                    s.push_str(p);
                }
            }
            output.push(s);
        }
        for _ in 0..boundary_count {
            output.push(String::new());
        }
    }
    Ok(output)
}
