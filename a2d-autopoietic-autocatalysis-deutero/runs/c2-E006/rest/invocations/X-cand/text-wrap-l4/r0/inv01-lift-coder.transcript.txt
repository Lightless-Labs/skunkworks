                gaps: gap_count,
            },
        );
        candidate.consumed_vector.insert(0, used);

        if best_first.as_ref().is_none_or(|old| better(&candidate, old)) {
            best_first = Some(candidate);
        }
    }

    let mut layout = best_first.unwrap_or(Layout {
        cost: 0,
        marker_count: 0,
        lines: Vec::new(),
        consumed_vector: Vec::new(),
    });

    if options.justify && layout.lines.len() > 1 {
        for i in 0..layout.lines.len() - 1 {
            let line = &mut layout.lines[i];
            if line.pieces < 2 || line.gaps == 0 {
                continue;
            }
            let used = line.text.chars().count();
            let extra = width - used;
            let base = extra / line.gaps;
            let remainder = extra % line.gaps;
            let indent_len = if i == 0 { options.first_indent } else { options.rest_indent };
            let body: String = line.text.chars().skip(indent_len).collect();
            let parts: Vec<&str> = body.split(' ').collect();
            if parts.len() != line.gaps + 1 {
                continue;
            }
            let mut rebuilt = " ".repeat(indent_len);
            for (g, part) in parts.iter().enumerate() {
                if g > 0 {
                    let start = i % line.gaps;
                    let distance = (g - 1 + line.gaps - start) % line.gaps;
                    rebuilt.push_str(&" ".repeat(1 + base + usize::from(distance < remainder)));
                }
                rebuilt.push_str(part);
            }
            line.text = rebuilt;
        }
    }

    layout.lines.into_iter().map(|line| line.text).collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_neighbor = |j: usize| {
                j < chars.len() && chars[j].1 != '~' && !separators(chars[j].1)
            };
            if i == 0 || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut pos = 0;

    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let mut end = pos;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - pos >= 2 {
                let words = parse_words(&segment);
                output.extend(solve_paragraph(&words, options));
                segment.clear();
                for _ in 1..(end - pos) {
                    output.push(String::new());
                }
                pos = end;
                continue;
            }
        }
        segment.push(normalized[pos]);
        pos += 1;
    }

    let words = parse_words(&segment);
    output.extend(solve_paragraph(&words, options));
    Ok(output)
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut spelling = Vec::new();
    let mut markers = Vec::new();

    for &c in chars {
        if separators(c) {
            if !spelling.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut spelling),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(spelling.len());
        } else {
            spelling.push(c);
        }
    }

    if !spelling.is_empty() {
        words.push(Word { chars: spelling, markers });
    }
    words
}
