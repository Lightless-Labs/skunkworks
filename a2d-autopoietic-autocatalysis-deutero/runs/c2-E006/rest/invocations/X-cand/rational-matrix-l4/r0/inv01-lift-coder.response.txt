```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        if i == bytes.len() {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            i += 1;
            let start = i;
            if i == bytes.len() || !bytes[i].is_ascii_digit() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            den_start = Some(start);
            den_end = Some(i);
            if i < bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(parse_err(
                if bytes[0] == b'-' { 2 } else { 1 },
                ParseKind::NonCanonical,
            ));
        }
        if bytes[0] == b'-' && &text[..num_end] == "-0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let (Some(s), Some(e)) = (den_start, den_end) {
            if e - s >= 2 && bytes[s] == b'0' {
                return Err(parse_err(s + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let mut nmag: u128 = 0;
        for &b in &bytes[num_start..num_end] {
            nmag = nmag * 10 + (b - b'0') as u128;
            if nmag > (1u128 << 63) {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }
        if !negative && nmag > i64::MAX as u128 {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }

        let mut dmag: u128 = 1;
        if let (Some(s), Some(e)) = (den_start, den_end) {
            dmag = 0;
            for &b in &bytes[s..e] {
                dmag = dmag * 10 + (b - b'0') as u128;
                if dmag > i64::MAX as u128 {
                    return Err(parse_err(s, ParseKind::OutOfRange));
                }
            }
        }

        if let Some(s) = den_start {
            if dmag == 0 {
                return Err(parse_err(s, ParseKind::ZeroDenominator));
            }
            let slash = s - 1;
            if dmag == 1 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
            if nmag == 0 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(nmag, dmag) > 1 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
        }

        let signed_n = if negative {
            if nmag == (1u128 << 63) {
                i64::MIN as i128
            } else {
                -(nmag as i128)
            }
        } else {
            nmag as i128
        };
        Ok(Rational {
            num: signed_n as i64,
            den: dmag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

fn zero() -> Rational {
    Rational { num: 0, den: 1 }
}

fn one() -> Rational {
    Rational { num: 1, den: 1 }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    fn at(&self, row: usize, col: usize) -> Rational {
        self.data[row * self.cols + col]
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.at(row, col))
    }

    pub fn transpose(&self) -> Self {
        let mut data = vec![zero(); self.data.len()];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[c * self.rows + r] = self.at(r, c);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero();
                for k in 0..self.cols {
                    let product = self.at(r, k).mul(&rhs.at(k, c)).map_err(|source| {
                        MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: k,
                            source,
                        }
                    })?;
                    acc = acc.add(&product).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        mut data: Vec<Rational>,
        rows: usize,
        cols: usize,
        pivot_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut p = 0;
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();

        for col in 0..pivot_cols {
            if p == rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for r in p..rows {
                let value = data[r * cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), r);
                if chosen.is_none_or(|q| {
                    let v = data[q * cols + col];
                    key < (v.den, v.num.unsigned_abs(), q)
                }) {
                    chosen = Some(r);
                }
            }
            let Some(q) = chosen else {
                continue;
            };

            if q != p {
                for c in 0..cols {
                    data.swap(p * cols + c, q * cols + c);
                }
                swaps.push((p, q));
            }
            let pivot = data[p * cols + col];

            for c in 0..cols {
                let old = data[p * cols + c];
                data[p * cols + c] = old.div(&pivot).map_err(|source| {
                    MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    }
                })?;
            }

            for r in 0..rows {
                if r == p {
                    continue;
                }
                let factor = data[r * cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..cols {
                    let product = factor
                        .mul(&data[p * cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        })?;
                    data[r * cols + c] = data[r * cols + c]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        })?;
                }
            }
            pivots.push(col);
            p += 1;
        }

        Ok(Reduction {
            matrix: Matrix { rows, cols, data },
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        Self::eliminate(self.data.clone(), self.rows, self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.at(r, c));
            }
            for c in 0..n {
                data.push(if r == c { one() } else { zero() });
            }
        }
        let red = Self::eliminate(data, n, width, n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: red.pivots.len(),
            });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(red.matrix.at(r, n + c));
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: result,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

impl Matrix {
    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let m = self.rows;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented = Vec::with_capacity(len);
        for r in 0..m {
            for c in 0..n {
                augmented.push(self.at(r, c));
            }
            for c in 0..k {
                augmented.push(rhs.at(r, c));
            }
        }

        let red = Matrix::eliminate(augmented, m, width, n).map_err(SolveError::Matrix)?;

        for r in 0..m {
            let coeff_zero = (0..n).all(|c| red.matrix.at(r, c).num == 0);
            if coeff_zero {
                for c in 0..k {
                    if red.matrix.at(r, n + c).num != 0 {
                        return Ok(Solution::None {
                            row: r,
                            rhs_col: c,
                            witness: red.matrix.data[r * width..(r + 1) * width].to_vec(),
                        });
                    }
                }
            }
        }

        let pivots = red.pivots;
        let rank = pivots.len();
        let mut is_pivot = vec![false; n];
        for &c in &pivots {
            is_pivot[c] = true;
        }
        let free: Vec<usize> = (0..n).filter(|&c| !is_pivot[c]).collect();

        let particular_len = n.checked_mul(k).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let mut particular_data = vec![zero(); particular_len];
        for (j, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] = red.matrix.at(j, n + c);
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let basis_len = n.checked_mul(free.len()).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let mut basis_data = vec![zero(); basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + j] = one();
            for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
                let coefficient = red.matrix.at(pivot_row, free_col);
                let negated = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: rank,
                        source,
                    })
                })?;
                basis_data[pivot_col * free.len() + j] = negated;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: basis_data.len().checked_div(n.max(1)).unwrap_or(0),
                data: basis_data,
            },
            pivots,
        })
    }
}
```
