use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn signed_from_magnitude(magnitude: u128, negative: bool) -> Result<i64, RationalError> {
    if negative {
        if magnitude > (1u128 << 63) {
            return Err(RationalError::Overflow);
        }
        if magnitude == (1u128 << 63) {
            Ok(i64::MIN)
        } else {
            Ok(-(magnitude as i64))
        }
    } else {
        if magnitude > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(magnitude as i64)
    }
}

fn rational_from_signed_magnitudes(
    num_mag: u128,
    den_mag: u128,
    negative: bool,
) -> Result<Rational, RationalError> {
    if den_mag == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num_mag == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(num_mag, den_mag);
    let n = num_mag / g;
    let d = den_mag / g;

    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    let num = signed_from_magnitude(n, negative)?;
    Ok(Rational {
        num,
        den: d as i64,
    })
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let negative = (num < 0) ^ (den < 0);
    rational_from_signed_magnitudes(num.unsigned_abs(), den.unsigned_abs(), negative)
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_signed_magnitudes(
            num.unsigned_abs() as u128,
            den.unsigned_abs() as u128,
            (num < 0) ^ (den < 0),
        )
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.num as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let negative = (self.num < 0) ^ (rhs.num < 0);
        let n = (self.num.unsigned_abs() as u128) * (rhs.den as u128);
        let d = (self.den as u128) * (rhs.num.unsigned_abs() as u128);
        rational_from_signed_magnitudes(n, d, negative)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let num = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let negative = bytes[0] == b'-';
        let numerator_digits = &bytes[num_start..num_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if negative && numerator_digits == b"0" {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(RationalError::Parse {
                    at: start + 1,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let mut num_mag: u128 = 0;
        let num_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        for &b in numerator_digits {
            num_mag = num_mag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
            if num_mag > num_limit {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: ParseKind::OutOfRange,
                });
            }
        }

        let mut den_mag = 1u128;
        if let (Some(start), Some(end)) = (den_start, den_end) {
            den_mag = 0;
            for &b in &bytes[start..end] {
                den_mag = den_mag
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
                if den_mag > i64::MAX as u128 {
                    return Err(RationalError::Parse {
                        at: start,
                        kind: ParseKind::OutOfRange,
                    });
                }
            }
            if den_mag == 0 {
                return Err(RationalError::Parse {
                    at: start,
                    kind: ParseKind::ZeroDenominator,
                });
            }
            let slash = start - 1;
            if den_mag == 1 {
                return Err(RationalError::Parse {
                    at: slash,
                    kind: ParseKind::NonCanonical,
                });
            }
            if num_mag == 0 {
                return Err(RationalError::Parse {
                    at: slash,
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(RationalError::Parse {
                    at: slash,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        rational_from_signed_magnitudes(num_mag, den_mag, negative).map_err(|_| {
            RationalError::Parse {
                at: 0,
                kind: ParseKind::OutOfRange,
            }
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

fn checked_len(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

fn zero() -> Rational {
    Rational { num: 0, den: 1 }
}

fn one() -> Rational {
    Rational { num: 1, den: 1 }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = checked_len(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = vec![zero(); self.data.len()];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[c * self.rows + r] = self.data[r * self.cols + c];
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = checked_len(self.rows, rhs.cols)?;
        let mut out = Vec::with_capacity(out_len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero();
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                out.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data: out,
        })
    }

    fn eliminate(
        &self,
        candidate_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                match chosen {
                    None => chosen = Some(row),
                    Some(old) => {
                        let old_value = matrix.data[old * matrix.cols + col];
                        let key = (
                            value.den,
                            value.num.unsigned_abs(),
                            row,
                        );
                        let old_key = (
                            old_value.den,
                            old_value.num.unsigned_abs(),
                            old,
                        );
                        if key.cmp(&old_key) == Ordering::Less {
                            chosen = Some(row);
                        }
                    }
                }
            }
            let Some(q) = chosen else {
                continue;
            };

            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..matrix.cols {
                let value = matrix.data[p * matrix.cols + c];
                matrix.data[p * matrix.cols + c] =
                    value.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let product = factor
                        .mul(&matrix.data[p * matrix.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    matrix.data[row * matrix.cols + c] = matrix.data[row * matrix.cols + c]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                }
            }

            pivots.push(col);
            p += 1;
        }

        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = checked_len(n, width)?;
        let mut data = vec![zero(); len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = one();
        }

        let augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let red = augmented.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: red.pivots.len(),
            });
        }

        let mut inverse = vec![zero(); n * n];
        for r in 0..n {
            for c in 0..n {
                inverse[r * n + c] = red.matrix.data[r * width + n + c];
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let width = self
            .cols
            .checked_add(rhs.cols)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = vec![zero(); len];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[r * width + c] = self.data[r * self.cols + c];
            }
            for c in 0..rhs.cols {
                data[r * width + self.cols + c] = rhs.data[r * rhs.cols + c];
            }
        }
        let augmented = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let red = augmented
            .eliminate(self.cols)
            .map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let coeff_zero = (0..self.cols)
                .all(|c| red.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                for rhs_col in 0..rhs.cols {
                    let value = red.matrix.data[row * width + self.cols + rhs_col];
                    if value.num != 0 {
                        let witness = red.matrix.data[row * width..(row + 1) * width].to_vec();
                        return Ok(Solution::None {
                            row,
                            rhs_col,
                            witness,
                        });
                    }
                }
            }
        }

        let n = self.cols;
        let k = rhs.cols;
        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![zero(); particular_len];

        for (pivot_row, &pivot_col) in red.pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular[pivot_col * k + rhs_col] =
                    red.matrix.data[pivot_row * width + n + rhs_col];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular,
        };

        if red.pivots.len() == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: red.pivots,
            });
        }

        let mut free = Vec::new();
        let mut is_pivot = vec![false; n];
        for &pivot in &red.pivots {
            is_pivot[pivot] = true;
        }
        for c in 0..n {
            if !is_pivot[c] {
                free.push(c);
            }
        }

        let f = free.len();
        let basis_len = n
            .checked_mul(f)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero(); basis_len];

        for (basis_col, &free_col) in free.iter().enumerate() {
            basis[free_col * f + basis_col] = one();
            for (pivot_row, &pivot_col) in red.pivots.iter().enumerate() {
                let coefficient = red.matrix.data[pivot_row * width + free_col];
                basis[pivot_col * f + basis_col] =
                    coefficient.neg().map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: basis_col,
                            step: red.pivots.len(),
                            source,
                        })
                    })?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: f,
                data: basis,
            },
            pivots: red.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
