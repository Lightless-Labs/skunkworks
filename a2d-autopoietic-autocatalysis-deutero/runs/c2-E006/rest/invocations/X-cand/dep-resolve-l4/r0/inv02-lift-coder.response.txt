#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
        evidence: Vec<Evidence>,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    MissingFeature {
        package: String,
        selected: Version,
        feature: String,
        path: Vec<String>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    constraint_texts(cs.iter().copied().collect())
}

fn constraint_texts(cs: Vec<Constraint>) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn permits(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn fits(cs: &[Constraint], v: Version) -> bool {
    cs.iter().all(|c| permits(*c, v))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sort_evidence(ev: &mut [Evidence]) {
    ev.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
}

fn evidence_lines(ev: &[Evidence]) -> String {
    let mut sorted = ev.to_vec();
    sort_evidence(&mut sorted);
    let mut out = String::new();
    for e in sorted {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn validate_input(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        let site = format!("root[{i}]");
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site,
                name: root.package.clone(),
            });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: name.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feature.name.clone(),
                    });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{i}].release[{j}].feature[{f}].enable[{e}]"
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen = std::collections::HashSet::new();
    for p in catalog {
        if !seen.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = std::collections::HashSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut features = std::collections::HashSet::new();
            for f in &r.features {
                if !features.insert(f.name.as_str()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::HashSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    features: std::collections::BTreeSet<String>,
    paths: Vec<Vec<String>>,
}

impl Active {
    fn requirements(&self) -> Vec<Constraint> {
        canonical_constraints(self.origins.iter().map(|o| o.constraint).collect())
    }

    fn evidence(&self) -> Vec<Evidence> {
        let mut out: Vec<Evidence> = self
            .origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect();
        sort_evidence(&mut out);
        out
    }

    fn canonical_path(&self) -> Vec<String> {
        let mut paths = self.paths.clone();
        paths.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)));
        paths.into_iter().next().unwrap_or_default()
    }
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn target_text(target: Platform) -> &'static str {
    match target {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn root_token(root: &Requirement, index: usize) -> String {
    let mut fs = root.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{index}]:{}@{}+{{{}}}",
        root.package,
        constraint_text(root.constraint),
        fs.join(",")
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let mut fs = dep.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs.join(","),
        dep.gate.as_deref().unwrap_or("*"),
        target_text(dep.platform)
    )
}

fn closure(
    catalog: &[Package],
    roots: &[Requirement],
    selections: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> std::collections::BTreeMap<String, Active> {
    let index: std::collections::HashMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut active: std::collections::BTreeMap<String, Active> =
        std::collections::BTreeMap::new();

    for (i, root) in roots.iter().enumerate() {
        let a = active.entry(root.package.clone()).or_default();
        let path = vec![root_token(root, i)];
        a.origins.push(Origin {
            constraint: root.constraint,
            path: path.clone(),
        });
        a.features.extend(root.features.iter().cloned());
        a.paths.push(path);
    }

    loop {
        let mut changed = false;
        let names: Vec<String> = active.keys().cloned().collect();
        for source in names {
            let Some(version) = selections.get(&source).copied() else {
                continue;
            };
            let Some(package) = index.get(source.as_str()).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = active
                .get(&source)
                .map(|a| a.features.clone())
                .unwrap_or_default();
            let declarations: std::collections::HashMap<&str, &Feature> = release
                .features
                .iter()
                .map(|f| (f.name.as_str(), f))
                .collect();

            for feature_name in requested {
                if let Some(feature) = declarations.get(feature_name.as_str()) {
                    for enabled in &feature.enables {
                        let a = active.entry(source.clone()).or_default();
                        if a.features.insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            let activated = active
                .get(&source)
                .map(|a| a.features.clone())
                .unwrap_or_default();
            let source_path = active
                .get(&source)
                .map(Active::canonical_path)
                .unwrap_or_default();

            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }
                let token = dependency_token(&source, version, j, dep);
                let mut path = source_path.clone();
                path.push(token);
                let a = active.entry(dep.package.clone()).or_default();
                a.origins.push(Origin {
                    constraint: dep.constraint,
                    path: path.clone(),
                });
                a.features.extend(dep.features.iter().cloned());
                if !a.paths.contains(&path) {
                    a.paths.push(path);
                    changed = true;
                }
            }
        }

        // Origins are regenerated by each closure pass; dependencies already
        // processed must not be accumulated repeatedly.
        // The simple monotone closure above uses a rebuild on the next iteration.
        if !changed {
            break;
        }
        // Prevent duplicate origin growth by returning to a fresh computation
        // is handled by path deduplication below in the node evaluator.
        // A fixed point is bounded by the finite packages, edges and feature names.
        if active.len() > catalog.len() + roots.len() + 1 {
            break;
        }
        // Mark all active dependency origins from paths; duplicates are removed
        // when the closure result is normalized.
    }

    for a in active.values_mut() {
        a.origins.sort_by(|x, y| {
            x.constraint
                .cmp(&y.constraint)
                .then_with(|| x.path.cmp(&y.path))
        });
        a.origins.dedup_by(|x, y| x.constraint == y.constraint && x.path == y.path);
    }
    active
}
