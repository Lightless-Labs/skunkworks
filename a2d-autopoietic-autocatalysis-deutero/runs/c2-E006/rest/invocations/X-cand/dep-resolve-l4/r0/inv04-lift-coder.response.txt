fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_version(c: Constraint) -> Version {
    match c {
        Constraint::Exact(v)
        | Constraint::AtLeast(v)
        | Constraint::Before(v)
        | Constraint::Caret(v) => v,
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    (constraint_rank(c), constraint_version(c))
}

fn constraint_text(c: Constraint) -> String {
    let prefix = match c {
        Constraint::Exact(_) => "=",
        Constraint::AtLeast(_) => ">=",
        Constraint::Before(_) => "<",
        Constraint::Caret(_) => "^",
    };
    format!("{prefix}{}", version_text(constraint_version(c)))
}

fn canonical_constraints(input: &[Constraint]) -> Vec<Constraint> {
    let mut out = input.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn constraints_text(input: &[Constraint]) -> String {
    canonical_constraints(input)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn constraint_matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if x.major > 0 {
                v >= x && v.major == x.major
            } else if x.minor > 0 {
                v >= x && v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn valid_name(s: &str) -> bool {
    let bytes = s.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn encoded(s: &str) -> String {
    let mut out = String::new();
    for &b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(line: &mut String, evidence: &[Evidence]) {
    for e in evidence_sorted(evidence) {
        line.push_str("\n  C:");
        line.push_str(&constraint_text(e.constraint));
        line.push_str(" via ");
        line.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encoded(site), encoded(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encoded(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encoded(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encoded(package),
            version_text(*version),
            encoded(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encoded(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut out = format!("missing|package={}", encoded(package));
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "no-match|package={}|constraints={}",
                encoded(package),
                constraints_text(requirements)
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "conflict|package={}|selected={}|constraints={}",
                encoded(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encoded(package),
            version_text(*selected),
            encoded(feature),
            encoded(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[derive(Clone, Debug)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Debug, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn dependency_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    let mut features = dep.features.clone();
    features.sort();
    features.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features.join(","),
        gate,
        platform
    )
}

fn root_token(index: usize, req: &Requirement) -> String {
    let mut features = req.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        req.package,
        constraint_text(req.constraint),
        features.join(",")
    )
}

fn improve_path(paths: &mut BTreeMap<String, Vec<String>>, package: &str, candidate: Vec<String>) {
    match paths.get(package) {
        Some(current)
            if (current.len(), current) <= (candidate.len(), &candidate) => {}
        _ => {
            paths.insert(package.to_string(), candidate);
        }
    }
}

fn compute_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut closure = Closure::default();

    for (i, req) in roots.iter().enumerate() {
        let token = root_token(i, req);
        closure
            .constraints
            .entry(req.package.clone())
            .or_default()
            .push(req.constraint);
        closure
            .origins
            .entry(req.package.clone())
            .or_default()
            .push(Origin {
                constraint: req.constraint,
                path: vec![token.clone()],
            });
        closure
            .requests
            .entry(req.package.clone())
            .or_default()
            .extend(req.features.iter().cloned());
        improve_path(&mut closure.paths, &req.package, vec![token]);
    }

    // Monotone worklist closure. Adding constraints and origins is independent
    // of feature activation; feature and dependency processing is revisited
    // only when its source gains new requests or a better provenance path.
    let mut queue: VecDeque<String> = selected.keys().cloned().collect();
    let mut queued: BTreeSet<String> = queue.iter().cloned().collect();
    let mut processed_requests: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    let mut processed_edges: BTreeMap<String, BTreeSet<usize>> = BTreeMap::new();

    while let Some(source) = queue.pop_front() {
        queued.remove(&source);
        let Some(&version) = selected.get(&source) else {
            continue;
        };
        let Some(package) = catalog.get(&source) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };

        let requested = closure.requests.get(&source).cloned().unwrap_or_default();
        let done = processed_requests.entry(source.clone()).or_default();
        let new_requests: Vec<String> = requested.difference(done).cloned().collect();
        done.extend(new_requests.iter().cloned());

        for feature_name in new_requests {
            if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                closure
                    .active_features
                    .entry(source.clone())
                    .or_default()
                    .insert(feature_name);
                closure
                    .requests
                    .entry(source.clone())
                    .or_default()
                    .extend(feature.enables.iter().cloned());
            }
        }

        // Process dependencies whose platform and gate are currently active.
        // If later feature requests enable another gate, the source is queued
        // again and the additional edge is processed then.
        let active = closure
            .active_features
            .get(&source)
            .cloned()
            .unwrap_or_default();
        let done_edges = processed_edges.entry(source.clone()).or_default();
        let source_path = closure.paths.get(&source).cloned().unwrap_or_default();

        for (index, dep) in release.deps.iter().enumerate() {
            if !platform_matches(dep.platform, target)
                || dep.gate.as_ref().is_some_and(|gate| !active.contains(gate))
                || done_edges.contains(&index)
            {
                continue;
            }
            done_edges.insert(index);

            let token = dependency_token(&source, version, index, dep);
            let mut path = source_path.clone();
            path.push(token);

            closure
                .constraints
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
            closure
                .origins
                .entry(dep.package.clone())
                .or_default()
                .push(Origin {
                    constraint: dep.constraint,
                    path: path.clone(),
                });
            closure
                .requests
                .entry(dep.package.clone())
                .or_default()
                .extend(dep.features.iter().cloned());

            let old_path = closure.paths.get(&dep.package).cloned();
            improve_path(&mut closure.paths, &dep.package, path);
            let path_improved = closure.paths.get(&dep.package).cloned() != old_path;

            if selected.contains_key(&dep.package)
                && (path_improved || !dep.features.is_empty())
                && queued.insert(dep.package.clone())
            {
                queue.push_back(dep.package.clone());
            }
        }

        // Enables added above may require another pass for this same package.
        let now_requested = closure.requests.get(&source).cloned().unwrap_or_default();
        if now_requested
            .difference(processed_requests.get(&source).unwrap_or(&BTreeSet::new()))
            .next()
            .is_some()
            && queued.insert(source.clone())
        {
            queue.push_back(source.clone());
        }
    }

    // Provenance may improve after an edge has already activated. Recompute
    // shortest paths over the final active dependency graph, then rebuild each
    // dependency origin using its source's canonical path.
    let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
    let mut path_queue: VecDeque<String> = VecDeque::new();
    for (i, req) in roots.iter().enumerate() {
        let path = vec![root_token(i, req)];
        improve_path(&mut paths, &req.package, path.clone());
        path_queue.push_back(req.package.clone());
    }
    let mut seen_path_states: BTreeSet<(String, Vec<String>)> = BTreeSet::new();
    while let Some(source) = path_queue.pop_front() {
        let Some(source_path) = paths.get(&source).cloned() else {
            continue;
        };
        if !seen_path_states.insert((source.clone(), source_path.clone())) {
            continue;
        }
        let Some(&version) = selected.get(&source) else {
            continue;
        };
        let Some(package) = catalog.get(&source) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        let active = closure
            .active_features
            .get(&source)
            .cloned()
            .unwrap_or_default();

        for (index, dep) in release.deps.iter().enumerate() {
            if !platform_matches(dep.platform, target)
                || dep.gate.as_ref().is_some_and(|gate| !active.contains(gate))
            {
                continue;
            }
            let mut candidate = source_path.clone();
            candidate.push(dependency_token(&source, version, index, dep));
            let old = paths.get(&dep.package).cloned();
            improve_path(&mut paths, &dep.package, candidate);
            if paths.get(&dep.package).cloned() != old {
                path_queue.push_back(dep.package.clone());
            }
        }
    }

    closure.paths = paths;
    for origins in closure.origins.values_mut() {
        // Root origins already have their canonical one-token path. Dependency
        // origins are rebuilt below from their dependency token identity.
        for origin in origins.iter_mut() {
            if origin.path.len() > 1 {
                let last = origin.path.last().cloned().unwrap_or_default();
                let source = last.split_once("@/deps[").map(|x| x.0);
                if let Some(source) = source {
                    if let Some(path) = closure.paths.get(source) {
                        let mut rebuilt = path.clone();
                        rebuilt.push(last);
                        origin.path = rebuilt;
                    }
                }
            }
        }
    }

    for constraints in closure.constraints.values_mut() {
        *constraints = canonical_constraints(constraints);
    }
    closure
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
    all_names: Vec<String>,
}

impl Search<'_> {
    fn score(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count()
    }

    fn vector(&self, selected: &BTreeMap<String, Version>) -> Vec<Option<Version>> {
        self.all_names
            .iter()
            .map(|name| selected.get(name).copied())
            .collect()
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn evidence_for(closure: &Closure, name: &str) -> Vec<Evidence> {
        closure
            .origins
            .get(name)
            .into_iter()
            .flatten()
            .map(|origin| Evidence {
                constraint: origin.constraint,
                path: origin.path.clone(),
            })
            .collect()
    }

    fn inspect(&mut self, selected: BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = compute_closure(self.catalog, self.roots, &selected, self.target);

        if let Some(name) = closure
            .constraints
            .keys()
            .filter(|name| !self.catalog.contains_key(*name))
            .min()
            .cloned()
        {
            return Err(ResolveError::Missing {
                evidence: Self::evidence_for(&closure, &name),
                package: name,
            });
        }

        for (name, version) in &selected {
            let Some(package) = self.catalog.get(name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            if let Some(feature) = closure
                .active_features
                .get(name)
                .and_then(|features| features.iter().find(|f| !release.features.iter().any(|d| d.name == **f)))
            {
                return Err(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: closure.paths.get(name).cloned().unwrap_or_default(),
                });
            }
        }

        for (name, version) in selected {
            let requirements = closure.constraints.get(&name).cloned().unwrap_or_default();
            if requirements.iter().any(|c| !constraint_matches(*c, version)) {
                return Err(ResolveError::Conflict {
                    evidence: Self::evidence_for(&closure, &name),
                    package: name,
                    selected: version,
                    requirements,
                });
            }
        }

        let unselected: Vec<String> = closure
            .constraints
            .keys()
            .filter(|name| !selected.contains_key(*name))
            .cloned()
            .collect();

        for name in &unselected {
            let requirements = closure.constraints.get(name).cloned().unwrap_or_default();
            let package = self.catalog[name];
            if !package.releases.iter().any(|release| {
                requirements
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, release.version))
            }) {
                return Err(ResolveError::NoMatch {
                    evidence: Self::evidence_for(&closure, name),
                    package: name.clone(),
                    requirements,
                });
            }
        }

        let Some(name) = unselected.first().cloned() else {
            let score = self.score(&selected);
            let vector = self.vector(&selected);
            let replace = self.best.as_ref().is_none_or(|(best_score, best_vector, _, _)| {
                score < *best_score || (score == *best_score && vector > *best_vector)
            });
            if replace {
                self.best = Some((score, vector, selected, closure));
            }
            return Ok(());
        };

        let requirements = closure.constraints.get(&name).cloned().unwrap_or_default();
        let package = self.catalog[&name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| {
                requirements
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, *version))
            })
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                candidates.remove(pos);
                candidates.insert(0, *lock);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let mut child = selected.clone();
            child.insert(name.clone(), candidate);
            match self.inspect(child) {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => return Err(error),
                Err(error) => self.record_error(error),
            }
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{base}.gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{base}.feature[{f}]"),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut packages_seen = BTreeSet::new();
    for package in catalog {
        if !packages_seen.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions_seen = BTreeSet::new();
        for release in &package.releases {
            if !versions_seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features_seen = BTreeSet::new();
            for feature in &release.features {
                if !features_seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let all_names: Vec<String> = catalog_map.keys().cloned().collect();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        all_names,
    };

    match search.inspect(BTreeMap::new()) {
        Err(error @ ResolveError::Limit { .. }) => return Err(error),
        Err(error) => search.record_error(error),
        Ok(()) => {}
    }

    let Some((_, _, selected_map, closure)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut changed_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    changed_names.extend(selected_map.keys().cloned());
    let mut changes = Vec::new();

    for package in changed_names {
        let before = lock_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let available = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !available {
                    ChangeReason::Unavailable
                } else if closure
                    .constraints
                    .get(&package)
                    .into_iter()
                    .flatten()
                    .any(|constraint| !constraint_matches(*constraint, old))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };

        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
