#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        if ch == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let run_len = i - start;
            if run_len >= 2 {
                append_segment(&segment, options, &mut output);
                segment.clear();
                for _ in 1..run_len {
                    output.push(String::new());
                }
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[i]);
            i += 1;
        }
    }
    append_segment(&segment, options, &mut output);

    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn append_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }

    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut indent = options.first_indent;
    let mut content = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let capacity = options.width - indent;
            let suffix_len = word.len() - offset;

            if suffix_len <= capacity {
                let needed = suffix_len + usize::from(content_len > 0);
                if content_len + needed <= capacity {
                    if content_len > 0 {
                        content.push(' ');
                        content_len += 1;
                    }
                    for &ch in &word[offset..] {
                        content.push(ch);
                    }
                    content_len += suffix_len;
                    offset = word.len();
                } else {
                    emit_line(indent, &content, output);
                    first_line = false;
                    indent = options.rest_indent;
                    content.clear();
                    content_len = 0;
                }
            } else if content_len == 0 {
                for &ch in &word[offset..offset + capacity] {
                    content.push(ch);
                }
                emit_line(indent, &content, output);
                offset += capacity;
                first_line = false;
                indent = options.rest_indent;
                content.clear();
                content_len = 0;
            } else {
                let available = capacity - content_len - 1;
                if available >= 2 {
                    content.push(' ');
                    for &ch in &word[offset..offset + available] {
                        content.push(ch);
                    }
                    emit_line(indent, &content, output);
                    offset += available;
                    first_line = false;
                    indent = options.rest_indent;
                    content.clear();
                    content_len = 0;
                } else {
                    emit_line(indent, &content, output);
                    first_line = false;
                    indent = options.rest_indent;
                    content.clear();
                    content_len = 0;
                }
            }
        }
    }

    if content_len > 0 {
        emit_line(indent, &content, output);
    }

    let _ = first_line;
}

fn emit_line(indent: usize, content: &str, output: &mut Vec<String>) {
    let mut line = String::with_capacity(indent + content.len());
    line.extend(std::iter::repeat(' ').take(indent));
    line.push_str(content);
    output.push(line);
}
