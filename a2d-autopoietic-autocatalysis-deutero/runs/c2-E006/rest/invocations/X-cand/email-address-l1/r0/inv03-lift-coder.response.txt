#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();

    match separators.len() {
        0 => return Err(EmailError::MissingAt),
        1 => {}
        _ => return Err(EmailError::MultipleAt { at: separators[1] }),
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &b) in local.iter().enumerate() {
        let at = i;
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at });
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return Err(EmailError::LocalChar { at });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let root_dot = raw_domain.last().copied() == Some(b'.');
    let domain_len = raw_domain.len() - usize::from(root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw_domain[..domain_len];
    let root_dot_at = root_dot.then_some(domain_start + domain_len);

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;

    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            if i == label_start {
                let at = if i == domain.len() {
                    root_dot_at.unwrap_or(domain_start + i)
                } else {
                    domain_start + i
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label = &domain[label_start..i];
            let label_at = domain_start + label_start;

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong { at: label_at + 63 });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_at });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: label_at + label.len() - 1,
                });
            }
            if let Some(offset) = label
                .iter()
                .position(|&b| !b.is_ascii_alphanumeric() && b != b'-')
            {
                return Err(EmailError::DomainChar {
                    at: label_at + offset,
                });
            }

            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if domain[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII local part");
    let domain_string = domain
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}
