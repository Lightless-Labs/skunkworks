#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut result = encoder.push(input).expect("new encoder is open");
    result.push_str(&encoder.finish().expect("encoder is open"));
    result
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut i = 0;

        if self.tail_len != 0 {
            while self.tail_len < 3 && i < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[i];
                } else {
                    let a = self.tail[0];
                    let b = self.tail[1];
                    let c = input[i];
                    output.push(ALPHABET[(a >> 2) as usize] as char);
                    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
                    output.push(ALPHABET[(c & 0x3f) as usize] as char);
                    self.tail_len = 0;
                    i += 1;
                    break;
                }
                self.tail_len += 1;
                i += 1;
            }
        }

        while i + 3 <= input.len() {
            let a = input[i];
            let b = input[i + 1];
            let c = input[i + 2];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
            output.push(ALPHABET[(c & 0x3f) as usize] as char);
            i += 3;
        }

        while i < input.len() {
            self.tail[self.tail_len] = input[i];
            self.tail_len += 1;
            i += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    next_offset: usize,
    terminal: bool,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            next_offset: 0,
            terminal: false,
            finished: false,
            error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            let failure = if self.terminal {
                Some(Error::BadPadding { offset })
            } else if self.len == 3 && self.group[2] == b'=' && byte != b'=' {
                Some(Error::BadPadding { offset })
            } else if byte == b'=' && self.len < 2 {
                Some(Error::BadPadding { offset })
            } else if byte != b'=' && alphabet_value(byte).is_none() {
                Some(Error::InvalidByte { offset, byte })
            } else {
                None
            };

            if let Some(error) = failure {
                self.error = Some(error.clone());
                return Err(error);
            }

            self.group[self.len] = byte;
            self.offsets[self.len] = offset;
            self.len += 1;

            if self.len == 4 {
                match (self.group[2], self.group[3]) {
                    (b'=', b'=') => {
                        let a = value(self.group[0]);
                        let b = value(self.group[1]);
                        output.push((a << 2) | (b >> 4));
                        self.terminal = true;
                    }
                    (b'=', _) => {
                        let error = Error::BadPadding {
                            offset: self.offsets[3],
                        };
                        self.error = Some(error.clone());
                        return Err(error);
                    }
                    (_, b'=') => {
                        let a = value(self.group[0]);
                        let b = value(self.group[1]);
                        let c = value(self.group[2]);
                        output.push((a << 2) | (b >> 4));
                        output.push((b << 4) | (c >> 2));
                        self.terminal = true;
                    }
                    _ => {
                        let a = value(self.group[0]);
                        let b = value(self.group[1]);
                        let c = value(self.group[2]);
                        let d = value(self.group[3]);
                        output.push((a << 2) | (b >> 4));
                        output.push((b << 4) | (c >> 2));
                        output.push((c << 6) | d);
                    }
                }
                self.len = 0;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if self.len != 0 {
            let error = if self.len == 3 && self.group[2] == b'=' {
                Error::BadPadding {
                    offset: self.offsets[2],
                }
            } else {
                Error::Truncated {
                    offset: self.offsets[0],
                }
            };
            self.error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

fn alphabet_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn value(byte: u8) -> u8 {
    alphabet_value(byte).expect("validated alphabet digit")
}
