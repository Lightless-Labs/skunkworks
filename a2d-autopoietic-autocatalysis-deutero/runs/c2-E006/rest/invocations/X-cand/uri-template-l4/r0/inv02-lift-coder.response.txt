#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

struct Expression<'a> {
    operator: Operator,
    specs: Vec<Specifier<'a>>,
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let expression = parse_expression(template, start + 1, close)?;
                expressions.push((start, close, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn parse_expression(
    template: &str,
    content_start: usize,
    close: usize,
) -> Result<Expression<'_>, Error> {
    let bytes = template.as_bytes();
    let mut operator = Operator::Simple;
    let mut start = content_start;

    if start < close {
        operator = match bytes[start] {
            b'+' => Operator::Reserved,
            b'#' => Operator::Fragment,
            b'.' => Operator::Dot,
            b'/' => Operator::Slash,
            b';' => Operator::Semicolon,
            b'?' => Operator::Question,
            b'&' => Operator::Ampersand,
            _ => Operator::Simple,
        };
        if !matches!(operator, Operator::Simple) {
            start += 1;
        }
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let content = &template[start..close];
    let mut specs = Vec::new();
    let mut relative_start = 0;

    for part in content.split(',') {
        let spec_pos = start + relative_start;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let modifier_offset = part
            .as_bytes()
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(part.len());
        let name = &part[..modifier_offset];

        validate_name(name, spec_pos)?;

        let modifier = if modifier_offset == part.len() {
            Modifier::None
        } else {
            let suffix = &part[modifier_offset..];
            match suffix.as_bytes() {
                [b'*'] => Modifier::Explode,
                [b':', first @ b'1'..=b'9', rest @ ..]
                    if rest.len() <= 3
                        && rest.iter().all(u8::is_ascii_digit) =>
                {
                    let mut value = usize::from(first - b'0');
                    for digit in rest {
                        value = value * 10 + usize::from(*digit - b'0');
                    }
                    Modifier::Prefix(value)
                }
                _ => {
                    return Err(Error::InvalidModifier {
                        pos: spec_pos + modifier_offset,
                    });
                }
            }
        };

        if specs.iter().any(|spec: &Specifier<'_>| spec.name == name) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }

        specs.push(Specifier {
            name,
            modifier,
            pos: spec_pos,
        });
        relative_start += part.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn validate_name(name: &str, pos: usize) -> Result<(), Error> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Err(Error::InvalidName { pos });
    }

    let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
    let valid_rest = |b: u8| b.is_ascii_alphanumeric() || b == b'_';

    if !valid_first(bytes[0]) {
        return Err(Error::InvalidName { pos });
    }

    if let Some(offset) = bytes.iter().enumerate().skip(1).find_map(|(i, &b)| {
        if valid_rest(b) {
            None
        } else {
            Some(i)
        }
    }) {
        return Err(Error::InvalidName { pos: pos + offset });
    }

    Ok(())
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let selected = vars
            .iter()
            .rev()
            .find(|(name, _)| *name == spec.name)
            .map(|(_, value)| value);

        let Some(value) = selected else {
            continue;
        };

        match (spec.modifier, value) {
            (Modifier::Prefix(_), Value::List(_) | Value::Map(_)) => {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
            (_, Value::Scalar(s)) => {
                let scalar = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                fields.push(scalar_field(
                    spec.name,
                    &scalar,
                    expression.operator,
                ));
            }
            (modifier, Value::List(items)) => {
                let explode = matches!(modifier, Modifier::Explode);
                if items.is_empty() {
                    continue;
                }

                if explode {
                    for item in items {
                        fields.push(match expression.operator {
                            Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                                scalar_field(spec.name, item, expression.operator)
                            }
                            _ => encode(item, uses_reserved(expression.operator)),
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, uses_reserved(expression.operator)))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(named_value(spec.name, &joined, expression.operator));
                }
            }
            (modifier, Value::Map(entries)) => {
                let mut seen = Vec::<&str>::new();
                for (key, _) in entries {
                    if seen.iter().any(|previous| *previous == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    seen.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                let reserved = uses_reserved(expression.operator);
                if matches!(modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        fields.push(if matches!(expression.operator, Operator::Semicolon)
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else {
                    let flattened = entries
                        .iter()
                        .flat_map(|(key, value)| {
                            [encode(key, reserved), encode(value, reserved)]
                        })
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(named_value(spec.name, &flattened, expression.operator));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

fn scalar_field(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, uses_reserved(operator));
    named_value(name, &encoded, operator)
}

fn named_value(name: &str, value: &str, operator: Operator) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
            format!("{name}={value}")
        }
        _ => value.to_owned(),
    }
}

fn uses_reserved(operator: Operator) -> bool {
    matches!(operator, Operator::Reserved | Operator::Fragment)
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
