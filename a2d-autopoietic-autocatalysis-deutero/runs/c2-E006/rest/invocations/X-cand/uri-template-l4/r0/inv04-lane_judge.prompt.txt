fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_specifier<'a>(text: &'a str, base: usize) -> Result<Specifier<'a>, Error> {
    let bytes = text.as_bytes();
    let modifier_at = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(bytes.len());
    let name = &text[..name_end];

    if name.is_empty() {
        return Err(Error::InvalidName { pos: base });
    }

    let name_bytes = name.as_bytes();
    if !is_name_start(name_bytes[0]) {
        return Err(Error::InvalidName { pos: base });
    }
    for (i, &byte) in name_bytes.iter().enumerate().skip(1) {
        if !is_name_continue(byte) {
            return Err(Error::InvalidName { pos: base + i });
        }
    }

    let modifier = if let Some(index) = modifier_at {
        let suffix = &text[index..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let digit_bytes = digits.as_bytes();
            if digit_bytes.is_empty()
                || digit_bytes.len() > 4
                || !digit_bytes[0].is_ascii_digit()
                || digit_bytes[0] == b'0'
                || !digit_bytes.iter().all(u8::is_ascii_digit)
            {
                return Err(Error::InvalidModifier { pos: base + index });
            }
            let mut value = 0usize;
            for &digit in digit_bytes {
                value = value * 10 + usize::from(digit - b'0');
            }
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: base + index });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: base + index });
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name,
        modifier,
        pos: base,
    })
}

fn parse_expression<'a>(content: &'a str, content_pos: usize, close_pos: usize) -> Result<Expression<'a>, Error> {
    let (operator, offset) = match content.as_bytes().first().copied() {
        Some(b'+') => (Operator::Plus, 1),
        Some(b'#') => (Operator::Hash, 1),
        Some(b'.') => (Operator::Dot, 1),
        Some(b'/') => (Operator::Slash, 1),
        Some(b';') => (Operator::Semicolon, 1),
        Some(b'?') => (Operator::Question, 1),
        Some(b'&') => (Operator::Ampersand, 1),
        _ => (Operator::Simple, 0),
    };

    if offset == content.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let remainder = &content[offset..];
    let remainder_pos = content_pos + offset;
    let mut specifiers = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut start = 0;

    loop {
        let end = remainder[start..]
            .find(',')
            .map(|relative| start + relative)
            .unwrap_or(remainder.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: remainder_pos + start,
            });
        }

        let specifier = parse_specifier(&remainder[start..end], remainder_pos + start)?;
        if seen.iter().any(|name| *name == specifier.name) {
            return Err(Error::DuplicateVariable { pos: specifier.pos });
        }
        seen.push(specifier.name);
        specifiers.push(specifier);

        if end == remainder.len() {
            break;
        }
        start = end + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn parse_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&byte| byte == b'}') {
                    Some(relative) => i + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(relative) = bytes[open + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative,
                    });
                }

                let expression =
                    parse_expression(&template[open + 1..close], open + 1, close)?;
                expressions.push(expression);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            let high = bytes[i + 1];
            let low = bytes[i + 2];
            if high.is_ascii_hexdigit() && low.is_ascii_hexdigit() {
                result.push('%');
                result.push((high as char).to_ascii_uppercase());
                result.push((low as char).to_ascii_uppercase());
                i += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/'
                    | b'?'
                    | b'#'
                    | b'['
                    | b']'
                    | b'@'
                    | b'!'
                    | b'$'
                    | b'&'
                    | b'\''
                    | b'('
                    | b')'
                    | b'*'
                    | b'+'
                    | b','
                    | b';'
                    | b'='
            );

        if unreserved || reserved_byte {
            result.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    result
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + value.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&value);
        field
    }
}

fn truncate_chars(value: &str, count: usize) -> String {
    value.chars().take(count).collect()
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let reserved = matches!(operator, Operator::Plus | Operator::Hash);
    let separator = match operator {
        Operator::Simple | Operator::Plus | Operator::Hash => ',',
        Operator::Dot => '.',
        Operator::Slash => '/',
        Operator::Semicolon => ';',
        Operator::Question | Operator::Ampersand => '&',
    };

    let mut fields = Vec::new();

    for specifier in &expression.specifiers {
        let Some(value) = lookup(specifier.name, vars) else {
            continue;
        };

        if matches!(specifier.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: specifier.pos });
        }

        match value {
            Value::Scalar(text) => {
                let selected = match specifier.modifier {
                    Modifier::Prefix(count) => truncate_chars(text, count),
                    Modifier::None | Modifier::Explode => text.clone(),
                };
                let encoded = encode(&selected, reserved);
                if matches!(
                    operator,
                    Operator::Semicolon | Operator::Question | Operator::Ampersand
                ) {
                    fields.push(named_field(specifier.name, encoded, operator));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let encoded_items: Vec<String> =
                    items.iter().map(|item| encode(item, reserved)).collect();

                if specifier.modifier == Modifier::Explode {
                    for item in encoded_items {
                        if matches!(
                            operator,
                            Operator::Semicolon | Operator::Question | Operator::Ampersand
                        ) {
                            fields.push(named_field(specifier.name, item, operator));
                        } else {
                            fields.push(item);
                        }
                    }
                } else {
                    let joined = encoded_items.join(",");
                    if matches!(
                        operator,
                        Operator::Semicolon | Operator::Question | Operator::Ampersand
                    ) {
                        fields.push(named_field(specifier.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys: Vec<&str> = Vec::new();
                for (key, _) in entries {
                    if keys.iter().any(|seen| *seen == key) {
                        return Err(Error::DuplicateKey { pos: specifier.pos });
                    }
                    keys.push(key);
                }
                if entries.is_empty() {
                    continue;
                }

                let encoded_entries: Vec<(String, String)> = entries
                    .iter()
                    .map(|(key, value)| (encode(key, reserved), encode(value, reserved)))
                    .collect();

                if specifier.modifier == Modifier::Explode {
                    for (key, value) in encoded_entries {
                        let mut field = key;
                        if operator != Operator::Semicolon || !value.is_empty() {
                            field.push('=');
                        }
                        field.push_str(&value);
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(encoded_entries.len() * 2);
                    for (key, value) in encoded_entries {
                        components.push(key);
                        components.push(value);
                    }
                    let joined = components.join(",");
                    if matches!(
                        operator,
                        Operator::Semicolon | Operator::Question | Operator::Ampersand
                    ) {
                        fields.push(named_field(specifier.name, joined, operator));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let prefix = match operator {
        Operator::Simple | Operator::Plus => "",
        Operator::Hash => "#",
        Operator::Dot => ".",
        Operator::Slash => "/",
        Operator::Semicolon => ";",
        Operator::Question => "?",
        Operator::Ampersand => "&",
    };

    let mut result = String::new();
    result.push_str(prefix);
    result.push_str(&fields.join(&separator.to_string()));
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::with_capacity(template.len());
    let mut cursor = 0;

    for expression in &expressions {
        // Locate each parsed expression's source span in order.
        let open = template[cursor..]
            .find('{')
            .map(|relative| cursor + relative)
            .expect("parsed expression has an opening brace");
        let close = template[open + 1..]
            .find('}')
            .map(|relative| open + 1 + relative)
            .expect("parsed expression has a closing brace");

        output.push_str(&template[cursor..open]);
        output.push_str(&expand_expression(expression, vars)?);
        cursor = close + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn valid_name(name: &str) -> Result<(), usize> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Err(0);
    }

    let first = bytes[0];
    if !(first.is_ascii_alphabetic() || first == b'_') {

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression<'_>, Error> {
    let (operator, offset) = match content.as_bytes().first().copied() {
        Some(b'+') => (Operator::Reserved, 1),
        Some(b'#') => (Operator::Fragment, 1),
        Some(b'.') => (Operator::Dot, 1),
        Some(b'/') => (Operator::Slash, 1),
        Some(b';') => (Operator::Semicolon, 1),
        Some(b'?') => (Operator::Query, 1),
        Some(b'&') => (Operator::Continuation, 1),
        _ => (Operator::Simple, 0),
    };

    if offset == content.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let body = &content[offset..];
    let body_pos = content_pos + offset;
    let mut specifiers = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut start = 0;

        let end = body[start..]
            .find(',')
            .map(|n| start + n)
            .unwrap_or(body.len());

        if end == start {
            return Err(Error::EmptyVariable {
                pos: body_pos + start,
            });
        }

        let raw = &body[start..end];
        let modifier_index = raw.find(['*', ':']);
        let (name, modifier_suffix) = match modifier_index {
            Some(i) => (&raw[..i], Some((i, &raw[i..]))),
            None => (raw, None),
        };

        if name.is_empty() {
            return Err(Error::InvalidName {
                pos: body_pos + start,
            });
        }

        if let Err(bad_offset) = valid_name(name) {
            return Err(Error::InvalidName {
                pos: body_pos + start + bad_offset,
            });
        }

        let modifier = match modifier_suffix {
            None => Modifier::None,
            Some((_, "*")) => Modifier::Explode,
            Some((_, suffix)) if suffix.starts_with(':') => {
                let digits = &suffix[1..];
                let bytes = digits.as_bytes();
                let valid = (1..=4).contains(&bytes.len())
                    && bytes[0].is_ascii_digit()
                    && bytes[0] != b'0'
                    && bytes.iter().all(u8::is_ascii_digit);

                if !valid {
                    return Err(Error::InvalidModifier {
                        pos: body_pos + start + modifier_index.unwrap(),
                    });
                }

                let value = digits.parse::<usize>().unwrap_or(0);
                if !(1..=9999).contains(&value) {
                    return Err(Error::InvalidModifier {
                        pos: body_pos + start + modifier_index.unwrap(),
                    });
                }
                Modifier::Prefix(value)
            }
            Some((index, _)) => {
                return Err(Error::InvalidModifier {
                    pos: body_pos + start + index,
                });
            }
        };

        if seen.iter().any(|prior| *prior == name) {
            return Err(Error::DuplicateVariable {
                pos: body_pos + start,
            });
        }
        seen.push(name);
        specifiers.push(Specifier {
            name,
            modifier,
            pos: body_pos + start,
        });

        if end == body.len() {
            break;
        }
        start = end + 1;
        if start == body.len() {
            return Err(Error::EmptyVariable {
                pos: body_pos + start,
            });
        }
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(n) => i + 1 + n,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(nested) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + nested,
                    });
                }

                let content = &template[open + 1..close];
                expressions.push(parse_expression(content, open + 1, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        if unreserved || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        Operator::Semicolon | Operator::Query | Operator::Continuation => {
            format!("{name}={value}")
        }
        _ => value,
    }
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let reserved = matches!(operator, Operator::Reserved | Operator::Fragment);
    let separator = match operator {
        Operator::Dot => ".",
        Operator::Slash => "/",
        Operator::Semicolon => ";",
        Operator::Query | Operator::Continuation => "&",
        _ => ",",
    };
    let prefix = match operator {
        Operator::Fragment => "#",
        Operator::Dot => ".",
        Operator::Slash => "/",
        Operator::Semicolon => ";",
        Operator::Query => "?",
        Operator::Continuation => "&",
        _ => "",
    };

    let mut fields = Vec::new();

    for specifier in &expression.specifiers {
        let Some(value) = selected_value(specifier.name, vars) else {
            continue;
        };

        if matches!(specifier.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: specifier.pos });
        }

        match value {
            Value::Scalar(value) => {
                let truncated;
                let value = match specifier.modifier {
                    Modifier::Prefix(n) => {
                        truncated = value.chars().take(n).collect::<String>();
                        truncated.as_str()
                    }
                    _ => value.as_str(),
                };
                fields.push(named_field(
                    specifier.name,
                    encode(value, reserved),
                    operator,
                ));
            }
            Value::List(items) => {
                let exploded = specifier.modifier == Modifier::Explode;
                if exploded {
                    for item in items {
                        fields.push(named_field(
                            specifier.name,
                            encode(item, reserved),
                            operator,
                        ));
                    }
                } else if !items.is_empty() {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(named_field(specifier.name, joined, operator));
                }
            }
            Value::Map(entries) => {
                let mut keys: Vec<&str> = Vec::new();
                for (key, _) in entries {
                    if keys.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: specifier.pos });
                    }
                    keys.push(key);
                }

                if specifier.modifier == Modifier::Explode {
                    for (key, value) in entries {
                        let key = encode(key, reserved);
                        let value = encode(value, reserved);
                        if operator == Operator::Semicolon && value.is_empty() {
                            fields.push(key);
                        } else {
                            fields.push(format!("{key}={value}"));
                        }
                    }
                } else if !entries.is_empty() {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    fields.push(named_field(
                        specifier.name,
                        components.join(","),
                        operator,
                    ));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    for expression in &expressions {
        // The expression's source span follows the preceding literal, found from the
        // byte position of its first specifier and operator-independent template scan.
        let _ = expression;
    }

    let mut i = 0;
    let mut expression_index = 0;
    while i < bytes.len() {
        if bytes[i] == b'{' {
            let close = i + 1 + bytes[i + 1..].iter().position(|&b| b == b'}').unwrap_or(0);
            output.push_str(&template[cursor..i]);
            output.push_str(&expand_expression(&expressions[expression_index], vars)?);
            expression_index += 1;
            cursor = close + 1;
            i = close + 1;
        } else {
            i += 1;
        }
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

struct Expression<'a> {
    operator: Operator,
    specs: Vec<Specifier<'a>>,

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let expression = parse_expression(template, start + 1, close)?;
                expressions.push((start, close, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn parse_expression(
    template: &str,
    content_start: usize,
    close: usize,
) -> Result<Expression<'_>, Error> {
    let bytes = template.as_bytes();
    let mut operator = Operator::Simple;
    let mut start = content_start;

    if start < close {
        operator = match bytes[start] {
            b'+' => Operator::Reserved,
            b'#' => Operator::Fragment,
            b'.' => Operator::Dot,
            b'/' => Operator::Slash,
            b';' => Operator::Semicolon,
            b'?' => Operator::Question,
            b'&' => Operator::Ampersand,
            _ => Operator::Simple,
        };
        if !matches!(operator, Operator::Simple) {
            start += 1;
        }
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let content = &template[start..close];
    let mut specs = Vec::new();
    let mut relative_start = 0;

    for part in content.split(',') {
        let spec_pos = start + relative_start;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let modifier_offset = part
            .as_bytes()
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(part.len());
        let name = &part[..modifier_offset];

        validate_name(name, spec_pos)?;

        let modifier = if modifier_offset == part.len() {
            Modifier::None
        } else {
            let suffix = &part[modifier_offset..];
            match suffix.as_bytes() {
                [b'*'] => Modifier::Explode,
                [b':', first @ b'1'..=b'9', rest @ ..]
                    if rest.len() <= 3
                        && rest.iter().all(u8::is_ascii_digit) =>
                {
                    let mut value = usize::from(first - b'0');
                    for digit in rest {
                        value = value * 10 + usize::from(*digit - b'0');
                    }
                    Modifier::Prefix(value)
                }
                _ => {
                    return Err(Error::InvalidModifier {
                        pos: spec_pos + modifier_offset,
                    });
                }
            }
        };

        if specs.iter().any(|spec: &Specifier<'_>| spec.name == name) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }

        specs.push(Specifier {
            name,
            modifier,
            pos: spec_pos,
        });
        relative_start += part.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn validate_name(name: &str, pos: usize) -> Result<(), Error> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Err(Error::InvalidName { pos });
    }

    let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
    let valid_rest = |b: u8| b.is_ascii_alphanumeric() || b == b'_';

    if !valid_first(bytes[0]) {
        return Err(Error::InvalidName { pos });
    }

    if let Some(offset) = bytes.iter().enumerate().skip(1).find_map(|(i, &b)| {
        if valid_rest(b) {
            None
        } else {
            Some(i)
        }
    }) {
        return Err(Error::InvalidName { pos: pos + offset });
    }

    Ok(())
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let selected = vars
            .iter()
            .rev()
            .find(|(name, _)| *name == spec.name)
            .map(|(_, value)| value);

        let Some(value) = selected else {
            continue;
        };

        match (spec.modifier, value) {
            (Modifier::Prefix(_), Value::List(_) | Value::Map(_)) => {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
            (_, Value::Scalar(s)) => {
                let scalar = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                fields.push(scalar_field(
                    spec.name,
                    &scalar,
                    expression.operator,
                ));
            }
            (modifier, Value::List(items)) => {
                let explode = matches!(modifier, Modifier::Explode);
                if items.is_empty() {
                    continue;
                }

                if explode {
                    for item in items {
                        fields.push(match expression.operator {
                            Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                                scalar_field(spec.name, item, expression.operator)
                            }
                            _ => encode(item, uses_reserved(expression.operator)),
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, uses_reserved(expression.operator)))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(named_value(spec.name, &joined, expression.operator));
                }
            }
            (modifier, Value::Map(entries)) => {
                let mut seen = Vec::<&str>::new();
                for (key, _) in entries {
                    if seen.iter().any(|previous| *previous == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    seen.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                let reserved = uses_reserved(expression.operator);
                if matches!(modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        fields.push(if matches!(expression.operator, Operator::Semicolon)
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else {
                    let flattened = entries
                        .iter()
                        .flat_map(|(key, value)| {
                            [encode(key, reserved), encode(value, reserved)]
                        })
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(named_value(spec.name, &flattened, expression.operator));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

fn scalar_field(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, uses_reserved(operator));
    named_value(name, &encoded, operator)
}

fn named_value(name: &str, value: &str, operator: Operator) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
            format!("{name}={value}")
        }
        _ => value.to_owned(),
    }
}

fn uses_reserved(operator: Operator) -> bool {
    matches!(operator, Operator::Reserved | Operator::Fragment)
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
