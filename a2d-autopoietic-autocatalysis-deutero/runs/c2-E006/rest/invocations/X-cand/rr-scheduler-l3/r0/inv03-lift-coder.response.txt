#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut event_order: Vec<usize> = (0..jobs.len()).collect();
    event_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let mut event_cursor = 0;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut completed_count = 0usize;

    fn admit_through(
        jobs: &[Job],
        event_order: &[usize],
        event_cursor: &mut usize,
        time: u64,
        queues: &mut [std::collections::VecDeque<usize>],
        completion: &mut [Option<u64>],
        completed_count: &mut usize,
    ) {
        while *event_cursor < event_order.len() {
            let index = event_order[*event_cursor];
            let job = &jobs[index];
            if u64::from(job.arrival) > time {
                break;
            }
            *event_cursor += 1;

            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
                *completed_count += 1;
            } else {
                queues[job.level].push_back(index);
            }
        }
    }

    admit_through(
        jobs,
        &event_order,
        &mut event_cursor,
        time,
        &mut queues,
        &mut completion,
        &mut completed_count,
    );

    while completed_count < jobs.len() {
        admit_through(
            jobs,
            &event_order,
            &mut event_cursor,
            time,
            &mut queues,
            &mut completion,
            &mut completed_count,
        );

        let selected = queues.iter().position(|queue| !queue.is_empty());
        let Some(level_index) = selected else {
            if event_cursor >= event_order.len() {
                break;
            }

            let next_index = event_order[event_cursor];
            let next_time = u64::from(jobs[next_index].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let selected_index = queues[level_index].pop_front().unwrap();
        let job_id = jobs[selected_index].id;

        if let Some(previous_id) = last_dispatched_id {
            if previous_id != job_id {
                time += u64::from(switch_cost);
            }
        }

        admit_through(
            jobs,
            &event_order,
            &mut event_cursor,
            time,
            &mut queues,
            &mut completion,
            &mut completed_count,
        );

        let run_start = time;
        let quantum = quanta[level_index];
        let consumed = remaining[selected_index].min(quantum);
        time += u64::from(consumed);
        remaining[selected_index] -= consumed;
        let finished = remaining[selected_index] == 0;

        runs.push(Run {
            id: job_id,
            start: run_start,
            end: time,
            finished,
            level: level_index,
        });

        admit_through(
            jobs,
            &event_order,
            &mut event_cursor,
            time,
            &mut queues,
            &mut completion,
            &mut completed_count,
        );

        if finished {
            completion[selected_index] = Some(time);
            completed_count += 1;
        } else {
            let next_level = (level_index + 1).min(quanta.len() - 1);
            levels[selected_index] = next_level;
            queues[next_level].push_back(selected_index);
        }

        last_dispatched_id = Some(job_id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
