pub fn encode(input: &[u8]) -> String {

    let mut chunks = input.chunks_exact(3);

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut output = Vec::with_capacity(input.len() / 4 * 3);
    let mut group = [0u8; 4];
    let mut offsets = [0usize; 4];
    let mut len = 0usize;
    let mut pending_padding: Option<usize> = None;
    let mut terminal_padding = false;

    for (offset, &byte) in input.iter().enumerate() {
        if terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if pending_padding.is_some() {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }

            terminal_padding = true;
            pending_padding = None;
            emit_one(&mut output, group);
            len = 0;
            continue;
        }

        if byte == b'=' {
            match len {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    pending_padding = Some(offset);
                }
                3 => {
                    terminal_padding = true;
                    emit_two(&mut output, group);
                    len = 0;
                }
                _ => unreachable!(),
            }
            continue;
        }

        let Some(value) = decode_digit(byte) else {
            return Err(Error::InvalidByte { offset, byte });
        };

        group[len] = value;
        offsets[len] = offset;
        len += 1;

        if len == 4 {
            emit_three(&mut output, group);
            len = 0;
        }
    }

    if let Some(offset) = pending_padding {
        return Err(Error::BadPadding { offset });
    }
    if len != 0 {
        return Err(Error::Truncated {
            offset: offsets[0],
        });
    }

    Ok(output)
}
