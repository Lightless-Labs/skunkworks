    selected: &mut [bool],

            offset: item_offset,

        .iter()

        (item, None, None)
    };

        return Err(syntax_error(field, item_offset));
    }

        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let (start_atom, end_atom, range) = if star {
        if !hyphens.is_empty() {
            return Err(syntax_error(field, item_offset));

        (None, None, false)

            return Err(syntax_error(field, item_offset));

            .map_err(|_| syntax_error(field, item_offset))?;
        let b = classify_atom(field, &base[h + 1..], item_offset + h + 1)
            .map_err(|_| syntax_error(field, item_offset))?;
        (Some(a), Some(b), true)
    } else {
        return Err(syntax_error(field, item_offset));
    };

    if let Some(step) = step_bytes {
        if !step.iter().all(u8::is_ascii_digit) {
            return Err(syntax_error(field, item_offset));
        }
        if step.is_empty() {
            return Err(syntax_error(field, item_offset));
        }

            Atom::Name(value) => Ok(value),

                } else {
                    Ok(value as u8)
                }
            }

            if bytes.len() > 1 && bytes[0] == b'0' {
                return Err(ParseError::LeadingZero { field, offset });
            }
        }
        Ok(())

        (d.min, d.max)
    } else {

            hyphens[0]
        } else {
            base.len()

                    Atom::Number(_, offset) => offset,
                    Atom::Name(_) => item_offset + end_start,

                return Err(ParseError::ReversedRange { field, offset });

            (start_value, end_value)
        } else if step_bytes.is_some() {
            (start_value, d.max)
        } else {
            (start_value, start_value)
        }

fn parse_field(field: Field, text: &str) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let d = domain(field);
    let mut selected = vec![false; d.width() as usize];
    let mut start = 0usize;

    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(field, &bytes[start..i], start, &mut selected)?;
            start = i + 1;
        }
    }

    let allowed = selected
        .iter()
        .enumerate()
        .filter_map(|(i, &yes)| yes.then_some(d.min + i as u8))
        .collect();

    Ok((allowed, text == "*"))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed = Vec::with_capacity(5);
    let mut unrestricted = [false; 5];

    for (i, (&field, part)) in fields.iter().zip(parts.iter()).enumerate() {
        let (allowed, is_unrestricted) = parse_field(field, part)?;
        parsed.push(allowed);
        unrestricted[i] = is_unrestricted;
    }

    Ok(Cron {
        allowed: [
            parsed.remove(0),
            parsed.remove(0),
            parsed.remove(0),
            parsed.remove(0),
            parsed.remove(0),
        ],
        dom_unrestricted: unrestricted[2],
        dow_unrestricted: unrestricted[4],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.allowed[0].contains(&at.minute)
            || !self.allowed[1].contains(&at.hour)
            || !self.allowed[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.allowed[2].contains(&at.day);
        let dow = self.allowed[4].contains(&weekday(at.year, at.month, at.day));

        match (self.dom_unrestricted, self.dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;

fn is_digits(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())

fn valid_atom(field: Field, atom: &str) -> bool {
    is_digits(atom) || name_value(field, atom).is_some()

fn parse_number(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 2));
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 3))?;
    }
    Ok(n)
}

fn parse_atom_value(
    field: Field,
    atom: &str,
    offset: usize,
) -> Result<u32, ParseError> {
    if let Some(v) = name_value(field, atom) {
        return Ok(v);
    }
    let n = parse_number(field, atom, offset)?;
    let d = domain(field);
    if n < d.min || n > d.max {
        return Err(parse_error(field, offset, 4));
    }
    Ok(n)
}

fn parse_step(field: Field, step: &str, offset: usize) -> Result<u32, ParseError> {
    if !is_digits(step) {
        return Err(parse_error(field, offset, 1));
    }
    let n = parse_number(field, step, offset)?;
    if n == 0 || n > domain(field).width() {
        return Err(parse_error(field, offset, 6));
    }
    Ok(n)
}

fn parse_field(
    field: Field,
    text: &str,
) -> Result<Vec<bool>, ParseError> {
    let d = domain(field);
    let mut selected = vec![false; (d.max + 1) as usize];

    for (i, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut item_start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(parse_error(field, item_start, 0));
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(parse_error(field, item_start, 1));
        }
        let (base, step_part) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((&item[pos + 1..], item_start + pos + 1)))
        } else {
            (item, None)
        };
        if base.is_empty() || step_part.is_some_and(|(s, _)| s.is_empty()) {
            return Err(parse_error(field, item_start, 1));
        }

        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        if hyphen_count > 1 {
            return Err(parse_error(field, item_start, 1));
        }

        let (kind, first, second, first_offset, second_offset) =
            if base == "*" {
                (0u8, "", None, item_start, item_start)
            } else if let Some(pos) = base.find('-') {
                let left = &base[..pos];
                let right = &base[pos + 1..];
                if left.is_empty()
                    || right.is_empty()
                    || left == "*"
                    || right == "*"
                    || !valid_atom(field, left)
                    || !valid_atom(field, right)
                {
                    return Err(parse_error(field, item_start, 1));
                }
                (2u8, left, Some(right), item_start, item_start + pos + 1)
            } else {
                if base == "*" || !valid_atom(field, base) {
                    return Err(parse_error(field, item_start, 1));
                }
                (1u8, base, None, item_start, item_start)
            };

        let step = if let Some((step_text, step_offset)) = step_part {
            Some(parse_step(field, step_text, step_offset)?)
        } else {
            None
        };

        let (start, end) = match kind {
            0 => (d.min, d.max),
            1 => {
                let value = parse_atom_value(field, first, first_offset)?;
                let end = if step.is_some() { d.max } else { value };
                (value, end)
            }
            _ => {
                let start = parse_atom_value(field, first, first_offset)?;
                let end = parse_atom_value(field, second.unwrap(), second_offset)?;
                if start > end {
                    return Err(parse_error(field, second_offset, 5));
                }
                (start, end)
            }
        };

        let stride = step.unwrap_or(1);
        let mut value = start;
        loop {
            selected[value as usize] = true;
            if value > end.saturating_sub(stride) {
                break;
            }
            value += stride;
        }

        item_start += item.len() + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];

    for i in 0..5 {
        selected[i] = parse_field(fields[i], parts[i])?;
    }

    Ok(Cron {
        selected,
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| if yes { Some(i as u8) } else { None })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            current = next_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn next_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
