    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Clone)]
struct Row {
    prefix: u8,
    payload: String,
}

#[derive(Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.starts_with('0') {
        return None;
    }
    let value = s.parse::<usize>().ok()?;
    (value <= 128).then_some(value)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let inner = line.strip_prefix("@@ -")?.strip_suffix(" @@")?;
    let mut parts = inner.split(' ');
    let old = parts.next()?;
    let new = parts.next()?;
    if parts.next().is_some() {
        return None;
    }

    let (a, b) = old.split_once(',')?;
    let (c, d) = new.strip_prefix('+')?.split_once(',')?;
    let a = parse_number(a)?;
    let b = parse_number(b)?;
    let c = parse_number(c)?;
    let d = parse_number(d)?;

    if old.bytes().filter(|&x| x == b',').count() != 1
        || new.bytes().filter(|&x| x == b',').count() != 1
        || (b > 0 && a == 0)
        || (d > 0 && c == 0)
    {
        return None;
    }

    Some((a, b, c, d))
}

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 { start - 1 } else { start }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }

    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let content = &patch[..patch.len() - 1];
    let lines: Vec<&str> = content.split('\n').collect();

    if lines.first().copied() != Some("--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if lines.get(1).copied() != Some("+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut pos = 2usize;
    let mut previous_old_end = 0usize;
    let mut preceding_delta: i128 = 0;
    let mut have_previous = false;

    while pos < lines.len() {
        let header_index = pos;
        let header_line = pos + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[pos]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);
        pos += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while pos < lines.len() && !lines[pos].starts_with("@@") {
            let row_line = pos + 1;
            let text = lines[pos];
            let Some((&prefix, payload)) = text.as_bytes().split_first() else {
                return Err(ApplyError::BadBody { line: row_line });
            };
            if !matches!(prefix, b' ' | b'-' | b'+') || payload.contains(&b'\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = std::str::from_utf8(payload).unwrap().to_owned();
            if prefix == b' ' || prefix == b'-' {
                old_seen += 1;
            }
            if prefix == b' ' || prefix == b'+' {
                new_seen += 1;
            }
            rows.push(Row { prefix, payload });
            pos += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if have_previous && old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i128 + preceding_delta;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        previous_old_end = old_start + b;
        have_previous = true;
        preceding_delta += d as i128 - b as i128;
        hunks.push(Hunk {
            line: header_index + 1,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }

    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (index, hunk) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < hunk.rows.len() && hunk.rows[lead].prefix == b' ' {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < hunk.rows.len()
            && hunk.rows[hunk.rows.len() - 1 - trail].prefix == b' '
        {
            trail += 1;
        }

        let expected = hunk.old_start as i128 + net + drift;
        let max_left = options.fuzz.min(lead);
        let max_right = options.fuzz.min(trail);
        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_choices: Vec<(usize, usize, usize)> = Vec::new();

        for left in 0..=max_left {
            for right in 0..=max_right {
                if left + right > hunk.old_count {
                    continue;
                }

                let mut old_payloads = Vec::new();
                for row in &hunk.rows[left..hunk.rows.len() - right] {
                    if row.prefix == b' ' || row.prefix == b'-' {
                        old_payloads.push(row.payload.as_str());
                    }
                }

                for start in 0..=working.len() {
                    if start + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (start as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }

                    let match_start = start + left;
                    let match_end = start + hunk.old_count - right;
                    if old_payloads.len() != match_end - match_start
                        || !old_payloads
                            .iter()
                            .zip(&working[match_start..match_end])
                            .all(|(expected, actual)| *expected == actual)
                    {
                        continue;
                    }

                    let rank = (left + right, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_choices.clear();
                            best_choices.push((start, left, right));
                        }
                        Some(best) if rank < best => {
                            best_rank = Some(rank);
                            best_choices.clear();
                            best_choices.push((start, left, right));
                        }
                        Some(best) if rank == best => {
                            best_choices.push((start, left, right));
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_choices.is_empty() {
            return Err(ApplyError::NoMatch { hunk: index + 1 });
        }

        best_choices.sort_by_key(|&(start, left, right)| (start, left, right));
        let mut starts: Vec<usize> = best_choices.iter().map(|choice| choice.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() >= 2 {
            return Err(ApplyError::Ambiguous {
                hunk: index + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        let (start, left, right) = best_choices[0];
        let previous_net = net;
        let replace_start = start + left;
        let replace_end = start + hunk.old_count - right;
        let trimmed = &hunk.rows[left..hunk.rows.len() - right];
        let replacement: Vec<String> = trimmed
            .iter()
            .filter(|row| row.prefix == b' ' || row.prefix == b'+')
            .map(|row| row.payload.clone())
            .collect();

        working.splice(replace_start..replace_end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = start as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let rest = line.strip_prefix("@@ -").ok_or_else(bad)?;
    let (old_field, rest) = rest.split_once(" +").ok_or_else(bad)?;
    let (new_field, tail) = rest.split_once(" @@").ok_or_else(bad)?;
    if !tail.is_empty() {
        return Err(bad());
    }
    let (a, b) = old_field.split_once(',').ok_or_else(bad)?;
    let (c, d) = new_field.split_once(',').ok_or_else(bad)?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return Err(bad());
    }
    let a = parse_number(a).ok_or_else(bad)?;
    let b = parse_number(b).ok_or_else(bad)?;
    let c = parse_number(c).ok_or_else(bad)?;
    let d = parse_number(d).ok_or_else(bad)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    Ok((a, b, c, d))
}

fn physical_lines(s: &str) -> Vec<&str> {
    if s.is_empty() {
        return Vec::new();
    }
    s[..s.len() - 1].split('\n').collect()
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines = physical_lines(patch);
    for (idx, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(idx).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: idx + 1 });
        }
    }
    if lines.len() < 3 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end: Option<i128> = None;
    let mut net_before = 0i128;

    while i < lines.len() {
        let header_line = i + 1;
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let (a, b, c, d) = parse_hunk_header(lines[i], header_line)?;
        let old_start = if b > 0 { (a - 1) as i128 } else { a as i128 };
        let new_start = if d > 0 { (c - 1) as i128 } else { c as i128 };

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let bytes = lines[i].as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            if lines[i].contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let kind = bytes[0];
            let payload = lines[i][1..].to_owned();
            match kind {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                }
                b'-' => old_seen += 1,
                b'+' => new_seen += 1,
                _ => unreachable!(),
            }
            rows.push(BodyRow { kind, payload });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }
        if previous_old_end.is_some_and(|end| old_start < end)
            || new_start != old_start + net_before
            || old_start + net_before < 0
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        previous_old_end = Some(old_start + b as i128);
        net_before += d as i128 - b as i128;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {

        return Err(ApplyError::InvalidOptions);
    }

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });

        return Err(ApplyError::SourceMissingLf);
    }

        Vec::new()

        physical_lines(source).into_iter().map(str::to_owned).collect()
    };

        return Err(ApplyError::TooLarge);

    let mut drift = 0i128;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
        let max_left = options.fuzz.min(lead);
        let max_right = options.fuzz.min(trail);

        let expected = hunk.old_start + net + drift;
        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_starts = Vec::new();
        let mut best_choice: Option<(usize, usize, usize)> = None;

        for left in 0..=max_left {
            for right in 0..=max_right {
                if left + right > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                let mut new_payloads = Vec::new();
                for row in hunk.rows.iter().skip(left).take(hunk.rows.len().saturating_sub(left + right)) {
                    match row.kind {
                        b' ' => {
                            old_payloads.push(row.payload.as_str());
                            new_payloads.push(row.payload.as_str());
                        }
                        b'-' => old_payloads.push(row.payload.as_str()),
                        b'+' => new_payloads.push(row.payload.as_str()),
                        _ => unreachable!(),
                    }
                }

                for start in 0..=working.len() {
                    if start + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (start as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    if start + left + right > working.len() {
                        continue;
                    }
                    let candidate = &working[start + left..start + hunk.old_count - right];
                    if candidate.len() != old_payloads.len()
                        || !candidate
                            .iter()
                            .zip(old_payloads.iter())
                            .all(|(line, payload)| line == payload)
                    {
                        continue;
                    }

                    let rank = (left + right, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            best_choice = Some((start, left, right));
                        }
                        Some(old_rank) if rank < old_rank => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            best_choice = Some((start, left, right));
                        }
                        Some(old_rank) if rank == old_rank => {
                            if !best_starts.contains(&start) {
                                best_starts.push(start);
                            }
                            if let Some((chosen_start, chosen_left, _)) = best_choice {
                                if start == chosen_start && left < chosen_left {
                                    best_choice = Some((start, left, right));
                                }
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        let Some((start, left, right)) = best_choice else {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        };

        if best_starts.len() > 1 {
            best_starts.sort_unstable();
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let mut replacement = Vec::new();
        for row in hunk
            .rows
            .iter()
            .skip(left)
            .take(hunk.rows.len().saturating_sub(left + right))
        {
            match row.kind {
                b' ' | b'+' => replacement.push(row.payload.clone()),
                b'-' => {}
                _ => unreachable!(),
            }
        }

        let range_start = start + left;
        let range_end = start + hunk.old_count - right;
        working.splice(range_start..range_end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = start as i128 - (hunk.old_start + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Debug, Clone)]
enum Row {
    Context(String),
    Remove(String),
    Add(String),
}

#[derive(Debug, Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
    lead: usize,
    trail: usize,
}

fn canonical_number(s: &str) -> Option<usize> {
    if s.is_empty()
        || !s.bytes().all(|b| b.is_ascii_digit())
        || (s.len() > 1 && s.starts_with('0'))
    {
        return None;
    }

    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old_part, rest) = rest.split_once(" +")?;
    let (new_part, suffix) = rest.split_once(" @@")?;
    if suffix != "" {
        return None;
    }

    let (old_start, old_count) = old_part.split_once(',')?;
    let (new_start, new_count) = new_part.split_once(',')?;

    let a = canonical_number(old_start)?;
    let b = canonical_number(old_count)?;
    let c = canonical_number(new_start)?;
    let d = canonical_number(new_count)?;

    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }

    Some((a, b, c, d))
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }

    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();

    for i in 0..2 {
        let expected = if i == 0 { "--- before" } else { "+++ after" };
        if lines.get(i).copied() != Some(expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }

    if lines.len() < 3 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end: i128 = 0;
    let mut cumulative_delta: i128 = 0;

    while i < lines.len() {
        let header_index = i;
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };

        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let text = lines[i];
            if text.contains('\r') || text.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let (prefix, payload) = text.split_at(1);
            match prefix {
                " " => {
                    old_seen += 1;
                    new_seen += 1;
                    rows.push(Row::Context(payload.to_owned()));
                }
                "-" => {
                    old_seen += 1;
                    rows.push(Row::Remove(payload.to_owned()));
                }
                "+" => {
                    new_seen += 1;
                    rows.push(Row::Add(payload.to_owned()));
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            }
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }

        if !rows.iter().any(|row| matches!(row, Row::Add(_) | Row::Remove(_))) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_start_i = old_start as i128;
        let new_start_i = new_start as i128;
        if old_start_i < previous_old_end
            || new_start_i != old_start_i + cumulative_delta
        {
            return Err(ApplyError::Misordered { line: header_line });
        }

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        let lead = rows
            .iter()
            .take_while(|row| matches!(row, Row::Context(_)))
            .count();
        let trail = rows
            .iter()
            .rev()
            .take_while(|row| matches!(row, Row::Context(_)))
            .count();

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
            lead,
            trail,
        });

        previous_old_end = old_start_i + b as i128;
        cumulative_delta += d as i128 - b as i128;

        if i == header_index {
            unreachable!();
        }
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.bytes().position(|b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }

    let body = &source[..source.len() - 1];
    let lines: Vec<String> = body.split('\n').map(str::to_owned).collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_distance(a: i128, b: i128) -> i128 {
    (a - b).abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let expected = hunk.old_start as i128 + net + drift;
        let max_left = options.fuzz.min(hunk.lead);
        let max_right = options.fuzz.min(hunk.trail);

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_starts = Vec::new();
        let mut best_choice: Option<(usize, usize, usize)> = None;

        for left in 0..=max_left {
            for right in 0..=max_right {
                if left + right > hunk.old_count {
                    continue;
                }
                let mut match_payloads = Vec::new();
                for row in &hunk.rows[left..hunk.rows.len() - right] {
                    match row {
                        Row::Context(s) | Row::Remove(s) => {
                            match_payloads.push(s.as_str())
                        }
                        Row::Add(_) => {}
                    }
                }

                let replacement: Vec<String> = hunk.rows[left..hunk.rows.len() - right]
                    .iter()
                    .filter_map(|row| match row {
                        Row::Context(s) | Row::Add(s) => Some(s.clone()),
                        Row::Remove(_) => None,
                    })
                    .collect();

                if match_payloads.len() != hunk.old_count - left - right {
                    continue;
                }

                if hunk.old_count - left - right > working.len() {
                    continue;
                }

                for start in 0..=working.len().saturating_sub(hunk.old_count) {
                    if start + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_distance(start as i128, expected);
                    if distance > options.max_offset as i128 {
                        continue;
                    }

                    let segment_matches = match_payloads
                        .iter()
                        .enumerate()
                        .all(|(j, payload)| working[start + left + j] == *payload);
                    if !segment_matches {
                        continue;
                    }

                    let rank = (left + right, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            best_choice = Some((start, left, right));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            best_choice = Some((start, left, right));
                        }
                        Some(current) if rank == current => {
                            if !best_starts.contains(&start) {
                                best_starts.push(start);
                            }
                            match best_choice {
                                Some((old_start, old_left, _))
                                    if old_start == start && left < old_left =>
                                {
                                    best_choice = Some((start, left, right));
                                }
                                None => best_choice = Some((start, left, right)),
                                _ => {}
                            }
                        }
                        _ => {}
                    }

                    let _ = &replacement;
                }
            }
        }

        let Some((start, left, right)) = best_choice else {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        };

        best_starts.sort_unstable();
        best_starts.dedup();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let begin = start + left;
        let end = start + hunk.old_count - right;
        let replacement: Vec<String> = hunk.rows[left..hunk.rows.len() - right]
            .iter()
            .filter_map(|row| match row {
                Row::Context(s) | Row::Add(s) => Some(s.clone()),
                Row::Remove(_) => None,
            })
            .collect();

        working.splice(begin..end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = start as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
