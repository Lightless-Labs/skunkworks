#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Debug, Clone)]
enum Row {
    Context(String),
    Remove(String),
    Add(String),
}

#[derive(Debug, Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
    lead: usize,
    trail: usize,
}

fn canonical_number(s: &str) -> Option<usize> {
    if s.is_empty()
        || !s.bytes().all(|b| b.is_ascii_digit())
        || (s.len() > 1 && s.starts_with('0'))
    {
        return None;
    }

    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old_part, rest) = rest.split_once(" +")?;
    let (new_part, suffix) = rest.split_once(" @@")?;
    if suffix != "" {
        return None;
    }

    let (old_start, old_count) = old_part.split_once(',')?;
    let (new_start, new_count) = new_part.split_once(',')?;

    let a = canonical_number(old_start)?;
    let b = canonical_number(old_count)?;
    let c = canonical_number(new_start)?;
    let d = canonical_number(new_count)?;

    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }

    Some((a, b, c, d))
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }

    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();

    for i in 0..2 {
        let expected = if i == 0 { "--- before" } else { "+++ after" };
        if lines.get(i).copied() != Some(expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }

    if lines.len() < 3 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end: i128 = 0;
    let mut cumulative_delta: i128 = 0;

    while i < lines.len() {
        let header_index = i;
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };

        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let text = lines[i];
            if text.contains('\r') || text.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let (prefix, payload) = text.split_at(1);
            match prefix {
                " " => {
                    old_seen += 1;
                    new_seen += 1;
                    rows.push(Row::Context(payload.to_owned()));
                }
                "-" => {
                    old_seen += 1;
                    rows.push(Row::Remove(payload.to_owned()));
                }
                "+" => {
                    new_seen += 1;
                    rows.push(Row::Add(payload.to_owned()));
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            }
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }

        if !rows.iter().any(|row| matches!(row, Row::Add(_) | Row::Remove(_))) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_start_i = old_start as i128;
        let new_start_i = new_start as i128;
        if old_start_i < previous_old_end
            || new_start_i != old_start_i + cumulative_delta
        {
            return Err(ApplyError::Misordered { line: header_line });
        }

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        let lead = rows
            .iter()
            .take_while(|row| matches!(row, Row::Context(_)))
            .count();
        let trail = rows
            .iter()
            .rev()
            .take_while(|row| matches!(row, Row::Context(_)))
            .count();

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
            lead,
            trail,
        });

        previous_old_end = old_start_i + b as i128;
        cumulative_delta += d as i128 - b as i128;

        if i == header_index {
            unreachable!();
        }
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.bytes().position(|b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }

    let body = &source[..source.len() - 1];
    let lines: Vec<String> = body.split('\n').map(str::to_owned).collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_distance(a: i128, b: i128) -> i128 {
    (a - b).abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let expected = hunk.old_start as i128 + net + drift;
        let max_left = options.fuzz.min(hunk.lead);
        let max_right = options.fuzz.min(hunk.trail);

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_starts = Vec::new();
        let mut best_choice: Option<(usize, usize, usize)> = None;

        for left in 0..=max_left {
            for right in 0..=max_right {
                if left + right > hunk.old_count {
                    continue;
                }
                let mut match_payloads = Vec::new();
                for row in &hunk.rows[left..hunk.rows.len() - right] {
                    match row {
                        Row::Context(s) | Row::Remove(s) => {
                            match_payloads.push(s.as_str())
                        }
                        Row::Add(_) => {}
                    }
                }

                let replacement: Vec<String> = hunk.rows[left..hunk.rows.len() - right]
                    .iter()
                    .filter_map(|row| match row {
                        Row::Context(s) | Row::Add(s) => Some(s.clone()),
                        Row::Remove(_) => None,
                    })
                    .collect();

                if match_payloads.len() != hunk.old_count - left - right {
                    continue;
                }

                if hunk.old_count - left - right > working.len() {
                    continue;
                }

                for start in 0..=working.len().saturating_sub(hunk.old_count) {
                    if start + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_distance(start as i128, expected);
                    if distance > options.max_offset as i128 {
                        continue;
                    }

                    let segment_matches = match_payloads
                        .iter()
                        .enumerate()
                        .all(|(j, payload)| working[start + left + j] == *payload);
                    if !segment_matches {
                        continue;
                    }

                    let rank = (left + right, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            best_choice = Some((start, left, right));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            best_choice = Some((start, left, right));
                        }
                        Some(current) if rank == current => {
                            if !best_starts.contains(&start) {
                                best_starts.push(start);
                            }
                            match best_choice {
                                Some((old_start, old_left, _))
                                    if old_start == start && left < old_left =>
                                {
                                    best_choice = Some((start, left, right));
                                }
                                None => best_choice = Some((start, left, right)),
                                _ => {}
                            }
                        }
                        _ => {}
                    }

                    let _ = &replacement;
                }
            }
        }

        let Some((start, left, right)) = best_choice else {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        };

        best_starts.sort_unstable();
        best_starts.dedup();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let begin = start + left;
        let end = start + hunk.old_count - right;
        let replacement: Vec<String> = hunk.rows[left..hunk.rows.len() - right]
            .iter()
            .filter_map(|row| match row {
                Row::Context(s) | Row::Add(s) => Some(s.clone()),
                Row::Remove(_) => None,
            })
            .collect();

        working.splice(begin..end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = start as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
