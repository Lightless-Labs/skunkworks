#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut event_index = 0;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let quantum = u64::from(quantum);
    let switch_cost = u64::from(switch_cost);
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let index = events[event_index];
            let job = &jobs[index];
            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
                completed_count += 1;
            } else {
                ready.push_back(index);
            }
            event_index += 1;
        }

        if let Some(index) = ready.pop_front() {
            let job = &jobs[index];
            if last_dispatched_id.is_some_and(|previous| previous != job.id) {
                time += switch_cost;
            }

            while event_index < events.len()
                && u64::from(jobs[events[event_index]].arrival) <= time
            {
                let arriving_index = events[event_index];
                let arriving_job = &jobs[arriving_index];
                if arriving_job.ticks == 0 {
                    completion[arriving_index] = Some(u64::from(arriving_job.arrival));
                    completed_count += 1;
                } else {
                    ready.push_back(arriving_index);
                }
                event_index += 1;
            }

            let consumed = u64::from(job.ticks).min(quantum);
            let start = time;
            let end = start + consumed;
            let finished = consumed == u64::from(job.ticks);

            runs.push(Run {
                id: job.id,
                start,
                end,
                finished,
            });
            time = end;

            while event_index < events.len()
                && u64::from(jobs[events[event_index]].arrival) <= time
            {
                let arriving_index = events[event_index];
                let arriving_job = &jobs[arriving_index];
                if arriving_job.ticks == 0 {
                    completion[arriving_index] = Some(u64::from(arriving_job.arrival));
                    completed_count += 1;
                } else {
                    ready.push_back(arriving_index);
                }
                event_index += 1;
            }

            if finished {
                completion[index] = Some(end);
                completed_count += 1;
            } else {
                ready.push_back(index);
            }
            last_dispatched_id = Some(job.id);
        } else if event_index < events.len() {
            let next_time = u64::from(jobs[events[event_index]].arrival);
            if next_time > time {
                if let Some((_, previous_end)) = idle.last_mut() {
                    if *previous_end == time {
                        *previous_end = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
        }
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
