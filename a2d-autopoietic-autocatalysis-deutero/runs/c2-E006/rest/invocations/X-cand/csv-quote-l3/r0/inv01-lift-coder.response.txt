use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_config(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_config(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let records = parse_records(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            (Some(records[0].clone()), records[1..].to_vec())
        }
    } else {
        (None, records)
    };

    if let Some(header_values) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in header_values.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name.as_str(), second);
            }
        }

        let expected = header_values.len();
        for (row_index, row) in rows.iter().enumerate() {
            if row.len() != expected {
                return Err(CsvError::HeaderWidth {
                    record: row_index + 1,
                    expected,
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

fn parse_records(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    let mut fields: Vec<String> = Vec::new();
    let mut field = Vec::new();

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;
    let mut record_index = 0usize;
    let mut i = start;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        // Outside quotes, bare CR has precedence over state-specific errors.
        let inside_quote = matches!(state, State::Quoted { .. });
        if !inside_quote && b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
        }

        match state {
            State::FieldStart => {
                record_started = true;
                if b == quote {
                    state = State::Quoted { opening: i };
                    i += 1;
                } else if b == delimiter {
                    fields.push(String::new());
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                    record_started = false;
                } else {
                    field.push(b);
                    state = State::Unquoted;
                    i += 1;
                }
            }
            State::Unquoted => {
                if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else if b == delimiter {
                    fields.push(bytes_to_string(std::mem::take(&mut field)));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    fields.push(bytes_to_string(std::mem::take(&mut field)));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                    record_started = false;
                } else {
                    field.push(b);
                    i += 1;
                }
            }
            State::Quoted { opening } => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(b);
                    i += 1;
                }
                let _ = opening;
            }
            State::AfterQuote => {
                if b == delimiter {
                    fields.push(bytes_to_string(std::mem::take(&mut field)));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    fields.push(bytes_to_string(std::mem::take(&mut field)));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                    record_started = false;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    match state {
        State::Unquoted | State::AfterQuote => {
            fields.push(bytes_to_string(field));
            records.push(fields);
        }
        State::FieldStart if record_started || !fields.is_empty() => {
            fields.push(String::new());
            records.push(fields);
        }
        State::FieldStart => {}
        State::Quoted { .. } => unreachable!(),
    }

    Ok(records)
}

fn bytes_to_string(bytes: Vec<u8>) -> String {
    String::from_utf8(bytes).expect("CSV field bytes originate from valid UTF-8 input")
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut output_rows = Vec::new();
    let mut errors = Vec::new();
    let mut stop_all = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stop_all {
            break;
        }
        let mut output = vec![None; columns.len()];

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (column, cell) = match resolved {
                None => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    });
                    if policy == ErrorPolicy::Stop {
                        stop_all = true;
                    }
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                    continue;
                }
                Some(index) => match row.get(index) {
                    None => {
                        errors.push(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(index),
                            kind: CellErrorKind::MissingCell,
                        });
                        if policy == ErrorPolicy::Stop {
                            stop_all = true;
                        }
                        if policy != ErrorPolicy::Collect {
                            break;
                        }
                        continue;
                    }
                    Some(cell) => (index, cell),
                },
            };

            let result = parse_cell(cell, spec);
            match result {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column),
                        kind,
                    });
                    if policy == ErrorPolicy::Stop {
                        stop_all = true;
                    }
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }

        output_rows.push(output);
    }

    TypedResult {
        rows: output_rows,
        errors,
    }
}

fn parse_cell(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

fn integer_syntax(s: &str) -> Option<(bool, &str)> {
    let (negative, digits) = if let Some(rest) = s.strip_prefix('-') {
        (true, rest)
    } else {
        (false, s)
    };
    let bytes = digits.as_bytes();
    if bytes == b"0" {
        return if negative { None } else { Some((false, digits)) };
    }
    if bytes.is_empty() || !(b'1'..=b'9').contains(&bytes[0]) {
        return None;
    }
    if !bytes[1..].iter().all(u8::is_ascii_digit) {
        return None;
    }
    Some((negative, digits))
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let (negative, digits) = integer_syntax(s).ok_or(CellErrorKind::InvalidInteger)?;
    let magnitude = digits
        .parse::<u64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)?;

    if negative {
        let limit = (i64::MAX as u64) + 1;
        if magnitude > limit {
            Err(CellErrorKind::IntegerOverflow)
        } else if magnitude == limit {
            Ok(i64::MIN)
        } else {
            Ok(-(magnitude as i64))
        }
    } else if magnitude > i64::MAX as u64 {
        Err(CellErrorKind::IntegerOverflow)
    } else {
        Ok(magnitude as i64)
    }
}

fn decimal_syntax(s: &str) -> Option<(bool, &str, &str)> {
    let (negative, body) = if let Some(rest) = s.strip_prefix('-') {
        (true, rest)
    } else {
        (false, s)
    };
    let (whole, fraction) = body.split_once('.')?;
    if fraction.len() != 2 || !fraction.as_bytes().iter().all(u8::is_ascii_digit) {
        return None;
    }
    let whole_bytes = whole.as_bytes();
    if whole_bytes == b"0" {
        if negative && fraction == "00" {
            return None;
        }
    } else if whole_bytes.is_empty()
        || !(b'1'..=b'9').contains(&whole_bytes[0])
        || !whole_bytes[1..].iter().all(u8::is_ascii_digit)
    {
        return None;
    }
    Some((negative, whole, fraction))
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    let (negative, whole, fraction) = decimal_syntax(s).ok_or(CellErrorKind::InvalidDecimal)?;

    let whole_value = whole
        .parse::<u64>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let fraction_value = fraction.parse::<u64>().expect("two ASCII digits");
    let magnitude = whole_value
        .checked_mul(100)
        .and_then(|v| v.checked_add(fraction_value))
        .ok_or(CellErrorKind::DecimalOverflow)?;

    if negative {
        let limit = (i64::MAX as u64) + 1;
        if magnitude > limit {
            Err(CellErrorKind::DecimalOverflow)
        } else if magnitude == limit {
            Ok(i64::MIN)
        } else {
            Ok(-(magnitude as i64))
        }
    } else if magnitude > i64::MAX as u64 {
        Err(CellErrorKind::DecimalOverflow)
    } else {
        Ok(magnitude as i64)
    }
}
