pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_byte(alphabet: Alphabet, value: u8) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 if alphabet == Alphabet::Standard => b'+',
        62 => b'-',
        _ if alphabet == Alphabet::Standard => b'/',
        _ => b'_',
    }
}

fn digit_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8], alphabet: Alphabet, padded: bool) -> Vec<u8> {
    let mut out = Vec::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(alphabet_byte(alphabet, a >> 2));
    out.push(alphabet_byte(alphabet, ((a & 0x03) << 4) | (b >> 4)));

    if bytes.len() > 1 {
        out.push(alphabet_byte(alphabet, ((b & 0x0f) << 2) | (c >> 6)));
    } else if padded {
        out.push(b'=');
    }

    if bytes.len() > 2 {
        out.push(alphabet_byte(alphabet, c & 0x3f));
    } else if padded {
        out.push(b'=');
    }

    out
}

fn append_wrapped(out: &mut String, symbols: &[u8], options: Options, column: &mut usize) {
    match options.whitespace {
        Whitespace::Wrapped { width, ending } => {
            for &symbol in symbols {
                if *column == width {
                    match ending {
                        LineEnding::Lf => out.push('\n'),
                        LineEnding::CrLf => out.push_str("\r\n"),
                    }
                    *column = 0;
                }
                out.push(symbol as char);
                *column += 1;
            }
        }
        _ => {
            for &symbol in symbols {
                out.push(symbol as char);
            }
        }
    }
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    finished: bool,
    column: usize,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            buffered: Vec::with_capacity(2),
            finished: false,
            column: 0,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.buffered.len() + input.len());
        data.extend_from_slice(&self.buffered);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut result = String::new();
        for group in data[..complete_len].chunks_exact(3) {
            let symbols = encode_group(group, self.options.alphabet, true);
            append_wrapped(&mut result, &symbols, self.options, &mut self.column);
        }

        self.buffered.clear();
        self.buffered.extend_from_slice(&data[complete_len..]);
        Ok(result)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut result = String::new();
        if !self.buffered.is_empty() {
            let padded = self.options.padding != Padding::Forbidden;
            let symbols = encode_group(&self.buffered, self.options.alphabet, padded);
            append_wrapped(&mut result, &symbols, self.options, &mut self.column);
            self.buffered.clear();
        }
        Ok(result)
    }
}

enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: Vec<(u8, usize)>,
    offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
    frame_symbols: usize,
    frame_state: FrameState,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
            frame_symbols: 0,
            frame_state: FrameState::Normal,
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn decode_completed_group(&mut self, out: &mut Vec<u8>) {
        let g = &self.group;
        let a = digit_value(g[0].0, self.options.alphabet).unwrap();
        let b = digit_value(g[1].0, self.options.alphabet).unwrap();

        if g[2].0 == b'=' {
            out.push((a << 2) | (b >> 4));
            self.terminal_padding = true;
        } else {
            let c = digit_value(g[2].0, self.options.alphabet).unwrap();
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));

            if g[3].0 == b'=' {
                self.terminal_padding = true;
            } else {
                let d = digit_value(g[3].0, self.options.alphabet).unwrap();
                out.push((c << 6) | d);
            }
        }
        self.group.clear();
    }

    fn scan_symbol(&mut self, byte: u8, physical_offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset: physical_offset });
        }

        if let Some(pending_offset) = self.pending_second_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset: physical_offset });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: physical_offset });
            }
            self.group.push((byte, physical_offset));
            self.pending_second_padding = None;
            self.decode_completed_group(out);
            let _ = pending_offset;
            return Ok(());
        }

        let position = self.group.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden || position < 2 {
                return Err(Error::BadPadding { offset: physical_offset });
            }
            if position == 2 {
                self.group.push((byte, physical_offset));
                self.pending_second_padding = Some(physical_offset);
                return Ok(());
            }

            // Position three padding is valid only after three alphabet digits.
            if position == 3
                && self.group.iter().all(|(b, _)| digit_value(*b, self.options.alphabet).is_some())
            {
                self.group.push((byte, physical_offset));
                self.decode_completed_group(out);
                return Ok(());
            }
            return Err(Error::BadPadding { offset: physical_offset });
        }

        if digit_value(byte, self.options.alphabet).is_none() {
            return Err(Error::InvalidByte { offset: physical_offset, byte });
        }

        self.group.push((byte, physical_offset));
        if self.group.len() == 4 {
            self.decode_completed_group(out);
        }
        Ok(())
    }

    fn process_wrapped_byte(
        &mut self,
        byte: u8,
        physical_offset: usize,
        out: &mut Vec<u8>,
    ) -> Result<(), Error> {
        let (width, ending) = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => (width, ending),
            _ => unreachable!(),
        };

        match self.frame_state {
            FrameState::NeedLf { cr_offset } => {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset: physical_offset });
                }
                self.frame_state = FrameState::NeedSymbol { separator_offset: cr_offset };
                return Ok(());
            }
            FrameState::NeedSymbol { .. } => {
                if matches!(byte, b'\r' | b'\n' | b' ' | b'\t' | 0x0b | 0x0c) {
                    return Err(Error::BadWrap { offset: physical_offset });
                }
                self.frame_state = FrameState::Normal;
                self.frame_symbols = 0;
            }
            FrameState::Normal => {}
        }

        if self.frame_symbols == width {
            match (ending, byte) {
                (LineEnding::Lf, b'\n') => {
                    self.frame_state = FrameState::NeedSymbol { separator_offset: physical_offset };
                    return Ok(());
                }
                (LineEnding::CrLf, b'\r') => {
                    self.frame_state = FrameState::NeedLf { cr_offset: physical_offset };
                    return Ok(());
                }
                _ => return Err(Error::BadWrap { offset: physical_offset }),
            }
        }

        if matches!(byte, b'\r' | b'\n' | b' ' | b'\t' | 0x0b | 0x0c) {
            return Err(Error::BadWrap { offset: physical_offset });
        }

        self.frame_symbols += 1;
        self.scan_symbol(byte, physical_offset, out)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let mut out = Vec::new();
        for &byte in input {
            let physical_offset = self.offset;
            self.offset += 1;

            let result = match self.options.whitespace {
                Whitespace::None => self.scan_symbol(byte, physical_offset, &mut out),
                Whitespace::IgnoreAscii => {
                    if matches!(byte, 9..=13 | 32) {
                        Ok(())
                    } else {
                        self.scan_symbol(byte, physical_offset, &mut out)
                    }
                }
                Whitespace::Wrapped { .. } => {
                    self.process_wrapped_byte(byte, physical_offset, &mut out)
                }
            };

            if let Err(error) = result {
                return Err(self.fail(error));
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if let FrameState::NeedLf { cr_offset } = self.frame_state {
            let error = Error::BadWrap { offset: cr_offset };
            return Err(self.fail(error));
        }
        if let FrameState::NeedSymbol { separator_offset } = self.frame_state {
            let error = Error::BadWrap { offset: separator_offset };
            return Err(self.fail(error));
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            return Err(self.fail(error));
        }

        let mut out = Vec::new();
        if !self.group.is_empty() {
            match self.options.padding {
                Padding::Required => {
                    let error = Error::Truncated { offset: self.group[0].1 };
                    return Err(self.fail(error));
                }
                Padding::Forbidden | Padding::Optional => {
                    match self.group.len() {
                        1 => {
                            let error = Error::Truncated { offset: self.group[0].1 };
                            return Err(self.fail(error));
                        }
                        2 => {
                            let a = digit_value(self.group[0].0, self.options.alphabet).unwrap();
                            let b = digit_value(self.group[1].0, self.options.alphabet).unwrap();
                            out.push((a << 2) | (b >> 4));
                        }
                        3 => {
                            let a = digit_value(self.group[0].0, self.options.alphabet).unwrap();
                            let b = digit_value(self.group[1].0, self.options.alphabet).unwrap();
                            let c = digit_value(self.group[2].0, self.options.alphabet).unwrap();
                            out.push((a << 2) | (b >> 4));
                            out.push((b << 4) | (c >> 2));
                        }
                        _ => {}
                    }
                    self.group.clear();
                }
            }
        }

        self.finished = true;
        Ok(out)
    }
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/",
        Alphabet::UrlSafe => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_",
    }
}

fn encode_symbols(input: &[u8], padding: Padding, alphabet: Alphabet) -> Vec<u8> {
    let table = alphabet_bytes(alphabet);
    let mut out = Vec::with_capacity((input.len() + 2) / 3 * 4);

    for chunk in input.chunks(3) {
        let a = chunk[0];
        let b = *chunk.get(1).unwrap_or(&0);
        let c = *chunk.get(2).unwrap_or(&0);

        out.push(table[(a >> 2) as usize]);
        out.push(table[(((a & 0x03) << 4) | (b >> 4)) as usize]);

        if chunk.len() >= 2 {
            out.push(table[(((b & 0x0f) << 2) | (c >> 6)) as usize]);
        } else if padding != Padding::Forbidden {
            out.push(b'=');
        }

        if chunk.len() == 3 {
            out.push(table[(c & 0x3f) as usize]);
        } else if padding != Padding::Forbidden {
            out.push(b'=');
        }
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let symbols = encode_symbols(input, options.padding, options.alphabet);

    let bytes = match options.whitespace {
        Whitespace::Wrapped { width, ending } => {
            let separator: &[u8] = match ending {
                LineEnding::Lf => b"\n",
                LineEnding::CrLf => b"\r\n",
            };
            let mut out = Vec::with_capacity(symbols.len() + symbols.len() / width * separator.len());
            for (i, chunk) in symbols.chunks(width).enumerate() {
                if i > 0 {
                    out.extend_from_slice(separator);
                }
                out.extend_from_slice(chunk);
            }
            out
        }
        _ => symbols,
    };

    Ok(String::from_utf8(bytes).expect("base64 output is ASCII"))
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    column: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            column: 0,
            finished: false,
        })
    }

    fn format_symbols(&mut self, symbols: &[u8]) -> String {
        let mut out = Vec::new();
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            for &symbol in symbols {
                if self.column == width {
                    match ending {
                        LineEnding::Lf => out.push(b'\n'),
                        LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
                    }
                    self.column = 0;
                }
                out.push(symbol);
                self.column += 1;
            }
        } else {
            out.extend_from_slice(symbols);
        }
        String::from_utf8(out).expect("base64 output is ASCII")
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let symbols = encode_symbols(
            &combined[..complete_len],
            Padding::Forbidden,
            self.options.alphabet,
        );
        self.tail.clear();
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(self.format_symbols(&symbols))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let symbols = encode_symbols(&self.tail, self.options.padding, self.options.alphabet);
        self.tail.clear();
        self.finished = true;
        Ok(self.format_symbols(&symbols))
    }
}

pub struct Decoder {
    options: Options,
    group: Vec<(u8, usize)>,
    pending_second_equals: Option<usize>,
    terminal_padding: bool,
    offset: usize,
    column: usize,

fn digit_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let table = alphabet_bytes(alphabet);
    table.iter().position(|&b| b == byte).map(|v| v as u8)
}

fn is_ignored(byte: u8, whitespace: Whitespace) -> bool {
    matches!(whitespace, Whitespace::IgnoreAscii)
        && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            pending_second_equals: None,
            terminal_padding: false,
            offset: 0,
            column: 0,
            frame: FrameState::Row,
            finished: false,
            sticky: None,
        })
    }

    fn record_error(&mut self, error: Error) -> Error {
        self.sticky = Some(error.clone());
        error
    }

    fn framed_symbol(&mut self, byte: u8, physical: usize) -> Result<bool, Error> {
        match self.options.whitespace {
            Whitespace::None | Whitespace::IgnoreAscii => Ok(true),
            Whitespace::Wrapped { width, ending } => {
                match self.frame {
                    FrameState::NeedLf { cr_offset } => {
                        if byte == b'\n' {
                            self.frame = FrameState::NeedSymbol {
                                first_offset: cr_offset,
                            };
                            Ok(false)
                        } else {
                            Err(Error::BadWrap { offset: physical })
                        }
                    }
                    FrameState::NeedSymbol { first_offset } => {
                        self.frame = FrameState::Row;
                        self.column = 0;
                        if byte == b'\r' || byte == b'\n' || byte == b' ' || byte == b'\t'
                            || byte == 11 || byte == 12
                        {
                            Err(Error::BadWrap { offset: physical })
                        } else {
                            let _ = first_offset;
                            Ok(true)
                        }
                    }
                    FrameState::Row => {
                        if self.column == width {
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.frame = FrameState::NeedSymbol {
                                        first_offset: physical,
                                    };
                                    Ok(false)
                                }
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.frame = FrameState::NeedLf {
                                        cr_offset: physical,
                                    };
                                    Ok(false)
                                }
                                _ => Err(Error::BadWrap { offset: physical }),
                            }
                        } else if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                            Err(Error::BadWrap { offset: physical })
                        } else {
                            self.column += 1;
                            Ok(true)
                        }
                    }
                }
            }
        }
    }

    fn consume_symbol(&mut self, byte: u8, physical: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset: physical });
        }

        if let Some(pending) = self.pending_second_equals {
            if byte != b'=' {
                return Err(Error::BadPadding { offset: physical });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: physical });
            }
            self.pending_second_equals = None;
            self.group.push((byte, physical));
            out.push((digit_value(self.group[0].0, self.options.alphabet).unwrap() << 2)
                | (digit_value(self.group[1].0, self.options.alphabet).unwrap() >> 4));
            self.group.clear();
            self.terminal_padding = true;
            let _ = pending;
            return Ok(());
        }

        let pos = self.group.len();
        if byte == b'=' {
            if pos <= 1 || self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: physical });
            }
            if pos == 2 {
                self.pending_second_equals = Some(physical);
                self.group.push((byte, physical));
                return Ok(());
            }
            if pos == 3 {
                let a = digit_value(self.group[0].0, self.options.alphabet).unwrap();
                let b = digit_value(self.group[1].0, self.options.alphabet).unwrap();
                let c = digit_value(self.group[2].0, self.options.alphabet).unwrap();
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                self.group.clear();
                self.terminal_padding = true;
                return Ok(());
            }
        }

        let Some(value) = digit_value(byte, self.options.alphabet) else {
            return Err(Error::InvalidByte {
                offset: physical,
                byte,
            });
        };

        self.group.push((byte, physical));
        if self.group.len() == 4 {
            let a = digit_value(self.group[0].0, self.options.alphabet).unwrap();
            let b = digit_value(self.group[1].0, self.options.alphabet).unwrap();
            let c = digit_value(self.group[2].0, self.options.alphabet).unwrap();
            let d = value;
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);
            self.group.clear();
        }
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky = Some(error.clone());
            return Err(error);
        }

        let mut out = Vec::new();
        for &byte in input {
            let physical = self.offset;
            self.offset += 1;

            if is_ignored(byte, self.options.whitespace) {
                continue;
            }
            match self.framed_symbol(byte, physical) {
                Ok(false) => continue,
                Ok(true) => {}
                Err(error) => return Err(self.record_error(error)),
            }
            if let Err(error) = self.consume_symbol(byte, physical, &mut out) {
                return Err(self.record_error(error));
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky = Some(error.clone());
            return Err(error);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame {
                FrameState::NeedLf { cr_offset } => {
                    let error = Error::BadWrap { offset: cr_offset };
                    return Err(self.record_error(error));
                }
                FrameState::NeedSymbol { first_offset } => {
                    let error = Error::BadWrap {
                        offset: first_offset,
                    };
                    return Err(self.record_error(error));
                }
                FrameState::Row => {}
            }
        }

        if let Some(offset) = self.pending_second_equals {
            let error = Error::BadPadding { offset };
            return Err(self.record_error(error));
        }

        let mut out = Vec::new();
        if self.group.is_empty() {
            self.finished = true;
            return Ok(out);
        }

        let len = self.group.len();
        if self.options.padding == Padding::Required {
            let error = Error::Truncated {
                offset: self.group[0].1,
            };
            return Err(self.record_error(error));
        }

        if len == 1 || self.group.iter().any(|(b, _)| *b == b'=') {
            let error = Error::Truncated {
                offset: self.group[0].1,
            };
            return Err(self.record_error(error));
        }

        let a = digit_value(self.group[0].0, self.options.alphabet).unwrap();
        let b = digit_value(self.group[1].0, self.options.alphabet).unwrap();
        out.push((a << 2) | (b >> 4));
        if len == 3 {
            let c = digit_value(self.group[2].0, self.options.alphabet).unwrap();
            out.push((b << 4) | (c >> 2));
        }
        self.group.clear();
        self.finished = true;
        Ok(out)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

        52..=61 => b'0' + value - 52,

    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_symbols(input: &[u8], options: Options) -> Vec<u8> {
    let mut output = Vec::with_capacity((input.len() + 2) / 3 * 4);
    let mut i = 0;
    while i + 3 <= input.len() {
        let a = input[i];
        let b = input[i + 1];
        let c = input[i + 2];
        output.push(alphabet_byte(a >> 2, options.alphabet));
        output.push(alphabet_byte(((a & 0x03) << 4) | (b >> 4), options.alphabet));
        output.push(alphabet_byte(((b & 0x0f) << 2) | (c >> 6), options.alphabet));
        output.push(alphabet_byte(c & 0x3f, options.alphabet));
        i += 3;
    }

    let remaining = input.len() - i;
    if remaining == 1 {
        let a = input[i];
        output.push(alphabet_byte(a >> 2, options.alphabet));
        output.push(alphabet_byte((a & 0x03) << 4, options.alphabet));
        if options.padding != Padding::Forbidden {
            output.extend_from_slice(b"==");
        }
    } else if remaining == 2 {
        let a = input[i];
        let b = input[i + 1];
        output.push(alphabet_byte(a >> 2, options.alphabet));
        output.push(alphabet_byte(((a & 0x03) << 4) | (b >> 4), options.alphabet));
        output.push(alphabet_byte((b & 0x0f) << 2, options.alphabet));
        if options.padding != Padding::Forbidden {
            output.push(b'=');
        }
    }
    output
}

fn wrap_symbols(symbols: &[u8], whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return String::from_utf8(symbols.to_vec()).unwrap();
    };

    let mut output = Vec::with_capacity(symbols.len() + symbols.len() / width * 2);
    for (i, &byte) in symbols.iter().enumerate() {

            match ending {
                LineEnding::Lf => output.push(b'\n'),
                LineEnding::CrLf => output.extend_from_slice(b"\r\n"),
            }
        }
        output.push(byte);
    }
    String::from_utf8(output).unwrap()
}

struct Symbol {

struct Framer {
    whitespace: Whitespace,
    offset: usize,
    row_count: usize,
    separator_state: u8,
    separator_start: usize,
    after_separator: bool,
}

impl Framer {
    fn new(whitespace: Whitespace) -> Self {
        Self {
            whitespace,
            offset: 0,
            row_count: 0,
            separator_state: 0,
            separator_start: 0,
            after_separator: false,
        }
    }

    fn feed(&mut self, byte: u8) -> Result<Option<Symbol>, Error> {
        let offset = self.offset;
        self.offset += 1;

        match self.whitespace {
            Whitespace::None => Ok(Some(Symbol { byte, offset })),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9..=13 | 32) {
                    Ok(None)
                } else {
                    Ok(Some(Symbol { byte, offset }))
                }
            }
            Whitespace::Wrapped { width, ending } => {
                if self.separator_state == 1 {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.separator_state = 0;
                    self.after_separator = true;
                    return Ok(None);
                }

                if self.row_count == width {
                    match ending {
                        LineEnding::Lf => {
                            if byte != b'\n' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.separator_start = offset;
                            self.after_separator = true;
                            self.row_count = 0;
                            return Ok(None);
                        }
                        LineEnding::CrLf => {
                            if byte != b'\r' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.separator_start = offset;
                            self.separator_state = 1;
                            self.row_count = 0;
                            return Ok(None);
                        }
                    }
                }

                if matches!(byte, 9..=13 | 32) {
                    return Err(Error::BadWrap { offset });
                }

                self.row_count += 1;
                self.after_separator = false;
                Ok(Some(Symbol { byte, offset }))
            }
        }
    }

    fn finish(&self) -> Result<(), Error> {
        if matches!(self.whitespace, Whitespace::Wrapped { .. }) {
            if self.separator_state == 1 {
                return Err(Error::BadWrap {
                    offset: self.separator_start,
                });
            }
            if self.after_separator {
                return Err(Error::BadWrap {
                    offset: self.separator_start,
                });
            }
        }
        Ok(())
    }
}

struct DecodeCore {
    options: Options,
    framer: Framer,
    group: Vec<Symbol>,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
}

impl DecodeCore {
    fn new(options: Options) -> Self {
        Self {
            options,
            framer: Framer::new(options.whitespace),
            group: Vec::with_capacity(4),
            pending_second_padding: None,
            terminal_padding: false,
        }
    }

    fn feed(&mut self, byte: u8, output: &mut Vec<u8>) -> Result<(), Error> {
        let Some(symbol) = self.framer.feed(byte)? else {
            return Ok(());
        };

        if self.terminal_padding {
            return Err(Error::BadPadding {
                offset: symbol.offset,
            });
        }

        if let Some(first_padding) = self.pending_second_padding.take() {
            if symbol.byte != b'=' {
                return Err(Error::BadPadding {
                    offset: symbol.offset,
                });
            }
            self.group.push(symbol);
            self.complete_padded_group(output);
            let _ = first_padding;
            return Ok(());
        }

        let position = self.group.len();
        if symbol.byte == b'=' {
            match position {
                0 | 1 => return Err(Error::BadPadding { offset: symbol.offset }),
                2 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset: symbol.offset });
                    }
                    self.pending_second_padding = Some(symbol.offset);
                    self.group.push(symbol);
                    return Ok(());
                }
                3 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset: symbol.offset });
                    }
                    self.group.push(symbol);
                    self.complete_padded_group(output);
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        if position == 3 && self.group.get(2).is_some_and(|s| s.byte == b'=') {
            return Err(Error::BadPadding {
                offset: symbol.offset,
            });
        }

        let Some(value) = decode_digit(symbol.byte, self.options.alphabet) else {
            return Err(Error::InvalidByte {
                offset: symbol.offset,
                byte: symbol.byte,
            });
        };
        let _ = value;
        self.group.push(symbol);

        if self.group.len() == 4 {
            let values = self.group_values()?;
            output.extend_from_slice(&[
                (values[0] << 2) | (values[1] >> 4),
                (values[1] << 4) | (values[2] >> 2),
                (values[2] << 6) | values[3],
            ]);
            self.group.clear();
        }
        Ok(())
    }

    fn group_values(&self) -> Result<[u8; 4], Error> {
        let mut values = [0; 4];
        for (i, symbol) in self.group.iter().enumerate() {
            values[i] = decode_digit(symbol.byte, self.options.alphabet)
                .ok_or(Error::InvalidByte {
                    offset: symbol.offset,
                    byte: symbol.byte,
                })?;
        }
        Ok(values)
    }

    fn complete_padded_group(&mut self, output: &mut Vec<u8>) {
        if self.group.len() == 4 {
            let first = decode_digit(self.group[0].byte, self.options.alphabet).unwrap_or(0);
            let second = decode_digit(self.group[1].byte, self.options.alphabet).unwrap_or(0);
            if self.group[2].byte == b'=' {
                output.push((first << 2) | (second >> 4));
            } else {
                let third = decode_digit(self.group[2].byte, self.options.alphabet).unwrap_or(0);
                output.push((first << 2) | (second >> 4));
                output.push((second << 4) | (third >> 2));
            }
            self.group.clear();
            self.terminal_padding = true;
        }
    }

    fn finish(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        self.framer.finish()?;

        if let Some(offset) = self.pending_second_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group.is_empty() {
            return Ok(());
        }

        match self.group.len() {
            1 => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
            2 | 3 if self.options.padding != Padding::Required => {
                let values = self.group_values()?;
                output.push((values[0] << 2) | (values[1] >> 4));
                if self.group.len() == 3 {
                    output.push((values[1] << 4) | (values[2] >> 2));
                }
                self.group.clear();
                Ok(())
            }
            _ => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut core = DecodeCore::new(options);
    let mut output = Vec::new();
    for &byte in input {
        core.feed(byte, &mut output)?;
    }
    core.finish(&mut output)?;
    Ok(output)
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    wrapped_symbols: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffered: Vec::with_capacity(2),
            wrapped_symbols: 0,
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        if input.is_empty() {
            return Ok(String::new());
        }

        let mut data = Vec::with_capacity(self.buffered.len() + input.len());
        data.extend_from_slice(&self.buffered);
        data.extend_from_slice(input);
        let complete_len = data.len() / 3 * 3;
        self.buffered.clear();
        self.buffered.extend_from_slice(&data[complete_len..]);

        let symbols = encode_symbols(&data[..complete_len], self.options);
        Ok(self.format_new_symbols(&symbols))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let symbols = encode_symbols(&self.buffered, self.options);
        self.buffered.clear();
        Ok(self.format_new_symbols(&symbols))
    }

    fn format_new_symbols(&mut self, symbols: &[u8]) -> String {
        let Whitespace::Wrapped { width, ending } = self.options.whitespace else {
            return String::from_utf8(symbols.to_vec()).unwrap();
        };

        let mut output = Vec::new();
        for &byte in symbols {
            if self.wrapped_symbols > 0 && self.wrapped_symbols % width == 0 {
                match ending {
                    LineEnding::Lf => output.push(b'\n'),
                    LineEnding::CrLf => output.extend_from_slice(b"\r\n"),
                }
            }
            output.push(byte);
            self.wrapped_symbols += 1;
        }
        String::from_utf8(output).unwrap()
    }
}

pub struct Decoder {
    core: DecodeCore,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            core: DecodeCore::new(options),
            finished: false,
            sticky_error: None,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let mut output = Vec::new();
        for &byte in input {
            if let Err(error) = self.core.feed(byte, &mut output) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let mut output = Vec::new();
        match self.core.finish(&mut output) {
            Ok(()) => {
                self.finished = true;
                Ok(output)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }
}
