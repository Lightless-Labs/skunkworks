    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(id, Transaction {
            revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let tx = self.transactions.get(&id).ok_or(Error::UnknownTransaction(id))?;
        if !tx.writes.is_empty() {
            for key in tx.writes.keys() {
                if self.data.get(key)
                    .and_then(|vs| vs.last())
                    .map_or(false, |v| v.revision > tx.revision)
                {
                    return Err(Error::Conflict { tx: id, key: key.clone() });
                }
            }
        }
        let transaction = self.transactions.remove(&id).expect("validated transaction");
        if transaction.writes.is_empty() {
            return Ok(self.revision);
        }
        self.revision += 1;
        let revision = self.revision;
        for (key, entry) in transaction.writes {
            self.append_version(key, entry, revision);
        }
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(&mut self, now: u64, tx: Option<u64>, key: &str, value: &str, ttl: Option<u64>) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let entry = match ttl {
            Some(0) => Entry::Tombstone,
            Some(d) => Entry::Value(Value {
                value: value.to_string(),
                deadline: Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            }),
            None => Entry::Value(Value { value: value.to_string(), deadline: None }),
        };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), entry);
        } else {
            self.revision += 1;
            let rev = self.revision;
            self.append_version(key.to_string(), entry, rev);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let existed = self.visible_value(key, tx, now).is_some();
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), Entry::Tombstone);
        } else {
            self.revision += 1;
            let rev = self.revision;
            self.append_version(key.to_string(), Entry::Tombstone, rev);
        }
        Ok(existed)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_key(key)?;
        Ok(self.visible_value(key, tx, now))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self.visible_entries(tx, now).values().filter(|v| v.as_str() == value).count())
    }

    fn in_range(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.map_or(false, |e| e <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::in_range(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }
        if limit == 0 {
            return Ok(Page { items: Vec::new(), next: None });
        }
        let mut entries: Vec<(String, String)> = self.visible_entries(tx, now)
            .into_iter()
            .filter(|(k, _)| Self::in_range(k, start, end, prefix)
                && cursor.map_or(true, |c| k.as_str() > c.last.as_str()))
            .collect();
        if cursor.is_none() {
            let skip = offset.min(entries.len());
            entries.drain(0..skip);
        }
        let take = limit.min(entries.len());
        let has_more = take < entries.len();
        let items: Vec<_> = entries.into_iter().take(take).collect();
        let next = if has_more {
            let last = items.last().unwrap().0.clone();
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last,
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.map_or(false, |e| e <= start) {
            return Err(Error::InvalidRange);
        }
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.visible_entries(tx, now) {
            if !Self::in_range(&key, start, end, prefix) {
                continue;
            }
            let number = Self::parse_number(&key, &value)?;
            sum = sum.checked_add(number).ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    fn parse_number(key: &str, value: &str) -> Result<i64, Error> {
        let bytes = value.as_bytes();
        let invalid = || Error::InvalidNumber { key: key.to_string(), value: value.to_string() };
        if bytes.is_empty() {
            return Err(invalid());
        }
        let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
        if digits.is_empty() || !digits.iter().all(u8::is_ascii_digit) {
            return Err(invalid());
        }
        if digits[0] == b'0' && (digits.len() > 1 || bytes[0] == b'-') {
            return Err(invalid());
        }
        if bytes[0] == b'0' && bytes.len() > 1 {
            return Err(invalid());
        }
        value.parse::<i64>().map_err(|_| invalid())
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let mut saved = BTreeMap::new();
        for (key, versions) in &self.data {
            if let Some(version) = versions.last() {
                if let Some(value) = Self::live(&version.entry, now) {
                    saved.insert(key.clone(), value.clone());
                }
            }
        }
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let saved = self.snapshots.get(name).ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }
        let saved_live: BTreeMap<String, Value> = saved.iter()
            .filter(|(_, v)| v.deadline.map_or(true, |d| d > now))
            .map(|(k, v)| (k.clone(), v.clone()))
            .collect();
        let current_live: BTreeMap<String, ()> = self.data.iter()
            .filter_map(|(k, vs)| {
                vs.last().and_then(|v| Self::live(&v.entry, now)).map(|_| (k.clone(), ()))
            })
            .collect();
        let keys: BTreeSet<String> = current_live.keys().cloned().chain(saved_live.keys().cloned()).collect();
        if keys.is_empty() {
            return Ok(self.revision);
        }
        self.revision += 1;
        let rev = self.revision;
        for key in keys {
            let entry = match saved_live.get(&key) {
                Some(v) => Entry::Value(v.clone()),
                None => Entry::Tombstone,
            };
            self.append_version(key, entry, rev);
        }
        Ok(rev)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0usize;
        for versions in self.data.values_mut() {
            let mut keep: BTreeSet<usize> = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for transaction in self.transactions.values() {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= transaction.revision) {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let live_keys = self.data.values().filter(|vs| {
            vs.last().map_or(false, |v| Self::live(&v.entry, now).is_some())
        }).count();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values().flat_map(|vs| vs.iter())
            .filter(|v| matches!(v.entry, Entry::Tombstone))
            .count();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        })
    }
