#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    next_offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_group(bytes: &[u8]) -> String {
    let mut out = String::with_capacity(4);
    match bytes.len() {
        3 => {
            let a = bytes[0];
            let b = bytes[1];
            let c = bytes[2];
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
            out.push(ALPHABET[(c & 0x3f) as usize] as char);
        }
        1 => {
            let a = bytes[0];
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            out.push_str("==");
        }
        2 => {
            let a = bytes[0];
            let b = bytes[1];
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            out.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            out.push('=');
        }
        _ => {}
    }
    out
}

fn alphabet_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn decode_group(group: &[u8; 4]) -> Vec<u8> {
    let a = alphabet_value(group[0]).unwrap();
    let b = alphabet_value(group[1]).unwrap();
    let mut out = Vec::with_capacity(3);

    if group[2] == b'=' {
        out.push((a << 2) | (b >> 4));
    } else {
        let c = alphabet_value(group[2]).unwrap();
        if group[3] == b'=' {
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
        } else {
            let d = alphabet_value(group[3]).unwrap();
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);
        }
    }
    out
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::new();
        let mut i = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && i < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[i];
                }
                self.tail_len += 1;
                i += 1;
            }
            if self.tail_len == 3 {
                let group = [self.tail[0], self.tail[1], input[i - 1]];
                out.push_str(&encode_group(&group));
                self.tail_len = 0;
            }
        }

        while i + 3 <= input.len() {
            out.push_str(&encode_group(&input[i..i + 3]));
            i += 3;
        }

        while i < input.len() {
            self.tail[self.tail_len] = input[i];
            self.tail_len += 1;
            i += 1;
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let out = encode_group(&self.tail[..self.tail_len]);
        self.tail_len = 0;
        Ok(out)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            next_offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            if self.terminal_padding {
                return Err(self.fail(Error::BadPadding { offset }));
            }

            if let Some(pending_offset) = self.pending_second_padding {
                if byte != b'=' {
                    return Err(self.fail(Error::BadPadding { offset }));
                }
                self.group[3] = b'=';
                self.offsets[3] = offset;
                self.group_len = 4;
                self.pending_second_padding = None;
                self.terminal_padding = true;
                out.extend(decode_group(&self.group));
                let _ = pending_offset;
                continue;
            }

            let position = self.group_len;
            if byte == b'=' {
                match position {
                    0 | 1 => return Err(self.fail(Error::BadPadding { offset })),
                    2 => {
                        self.group[2] = b'=';
                        self.offsets[2] = offset;
                        self.group_len = 3;
                        self.pending_second_padding = Some(offset);
                    }
                    3 => {
                        self.group[3] = b'=';
                        self.offsets[3] = offset;
                        self.group_len = 4;
                        self.terminal_padding = true;
                        out.extend(decode_group(&self.group));
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            if alphabet_value(byte).is_none() {
                return Err(self.fail(Error::InvalidByte { offset, byte }));
            }

            self.group[position] = byte;
            self.offsets[position] = offset;
            self.group_len += 1;

            if self.group_len == 4 {
                out.extend(decode_group(&self.group));
                self.group_len = 0;
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            return Err(self.fail(error));
        }

        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            return Err(self.fail(error));
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut out = encoder.push(input).unwrap();
    out.push_str(&encoder.finish().unwrap());
    out
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
