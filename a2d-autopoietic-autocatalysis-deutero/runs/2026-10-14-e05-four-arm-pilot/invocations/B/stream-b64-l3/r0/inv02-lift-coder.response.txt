use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{self:?}")
    }
}

impl std::error::Error for Error {}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/",
        Alphabet::UrlSafe => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_",
    }
}

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8], alphabet: Alphabet, padding: Padding) -> Vec<u8> {
    let table = alphabet_bytes(alphabet);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };
    let mut out = vec![
        table[(a >> 2) as usize],
        table[(((a & 0x03) << 4) | (b >> 4)) as usize],
    ];
    if bytes.len() > 1 {
        out.push(table[(((b & 0x0f) << 2) | (c >> 6)) as usize]);
    } else if padding != Padding::Forbidden {
        out.extend_from_slice(b"==");
        return out;
    }
    if bytes.len() > 2 {
        out.push(table[(c & 0x3f) as usize]);
    } else if padding != Padding::Forbidden {
        out.push(b'=');
    }
    out
}

fn wrap_symbols(symbols: &[u8], whitespace: Whitespace) -> String {
    let mut out = Vec::new();
    match whitespace {
        Whitespace::Wrapped { width, ending } => {
            for (i, &b) in symbols.iter().enumerate() {
                if i > 0 && i % width == 0 {
                    match ending {
                        LineEnding::Lf => out.push(b'\n'),
                        LineEnding::CrLf => out.extend_from_slice(b"\r\n"),
                    }
                }
                out.push(b);
            }
        }
        _ => out.extend_from_slice(symbols),
    }
    String::from_utf8(out).expect("base64 output is ASCII")
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut out = encoder.push(input)?;
    out.push_str(&encoder.finish()?);
    Ok(out)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            pending: Vec::new(),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut data = std::mem::take(&mut self.pending);
        data.extend_from_slice(input);
        let complete_len = data.len() / 3 * 3;
        let mut symbols = Vec::with_capacity((complete_len / 3) * 4);
        for chunk in data[..complete_len].chunks_exact(3) {
            symbols.extend(encode_group(chunk, self.options.alphabet, self.options.padding));
        }
        self.pending.extend_from_slice(&data[complete_len..]);
        Ok(wrap_symbols(&symbols, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        if self.pending.is_empty() {
            return Ok(String::new());
        }
        let symbols = encode_group(
            &self.pending,
            self.options.alphabet,
            self.options.padding,
        );
        self.pending.clear();
        Ok(wrap_symbols(&symbols, self.options.whitespace))
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    value: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: Vec<Symbol>,
    terminal_padding: bool,
    offset: usize,
    row_symbols: usize,
    frame_state: FrameState,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::new(),
            terminal_padding: false,
            offset: 0,
            row_symbols: 0,
            frame_state: FrameState::Normal,
            finished: false,
            sticky_error: None,
        })
    }

    fn record_error(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn append_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        let position = self.group.len();

        if position == 0 || position == 1 {
            if byte == b'=' {
                return Err(Error::BadPadding { offset });
            }
            let value = digit(byte, self.options.alphabet)
                .ok_or(Error::InvalidByte { offset, byte })?;
            self.group.push(Symbol { value, offset });
            return Ok(());
        }

        if position == 2 {
            if byte == b'=' {
                self.group.push(Symbol { value: 64, offset });
                return Ok(());
            }
            let value = digit(byte, self.options.alphabet)
                .ok_or(Error::InvalidByte { offset, byte })?;
            self.group.push(Symbol { value, offset });
            return Ok(());
        }

        if self.group[2].value == 64 {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.group.push(Symbol { value: 64, offset });
            out.push((self.group[0].value << 2) | (self.group[1].value >> 4));
            self.group.clear();
            self.terminal_padding = true;
            return Ok(());
        }

        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            out.push((self.group[0].value << 2) | (self.group[1].value >> 4));
            out.push((self.group[1].value << 4) | (self.group[2].value >> 2));
            self.group.clear();
            self.terminal_padding = true;
            return Ok(());
        }

        let value = digit(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset, byte })?;
        out.push((self.group[0].value << 2) | (self.group[1].value >> 4));
        out.push((self.group[1].value << 4) | (self.group[2].value >> 2));
        out.push((self.group[2].value << 6) | value);
        self.group.clear();
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.append_symbol(byte, offset, out),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.append_symbol(byte, offset, out)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.frame_state {
                    FrameState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.frame_state = FrameState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        return Ok(());
                    }
                    FrameState::NeedSymbol { .. } => {
                        self.frame_state = FrameState::Normal;
                        self.row_symbols = 0;
                    }
                    FrameState::Normal => {}
                }

                if self.row_symbols == width {
                    match ending {
                        LineEnding::Lf => {
                            if byte != b'\n' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.frame_state = FrameState::NeedSymbol {
                                separator_offset: offset,
                            };
                            return Ok(());
                        }
                        LineEnding::CrLf => {
                            if byte != b'\r' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.frame_state = FrameState::NeedLf { cr_offset: offset };
                            return Ok(());
                        }
                    }
                }

                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset });
                }

                self.append_symbol(byte, offset, out)?;
                self.row_symbols += 1;
                Ok(())
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut out) {
                return Err(self.record_error(error));
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedLf { cr_offset } => {
                    return Err(self.record_error(Error::BadWrap { offset: cr_offset }));
                }
                FrameState::NeedSymbol { separator_offset } => {
                    return Err(self.record_error(Error::BadWrap {
                        offset: separator_offset,
                    }));
                }
                FrameState::Normal => {}
            }
        }

        if self.group.len() == 3 && self.group[2].value == 64 {
            let offset = self.group[2].offset;
            return Err(self.record_error(Error::BadPadding { offset }));
        }

        if !self.group.is_empty() {
            match self.options.padding {
                Padding::Required => {
                    let offset = self.group[0].offset;
                    return Err(self.record_error(Error::Truncated { offset }));
                }
                Padding::Forbidden | Padding::Optional => match self.group.len() {
                    1 => {
                        let offset = self.group[0].offset;
                        return Err(self.record_error(Error::Truncated { offset }));
                    }
                    2 => {
                        let out = vec![
                            (self.group[0].value << 2) | (self.group[1].value >> 4),
                        ];
                        self.group.clear();
                        self.finished = true;
                        return Ok(out);
                    }
                    3 => {
                        let out = vec![
                            (self.group[0].value << 2) | (self.group[1].value >> 4),
                            (self.group[1].value << 4) | (self.group[2].value >> 2),
                        ];
                        self.group.clear();
                        self.finished = true;
                        return Ok(out);
                    }
                    _ => {}
                },
            }
        }

        self.finished = true;
        Ok(Vec::new())
    }
}
