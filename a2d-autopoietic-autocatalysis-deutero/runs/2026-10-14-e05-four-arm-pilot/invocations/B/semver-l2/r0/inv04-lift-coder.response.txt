use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_digits(s: &str, start: usize) -> Result<(u64, usize), VersionError> {
    let bytes = s.as_bytes();
    let mut end = start;
    while end < bytes.len() && bytes[end].is_ascii_digit() {
        end += 1;
    }
    if end == start {
        let at = start;
        return Err(if at == bytes.len() || matches!(bytes[at], b'.' | b'-' | b'+') {
            VersionError::MissingCore { at }
        } else {
            VersionError::InvalidChar { at }
        });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &b in &bytes[start..end] {
        let digit = (b - b'0') as u64;
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(digit))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok((value, end))
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut pos = start;
    loop {
        let ident_start = pos;
        while pos < end && bytes[pos] != b'.' {
            let b = bytes[pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
        if ident_start == pos {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..pos];
        if prerelease
            && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.len() > 1
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        out.push(ident.to_string());
        if pos == end {
            break;
        }
        pos += 1;
        if pos == end {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let (major, mut pos) = parse_digits(s, 0)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;

    let (minor, next) = parse_digits(s, pos)?;
    pos = next;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;

    let (patch, next) = parse_digits(s, pos)?;
    pos = next;

    let bytes = s.as_bytes();
    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let start = pos + 1;
        let plus = bytes[start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| start + i)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(s, start, plus, true)?;
        pos = plus;
    }
    if pos < bytes.len() {
        // The only remaining valid delimiter is the build separator.
        let start = pos + 1;
        build = parse_identifiers(s, start, bytes.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    let an = is_numeric_identifier(a);
    let bn = is_numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_identifier(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    admission: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut pos = 0;
    while pos < bytes.len() {
        while pos < bytes.len() && is_range_space(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }
        let start = pos;
        while pos < bytes.len() && !is_range_space(bytes[pos]) {
            pos += 1;
        }
        tokens.push(Token {
            text: &s[start..pos],
            start,
        });
    }
    tokens
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    }
}

fn padded(fields: &[(u64, bool)], full: Option<&Version>) -> Version {
    if let Some(v) = full {
        return v.clone();
    }
    Version {
        major: fields.first().map(|x| x.0).unwrap_or(0),
        minor: fields.get(1).map(|x| x.0).unwrap_or(0),
        patch: fields.get(2).map(|x| x.0).unwrap_or(0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    version: Version,
    numeric_count: usize,
    full: bool,
}

fn parse_operand(s: &str, token: &Token<'_>, operand: &str, offset: usize) -> Result<Partial, RangeError> {
    let is_full_candidate = operand.contains('-')
        || operand.contains('+')
        || (operand.split('.').count() >= 3
            && !operand.split('.').any(|f| matches!(f, "*" | "x" | "X")));

    if is_full_candidate {
        let v = parse_version(operand).map_err(|error| RangeError::Version {
            at: token.start + offset,
            error,
        })?;
        return Ok(Partial {
            version: v,
            numeric_count: 3,
            full: true,
        });
    }

    let fields: Vec<&str> = operand.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: token.start });
    }
    let mut values = Vec::new();
    let mut saw_wild = false;
    let mut numeric_count = 0;
    for field in fields {
        if matches!(field, "*" | "x" | "X") {
            saw_wild = true;
        } else {
            if saw_wild
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: token.start });
            }
            let n = field
                .parse::<u64>()
                .map_err(|_| RangeError::Syntax { at: token.start })?;
            numeric_count += 1;
            values.push((n, true));
        }
    }
    let version = Version {
        major: values.first().map(|x| x.0).unwrap_or(0),
        minor: values.get(1).map(|x| x.0).unwrap_or(0),
        patch: values.get(2).map(|x| x.0).unwrap_or(0),
        pre: Vec::new(),
        build: Vec::new(),
    };
    let _ = s;
    Ok(Partial {
        version,
        numeric_count,
        full: false,
    })
}

fn add_check(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.admission.push(tuple(&version));
    }
    arm.checks.push(Check { op, version });
}

fn bump_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut major = v.major;
    let mut minor = v.minor;
    let mut patch = v.patch;
    match field {
        0 => {
            major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            minor = 0;
            patch = 0;
        }
        1 => {
            minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            patch = 0;
        }
        _ => patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    Ok(endpoint(major, minor, patch))
}

fn add_bare(arm: &mut Arm, p: Partial, at: usize) -> Result<(), RangeError> {
    if p.full {
        add_check(arm, Op::Eq, p.version.clone(), true);
        return Ok(());
    }
    match p.numeric_count {
        0 => {}
        1 => {
            add_check(arm, Op::Ge, p.version.clone(), false);
            add_check(arm, Op::Lt, bump_upper(&p.version, 0, at)?, false);
        }
        2 => {
            add_check(arm, Op::Ge, p.version.clone(), false);
            add_check(arm, Op::Lt, bump_upper(&p.version, 1, at)?, false);
        }
        _ => add_check(arm, Op::Eq, p.version, false),
    }
    Ok(())
}

fn parse_arm(tokens: &[Token<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    let standalone_minus = tokens.iter().position(|t| t.text == "-");
    if let Some(minus_idx) = standalone_minus {
        if tokens.len() != 3 || minus_idx != 1 {
            return Err(RangeError::Syntax {
                at: tokens[minus_idx].start,
            });
        }
        let lower = parse_operand("", &tokens[0], tokens[0].text, 0)?;
        let upper = parse_operand("", &tokens[2], tokens[2].text, 0)?;
        if lower.numeric_count == 0 {
            return Err(RangeError::Syntax { at: tokens[0].start });
        }
        if upper.numeric_count == 0 {
            return Err(RangeError::Syntax { at: tokens[2].start });
        }
        add_check(arm, Op::Ge, lower.version.clone(), true);
        if upper.full {
            add_check(arm, Op::Le, upper.version.clone(), true);
        } else {
            let field = if upper.numeric_count == 1 { 0 } else { 1 };
            add_check(arm, Op::Lt, bump_upper(&upper.version, field, tokens[2].start)?, false);
        }
        return Ok(());
    }

    for token in tokens {
        let text = token.text;
        if text == "&" || text.contains(',') || text.contains('(') || text.contains(')') {
            return Err(RangeError::Syntax { at: token.start });
        }
        if text.starts_with('^') || text.starts_with('~') {
            let operand = &text[1..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() && !matches!(operand, "*" | "x" | "X") {
                return Err(RangeError::Syntax { at: token.start });
            }
            let p = parse_operand("", token, operand, 1)?;
            if p.numeric_count == 0 {
                return Err(RangeError::Syntax { at: token.start });
            }
            add_check(arm, Op::Ge, p.version.clone(), true);
            let upper_field = if text.starts_with('~') {
                if p.numeric_count == 1 { 0 } else { 1 }
            } else if p.version.major > 0 || p.numeric_count == 1 {
                0
            } else if p.version.minor > 0 || p.numeric_count == 2 {
                1
            } else {
                2
            };
            add_check(arm, Op::Lt, bump_upper(&p.version, upper_field, token.start + 1)?, false);
            continue;
        }

        let (op, prefix_len) = if text.starts_with(">=") {
            (Some(Op::Ge), 2)
        } else if text.starts_with("<=") {
            (Some(Op::Le), 2)
        } else if text.starts_with('>') {
            (Some(Op::Gt), 1)
        } else if text.starts_with('<') {
            (Some(Op::Lt), 1)
        } else if text.starts_with('=') {
            (Some(Op::Eq), 1)
        } else {
            (None, 0)
        };
        if let Some(op) = op {
            let operand = &text[prefix_len..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: token.start });
            }
            let v = parse_version(operand).map_err(|error| RangeError::Version {
                at: token.start + prefix_len,
                error,
            })?;
            add_check(arm, op, v, true);
            continue;
        }

        if !(text.as_bytes()[0].is_ascii_digit() || matches!(text, "*" | "x" | "X")) {
            return Err(RangeError::Syntax { at: token.start });
        }
        let p = parse_operand("", token, text, 0)?;
        add_bare(arm, p, token.start)?;
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut last_or_end = None;
    for token in tokens {
        if token.text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: token.start });
            }
            let mut arm = Arm {
                checks: Vec::new(),
                admission: Vec::new(),
            };
            parse_arm(&current, &mut arm)?;
            arms.push(arm);
            current.clear();
            last_or_end = Some(token.start + token.text.len());
        } else {
            current.push(token);
            last_or_end = None;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty {
            at: last_or_end.unwrap_or(s.len()),
        });
    }
    let mut arm = Arm {
        checks: Vec::new(),
        admission: Vec::new(),
    };
    parse_arm(&current, &mut arm)?;
    arms.push(arm);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm) -> bool {
    if !v.pre.is_empty() && !arm.admission.contains(&tuple(v)) {
        return false;
    }
    arm.checks.iter().all(|check| {
        let ord = compare_versions(v, &check.version);
        match check.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<(usize, Ordering)> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some((index, Ordering::Equal)),
            Some((best_index, _)) => {
                let ord = compare_versions(version, &parsed[best_index]);
                if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                    best = Some((index, ord));
                }
            }
        }
    }
    Ok(best.map(|(index, _)| candidates[index].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    fn pr(s: &str) -> Range {
        parse_range(s).unwrap()
    }

    #[test]
    fn public_version() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(
            compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")),
            Ordering::Less
        );
    }

    #[test]
    fn public_comparators() {
        let r = pr(">=2.1.0 <3.0.0");
        assert!(matches(&pv("2.5.0"), &r));
        assert!(!matches(&pv("3.0.0"), &r));
        assert!(matches(
            &pv("2.1.3-rc"),
            &pr(">=2.1.3-beta <2.1.3")
        ));
    }

    #[test]
    fn public_expanded() {
        for s in ["^2.1.0", "~2.5.0", "2.5.x", "2.1 - 2.8"] {
            assert!(matches(&pv("2.5.1"), &pr(s)));
        }
        assert!(matches(&pv("4.0.0"), &pr("2.x || 4.0.0")));
    }

    #[test]
    fn public_extrema() {
        let cs = vec![
            "2.2.0".to_string(),
            "2.8.0".to_string(),
            "3.0.0".to_string(),
        ];
        let r = pr("2.x");
        assert_eq!(max_satisfying(&cs, &r), Ok(Some("2.8.0".into())));
        assert_eq!(min_satisfying(&cs, &r), Ok(Some("2.2.0".into())));
    }
}
