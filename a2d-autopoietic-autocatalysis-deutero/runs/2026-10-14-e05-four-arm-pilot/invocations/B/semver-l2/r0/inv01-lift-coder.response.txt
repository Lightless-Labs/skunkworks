use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }

    let field = &s[start..*pos];
    if field.len() > 1 && field.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    field.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..i];
        if is_pre
            && ident.len() > 1
            && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(ident.to_string());

        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start > end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    for _ in 0..2 {
        match byte_at(s, pos) {
            Some(b'.') => pos += 1,
            None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        let value = parse_core_field(s, &mut pos)?;
        if _ == 0 {
            // Kept separate below to preserve left-to-right error ordering.
            let _ = value;
        }
    }
    // Reparse the already validated fields into their values, preserving the
    // exact same validation order.
    let mut p = 0;
    let major = parse_core_field(s, &mut p)?;
    p += 1;
    let minor = parse_core_field(s, &mut p)?;
    p += 1;
    let patch = parse_core_field(s, &mut p)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    match byte_at(s, p) {
        None => {}
        Some(b'-') => {
            let pre_start = p + 1;
            let build_marker = s.as_bytes()[pre_start..]
                .iter()
                .position(|&b| b == b'+')
                .map(|i| pre_start + i);
            let pre_end = build_marker.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(plus) = build_marker {
                build = parse_identifiers(s, plus + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, p + 1, s.len(), false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: p }),
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.pre.iter().zip(&b.pre) {
        let left_numeric = is_numeric_identifier(left);
        let right_numeric = is_numeric_identifier(right);
        let ordering = match (left_numeric, right_numeric) {
            (true, true) => compare_numeric_identifiers(left, right),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };
        if ordering != Ordering::Equal {
            return ordering;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && whitespace(bytes[i]) {
            i += 1;
        }
        if i == end {
            break;
        }
        let token_start = i;
        while i < end && !whitespace(bytes[i]) {
            i += 1;
        }
        out.push((token_start, &s[token_start..i]));
    }
    out
}

fn full_version(
    operand: &str,
    operand_start: usize,
) -> Result<Version, RangeError> {
    parse_version(operand).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    numeric_count: usize,
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn should_parse_full(operand: &str) -> bool {
    let fields: Vec<&str> = operand.split('.').collect();
    operand.contains('-')
        || operand.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f)))
}

fn parse_partial(operand: &str) -> Result<Partial, ()> {
    if operand.is_empty() {
        return Err(());
    }
    let fields: Vec<&str> = operand.split('.').collect();
    if fields.len() > 3 {
        return Err(());
    }

    let mut parsed = Vec::new();
    let mut saw_wildcard = false;
    let mut numeric_count = 0;
    for field in fields {
        if field.is_empty() {
            return Err(());
        }
        if is_wildcard(field) {
            saw_wildcard = true;
            parsed.push(None);
            continue;
        }
        if saw_wildcard
            || !field.bytes().all(|b| b.is_ascii_digit())
            || (field.len() > 1 && field.starts_with('0'))
        {
            return Err(());
        }
        let value = field.parse::<u64>().map_err(|_| ())?;
        numeric_count += 1;
        parsed.push(Some(value));
    }
    Ok(Partial {
        fields: parsed,
        numeric_count,
    })
}

fn padded_tuple(p: &Partial) -> (u64, u64, u64) {
    let mut values = [0u64; 3];
    for (i, field) in p.fields.iter().enumerate() {
        if let Some(value) = field {
            values[i] = *value;
        }
    }
    (values[0], values[1], values[2])
}

fn version_from_tuple(tuple: (u64, u64, u64), pre: Vec<String>, build: Vec<String>) -> Version {
    Version {
        major: tuple.0,
        minor: tuple.1,
        patch: tuple.2,
        pre,
        build,
    }
}

fn bump(tuple: (u64, u64, u64), field: usize) -> Option<(u64, u64, u64)> {
    match field {
        0 => Some((tuple.0.checked_add(1)?, 0, 0)),
        1 => Some((tuple.0, tuple.1.checked_add(1)?, 0)),
        _ => Some((tuple.0, tuple.1, tuple.2.checked_add(1)?)),
    }
}

fn push_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator { op, version });
}

fn syntax(at: usize) -> RangeError {
    RangeError::Syntax { at }
}

fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut i = 0usize;
    let mut saw_any_whitespace_token = false;
    let mut separators = Vec::new();

    while i < bytes.len() {
        if whitespace(bytes[i]) {
            i += 1;
            continue;
        }
        let start = i;
        while i < bytes.len() && !whitespace(bytes[i]) {
            i += 1;
        }
        saw_any_whitespace_token = true;
        if &s[start..i] == "||" {
            separators.push((start, i));
        }
    }
    if !saw_any_whitespace_token {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut cursor = 0usize;
    for (sep_start, sep_end) in separators.iter().copied() {
        let left_tokens = split_tokens(s, cursor, sep_start);
        if left_tokens.is_empty() {
            return Err(RangeError::Empty { at: sep_start });
        }
        arms.push(parse_arm(s, cursor, sep_start)?);
        cursor = sep_end;
        while cursor < s.len() && whitespace(bytes[cursor]) {
            cursor += 1;
        }
        if cursor == s.len() {
            return Err(RangeError::Empty { at: s.len() });
        }
        if s[cursor..].starts_with("||") {
            return Err(RangeError::Empty { at: cursor });
        }
    }
    if split_tokens(s, cursor, s.len()).is_empty() {
        if !separators.is_empty() {
            return Err(RangeError::Empty { at: s.len() });
        }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(s, cursor, s.len())?);
    Ok(Range { arms })
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let tokens = split_tokens(s, start, end);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: start });
    }

    if let Some((minus_index, (minus_start, _))) =
        tokens.iter().enumerate().find(|(_, (_, token))| *token == "-")
    {
        let _ = minus_index;
        if tokens.len() != 3 {
            return Err(syntax(*minus_start));
        }
        return parse_hyphen_arm(&tokens);
    }

    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    for (token_start, token) in tokens {
        parse_token(&mut arm, token, token_start)?;
    }
    Ok(arm)
}

fn parse_hyphen_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    if tokens.len() != 3 || tokens[1].1 != "-" {
        let at = tokens.iter().find(|(_, token)| *token == "-").map(|t| t.0).unwrap_or(tokens[0].0);
        return Err(syntax(at));
    }
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    let (lower_start, lower_text) = tokens[0];
    let lower = if should_parse_full(lower_text) {
        let v = full_version(lower_text, lower_start)?;
        Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            numeric_count: 3,
        }
    } else {
        parse_partial(lower_text).map_err(|_| syntax(lower_start))?
    };
    let (upper_start, upper_text) = tokens[2];
    let upper_full = should_parse_full(upper_text);
    let upper_version = if upper_full {
        Some(full_version(upper_text, upper_start)?)
    } else {
        None
    };
    let upper = if let Some(v) = &upper_version {
        Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            numeric_count: 3,
        }
    } else {
        parse_partial(upper_text).map_err(|_| syntax(upper_start))?
    };

    if lower.numeric_count == 0 {
        return Err(syntax(lower_start));
    }
    if upper.numeric_count == 0 {
        return Err(syntax(upper_start));
    }

    let lower_tuple = padded_tuple(&lower);
    push_cmp(
        &mut arm,
        Op::Ge,
        version_from_tuple(lower_tuple, Vec::new(), Vec::new()),
        false,
    );
    if let Some(v) = upper_version {
        push_cmp(&mut arm, Op::Le, v, true);
    } else {
        let tuple = padded_tuple(&upper);
        let field = if upper.numeric_count == 1 { 0 } else { 1 };
        let next = bump(tuple, field).ok_or_else(|| syntax(upper_start))?;
        push_cmp(
            &mut arm,
            Op::Lt,
            version_from_tuple(next, vec!["0".to_string()], Vec::new()),
            false,
        );
    }
    Ok(arm)
}

fn parse_token(arm: &mut Arm, token: &str, token_start: usize) -> Result<(), RangeError> {
    if token.is_empty() {
        return Err(syntax(token_start));
    }

    let first = token.as_bytes()[0];
    if matches!(first, b'<' | b'>' | b'=') {
        let (op, op_len) = if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with(">=") {
            (Op::Ge, 2)
        } else if token.starts_with('<') {
            (Op::Lt, 1)
        } else if token.starts_with('>') {
            (Op::Gt, 1)
        } else {
            (Op::Eq, 1)
        };
        let operand = &token[op_len..];
        let operand_start = token_start + op_len;
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(syntax(token_start));
        }
        let version = full_version(operand, operand_start)?;
        push_cmp(arm, op, version, true);
        return Ok(());
    }

    if first == b'^' || first == b'~' {
        let operand_start = token_start + 1;
        let operand = &token[1..];
        let partial = if should_parse_full(operand) {
            let v = full_version(operand, operand_start)?;
            Partial {
                fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
                numeric_count: 3,
            }
        } else {
            parse_partial(operand).map_err(|_| syntax(token_start))?
        };
        if partial.numeric_count == 0 {
            return Err(syntax(token_start));
        }
        let full = if should_parse_full(operand) {
            Some(full_version(operand, operand_start)?)
        } else {
            None
        };
        let lower_tuple = padded_tuple(&partial);
        let lower_version = full.unwrap_or_else(|| version_from_tuple(lower_tuple, Vec::new(), Vec::new()));
        push_cmp(arm, Op::Ge, lower_version, true);

        let field = if first == b'^' {
            if lower_tuple.0 > 0 || partial.numeric_count == 1 {
                0
            } else if lower_tuple.1 > 0 || partial.numeric_count == 2 {
                1
            } else {
                2
            }
        } else if partial.numeric_count == 1 {
            0
        } else {
            1
        };
        let next = bump(lower_tuple, field).ok_or_else(|| syntax(operand_start))?;
        push_cmp(
            arm,
            Op::Lt,
            version_from_tuple(next, vec!["0".to_string()], Vec::new()),
            false,
        );
        return Ok(());
    }

    if first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X') {
        if should_parse_full(token) {
            let version = full_version(token, token_start)?;
            push_cmp(arm, Op::Eq, version, true);
            return Ok(());
        }
        let partial = parse_partial(token).map_err(|_| syntax(token_start))?;
        match partial.numeric_count {
            0 => {}
            1 | 2 => {
                let tuple = padded_tuple(&partial);
                push_cmp(
                    arm,
                    Op::Ge,
                    version_from_tuple(tuple, Vec::new(), Vec::new()),
                    false,
                );
                let next = bump(tuple, partial.numeric_count - 1)
                    .ok_or_else(|| syntax(token_start))?;
                push_cmp(
                    arm,
                    Op::Lt,
                    version_from_tuple(next, vec!["0".to_string()], Vec::new()),
                    false,
                );
            }
            _ => {
                let version = full_version(token, token_start)?;
                push_cmp(arm, Op::Eq, version, true);
            }
        }
        return Ok(());
    }

    Err(syntax(token_start))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let tuple = (v.major, v.minor, v.patch);
        if !v.pre.is_empty() && !arm.anchors.iter().any(|anchor| *anchor == tuple) {
            return false;
        }
        arm.comparators.iter().all(|cmp| {
            let ordering = compare_versions(v, &cmp.version);
            match cmp.op {
                Op::Eq => ordering == Ordering::Equal,
                Op::Lt => ordering == Ordering::Less,
                Op::Le => ordering != Ordering::Greater,
                Op::Gt => ordering == Ordering::Greater,
                Op::Ge => ordering != Ordering::Less,
            }
        })
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(current) => {
                let ordering = compare_versions(version, &parsed[current]);
                if (want_max && ordering == Ordering::Greater)
                    || (!want_max && ordering == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }
    Ok(best.map(|index| candidates[index].clone()))
}
