use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &b in &bytes[start..*pos] {
        value = value.checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn is_identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut ident_start = start;
    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            if !is_identifier_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        if pre {
            let all_digits = bytes[ident_start..i].iter().all(u8::is_ascii_digit);
            if all_digits && i - ident_start > 1 && bytes[ident_start] == b'0' {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
        }
        out.push(s[ident_start..i].to_string());
        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start > end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            if matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let n = parse_core_number(s, &mut pos)?;
        if expected == 0 {
            let minor = n;
            if pos >= s.len() {
                return Err(VersionError::MissingCore { at: pos });
            }
            if s.as_bytes()[pos] != b'.' {
                if matches!(s.as_bytes()[pos], b'-' | b'+') {
                    return Err(VersionError::MissingCore { at: pos });
                }
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core_number(s, &mut pos)?;
            return parse_version_tail(s, pos, major, minor, patch);
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..].iter().position(|&b| b == b'+')
            .map(|i| pre_start + i).unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, plus, true)?;
        pos = plus;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }
    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn cmp_numeric_ident(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric_identifier(x);
                    let yn = is_numeric_identifier(y);
                    let ord = match (xn, yn) {
                        (true, true) => cmp_numeric_ident(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    terms: Vec<Term>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Term {
    op: Op,
    version: Version,
    anchor: bool,
}

fn cmp_op(op: &Op, v: &Version, target: &Version) -> bool {
    let c = compare_versions(v, target);
    match op {
        Op::Eq => c == Ordering::Equal,
        Op::Lt => c == Ordering::Less,
        Op::Le => c != Ordering::Greater,
        Op::Gt => c == Ordering::Greater,
        Op::Ge => c != Ordering::Less,
    }
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields { nums: Vec<u64>, k: usize },
}

fn operand_is_full(s: &str) -> bool {
    s.contains('-') || s.contains('+')
        || (s.split('.').count() >= 3
            && !s.split('.').any(|x| matches!(x, "*" | "x" | "X")))
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if operand_is_full(s) {
        return parse_full_operand(s, at).map(Partial::Full);
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial::Fields { k: nums.len(), nums })
}

fn padded(nums: &[u64]) -> Version {
    Version {
        major: *nums.first().unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_next(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut u = v.clone();
    u.pre = vec!["0".to_string()];
    u.build.clear();
    match field {
        0 => {
            u.major = u.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            u.minor = 0;
            u.patch = 0;
        }
        1 => {
            u.minor = u.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            u.patch = 0;
        }
        _ => {
            u.patch = u.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
        }
    }
    Ok(u)
}

fn add_term(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
    arm.terms.push(Term { op, version, anchor });
}

fn parse_bare_or_expanded(
    arm: &mut Arm,
    token: &str,
    at: usize,
    mode: char,
) -> Result<(), RangeError> {
    let p = parse_partial(token, at)?;
    match p {
        Partial::Full(v) if mode == 'b' => add_term(arm, Op::Eq, v, true),
        Partial::Full(v) => {
            let (field, lower) = if mode == '^' {
                let field = if v.major > 0 { 0 } else if v.minor > 0 { 1 } else { 2 };
                (field, v.clone())
            } else {
                (1, v.clone())
            };
            add_term(arm, Op::Ge, lower, true);
            let upper = upper_next(&v, field, at)?;
            add_term(arm, Op::Lt, upper, false);
        }
        Partial::Fields { nums, k } => {
            if mode != 'b' && k == 0 {
                return Err(RangeError::Syntax { at });
            }
            if mode == 'b' {
                if k == 0 {
                    return Ok(());
                }
                let lower = padded(&nums);
                add_term(arm, Op::Ge, lower.clone(), false);
                if k < 3 {
                    let upper = upper_next(&lower, k - 1, at)?;
                    add_term(arm, Op::Lt, upper, false);
                }
            } else {
                let lower = padded(&nums);
                add_term(arm, Op::Ge, lower.clone(), false);
                let field = if mode == '^' {
                    if lower.major > 0 || k == 1 { 0 }
                    else if lower.minor > 0 || k == 2 { 1 }
                    else { 2 }
                } else if k == 1 {
                    0
                } else {
                    1
                };
                let upper = upper_next(&lower, field, at)?;
                add_term(arm, Op::Lt, upper, false);
            }
        }
    }
    Ok(())
}

fn split_range(s: &str) -> Vec<(usize, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') { i += 1; }
        if i == b.len() { break; }
        let start = i;
        while i < b.len() && !matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') { i += 1; }
        out.push((start, i));
    }
    out
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = split_range(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms: Vec<Vec<(usize, usize)>> = vec![Vec::new()];
    let mut i = 0;
    while i < tokens.len() {
        let (st, en) = tokens[i];
        if &s[st..en] == "||" {
            if arms.last().unwrap().is_empty() {
                return Err(RangeError::Empty { at: st });
            }
            arms.push(Vec::new());
        } else {
            arms.last_mut().unwrap().push((st, en));
        }
        i += 1;
    }
    if arms.last().unwrap().is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }

    let mut parsed_arms = Vec::new();
    for ts in arms {
        if let Some((idx, &(st, _))) = ts.iter().enumerate().find(|(_, &(a, b))| &s[a..b] == "-") {
            if ts.len() != 3 || idx != 1 {
                return Err(RangeError::Syntax { at: st });
            }
            let (ls, le) = ts[0];
            let (us, ue) = ts[2];
            let lp = parse_partial(&s[ls..le], ls)?;
            let up = parse_partial(&s[us..ue], us)?;
            let (ln, lk) = match lp {
                Partial::Fields { nums, k } => (nums, k),
                Partial::Full(v) => (vec![v.major, v.minor, v.patch], 3),
            };
            let (un, uk, full_upper) = match up {
                Partial::Fields { nums, k } => (nums, k, None),
                Partial::Full(v) => (vec![v.major, v.minor, v.patch], 3, Some(v)),
            };
            if lk == 0 { return Err(RangeError::Syntax { at: ls }); }
            if uk == 0 { return Err(RangeError::Syntax { at: us }); }
            let mut arm = Arm { terms: Vec::new(), anchors: Vec::new() };
            let lower = padded(&ln);
            add_term(&mut arm, Op::Ge, lower, false);
            let upper = if let Some(v) = full_upper {
                add_term(&mut arm, Op::Le, v, true);
                None
            } else if uk < 3 {
                Some(upper_next(&padded(&un), uk - 1, us)?)
            } else {
                Some(padded(&un))
            };
            if let Some(v) = upper {
                add_term(&mut arm, if uk < 3 { Op::Lt } else { Op::Le }, v, false);
            }
            parsed_arms.push(arm);
            continue;
        }

        let mut arm = Arm { terms: Vec::new(), anchors: Vec::new() };
        let mut j = 0;
        while j < ts.len() {
            let (st, en) = ts[j];
            let token = &s[st..en];
            let bytes = token.as_bytes();
            if token == "-" { return Err(RangeError::Syntax { at: st }); }
            if token.starts_with(',') || token.contains(',') || token.contains('(')
                || token.contains(')') || token == "&"
            {
                return Err(RangeError::Syntax { at: st });
            }
            if token.starts_with('^') || token.starts_with('~') {
                let mode = bytes[0] as char;
                let operand = &token[1..];
                if operand.is_empty() {
                    return Err(RangeError::Syntax { at: st });
                }
                parse_bare_or_expanded(&mut arm, operand, st + 1, mode)?;
            } else if matches!(bytes[0], b'<' | b'>' | b'=') {
                let (op, n) = if token.starts_with("<=") { (Op::Le, 2) }
                    else if token.starts_with(">=") { (Op::Ge, 2) }
                    else if token.starts_with('<') { (Op::Lt, 1) }
                    else if token.starts_with('>') { (Op::Gt, 1) }
                    else { (Op::Eq, 1) };
                let operand = &token[n..];
                if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                    return Err(RangeError::Syntax { at: st });
                }
                let v = parse_full_operand(operand, st + n)?;
                add_term(&mut arm, op, v, true);
            } else if bytes[0].is_ascii_digit() || matches!(bytes[0], b'*' | b'x' | b'X') {
                parse_bare_or_expanded(&mut arm, token, st, 'b')?;
            } else {
                return Err(RangeError::Syntax { at: st });
            }
            j += 1;
        }
        parsed_arms.push(arm);
    }
    Ok(Range { arms: parsed_arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.iter().any(|t| *t == (v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.terms.iter().all(|term| cmp_op(&term.op, v, &term.version))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(prev) = best {
            let ord = compare_versions(v, &parsed[prev]);
            if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
