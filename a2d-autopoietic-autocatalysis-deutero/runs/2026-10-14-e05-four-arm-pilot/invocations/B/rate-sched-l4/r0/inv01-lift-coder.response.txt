#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Runtime {
    state: State,
    ready_at: u64,
    attempt: u8,
    remaining: u64,
    started_at: u64,
}

pub struct Pool {
    jobs: Vec<Job>,
    states: Vec<Runtime>,
    dependencies: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_deref().is_some_and(|s| !valid_token(s, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut deps = Vec::new();
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(spelling) {
                    Some(&index) => deps.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(deps);
        }

        fn visit(
            i: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[i] == 1 {
                return false;
            }
            if marks[i] == 2 {
                return true;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[i] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|job| Runtime {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                ready_at: job.release_at,
                attempt: 0,
                remaining: job.duration,
                started_at: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            dependencies,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.jobs
            .iter()
            .position(|job| job.id == id)
            .map(|i| self.states[i].state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.has_run { self.horizon } else { 0 };
        let first_phase = !self.has_run;

        loop {
            let mut completions: Vec<usize> = self
                .states
                .iter()
                .enumerate()
                .filter_map(|(i, r)| match r.state {
                    State::Running { finish_at, .. } if finish_at == time => Some(i),
                    _ => None,
                })
                .collect();
            completions.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));

            for i in completions {
                let attempt = self.states[i].attempt;
                let ok = attempt > self.jobs[i].failures;
                if ok {
                    self.states[i].state = State::Succeeded;
                    lines.push(format!("{time} finish {} {attempt} ok", self.jobs[i].id));
                } else {
                    lines.push(format!("{time} finish {} {attempt} fail", self.jobs[i].id));
                    if attempt >= self.jobs[i].max_attempts {
                        self.states[i].state = State::Failed;
                    } else {
                        let shift = u32::from(attempt.saturating_sub(1));
                        let delay = self.retry_base.checked_shl(shift).unwrap_or(u64::MAX)
                            .min(self.retry_cap);
                        let ready = time.saturating_add(delay);
                        self.states[i].ready_at = ready;
                        self.states[i].remaining = self.jobs[i].duration;
                        self.states[i].state = State::Waiting {
                            ready_at: ready,
                            attempt,
                        };
                        lines.push(format!("{time} retry {} {attempt} {ready}", self.jobs[i].id));
                    }
                }
            }

            let mut newly_blocked = Vec::new();
            loop {
                let mut changed = false;
                for i in 0..self.jobs.len() {
                    if !matches!(self.states[i].state, State::Waiting { .. }) {
                        continue;
                    }
                    if let Some(&dep) = self.dependencies[i].iter().find(|&&d| {
                        matches!(self.states[d].state, State::Failed | State::Blocked)
                    }) {
                        self.states[i].state = State::Blocked;
                        newly_blocked.push((i, dep));
                        changed = true;
                    }
                }
                if !changed {
                    break;
                }
            }
            newly_blocked.sort_by(|(a, _), (b, _)| self.jobs[*a].id.cmp(&self.jobs[*b].id));
            for (i, dep) in newly_blocked {
                lines.push(format!("{time} skip {} {}", self.jobs[i].id, self.jobs[dep].id));
            }

            let should_schedule = first_phase || time > self.horizon || time == self.horizon;
            if should_schedule {
                self.schedule(time, &mut lines);
            }

            if time == until {
                break;
            }

            let next = self
                .states
                .iter()
                .filter_map(|r| match r.state {
                    State::Waiting { ready_at, .. } if ready_at > time => Some(ready_at),
                    State::Running { finish_at, .. } if finish_at > time => Some(finish_at),
                    _ => None,
                })
                .min();

            match next {
                Some(t) if t <= until => time = t,
                _ => break,
            }
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn schedule(&mut self, time: u64, lines: &mut Vec<String>) {
        loop {
            let running: Vec<usize> = self
                .states
                .iter()
                .enumerate()
                .filter_map(|(i, r)| matches!(r.state, State::Running { .. }).then_some(i))
                .collect();

            let mut candidates: Vec<usize> = (0..self.jobs.len())
                .filter(|&i| {
                    matches!(
                        self.states[i].state,
                        State::Waiting { ready_at, .. } if ready_at <= time
                    ) && self.dependencies[i]
                        .iter()
                        .all(|&d| matches!(self.states[d].state, State::Succeeded))
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut selected = None;
            for candidate in candidates {
                let tag = self.jobs[candidate].tag.as_deref();
                let tag_busy = tag.is_some_and(|t| {
                    running
                        .iter()
                        .any(|&r| self.jobs[r].tag.as_deref() == Some(t))
                });
                if running.len() < self.capacity && !tag_busy {
                    selected = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&v| {
                        self.jobs[candidate].priority
                            >= self.jobs[v].priority.saturating_add(2)
                    })
                    .filter(|&v| {
                        let remaining = running.len() - 1;
                        let slot_ok = remaining < self.capacity;
                        let tag_ok = tag.is_none()
                            || !running.iter().any(|&r| {
                                r != v && self.jobs[r].tag.as_deref() == tag
                            });
                        slot_ok && tag_ok
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .priority
                        .cmp(&self.jobs[b].priority)
                        .then_with(|| self.states[b].started_at.cmp(&self.states[a].started_at))
                        .then_with(|| self.jobs[b].id.cmp(&self.jobs[a].id))
                });

                if let Some(victim) = victims.first().copied() {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(v) = victim {
                let finish_at = match self.states[v].state {
                    State::Running { finish_at, .. } => finish_at,
                    _ => unreachable!(),
                };
                let remaining = finish_at.saturating_sub(time);
                self.states[v].remaining = remaining;
                self.states[v].ready_at = time;
                self.states[v].state = State::Waiting {
                    ready_at: time,
                    attempt: self.states[v].attempt,
                };
                lines.push(format!(
                    "{time} preempt {} by {}",
                    self.jobs[v].id, self.jobs[candidate].id
                ));
            }

            let resuming = self.states[candidate].attempt > 0
                && self.states[candidate].remaining != self.jobs[candidate].duration;
            let attempt = if resuming {
                self.states[candidate].attempt
            } else {
                self.states[candidate].attempt.saturating_add(1)
            };
            let remaining = if resuming {
                self.states[candidate].remaining
            } else {
                self.jobs[candidate].duration
            };
            let finish_at = time.saturating_add(remaining);
            self.states[candidate].attempt = attempt;
            self.states[candidate].started_at = time;
            self.states[candidate].state = State::Running {
                attempt,
                finish_at,
            };
            lines.push(format!(
                "{time} start {} {attempt} {remaining}",
                self.jobs[candidate].id
            ));
        }
    }

    fn effective_priority(&self, i: usize, time: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[i].priority;
        }
        let waited = time.saturating_sub(self.states[i].ready_at);
        let age = waited / self.aging_ms;
        self.jobs[i]
            .priority
            .saturating_add(age.min(9) as u8)
            .min(9)
    }
}

fn valid_token(s: &str, min_len: usize, max_len: usize) -> bool {
    (min_len..=max_len).contains(&s.len())
        && s.bytes()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}
