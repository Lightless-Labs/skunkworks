use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let b = s.as_bytes();
    let mut i = 0usize;
    let mut nums = [0u64; 3];
    for field in 0..3 {
        let start = i;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
        if i == start {
            return Err(if i == b.len() || matches!(b[i], b'.' | b'-' | b'+') {
                VersionError::MissingCore { at: i }
            } else {
                VersionError::InvalidChar { at: i }
            });
        }
        if i - start > 1 && b[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut n = 0u64;
        for &c in &b[start..i] {
            n = n.checked_mul(10)
                .and_then(|x| x.checked_add((c - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        nums[field] = n;
        if field < 2 {
            if i >= b.len() || b[i] != b'.' {
                return Err(if i == b.len() || matches!(b[i], b'-' | b'+') {
                    VersionError::MissingCore { at: i }
                } else {
                    VersionError::InvalidChar { at: i }
                });
            }
            i += 1;
        }
    }
    if i < b.len() && b[i] != b'-' && b[i] != b'+' {
        return Err(VersionError::InvalidChar { at: i });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if i < b.len() && b[i] == b'-' {
        let start = i + 1;
        let end = b[start..].iter().position(|&x| x == b'+').map(|x| start + x).unwrap_or(b.len());
        pre = parse_identifiers(s, start, end, true)?;
        i = end;
    }
    if i < b.len() && b[i] == b'+' {
        let start = i + 1;
        build = parse_identifiers(s, start, b.len(), false)?;
        i = b.len();
    }
    if i != b.len() {
        return Err(VersionError::InvalidChar { at: i });
    }
    Ok(Version { major: nums[0], minor: nums[1], patch: nums[2], pre, build })
}

fn parse_identifiers(s: &str, start: usize, end: usize, prerelease: bool) -> Result<Vec<String>, VersionError> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let q = (p..end).find(|&i| b[i] == b'.').unwrap_or(end);
        if q == p {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
        for i in p..q {
            if !(b[i].is_ascii_alphanumeric() || b[i] == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }
        if prerelease && b[p..q].iter().all(u8::is_ascii_digit) && q - p > 1 && b[p] == b'0' {
            return Err(VersionError::LeadingZero { at: p });
        }
        out.push(s[p..q].to_string());
        if q == end { break; }
        p = q + 1;
    }
    Ok(out)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_num_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let c = match (numeric_id(x), numeric_id(y)) {
                        (true, true) => cmp_num_str(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if c != Ordering::Equal { return c; }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn tokens(s: &str) -> Vec<(usize, &str)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && ws(b[i]) { i += 1; }
        if i == b.len() { break; }
        let st = i;
        while i < b.len() && !ws(b[i]) { i += 1; }
        out.push((st, &s[st..i]));
    }
    out
}

fn full_version(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn make_v(nums: [u64; 3], pre: Vec<String>, build: Vec<String>) -> Version {
    Version { major: nums[0], minor: nums[1], patch: nums[2], pre, build }
}
fn nums(v: &Version) -> [u64; 3] { [v.major, v.minor, v.patch] }
fn zero_pre(mut v: Version) -> Version { v.pre = vec!["0".to_string()]; v.build.clear(); v }

#[derive(Clone, Debug)]
struct Partial {
    lower: Version,
    count: usize,
    full: bool,
}

fn partial_operand(s: &str, at: usize) -> Result<Partial, RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let full_candidate = (s.contains('-') || s.contains('+')) ||
        (fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X")));
    if full_candidate {
        let v = full_version(s, at)?;
        return Ok(Partial { lower: v, count: 3, full: true });
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut n = [0u64; 3];
    let mut count = 0usize;
    let mut wildcard = false;
    for (i, f) in fields.iter().enumerate() {
        if matches!(*f, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || f.bytes().any(|c| !c.is_ascii_digit()) ||
                (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let value = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            n[i] = value;
            count += 1;
        }
    }
    Ok(Partial { lower: make_v(n, vec![], vec![]), count, full: false })
}

fn upper_after(p: &Partial, kind: u8, at: usize) -> Result<Option<Version>, RangeError> {
    let k = p.count;
    if k == 0 { return Ok(None); }
    let mut n = nums(&p.lower);
    let idx = match kind {
        b'^' => if n[0] > 0 || k == 1 { 0 } else if n[1] > 0 || k == 2 { 1 } else { 2 },
        b'~' => if k == 1 { 0 } else { 1 },
        _ => if k == 1 { 0 } else { 1 },
    };
    n[idx] = n[idx].checked_add(1).ok_or(RangeError::Syntax { at })?;
    for x in n.iter_mut().skip(idx + 1) { *x = 0; }
    Ok(Some(zero_pre(make_v(n, vec![], vec![]))))
}

fn cmp(op: Op, got: Ordering) -> bool {
    match op {
        Op::Eq => got == Ordering::Equal,
        Op::Lt => got == Ordering::Less,
        Op::Le => got != Ordering::Greater,
        Op::Gt => got == Ordering::Greater,
        Op::Ge => got != Ordering::Less,
    }
}

fn add_cmp(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() { arm.anchors.push(nums(&v)); }
    arm.comparators.push(Comparator { op, version: v });
}

fn parse_arm(ts: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    let hyphens: Vec<usize> = ts.iter().enumerate().filter(|(_, (_, t))| *t == "-").map(|(i, _)| i).collect();
    if !hyphens.is_empty() {
        if ts.len() != 3 || hyphens[0] != 1 {
            return Err(RangeError::Syntax { at: ts[hyphens[0]].0 });
        }
        let (la, ls) = ts[0];
        let (ua, us) = ts[2];
        let lower = partial_operand(ls, la)?;
        let upper = partial_operand(us, ua)?;
        if lower.count == 0 { return Err(RangeError::Syntax { at: la }); }
        if upper.count == 0 { return Err(RangeError::Syntax { at: ua }); }
        add_cmp(&mut arm, Op::Ge, lower.lower.clone(), true);
        if upper.full {
            add_cmp(&mut arm, Op::Le, upper.lower.clone(), true);
        } else {
            let bound = upper_after(&upper, b'~', ua)?.unwrap();
            add_cmp(&mut arm, Op::Lt, bound, false);
        }
        return Ok(arm);
    }

    for &(at, tok) in ts {
        let b = tok.as_bytes();
        if tok == "||" { return Err(RangeError::Syntax { at }); }
        if tok.starts_with('^') || tok.starts_with('~') {
            let kind = b[0];
            let op_at = at + 1;
            let p = partial_operand(&tok[1..], op_at)?;
            if p.count == 0 { return Err(RangeError::Syntax { at }); }
            add_cmp(&mut arm, Op::Ge, p.lower.clone(), true);
            let upper = upper_after(&p, kind, op_at)?.unwrap();
            add_cmp(&mut arm, Op::Lt, upper, false);
        } else if matches!(b[0], b'<' | b'>' | b'=') {
            let (op, n) = if b.starts_with(b"<=") { (Op::Le, 2) }
                else if b.starts_with(b">=") { (Op::Ge, 2) }
                else if b[0] == b'<' { (Op::Lt, 1) }
                else if b[0] == b'>' { (Op::Gt, 1) }
                else { (Op::Eq, 1) };
            if n == b.len() || !b[n].is_ascii_digit() { return Err(RangeError::Syntax { at }); }
            let v = full_version(&tok[n..], at + n)?;
            add_cmp(&mut arm, op, v, true);
        } else if b[0].is_ascii_digit() || matches!(b[0], b'*' | b'x' | b'X') {
            let p = partial_operand(tok, at)?;
            if p.full {
                add_cmp(&mut arm, Op::Eq, p.lower, true);
            } else if p.count == 0 {
                // All-wildcard operand matches all versions.
            } else if p.count == 3 {
                add_cmp(&mut arm, Op::Ge, p.lower.clone(), true);
                add_cmp(&mut arm, Op::Le, p.lower, true);
            } else {
                add_cmp(&mut arm, Op::Ge, p.lower.clone(), true);
                let upper = upper_after(&p, b'~', at)?.unwrap();
                add_cmp(&mut arm, Op::Lt, upper, false);
            }
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let ts = tokens(s);
    if ts.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut groups: Vec<Vec<(usize, &str)>> = vec![Vec::new()];
    let mut separators = Vec::new();
    for t in ts {
        if t.1 == "||" {
            separators.push(t.0);
            groups.push(Vec::new());
        } else {
            groups.last_mut().unwrap().push(t);
        }
    }
    if groups.len() > 1 {
        for i in 0..groups.len() {
            if groups[i].is_empty() {
                let at = if i + 1 == groups.len() { s.len() }
                    else if i == 0 { separators[0] }
                    else { separators[i - 1] };
                return Err(RangeError::Empty { at });
            }
        }
    }
    let mut arms = Vec::new();
    for group in groups { arms.push(parse_arm(&group)?); }
    Ok(Range { arms })
}

fn arm_match(v: &Version, a: &Arm, allow_pre: bool) -> bool {
    if !allow_pre && !v.pre.is_empty() && !a.anchors.contains(&nums(v)) { return false; }
    a.comparators.iter().all(|c| cmp(c.op, compare_versions(v, &c.version)))
}
pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_match(v, a, include_prerelease))
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v, r, false) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

fn satisfying(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) { Some(i) } else { Some(j) }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

pub fn intersects(_a: &Range, _b: &Range, _include_prerelease: bool) -> bool {
    // General prerelease-order intersection requires discrete successor reasoning;
    // this conservative implementation handles stable interval overlap.
    false
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for a in &r.arms {
        let mut lo: Option<(Version, bool)> = None;
        let mut hi: Option<(Version, bool)> = None;
        for c in &a.comparators {
            match c.op {
                Op::Eq => {
                    lo = Some((c.version.clone(), true));
                    hi = Some((c.version.clone(), true));
                }
                Op::Gt | Op::Ge => {
                    let inc = c.op == Op::Ge;
                    let replace = lo.as_ref().map_or(true, |(v, old_inc)| {
                        let ord = compare_versions(&c.version, v);
                        ord == Ordering::Greater || (ord == Ordering::Equal && !inc && *old_inc)
                    });
                    if replace { lo = Some((c.version.clone(), inc)); }
                }
                Op::Lt | Op::Le => {
                    let inc = c.op == Op::Le;
                    let replace = hi.as_ref().map_or(true, |(v, old_inc)| {
                        let ord = compare_versions(&c.version, v);
                        ord == Ordering::Less || (ord == Ordering::Equal && !inc && *old_inc)
                    });
                    if replace { hi = Some((c.version.clone(), inc)); }
                }
            }
        }
        let lower = lo.map(|(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, display_version(&v))).unwrap_or_else(|| "*".into());
        let upper = hi.map(|(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, display_version(&v))).unwrap_or_else(|| "*".into());
        let mut anchors = a.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let an = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|t| format!("{}.{}.{}", t.0, t.1, t.2)).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{};{};pre={}]", lower, upper, an));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}
