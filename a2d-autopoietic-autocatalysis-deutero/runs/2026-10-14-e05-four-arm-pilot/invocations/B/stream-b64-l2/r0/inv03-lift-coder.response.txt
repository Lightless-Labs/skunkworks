#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8]) -> String {
    let mut out = String::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut result = encoder.push(input).expect("new encoder is open");
    result.push_str(&encoder.finish().expect("encoder is open"));
    result
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    buffered: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            buffered: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.buffered.len() + input.len());
        data.extend_from_slice(&self.buffered);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut output = String::with_capacity(complete_len / 3 * 4);
        for group in data[..complete_len].chunks_exact(3) {
            output.push_str(&encode_group(group));
        }

        self.buffered.clear();
        self.buffered.extend_from_slice(&data[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        if self.buffered.is_empty() {
            Ok(String::new())
        } else {
            let output = encode_group(&self.buffered);
            self.buffered.clear();
            Ok(output)
        }
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

pub struct Decoder {
    group: Vec<Symbol>,
    pending_padding: Option<usize>,
    terminal_padding: bool,
    next_offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            pending_padding: None,
            terminal_padding: false,
            next_offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) {
        let values: Vec<u8> = self
            .group
            .iter()
            .map(|symbol| digit(symbol.byte).unwrap_or(0))
            .collect();

        match self.group.len() {
            4 if self.group[2].byte == b'=' => {
                output.push((values[0] << 2) | (values[1] >> 4));
                self.terminal_padding = true;
            }
            4 if self.group[3].byte == b'=' => {
                output.push((values[0] << 2) | (values[1] >> 4));
                output.push((values[1] << 4) | (values[2] >> 2));
                self.terminal_padding = true;
            }
            4 => {
                output.push((values[0] << 2) | (values[1] >> 4));
                output.push((values[1] << 4) | (values[2] >> 2));
                output.push((values[2] << 6) | values[3]);
            }
            _ => {}
        }

        self.group.clear();
        self.pending_padding = None;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset });
            }

            if let Some(padding_offset) = self.pending_padding {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset });
                }
                self.group.push(Symbol { byte, offset });
                let _ = padding_offset;
                self.emit_group(&mut output);
                continue;
            }

            let position = self.group.len();
            if byte == b'=' {
                match position {
                    0 | 1 => return self.fail(Error::BadPadding { offset }),
                    2 => {
                        self.group.push(Symbol { byte, offset });
                        self.pending_padding = Some(offset);
                    }
                    3 => {
                        self.group.push(Symbol { byte, offset });
                        self.emit_group(&mut output);
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            if digit(byte).is_none() {
                return self.fail(Error::InvalidByte { offset, byte });
            }

            self.group.push(Symbol { byte, offset });
            if self.group.len() == 4 {
                self.emit_group(&mut output);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if let Some(first) = self.group.first() {
            return self.fail(Error::Truncated {
                offset: first.offset,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }
}
