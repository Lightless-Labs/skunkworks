#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|n| n as u8)
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 2 && index < input.len() {
                self.tail[self.tail_len] = input[index];
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 2 && index < input.len() {
                encode_triplet(
                    &mut output,
                    self.tail[0],
                    self.tail[1],
                    input[index],
                );
                self.tail_len = 0;
                index += 1;
            }
        }

        while index + 3 <= input.len() {
            encode_triplet(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        while index < input.len() {
            self.tail[self.tail_len] = input[index];
            self.tail_len += 1;
            index += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        self.finished = true;
        let mut output = String::new();

        if self.tail_len == 1 {
            let a = self.tail[0];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push_str("==");
        } else if self.tail_len == 2 {
            let a = self.tail[0];
            let b = self.tail[1];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }

        self.tail_len = 0;
        Ok(output)
    }
}

fn encode_triplet(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub struct Decoder {
    values: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    next_offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            values: [0; 4],
            offsets: [0; 4],
            len: 0,
            pending_second_padding: None,
            terminal_padding: false,
            next_offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset = self.next_offset.saturating_add(1);

            let error = if self.terminal_padding {
                Some(Error::BadPadding { offset })
            } else if self.pending_second_padding.is_some() && byte != b'=' {
                Some(Error::BadPadding { offset })
            } else if byte == b'=' {
                if self.len <= 1 {
                    Some(Error::BadPadding { offset })
                } else if self.len == 2 {
                    self.pending_second_padding = Some(offset);
                    None
                } else {
                    self.complete_group(&mut output, true);
                    self.terminal_padding = true;
                    None
                }
            } else if let Some(value) = digit(byte) {
                self.values[self.len] = value;
                self.offsets[self.len] = offset;
                self.len += 1;
                if self.len == 4 {
                    self.complete_group(&mut output, false);
                }
                None
            } else {
                Some(Error::InvalidByte { offset, byte })
            };

            if let Some(error) = error {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let result = if let Some(offset) = self.pending_second_padding {
            Err(Error::BadPadding { offset })
        } else if self.len != 0 {
            Err(Error::Truncated {
                offset: self.offsets[0],
            })
        } else {
            Ok(Vec::new())
        };

        match result {
            Ok(output) => {
                self.finished = true;
                Ok(output)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }

    fn complete_group(&mut self, output: &mut Vec<u8>, one_byte: bool) {
        let a = self.values[0];
        let b = self.values[1];
        output.push((a << 2) | (b >> 4));

        if !one_byte {
            let c = self.values[2];
            output.push((b << 4) | (c >> 2));

            if self.len == 4 {
                let d = self.values[3];
                output.push((c << 6) | d);
            }
        }

        self.len = 0;
        self.pending_second_padding = None;
    }
}
