#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|n| n as u8)
}

fn encode_group(bytes: &[u8], out: &mut String) {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::new();
    for chunk in input.chunks_exact(3) {
        encode_group(chunk, &mut out);
    }

    let tail = input.chunks_exact(3).remainder();
    if !tail.is_empty() {
        encode_group(tail, &mut out);
    }
    out
}

#[derive(Clone, Copy)]
enum DecodeState {
    Open,
    PendingThird { first: u8, second: u8, offset: usize },
    Terminal,
}

pub struct Encoder {
    buffered: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            buffered: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.buffered.len() + input.len());
        data.extend_from_slice(&self.buffered);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut out = String::new();
        for chunk in data[..complete_len].chunks_exact(3) {
            encode_group(chunk, &mut out);
        }
        self.buffered.clear();
        self.buffered.extend_from_slice(&data[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        if !self.buffered.is_empty() {
            encode_group(&self.buffered, &mut out);
            self.buffered.clear();
        }
        self.finished = true;
        Ok(out)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    state: DecodeState,
    group: [u8; 4],
    len: usize,
    offset: usize,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            state: DecodeState::Open,
            group: [0; 4],
            len: 0,
            offset: 0,
            finished: false,
            error: None,
        }
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.error = Some(error.clone());
        Err(error)
    }

    fn complete_group(&mut self, out: &mut Vec<u8>) {
        let a = self.group[0];
        let b = self.group[1];
        let c = self.group[2];
        let d = self.group[3];

        out.push((a << 2) | (b >> 4));
        if c != 64 {
            out.push((b << 4) | (c >> 2));
        }
        if d != 64 {
            out.push((c << 6) | d);
        }

        if c == 64 || d == 64 {
            self.state = DecodeState::Terminal;
        } else {
            self.state = DecodeState::Open;
        }
        self.len = 0;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            match self.state {
                DecodeState::Terminal => {
                    return self.fail(Error::BadPadding {
                        offset: current_offset,
                    });
                }
                DecodeState::PendingThird {
                    first,
                    second,
                    offset: _,
                } => {
                    if byte != b'=' {
                        return self.fail(Error::BadPadding {
                            offset: current_offset,
                        });
                    }
                    self.group = [first, second, 64, 64];
                    self.complete_group(&mut out);
                }
                DecodeState::Open => {
                    let position = self.len;
                    if position <= 1 && byte == b'=' {
                        return self.fail(Error::BadPadding {
                            offset: current_offset,
                        });
                    }
                    if position == 2 && byte == b'=' {
                        self.state = DecodeState::PendingThird {
                            first: self.group[0],
                            second: self.group[1],
                            offset: current_offset,
                        };
                        continue;
                    }
                    if position == 3 && byte == b'=' {
                        self.group[3] = 64;
                        self.complete_group(&mut out);
                        continue;
                    }

                    let Some(value) = digit(byte) else {
                        return self.fail(Error::InvalidByte {
                            offset: current_offset,
                            byte,
                        });
                    };
                    self.group[position] = value;
                    self.len += 1;
                    if self.len == 4 {
                        self.complete_group(&mut out);
                    }
                }
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.state {
            DecodeState::PendingThird { offset, .. } => {
                self.fail(Error::BadPadding { offset })
            }
            _ if self.len != 0 => {
                let first_offset = self.offset - self.len;
                self.fail(Error::Truncated {
                    offset: first_offset,
                })
            }
            _ => {
                self.finished = true;
                Ok(Vec::new())
            }
        }
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
