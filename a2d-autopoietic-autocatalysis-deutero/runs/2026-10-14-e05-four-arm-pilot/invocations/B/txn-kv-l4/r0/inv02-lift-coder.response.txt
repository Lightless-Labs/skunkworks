use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Entry {
    value: Option<String>,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    entry: Entry,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Entry>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit_time(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn validate_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn live(entry: &Entry, now: u64) -> Option<&String> {
        if entry.value.is_some() && entry.deadline.map_or(true, |d| d > now) {
            entry.value.as_ref()
        } else {
            None
        }
    }

    fn selected_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return Some(entry.clone());
            }
            self.data.get(key)?.iter()
                .rev()
                .find(|v| v.revision <= transaction.begin_revision)
                .map(|v| v.entry.clone())
        } else {
            self.data.get(key)?.last().map(|v| v.entry.clone())
        }
    }

    fn visible(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.data.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                keys.extend(transaction.writes.keys().cloned());
            }
        }

        let mut result = BTreeMap::new();
        for key in keys {
            if let Some(entry) = self.selected_entry(&key, tx) {
                if let Some(value) = Self::live(&entry, now) {
                    result.insert(key, value.clone());
                }
            }
        }
        result
    }

    fn append(&mut self, key: String, entry: Entry) -> u64 {
        self.revision += 1;
        self.data.entry(key).or_default().push(Version {
            revision: self.revision,
            entry,
        });
        self.revision
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit_time(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(id, Transaction {
            begin_revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit_time(now)?;
        let transaction = self.transactions.get(&id).ok_or(Error::UnknownTransaction(id))?;

        for key in transaction.writes.keys() {
            if self.data.get(key)
                .and_then(|versions| versions.last())
                .is_some_and(|version| version.revision > transaction.begin_revision)
            {
                return Err(Error::Conflict { tx: id, key: key.clone() });
            }
        }

        if transaction.writes.is_empty() {
            self.transactions.remove(&id);
            return Ok(self.revision);
        }

        let writes = transaction.writes.clone();
        self.revision += 1;
        let revision = self.revision;
        for (key, entry) in writes {
            self.data.entry(key).or_default().push(Version { revision, entry });
        }
        self.transactions.remove(&id);
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit_time(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit_time(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let entry = match ttl {
            Some(0) => Entry { value: None, deadline: None },
            Some(duration) => Entry {
                value: Some(value.to_string()),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            },
            None => Entry { value: Some(value.to_string()), deadline: None },
        };

        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), entry);
        } else {
            self.append(key.to_string(), entry);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit_time(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let existed = self.selected_entry(key, tx)
            .as_ref()
            .is_some_and(|entry| Self::live(entry, now).is_some());

        let tombstone = Entry { value: None, deadline: None };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), tombstone);
        } else {
            self.append(key.to_string(), tombstone);
        }
        Ok(existed)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit_time(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.selected_entry(key, tx)
            .as_ref()
            .and_then(|entry| Self::live(entry, now).cloned()))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit_time(now)?;
        self.validate_tx(tx)?;
        Ok(self.visible(tx, now).values().filter(|v| v.as_str() == value).count())
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit_time(now)?;
        self.validate_tx(tx)?;
        Self::validate_range(start, end)?;

        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page { items: Vec::new(), next: None });
        }

        let visible = self.visible(tx, now);
        let mut matching: Vec<(String, String)> = visible.into_iter()
            .filter(|(key, _)| {
                Self::matches(key, start, end, prefix)
                    && cursor.map_or(true, |c| key.as_str() > c.last.as_str())
            })
            .collect();

        if cursor.is_none() {
            let skip = offset.min(matching.len());
            matching.drain(..skip);
        }

        let take = limit.min(matching.len());
        let has_more = take < matching.len();
        matching.truncate(take);

        let next = if has_more {
            matching.last().map(|(last, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };

        Ok(Page { items: matching, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit_time(now)?;
        self.validate_tx(tx)?;
        Self::validate_range(start, end)?;

        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.visible(tx, now) {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            let parsed = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum.checked_add(parsed).ok_or_else(|| Error::SumOverflow {
                key: key.clone(),
            })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit_time(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        Self::validate_name(name)?;

        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }

        let mut saved = BTreeMap::new();
        for (key, versions) in &self.data {
            if let Some(version) = versions.last() {
                if Self::live(&version.entry, now).is_some() {
                    saved.insert(key.clone(), version.entry.clone());
                }
            }
        }
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit_time(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        Self::validate_name(name)?;

        let saved = self.snapshots.get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let current = self.visible(None, now);
        let copied: BTreeMap<String, Entry> = saved.into_iter()
            .filter(|(_, entry)| Self::live(entry, now).is_some())
            .collect();

        let mut keys: BTreeSet<String> = current.keys().cloned().collect();
        keys.extend(copied.keys().cloned());
        if keys.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in keys {
            let entry = copied.get(&key).cloned()
                .unwrap_or(Entry { value: None, deadline: None });
            self.data.entry(key).or_default().push(Version { revision, entry });
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit_time(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        Self::validate_name(name)?;
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit_time(now)?;
        let mut removed = 0usize;

        for versions in self.data.values_mut() {
            if versions.len() <= 1 {
                continue;
            }
            let mut keep = BTreeSet::new();
            keep.insert(versions.len() - 1);
            for transaction in self.transactions.values() {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= transaction.begin_revision) {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit_time(now)?;
        let live_keys = self.visible(None, now).len();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values()
            .flat_map(|v| v.iter())
            .filter(|v| v.entry.value.is_none())
            .count();
        let read_only_transactions = self.transactions.values().filter(|t| t.read_only).count();
        let staged_writes = self.transactions.values().map(|t| t.writes.len()).sum();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}

fn parse_canonical_i64(value: &str) -> Option<i64> {
    if value.is_empty() {
        return None;
    }

    let digits = if let Some(rest) = value.strip_prefix('-') {
        if rest.is_empty() || rest == "0" {
            return None;
        }
        rest
    } else {
        value
    };

    if digits == "0" {
        return value.parse::<i64>().ok();
    }

    let bytes = digits.as_bytes();
    if bytes.first().is_none_or(|b| !(b'1'..=b'9').contains(b))
        || bytes.iter().skip(1).any(|b| !b.is_ascii_digit())
    {
        return None;
    }

    value.parse::<i64>().ok()
}
