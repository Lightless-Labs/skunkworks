```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Value {
    value: String,
    deadline: Option<u64>,
}

#[derive(Clone)]
enum VersionData {
    Value(Value),
    Tombstone,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    data: VersionData,
}

#[derive(Clone)]
struct Write {
    data: VersionData,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Value>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn tx_exists(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn selected_data(&self, key: &str, revision: Option<u64>) -> Option<VersionData> {
        let versions = self.data.get(key)?;
        let v = match revision {
            Some(r) => versions.iter().rev().find(|v| v.revision <= r)?,
            None => versions.last()?,
        };
        Some(v.data.clone())
    }

    fn visible_value(&self, key: &str, tx: Option<u64>, now: u64) -> Option<String> {
        let data = if let Some(id) = tx {
            let t = self.transactions.get(&id)?;
            if let Some(w) = t.writes.get(key) {
                w.data.clone()
            } else {
                self.selected_data(key, Some(t.begin_revision))?
            }
        } else {
            self.selected_data(key, None)?
        };

        match data {
            VersionData::Value(v) if v.deadline.map_or(true, |d| d > now) => Some(v.value),
            _ => None,
        }
    }

    fn visible_map(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.data.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(t) = self.transactions.get(&id) {
                keys.extend(t.writes.keys().cloned());
            }
        }

        keys.into_iter()
            .filter_map(|key| self.visible_value(&key, tx, now).map(|v| (key, v)))
            .collect()
    }

    fn append(&mut self, key: String, data: VersionData, revision: u64) {
        self.data
            .entry(key)
            .or_default()
            .push(Version { revision, data });
    }

    fn check_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    fn check_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        Ok(())
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    fn matching_map(
        &self,
        tx: Option<u64>,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Vec<(String, String)> {
        self.visible_map(tx, now)
            .into_iter()
            .filter(|(k, _)| Self::matches(k, start, end, prefix))
            .collect()
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let Some(tx) = self.transactions.get(&id) else {
            return Err(Error::UnknownTransaction(id));
        };

        for key in tx.writes.keys() {
            if self
                .data
                .get(key)
                .and_then(|vs| vs.last())
                .is_some_and(|v| v.revision > tx.begin_revision)
            {
                return Err(Error::Conflict {
                    tx: id,
                    key: key.clone(),
                });
            }
        }

        if tx.writes.is_empty() {
            self.transactions.remove(&id);
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        let writes = self.transactions.remove(&id).unwrap().writes;
        for (key, write) in writes {
            self.append(key, write.data, revision);
        }
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let data = match ttl {
            Some(0) => VersionData::Tombstone,
            Some(d) => VersionData::Value(Value {
                value: value.to_string(),
                deadline: Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            }),
            None => VersionData::Value(Value {
                value: value.to_string(),
                deadline: None,
            }),
        };

        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(
                key.to_string(),
                Write { data },
            );
        } else {
            self.revision += 1;
            self.append(key.to_string(), data, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let existed = self.visible_value(key, tx, now).is_some();
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(
                key.to_string(),
                Write {
                    data: VersionData::Tombstone,
                },
            );
        } else {
            self.revision += 1;
            self.append(key.to_string(), VersionData::Tombstone, self.revision);
        }
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.visible_value(key, tx, now))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        Ok(self
            .visible_map(tx, now)
            .values()
            .filter(|v| v.as_str() == value)
            .count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        Self::check_range(start, end)?;

        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut entries = self.matching_map(tx, now, start, end, prefix);
        if let Some(c) = cursor {
            entries.retain(|(k, _)| k > &c.last);
        } else {
            entries = entries.into_iter().skip(offset).collect();
        }

        let has_more = entries.len() > limit;
        let items: Vec<_> = entries.into_iter().take(limit).collect();
        let next = if has_more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };

        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        Self::check_range(start, end)?;

        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.matching_map(tx, now, start, end, prefix) {
            if !Self::canonical_i64(&value) {
                return Err(Error::InvalidNumber { key, value });
            }
            let parsed = value
                .parse::<i64>()
                .map_err(|_| Error::InvalidNumber {
                    key: key.clone(),
                    value: value.clone(),
                })?;
            sum = sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    fn canonical_i64(value: &str) -> bool {
        let bytes = value.as_bytes();
        if bytes.is_empty() {
            return false;
        }
        let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
        if digits.is_empty() || !digits.iter().all(u8::is_ascii_digit) {
            return false;
        }
        if digits.len() > 1 && digits[0] == b'0' {
            return false;
        }
        if bytes[0] == b'-' && digits == b"0" {
            return false;
        }
        true
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::check_name(name)?;
        if self.snapshots.len() >= 3 && !self.snapshots.contains_key(name) {
            return Err(Error::SnapshotLimit);
        }
        let saved = self
            .visible_map(None, now)
            .into_iter()
            .filter_map(|(key, _)| {
                let data = self.selected_data(&key, None)?;
                match data {
                    VersionData::Value(v) => Some((key, v)),
                    VersionData::Tombstone => None,
                }
            })
            .collect();
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let Some(saved) = self.snapshots.get(name) else {
            return Err(Error::SnapshotMissing(name.to_string()));
        };
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let saved_live: BTreeMap<String, Value> = saved
            .iter()
            .filter(|(_, v)| v.deadline.map_or(true, |d| d > now))
            .map(|(k, v)| (k.clone(), v.clone()))
            .collect();

        let current = self.visible_map(None, now);
        let keys: BTreeSet<String> = current
            .keys()
            .chain(saved_live.keys())
            .cloned()
            .collect();

        if keys.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in keys {
            let data = saved_live
                .get(&key)
                .cloned()
                .map(VersionData::Value)
                .unwrap_or(VersionData::Tombstone);
            self.append(key, data, revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let begin_revisions: Vec<u64> = self
            .transactions
            .values()
            .map(|t| t.begin_revision)
            .collect();
        let mut removed = 0;

        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for begin in &begin_revisions {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= *begin) {
                    keep.insert(index);
                }
            }

            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let live_keys = self.visible_map(None, now).len();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self
            .data
            .values()
            .flat_map(|vs| vs.iter())
            .filter(|v| matches!(v.data, VersionData::Tombstone))
            .count();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self
                .transactions
                .values()
                .filter(|t| t.read_only)
                .count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        })
    }
}
```
