use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let b = s.as_bytes();
    let mut p = 0usize;
    let mut nums = [0u64; 3];

    for i in 0..3 {
        let start = p;
        while p < b.len() && b[p].is_ascii_digit() {
            p += 1;
        }
        if p == start {
            return Err(if p == b.len() || matches!(b[p], b'.' | b'-' | b'+') {
                VersionError::MissingCore { at: p }
            } else {
                VersionError::InvalidChar { at: p }
            });
        }
        if p - start > 1 && b[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        nums[i] = s[start..p]
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;
        if i < 2 {
            if p >= b.len() {
                return Err(VersionError::MissingCore { at: p });
            }
            if b[p] != b'.' {
                return Err(if matches!(b[p], b'-' | b'+') {
                    VersionError::MissingCore { at: p }
                } else {
                    VersionError::InvalidChar { at: p }
                });
            }
            p += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < b.len() && b[p] == b'-' {
        let start = p + 1;
        let end = b[start..]
            .iter()
            .position(|&c| c == b'+')
            .map(|n| start + n)
            .unwrap_or(b.len());
        pre = parse_identifiers(s, start, end, true)?;
        p = end;
    }
    if p < b.len() && b[p] == b'+' {
        let start = p + 1;
        build = parse_identifiers(s, start, b.len(), false)?;
        p = b.len();
    }
    if p != b.len() {
        return Err(VersionError::InvalidChar { at: p });
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut id_start = start;
    loop {
        let mut p = id_start;
        while p < end && b[p] != b'.' {
            if !b[p].is_ascii_alphanumeric() && b[p] != b'-' {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if prerelease
            && b[id_start..p].iter().all(u8::is_ascii_digit)
            && p - id_start > 1
            && b[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(s[id_start..p].to_owned());
        if p == end {
            break;
        }
        id_start = p + 1;
    }
    Ok(out)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|c| c.is_ascii_digit())
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = match (numeric_id(x), numeric_id(y)) {
                        (true, true) => cmp_numeric_id(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn cmp_op(op: Op, v: &Version, rhs: &Version) -> bool {
    match op {
        Op::Eq => compare_versions(v, rhs) == Ordering::Equal,
        Op::Lt => compare_versions(v, rhs) == Ordering::Less,
        Op::Le => compare_versions(v, rhs) != Ordering::Greater,
        Op::Gt => compare_versions(v, rhs) == Ordering::Greater,
        Op::Ge => compare_versions(v, rhs) != Ordering::Less,
    }
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let b = s.as_bytes();
    let mut result = Vec::new();
    let mut p = 0usize;
    while p < b.len() {
        while p < b.len() && whitespace(b[p]) {
            p += 1;
        }
        if p == b.len() {
            break;
        }
        let start = p;
        while p < b.len() && !whitespace(b[p]) {
            p += 1;
        }
        result.push((s[start..p].to_owned(), base + start));
    }
    result
}

fn full_or_error(token: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(token).map_err(|error| RangeError::Version { at, error })
}

fn make_upper(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_owned()],
        build: Vec::new(),
    }
}

fn bump_tuple(
    nums: [u64; 3],
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut n = nums;
    n[field] = n[field]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at })?;
    for x in n.iter_mut().skip(field + 1) {
        *x = 0;
    }
    Ok(make_upper(n[0], n[1], n[2]))
}

#[derive(Clone, Debug)]
struct Partial {
    nums: [u64; 3],
    k: usize,
    full: Option<Version>,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        return Ok(Partial {
            nums: [0; 3],
            k: 3,
            full: Some(full_or_error(s, at)?),
        });
    }

    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")) {
        return Ok(Partial {
            nums: [0; 3],
            k: 3,
            full: Some(full_or_error(s, at)?),
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }

    let mut nums = [0u64; 3];
    let mut k = 0;
    let mut wildcard_seen = false;
    for (i, field) in fields.iter().enumerate() {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(*field, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !field.bytes().all(|c| c.is_ascii_digit())
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            nums[i] = field.parse().map_err(|_| RangeError::Syntax { at })?;
            k += 1;
        }
    }
    Ok(Partial {
        nums,
        k,
        full: None,
    })
}

fn add_clause(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    arm.clauses.push(Clause {
        op,
        version: v,
        anchor,
    });
}

fn parse_general_token(arm: &mut Arm, token: &str, at: usize) -> Result<(), RangeError> {
    if token.starts_with('^') || token.starts_with('~') {
        let caret = token.starts_with('^');
        let operand = &token[1..];
        let op_at = at + 1;
        let p = parse_partial(operand, op_at)?;
        if p.k == 0 {
            return Err(RangeError::Syntax { at });
        }
        let lower = p.full.clone().unwrap_or_else(|| Version {
            major: p.nums[0],
            minor: p.nums[1],
            patch: p.nums[2],
            pre: Vec::new(),
            build: Vec::new(),
        });
        let anchor = !lower.pre.is_empty();
        add_clause(arm, Op::Ge, lower.clone(), anchor);
        let field = if caret {
            if p.nums[0] > 0 || p.k == 1 {
                0
            } else if p.nums[1] > 0 || p.k == 2 {
                1
            } else {
                2
            }
        } else if p.k == 1 {
            0
        } else {
            1
        };
        let upper = bump_tuple(p.nums, field, op_at)?;
        add_clause(arm, Op::Lt, upper, false);
        return Ok(());
    }

    let explicit = token.starts_with('<')
        || token.starts_with('>')
        || token.starts_with('=');
    if explicit {
        let (op, n) = if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with(">=") {
            (Op::Ge, 2)
        } else {
            match token.as_bytes()[0] {
                b'<' => (Op::Lt, 1),
                b'>' => (Op::Gt, 1),
                _ => (Op::Eq, 1),
            }
        };
        let operand_at = at + n;
        if operand_at >= at + token.len()
            || !token.as_bytes()[n].is_ascii_digit()
        {
            return Err(RangeError::Syntax { at });
        }
        let v = full_or_error(&token[n..], operand_at)?;
        add_clause(arm, op, v, !v.pre.is_empty());
        return Ok(());
    }

    if !token.as_bytes()[0].is_ascii_digit()
        && !matches!(token, "*" | "x" | "X")
    {
        return Err(RangeError::Syntax { at });
    }

    let p = parse_partial(token, at)?;
    if let Some(v) = p.full {
        add_clause(arm, Op::Eq, v.clone(), !v.pre.is_empty());
    } else if p.k == 0 {
        // All releases.
    } else if p.k == 3 {
        let v = Version {
            major: p.nums[0],
            minor: p.nums[1],
            patch: p.nums[2],
            pre: Vec::new(),
            build: Vec::new(),
        };
        add_clause(arm, Op::Eq, v, false);
    } else {
        let lower = Version {
            major: p.nums[0],
            minor: p.nums[1],
            patch: p.nums[2],
            pre: Vec::new(),
            build: Vec::new(),
        };
        add_clause(arm, Op::Ge, lower, false);
        let upper = bump_tuple(p.nums, p.k - 1, at)?;
        add_clause(arm, Op::Lt, upper, false);
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut segments: Vec<(usize, usize)> = Vec::new();
    let mut p = 0usize;
    let mut start = 0usize;
    while p < b.len() {
        if b[p..].starts_with(b"||") {
            if p > 0 && !whitespace(b[p - 1]) || p + 2 < b.len() && !whitespace(b[p + 2]) {
                p += 2;
                continue;
            }
            segments.push((start, p));
            p += 2;
            start = p;
        } else {
            p += 1;
        }
    }
    segments.push((start, b.len()));

    if s.bytes().all(whitespace) {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    for (lo, hi) in segments {
        let text = &s[lo..hi];
        let tokens = split_tokens(text, lo);
        if tokens.is_empty() {
            let at = if hi == b.len() && s[..lo].trim_matches(|c: char| c.is_ascii_whitespace()).ends_with("||") {
                b.len()
            } else {
                lo
            };
            return Err(RangeError::Empty { at });
        }
        let hyphens: Vec<usize> = tokens
            .iter()
            .filter(|(t, _)| t == "-")
            .map(|(_, at)| *at)
            .collect();
        if !hyphens.is_empty() {
            if tokens.len() != 3 || tokens[1].0 != "-" {
                return Err(RangeError::Syntax { at: hyphens[0] });
            }
            let lower = parse_partial(&tokens[0].0, tokens[0].1)?;
            let upper = parse_partial(&tokens[2].0, tokens[2].1)?;
            if lower.k == 0 {
                return Err(RangeError::Syntax { at: tokens[0].1 });
            }
            if upper.k == 0 {
                return Err(RangeError::Syntax { at: tokens[2].1 });
            }
            let lv = lower.full.clone().unwrap_or_else(|| Version {
                major: lower.nums[0],
                minor: lower.nums[1],
                patch: lower.nums[2],
                pre: Vec::new(),
                build: Vec::new(),
            });
            add_clause(&mut Arm { clauses: Vec::new() }, Op::Ge, lv, false);
            let mut arm = Arm { clauses: Vec::new() };
            add_clause(&mut arm, Op::Ge, lv.clone(), !lv.pre.is_empty());
            if let Some(v) = upper.full {
                add_clause(&mut arm, Op::Le, v.clone(), !v.pre.is_empty());
            } else {
                let upper_bound = if upper.k == 1 { 0 } else { 1 };
                add_clause(
                    &mut arm,
                    Op::Lt,
                    bump_tuple(upper.nums, upper_bound, tokens[2].1)?,
                    false,
                );
            }
            arms.push(arm);
        } else {
            let mut arm = Arm { clauses: Vec::new() };
            for (token, at) in tokens {
                parse_general_token(&mut arm, &token, at)?;
            }
            arms.push(arm);
        }
    }

    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty()
        && !include_prerelease
        && !arm.clauses.iter().any(|c| {
            c.anchor
                && !c.version.pre.is_empty()
                && (c.version.major, c.version.minor, c.version.patch)
                    == (v.major, v.minor, v.patch)
        })
    {
        return false;
    }
    arm.clauses.iter().all(|c| cmp_op(c.op, v, &c.version))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..candidates.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(&parsed[i], &parsed[j]);
                if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn visit_ranges<F: FnMut(&str, &[usize])>(
    expr: &SetExpr,
    path: &mut Vec<usize>,
    f: &mut F,
) {
    match expr {
        SetExpr::Range(s) => f(s, path),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                visit_ranges(child, path, f);
                path.pop();
            }
        }
    }
}

fn expr_matches(expr: &SetExpr, v: &Version, include: bool, parsed: &[Range]) -> bool {
    fn walk(
        expr: &SetExpr,
        v: &Version,
        include: bool,
        parsed: &[Range],
        cursor: &mut usize,
    ) -> bool {
        match expr {
            SetExpr::Range(_) => {
                let i = *cursor;
                *cursor += 1;
                matches_with(v, &parsed[i], include)
            }
            SetExpr::Union(children) => children.iter().any(|e| walk(e, v, include, parsed, cursor)),
            SetExpr::Intersection(children) => children.iter().all(|e| walk(e, v, include, parsed, cursor)),
        }
    }
    walk(expr, v, include, parsed, &mut 0)
}

fn count_leaves(expr: &SetExpr) -> usize {
    match expr {
        SetExpr::Range(_) => 1,
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => xs.iter().map(count_leaves).sum(),
    }
}

fn ascii_vec_cmp(a: &[String], b: &[String]) -> Ordering {
    for (x, y) in a.iter().zip(b) {
        let c = x.as_bytes().cmp(y.as_bytes());
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = Vec::<&str>::new();
    for p in packages {
        if names.contains(&p.name.as_str()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
        names.push(&p.name);
    }

    for p in packages {
        let mut labels = Vec::<&str>::new();
        for c in &p.constraints {
            if labels.contains(&c.label.as_str()) {
                return Err(SelectError::DuplicateLabel {
                    package: p.name.clone(),
                    label: c.label.clone(),
                });
            }
            labels.push(&c.label);
        }
    }

    let mut parsed_candidates: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            versions.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(),
                index: i,
                error,
            })?);
        }
        parsed_candidates.push(versions);
    }

    let mut parsed_ranges: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut package_ranges = Vec::new();
        for c in &p.constraints {
            let mut ranges = Vec::new();
            let mut path = Vec::new();
            let mut failure = None;
            visit_ranges(&c.expr, &mut path, &mut |s, path| {
                if failure.is_some() {
                    return;
                }
                match parse_range(s) {
                    Ok(r) => ranges.push(r),
                    Err(error) => {
                        failure = Some(SelectError::InvalidRange {
                            package: p.name.clone(),
                            label: c.label.clone(),
                            path: path.to_vec(),
                            error,
                        })
                    }
                }
            });
            if let Some(e) = failure {
                return Err(e);
            }
            package_ranges.push(ranges);
        }
        parsed_ranges.push(package_ranges);
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &parsed_candidates[pi];
        let mut qualifying = Vec::new();
        for (i, v) in versions.iter().enumerate() {
            let mut ok = true;
            for (ci, c) in p.constraints.iter().enumerate() {
                if !expr_matches(&c.expr, v, include_prerelease, &parsed_ranges[pi][ci]) {
                    ok = false;
                    break;
                }
            }
            if ok {
                qualifying.push(i);
            }
        }
        if qualifying.is_empty() {
            let labels = if p.candidates.is_empty() {
                Vec::new()
            } else {
                let n = p.constraints.len();
                let mut best: Option<Vec<String>> = None;
                for mask in 1usize..(1usize << n) {
                    let count = mask.count_ones() as usize;
                    if best.as_ref().is_some_and(|b| b.len() < count) {
                        continue;
                    }
                    let any = versions.iter().enumerate().any(|(vi, v)| {
                        (0..n).all(|ci| {
                            (mask & (1usize << ci)) == 0
                                || expr_matches(
                                    &p.constraints[ci].expr,
                                    v,
                                    include_prerelease,
                                    &parsed_ranges[pi][ci],
                                )
                        })
                    });
                    if !any {
                        let mut candidate: Vec<String> = (0..n)
                            .filter(|ci| (mask & (1usize << ci)) != 0)
                            .map(|ci| p.constraints[ci].label.clone())
                            .collect();
                        candidate.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                        let replace = best.as_ref().is_none_or(|b| {
                            candidate.len() < b.len()
                                || candidate.len() == b.len() && ascii_vec_cmp(&candidate, b) == Ordering::Less
                        });
                        if replace {
                            best = Some(candidate);
                        }
                    }
                }
                best.unwrap_or_default()
            };
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels,
            });
        }

        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let choice = match preference {
                Preference::Highest => {
                    compare_versions(&versions[i], &versions[best]) == Ordering::Greater
                }
                Preference::Lowest => {
                    compare_versions(&versions[i], &versions[best]) == Ordering::Less
                }
                Preference::StableFirst => {
                    let stable_i = versions[i].pre.is_empty();
                    let stable_best = versions[best].pre.is_empty();
                    if stable_i != stable_best {
                        stable_i
                    } else {
                        compare_versions(&versions[i], &versions[best]) == Ordering::Greater
                    }
                }
            };
            if choice {
                best = i;
            }
        }
        selections.push(Selection {
            package: p.name.clone(),
            version: p.candidates[best].clone(),
        });
    }
    Ok(selections)
}

#[derive(Clone, Debug)]
struct Bound {
    version: Version,
    inclusive: bool,
}

fn canonical_arm(arm: &Arm) -> Option<String> {
    let mut lower: Option<Bound> = None;
    let mut upper: Option<Bound> = None;
    let mut anchors = Vec::new();
    for c in &arm.clauses {
        if c.anchor && !c.version.pre.is_empty() {
            anchors.push((c.version.major, c.version.minor, c.version.patch));
        }
        match c.op {
            Op::Eq => {
                let b = Bound { version: c.version.clone(), inclusive: true };
                lower = Some(stronger_lower(lower, b.clone()));
                upper = Some(stronger_upper(upper, b));
            }
            Op::Gt | Op::Ge => {
                let b = Bound { version: c.version.clone(), inclusive: matches!(c.op, Op::Ge) };
                lower = Some(stronger_lower(lower, b));
            }
            Op::Lt | Op::Le => {
                let b = Bound { version: c.version.clone(), inclusive: matches!(c.op, Op::Le) };
                upper = Some(stronger_upper(upper, b));
            }
        }
    }
    anchors.sort();
    anchors.dedup();
    let lo = lower.as_ref().map(|b| format!("{}{}", if b.inclusive { ">=" } else { ">" }, version_string(&b.version, false))).unwrap_or_else(|| "*".to_owned());
    let hi = upper.as_ref().map(|b| format!("{}{}", if b.inclusive { "<=" } else { "<" }, version_string(&b.version, false))).unwrap_or_else(|| "*".to_owned());
    let anchor_text = if anchors.is_empty() {
        "-".to_owned()
    } else {
        anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
    };
    Some(format!("[{lo};{hi};pre={anchor_text}]"))
}

fn stronger_lower(old: Option<Bound>, new: Bound) -> Bound {
    match old {
        None => new,
        Some(o) => match compare_versions(&new.version, &o.version) {
            Ordering::Greater => new,
            Ordering::Less => o,
            Ordering::Equal => {
                if !new.inclusive { new } else { o }
            }
        },
    }
}

fn stronger_upper(old: Option<Bound>, new: Bound) -> Bound {
    match old {
        None => new,
        Some(o) => match compare_versions(&new.version, &o.version) {
            Ordering::Less => new,
            Ordering::Greater => o,
            Ordering::Equal => {
                if !new.inclusive { new } else { o }
            }
        },
    }
}

fn version_string(v: &Version, build: bool) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    if build && !v.build.is_empty() {
        s.push('+');
        s.push_str(&v.build.join("."));
    }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut items = Vec::new();
    for arm in &r.arms {
        if let Some(s) = canonical_arm(arm) {
            items.push(s);
        }
    }
    items.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    items.dedup();
    if items.is_empty() {
        "EMPTY".to_owned()
    } else {
        items.join(" || ")
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    let anchors_a: Vec<_> = a.arms.iter().flat_map(|x| x.clauses.iter())
        .filter(|c| c.anchor && !c.version.pre.is_empty())
        .map(|c| (c.version.major,c.version.minor,c.version.patch))
        .collect();
    let anchors_b: Vec<_> = b.arms.iter().flat_map(|x| x.clauses.iter())
        .filter(|c| c.anchor && !c.version.pre.is_empty())
        .map(|c| (c.version.major,c.version.minor,c.version.patch))
        .collect();
    for aa in &a.arms {
        for bb in &b.arms {
            let mut candidates = Vec::new();
            for major in [0, 1, u64::MAX] {
                candidates.push(Version { major, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new() });
            }
            for c in aa.clauses.iter().chain(bb.clauses.iter()) {
                candidates.push(c.version.clone());
            }
            for tuple in anchors_a.iter().chain(anchors_b.iter()) {
                candidates.push(Version { major: tuple.0, minor: tuple.1, patch: tuple.2, pre: vec!["0".into()], build: Vec::new() });
            }
            if !include_prerelease {
                for tuple in anchors_a.iter().filter(|x| anchors_b.contains(x)) {
                    candidates.push(Version { major: tuple.0, minor: tuple.1, patch: tuple.2, pre: vec!["0".into()], build: Vec::new() });
                }
            }
            if candidates.iter().any(|v| {
                arm_matches(v, aa, include_prerelease) && arm_matches(v, bb, include_prerelease)
            }) {
                return true;
            }
        }
    }
    false
}
