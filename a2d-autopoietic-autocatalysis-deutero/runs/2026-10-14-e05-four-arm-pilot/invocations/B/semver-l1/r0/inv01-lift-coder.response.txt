use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in &bytes[start..pos] {
            value = match value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
            {
                Some(n) => n,
                None => return Err(VersionError::Overflow { at: start }),
            };
        }
        core[field] = value;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let pre_end = bytes[pos..]
                    .iter()
                    .position(|&b| b == b'+')
                    .map(|i| pos + i)
                    .unwrap_or(bytes.len());
                pre = parse_identifiers(s, bytes, pos, pre_end, true)?;
                pos = pre_end;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    if pos < bytes.len() {
        if bytes[pos] != b'+' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        build = parse_identifiers(s, bytes, pos, bytes.len(), false)?;
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    bytes: &[u8],
    start: usize,
    end: usize,
    check_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let ident_end = bytes[ident_start..end]
            .iter()
            .position(|&b| b == b'.')
            .map(|i| ident_start + i)
            .unwrap_or(end);

        if ident_start == ident_end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &bytes[ident_start..ident_end];
        for (offset, &b) in ident.iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar {
                    at: ident_start + offset,
                });
            }
        }

        if check_numeric_leading_zero
            && ident.len() > 1
            && ident[0] == b'0'
            && ident.iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        result.push(s[ident_start..ident_end].to_owned());

        if ident_end == end {
            break;
        }
        ident_start = ident_end + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (left, right) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_identifiers(left, right);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CompareOp,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum CompareOp {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = vec![Vec::new()];
    let mut pos = 0;
    let mut token_count = 0;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        token_count += 1;

        let token = &s[start..pos];
        let (op, operand_offset) = if token.starts_with(">=") {
            (CompareOp::GreaterEqual, 2)
        } else if token.starts_with("<=") {
            (CompareOp::LessEqual, 2)
        } else if token.starts_with('>') {
            (CompareOp::Greater, 1)
        } else if token.starts_with('<') {
            (CompareOp::Less, 1)
        } else if token.starts_with('=') {
            (CompareOp::Equal, 1)
        } else {
            (CompareOp::Equal, 0)
        };

        if operand_offset >= token.len() || !token.as_bytes()[operand_offset].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }

        let operand_start = start + operand_offset;
        let version = parse_version(&s[operand_start..pos]).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        arms[0].push(Comparator { op, version });
    }

    if token_count == 0 {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { arms })
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted = arm.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && (comparator.version.major, comparator.version.minor, comparator.version.patch)
                    == (v.major, v.minor, v.patch)
        });

        if !v.pre.is_empty() && !admitted {
            return false;
        }

        arm.iter().all(|comparator| {
            let ordering = compare_versions(v, &comparator.version);
            match comparator.op {
                CompareOp::Equal => ordering == Ordering::Equal,
                CompareOp::Less => ordering == Ordering::Less,
                CompareOp::LessEqual => ordering != Ordering::Greater,
                CompareOp::Greater => ordering == Ordering::Greater,
                CompareOp::GreaterEqual => ordering != Ordering::Less,
            }
        })
    })
}
