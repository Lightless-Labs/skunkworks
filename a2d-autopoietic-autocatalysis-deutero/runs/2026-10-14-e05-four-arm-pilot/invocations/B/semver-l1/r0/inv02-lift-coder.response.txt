use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in &bytes[start..pos] {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        core[field] = value;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let (ids, next) = parse_identifiers(bytes, pos, true)?;
        pre = ids;
        pos = next;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        let (ids, next) = parse_identifiers(bytes, pos, false)?;
        build = ids;
        pos = next;
    }

    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    is_pre: bool,
) -> Result<(Vec<String>, usize), VersionError> {
    let mut ids = Vec::new();
    let mut pos = start;

    loop {
        let id_start = pos;
        while pos < bytes.len()
            && (bytes[pos].is_ascii_alphanumeric() || bytes[pos] == b'-')
        {
            pos += 1;
        }

        if pos == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        if pos - id_start > 1
            && bytes[id_start] == b'0'
            && bytes[id_start..pos].iter().all(u8::is_ascii_digit)
            && is_pre
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        ids.push(String::from_utf8(bytes[id_start..pos].to_vec()).unwrap());

        if pos == bytes.len() || (is_pre && bytes[pos] == b'+') {
            return Ok((ids, pos));
        }

        if bytes[pos] == b'.' {
            pos += 1;
            continue;
        }

        return Err(VersionError::InvalidChar { at: pos });
    }
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core_order = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core_order != Ordering::Equal {
        return core_order;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (ai, bi) in a.pre.iter().zip(&b.pre) {
                let an = is_numeric_identifier(ai);
                let bn = is_numeric_identifier(bi);
                let order = match (an, bn) {
                    (true, true) => ai
                        .len()
                        .cmp(&bi.len())
                        .then_with(|| ai.as_bytes().cmp(bi.as_bytes())),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    (false, false) => ai.as_bytes().cmp(bi.as_bytes()),
                };
                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    admits_pre: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();
    let mut admits_pre = Vec::new();

    while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
        pos += 1;
    }
    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let mut operand_start = token_start;
        let op = match bytes[token_start] {
            b'=' => {
                operand_start += 1;
                Op::Eq
            }
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Op::Le
            }
            b'<' => {
                operand_start += 1;
                Op::Lt
            }
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Op::Ge
            }
            b'>' => {
                operand_start += 1;
                Op::Gt
            }
            _ => Op::Eq,
        };

        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            admits_pre.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { op, version });

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        comparators,
        admits_pre,
    })
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .admits_pre
            .iter()
            .any(|&(major, minor, patch)| (v.major, v.minor, v.patch) == (major, minor, patch))
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let order = compare_versions(v, &comparator.version);
        match comparator.op {
            Op::Eq => order == Ordering::Equal,
            Op::Lt => order == Ordering::Less,
            Op::Le => order != Ordering::Greater,
            Op::Gt => order == Ordering::Greater,
            Op::Ge => order != Ordering::Less,
        }
    })
}
