#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];

        output.push(ALPHABET[(a >> 2) as usize] as char);
        output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
        output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
        output.push(ALPHABET[(c & 0x3f) as usize] as char);
    }

    let tail = chunks.remainder();
    match tail.len() {
        1 => {
            let a = tail[0];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push_str("==");
        }
        2 => {
            let a = tail[0];
            let b = tail[1];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        _ => {}
    }

    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut output = Vec::with_capacity(input.len() / 4 * 3);
    let mut group: [(u8, usize); 4] = [(0, 0); 4];
    let mut count = 0usize;
    let mut terminal_padding = false;
    let mut pending_second_padding = false;
    let mut pending_offset = 0usize;

    for (offset, &byte) in input.iter().enumerate() {
        if terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if pending_second_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            group[3] = (byte, offset);
            output.push((digit(group[0].0).unwrap() << 2) | (digit(group[1].0).unwrap() >> 4));
            count = 0;
            pending_second_padding = false;
            terminal_padding = true;
            continue;
        }

        if count == 0 || count == 1 {
            if byte == b'=' {
                return Err(Error::BadPadding { offset });
            }
            let Some(value) = digit(byte) else {
                return Err(Error::InvalidByte { offset, byte });
            };
            group[count] = (byte, offset);
            count += 1;
            let _ = value;
        } else if count == 2 {
            if byte == b'=' {
                group[2] = (byte, offset);
                pending_second_padding = true;
                pending_offset = offset;
            } else {
                let Some(_) = digit(byte) else {
                    return Err(Error::InvalidByte { offset, byte });
                };
                group[2] = (byte, offset);
                count += 1;
            }
        } else {
            if byte == b'=' {
                let a = digit(group[0].0).unwrap();
                let b = digit(group[1].0).unwrap();
                let c = digit(group[2].0).unwrap();
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                count = 0;
                terminal_padding = true;
            } else {
                let Some(_) = digit(byte) else {
                    return Err(Error::InvalidByte { offset, byte });
                };
                let a = digit(group[0].0).unwrap();
                let b = digit(group[1].0).unwrap();
                let c = digit(group[2].0).unwrap();
                let d = digit(byte).unwrap();
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
                group = [(0, 0); 4];
                count = 0;
                continue;
            }
        }
    }

    if pending_second_padding {
        return Err(Error::BadPadding {
            offset: pending_offset,
        });
    }
    if count != 0 {
        return Err(Error::Truncated {
            offset: group[0].1,
        });
    }

    Ok(output)
}
