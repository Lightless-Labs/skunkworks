```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    job: Job,
    dependencies: Vec<usize>,
    attempts: u8,
    state: Option<State>,
}

pub struct Runner {
    jobs: Vec<JobData>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();

        for (i, job) in jobs.iter().enumerate() {
            let valid_id = !job.id.is_empty()
                && job.id.len() <= 16
                && job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
            if !valid_id {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dep) {
                    Some(&index) => deps.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(deps);
        }

        fn visit(
            node: usize,
            dependencies: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return true;
            }
            if marks[node] == 2 {
                return false;
            }
            marks[node] = 1;
            for &dep in &dependencies[node] {
                if visit(dep, dependencies, marks) {
                    return true;
                }
            }
            marks[node] = 2;
            false
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let data = jobs
            .into_iter()
            .zip(dependencies)
            .map(|(job, dependencies)| JobData {
                job,
                dependencies,
                attempts: 0,
                state: None,
            })
            .collect();

        Ok(Self {
            jobs: data,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn index_of(&self, id: &str) -> Result<usize, Error> {
        self.jobs
            .iter()
            .position(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.jobs.iter().any(|entry| {
            matches!(
                entry.state,
                Some(State::Waiting { .. } | State::Running { .. })
            )
        }) {
            return Err(Error::Busy);
        }

        for entry in &mut self.jobs {
            entry.attempts = 0;
            entry.state = Some(State::Waiting { ready_at: now });
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;
        for (i, entry) in self.jobs.iter().enumerate() {
            let ready_at = match entry.state {
                Some(State::Waiting { ready_at }) if ready_at <= now => ready_at,
                _ => continue,
            };
            if !entry.dependencies.iter().all(|&dep| {
                matches!(self.jobs[dep].state, Some(State::Succeeded))
            }) {
                continue;
            }

            let priority = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                let gained = (now - ready_at) / self.aging_ms;
                (u64::from(entry.job.priority) + gained).min(9) as u8
            };

            if selected.map_or(true, |(_, best)| priority > best) {
                selected = Some((i, priority));
            }
        }

        self.last_time = now;
        if let Some((i, _)) = selected {
            let entry = &mut self.jobs[i];
            entry.attempts += 1;
            entry.state = Some(State::Running {
                attempt: entry.attempts,
            });
            Ok(Some(Ticket {
                id: entry.job.id.clone(),
                attempt: entry.attempts,
            }))
        } else {
            Ok(None)
        }
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let index = self.index_of(id)?;
        if !matches!(self.jobs[index].state, Some(State::Running { .. })) {
            return Err(Error::NotRunning);
        }

        let job = &self.jobs[index].job;
        let attempts = self.jobs[index].attempts;
        let new_state = if success {
            State::Succeeded
        } else if attempts == job.max_attempts {
            State::Failed
        } else {
            let shift = attempts.saturating_sub(1) as u32;
            let delay = self
                .retry_base
                .checked_shl(shift)
                .unwrap_or(u64::MAX)
                .min(self.retry_cap);
            State::Waiting {
                ready_at: now.saturating_add(delay),
            }
        };

        self.jobs[index].state = Some(new_state);

        if matches!(self.jobs[index].state, Some(State::Failed)) {
            loop {
                let mut changed = false;
                for i in 0..self.jobs.len() {
                    if !matches!(self.jobs[i].state, Some(State::Waiting { .. })) {
                        continue;
                    }
                    let should_block = self.jobs[i].dependencies.iter().any(|&dep| {
                        matches!(
                            self.jobs[dep].state,
                            Some(State::Failed | State::Blocked)
                        )
                    });
                    if should_block {
                        self.jobs[i].state = Some(State::Blocked);
                        changed = true;
                    }
                }
                if !changed {
                    break;
                }
            }
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = self.index_of(id)?;
        self.jobs[index].state.clone().ok_or(Error::NoCycle)
    }
}
```
