#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(cs: &[Constraint]) -> String {
    let mut canonical = cs.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn active_constraints(
    roots: &[Requirement],
    selected: &[(String, Version)],
    catalog_by_name: &std::collections::BTreeMap<&str, &Package>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in selected {
        if let Some(package) = catalog_by_name.get(name.as_str()) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        canonicalize(constraints);
    }
    active
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog_by_name: std::collections::BTreeMap<&'a str, &'a Package>,
    locks: &'a [Lock],
    all_names: Vec<String>,
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, selected, &self.catalog_by_name);

        if let Some(name) = active
            .keys()
            .find(|name| !self.catalog_by_name.contains_key(name.as_str()))
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        let assigned: std::collections::BTreeMap<&str, Version> = selected
            .iter()
            .map(|(name, version)| (name.as_str(), *version))
            .collect();

        for (name, version) in &assigned {
            if let Some(constraints) = active.get(*name) {
                if !constraints.iter().all(|c| matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: (*name).to_string(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if assigned.contains_key(name.as_str()) {
                continue;
            }
            let package = self.catalog_by_name[name.as_str()];
            if !package.releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active
            .keys()
            .find(|name| !assigned.contains_key(name.as_str()))
            .cloned();

        let Some(name) = next else {
            self.consider(selected);
            return Ok(());
        };

        let constraints = &active[&name];
        let package = self.catalog_by_name[name.as_str()];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|version| constraints.iter().all(|c| matches(*version, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        let mut local_first_error = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.push((name.clone(), version));
            match self.visit(selected) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    selected.pop();
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if local_first_error.is_none() {
                        local_first_error = Some(error);
                    }
                }
            }
            selected.pop();
        }

        if let Some(error) = local_first_error {
            Err(error)
        } else {
            Ok(())
        }
    }

    fn consider(&mut self, selected: &[(String, Version)]) {
        let cost = self
            .locks
            .iter()
            .filter(|lock| {
                !selected
                    .iter()
                    .any(|(name, version)| name == &lock.package && *version == lock.version)
            })
            .count();

        let better = match &self.best {
            None => true,
            Some((best_cost, best_selection)) => {
                if cost != *best_cost {
                    cost < *best_cost
                } else {
                    self.compare_vectors(selected, best_selection)
                        == std::cmp::Ordering::Greater
                }
            }
        };

        if better {
            self.best = Some((cost, selected.to_vec()));
        }
    }

    fn compare_vectors(
        &self,
        left: &[(String, Version)],
        right: &[(String, Version)],
    ) -> std::cmp::Ordering {
        let left_map: std::collections::BTreeMap<&str, Version> = left
            .iter()
            .map(|(name, version)| (name.as_str(), *version))
            .collect();
        let right_map: std::collections::BTreeMap<&str, Version> = right
            .iter()
            .map(|(name, version)| (name.as_str(), *version))
            .collect();

        for name in &self.all_names {
            let ordering = match (left_map.get(name.as_str()), right_map.get(name.as_str())) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(a), Some(b)) => a.cmp(b),
            };
            if ordering != std::cmp::Ordering::Equal {
                return ordering;
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_by_name: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();
    let mut all_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    all_names.sort();

    let mut search = Search {
        roots,
        catalog_by_name,
        locks,
        all_names,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };

    let mut selected = Vec::new();
    match search.visit(&mut selected) {
        Err(error @ ResolveError::Limit { .. }) => return Err(error),
        Err(error) => {
            search.first_error = Some(error);
        }
        Ok(()) => {}
    }

    let Some((_, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| {
            ResolveError::NoMatch {
                package: String::new(),
                requirements: Vec::new(),
            }
        }));
    };

    let final_active = active_constraints(roots, &chosen, &search.catalog_by_name);
    let mut selected_out: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selected_out.sort_by(|a, b| a.package.cmp(&b.package));

    let before: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect();
    let after: std::collections::BTreeMap<&str, Version> = chosen
        .iter()
        .map(|(name, version)| (name.as_str(), *version))
        .collect();

    let mut names: std::collections::BTreeSet<&str> = before.keys().copied().collect();
    names.extend(after.keys().copied());

    let mut changes = Vec::new();
    for name in names {
        let b = before.get(name).copied();
        let a = after.get(name).copied();
        if b == a {
            continue;
        }

        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_by_name
                    .get(name)
                    .is_some_and(|package| package.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(name)
                    .is_some_and(|constraints| !constraints.iter().all(|c| matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };

        changes.push(Change {
            package: name.to_string(),
            before: b,
            after: a,
            reason,
        });
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
