#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphabetic()
            || b.is_ascii_digit()
            || *b == b'.'
            || *b == b'_'
            || *b == b'-'
        {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode_text(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| match *c {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    })
}

fn collect_active(
    roots: &[Requirement],
    catalog: &[Package],
    assignments: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in assignments {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best_cost: Option<usize>,
    best: Option<Vec<(String, Version)>>,
    first_failure: Option<ResolveError>,
}

impl Search<'_> {
    fn inspect(
        &mut self,
        assignments: &mut Vec<(String, Version)>,
    ) -> Result<(), ResolveError> {
        let active = collect_active(self.roots, self.catalog, assignments);

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        let assigned: std::collections::BTreeMap<String, Version> =
            assignments.iter().cloned().collect();

        for (name, constraints) in &active {
            if let Some(version) = assigned.get(name) {
                if !matches(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !package
                .releases
                .iter()
                .any(|r| matches(r.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !assigned.contains_key(*name));
        let Some(name) = next else {
            self.record_solution(assignments);
            return Ok(());
        };

        let constraints = &active[name];
        let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|r| matches(r.version, constraints))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.iter().find(|l| l.package == *name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        let mut first_here = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assignments.push((name.clone(), candidate));
            match self.inspect(assignments) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    assignments.pop();
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if first_here.is_none() {
                        first_here = Some(error);
                    }
                }
            }
            assignments.pop();
        }

        if self.best.is_some() {
            Ok(())
        } else if let Some(error) = first_here {
            Err(error)
        } else {
            Ok(())
        }
    }

    fn record_solution(&mut self, assignments: &[(String, Version)]) {
        let mut solution = assignments.to_vec();
        solution.sort_by(|a, b| a.0.cmp(&b.0));

        let cost = self
            .locks
            .iter()
            .filter(|lock| {
                !solution
                    .iter()
                    .any(|(name, version)| name == &lock.package && *version == lock.version)
            })
            .count();

        let better = match (&self.best_cost, &self.best) {
            (None, _) => true,
            (Some(old_cost), Some(old)) if cost < *old_cost => true,
            (Some(old_cost), Some(old)) if cost == *old_cost => {
                let names: std::collections::BTreeSet<String> = self
                    .catalog
                    .iter()
                    .map(|p| p.name.clone())
                    .collect();
                let old_map: std::collections::BTreeMap<String, Version> =
                    old.iter().cloned().collect();
                let new_map: std::collections::BTreeMap<String, Version> =
                    solution.iter().cloned().collect();
                names.into_iter().find_map(|name| {
                    let a = new_map.get(&name);
                    let b = old_map.get(&name);
                    match (a, b) {
                        (None, None) => None,
                        (None, Some(_)) => Some(false),
                        (Some(_), None) => Some(true),
                        (Some(a), Some(b)) if a != b => Some(a > b),
                        _ => None,
                    }
                }).unwrap_or(false)
            }
            _ => false,
        };

        if better {
            self.best_cost = Some(cost);
            self.best = Some(solution);
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best_cost: None,
        best: None,
        first_failure: None,
    };

    let mut assignments = Vec::new();
    let result = search.inspect(&mut assignments);

    if let Err(ResolveError::Limit { attempts }) = result {
        return Err(ResolveError::Limit { attempts });
    }

    let Some(solution) = search.best else {
        if let Err(error) = result {
            return Err(error);
        }
        return Ok(Resolution {
            selected: Vec::new(),
            changes: locks
                .iter()
                .map(|lock| Change {
                    package: lock.package.clone(),
                    before: Some(lock.version),
                    after: None,
                    reason: ChangeReason::Removed,
                })
                .collect(),
        });
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let active = collect_active(roots, catalog, &solution);
    let selected_map: std::collections::BTreeMap<String, Version> =
        solution.iter().cloned().collect();
    let lock_map: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let all_names: std::collections::BTreeSet<String> = lock_map
        .keys()
        .chain(selected_map.keys())
        .cloned()
        .collect();

    let mut changes = Vec::new();
    for name in all_names {
        let before = lock_map.get(&name).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog.iter().any(|p| {
                    p.name == name && p.releases.iter().any(|r| r.version == old)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else if !active
                    .get(&name)
                    .map(|cs| matches(old, cs))
                    .unwrap_or(true)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
