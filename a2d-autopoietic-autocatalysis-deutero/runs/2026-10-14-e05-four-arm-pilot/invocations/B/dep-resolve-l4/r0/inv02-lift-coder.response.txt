pub enum ResolveError {
    InvalidName {
        site: String,

    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
        evidence: Vec<Evidence>,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,

        package: String,

        evidence: Vec<Evidence>,

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn satisfies(cs: &[Constraint], v: Version) -> bool {
    cs.iter().all(|c| matches(*c, v))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn evidence_lines(evidence: &[Evidence]) -> String {
    evidence_sorted(evidence)
        .iter()
        .map(|e| {
            format!(
                "\n  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect::<String>()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            format!("missing|package={}{}", encode(package), evidence_lines(evidence))
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\
             \n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct NodeInfo {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    feature_requests: BTreeSet<String>,
    features: BTreeSet<String>,
    provenance: Option<Vec<String>>,
}

struct Search<'a> {
    catalog: &'a [Package],
    package_indices: BTreeMap<String, usize>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn target_matches(&self, platform: Platform) -> bool {
        matches!(
            (platform, self.target),
            (Platform::Any, _)
                | (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
    }

    fn closure(
        &self,
        selected: &BTreeMap<String, Version>,
    ) -> (BTreeMap<String, NodeInfo>, BTreeMap<String, Vec<String>>) {
        let mut info: BTreeMap<String, NodeInfo> = BTreeMap::new();

        for (i, root) in self.roots.iter().enumerate() {
            let mut fs = root.features.clone();
            fs.sort();
            fs.dedup();
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                root.package,
                constraint_text(root.constraint),
                fs.join(",")
            );
            let entry = info.entry(root.package.clone()).or_default();
            entry.constraints.push(root.constraint);
            entry.origins.push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
            entry.feature_requests.extend(root.features.iter().cloned());
            let path = vec![token];
            if entry
                .provenance
                .as_ref()
                .map_or(true, |old| path.len() < old.len() || (path.len() == old.len() && path < *old))
            {
                entry.provenance = Some(path);
            }
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = info.keys().cloned().collect();

            for package_name in names {
                let Some(version) = selected.get(&package_name).copied() else {
                    continue;
                };
                let Some(&pi) = self.package_indices.get(&package_name) else {
                    continue;
                };
                let Some(release) = self.catalog[pi]
                    .releases
                    .iter()
                    .find(|r| r.version == version)
                else {
                    continue;
                };

                let requests: Vec<String> = info
                    .get(&package_name)
                    .map(|n| n.feature_requests.iter().cloned().collect())
                    .unwrap_or_default();

                for request in requests {
                    let declared = release.features.iter().find(|f| f.name == request);
                    let newly_activated = info
                        .get_mut(&package_name)
                        .unwrap()
                        .features
                        .insert(request.clone());
                    if newly_activated {
                        changed = true;
                    }
                    if let Some(feature) = declared {
                        for enabled in &feature.enables {
                            if info
                                .get_mut(&package_name)
                                .unwrap()
                                .feature_requests
                                .insert(enabled.clone())
                            {
                                changed = true;
                            }
                        }
                    }
                }

                let activated: BTreeSet<String> = info
                    .get(&package_name)
                    .map(|n| n.features.clone())
                    .unwrap_or_default();

                for (dep_index, dep) in release.deps.iter().enumerate() {
                    if !self.target_matches(dep.platform)
                        || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }

                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        package_name,
                        version_text(version),
                        dep_index,
                        dep.package,
                        constraint_text(dep.constraint),
                        fs.join(","),
                        gate,
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );

                    let source_path = info
                        .get(&package_name)
                        .and_then(|n| n.provenance.clone())
                        .unwrap_or_default();
                    let mut dest_path = source_path.clone();
                    dest_path.push(token);
                    let dest = info.entry(dep.package.clone()).or_default();
                    if !dest.constraints.contains(&dep.constraint)
                        || !dest.origins.iter().any(|o| {
                            o.constraint == dep.constraint && o.path == dest_path
                        })
                    {
                        dest.constraints.push(dep.constraint);
                        dest.origins.push(Origin {
                            constraint: dep.constraint,
                            path: dest_path.clone(),
                        });
                        changed = true;
                    }
                    for f in &dep.features {
                        if dest.feature_requests.insert(f.clone()) {
                            changed = true;
                        }
                    }
                    if dest.provenance.as_ref().map_or(true, |old| {
                        dest_path.len() < old.len()
                            || (dest_path.len() == old.len() && dest_path < *old)
                    }) {
                        dest.provenance = Some(dest_path);
                        changed = true;
                    }
                }
            }

            if !changed {
                break;
            }
        }

        let paths = info
            .iter()
            .map(|(name, n)| (name.clone(), n.provenance.clone().unwrap_or_default()))
            .collect();
        (info, paths)
    }

    fn dynamic_error(
        &self,
        selected: &BTreeMap<String, Version>,
        info: &BTreeMap<String, NodeInfo>,
        paths: &BTreeMap<String, Vec<String>>,
    ) -> Option<ResolveError> {
        for name in info.keys() {
            if !self.package_indices.contains_key(name) {
                let evidence = info[name]
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                return Some(ResolveError::Missing {
                    package: name.clone(),
                    evidence,
                });
            }
        }

        for (name, version) in selected {
            let Some(node) = info.get(name) else { continue };
            let Some(&pi) = self.package_indices.get(name) else {
                continue;
            };
            let selected_exists = self.catalog[pi]
                .releases
                .iter()
                .any(|r| r.version == *version);
            if !selected_exists {
                continue;
            }
            let release = self.catalog[pi]
                .releases
                .iter()
                .find(|r| r.version == *version)
                .unwrap();

            let activated: BTreeSet<String> = node.features.clone();
            for feature in activated {
                if !release.features.iter().any(|f| f.name == feature) {
                    return Some(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature,
                        path: paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        for (name, version) in selected {
            if let Some(node) = info.get(name) {
                if !satisfies(&node.constraints, *version) {
                    return Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical_constraints(&node.constraints),
                        evidence: node
                            .origins
                            .iter()
                            .map(|o| Evidence {
                                constraint: o.constraint,
                                path: o.path.clone(),
                            })
                            .collect(),
                    });
                }
            }
        }

        for (name, node) in info {
            if selected.contains_key(name) {
                continue;
            }
            let Some(&pi) = self.package_indices.get(name) else {
                continue;
            };
            let has_candidate = self.catalog[pi]
                .releases
                .iter()
                .any(|r| satisfies(&node.constraints, r.version));
            if !has_candidate {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(&node.constraints),
                    evidence: node
                        .origins
                        .iter()
                        .map(|o| Evidence {
                            constraint: o.constraint,
                            path: o.path.clone(),
                        })
                        .collect(),
                });
            }
        }
        None
    }

    fn score(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count()
    }

    fn better(
        &self,
        score: usize,
        candidate: &BTreeMap<String, Version>,
        incumbent: &(usize, BTreeMap<String, Version>),
    ) -> bool {
        if score != incumbent.0 {
            return score < incumbent.0;
        }
        let mut names: Vec<String> = self.package_indices.keys().cloned().collect();
        names.sort();
        for name in names {
            match (candidate.get(&name), incumbent.1.get(&name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn search(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (info, paths) = self.closure(selected);
        if let Some(error) = self.dynamic_error(selected, &info, &paths) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = info.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(package_name) = next else {
            let score = self.score(selected);
            if self
                .best
                .as_ref()
                .map_or(true, |best| self.better(score, selected, best))
            {
                self.best = Some((score, selected.clone()));
            }
            return Ok(());
        };

        let pi = self.package_indices[&package_name];
        let node = &info[&package_name];
        let mut candidates: Vec<Version> = self.catalog[pi]
            .releases
            .iter()
            .filter(|r| satisfies(&node.constraints, r.version))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(&package_name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), candidate);
            self.search(selected)?;
            selected.remove(&package_name);
        }
        Ok(())
    }
}

fn validate_names(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!(
                            "catalog[{}].release[{}].feature[{}]",
                            i, j, f
                        ),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_indices: BTreeMap<String, usize> = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        catalog,
        package_indices,
        roots,
        locks: lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    let mut selected = BTreeMap::new();
    search.search(&mut selected)?;

    let chosen = match search.best {
        Some((_, chosen)) => chosen,
        None => return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        })),
    };

    let (final_info, _) = search.closure(&chosen);
    let selected_out: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names: BTreeSet<String> = search.locks.keys().cloned().collect();
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();

    for package in all_names {
        let before = search.locks.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .package_indices
                    .get(&package)
                    .map(|&pi| catalog[pi].releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_info
                    .get(&package)
                    .is_some_and(|n| !satisfies(&n.constraints, old))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
