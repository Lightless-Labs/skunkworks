```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version { pub major: u32, pub minor: u32, pub patch: u32 }

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint { Exact(Version), AtLeast(Version), Before(Version), Caret(Version) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package: String, pub version: Version }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package { pub name: String, pub releases: Vec<Release> }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Target { Linux, Mac, Windows }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Platform { Any, Linux, Mac, Windows }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement { pub package: String, pub constraint: Constraint, pub features: Vec<String> }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Dependency {
    pub package: String,
    pub constraint: Constraint,
    pub features: Vec<String>,
    pub gate: Option<String>,
    pub platform: Platform,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Feature { pub name: String, pub enables: Vec<String> }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Dependency>,
    pub features: Vec<Feature>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Evidence { pub constraint: Constraint, pub path: Vec<String> }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock { pub package: String, pub version: Version }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason { Added, Removed, Unavailable, Rejected, Rebalanced }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution { pub selected: Vec<Selection>, pub changes: Vec<Change> }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateFeature { package: String, version: Version, feature: String },
    DuplicateLock { package: String },
    Missing { package: String, evidence: Vec<Evidence> },
    NoMatch { package: String, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    Conflict { package: String, selected: Version, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    MissingFeature { package: String, selected: Version, feature: String, path: Vec<String> },
    Limit { attempts: usize },
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_rank(*c));
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs).iter().map(|c| constraint_text(*c)).collect::<Vec<_>>().join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x && if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn pct(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut e = evidence.to_vec();
    e.sort_by(|a, b| constraint_rank(a.constraint).cmp(&constraint_rank(b.constraint))
        .then_with(|| a.path.cmp(&b.path)));
    e
}

fn append_evidence(out: &mut String, evidence: &[Evidence]) {
    for e in evidence_sorted(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } =>
            format!("invalid-name|site={}|name={}", pct(site), pct(name)),
        ResolveError::DuplicatePackage { package } =>
            format!("duplicate-package|package={}", pct(package)),
        ResolveError::DuplicateVersion { package, version } =>
            format!("duplicate-version|package={}|version={}", pct(package), version_text(*version)),
        ResolveError::DuplicateFeature { package, version, feature } =>
            format!("duplicate-feature|package={}|version={}|feature={}", pct(package), version_text(*version), pct(feature)),
        ResolveError::DuplicateLock { package } =>
            format!("duplicate-lock|package={}", pct(package)),
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", pct(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch { package, requirements, evidence } => {
            let mut s = format!("no-match|package={}|constraints={}", pct(package), constraints_text(requirements));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict { package, selected, requirements, evidence } => {
            let mut s = format!("conflict|package={}|selected={}|constraints={}",
                pct(package), version_text(*selected), constraints_text(requirements));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature { package, selected, feature, path } =>
            format!("missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
                pct(package), version_text(*selected), pct(feature), pct(feature), path.join(" -> ")),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone)]
struct Active {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    features: BTreeSet<String>,
    provenance: Vec<String>,
}

use std::collections::{BTreeMap, BTreeSet};

struct Search<'a> {
    packages: &'a BTreeMap<String, &'a Package>,
    locks: BTreeMap<String, Version>,
    target: Target,
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!((platform, target),
            (Platform::Linux, Target::Linux) |
            (Platform::Mac, Target::Mac) |
            (Platform::Windows, Target::Windows))
}

fn path_key(path: &[String]) -> (usize, Vec<String>) {
    (path.len(), path.to_vec())
}

impl<'a> Search<'a> {
    fn closure(&self, roots: &[Requirement], assigned: &BTreeMap<String, Version>) -> BTreeMap<String, Active> {
        let mut active: BTreeMap<String, Active> = BTreeMap::new();
        for (i, root) in roots.iter().enumerate() {
            let mut fs: Vec<String> = root.features.clone();
            fs.sort();
            fs.dedup();
            let token = format!("root[{}]:{}@{}+{{{}}}", i, root.package,
                constraint_text(root.constraint), fs.join(","));
            let a = active.entry(root.package.clone()).or_insert_with(|| Active {
                constraints: vec![], origins: vec![], features: BTreeSet::new(), provenance: vec![],
            });
            a.constraints.push(root.constraint);
            a.origins.push(Origin { constraint: root.constraint, path: vec![token] });
            a.features.extend(root.features.iter().cloned());
        }

        loop {
            let snapshot = active.clone();
            let mut changed = false;
            for (name, version) in assigned {
                let Some(a) = snapshot.get(name) else { continue };
                let Some(pkg) = self.packages.get(name) else { continue };
                let Some(rel) = pkg.releases.iter().find(|r| r.version == *version) else { continue };

                let mut feature_queue: Vec<String> = a.features.iter().cloned().collect();
                let mut activated = BTreeSet::new();
                while let Some(f) = feature_queue.pop() {
                    if !activated.insert(f.clone()) { continue; }
                    if let Some(decl) = rel.features.iter().find(|d| d.name == f) {
                        for en in &decl.enables { feature_queue.push(en.clone()); }
                    }
                }
                for f in activated {
                    if !active.get(name).map_or(false, |cur| cur.features.contains(&f)) {
                        if let Some(cur) = active.get_mut(name) {
                            cur.features.insert(f);
                            changed = true;
                        }
                    }
                }

                let current_features = active.get(name).map(|x| x.features.clone()).unwrap_or_default();
                for (j, dep) in rel.deps.iter().enumerate() {
                    if !target_matches(dep.platform, self.target)
                        || dep.gate.as_ref().map_or(false, |g| !current_features.contains(g)) {
                        continue;
                    }
                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let platform = match dep.platform {
                        Platform::Any => "any", Platform::Linux => "linux",
                        Platform::Mac => "mac", Platform::Windows => "windows",
                    };
                    let token = format!("{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        name, version_text(*version), j, dep.package, constraint_text(dep.constraint),
                        fs.join(","), gate, platform);
                    let source_path = active.get(name).map(|x| x.provenance.clone()).unwrap_or_default();
                    let mut path = source_path;
                    path.push(token);
                    let dest = active.entry(dep.package.clone()).or_insert_with(|| Active {
                        constraints: vec![], origins: vec![], features: BTreeSet::new(), provenance: vec![],
                    });
                    let before = (dest.constraints.len(), dest.origins.len(), dest.features.len());
                    dest.constraints.push(dep.constraint);
                    dest.origins.push(Origin { constraint: dep.constraint, path });
                    dest.features.extend(dep.features.iter().cloned());
                    if before != (dest.constraints.len(), dest.origins.len(), dest.features.len()) {
                        changed = true;
                    }
                }
            }

            // Compute canonical shortest provenance paths over active dependency edges.
            let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
            for (name, a) in &active {
                for origin in &a.origins {
                    if origin.path.len() == 1 && origin.path[0].starts_with("root[") {
                        let p = origin.path.clone();
                        if paths.get(name).map_or(true, |old| path_key(&p) < path_key(old)) {
                            paths.insert(name.clone(), p);
                        }
                    }
                }
            }
            loop {
                let mut any = false;
                for (src, version) in assigned {
                    let Some(base) = paths.get(src).cloned() else { continue };
                    let Some(pkg) = self.packages.get(src) else { continue };
                    let Some(rel) = pkg.releases.iter().find(|r| r.version == *version) else { continue };
                    let features = active.get(src).map(|a| &a.features);
                    for (j, dep) in rel.deps.iter().enumerate() {
                        if !target_matches(dep.platform, self.target)
                            || dep.gate.as_ref().map_or(false, |g| !features.map_or(false, |f| f.contains(g))) {
                            continue;
                        }
                        let mut fs = dep.features.clone();
                        fs.sort(); fs.dedup();
                        let token = format!("{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                            src, version_text(*version), j, dep.package, constraint_text(dep.constraint),
                            fs.join(","), dep.gate.as_deref().unwrap_or("*"),
                            match dep.platform { Platform::Any=>"any", Platform::Linux=>"linux", Platform::Mac=>"mac", Platform::Windows=>"windows" });
                        let mut candidate = base.clone(); candidate.push(token);
                        if paths.get(&dep.package).map_or(true, |old| path_key(&candidate) < path_key(old)) {
                            paths.insert(dep.package.clone(), candidate);
                            any = true;
                        }
                    }
                }
                if !any { break; }
            }
            for (name, path) in paths {
                if let Some(a) = active.get_mut(&name) {
                    if a.provenance != path { a.provenance = path; changed = true; }
                }
            }
            if !changed { break; }
        }

        for a in active.values_mut() {
            a.constraints = canonical_constraints(&a.constraints);
            a.origins.sort_by(|x, y| constraint_rank(x.constraint).cmp(&constraint_rank(y.constraint))
                .then_with(|| x.path.cmp(&y.path)));
        }
        active
    }

    fn evidence(a: &Active) -> Vec<Evidence> {
        a.origins.iter().map(|o| Evidence { constraint: o.constraint, path: o.path.clone() }).collect()
    }

    fn inspect(&mut self, roots: &[Requirement], assigned: &mut BTreeMap<String, Version>) -> Result<(), ()> {
        let active = self.closure(roots, assigned);

        if let Some(name) = active.keys().find(|n| !self.packages.contains_key(*n)).cloned() {
            let evidence = Self::evidence(&active[&name]);
            let err = ResolveError::Missing { package: name, evidence };
            if self.first_error.is_none() { self.first_error = Some(err); }
            return Ok(());
        }

        for (name, version) in assigned.iter() {
            let Some(a) = active.get(name) else { continue };
            if a.constraints.iter().any(|c| !matches(*version, *c)) {
                let err = ResolveError::Conflict {
                    package: name.clone(), selected: *version,
                    requirements: a.constraints.clone(), evidence: Self::evidence(a),
                };
                if self.first_error.is_none() { self.first_error = Some(err); }
                return Ok(());
            }
        }

        for (name, version) in assigned.iter() {
            let Some(a) = active.get(name) else { continue };
            let Some(pkg) = self.packages.get(name) else { continue };
            let rel = pkg.releases.iter().find(|r| r.version == *version).unwrap();
            for feature in &a.features {
                if !rel.features.iter().any(|f| f.name == *feature) {
                    let err = ResolveError::MissingFeature {
                        package: name.clone(), selected: *version, feature: feature.clone(),
                        path: a.provenance.clone(),
                    };
                    if self.first_error.is_none() { self.first_error = Some(err); }
                    return Ok(());
                }
            }
        }

        for name in active.keys().filter(|n| !assigned.contains_key(*n)) {
            let a = &active[name];
            let pkg = self.packages[name];
            if !pkg.releases.iter().any(|r| a.constraints.iter().all(|c| matches(r.version, *c))) {
                let err = ResolveError::NoMatch {
                    package: name.clone(), requirements: a.constraints.clone(),
                    evidence: Self::evidence(a),
                };
                if self.first_error.is_none() { self.first_error = Some(err); }
                return Ok(());
            }
        }

        let next = active.keys().find(|n| !assigned.contains_key(*n)).cloned();
        if let Some(name) = next {
            let a = &active[&name];
            let pkg = self.packages[&name];
            let mut candidates: Vec<Version> = pkg.releases.iter()
                .map(|r| r.version)
                .filter(|v| a.constraints.iter().all(|c| matches(*v, *c)))
                .collect();
            candidates.sort_by(|x, y| y.cmp(x));
            candidates.dedup();
            if let Some(lock) = self.locks.get(&name) {
                if let Some(i) = candidates.iter().position(|v| v == lock) {
                    let v = candidates.remove(i);
                    candidates.insert(0, v);
                }
            }
            let mut first_local_error = None;
            for candidate in candidates {
                if self.attempts == self.max_attempts {
                    return Err(());
                }
                self.attempts += 1;
                assigned.insert(name.clone(), candidate);
                let before_err = self.first_error.clone();
                self.inspect(roots, assigned)?;
                assigned.remove(&name);
                if self.first_error.is_none() {
                    let _ = before_err;
                }
            }
            return Ok(());
        }

        let cost = self.locks.iter().filter(|(p, v)| assigned.get(*p) != Some(*v)).count();
        let replace = match &self.best {
            None => true,
            Some((old_cost, old)) if cost < *old_cost => true,
            Some((old_cost, old)) if cost == *old_cost => {
                let names: BTreeSet<String> = self.packages.keys().cloned().collect();
                let mut better = false;
                for n in names {
                    match (assigned.get(&n), old.get(&n)) {
                        (Some(a), Some(b)) if a != b => { better = a > b; break; }
                        (Some(_), None) => { better = true; break; }
                        (None, Some(_)) => { better = false; break; }
                        _ => {}
                    }
                }
                better
            }
            _ => false,
        };
        if replace { self.best = Some((cost, assigned.clone())); }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]", i), name: root.package.clone() });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName { site: format!("root[{}].feature[{}]", i, f), name: feature.clone() });
            }
        }
    }
    for (i, pkg) in catalog.iter().enumerate() {
        if !valid_name(&pkg.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]", i), name: pkg.name.clone() });
        }
        for (j, rel) in pkg.releases.iter().enumerate() {
            for (k, dep) in rel.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in rel.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].feature[{}].enable[{}]", i, j, f, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for pkg in catalog {
        if !seen_packages.insert(pkg.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: pkg.name.clone() });
        }
    }
    for pkg in catalog {
        let mut versions = BTreeSet::new();
        for rel in &pkg.releases {
            if !versions.insert(rel.version) {
                return Err(ResolveError::DuplicateVersion { package: pkg.name.clone(), version: rel.version });
            }
        }
    }
    for pkg in catalog {
        for rel in &pkg.releases {
            let mut features = BTreeSet::new();
            for feature in &rel.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: pkg.name.clone(), version: rel.version, feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{}]", i), name: lock.package.clone() });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }

    let package_map: BTreeMap<String, &Package> = catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks.iter().map(|l| (l.package.clone(), l.version)).collect();
    let mut search = Search {
        packages: &package_map,
        locks: lock_map.clone(),
        target,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };
    let mut assigned = BTreeMap::new();
    if search.inspect(roots, &mut assigned).is_err() {
        return Err(ResolveError::Limit { attempts: max_attempts });
    }
    let Some((_, selected)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(), requirements: vec![], evidence: vec![],
        }));
    };

    let selected_vec: Vec<Selection> = selected.iter().map(|(package, version)| Selection {
        package: package.clone(), version: *version,
    }).collect();

    let final_active = search.closure(roots, &selected);
    let mut all_names: BTreeSet<String> = locks.iter().map(|l| l.package.clone()).collect();
    all_names.extend(selected.keys().cloned());
    let mut changes = Vec::new();
    for name in all_names {
        let before = lock_map.get(&name).copied();
        let after = selected.get(&name).copied();
        if before == after { continue; }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = package_map.get(&name).map_or(false, |p| p.releases.iter().any(|r| r.version == old));
                if !exists { ChangeReason::Unavailable }
                else if final_active.get(&name).map_or(false, |a| a.constraints.iter().any(|c| !matches(old, *c))) {
                    ChangeReason::Rejected
                } else { ChangeReason::Rebalanced }
            }
            _ => unreachable!(),
        };
        changes.push(Change { package: name, before, after, reason });
    }
    Ok(Resolution { selected: selected_vec, changes })
}
```
