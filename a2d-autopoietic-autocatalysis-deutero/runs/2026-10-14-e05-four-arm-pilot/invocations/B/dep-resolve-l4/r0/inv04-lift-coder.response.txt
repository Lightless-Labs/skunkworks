fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(values: &[Constraint]) -> Vec<Constraint> {
    let mut out = values.to_vec();
    out.sort();
    out.dedup();
    out
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || *b == b'.' || *b == b'_' || *b == b'-' {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn constraint_list(values: &[Constraint]) -> String {
    canonical_constraints(values)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn sorted_evidence(values: &[Evidence]) -> Vec<Evidence> {
    let mut out = values.to_vec();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    out
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraint_list(requirements)
            );
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraint_list(requirements)
            );
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn permits(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn matches_all(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| permits(*c, v))
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!(
                            "catalog[{}].release[{}].feature[{}]",
                            i, j, f
                        ),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }
    Ok(())
}

fn validate_catalog(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut names = std::collections::BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for package in catalog {
        for release in &package.releases {
            let mut features = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Origin {

    path: Vec<String>,

#[derive(Clone, Default)]
struct NodeInfo {
    origins: std::collections::BTreeMap<String, Vec<Origin>>,
    requests: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    active_features: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
}

fn dedup_sorted(values: &[String]) -> Vec<String> {
    let mut result = values.to_vec();
    result.sort();
    result.dedup();
    result
}

fn make_root_token(index: usize, r: &Requirement) -> String {
    let features = dedup_sorted(&r.features).join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        r.package,
        constraint_text(r.constraint),
        features
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let features = dedup_sorted(&dep.features).join(",");
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features,
        gate,
        platform
    )
}

fn update_path(paths: &mut std::collections::BTreeMap<String, Vec<String>>, package: &str, path: Vec<String>) -> bool {
    match paths.get(package) {
        None => {
            paths.insert(package.to_string(), path);
            true
        }
        Some(old) if (path.len(), &path) < (old.len(), old) => {
            paths.insert(package.to_string(), path);
            true
        }
        _ => false,
    }
}

fn compute_info(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> NodeInfo {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut info = NodeInfo::default();

    for (i, root) in roots.iter().enumerate() {
        let token = make_root_token(i, root);
        info.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        info.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        update_path(&mut info.paths, &root.package, vec![token]);
    }

    loop {
        let mut changed = false;
        let packages: Vec<String> = selected.keys().cloned().collect();
        for source in packages {
            let Some(&version) = selected.get(&source) else {
                continue;
            };
            let Some(package) = by_name.get(source.as_str()).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requests = info.requests.entry(source.clone()).or_default().clone();
            let mut activated = info.active_features.entry(source.clone()).or_default().clone();
            let mut queue: Vec<String> = requests.iter().cloned().collect();
            while let Some(feature_name) = queue.pop() {
                if !activated.insert(feature_name.clone()) {
                    continue;
                }
                if let Some(declaration) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &declaration.enables {
                        if !activated.contains(enabled) {
                            queue.push(enabled.clone());
                        }
                    }
                }
            }
            if info.active_features.get(&source) != Some(&activated) {
                info.active_features.insert(source.clone(), activated.clone());
                changed = true;
            }

            for (dep_index, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }
                let path = info
                    .paths
                    .get(&source)
                    .cloned()
                    .unwrap_or_default();
                let mut dep_path = path;
                dep_path.push(dep_token(&source, version, dep_index, dep));

                let entry = info.origins.entry(dep.package.clone()).or_default();
                if !entry.iter().any(|o| o.constraint == dep.constraint && o.path == dep_path) {
                    entry.push(Origin {
                        constraint: dep.constraint,
                        path: dep_path.clone(),
                    });
                    changed = true;
                }
                if update_path(&mut info.paths, &dep.package, dep_path) {
                    changed = true;
                }
                let req = info.requests.entry(dep.package.clone()).or_default();
                for f in &dep.features {
                    if req.insert(f.clone()) {
                        changed = true;
                    }
                }
            }
        }
        if !changed {
            break;
        }
    }
    info
}

fn canonical_origins(origins: &[Origin]) -> Vec<Evidence> {
    let mut result: Vec<Evidence> = origins
        .iter()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    result.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    result
}

fn requirements(info: &NodeInfo, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        &info
            .origins
            .get(package)
            .map(|v| v.iter().map(|o| o.constraint).collect::<Vec<_>>())
            .unwrap_or_default(),
    )
}

fn active_names(info: &NodeInfo) -> Vec<String> {
    info.origins.keys().cloned().collect()
}

fn node_error(
    catalog: &[Package],
    selected: &std::collections::BTreeMap<String, Version>,
    info: &NodeInfo,
) -> Option<ResolveError> {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let active = active_names(info);

    for name in &active {
        if !by_name.contains_key(name.as_str()) {
            return Some(ResolveError::Missing {
                package: name.clone(),
                evidence: canonical_origins(info.origins.get(name).map(Vec::as_slice).unwrap_or(&[])),
            });
        }
    }

    for name in selected.keys() {
        let cs = requirements(info, name);
        if let Some(version) = selected.get(name) {
            if !matches_all(*version, &cs) {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: cs,
                    evidence: canonical_origins(info.origins.get(name).map(Vec::as_slice).unwrap_or(&[])),
                });
            }
            let package = by_name.get(name.as_str()).copied();
            if let Some(package) = package {
                let release = package.releases.iter().find(|r| r.version == *version);
                if let Some(release) = release {
                    let features = info
                        .active_features
                        .get(name)
                        .cloned()
                        .unwrap_or_default();
                    for feature in features {
                        if !release.features.iter().any(|f| f.name == feature) {
                            return Some(ResolveError::MissingFeature {
                                package: name.clone(),
                                selected: *version,
                                feature,
                                path: info.paths.get(name).cloned().unwrap_or_default(),
                            });
                        }
                    }
                }
            }
        }
    }

    for name in active {
        if selected.contains_key(&name) {
            continue;
        }
        let cs = requirements(info, &name);
        let has_match = by_name
            .get(name.as_str())
            .map(|p| p.releases.iter().any(|r| matches_all(r.version, &cs)))
            .unwrap_or(false);
        if !has_match {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: cs,
                evidence: canonical_origins(info.origins.get(&name).map(Vec::as_slice).unwrap_or(&[])),
            });
        }
    }

    None
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>, NodeInfo)>,
    first_failure: Option<ResolveError>,
}

impl Search<'_> {
    fn lock_cost(&self, selected: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn better(
        &self,
        cost: usize,
        selected: &std::collections::BTreeMap<String, Version>,
        best: &(usize, Vec<Selection>, NodeInfo),
    ) -> bool {
        if cost != best.0 {
            return cost < best.0;
        }
        let mut names: Vec<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
        names.sort();
        for name in names {
            let a = selected.get(&name);
            let b = best.1.iter().find(|s| s.package == name).map(|s| &s.version);
            match (a, b) {
                (None, None) => {}
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn visit(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let info = compute_info(self.catalog, self.roots, selected, self.target);
        if let Some(error) = node_error(self.catalog, selected, &info) {
            if self.first_failure.is_none() {
                self.first_failure = Some(error);
            }
            return Ok(());
        }

        let mut branch: Option<String> = None;
        for name in info.origins.keys() {
            if !selected.contains_key(name) {
                branch = Some(name.clone());
                break;
            }
        }
        let Some(package_name) = branch else {
            let selections: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self.lock_cost(selected);
            let candidate = (cost, selections, info);
            if self
                .best
                .as_ref()
                .is_none_or(|best| self.better(cost, selected, best))
            {
                self.best = Some(candidate);
            }
            return Ok(());
        };

        let constraints = requirements(&info, &package_name);
        let package = self
            .catalog
            .iter()
            .find(|p| p.name == package_name)
            .expect("active package was validated");
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches_all(*v, &constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|l| l.package == package_name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), version);
            self.visit(selected)?;
            selected.remove(&package_name);
        }
        Ok(())
    }
}

fn build_changes(
    catalog: &[Package],
    locks: &[Lock],
    selected: &[Selection],
    final_info: &NodeInfo,
) -> Vec<Change> {
    let lock_map: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|l| (l.package.as_str(), l.version))
        .collect();
    let selected_map: std::collections::BTreeMap<&str, Version> = selected
        .iter()
        .map(|s| (s.package.as_str(), s.version))
        .collect();
    let mut names = std::collections::BTreeSet::new();
    names.extend(lock_map.keys().copied());
    names.extend(selected_map.keys().copied());

    let mut changes = Vec::new();
    for name in names {
        let before = lock_map.get(name).copied();
        let after = selected_map.get(name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_new)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == name)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !matches_all(old, &requirements(final_info, name)) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name.to_string(),
            before,
            after,
            reason,
        });
    }
    changes
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;
    validate_catalog(catalog)?;

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_failure: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    search.visit(&mut selected)?;

    if let Some((_, selected, info)) = search.best {
        let changes = build_changes(catalog, locks, &selected, &info);
        Ok(Resolution { selected, changes })
    } else {
        Err(search.first_failure.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }))
    }
}
