fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| match (*a, *b) {
            (Constraint::Exact(x), Constraint::Exact(y))
            | (Constraint::AtLeast(x), Constraint::AtLeast(y))
            | (Constraint::Before(x), Constraint::Before(y))
            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(&y),
            _ => std::cmp::Ordering::Equal,
        })
}

fn canonical_constraints(values: &[Constraint]) -> Vec<Constraint> {
    let mut out = values.to_vec();
    out.sort_by(constraint_cmp);
    out.dedup();
    out
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut result = evidence.to_vec();
    result.sort_by(|a, b| {
        constraint_cmp(&a.constraint, &b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    result
}

fn append_evidence(out: &mut String, evidence: &[Evidence]) {
    for e in sorted_evidence(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut out = format!("missing|package={}", encode(package));
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn valid_name(s: &str) -> bool {
    let bytes = s.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes.last() != Some(&b'-')
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn satisfies(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x
                && if x.major > 0 {
                    version.major == x.major
                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x
                }
        }
    }
}

fn matches_all(version: Version, requirements: &[Constraint]) -> bool {
    requirements.iter().all(|c| satisfies(version, *c))
}

fn target_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn target_text(platform: Platform) -> &'static str {
    match platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn sorted_features(features: &[String]) -> Vec<String> {
    let mut result = features.to_vec();
    result.sort();
    result.dedup();
    result
}

fn feature_text(features: &[String]) -> String {
    sorted_features(features).join(",")
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut names = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !names.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    Ok(())
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct NodeState {
    constraints: std::collections::BTreeMap<String, Vec<Constraint>>,
    origins: std::collections::BTreeMap<String, Vec<Origin>>,
    features: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
    active: std::collections::BTreeSet<String>,
}

fn canonical_path_better(candidate: &[String], current: &[String]) -> bool {
    candidate.len() < current.len()
        || (candidate.len() == current.len() && candidate < current)
}

fn add_origin(state: &mut NodeState, package: &str, constraint: Constraint, path: Vec<String>) {
    state.active.insert(package.to_string());
    state
        .constraints
        .entry(package.to_string())
        .or_default()
        .push(constraint);
    state
        .origins
        .entry(package.to_string())
        .or_default()
        .push(Origin { constraint, path: path.clone() });
    match state.paths.get(package) {
        Some(current) if !canonical_path_better(&path, current) => {}
        _ => {
            state.paths.insert(package.to_string(), path);
        }
    }
}

fn build_state(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> NodeState {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut state = NodeState::default();

    for (i, root) in roots.iter().enumerate() {
        let features = feature_text(&root.features);
        let token = format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            root.package,
            constraint_text(root.constraint),
            features
        );
        add_origin(
            &mut state,
            &root.package,
            root.constraint,
            vec![token],
        );
        state
            .features
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
    }

    let mut processed_features: std::collections::BTreeMap<String, std::collections::BTreeSet<String>> =
        std::collections::BTreeMap::new();

    loop {
        let mut changed = false;
        let names: Vec<String> = selected.keys().cloned().collect();
        for source in names {
            let version = selected[&source];
            let Some(package) = by_name.get(source.as_str()).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = state.features.entry(source.clone()).or_default().clone();
            let done = processed_features.entry(source.clone()).or_default();
            for feature_name in requested {
                if !done.insert(feature_name.clone()) {
                    continue;
                }
                if let Some(declaration) = release.features.iter().find(|f| f.name == feature_name) {
                    state
                        .features
                        .entry(source.clone())
                        .or_default()
                        .extend(declaration.enables.iter().cloned());
                    changed = true;
                }
            }

            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target) {
                    continue;
                }
                if let Some(gate) = &dep.gate {
                    if !state
                        .features
                        .get(&source)
                        .is_some_and(|set| set.contains(gate))
                    {
                        continue;
                    }
                }

                let Some(source_path) = state.paths.get(&source).cloned() else {
                    continue;
                };
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    source,
                    version_text(version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    feature_text(&dep.features),
                    dep.gate.as_deref().unwrap_or("*"),
                    target_text(dep.platform)
                );
                let mut path = source_path;
                path.push(token);
                let existed = state.origins.get(&dep.package).is_some_and(|origins| {
                    origins.iter().any(|o| o.path == path && o.constraint == dep.constraint)
                });
                if !existed {
                    add_origin(&mut state, &dep.package, dep.constraint, path);
                    changed = true;
                }
                let dest_features = state.features.entry(dep.package.clone()).or_default();
                for feature in &dep.features {
                    if dest_features.insert(feature.clone()) {
                        changed = true;
                    }
                }
            }
        }
        if !changed {
            break;
        }
    }
    state
}

fn error_evidence(state: &NodeState, package: &str) -> Vec<Evidence> {
    let mut result: Vec<Evidence> = state
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|origin| Evidence {
            constraint: origin.constraint,
            path: origin.path.clone(),
        })
        .collect();
    result.sort_by(|a, b| {
        constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
    });
    result
}

fn requirements(state: &NodeState, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        state
            .constraints
            .get(package)
            .map(Vec::as_slice)
            .unwrap_or(&[]),
    )
}

fn dynamic_error(
    catalog: &[Package],
    selected: &std::collections::BTreeMap<String, Version>,
    state: &NodeState,
) -> Option<ResolveError> {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();

    for package in &state.active {
        if !by_name.contains_key(package.as_str()) {
            return Some(ResolveError::Missing {
                package: package.clone(),
                evidence: error_evidence(state, package),
            });
        }
    }

    for (package_name, version) in selected {
        let Some(cs) = state.constraints.get(package_name) else {
            continue;
        };
        if !matches_all(*version, cs) {
            return Some(ResolveError::Conflict {
                package: package_name.clone(),
                selected: *version,
                requirements: requirements(state, package_name),
                evidence: error_evidence(state, package_name),
            });
        }
    }

    for package_name in selected.keys() {
        let Some(package) = by_name.get(package_name.as_str()) else {
            continue;
        };
        let version = selected[package_name];
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        let activated = state.features.get(package_name);
        if let Some(activated) = activated {
            for feature in activated {
                if !release.features.iter().any(|f| &f.name == feature) {
                    return Some(ResolveError::MissingFeature {
                        package: package_name.clone(),
                        selected: version,
                        feature: feature.clone(),
                        path: state.paths.get(package_name).cloned().unwrap_or_default(),
                    });
                }
            }
        }
    }

    for package_name in &state.active {
        if selected.contains_key(package_name) {
            continue;
        }
        let package = by_name[package_name.as_str()];
        let cs = state.constraints.get(package_name).map(Vec::as_slice).unwrap_or(&[]);
        if !package.releases.iter().any(|release| matches_all(release.version, cs)) {
            return Some(ResolveError::NoMatch {
                package: package_name.clone(),
                requirements: requirements(state, package_name),
                evidence: error_evidence(state, package_name),
            });
        }
    }
    None
}

fn compare_vectors(
    a: &std::collections::BTreeMap<String, Version>,
    b: &std::collections::BTreeMap<String, Version>,
) -> std::cmp::Ordering {
    let names: std::collections::BTreeSet<String> =
        a.keys().chain(b.keys()).cloned().collect();
    for name in names {
        match (a.get(&name), b.get(&name)) {
            (Some(x), Some(y)) => {
                let order = x.cmp(y);
                if order != std::cmp::Ordering::Equal {
                    return order;
                }
            }
            (Some(_), None) => return std::cmp::Ordering::Greater,
            (None, Some(_)) => return std::cmp::Ordering::Less,
            (None, None) => {}
        }
    }
    std::cmp::Ordering::Equal
}

struct SearchResult {
    best: Option<std::collections::BTreeMap<String, Version>>,
    first_error: Option<ResolveError>,
    best_cost: usize,
}

fn lock_cost(
    selected: &std::collections::BTreeMap<String, Version>,
    locks: &[Lock],
) -> usize {
    locks
        .iter()
        .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
        .count()
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    budget: usize,
    attempts: &mut usize,
    selected: &mut std::collections::BTreeMap<String, Version>,
    result: &mut SearchResult,
) -> Result<(), ResolveError> {
    let state = build_state(catalog, roots, selected, target);
    if let Some(error) = dynamic_error(catalog, selected, &state) {
        if result.first_error.is_none() {
            result.first_error = Some(error);
        }
        return Ok(());
    }

    let next = state
        .active
        .iter()
        .find(|name| !selected.contains_key(*name))
        .cloned();
    let Some(package_name) = next else {
        let cost = lock_cost(selected, locks);
        let replace = match &result.best {
            None => true,
            Some(best) => {
                cost < result.best_cost
                    || (cost == result.best_cost
                        && compare_vectors(selected, best) == std::cmp::Ordering::Greater)
            }
        };
        if replace {
            result.best = Some(selected.clone());
            result.best_cost = cost;
        }
        return Ok(());
    };

    let package = catalog
        .iter()
        .find(|p| p.name == package_name)
        .expect("active package passed missing check");
    let cs = state.constraints.get(&package_name).map(Vec::as_slice).unwrap_or(&[]);
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|r| matches_all(r.version, cs))
        .map(|r| r.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(lock) = locks.iter().find(|l| l.package == package_name) {
        if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
            let locked = candidates.remove(index);
            candidates.insert(0, locked);
        }
    }

    for candidate in candidates {
        if *attempts == budget {
            return Err(ResolveError::Limit { attempts: budget });
        }
        *attempts += 1;
        selected.insert(package_name.clone(), candidate);
        let child = search(
            catalog, roots, locks, target, budget, attempts, selected, result,
        );
        selected.remove(&package_name);
        child?;
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let mut attempts = 0usize;
    let mut selected = std::collections::BTreeMap::new();
    let mut search_result = SearchResult {
        best: None,
        first_error: None,
        best_cost: usize::MAX,
    };
    search(
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        &mut attempts,
        &mut selected,
        &mut search_result,
    )?;

    let Some(best) = search_result.best else {
        return Err(search_result.first_error.unwrap_or_else(|| {
            ResolveError::NoMatch {
                package: String::new(),
                requirements: Vec::new(),
                evidence: Vec::new(),
            }
        }));
    };

    let final_state = build_state(catalog, roots, &best, target);
    let selected_out: Vec<Selection> = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut before: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let mut after = best.clone();
    let names: std::collections::BTreeSet<String> =
        before.keys().chain(after.keys()).cloned().collect();
    let mut changes = Vec::new();
    for package in names {
        let b = before.remove(&package);
        let a = after.remove(&package);
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !matches_all(
                    old,
                    final_state
                        .constraints
                        .get(&package)
                        .map(Vec::as_slice)
                        .unwrap_or(&[]),
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
