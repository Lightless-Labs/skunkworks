use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, at: usize) -> Option<u8> {
    s.as_bytes().get(at).copied()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while byte_at(s, *pos).is_some_and(|b| b.is_ascii_digit()) {
        *pos += 1;
    }
    if *pos == start {
        return match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn is_identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            if !is_identifier_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let identifier = &s[ident_start..i];
        if prerelease
            && identifier.len() > 1
            && identifier.starts_with('0')
            && identifier.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        result.push(identifier.to_owned());
        if i == end {
            break;
        }
        ident_start = i + 1;
    }

    Ok(result)
}

/// Parses a full semantic version string.
pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        match byte_at(s, pos) {
            Some(b'.') => pos += 1,
            None | Some(b'-' | b'+') => {
                return Err(VersionError::MissingCore { at: pos });
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        if _ == 0 {
            break;
        }
    }

    // Parse the remaining two core fields explicitly to preserve left-to-right errors.
    let minor = parse_core_after_major(s, &mut pos)?;
    let patch = parse_core_after_major(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();

    match byte_at(s, pos) {
        Some(b'-') => {
            let pre_start = pos + 1;
            let build_sep = s[pre_start..].find('+').map(|i| pre_start + i);
            let pre_end = build_sep.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            pos = pre_end;
            if byte_at(s, pos) == Some(b'+') {
                let build_start = pos + 1;
                build = parse_identifiers(s, build_start, s.len(), false)?;
                pos = s.len();
            }
        }
        Some(b'+') => {
            let build_start = pos + 1;
            build = parse_identifiers(s, build_start, s.len(), false)?;
            pos = s.len();
        }
        None => {}
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn parse_core_after_major(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let value = parse_core(s, pos)?;
    if *pos == s.len() {
        return Err(VersionError::MissingCore { at: *pos });
    }

    // The caller needs a dot after this field only when another core field follows.
    Ok(value)
}

/// Compares versions by semantic precedence, ignoring build metadata.
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (left, right) in a.iter().zip(b.iter()) {
        let left_numeric = left.bytes().all(|b| b.is_ascii_digit());
        let right_numeric = right.bytes().all(|b| b.is_ascii_digit());
        let ordering = match (left_numeric, right_numeric) {
            (true, true) => left.len().cmp(&right.len()).then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };
        if ordering != Ordering::Equal {
            return ordering;
        }
    }
    a.len().cmp(&b.len())
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

/// Parses a conjunction of full-version comparators.
pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut i = 0;
    let mut comparators = Vec::new();

    while i < bytes.len() {
        while i < bytes.len() && is_range_space(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let token_start = i;
        while i < bytes.len() && !is_range_space(bytes[i]) {
            i += 1;
        }
        let token_end = i;
        let token = &s[token_start..token_end];

        let (op, operand_offset) = if token.starts_with(">=") {
            (Op::Ge, 2)
        } else if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with('>') {
            (Op::Gt, 1)
        } else if token.starts_with('<') {
            (Op::Lt, 1)
        } else if token.starts_with('=') {
            (Op::Eq, 1)
        } else if token.as_bytes().first().is_some_and(u8::is_ascii_digit) {
            (Op::Eq, 0)
        } else {
            return Err(RangeError::Syntax { at: token_start });
        };

        let operand_start = token_start + operand_offset;
        let operand = &s[operand_offset + token_start - token_start..token.len()];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { comparators })
}

/// Tests whether a version satisfies every comparator in a range.
pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r.comparators.iter().any(|c| {
            !c.version.pre.is_empty()
                && (v.major, v.minor, v.patch)
                    == (c.version.major, c.version.minor, c.version.patch)
        })
    {
        return false;
    }

    r.comparators.iter().all(|c| {
        let ordering = compare_versions(v, &c.version);
        match c.op {
            Op::Eq => ordering == Ordering::Equal,
            Op::Lt => ordering == Ordering::Less,
            Op::Le => ordering != Ordering::Greater,
            Op::Gt => ordering == Ordering::Greater,
            Op::Ge => ordering != Ordering::Less,
        }
    })
}

fn parse_core_after_major(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let value = parse_core(s, pos)?;
    Ok(value)
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    #[test]
    fn parses_and_compares_versions() {
        let version = pv("2.1.3-rc.1+build.7");
        assert_eq!((version.major, version.minor, version.patch), (2, 1, 3));
        assert_eq!(version.pre, vec!["rc", "1"]);
        assert_eq!(version.build, vec!["build", "7"]);
        assert_eq!(compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")), Ordering::Less);
    }

    #[test]
    fn rejects_invalid_core_and_identifiers() {
        assert_eq!(parse_version("01.0.0"), Err(VersionError::LeadingZero { at: 0 }));
        assert_eq!(parse_version("1.0"), Err(VersionError::MissingCore { at: 3 }));
        assert!(matches!(parse_version("1.0.0-01"), Err(VersionError::LeadingZero { at: 6 })));
    }

    #[test]
    fn range_matching_and_prerelease_admission() {
        let range = parse_range(">=2.1.3-beta <2.1.3").unwrap();
        assert!(matches(&pv("2.1.3-rc"), &range));
        assert!(!matches(&pv("2.1.4-rc"), &range));
        assert!(matches(&pv("2.1.3"), &parse_range("=2.1.3+other").unwrap()));
    }
}
