    fn tag_is_free(&self, candidate: usize, running: &[usize]) -> bool {
        let Some(tag) = self.entries[candidate].job.tag.as_ref() else {
            return true;
        };
        running
            .iter()
            .all(|&i| self.entries[i].job.tag.as_ref() != Some(tag))
    }
}

fn valid_text(s: &str, min: usize, max: usize) -> bool {
    let bytes = s.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes.iter().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn running_count(entries: &[Entry]) -> usize {
    entries
        .iter()
        .filter(|e| matches!(e.state, State::Running { .. }))
        .count()
}

fn running_indices(entries: &[Entry], excluding: Option<usize>) -> Vec<usize> {
    entries
        .iter()
        .enumerate()
        .filter_map(|(i, e)| {
            (Some(i) != excluding && matches!(e.state, State::Running { .. })).then_some(i)
        })
        .collect()
}
