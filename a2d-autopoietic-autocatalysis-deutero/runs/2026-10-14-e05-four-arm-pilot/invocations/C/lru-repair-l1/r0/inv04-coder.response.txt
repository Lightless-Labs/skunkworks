use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BackwardTime { last: u64, now: u64 },
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    DeadlineOverflow,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub expires_at: Option<u64>,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacity: u64,
    used: u64,
    clock: u64,
    entries: BTreeMap<u64, Entry>,
    order: VecDeque<u64>,
}

impl Cache {
    /// Creates an empty cache with the given weighted capacity.
    pub fn new(capacity: u64) -> Self {
        Self {
            capacity,
            used: 0,
            clock: 0,
            entries: BTreeMap::new(),
            order: VecDeque::new(),
        }
    }

    /// Returns the cache's current capacity.
    pub fn capacity(&self) -> u64 {
        self.capacity
    }

    /// Returns the total weight of resident entries.
    pub fn used(&self) -> u64 {
        self.used
    }

    /// Returns the current logical clock.
    pub fn clock(&self) -> u64 {
        self.clock
    }

    /// Returns cloned resident entries from least to most recently used.
    pub fn snapshot(&self) -> Vec<Entry> {
        self.order
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect()
    }

    fn validate_time(&self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::BackwardTime {
                last: self.clock,
                now,
            });
        }
        Ok(())
    }

    fn make_deadline(ttl: Option<u64>, now: u64) -> Result<Option<u64>, Error> {
        match ttl {
            Some(duration) => now
                .checked_add(duration)
                .map(Some)
                .ok_or(Error::DeadlineOverflow),
            None => Ok(None),
        }
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.used -= entry.weight;
        self.order.retain(|&existing| existing != key);
        Some(entry)
    }

    fn touch(&mut self, key: u64) {
        self.order.retain(|&existing| existing != key);
        self.order.push_back(key);
    }

    fn expire(&mut self, now: u64) -> Vec<u64> {
        // BTreeMap iteration visits keys in ascending numeric order.
        let expired: Vec<u64> = self
            .entries
            .iter()
            .filter_map(|(&key, entry)| {
                entry.expires_at.filter(|&deadline| deadline <= now).map(|_| key)
            })
            .collect();

        for &key in &expired {
            self.erase(key);
        }
        self.clock = now;
        expired
    }

    /// Purges expired entries at `now` and returns their keys in ascending order.
    pub fn advance(&mut self, now: u64) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        Ok(self.expire(now))
    }

    /// Looks up a key at `now`, updating its recency if it is present.
    pub fn get(&mut self, key: u64, now: u64) -> Result<Option<i64>, Error> {
        self.validate_time(now)?;
        self.expire(now);

        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.touch(key);
        }
        Ok(value)
    }

    /// Removes a key at `now` and returns its value if it was resident.
    pub fn remove(&mut self, key: u64, now: u64) -> Result<Option<i64>, Error> {
        self.validate_time(now)?;
        self.expire(now);
        Ok(self.erase(key).map(|entry| entry.value))
    }

    /// Inserts or replaces a weighted entry, returning expired and evicted keys.
    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        ttl: Option<u64>,
        now: u64,
    ) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > self.capacity {
            return Err(Error::TooHeavy {
                weight,
                capacity: self.capacity,
            });
        }
        let expires_at = Self::make_deadline(ttl, now)?;

        let mut removed = self.expire(now);
        self.erase(key);

        if ttl == Some(0) {
            return Ok(removed);
        }

        // Compare against remaining capacity to avoid overflowing `used + weight`.
        let available = self.capacity - weight;
        while self.used > available {
            let victim = self.order.front().copied().expect("used weight requires an entry");
            self.erase(victim);
            removed.push(victim);
        }

        self.used += weight;
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                weight,
                expires_at,
            },
        );
        self.touch(key);
        Ok(removed)
    }

    /// Changes the capacity at `now`, returning expired and evicted keys.
    pub fn resize(&mut self, capacity: u64, now: u64) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        let mut removed = self.expire(now);
        self.capacity = capacity;

        while self.used > capacity {
            let key = self.order.front().copied().expect("used weight requires an entry");
            self.erase(key);
            removed.push(key);
        }
        Ok(removed)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn weighted_lru_insert_and_get() {
        let mut cache = Cache::new(5);
        assert_eq!(cache.insert(1, 10, 2, None, 1), Ok(vec![]));
        assert_eq!(cache.insert(2, 20, 2, None, 2), Ok(vec![]));
        assert_eq!(cache.get(1, 3), Ok(Some(10)));
        assert_eq!(cache.insert(3, 30, 2, None, 4), Ok(vec![2]));
        assert_eq!(
            cache.snapshot().iter().map(|entry| entry.key).collect::<Vec<_>>(),
            vec![1, 3]
        );
        assert_eq!(cache.used(), 4);
    }

    #[test]
    fn expiry_includes_deadline_and_is_key_ordered() {
        let mut cache = Cache::new(10);
        cache.insert(9, 90, 1, Some(2), 1).unwrap();
        cache.insert(3, 30, 1, Some(1), 1).unwrap();
        assert_eq!(cache.advance(2), Ok(vec![3]));
        assert_eq!(cache.advance(3), Ok(vec![9]));
        assert!(cache.snapshot().is_empty());
    }

    #[test]
    fn validation_errors_leave_state_unchanged() {
        let mut cache = Cache::new(3);
        cache.insert(1, 1, 2, Some(1), 1).unwrap();
        let before = cache.clone();

        assert_eq!(cache.insert(2, 2, 0, None, 2), Err(Error::ZeroWeight));
        assert_eq!(cache, before);
        assert_eq!(
            cache.insert(2, 2, 4, None, 2),
            Err(Error::TooHeavy {
                weight: 4,
                capacity: 3
            })
        );
        assert_eq!(cache, before);
    }

    #[test]
    fn replacement_zero_ttl_resize_and_full_width_weights() {
        let mut cache = Cache::new(u64::MAX);
        cache.insert(1, 10, u64::MAX, None, 0).unwrap();
        cache.insert(1, 11, u64::MAX, None, 0).unwrap();
        assert_eq!(cache.used(), u64::MAX);
        assert_eq!(cache.insert(1, 12, 1, Some(0), 0), Ok(vec![]));
        assert_eq!(cache.used(), 0);
        assert_eq!(cache.insert(2, 20, u64::MAX, None, 0), Ok(vec![]));
        assert_eq!(cache.resize(0, 0), Ok(vec![2]));
        assert_eq!(cache.used(), 0);
    }
}
