#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_rank(constraint: &Constraint) -> u8 {
    match constraint {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn canonical_constraints(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut result = constraints.to_vec();
    result.sort_by(|a, b| {
        constraint_rank(a)
            .cmp(&constraint_rank(b))
            .then_with(|| match (a, b) {
                (Constraint::Exact(x), Constraint::Exact(y))
                | (Constraint::AtLeast(x), Constraint::AtLeast(y))
                | (Constraint::Before(x), Constraint::Before(y))
                | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(y),
                _ => std::cmp::Ordering::Equal,
            })
    });
    result.dedup();
    result
}

fn constraint_text(constraint: &Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(*v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(*v)),
        Constraint::Before(v) => format!("<{}", version_text(*v)),
        Constraint::Caret(v) => format!("^{}", version_text(*v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints)
        .iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut out = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{:02X}", byte));
        }
    }
    out
}

/// Formats a resolution error as its stable, percent-encoded text representation.
pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn permits(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Assignments = std::collections::BTreeMap<String, Version>;
type Active = std::collections::BTreeMap<String, Vec<Constraint>>;

fn active_constraints(roots: &[Requirement], assignments: &Assignments, catalog: &[Package]) -> Active {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut active = Active::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assignments {
        if let Some(package) = by_name.get(name.as_str()) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    active
}

fn canonical_active(mut active: Active) -> Active {
    for constraints in active.values_mut() {
        *constraints = canonical_constraints(constraints);
    }
    active
}

fn dynamic_error(
    catalog: &[Package],
    assignments: &Assignments,
    active: &Active,
) -> Option<ResolveError> {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();

    for name in active.keys() {
        if !by_name.contains_key(name.as_str()) {
            return Some(ResolveError::Missing {
                package: name.clone(),
            });
        }
    }

    for (name, selected) in assignments {
        if let Some(constraints) = active.get(name) {
            if constraints.iter().any(|c| !permits(*c, *selected)) {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *selected,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in active {
        if assignments.contains_key(name) {
            continue;
        }
        let package = by_name[name.as_str()];
        if !package.releases.iter().any(|release| {
            constraints
                .iter()
                .all(|constraint| permits(*constraint, release.version))
        }) {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }
    None
}

struct SearchResult {
    best: Option<(usize, Vec<Option<Version>>, Assignments)>,
    first_error: Option<ResolveError>,
}

fn vector_for(catalog: &[Package], assignments: &Assignments) -> Vec<Option<Version>> {
    let mut names: Vec<&str> = catalog.iter().map(|p| p.name.as_str()).collect();
    names.sort_unstable();
    names
        .into_iter()
        .map(|name| assignments.get(name).copied())
        .collect()
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &std::collections::BTreeMap<String, Version>,
    assignments: &mut Assignments,
    attempts: &mut usize,
    max_attempts: usize,
    result: &mut SearchResult,
) -> Result<(), ResolveError> {
    let active = canonical_active(active_constraints(roots, assignments, catalog));
    if let Some(error) = dynamic_error(catalog, assignments, &active) {
        if result.first_error.is_none() {
            result.first_error = Some(error);
        }
        return Ok(());
    }

    let next = active.keys().find(|name| !assignments.contains_key(*name)).cloned();
    let Some(package_name) = next else {
        let cost = locks
            .iter()
            .filter(|(name, version)| assignments.get(*name) != Some(*version))
            .count();
        let vector = vector_for(catalog, assignments);
        let replace = match &result.best {
            None => true,
            Some((best_cost, best_vector, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            result.best = Some((cost, vector, assignments.clone()));
        }
        return Ok(());
    };

    let package = catalog.iter().find(|p| p.name == package_name).unwrap();
    let constraints = &active[&package_name];
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|release| {
            constraints
                .iter()
                .all(|constraint| permits(*constraint, release.version))
        })
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(locked) = locks.get(&package_name) {
        if let Some(position) = candidates.iter().position(|version| version == locked) {
            let version = candidates.remove(position);
            candidates.insert(0, version);
        }
    }

    for candidate in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        assignments.insert(package_name.clone(), candidate);
        let branch_result = search(
            catalog,
            roots,
            locks,
            assignments,
            attempts,
            max_attempts,
            result,
        );
        assignments.remove(&package_name);
        branch_result?;
    }
    Ok(())
}

/// Resolves root requirements against a catalog, using locks for search ordering and reconciliation.
pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let lock_map: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let mut assignments = Assignments::new();
    let mut attempts = 0usize;
    let mut result = SearchResult {
        best: None,
        first_error: None,
    };

    search(
        catalog,
        roots,
        &lock_map,
        &mut assignments,
        &mut attempts,
        max_attempts,
        &mut result,
    )?;

    let Some((_, _, chosen)) = result.best else {
        return Err(result.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = canonical_active(active_constraints(roots, &chosen, catalog));
    let selected_map = &chosen;
    let mut names = std::collections::BTreeSet::new();
    names.extend(lock_map.keys().cloned());
    names.extend(selected_map.keys().cloned());

    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut changes = Vec::new();
    for name in names {
        let before = lock_map.get(&name).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = by_name
                    .get(name.as_str())
                    .is_some_and(|package| package.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&name)
                    .is_some_and(|constraints| constraints.iter().any(|c| !permits(*c, old)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

#[cfg(test)]
mod tests {
    use super::*;

    fn v(major: u32, minor: u32, patch: u32) -> Version {
        Version { major, minor, patch }
    }

    fn x(n: u32) -> Version {
        v(n, 0, 0)
    }

    fn q(package: &str, constraint: Constraint) -> Requirement {
        Requirement {
            package: package.into(),
            constraint,
        }
    }

    fn release(n: u32, deps: Vec<Requirement>) -> Release {
        Release { version: x(n), deps }
    }

    fn package(name: &str, releases: Vec<Release>) -> Package {
        Package {
            name: name.into(),
            releases,
        }
    }

    #[test]
    fn resolves_transitive_dependencies_in_name_order() {
        let catalog = vec![
            package("engine", vec![release(1, vec![q("util", Constraint::AtLeast(x(1)))])]),
            package("util", vec![release(1, vec![])]),
        ];
        let resolution = resolve(
            &catalog,
            &[q("engine", Constraint::Exact(x(1)))],
            &[],
            10,
        )
        .unwrap();
        assert_eq!(resolution.selected.len(), 2);
        assert_eq!(resolution.selected[0].package, "engine");
        assert_eq!(resolution.selected[1].package, "util");
    }

    #[test]
    fn invalid_names_are_reported_before_resolution() {
        let error = resolve(&[], &[q("Bad Name", Constraint::Exact(x(1)))], &[], 0).unwrap_err();
        assert_eq!(
            error_text(&error),
            "invalid-name|site=root%5B0%5D|name=Bad%20Name"
        );
    }

    #[test]
    fn empty_catalog_for_active_root_reports_missing() {
        let error = resolve(&[], &[q("app", Constraint::AtLeast(x(1)))], &[], 0).unwrap_err();
        assert_eq!(error, ResolveError::Missing { package: "app".into() });
    }

    #[test]
    fn lock_retention_beats_newer_version() {
        let catalog = vec![package(
            "app",
            vec![release(1, vec![]), release(2, vec![])],
        )];
        let resolution = resolve(
            &catalog,
            &[q("app", Constraint::AtLeast(x(1)))],
            &[Lock {
                package: "app".into(),
                version: x(1),
            }],
            10,
        )
        .unwrap();
        assert_eq!(resolution.selected[0].version, x(1));
        assert!(resolution.changes.is_empty());
    }
}
