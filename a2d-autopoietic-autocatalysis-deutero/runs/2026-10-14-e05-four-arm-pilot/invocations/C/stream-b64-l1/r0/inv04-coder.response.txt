const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
}

/// Encodes bytes as padded Base64.
pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];

        output.push(ALPHABET[(a >> 2) as usize] as char);
        output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
        output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
        output.push(ALPHABET[(c & 0x3f) as usize] as char);
    }

    let tail = chunks.remainder();
    if tail.len() == 1 {
        let a = tail[0];
        output.push(ALPHABET[(a >> 2) as usize] as char);
        output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
        output.push_str("==");
    } else if tail.len() == 2 {
        let a = tail[0];
        let b = tail[1];
        output.push(ALPHABET[(a >> 2) as usize] as char);
        output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
        output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
        output.push('=');
    }

    output
}

/// Decodes padded Base64 bytes, reporting invalid bytes and padding positions.
pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut output = Vec::with_capacity(input.len() / 4 * 3);
    let mut group = [0u8; 4];
    let mut offsets = [0usize; 4];
    let mut count = 0usize;
    let mut pending_second_padding: Option<usize> = None;
    let mut terminal_padding = false;

    for (offset, &byte) in input.iter().enumerate() {
        if terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if let Some(padding_offset) = pending_second_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }

            output.push((group[0] << 2) | (group[1] >> 4));
            pending_second_padding = None;
            terminal_padding = true;
            count = 0;
            let _ = padding_offset;
            continue;
        }

        if byte == b'=' {
            match count {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    pending_second_padding = Some(offset);
                    continue;
                }
                3 => {
                    output.push((group[0] << 2) | (group[1] >> 4));
                    output.push((group[1] << 4) | (group[2] >> 2));
                    terminal_padding = true;
                    count = 0;
                    continue;
                }
                _ => unreachable!(),
            }
        }

        let value = match decode_digit(byte) {
            Some(value) => value,
            None => return Err(Error::InvalidByte { offset, byte }),
        };

        group[count] = value;
        offsets[count] = offset;
        count += 1;

        if count == 4 {
            output.push((group[0] << 2) | (group[1] >> 4));
            output.push((group[1] << 4) | (group[2] >> 2));
            output.push((group[2] << 6) | group[3]);
            count = 0;
        }
    }

    if let Some(offset) = pending_second_padding {
        return Err(Error::BadPadding { offset });
    }
    if count != 0 {
        return Err(Error::Truncated {
            offset: offsets[0],
        });
    }

    Ok(output)
}

fn decode_digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn encodes_and_decodes_normal_input() {
        assert_eq!(encode(b"Cat"), "Q2F0");
        assert_eq!(decode(b"Q2F0"), Ok(b"Cat".to_vec()));
    }

    #[test]
    fn handles_empty_and_padded_tails() {
        assert_eq!(encode(b""), "");
        assert_eq!(decode(b""), Ok(Vec::new()));
        assert_eq!(decode(b"Zg=="), Ok(b"f".to_vec()));
        assert_eq!(decode(b"Zm8="), Ok(b"fo".to_vec()));
    }

    #[test]
    fn reports_scan_errors_before_truncation() {
        assert_eq!(
            decode(b"Q?"),
            Err(Error::InvalidByte {
                offset: 1,
                byte: b'?'
            })
        );
        assert_eq!(decode(b"Q"), Err(Error::Truncated { offset: 0 }));
    }

    #[test]
    fn round_trips_all_byte_values() {
        let bytes: Vec<u8> = (0..=255).collect();
        assert_eq!(decode(encode(&bytes).as_bytes()), Ok(bytes));
    }
}
