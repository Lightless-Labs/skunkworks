use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroPeriod,
    Disabled,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub frequency: u8,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacity: usize,
    period: u32,
    pending: u32,
    entries: BTreeMap<u64, Entry>,
    recency: VecDeque<u64>,
}

impl Cache {
    /// Creates an empty cache with the given capacity and aging period.
    pub fn new(capacity: usize, period: u32) -> Result<Self, Error> {
        if period == 0 {
            return Err(Error::ZeroPeriod);
        }
        Ok(Self {
            capacity,
            period,
            pending: 0,
            entries: BTreeMap::new(),
            recency: VecDeque::new(),
        })
    }

    /// Returns the cache's entry capacity.
    pub fn capacity(&self) -> usize {
        self.capacity
    }

    /// Returns the number of operations since the last automatic aging round.
    pub fn pending(&self) -> u32 {
        self.pending
    }

    /// Returns cloned entries ordered by eviction priority.
    pub fn snapshot(&self) -> Vec<Entry> {
        let mut result = self.entries_by_recency();
        result.sort_by_key(|entry| entry.frequency);
        result
    }

    /// Returns the value for a key and updates its frequency and recency on a hit.
    pub fn get(&mut self, key: u64) -> Option<i64> {
        self.begin();
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.credit(key);
        }
        value
    }

    /// Inserts or updates a value, returning any key evicted to make room.
    pub fn put(&mut self, key: u64, value: i64) -> Result<Vec<u64>, Error> {
        if self.capacity == 0 {
            return Err(Error::Disabled);
        }

        self.begin();
        if self.entries.contains_key(&key) {
            if let Some(entry) = self.entries.get_mut(&key) {
                entry.value = value;
            }
            self.credit(key);
            return Ok(Vec::new());
        }

        let mut removed = Vec::new();
        if self.entries.len() >= self.capacity {
            if let Some(victim) = self.victim() {
                self.erase(victim);
                removed.push(victim);
            }
        }
        self.store_new(key, value);
        Ok(removed)
    }

    /// Removes a key and returns its value, if present.
    pub fn remove(&mut self, key: u64) -> Option<i64> {
        self.begin();
        self.erase(key).map(|entry| entry.value)
    }

    /// Ages all resident entries by the requested number of rounds.
    pub fn age(&mut self, rounds: u8) {
        for _ in 0..rounds {
            self.halve();
        }
    }

    /// Changes capacity and returns entries evicted in priority order.
    pub fn resize(&mut self, capacity: usize) -> Vec<u64> {
        self.capacity = capacity;
        let mut removed = Vec::new();
        while self.entries.len() > capacity {
            if let Some(victim) = self.victim() {
                self.erase(victim);
                removed.push(victim);
            } else {
                break;
            }
        }
        removed
    }

    fn entries_by_recency(&self) -> Vec<Entry> {
        self.recency
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect()
    }

    fn store_new(&mut self, key: u64, value: i64) {
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                frequency: 1,
            },
        );
        self.touch(key);
    }

    fn halve(&mut self) {
        for entry in self.entries.values_mut() {
            entry.frequency = (entry.frequency + 1) / 2;
        }
    }

    fn begin(&mut self) {
        self.pending += 1;
        if self.pending == self.period {
            self.halve();
            self.pending = 0;
        }
    }

    fn touch(&mut self, key: u64) {
        self.recency.retain(|existing| *existing != key);
        self.recency.push_back(key);
    }

    fn credit(&mut self, key: u64) {
        if let Some(entry) = self.entries.get_mut(&key) {
            entry.frequency = entry.frequency.saturating_add(1).min(7);
        }
        self.touch(key);
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.recency.retain(|existing| *existing != key);
        Some(entry)
    }

    fn victim(&self) -> Option<u64> {
        self.recency.iter().copied().min_by_key(|key| {
            self.entries
                .get(key)
                .map(|entry| entry.frequency)
                .unwrap_or(u8::MAX)
        })
    }
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn lfu_basic_path() {
        let mut cache = Cache::new(2, 100).unwrap();
        cache.put(1, 10).unwrap();
        cache.put(2, 20).unwrap();
        assert_eq!(cache.get(1), Some(10));
        assert_eq!(cache.put(3, 30), Ok(vec![2]));
        assert_eq!(
            cache.snapshot().iter().map(|entry| entry.key).collect::<Vec<_>>(),
            vec![3, 1]
        );
    }

    #[test]
    fn aging_and_boundary_operation() {
        let mut cache = Cache::new(2, 2).unwrap();
        cache.put(4, 40).unwrap();
        cache.get(4);
        assert_eq!(cache.pending(), 0);
        assert_eq!(cache.snapshot()[0].frequency, 1);
    }

    #[test]
    fn disabled_put_does_not_count() {
        let mut cache = Cache::new(0, 3).unwrap();
        assert_eq!(cache.put(1, 0), Err(Error::Disabled));
        assert_eq!(cache.pending(), 0);
        assert_eq!(cache.get(1), None);
        assert_eq!(cache.pending(), 1);
    }

    #[test]
    fn resize_evicts_by_frequency_then_recency() {
        let mut cache = Cache::new(3, 100).unwrap();
        cache.put(8, 80).unwrap();
        cache.put(9, 90).unwrap();
        cache.put(10, 100).unwrap();
        cache.get(8);
        assert_eq!(cache.resize(1), vec![9, 10]);
        assert_eq!(cache.snapshot()[0].key, 8);
    }
}
