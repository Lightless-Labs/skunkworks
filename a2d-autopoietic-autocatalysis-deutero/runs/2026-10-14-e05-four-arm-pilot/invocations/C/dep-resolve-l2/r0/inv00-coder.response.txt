use std::collections::{BTreeMap, BTreeSet};

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn canonical_constraints(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| matches(version, *c))
}

fn collect_active(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
) -> BTreeMap<String, Vec<Constraint>> {
    let mut active = BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in selected {
        if let Some(package) = catalog.get(name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        canonical_constraints(constraints);
    }
    active
}

enum SearchResult {
    Done(Result<Vec<Selection>, ResolveError>),
    Failed(ResolveError),
}

fn search(
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    selected: &mut BTreeMap<String, Version>,
    attempts: &mut usize,
    max_attempts: usize,
) -> SearchResult {
    let active = collect_active(roots, selected, catalog);

    if let Some(name) = active.keys().find(|name| !catalog.contains_key(*name)) {
        return SearchResult::Failed(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in selected.iter() {
        if let Some(constraints) = active.get(name) {
            if !satisfies(*version, constraints) {
                return SearchResult::Failed(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let mut candidates_by_name = BTreeMap::<String, Vec<&Release>>::new();
    for (name, constraints) in &active {
        if selected.contains_key(name) {
            continue;
        }
        let package = catalog[name];
        let mut releases: Vec<&Release> = package
            .releases
            .iter()
            .filter(|release| satisfies(release.version, constraints))
            .collect();
        releases.sort_by(|a, b| b.version.cmp(&a.version));
        if releases.is_empty() {
            return SearchResult::Failed(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
        candidates_by_name.insert(name.clone(), releases);
    }

    let Some((name, releases)) = candidates_by_name.iter().next() else {
        return SearchResult::Done(Ok(selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect()));
    };

    let name = name.clone();
    let releases = releases.clone();
    let mut first_error = None;

    for release in releases {
        if *attempts == max_attempts {
            return SearchResult::Done(Err(ResolveError::Limit {
                attempts: max_attempts,
            }));
        }
        *attempts += 1;
        selected.insert(name.clone(), release.version);

        match search(roots, catalog, selected, attempts, max_attempts) {
            SearchResult::Done(Ok(solution)) => return SearchResult::Done(Ok(solution)),
            SearchResult::Done(Err(error @ ResolveError::Limit { .. })) => {
                return SearchResult::Done(Err(error));
            }
            SearchResult::Done(Err(error)) | SearchResult::Failed(error) => {
                selected.remove(&name);
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    SearchResult::Failed(first_error.unwrap_or_else(|| ResolveError::NoMatch {
        package: name.clone(),
        requirements: active.get(&name).cloned().unwrap_or_default(),
    }))
}

/// Resolves root requirements against a catalog using a global assignment budget.
pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let mut selected = BTreeMap::new();
    let mut attempts = 0;

    match search(
        roots,
        &catalog_map,
        &mut selected,
        &mut attempts,
        max_attempts,
    ) {
        SearchResult::Done(result) => result,
        SearchResult::Failed(error) => Err(error),
    }
}

fn format_version(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn format_constraint(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", format_version(v)),
        Constraint::AtLeast(v) => format!(">={}", format_version(v)),
        Constraint::Before(v) => format!("<{}", format_version(v)),
        Constraint::Caret(v) => format!("^{}", format_version(v)),
    }
}

fn encode(text: &str) -> String {
    let mut output = String::new();
    for byte in text.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            output.push(byte as char);
        } else {
            output.push_str(&format!("%{byte:02X}"));
        }
    }
    output
}

/// Formats a resolution error as its deterministic, percent-encoded text representation.
pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            format_version(*version)
        ),
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => {
            let mut constraints = requirements.clone();
            canonical_constraints(&mut constraints);
            let text = constraints
                .iter()
                .map(|c| format_constraint(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!("no-match|package={}|constraints={text}", encode(package))
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => {
            let mut constraints = requirements.clone();
            canonical_constraints(&mut constraints);
            let text = constraints
                .iter()
                .map(|c| format_constraint(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={text}",
                encode(package),
                format_version(*selected)
            )
        }
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn v(major: u32, minor: u32, patch: u32) -> Version {
        Version { major, minor, patch }
    }
    fn x(n: u32) -> Version {
        v(n, 0, 0)
    }
    fn e(n: u32) -> Constraint {
        Constraint::Exact(x(n))
    }
    fn ge(n: u32) -> Constraint {
        Constraint::AtLeast(x(n))
    }
    fn q(package: &str, constraint: Constraint) -> Requirement {
        Requirement {
            package: package.into(),
            constraint,
        }
    }
    fn r(n: u32, deps: Vec<Requirement>) -> Release {
        Release { version: x(n), deps }
    }
    fn p(name: &str, releases: Vec<Release>) -> Package {
        Package { name: name.into(), releases }
    }
    fn sel(package: &str, n: u32) -> Selection {
        Selection {
            package: package.into(),
            version: x(n),
        }
    }
    fn run(catalog: &[Package], roots: &[Requirement]) -> Result<Vec<Selection>, ResolveError> {
        resolve(catalog, roots, 100_000)
    }

    #[test]
    fn public_transitive_selection() {
        let catalog = vec![
            p("engine", vec![r(1, vec![q("util", ge(1))])]),
            p("util", vec![r(1, vec![])]),
        ];
        assert_eq!(
            run(&catalog, &[q("engine", e(1))]).unwrap(),
            vec![sel("engine", 1), sel("util", 1)]
        );
    }

    #[test]
    fn public_ranges() {
        let catalog = vec![p("codec", vec![r(3, vec![]), r(1, vec![]), r(2, vec![])])];
        assert_eq!(
            run(
                &catalog,
                &[q("codec", ge(1)), q("codec", Constraint::Before(x(3)))]
            )
            .unwrap(),
            vec![sel("codec", 2)]
        );
    }

    #[test]
    fn public_error_format() {
        let error = ResolveError::InvalidName {
            site: "root[0]".into(),
            name: "Bad Name".into(),
        };
        assert_eq!(
            error_text(&error),
            "invalid-name|site=root%5B0%5D|name=Bad%20Name"
        );
    }

    #[test]
    fn public_backtracking() {
        let catalog = vec![p(
            "app",
            vec![r(2, vec![q("missing", e(1))]), r(1, vec![])],
        )];
        assert_eq!(
            run(&catalog, &[q("app", ge(1))]).unwrap(),
            vec![sel("app", 1)]
        );
    }

    #[test]
    fn validates_unreachable_catalog_data() {
        let catalog = vec![p("good", vec![r(1, vec![q("Bad", e(1))])])];
        assert_eq!(
            resolve(&catalog, &[], 0),
            Err(ResolveError::InvalidName {
                site: "catalog[0].release[0].dep[0]".into(),
                name: "Bad".into(),
            })
        );
    }

    #[test]
    fn limit_counts_assignments_globally() {
        let catalog = vec![p("app", vec![r(2, vec![]), r(1, vec![])])];
        assert_eq!(
            resolve(&catalog, &[q("app", ge(1))], 0),
            Err(ResolveError::Limit { attempts: 0 })
        );
    }

    #[test]
    fn canonicalizes_constructed_error_constraints() {
        let error = ResolveError::NoMatch {
            package: "pkg".into(),
            requirements: vec![Constraint::Before(x(2)), e(1), e(1)],
        };
        assert_eq!(
            error_text(&error),
            "no-match|package=pkg|constraints==1.0.0,<2.0.0"
        );
    }
}
