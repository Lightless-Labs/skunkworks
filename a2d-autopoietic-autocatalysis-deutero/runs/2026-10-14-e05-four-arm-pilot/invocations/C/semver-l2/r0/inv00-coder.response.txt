use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, at: usize) -> Option<u8> {
    s.as_bytes().get(at).copied()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut ident_start = start;
    loop {
        let mut p = ident_start;
        while p < end {
            let b = s.as_bytes()[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..p];
        if prerelease
            && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.len() > 1
            && ident.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(ident.to_string());
        if p == end {
            break;
        }
        ident_start = p + 1;
        if ident_start > end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }
    Ok(result)
}

/// Parses a strict semantic version string.
pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        match byte_at(s, pos) {
            Some(b'.') => pos += 1,
            None | Some(b'-' | b'+') => {
                return Err(VersionError::MissingCore { at: pos });
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        let value = parse_core(s, &mut pos)?;
        if _ == 0 {
            // The second core value is minor; retain it for the final construction.
            MINOR_HOLDER.with(|_| {});
            // handled below by the local helper flow
            let _ = value;
        }
    }

    // Reparse the already validated core fields to keep the parse flow simple.
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    p += 1;
    let minor = parse_core(s, &mut p)?;
    p += 1;
    let patch = parse_core(s, &mut p)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();

    match byte_at(s, p) {
        None => {}
        Some(b'-') => {
            let pre_start = p + 1;
            let plus = s[pre_start..]
                .find('+')
                .map(|i| pre_start + i);
            let pre_end = plus.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(plus_at) = plus {
                build = parse_identifiers(s, plus_at + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, p + 1, s.len(), false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: p }),
    }

    Ok(Version { major, minor, patch, pre, build })
}

thread_local! {
    static MINOR_HOLDER: () = const { () };
}

fn cmp_ident(a: &str, b: &str) -> Ordering {
    let an = a.bytes().all(|c| c.is_ascii_digit());
    let bn = b.bytes().all(|c| c.is_ascii_digit());
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

/// Compares semantic version precedence, ignoring build metadata.
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let c = cmp_ident(x, y);
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn is_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, base: usize) -> Vec<(&str, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_space(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_space(bytes[i]) {
            i += 1;
        }
        out.push((&s[start..i], base + start));
    }
    out
}

fn full_version(token: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(token).map_err(|error| RangeError::Version { at, error })
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: *fields.first().unwrap_or(&0),
        minor: *fields.get(1).unwrap_or(&0),
        patch: *fields.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    version: Version,
    numeric: usize,
    full: bool,
}

fn parse_partial(token: &str, at: usize) -> Result<Partial, RangeError> {
    if token.contains('-') || token.contains('+') {
        return Ok(Partial {
            version: full_version(token, at)?,
            numeric: 3,
            full: true,
        });
    }
    let fields: Vec<&str> = token.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if fields.len() >= 3 && !has_wild {
        return Ok(Partial {
            version: full_version(token, at)?,
            numeric: 3,
            full: true,
        });
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wild_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wild_seen = true;
        } else {
            if wild_seen
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial {
        version: padded(&nums),
        numeric: nums.len(),
        full: false,
    })
}

fn increment(version: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut v = version.clone();
    let slot = match field {
        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,
    };
    *slot = slot.checked_add(1).ok_or(RangeError::Syntax { at })?;
    v.pre.clear();
    v.build.clear();
    match field {
        0 => { v.minor = 0; v.patch = 0; }
        1 => v.patch = 0,
        _ => {}
    }
    Ok(v)
}

fn add_check(arm: &mut Arm, op: Op, version: Version, explicit: bool) {
    if explicit && !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
    arm.checks.push(Check { op, version });
}

fn expand_bare(arm: &mut Arm, p: Partial, at: usize) -> Result<(), RangeError> {
    if p.full {
        add_check(arm, Op::Eq, p.version.clone(), true);
    } else if p.numeric == 1 || p.numeric == 2 {
        add_check(arm, Op::Ge, p.version.clone(), false);
        let upper = increment(&p.version, p.numeric - 1, at)?;
        add_check(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn expand_caret_tilde(
    arm: &mut Arm,
    token: &str,
    at: usize,
    caret: bool,
) -> Result<(), RangeError> {
    let operand = &token[1..];
    let p = parse_partial(operand, at + 1)?;
    if p.numeric == 0 {
        return Err(RangeError::Syntax { at });
    }
    add_check(arm, Op::Ge, p.version.clone(), true);
    let field = if caret {
        if p.version.major > 0 || p.numeric == 1 {
            0
        } else if p.version.minor > 0 || p.numeric == 2 {
            1
        } else {
            2
        }
    } else if p.numeric == 1 {
        0
    } else {
        1
    };
    let upper = increment(&p.version, field, at + 1)?;
    add_check(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_arm(ts: &[(&str, usize)]) -> Result<Arm, RangeError> {
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    if let Some((_, minus_at)) = ts.iter().find(|(t, _)| *t == "-") {
        if ts.len() != 3 || ts[1].0 != "-" {
            return Err(RangeError::Syntax { at: *minus_at });
        }
        let lo = parse_partial(ts[0].0, ts[0].1)?;
        let hi = parse_partial(ts[2].0, ts[2].1)?;
        if lo.numeric == 0 {
            return Err(RangeError::Syntax { at: ts[0].1 });
        }
        if hi.numeric == 0 {
            return Err(RangeError::Syntax { at: ts[2].1 });
        }
        let mut arm = Arm { checks: Vec::new(), anchors: Vec::new() };
        add_check(&mut arm, Op::Ge, lo.version.clone(), true);
        if hi.full {
            add_check(&mut arm, Op::Le, hi.version, true);
        } else if hi.numeric == 1 || hi.numeric == 2 {
            let upper = increment(&hi.version, hi.numeric - 1, ts[2].1)?;
            add_check(&mut arm, Op::Lt, upper, false);
        } else {
            add_check(&mut arm, Op::Le, hi.version, false);
        }
        return Ok(arm);
    }

    let mut arm = Arm { checks: Vec::new(), anchors: Vec::new() };
    for &(token, at) in ts {
        let first = token.as_bytes().first().copied();
        match first {
            Some(b'^') | Some(b'~') => expand_caret_tilde(&mut arm, token, at, first == Some(b'^'))?,
            Some(b'<' | b'>' | b'=') => {
                let (op, n) = if token.starts_with(">=") {
                    (Op::Ge, 2)
                } else if token.starts_with("<=") {
                    (Op::Le, 2)
                } else if token.starts_with('>') {
                    (Op::Gt, 1)
                } else if token.starts_with('<') {
                    (Op::Lt, 1)
                } else {
                    (Op::Eq, 1)
                };
                if token.len() == n || !token.as_bytes()[n].is_ascii_digit() {
                    return Err(RangeError::Syntax { at });
                }
                let v = full_version(&token[n..], at + n)?;
                add_check(&mut arm, op, v, true);
            }
            Some(b'0'..=b'9' | b'*' | b'x' | b'X') => {
                let p = parse_partial(token, at)?;
                expand_bare(&mut arm, p, at)?;
            }
            _ => return Err(RangeError::Syntax { at }),
        }
    }
    Ok(arm)
}

/// Parses a conjunction or disjunction of semantic version range expressions.
pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let all = tokens(s, 0);
    if all.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i <= all.len() {
        if i == all.len() || all[i].0 == "||" {
            if i == start {
                let at = if i == all.len() {
                    s.len()
                } else if i == 0 {
                    all[i].1
                } else {
                    all[i].1
                };
                return Err(RangeError::Empty { at });
            }
            arms.push(parse_arm(&all[start..i])?);
            start = i + 1;
        }
        i += 1;
    }
    Ok(Range { arms })
}

fn satisfies(v: &Version, check: &Check) -> bool {
    let ord = compare_versions(v, &check.version);
    match check.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

/// Returns whether a version matches a parsed range.
pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.iter().any(|t| *t == (v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.checks.iter().all(|c| satisfies(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, value) in candidates.iter().enumerate() {
        let v = parse_version(value).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        if let Some(j) = best {
            let cmp = compare_versions(&parsed[i], &parsed[j]);
            if (greatest && cmp == Ordering::Greater) || (!greatest && cmp == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

/// Returns the greatest matching candidate, preserving the original input spelling.
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

/// Returns the least matching candidate, preserving the original input spelling.
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    fn pr(s: &str) -> Range {
        parse_range(s).unwrap()
    }

    #[test]
    fn version_parsing_and_precedence() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")), Ordering::Less);
    }

    #[test]
    fn invalid_version_and_partial_range() {
        assert_eq!(parse_version("01.2.3"), Err(VersionError::LeadingZero { at: 0 }));
        assert!(matches(&pv("2.5.1"), &pr("2.5.x")));
    }

    #[test]
    fn range_expansion_and_extrema() {
        let range = pr("^2.1.0 || 4.0.0");
        assert!(matches(&pv("2.5.1"), &range));
        assert!(matches(&pv("4.0.0"), &range));
        let candidates = vec!["2.2.0".to_string(), "2.8.0".to_string(), "3.0.0".to_string()];
        let r = pr("2.x");
        assert_eq!(max_satisfying(&candidates, &r), Ok(Some("2.8.0".into())));
        assert_eq!(min_satisfying(&candidates, &r), Ok(Some("2.2.0".into())));
    }
}
