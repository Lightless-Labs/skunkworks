use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, at: usize) -> Option<u8> {
    s.as_bytes().get(at).copied()
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while matches!(byte_at(s, *pos), Some(byte) if byte.is_ascii_digit()) {
        *pos += 1;
    }
    if *pos == start {
        return match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            Some(_) => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut id_start = start;
    loop {
        if id_start == end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let mut cursor = id_start;
        while cursor < end && byte_at(s, cursor) != Some(b'.') {
            let byte = byte_at(s, cursor).unwrap_or_default();
            let valid = byte.is_ascii_alphanumeric() || byte == b'-';
            if !valid {
                return Err(VersionError::InvalidChar { at: cursor });
            }
            cursor += 1;
        }
        let identifier = &s[id_start..cursor];
        let all_digits = identifier.bytes().all(|byte| byte.is_ascii_digit());
        if prerelease && all_digits && identifier.len() > 1 && identifier.starts_with('0') {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(identifier.to_string());
        if cursor == end {
            break;
        }
        id_start = cursor + 1;
    }
    Ok(result)
}

/// Parses a strict semantic version string.
pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for _index in 0..2 {
        match byte_at(s, pos) {
            Some(b'.') => pos += 1,
            None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
            Some(_) => return Err(VersionError::InvalidChar { at: pos }),
        }
        if _index == 0 {
            continue;
        }
    }

    // Parse minor and patch explicitly so each separator is checked after its number.
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    if byte_at(s, pos) != Some(b'.') {
        return match byte_at(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            Some(_) => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let minor = parse_core_number(s, &mut pos)?;
    if byte_at(s, pos) != Some(b'.') {
        return match byte_at(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            Some(_) => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let patch = parse_core_number(s, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    match byte_at(s, pos) {
        None => {}
        Some(b'-') => {
            let pre_start = pos + 1;
            let plus = s[pre_start..].find('+').map(|offset| pre_start + offset);
            let pre_end = plus.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(plus_pos) = plus {
                build = parse_identifiers(s, plus_pos + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, pos + 1, s.len(), false)?;
        }
        Some(_) => return Err(VersionError::InvalidChar { at: pos }),
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = a.bytes().all(|byte| byte.is_ascii_digit());
    let b_numeric = b.bytes().all(|byte| byte.is_ascii_digit());
    match (a_numeric, b_numeric) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

/// Compares semantic version precedence, ignoring build metadata.
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(b.pre.iter()) {
                let order = compare_identifiers(left, right);
                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    at: usize,
}

fn is_space(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str, base: usize) -> Vec<Token> {
    let mut tokens = Vec::new();
    let bytes = s.as_bytes();
    let mut pos = 0;
    while pos < bytes.len() {
        while pos < bytes.len() && is_space(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }
        let start = pos;
        while pos < bytes.len() && !is_space(bytes[pos]) {
            pos += 1;
        }
        tokens.push(Token { text: s[start..pos].to_string(), at: base + start });
    }
    tokens
}

fn syntax(at: usize) -> RangeError {
    RangeError::Syntax { at }
}

fn wrap_version(text: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(text).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    numeric_count: usize,
    full: Option<Version>,
}

fn parse_partial(text: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix_marker = text.contains('-') || text.contains('+');
    let fields: Vec<&str> = text.split('.').collect();
    let has_wildcard = fields.iter().any(|field| matches!(*field, "*" | "x" | "X"));
    if has_suffix_marker || (fields.len() >= 3 && !has_wildcard) {
        let version = wrap_version(text, at)?;
        return Ok(Partial {
            fields: vec![Some(version.major), Some(version.minor), Some(version.patch)],
            numeric_count: 3,
            full: Some(version),
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(syntax(at));
    }
    let mut parsed = Vec::new();
    let mut numeric_count = 0;
    let mut saw_wildcard = false;
    for field in fields {
        if field.is_empty() {
            return Err(syntax(at));
        }
        if matches!(field, "*" | "x" | "X") {
            saw_wildcard = true;
            parsed.push(None);
        } else {
            if saw_wildcard
                || !field.bytes().all(|byte| byte.is_ascii_digit())
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(syntax(at));
            }
            let number = field.parse::<u64>().map_err(|_| syntax(at))?;
            numeric_count += 1;
            parsed.push(Some(number));
        }
    }
    Ok(Partial { fields: parsed, numeric_count, full: None })
}

fn partial_lower(partial: &Partial) -> Version {
    Version {
        major: partial.fields.first().copied().flatten().unwrap_or(0),
        minor: partial.fields.get(1).copied().flatten().unwrap_or(0),
        patch: partial.fields.get(2).copied().flatten().unwrap_or(0),
        pre: partial.full.as_ref().map(|version| version.pre.clone()).unwrap_or_default(),
        build: partial.full.as_ref().map(|version| version.build.clone()).unwrap_or_default(),
    }
}

fn upper_increment(
    version: &Version,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut upper = Version {
        major: version.major,
        minor: version.minor,
        patch: version.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    let target = match field {
        0 => &mut upper.major,
        1 => &mut upper.minor,
        _ => &mut upper.patch,
    };
    *target = target.checked_add(1).ok_or_else(|| syntax(at))?;
    if field == 0 {
        upper.minor = 0;
        upper.patch = 0;
    } else if field == 1 {
        upper.patch = 0;
    }
    Ok(upper)
}

fn add_comparator(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator { op, version });
}

fn parse_bare(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let partial = parse_partial(&token.text, token.at)?;
    if let Some(version) = partial.full {
        add_comparator(arm, Op::Eq, version, true);
        return Ok(());
    }
    if partial.numeric_count == 0 {
        return Ok(());
    }
    let lower = partial_lower(&partial);
    add_comparator(arm, Op::Ge, lower.clone(), true);
    if partial.numeric_count == 1 {
        let upper = upper_increment(&lower, 0, token.at)?;
        add_comparator(arm, Op::Lt, upper, false);
    } else if partial.numeric_count == 2 {
        let upper = upper_increment(&lower, 1, token.at)?;
        add_comparator(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_caret_tilde(token: &Token, arm: &mut Arm, caret: bool) -> Result<(), RangeError> {
    let operand_at = token.at + 1;
    let operand = &token.text[1..];
    let partial = parse_partial(operand, operand_at)?;
    if partial.numeric_count == 0 {
        return Err(syntax(token.at));
    }
    let lower = partial_lower(&partial);
    add_comparator(arm, Op::Ge, lower.clone(), true);
    let field = if caret {
        if lower.major > 0 || partial.numeric_count == 1 {
            0
        } else if lower.minor > 0 || partial.numeric_count == 2 {
            1
        } else {
            2
        }
    } else if partial.numeric_count == 1 {
        0
    } else {
        1
    };
    let upper = upper_increment(&lower, field, token.at)?;
    add_comparator(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_explicit_comparator(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let bytes = token.text.as_bytes();
    let (length, op) = match bytes.first().copied() {
        Some(b'=') => (1, Op::Eq),
        Some(b'<') if bytes.get(1) == Some(&b'=') => (2, Op::Le),
        Some(b'<') => (1, Op::Lt),
        Some(b'>') if bytes.get(1) == Some(&b'=') => (2, Op::Ge),
        Some(b'>') => (1, Op::Gt),
        _ => return Err(syntax(token.at)),
    };
    let operand = &token.text[length..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(syntax(token.at));
    }
    let version = wrap_version(operand, token.at + length)?;
    add_comparator(arm, op, version, true);
    Ok(())
}

fn parse_hyphen(tokens: &[Token], arm: &mut Arm) -> Result<(), RangeError> {
    let minus_index = tokens.iter().position(|token| token.text == "-").unwrap_or(0);
    if tokens.len() != 3 || minus_index != 1 {
        return Err(syntax(tokens[minus_index].at));
    }
    let lower_partial = parse_partial(&tokens[0].text, tokens[0].at)?;
    let upper_partial = parse_partial(&tokens[2].text, tokens[2].at)?;
    if lower_partial.numeric_count == 0 {
        return Err(syntax(tokens[0].at));
    }
    if upper_partial.numeric_count == 0 {
        return Err(syntax(tokens[2].at));
    }
    let lower = partial_lower(&lower_partial);
    add_comparator(arm, Op::Ge, lower, true);
    if let Some(full_upper) = upper_partial.full {
        add_comparator(arm, Op::Le, full_upper, true);
    } else {
        let upper_base = partial_lower(&upper_partial);
        let upper = if upper_partial.numeric_count == 1 {
            upper_increment(&upper_base, 0, tokens[2].at)?
        } else if upper_partial.numeric_count == 2 {
            upper_increment(&upper_base, 1, tokens[2].at)?
        } else {
            add_comparator(arm, Op::Le, upper_base, false);
            return Ok(());
        };
        add_comparator(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_arm(tokens: &[Token]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    if tokens.iter().any(|token| token.text == "-") {
        parse_hyphen(tokens, &mut arm)?;
        return Ok(arm);
    }
    for token in tokens {
        let first = token.text.as_bytes().first().copied();
        match first {
            Some(b'^') => parse_caret_tilde(token, &mut arm, true)?,
            Some(b'~') => parse_caret_tilde(token, &mut arm, false)?,
            Some(b'<' | b'>' | b'=') => parse_explicit_comparator(token, &mut arm)?,
            Some(b'0'..=b'9' | b'*' | b'x' | b'X') => parse_bare(token, &mut arm)?,
            _ => return Err(syntax(token.at)),
        }
    }
    Ok(arm)
}

/// Parses a range expression with comparator, partial, and disjunction support.
pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s, 0);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut pending_separator = None;
    for token in tokens {
        if token.text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: token.at });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            pending_separator = Some(token.at);
        } else {
            current.push(token);
        }
    }
    if current.is_empty() {
        if pending_separator.is_some() {
            return Err(RangeError::Empty { at: s.len() });
        }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn comparator_matches(candidate: &Version, comparator: &Comparator) -> bool {
    let order = compare_versions(candidate, &comparator.version);
    match comparator.op {
        Op::Eq => order == Ordering::Equal,
        Op::Lt => order == Ordering::Less,
        Op::Le => order != Ordering::Greater,
        Op::Gt => order == Ordering::Greater,
        Op::Ge => order != Ordering::Less,
    }
}

/// Checks whether a version satisfies a parsed range.
pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.iter().any(|core| *core == (v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.comparators.iter().all(|comparator| comparator_matches(v, comparator))
    })
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        match parse_version(candidate) {
            Ok(version) => parsed.push(version),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(best_index) => {
                let order = compare_versions(version, &parsed[best_index]);
                let improves = if greatest {
                    order == Ordering::Greater
                } else {
                    order == Ordering::Less
                };
                if improves {
                    best = Some(index);
                }
            }
        }
    }
    Ok(best.map(|index| candidates[index].clone()))
}

/// Returns the original candidate with greatest satisfying precedence.
pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

/// Returns the original candidate with least satisfying precedence.
pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn version_parse_and_compare() {
        let version = parse_version("2.1.3-rc.1+build.7").unwrap();
        assert_eq!(version.pre, vec!["rc", "1"]);
        assert_eq!(compare_versions(
            &parse_version("2.1.3-alpha").unwrap(),
            &parse_version("2.1.3-beta").unwrap()
        ), Ordering::Less);
    }

    #[test]
    fn invalid_version_offsets() {
        assert_eq!(parse_version("01.2.3"), Err(VersionError::LeadingZero { at: 0 }));
        assert_eq!(parse_version("1.2"), Err(VersionError::MissingCore { at: 3 }));
    }

    #[test]
    fn range_expansion_and_matching() {
        let range = parse_range("^2.1.0 || 4.0.0").unwrap();
        assert!(matches(&parse_version("2.5.1").unwrap(), &range));
        assert!(matches(&parse_version("4.0.0").unwrap(), &range));
    }

    #[test]
    fn extrema_select_original_candidate() {
        let candidates = vec!["2.2.0".to_string(), "2.8.0".to_string(), "3.0.0".to_string()];
        let range = parse_range("2.x").unwrap();
        assert_eq!(max_satisfying(&candidates, &range), Ok(Some("2.8.0".to_string())));
        assert_eq!(min_satisfying(&candidates, &range), Ok(Some("2.2.0".to_string())));
    }
}
