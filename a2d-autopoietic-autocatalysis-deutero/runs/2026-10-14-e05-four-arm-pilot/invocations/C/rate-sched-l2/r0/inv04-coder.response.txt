use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

fn parse_number(text: &str, offset: usize, limit: usize) -> Result<usize, CronError> {
    if text.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some((i, _)) = text.bytes().enumerate().find(|(_, b)| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if text.len() > 1 && text.starts_with('0') {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for b in text.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(offset)),
        };
        if value > limit {
            return Err(CronError::Bounds(offset));
        }
    }
    Ok(value)
}

fn parse_field(
    text: &str,
    field_offset: usize,
    limit: usize,
    step_limit: usize,
) -> Result<Vec<bool>, CronError> {
    let mut selected = vec![false; limit + 1];
    let mut item_start = 0;

    loop {
        let comma = text[item_start..].find(',').map(|i| item_start + i);
        let item_end = comma.unwrap_or(text.len());
        let item = &text[item_start..item_end];
        let absolute_start = field_offset + item_start;

        if item.is_empty() {
            return Err(CronError::Syntax(absolute_start));
        }

        let slash_positions: Vec<usize> = item.match_indices('/').map(|(i, _)| i).collect();
        if slash_positions.len() > 1 {
            return Err(CronError::Syntax(absolute_start + slash_positions[1]));
        }
        let slash = slash_positions.first().copied();

        let base_end = slash.unwrap_or(item.len());
        let base = &item[..base_end];

        if let Some(slash_pos) = slash {
            if !base.contains('*') && !base.contains('-') {
                return Err(CronError::Syntax(absolute_start + slash_pos));
            }
        }

        let (start, end) = if base == "*" {
            (0, limit)
        } else if let Some(dash) = base.find('-') {
            let left = parse_number(&base[..dash], absolute_start, limit)?;
            let right_offset = absolute_start + dash + 1;
            let right = parse_number(&base[dash + 1..], right_offset, limit)?;
            if left > right {
                return Err(CronError::Descending(absolute_start + dash));
            }
            (left, right)
        } else {
            let n = parse_number(base, absolute_start, limit)?;
            (n, n)
        };

        let step = if let Some(slash_pos) = slash {
            let step_text = &item[slash_pos + 1..];
            let step_offset = absolute_start + slash_pos + 1;

            if step_text.is_empty() {
                return Err(CronError::Syntax(step_offset));
            }
            if let Some((i, _)) = step_text.bytes().enumerate().find(|(_, b)| !b.is_ascii_digit()) {
                return Err(CronError::Syntax(step_offset + i));
            }
            if step_text.len() > 1 && step_text.starts_with('0') {
                return Err(CronError::Number(step_offset));
            }

            let mut value = 0usize;
            for b in step_text.bytes() {
                value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as usize)) {
                    Some(v) => v,
                    None => return Err(CronError::Step(step_offset)),
                };
                if value > step_limit {
                    return Err(CronError::Step(step_offset));
                }
            }
            if value == 0 {
                return Err(CronError::Step(step_offset));
            }
            value
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        match comma {
            Some(pos) => item_start = pos + 1,
            None => break,
        }
    }

    Ok(selected)
}

impl Cron {
    /// Parses a minute and hour cron expression.
    pub fn parse(text: &str) -> Result<Self, CronError> {
        for (i, b) in text.bytes().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = text.match_indices(' ').map(|(i, _)| i).collect();
        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let sep = spaces[0];
        if sep == 0 || sep + 1 == text.len() {
            return Err(CronError::Layout(sep));
        }

        let minutes = parse_field(&text[..sep], 0, 59, 60)?;
        let hours = parse_field(&text[sep + 1..], sep + 1, 23, 24)?;
        let mut minute_set = [false; 60];
        let mut hour_set = [false; 24];
        minute_set.copy_from_slice(&minutes);
        hour_set.copy_from_slice(&hours);
        Ok(Self {
            minutes: minute_set,
            hours: hour_set,
        })
    }

    /// Returns whether an absolute minute matches this cron expression.
    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: HashSet<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    /// Creates an empty calendar.
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: HashSet::new(),
            last_poll: None,
        }
    }

    /// Adds a uniquely identified cron job.
    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.contains(id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        let owned_id = id.to_owned();
        let order = self.jobs.len();
        self.ids.insert(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order,
        });
        Ok(())
    }

    /// Returns occurrences in the interval since the previous successful poll.
    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let mut matching = Vec::new();
            let mut boundary = 0u64;
            while boundary <= now {
                let in_interval = match lower {
                    Some(last) => boundary > last,
                    None => true,
                };
                if in_interval && boundary % 60_000 == 0 && job.cron.matches(boundary / 60_000) {
                    matching.push(boundary);
                }
                match boundary.checked_add(60_000) {
                    Some(next) => boundary = next,
                    None => break,
                }
            }

            let chosen: Vec<u64> = match job.policy {
                Policy::All => matching,
                Policy::Once => matching.last().copied().into_iter().collect(),
                Policy::Skip => matching.into_iter().filter(|&at| at == now).collect(),
            };
            for at in chosen {
                emitted.push((
                    Occurrence {
                        id: job.id.clone(),
                        at,
                    },
                    job.priority,
                    job.order,
                ));
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}

impl Default for Calendar {
    fn default() -> Self {
        Self::new()
    }
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn parses_and_matches_normal_expression() {
        let cron = Cron::parse("5-15/5 2,4").unwrap();
        assert!(cron.matches(130));
        assert!(!cron.matches(131));
    }

    #[test]
    fn rejects_noncanonical_number() {
        assert!(matches!(
            Cron::parse("00 0"),
            Err(ErrorOrCron::Placeholder)
        ));
    }

    #[test]
    fn calendar_emits_catch_up_and_orders_jobs() {
        let mut calendar = Calendar::new();
        calendar.add("low", "*/1 0-23", 0, Policy::All).unwrap();
        calendar.add("high", "*/1 0-23", 9, Policy::All).unwrap();
        let occurrences = calendar.poll(60_000).unwrap();
        assert_eq!(occurrences.len(), 4);
        assert_eq!(occurrences[0].id, "high");
        assert_eq!(occurrences[0].at, 0);
        assert_eq!(occurrences[1].id, "low");
        assert_eq!(occurrences[2].id, "high");
        assert_eq!(occurrences[2].at, 60_000);
    }

    #[test]
    fn policies_consume_poll_interval() {
        let mut calendar = Calendar::new();
        calendar.add("once", "*/1 0-23", 1, Policy::Once).unwrap();
        calendar.add("skip", "*/1 0-23", 2, Policy::Skip).unwrap();
        let first = calendar.poll(90_000).unwrap();
        assert_eq!(first.len(), 1);
        assert_eq!(first[0].id, "once");
        assert_eq!(first[0].at, 60_000);
        assert!(calendar.poll(90_000).unwrap().is_empty());
    }
}
