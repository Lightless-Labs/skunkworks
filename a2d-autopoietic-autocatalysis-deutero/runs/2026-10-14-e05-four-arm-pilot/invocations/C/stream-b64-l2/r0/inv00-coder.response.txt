```rust
use std::fmt;

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Error::InvalidByte { offset, byte } =>
                write!(f, "invalid byte {byte} at offset {offset}"),
            Error::BadPadding { offset } => write!(f, "bad padding at offset {offset}"),
            Error::Truncated { offset } => write!(f, "truncated group at offset {offset}"),
            Error::AlreadyFinished => write!(f, "stream already finished"),
        }
    }
}

impl std::error::Error for Error {}

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|i| i as u8)
}

fn encode_group(input: &[u8], output: &mut String) {
    let a = input[0];
    let b = if input.len() > 1 { input[1] } else { 0 };
    let c = if input.len() > 2 { input[2] } else { 0 };
    let values = [
        a >> 2,
        ((a & 0x03) << 4) | (b >> 4),
        ((b & 0x0f) << 2) | (c >> 6),
        c & 0x3f,
    ];

    output.push(ALPHABET[values[0] as usize] as char);
    output.push(ALPHABET[values[1] as usize] as char);
    if input.len() > 1 {
        output.push(ALPHABET[values[2] as usize] as char);
    } else {
        output.push('=');
    }
    if input.len() > 2 {
        output.push(ALPHABET[values[3] as usize] as char);
    } else {
        output.push('=');
    }
}

/// Encodes bytes using the standard six-bit alphabet and padding.
pub fn encode(input: &[u8]) -> String {
    let mut output = String::new();
    for chunk in input.chunks(3) {
        encode_group(chunk, &mut output);
    }
    output
}

#[derive(Clone, Copy)]
enum Symbol {
    Digit(u8, usize),
    Padding(usize),
}

#[derive(Default)]
struct DecoderState {
    group: Vec<Symbol>,
    offset: usize,
    terminal: bool,
    sticky: Option<Error>,
    finished: bool,
}

fn bad_padding(offset: usize) -> Error {
    Error::BadPadding { offset }
}

fn decode_group(group: &[Symbol], output: &mut Vec<u8>) {
    let vals: Vec<u8> = group
        .iter()
        .filter_map(|s| match s {
            Symbol::Digit(v, _) => Some(*v),
            Symbol::Padding(_) => None,
        })
        .collect();

    let a = vals[0];
    let b = vals[1];
    output.push((a << 2) | (b >> 4));

    if vals.len() >= 3 {
        let c = vals[2];
        output.push((b << 4) | (c >> 2));
        if vals.len() == 4 {
            output.push((c << 6) | vals[3]);
        }
    }
}

fn process_byte(state: &mut DecoderState, byte: u8, output: &mut Vec<u8>) -> Result<(), Error> {
    let offset = state.offset;
    state.offset += 1;

    if state.terminal {
        return Err(bad_padding(offset));
    }

    if let Some(Symbol::Padding(pad_offset)) = state.group.get(2) {
        if byte != b'=' {
            return Err(bad_padding(offset));
        }
        state.group.push(Symbol::Padding(offset));
        let _ = pad_offset;
        decode_group(&state.group, output);
        state.group.clear();
        state.terminal = true;
        return Ok(());
    }

    let position = state.group.len();
    if byte == b'=' {
        match position {
            0 | 1 => return Err(bad_padding(offset)),
            2 => state.group.push(Symbol::Padding(offset)),
            3 => {
                state.group.push(Symbol::Padding(offset));
                decode_group(&state.group, output);
                state.group.clear();
                state.terminal = true;
            }
            _ => return Err(bad_padding(offset)),
        }
        return Ok(());
    }

    let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
    state.group.push(Symbol::Digit(value, offset));
    if state.group.len() == 4 {
        decode_group(&state.group, output);
        state.group.clear();
    }
    Ok(())
}

/// Decodes bytes, returning an error for invalid alphabet, padding, or truncation.
pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

#[derive(Debug, Clone)]
pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    /// Creates an empty open encoder.
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    /// Consumes input and encodes every newly available three-byte group.
    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut data = std::mem::take(&mut self.tail);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut output = String::new();
        for chunk in data[..complete_len].chunks(3) {
            encode_group(chunk, &mut output);
        }
        self.tail.extend_from_slice(&data[complete_len..]);
        Ok(output)
    }

    /// Encodes any remaining bytes and closes the encoder.
    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let mut output = String::new();
        if !self.tail.is_empty() {
            encode_group(&self.tail, &mut output);
            self.tail.clear();
        }
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Debug, Clone, Default)]
pub struct Decoder {
    state: DecoderState,
}

impl Decoder {
    /// Creates an empty open decoder.
    pub fn new() -> Self {
        Self::default()
    }

    /// Scans input and returns newly completed decoded groups.
    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.finished {
            let error = Error::AlreadyFinished;
            self.state.sticky = Some(error.clone());
            return Err(error);
        }

        let mut output = Vec::new();
        for &byte in input {
            if let Err(error) = process_byte(&mut self.state, byte, &mut output) {
                self.state.sticky = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    /// Validates the residual group and closes the decoder.
    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.finished {
            let error = Error::AlreadyFinished;
            self.state.sticky = Some(error.clone());
            return Err(error);
        }

        if let Some(Symbol::Padding(offset)) = self.state.group.get(2) {
            let error = bad_padding(*offset);
            self.state.sticky = Some(error.clone());
            return Err(error);
        }
        if !self.state.group.is_empty() {
            let offset = match self.state.group.first().unwrap() {
                Symbol::Digit(_, offset) | Symbol::Padding(offset) => *offset,
            };
            let error = Error::Truncated { offset };
            self.state.sticky = Some(error.clone());
            return Err(error);
        }
        self.state.finished = true;
        Ok(Vec::new())
    }
}

/// A minimal executable entry point.
pub fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn encode_and_decode_normal_path() {
        assert_eq!(encode(b"Cat"), "Q2F0");
        assert_eq!(decode(b"Q2F0"), Ok(b"Cat".to_vec()));
        assert_eq!(decode(b"T2s="), Ok(b"Ok".to_vec()));
    }

    #[test]
    fn rejects_invalid_and_truncated_input() {
        assert_eq!(
            decode(b"YWJj!"),
            Err(Error::BadPadding { offset: 4 })
        );
        assert_eq!(decode(b"YWJ"), Err(Error::Truncated { offset: 0 }));
        assert_eq!(
            decode(b"Y W="),
            Err(Error::InvalidByte { offset: 1, byte: b' ' })
        );
    }

    #[test]
    fn streaming_round_trip_and_sticky_error() {
        let mut encoder = Encoder::new();
        let mut encoded = encoder.push(b"C").unwrap();
        encoded.push_str(&encoder.push(b"at").unwrap());
        encoded.push_str(&encoder.finish().unwrap());
        assert_eq!(encoded, encode(b"Cat"));

        let mut decoder = Decoder::new();
        assert_eq!(decoder.push(b"YWJj").unwrap(), b"abc");
        assert_eq!(decoder.push(b"=").unwrap_err(), Error::BadPadding { offset: 4 });
        assert_eq!(decoder.finish().unwrap_err(), Error::BadPadding { offset: 4 });
    }

    #[test]
    fn accepts_noncanonical_tail_bits() {
        assert_eq!(decode(b"Zh=="), Ok(b"f".to_vec()));
        assert_eq!(decode(b"Zm9="), Ok(b"fo".to_vec()));
    }
}
```
