            }
        }
        let has_more = matching.next().is_some();
        let next = if has_more {
            items.last().map(|(key, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: key.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    /// Aggregates matching entries using canonical signed 64-bit decimal values.
    pub fn aggregate(
        &mut self,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        if end.is_some_and(|end| end <= start) {
            return Err(Error::InvalidRange);
        }

        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, entry) in self.visible().iter() {
            if !Self::matches(key, start, end, prefix) {
                continue;
            }
            let Some(value) = parse_canonical_i64(&entry.value) else {
                return Err(Error::InvalidNumber {
                    key: key.clone(),
                    value: entry.value.clone(),
                });
            };
            sum = sum.checked_add(value).ok_or_else(|| Error::SumOverflow {
                key: key.clone(),
            })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }
}

fn parse_canonical_i64(text: &str) -> Option<i64> {
    if text == "0" {
        return Some(0);
    }
    let digits = text.strip_prefix('-').unwrap_or(text);
    if digits.is_empty()
        || digits.starts_with('0')
        || !digits.bytes().all(|byte| byte.is_ascii_digit())
        || (text.starts_with('-') && digits == "0")
    {
        return None;
    }
    text.parse::<i64>().ok()
}
