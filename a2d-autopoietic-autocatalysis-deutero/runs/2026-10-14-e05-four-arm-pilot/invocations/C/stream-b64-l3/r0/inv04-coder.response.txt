#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet(options: Options, value: u8) -> u8 {
    match (options.alphabet, value) {
        (Alphabet::UrlSafe, b'+') => b'-',
        (Alphabet::UrlSafe, b'/') => b'_',
        _ => value,
    }
}

fn value(options: Options, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if options.alphabet == Alphabet::Standard => Some(62),
        b'/' if options.alphabet == Alphabet::Standard => Some(63),
        b'-' if options.alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if options.alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn emit_group(bytes: &[u8], options: Options, out: &mut String) {
    let a = bytes[0];
    let b = bytes.get(1).copied().unwrap_or(0);
    let c = bytes.get(2).copied().unwrap_or(0);
    let digits = [
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"[(a >> 2) as usize],
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
            [(((a & 3) << 4) | (b >> 4)) as usize],
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
            [(((b & 15) << 2) | (c >> 6)) as usize],
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"[(c & 63) as usize],
    ];
    out.push(alphabet(options, digits[0]) as char);
    out.push(alphabet(options, digits[1]) as char);
    if bytes.len() >= 2 {
        out.push(alphabet(options, digits[2]) as char);
    } else if options.padding != Padding::Forbidden {
        out.push('=');
    }
    if bytes.len() >= 3 {
        out.push(alphabet(options, digits[3]) as char);
    } else if options.padding != Padding::Forbidden {
        out.push('=');
    }
}

fn wrap_output(raw: String, whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return raw;
    };
    if raw.is_empty() {
        return raw;
    }
    let separator = match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };
    let mut result = String::with_capacity(raw.len() + raw.len() / width * separator.len());
    for (index, chunk) in raw.as_bytes().chunks(width).enumerate() {
        if index > 0 {
            result.push_str(separator);
        }
        result.push_str(std::str::from_utf8(chunk).unwrap());
    }
    result
}

/// Encodes bytes using the selected alphabet, padding, and wrapping options.
pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

#[derive(Debug)]
pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    raw_count: usize,
    finished: bool,
}

impl Encoder {
    /// Creates an open encoder stream.
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            raw_count: 0,
            finished: false,
        })
    }

    /// Encodes all complete three-byte groups in this chunk.
    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut data = std::mem::take(&mut self.tail);
        data.extend_from_slice(input);
        let complete = data.len() / 3 * 3;
        let mut raw = String::new();
        for group in data[..complete].chunks_exact(3) {
            emit_group(group, self.options, &mut raw);
        }
        self.tail.extend_from_slice(&data[complete..]);
        let mut output = String::new();
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            let separator = match ending {
                LineEnding::Lf => "\n",
                LineEnding::CrLf => "\r\n",
            };
            for byte in raw.bytes() {
                if self.raw_count > 0 && self.raw_count % width == 0 {
                    output.push_str(separator);
                }
                output.push(byte as char);
                self.raw_count += 1;
            }
        } else {
            output = raw;
        }
        Ok(output)
    }

    /// Emits any remaining tail and closes the encoder.
    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let mut raw = String::new();
        if !self.tail.is_empty() {
            emit_group(&self.tail, self.options, &mut raw);
            self.tail.clear();
        }
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            let separator = match ending {
                LineEnding::Lf => "\n",
                LineEnding::CrLf => "\r\n",
            };
            let mut output = String::new();
            for byte in raw.bytes() {
                if self.raw_count > 0 && self.raw_count % width == 0 {
                    output.push_str(separator);
                }
                output.push(byte as char);
                self.raw_count += 1;
            }
            Ok(output)
        } else {
            Ok(raw)
        }
    }
}

#[derive(Debug, Clone, Copy)]
enum FrameState {
    Data,
    NeedLf { cr_offset: usize },
    NeedSymbol { start_offset: usize },
}

#[derive(Debug)]
pub struct Decoder {
    options: Options,
    group: Vec<(u8, usize)>,
    offset: usize,
    finished: bool,
    sticky: Option<Error>,
    padded: bool,
    frame_state: FrameState,
    row_symbols: usize,
}

impl Decoder {
    /// Creates an open decoder stream.
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::new(),
            offset: 0,
            finished: false,
            sticky: None,
            padded: false,
            frame_state: FrameState::Data,
            row_symbols: 0,
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky = Some(error.clone());
        error
    }

    fn decode_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        let g = &self.group;
        if g.len() != 4 {
            return Ok(());
        }
        let a = value(self.options, g[0].0).unwrap();
        let b = value(self.options, g[1].0).unwrap();
        out.push((a << 2) | (b >> 4));
        if g[2].0 == b'=' {
            self.padded = true;
        } else {
            let c = value(self.options, g[2].0).unwrap();
            out.push((b << 4) | (c >> 2));
            if g[3].0 == b'=' {
                self.padded = true;
            } else {
                let d = value(self.options, g[3].0).unwrap();
                out.push((c << 6) | d);
            }
        }
        self.group.clear();
        Ok(())
    }

    fn scan_symbol(&mut self, byte: u8, at: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.padded {
            return Err(Error::BadPadding { offset: at });
        }
        if self.group.len() == 3 && self.group[2].0 == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset: at });
        }
        let pos = self.group.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: at });
            }
            if pos < 2 {
                return Err(Error::BadPadding { offset: at });
            }
            if pos == 2 {
                self.group.push((byte, at));
            } else {
                if self.group[2].0 != b'=' {
                    if self.group.iter().any(|(b, _)| value(self.options, *b).is_none()) {
                        return Err(Error::InvalidByte { offset: at, byte });
                    }
                }
                self.group.push((byte, at));
                self.decode_group(out)?;
            }
            return Ok(());
        }
        if value(self.options, byte).is_none() {
            return Err(Error::InvalidByte { offset: at, byte });
        }
        self.group.push((byte, at));
        if self.group.len() == 4 {
            self.decode_group(out)?;
        }
        Ok(())
    }

    /// Decodes newly completed groups from this chunk.
    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            let e = Error::AlreadyFinished;
            self.sticky = Some(e.clone());
            return Err(e);
        }
        let mut out = Vec::new();
        for &byte in input {
            let at = self.offset;
            self.offset += 1;

            match self.options.whitespace {
                Whitespace::None => {
                    if let Err(e) = self.scan_symbol(byte, at, &mut out) {
                        return Err(self.fail(e));
                    }
                }
                Whitespace::IgnoreAscii => {
                    if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                        continue;
                    }
                    if let Err(e) = self.scan_symbol(byte, at, &mut out) {
                        return Err(self.fail(e));
                    }
                }
                Whitespace::Wrapped { width, ending } => {
                    match self.frame_state {
                        FrameState::NeedLf { cr_offset } => {
                            if byte != b'\n' {
                                return Err(self.fail(Error::BadWrap { offset: at }));
                            }
                            self.frame_state = FrameState::NeedSymbol {
                                start_offset: cr_offset,
                            };
                            continue;
                        }
                        FrameState::NeedSymbol { .. } => {
                            self.frame_state = FrameState::Data;
                            self.row_symbols = 0;
                            if byte == b'\n' || byte == b'\r' || byte == b' ' || byte == b'\t' {
                                return Err(self.fail(Error::BadWrap { offset: at }));
                            }
                        }
                        FrameState::Data => {}
                    }

                    if self.row_symbols == width {
                        match ending {
                            LineEnding::Lf if byte == b'\n' => {
                                self.frame_state = FrameState::NeedSymbol { start_offset: at };
                                continue;
                            }
                            LineEnding::CrLf if byte == b'\r' => {
                                self.frame_state = FrameState::NeedLf { cr_offset: at };
                                continue;
                            }
                            _ => return Err(self.fail(Error::BadWrap { offset: at })),
                        }
                    }
                    if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                        return Err(self.fail(Error::BadWrap { offset: at }));
                    }
                    self.row_symbols += 1;
                    if let Err(e) = self.scan_symbol(byte, at, &mut out) {
                        return Err(self.fail(e));
                    }
                }
            }
        }
        Ok(out)
    }

    /// Validates and emits a permitted residual group, then closes the decoder.
    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            let e = Error::AlreadyFinished;
            self.sticky = Some(e.clone());
            return Err(e);
        }
        self.finished = true;
        if let FrameState::NeedLf { cr_offset } = self.frame_state {
            return Err(self.fail(Error::BadWrap { offset: cr_offset }));
        }
        if let FrameState::NeedSymbol { start_offset } = self.frame_state {
            return Err(self.fail(Error::BadWrap { offset: start_offset }));
        }
        if self.group.len() == 3 && self.group[2].0 == b'=' {
            return Err(self.fail(Error::BadPadding {
                offset: self.group[2].1,
            }));
        }
        let mut out = Vec::new();
        match self.group.len() {
            0 => {}
            1 => return Err(self.fail(Error::Truncated { offset: self.group[0].1 })),
            2 | 3 if self.options.padding != Padding::Required => {
                let a = value(self.options, self.group[0].0).unwrap();
                let b = value(self.options, self.group[1].0).unwrap();
                out.push((a << 2) | (b >> 4));
                if self.group.len() == 3 {
                    let c = value(self.options, self.group[2].0).unwrap();
                    out.push((b << 4) | (c >> 2));
                }
            }
            _ => return Err(self.fail(Error::Truncated { offset: self.group[0].1 })),
        }
        self.group.clear();
        Ok(out)
    }
}

/// Decodes encoded bytes, returning an error without exposing partial output.
pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

#[cfg(test)]
mod tests {
    use super::*;

    fn opts() -> Options {
        Options {
            alphabet: Alphabet::Standard,
            padding: Padding::Required,
            whitespace: Whitespace::None,
        }
    }

    #[test]
    fn normal_round_trip() {
        let data = b"Rust";
        let encoded = encode(data, opts()).unwrap();
        assert_eq!(encoded, "UnVzdA==");
        assert_eq!(decode(encoded.as_bytes(), opts()).unwrap(), data);
    }

    #[test]
    fn invalid_and_truncated_inputs() {
        assert_eq!(decode(b"=AAA", opts()), Err(Error::BadPadding { offset: 0 }));
        assert_eq!(decode(b"Y", opts()), Err(Error::Truncated { offset: 0 }));
        assert_eq!(
            decode(b"YWJj$", opts()),
            Err(Error::InvalidByte {
                offset: 4,
                byte: b'$'
            })
        );
    }

    #[test]
    fn streaming_and_wrapped_round_trip() {
        let mut options = opts();
        options.whitespace = Whitespace::Wrapped {
            width: 4,
            ending: LineEnding::Lf,
        };
        let mut encoder = Encoder::new(options).unwrap();
        let mut text = encoder.push(b"Qrs").unwrap();
        text.push_str(&encoder.push(b"T").unwrap());
        text.push_str(&encoder.finish().unwrap());
        assert_eq!(text, "UXJz\nVA==");
        assert_eq!(decode(text.as_bytes(), options).unwrap(), b"QrsT");
    }

    #[test]
    fn padded_tail_and_sticky_decoder_error() {
        let mut decoder = Decoder::new(opts()).unwrap();
        assert_eq!(decoder.push(b"YQ==").unwrap(), b"a");
        assert_eq!(
            decoder.push(b"!"),
            Err(Error::BadPadding { offset: 4 })
        );
        assert_eq!(
            decoder.finish(),
            Err(Error::BadPadding { offset: 4 })
        );
    }
}
