#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    const STD: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const URL: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
    let table = match alphabet {
        Alphabet::Standard => STD,
        Alphabet::UrlSafe => URL,
    };

    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };
    let mut out = String::with_capacity(4);
    out.push(table[(a >> 2) as usize] as char);
    out.push(table[(((a & 3) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(table[(((b & 15) << 2) | (c >> 6)) as usize] as char);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(table[(c & 63) as usize] as char);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }
    out
}

/// Encodes bytes using the selected alphabet, padding, and whitespace policy.
pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut out = encoder.push(input)?;
    out.push_str(&encoder.finish()?);
    Ok(out)
}

/// Decodes bytes using the selected alphabet, padding, and whitespace policy.
pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    /// Creates a new open encoder.
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            finished: false,
        })
    }

    /// Encodes newly available complete three-byte groups.
    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut bytes = std::mem::take(&mut self.tail);
        bytes.extend_from_slice(input);
        let complete = bytes.len() / 3 * 3;
        let mut raw = String::new();
        for chunk in bytes[..complete].chunks_exact(3) {
            raw.push_str(&encode_group(chunk, self.options.alphabet, self.options.padding));
        }
        self.tail.extend_from_slice(&bytes[complete..]);
        Ok(self.wrap_output(raw))
    }

    /// Encodes any remaining bytes and closes the encoder.
    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let raw = if self.tail.is_empty() {
            String::new()
        } else {
            encode_group(&self.tail, self.options.alphabet, self.options.padding)
        };
        self.tail.clear();
        Ok(self.wrap_output(raw))
    }

    fn wrap_output(&self, raw: String) -> String {
        let Whitespace::Wrapped { width, ending } = self.options.whitespace else {
            return raw;
        };
        let separator = match ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        };
        let mut out = String::new();
        for (i, ch) in raw.chars().enumerate() {
            if i > 0 && i % width == 0 {
                out.push_str(separator);
            }
            out.push(ch);
        }
        out
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum FrameState {
    Row,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: Vec<Symbol>,
    terminal: bool,
    offset: usize,
    finished: bool,
    sticky: Option<Error>,
    row_symbols: usize,
    frame: FrameState,
}

impl Decoder {
    /// Creates a new open decoder.
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            terminal: false,
            offset: 0,
            finished: false,
            sticky: None,
            row_symbols: 0,
            frame: FrameState::Row,
        })
    }

    /// Scans encoded bytes and returns newly completed decoded groups.
    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(err) = &self.sticky {
            return Err(err.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let at = self.offset;
            self.offset += 1;
            let result = self.process_byte(byte, at, &mut output);
            if let Err(err) = result {
                self.sticky = Some(err.clone());
                return Err(err);
            }
        }
        Ok(output)
    }

    /// Completes any permitted residual group and closes the decoder.
    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(err) = &self.sticky {
            return Err(err.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { ending, .. } = self.options.whitespace {
            let framing_error = match self.frame {
                FrameState::NeedLf { cr_offset } => Some(Error::BadWrap { offset: cr_offset }),
                FrameState::NeedSymbol { separator_offset } => {
                    Some(Error::BadWrap { offset: separator_offset })
                }
                FrameState::Row => {
                    let _ = ending;
                    None
                }
            };
            if let Some(err) = framing_error {
                self.sticky = Some(err.clone());
                return Err(err);
            }
        }

        let result = self.finish_group();
        match result {
            Ok(out) => {
                self.finished = true;
                Ok(out)
            }
            Err(err) => {
                self.sticky = Some(err.clone());
                Err(err)
            }
        }
    }

    fn process_byte(
        &mut self,
        byte: u8,
        at: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.process_symbol(byte, at, output),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.process_symbol(byte, at, output)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.frame {
                    FrameState::NeedLf { cr_offset: _ } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset: at });
                        }
                        self.frame = FrameState::NeedSymbol {
                            separator_offset: at.saturating_sub(1),
                        };
                        Ok(())
                    }
                    FrameState::NeedSymbol { .. } => {
                        self.frame = FrameState::Row;
                        self.row_symbols = 0;
                        self.process_wrapped_symbol(byte, at, width, ending, output)
                    }
                    FrameState::Row => {
                        if self.row_symbols == width {
                            let sep_offset = at;
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.frame = FrameState::NeedSymbol {
                                        separator_offset: sep_offset,
                                    };
                                    Ok(())
                                }
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.frame = FrameState::NeedLf { cr_offset: at };
                                    Ok(())
                                }
                                _ => Err(Error::BadWrap { offset: at }),
                            }
                        } else {
                            if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                                return Err(Error::BadWrap { offset: at });
                            }
                            self.process_wrapped_symbol(byte, at, width, ending, output)
                        }
                    }
                }
            }
        }
    }

    fn process_wrapped_symbol(
        &mut self,
        byte: u8,
        at: usize,
        _width: usize,
        _ending: LineEnding,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        self.row_symbols += 1;
        self.process_symbol(byte, at, output)
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        at: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal {
            return Err(Error::BadPadding { offset: at });
        }

        if self.group.len() == 3 && self.group[2].byte == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset: at });
        }

        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: at });
            }
            match self.group.len() {
                0 | 1 => return Err(Error::BadPadding { offset: at }),
                2 => {
                    self.group.push(Symbol { byte, offset: at });
                    return Ok(());
                }
                3 => {
                    if self.group[2].byte == b'=' {
                        let a = alphabet_digit(self.group[0].byte, self.options.alphabet)
                            .expect("validated symbol");
                        let b = alphabet_digit(self.group[1].byte, self.options.alphabet)
                            .expect("validated symbol");
                        output.push((a << 2) | (b >> 4));
                        self.group.clear();
                        self.terminal = true;
                        return Ok(());
                    }
                    let a = alphabet_digit(self.group[0].byte, self.options.alphabet)
                        .expect("validated symbol");
                    let b = alphabet_digit(self.group[1].byte, self.options.alphabet)
                        .expect("validated symbol");
                    let c = alphabet_digit(self.group[2].byte, self.options.alphabet)
                        .expect("validated symbol");
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                    self.group.clear();
                    self.terminal = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = alphabet_digit(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset: at, byte })?;
        self.group.push(Symbol { byte, offset: at });
        if self.group.len() == 4 {
            let a = alphabet_digit(self.group[0].byte, self.options.alphabet)
                .expect("validated symbol");
            let b = alphabet_digit(self.group[1].byte, self.options.alphabet)
                .expect("validated symbol");
            let c = alphabet_digit(self.group[2].byte, self.options.alphabet)
                .expect("validated symbol");
            let d = value;
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
            self.group.clear();
        }
        Ok(())
    }

    fn finish_group(&mut self) -> Result<Vec<u8>, Error> {
        let mut out = Vec::new();
        if self.group.is_empty() {
            return Ok(out);
        }
        if self.group.len() == 3 && self.group[2].byte == b'=' {
            return Err(Error::BadPadding {
                offset: self.group[2].offset,
            });
        }
        match self.options.padding {
            Padding::Required => {
                return Err(Error::Truncated {
                    offset: self.group[0].offset,
                });
            }
            Padding::Forbidden | Padding::Optional => match self.group.len() {
                1 => {
                    return Err(Error::Truncated {
                        offset: self.group[0].offset,
                    });
                }
                2 => {
                    let a = alphabet_digit(self.group[0].byte, self.options.alphabet)
                        .expect("validated symbol");
                    let b = alphabet_digit(self.group[1].byte, self.options.alphabet)
                        .expect("validated symbol");
                    out.push((a << 2) | (b >> 4));
                }
                3 => {
                    let a = alphabet_digit(self.group[0].byte, self.options.alphabet)
                        .expect("validated symbol");
                    let b = alphabet_digit(self.group[1].byte, self.options.alphabet)
                        .expect("validated symbol");
                    let c = alphabet_digit(self.group[2].byte, self.options.alphabet)
                        .expect("validated symbol");
                    out.push((a << 2) | (b >> 4));
                    out.push((b << 4) | (c >> 2));
                }
                _ => unreachable!(),
            },
        }
        self.group.clear();
        Ok(out)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn opts() -> Options {
        Options {
            alphabet: Alphabet::Standard,
            padding: Padding::Required,
            whitespace: Whitespace::None,
        }
    }

    #[test]
    fn normal_round_trip() {
        assert_eq!(encode(b"Cat", opts()).unwrap(), "Q2F0");
        assert_eq!(decode(b"Q2F0", opts()).unwrap(), b"Cat");
    }

    #[test]
    fn invalid_and_truncated_inputs() {
        assert_eq!(
            decode(b"@@", opts()),
            Err(Error::BadPadding { offset: 0 })
        );
        assert_eq!(
            decode(b"QQ", opts()),
            Err(Error::Truncated { offset: 0 })
        );
    }

    #[test]
    fn streaming_and_wrapped_round_trip() {
        let options = Options {
            alphabet: Alphabet::Standard,
            padding: Padding::Required,
            whitespace: Whitespace::Wrapped {
                width: 4,
                ending: LineEnding::Lf,
            },
        };
        let mut encoder = Encoder::new(options).unwrap();
        let mut encoded = encoder.push(b"Qrs") .unwrap();
        encoded.push_str(&encoder.push(b"T").unwrap());
        encoded.push_str(&encoder.finish().unwrap());
        assert_eq!(encoded, "UXJz\nVA==");
        assert_eq!(decode(encoded.as_bytes(), options).unwrap(), b"QrsT");
    }

    #[test]
    fn padded_tail_and_sticky_decoder_error() {
        assert_eq!(decode(b"Zh==", opts()).unwrap(), b"f");
        let mut decoder = Decoder::new(opts()).unwrap();
        let err = decoder.push(b"QQ=").unwrap_err();
        assert_eq!(err, Error::BadPadding { offset: 2 });
        assert_eq!(decoder.finish(), Err(err));
    }
}
