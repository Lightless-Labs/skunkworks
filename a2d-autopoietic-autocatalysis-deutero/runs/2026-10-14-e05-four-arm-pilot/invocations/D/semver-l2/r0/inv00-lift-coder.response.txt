use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for (field, slot) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let value = s[start..pos]
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;
        *slot = value;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let plus = bytes[pos..].iter().position(|&b| b == b'+').map(|i| pos + i);
        let pre_end = plus.unwrap_or(bytes.len());
        parse_identifiers(s, pre_start, pre_end, true, &mut pre)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    check_numeric_zero: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    if start == end {
        return Err(VersionError::EmptyIdentifier { at: start });
    }

    let bytes = s.as_bytes();
    let mut ident_start = start;
    let mut i = start;

    while i <= end {
        if i == end || bytes[i] == b'.' {
            if i == ident_start {
                return Err(VersionError::EmptyIdentifier { at: ident_start });
            }
            let ident = &s[ident_start..i];
            if check_numeric_zero
                && ident.len() > 1
                && ident.as_bytes()[0] == b'0'
                && ident.bytes().all(|b| b.is_ascii_digit())
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
            out.push(ident.to_string());
            ident_start = i + 1;
        } else if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = x.bytes().all(|c| c.is_ascii_digit());
        let yn = y.bytes().all(|c| c.is_ascii_digit());
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    constraints: Vec<Constraint>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Constraint {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut arm_tokens = Vec::new();
    let mut last_separator_end = None;

    for token in tokens {
        if token.text == "||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at: token.start });
            }
            arms.push(parse_arm(&arm_tokens)?);
            arm_tokens.clear();
            last_separator_end = Some(token.start + 2);
        } else {
            arm_tokens.push(token);
        }
    }

    if arm_tokens.is_empty() {
        return Err(RangeError::Empty {
            at: last_separator_end.unwrap_or(s.len()),
        });
    }
    arms.push(parse_arm(&arm_tokens)?);
    Ok(Range { arms })
}

fn tokenize(s: &str) -> Vec<Token> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if is_range_ws(bytes[i]) {
            i += 1;
            continue;
        }
        let start = i;
        while i < bytes.len() && !is_range_ws(bytes[i]) {
            i += 1;
        }
        out.push(Token {
            text: s[start..i].to_string(),
            start,
        });
    }
    out
}

fn is_range_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn parse_arm(tokens: &[Token]) -> Result<Arm, RangeError> {
    if let Some(t) = tokens.iter().find(|t| t.text == "-") {
        if tokens.len() != 3 || tokens[1].text != "-" {
            return Err(RangeError::Syntax { at: t.start });
        }
        return parse_hyphen_arm(&tokens[0], &tokens[2]);
    }

    let mut arm = Arm {
        constraints: Vec::new(),
        anchors: Vec::new(),
    };
    for token in tokens {
        parse_token(token, &mut arm)?;
    }
    Ok(arm)
}

fn parse_hyphen_arm(lower: &Token, upper: &Token) -> Result<Arm, RangeError> {
    let lo = parse_partial_operand(lower, &lower.text)?;
    let hi = parse_partial_operand(upper, &upper.text)?;

    if lo.numeric_count == 0 {
        return Err(RangeError::Syntax { at: lower.start });
    }
    if hi.numeric_count == 0 {
        return Err(RangeError::Syntax { at: upper.start });
    }

    let lo_version = lo.lower;
    let hi_version = if hi.full {
        hi.lower
    } else {
        next_partial_bound(&hi, upper.start, true)?
    };

    let mut arm = Arm {
        constraints: vec![
            Constraint {
                op: Op::Ge,
                version: lo_version.clone(),
            },
            Constraint {
                op: if hi.full { Op::Le } else { Op::Lt },
                version: hi_version,
            },
        ],
        anchors: Vec::new(),
    };
    add_anchor(&mut arm, &lo_version);
    if hi.full {
        add_anchor(&mut arm, &hi.lower);
    }
    Ok(arm)
}

fn parse_token(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let text = token.text.as_str();
    if let Some((op, n)) = comparator_prefix(text) {
        let operand = &text[n..];
        let operand_start = token.start + n;
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token.start });
        }
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        arm.constraints.push(Constraint { op, version: version.clone() });
        add_anchor(arm, &version);
        return Ok(());
    }

    let (prefix, operand) = if text.starts_with('^') || text.starts_with('~') {
        (&text[..1], &text[1..])
    } else {
        ("", text)
    };
    if !prefix.is_empty() {
        let operand_start = token.start + 1;
        let partial = parse_partial_operand(token, operand)?;
        if partial.numeric_count == 0 {
            return Err(RangeError::Syntax { at: token.start });
        }
        arm.constraints.push(Constraint {
            op: Op::Ge,
            version: partial.lower.clone(),
        });
        add_anchor(arm, &partial.lower);
        let upper = if prefix == "^" {
            caret_upper(&partial, token.start)?
        } else {
            tilde_upper(&partial, token.start)?
        };
        arm.constraints.push(Constraint { op: Op::Lt, version: upper });
        let _ = operand_start;
        return Ok(());
    }

    if !is_valid_token_start(text) {
        return Err(RangeError::Syntax { at: token.start });
    }

    let partial = parse_partial_operand(token, text)?;
    if partial.full {
        arm.constraints.push(Constraint {
            op: Op::Eq,
            version: partial.lower.clone(),
        });
        add_anchor(arm, &partial.lower);
    } else if partial.numeric_count == 1 || partial.numeric_count == 2 {
        arm.constraints.push(Constraint {
            op: Op::Ge,
            version: partial.lower,
        });
        let upper = next_partial_bound(&partial, token.start, true)?;
        arm.constraints.push(Constraint { op: Op::Lt, version: upper });
    }
    Ok(())
}

fn comparator_prefix(s: &str) -> Option<(Op, usize)> {
    if s.starts_with(">=") {
        Some((Op::Ge, 2))
    } else if s.starts_with("<=") {
        Some((Op::Le, 2))
    } else if s.starts_with('>') {
        Some((Op::Gt, 1))
    } else if s.starts_with('<') {
        Some((Op::Lt, 1))
    } else if s.starts_with('=') {
        Some((Op::Eq, 1))
    } else {
        None
    }
}

fn is_valid_token_start(s: &str) -> bool {
    s.as_bytes()
        .first()
        .map(|b| b.is_ascii_digit() || matches!(b, b'*' | b'x' | b'X'))
        .unwrap_or(false)
}

#[derive(Clone)]
struct Partial {
    lower: Version,
    numeric_count: usize,
    full: bool,
    fields: Vec<Option<u64>>,
}

fn parse_partial_operand(token: &Token, operand: &str) -> Result<Partial, RangeError> {
    let has_suffix = operand.contains('-') || operand.contains('+');
    let raw_fields: Vec<&str> = operand.split('.').collect();
    let any_wild = raw_fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));

    if has_suffix || (raw_fields.len() >= 3 && !any_wild) {
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: token.start + (token.text.len() - operand.len()),
            error,
        })?;
        return Ok(Partial {
            fields: vec![Some(version.major), Some(version.minor), Some(version.patch)],
            numeric_count: 3,
            full: true,
            lower: version,
        });
    }

    if raw_fields.is_empty() || raw_fields.len() > 3 || operand.is_empty() {
        return Err(RangeError::Syntax { at: token.start });
    }

    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    let mut numeric_count = 0;
    for field in raw_fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at: token.start });
        }
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
            fields.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: token.start });
            }
            let value = field
                .parse::<u64>()
                .map_err(|_| RangeError::Syntax { at: token.start })?;
            fields.push(Some(value));
            numeric_count += 1;
        }
    }

    while fields.len() < 3 {
        fields.push(None);
    }
    let mut core = [0u64; 3];
    for i in 0..3 {
        if let Some(v) = fields[i] {
            core[i] = v;
        }
    }
    Ok(Partial {
        lower: Version {
            major: core[0],
            minor: core[1],
            patch: core[2],
            pre: Vec::new(),
            build: Vec::new(),
        },
        numeric_count,
        full: false,
        fields,
    })
}

fn add_anchor(arm: &mut Arm, version: &Version) {
    if !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
}

fn next_partial_bound(p: &Partial, at: usize, implicit: bool) -> Result<Version, RangeError> {
    let k = p.numeric_count;
    let mut major = p.lower.major;
    let mut minor = p.lower.minor;
    let mut patch = p.lower.patch;
    if k == 1 {
        major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
        minor = 0;
        patch = 0;
    } else if k == 2 {
        minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
        patch = 0;
    } else if !implicit {
        patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    })
}

fn caret_upper(p: &Partial, at: usize) -> Result<Version, RangeError> {
    let k = p.numeric_count;
    let (mut major, mut minor, mut patch) = (p.lower.major, p.lower.minor, p.lower.patch);
    if major > 0 || k == 1 {
        major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
        minor = 0;
        patch = 0;
    } else if minor > 0 || k == 2 {
        minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
        patch = 0;
    } else {
        patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    }
    Ok(Version { major, minor, patch, pre: vec!["0".into()], build: Vec::new() })
}

fn tilde_upper(p: &Partial, at: usize) -> Result<Version, RangeError> {
    let k = p.numeric_count;
    let (mut major, mut minor, patch) = (p.lower.major, p.lower.minor, p.lower.patch);
    if k == 1 {
        major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
        minor = 0;
        Ok(Version { major, minor, patch: 0, pre: vec!["0".into()], build: Vec::new() })
    } else {
        minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
        Ok(Version { major, minor, patch: 0, pre: vec!["0".into()], build: Vec::new() })
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.iter().any(|&(ma, mi, pa)| (ma, mi, pa) == (v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.constraints.iter().all(|constraint| {
            let ord = compare_versions(v, &constraint.version);
            match constraint.op {
                Op::Eq => ord == Ordering::Equal,
                Op::Lt => ord == Ordering::Less,
                Op::Le => ord != Ordering::Greater,
                Op::Gt => ord == Ordering::Greater,
                Op::Ge => ord != Ordering::Less,
            }
        })
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        best = match best {
            None => Some(index),
            Some(current) => {
                let ord = compare_versions(version, &parsed[current]);
                if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                    Some(index)
                } else {
                    Some(current)
                }
            }
        };
    }
    Ok(best.map(|index| candidates[index].clone()))
}
