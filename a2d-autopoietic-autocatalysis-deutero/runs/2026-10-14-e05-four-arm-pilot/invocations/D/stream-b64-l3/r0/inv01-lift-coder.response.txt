#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn symbol(value: u8, alphabet: Alphabet) -> u8 {
    const STANDARD: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const URL_SAFE: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";

    match alphabet {
        Alphabet::Standard => STANDARD[value as usize],
        Alphabet::UrlSafe => URL_SAFE[value as usize],
    }
}

fn encode_group(bytes: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };
    let mut out = String::with_capacity(4);

    out.push(symbol(a >> 2, alphabet) as char);
    out.push(symbol(((a & 0x03) << 4) | (b >> 4), alphabet) as char);

    if bytes.len() > 1 {
        out.push(symbol(((b & 0x0f) << 2) | (c >> 6), alphabet) as char);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(symbol(c & 0x3f, alphabet) as char);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }

    out
}

fn wrap_encoded(raw: String, whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return raw;
    };

    if raw.len() <= width {
        return raw;
    }

    let separator = match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };
    let mut out = String::with_capacity(raw.len() + raw.len() / width * separator.len());

    for (index, chunk) in raw.as_bytes().chunks(width).enumerate() {
        if index > 0 {
            out.push_str(separator);
        }
        out.push_str(std::str::from_utf8(chunk).unwrap());
    }

    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            pending: Vec::with_capacity(2),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut bytes = Vec::with_capacity(self.pending.len() + input.len());
        bytes.extend_from_slice(&self.pending);
        bytes.extend_from_slice(input);

        let complete_len = bytes.len() / 3 * 3;
        self.pending.clear();
        self.pending.extend_from_slice(&bytes[complete_len..]);

        let mut raw = String::new();
        for chunk in bytes[..complete_len].chunks_exact(3) {
            raw.push_str(&encode_group(
                chunk,
                self.options.alphabet,
                self.options.padding,
            ));
        }
        Ok(wrap_encoded(raw, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let raw = if self.pending.is_empty() {
            String::new()
        } else {
            encode_group(
                &self.pending,
                self.options.alphabet,
                self.options.padding,
            )
        };
        self.pending.clear();
        Ok(wrap_encoded(raw, self.options.whitespace))
    }
}

#[derive(Clone, Copy)]
struct Token {
    byte: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum FrameState {
    Row,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: Vec<Token>,
    total_offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
    terminal_padding: bool,
    frame_state: FrameState,
    row_symbols: usize,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            total_offset: 0,
            finished: false,
            sticky_error: None,
            terminal_padding: false,
            frame_state: FrameState::Row,
            row_symbols: 0,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        let start = self.total_offset;

        for (index, &byte) in input.iter().enumerate() {
            let offset = start + index;
            if let Err(error) = self.process_byte(byte, offset, &mut out) {
                self.total_offset = offset + 1;
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        self.total_offset += input.len();
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let frame_error = match self.frame_state {
            FrameState::NeedLf { cr_offset } => Some(Error::BadWrap { offset: cr_offset }),
            FrameState::NeedSymbol { separator_offset } => {
                Some(Error::BadWrap { offset: separator_offset })
            }
            FrameState::Row => None,
        };

        if let Some(error) = frame_error {
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.group.len() == 2 && self.group[1].byte == b'=' {
            let error = Error::BadPadding {
                offset: self.group[1].offset,
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let result = match self.group.len() {
            0 => Ok(Vec::new()),
            1 => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
            2 | 3 if self.options.padding != Padding::Required => {
                let a = digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = digit(self.group[1].byte, self.options.alphabet).unwrap();
                let mut bytes = vec![(a << 2) | (b >> 4)];

                if self.group.len() == 3 {
                    let c = digit(self.group[2].byte, self.options.alphabet).unwrap();
                    bytes.push((b << 4) | (c >> 2));
                }
                Ok(bytes)
            }
            _ => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
        };

        match result {
            Ok(bytes) => {
                self.finished = true;
                Ok(bytes)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }

    fn process_byte(
        &mut self,
        byte: u8,
        offset: usize,
        out: &mut Vec<u8>,
    ) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => {}
            Whitespace::IgnoreAscii if is_ignored(byte) => return Ok(()),
            Whitespace::IgnoreAscii => {}
            Whitespace::Wrapped { width, ending } => match self.frame_state {
                FrameState::NeedLf { .. } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.frame_state = FrameState::NeedSymbol {
                        separator_offset: match self.frame_state {
                            FrameState::NeedLf { cr_offset } => cr_offset,
                            _ => unreachable!(),
                        },
                    };
                    return Ok(());
                }
                FrameState::NeedSymbol { .. } => {
                    if byte == b'\r' || byte == b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.frame_state = FrameState::Row;
                    self.row_symbols = 0;
                }
                FrameState::Row => {
                    if self.row_symbols == width {
                        match ending {
                            LineEnding::Lf if byte == b'\n' => {
                                self.frame_state = FrameState::NeedSymbol {
                                    separator_offset: offset,
                                };
                                return Ok(());
                            }
                            LineEnding::CrLf if byte == b'\r' => {
                                self.frame_state = FrameState::NeedLf { cr_offset: offset };
                                return Ok(());
                            }
                            _ => return Err(Error::BadWrap { offset }),
                        }
                    } else if byte == b'\r' || byte == b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                }
            },
        }

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group.len() == 2 && self.group[1].byte == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        match self.group.len() {
            0 | 1 => {
                if byte == b'=' {
                    return Err(Error::BadPadding { offset });
                }
                if digit(byte, self.options.alphabet).is_none() {
                    return Err(Error::InvalidByte { offset, byte });
                }
                self.group.push(Token { byte, offset });
                self.row_symbols += 1;
            }
            2 => {
                if byte == b'=' {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.group.push(Token { byte, offset });
                } else {
                    if digit(byte, self.options.alphabet).is_none() {
                        return Err(Error::InvalidByte { offset, byte });
                    }
                    self.group.push(Token { byte, offset });
                }
                self.row_symbols += 1;
            }
            3 => {
                if self.group[2].byte == b'=' {
                    if byte != b'=' {
                        return Err(Error::BadPadding { offset });
                    }
                    let a = digit(self.group[0].byte, self.options.alphabet).unwrap();
                    let b = digit(self.group[1].byte, self.options.alphabet).unwrap();
                    out.push((a << 2) | (b >> 4));
                    self.group.clear();
                    self.terminal_padding = true;
                    self.row_symbols += 1;
                    return Ok(());
                }

                if byte == b'=' {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    let a = digit(self.group[0].byte, self.options.alphabet).unwrap();
                    let b = digit(self.group[1].byte, self.options.alphabet).unwrap();
                    let c = digit(self.group[2].byte, self.options.alphabet).unwrap();
                    out.push((a << 2) | (b >> 4));
                    out.push((b << 4) | (c >> 2));
                    self.group.clear();
                    self.terminal_padding = true;
                    self.row_symbols += 1;
                    return Ok(());
                }

                let Some(d) = digit(byte, self.options.alphabet) else {
                    return Err(Error::InvalidByte { offset, byte });
                };
                let a = digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = digit(self.group[1].byte, self.options.alphabet).unwrap();
                let c = digit(self.group[2].byte, self.options.alphabet).unwrap();
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                out.push((c << 6) | d);
                self.group.clear();
                self.row_symbols += 1;
            }
            _ => unreachable!(),
        }

        Ok(())
    }
}

fn is_ignored(byte: u8) -> bool {
    matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}
