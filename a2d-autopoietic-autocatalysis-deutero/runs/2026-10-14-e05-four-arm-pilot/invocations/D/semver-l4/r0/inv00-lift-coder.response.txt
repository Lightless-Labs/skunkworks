use std::cmp::Ordering;
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_field(bytes: &[u8], pos: &mut usize, source_len: usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if start == *pos {
        if start == source_len || matches!(bytes.get(start), Some(b'.' | b'-' | b'+')) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let n = std::str::from_utf8(&bytes[start..*pos])
        .ok()
        .and_then(|s| s.parse::<u64>().ok())
        .ok_or(VersionError::Overflow { at: start })?;
    Ok(n)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    allow_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;
    let mut i = start;
    loop {
        let end = if i == bytes.len() { i } else { i };
        if end == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        for j in id_start..end {
            let b = bytes[j];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: j });
            }
        }
        if allow_numeric_leading_zero
            && end - id_start > 1
            && bytes[id_start..end].iter().all(|b| is_digit(*b))
            && bytes[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(s[id_start..end].to_string());
        if i == bytes.len() {
            break;
        }
        if bytes[i] != b'.' {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
        id_start = i;
        if i == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: i });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0;
    let major = parse_core_field(bytes, &mut pos, bytes.len())?;
    for _ in 0..2 {
        if pos >= bytes.len() || bytes[pos] != b'.' {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let n = parse_core_field(bytes, &mut pos, bytes.len())?;
        if pos == 0 {
            unreachable!();
        }
        if _ == 0 {
            // The first field after the major is minor.
            let minor = n;
            // Continue with the patch field below.
            if pos >= bytes.len() || bytes[pos] != b'.' {
                if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                    return Err(VersionError::MissingCore { at: pos });
                }
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core_field(bytes, &mut pos, bytes.len())?;
            let (pre, build) = parse_version_tail(s, pos)?;
            return Ok(Version { major, minor, patch, pre, build });
        }
    }
    unreachable!()
}

fn parse_version_tail(s: &str, mut pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos == bytes.len() {
        return Ok((pre, build));
    }
    if bytes[pos] == b'-' {
        let start = pos + 1;
        pos = start;
        while pos < bytes.len() && bytes[pos] != b'+' {
            pos += 1;
        }
        if start == pos {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
        let section = &s[start..pos];
        pre = parse_identifiers(section, 0, true)?;
        if pos < bytes.len() {
            let build_start = pos + 1;
            if build_start == bytes.len() {
                return Err(VersionError::EmptyIdentifier { at: build_start });
            }
            build = parse_identifiers(s, build_start, false)?;
        }
    } else if bytes[pos] == b'+' {
        let start = pos + 1;
        if start == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
        build = parse_identifiers(s, start, false)?;
    } else {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok((pre, build))
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core_tuple(a)
        .cmp(&core_tuple(b))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

fn token_spans(s: &str) -> Vec<(usize, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        out.push((start, i));
    }
    out
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_arms(s: &str) -> Result<Vec<(usize, usize)>, RangeError> {
    let bytes = s.as_bytes();
    let mut spans = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i..].starts_with(b"||")
            && (i == 0 || is_ws(bytes[i - 1]))
            && (i + 2 == bytes.len() || is_ws(bytes[i + 2]))
        {
            let mut left = arm_start;
            let mut right = i;
            while left < right && is_ws(bytes[left]) { left += 1; }
            while right > left && is_ws(bytes[right - 1]) { right -= 1; }
            if left == right {
                return Err(RangeError::Empty { at: i });
            }
            spans.push((left, right));
            i += 2;
            arm_start = i;
        } else {
            i += 1;
        }
    }
    let mut left = arm_start;
    let mut right = bytes.len();
    while left < right && is_ws(bytes[left]) { left += 1; }
    while right > left && is_ws(bytes[right - 1]) { right -= 1; }
    if left == right {
        if spans.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: bytes.len() });
    }
    spans.push((left, right));
    Ok(spans)
}

fn version_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn make_version(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version { major, minor, patch, pre, build: Vec::new() }
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = make_version(v.major, v.minor, v.patch, vec!["0".to_string()]);
    let result = match field {
        0 => v.major.checked_add(1).map(|x| { out.major = x; }),
        1 => v.minor.checked_add(1).map(|x| { out.major = v.major; out.minor = x; }),
        _ => v.patch.checked_add(1).map(|x| { out.major = v.major; out.minor = v.minor; out.patch = x; }),
    };
    if result.is_none() {
        return Err(RangeError::Syntax { at });
    }
    Ok(out)
}

fn push_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core_tuple(&version));
    }
    arm.comparators.push(Comparator { op, version, anchor });
}

fn satisfies_cmp(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn partial_or_full(
    operand: &str,
    operand_at: usize,
) -> Result<(Version, usize, bool), RangeError> {
    let full_like = operand.contains(['-', '+'])
        || (operand.split('.').count() >= 3
            && !operand.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if full_like {
        return Ok((version_operand(operand, operand_at)?, 3, true));
    }
    let fields: Vec<&str> = operand.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: operand_at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.starts_with('0'))
                || f.parse::<u64>().is_err()
            {
                return Err(RangeError::Syntax { at: operand_at });
            }
            nums.push(f.parse::<u64>().unwrap());
        }
    }
    if nums.len() < 3 && wildcard_seen {
        // Explicit wildcard fields are allowed; numeric fields after one were rejected above.
    }
    let k = nums.len();
    let mut vals = [0u64; 3];
    for (i, n) in nums.iter().enumerate() { vals[i] = *n; }
    Ok((make_version(vals[0], vals[1], vals[2], Vec::new()), k, false))
}

fn parse_bare(
    arm: &mut Arm,
    operand: &str,
    at: usize,
) -> Result<(), RangeError> {
    let (v, k, full) = partial_or_full(operand, at)?;
    if full {
        push_cmp(arm, Op::Eq, v, true);
    } else if k == 0 {
        // All-wildcard operand imposes no bounds.
    } else if k == 3 {
        push_cmp(arm, Op::Eq, v, false);
    } else {
        push_cmp(arm, Op::Ge, v.clone(), false);
        let upper = bump(&v, if k == 1 { 0 } else { 1 }, at)?;
        push_cmp(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_caret_tilde(
    arm: &mut Arm,
    operand: &str,
    at: usize,
    caret: bool,
) -> Result<(), RangeError> {
    let (v, k, full) = partial_or_full(operand, at)?;
    if k == 0 {
        return Err(RangeError::Syntax { at: at.saturating_sub(1) });
    }
    push_cmp(arm, Op::Ge, v.clone(), full);
    let field = if caret {
        if v.major > 0 || k == 1 { 0 } else if v.minor > 0 || k == 2 { 1 } else { 2 }
    } else if k == 1 { 0 } else { 1 };
    let upper = bump(&v, field, at)?;
    push_cmp(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_comparator(token: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let bytes = token.as_bytes();
    let (op, n) = if token.starts_with(">=") { (Op::Ge, 2) }
        else if token.starts_with("<=") { (Op::Le, 2) }
        else if token.starts_with(">") { (Op::Gt, 1) }
        else if token.starts_with("<") { (Op::Lt, 1) }
        else if token.starts_with("=") { (Op::Eq, 1) }
        else { return Err(RangeError::Syntax { at: start }); };
    if n >= bytes.len() || !is_digit(bytes[n]) {
        return Err(RangeError::Syntax { at: start });
    }
    let v = version_operand(&token[n..], start + n)?;
    push_cmp(arm, op, v, true);
    Ok(())
}

fn parse_hyphen(tokens: &[(usize, usize)], s: &str, arm: &mut Arm) -> Result<(), RangeError> {
    let minus = tokens.iter().position(|&(a, b)| &s[a..b] == "-").unwrap();
    if tokens.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax { at: tokens[minus].0 });
    }
    let (ls, le) = tokens[0];
    let (us, ue) = tokens[2];
    let (lower, lk, lf) = partial_or_full(&s[ls..le], ls)?;
    let (upper, uk, uf) = partial_or_full(&s[us..ue], us)?;
    if lk == 0 { return Err(RangeError::Syntax { at: ls }); }
    if uk == 0 { return Err(RangeError::Syntax { at: us }); }
    push_cmp(arm, Op::Ge, lower, lf);
    if uf || uk == 3 {
        push_cmp(arm, Op::Le, upper, uf);
    } else {
        let bound = bump(&upper, if uk == 1 { 0 } else { 1 }, us)?;
        push_cmp(arm, Op::Lt, bound, false);
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let arms_spans = split_arms(s)?;
    let mut arms = Vec::new();
    for (a, b) in arms_spans {
        let arm_text = &s[a..b];
        let local = token_spans(arm_text);
        let tokens: Vec<(usize, usize)> = local.iter().map(|(x,y)| (a+x, a+y)).collect();
        let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
        if let Some(idx) = tokens.iter().position(|&(x,y)| &s[x..y] == "-") {
            let _ = idx;
            parse_hyphen(&tokens, s, &mut arm)?;
        } else {
            for &(ts, te) in &tokens {
                let t = &s[ts..te];
                if t.starts_with('^') || t.starts_with('~') {
                    if t.len() == 1 { return Err(RangeError::Syntax { at: ts }); }
                    parse_caret_tilde(&mut arm, &t[1..], ts+1, t.starts_with('^'))?;
                } else if t.starts_with('<') || t.starts_with('>') || t.starts_with('=') {
                    parse_comparator(t, ts, &mut arm)?;
                } else if t.starts_with(|c: char| c.is_ascii_digit())
                    || t == "*" || t == "x" || t == "X"
                {
                    parse_bare(&mut arm, t, ts)?;
                } else {
                    return Err(RangeError::Syntax { at: ts });
                }
            }
        }
        arms.push(arm);
    }
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty() && !include_prerelease && !arm.anchors.contains(&core_tuple(v)) {
        return false;
    }
    arm.comparators.iter().all(|c| satisfies_cmp(v, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates.iter().enumerate().map(|(i, s)| {
        parse_version(s).map_err(|error| CandidateError { index: i, error })
    }).collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug, PartialEq, Eq)]
struct Bound {
    version: Version,
    inclusive: bool,
}

fn arm_bounds(arm: &Arm) -> (Option<Bound>, Option<Bound>, bool) {
    let mut lo: Option<Bound> = None;
    let mut hi: Option<Bound> = None;
    let mut impossible = false;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                let b = Bound { version: c.version.clone(), inclusive: true };
                lo = strongest_lower(lo, Some(b.clone()));
                hi = strongest_upper(hi, Some(b));
            }
            Op::Gt | Op::Ge => lo = strongest_lower(lo, Some(Bound { version: c.version.clone(), inclusive: c.op == Op::Ge })),
            Op::Lt | Op::Le => hi = strongest_upper(hi, Some(Bound { version: c.version.clone(), inclusive: c.op == Op::Le })),
        }
    }
    if let (Some(l), Some(h)) = (&lo, &hi) {
        let o = compare_versions(&l.version, &h.version);
        if o == Ordering::Greater || (o == Ordering::Equal && !(l.inclusive && h.inclusive)) {
            impossible = true;
        }
    }
    (lo, hi, impossible)
}

fn strongest_lower(old: Option<Bound>, new: Option<Bound>) -> Option<Bound> {
    match (old, new) {
        (None, x) | (x, None) => x,
        (Some(a), Some(b)) => match compare_versions(&a.version, &b.version) {
            Ordering::Less => Some(b),
            Ordering::Greater => Some(a),
            Ordering::Equal => Some(Bound { version: a.version, inclusive: a.inclusive && b.inclusive }),
        },
    }
}

fn strongest_upper(old: Option<Bound>, new: Option<Bound>) -> Option<Bound> {
    match (old, new) {
        (None, x) | (x, None) => x,
        (Some(a), Some(b)) => match compare_versions(&a.version, &b.version) {
            Ordering::Less => Some(a),
            Ordering::Greater => Some(b),
            Ordering::Equal => Some(Bound { version: a.version, inclusive: a.inclusive && b.inclusive }),
        },
    }
}

fn bound_text(b: &Option<Bound>, lower: bool) -> String {
    if let Some(b) = b {
        let mut v = b.version.clone();
        v.build.clear();
        let op = if lower {
            if b.inclusive { ">=" } else { ">" }
        } else if b.inclusive { "<=" } else { "<" };
        format!("{}{}", op, version_text(&v))
    } else {
        "*".to_string()
    }
}

fn version_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn core_text(t: &(u64,u64,u64)) -> String {
    format!("{}.{}.{}", t.0, t.1, t.2)
}

fn arm_string(arm: &Arm) -> Option<String> {
    let (lo, hi, impossible) = arm_bounds(arm);
    if impossible { return None; }
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let anchors_text = if anchors.is_empty() { "-".to_string() }
        else { anchors.iter().map(core_text).collect::<Vec<_>>().join(",") };
    Some(format!("[{};{};pre={}]", bound_text(&lo, true), bound_text(&hi, false), anchors_text))
}

pub fn canonical(r: &Range) -> String {
    let mut strings: Vec<String> = r.arms.iter().filter_map(arm_string).collect();
    strings.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() { "EMPTY".to_string() } else { strings.join(" || ") }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let (al, ah, ai) = arm_bounds(aa);
            let (bl, bh, bi) = arm_bounds(bb);
            if ai || bi { continue; }
            let lo = strongest_lower(al, bl);
            let hi = strongest_upper(ah, bh);
            if let (Some(l), Some(h)) = (&lo, &hi) {
                let cmp = compare_versions(&l.version, &h.version);
                if cmp == Ordering::Greater || (cmp == Ordering::Equal && !(l.inclusive && h.inclusive)) {
                    continue;
                }
            }
            if include_prerelease {
                return true;
            }
            let mut anchors = aa.anchors.iter().filter(|t| bb.anchors.contains(t));
            if anchors.any(|t| {
                let v = make_version(t.0, t.1, t.2, vec!["0".to_string()]);
                bounds_match(&v, &lo, &hi)
            }) {
                return true;
            }
            if let Some(v) = stable_witness(&lo, &hi) {
                if bounds_match(&v, &lo, &hi) { return true; }
            }
        }
    }
    false
}

fn bounds_match(v: &Version, lo: &Option<Bound>, hi: &Option<Bound>) -> bool {
    lo.as_ref().map_or(true, |b| {
        let o = compare_versions(v, &b.version);
        o == Ordering::Greater || (o == Ordering::Equal && b.inclusive)
    }) && hi.as_ref().map_or(true, |b| {
        let o = compare_versions(v, &b.version);
        o == Ordering::Less || (o == Ordering::Equal && b.inclusive)
    })
}

fn stable_witness(lo: &Option<Bound>, hi: &Option<Bound>) -> Option<Version> {
    if let Some(l) = lo {
        if l.version.pre.is_empty() && l.inclusive && bounds_match(&l.version, lo, hi) {
            return Some(l.version.clone());
        }
        let (m, n, p) = core_tuple(&l.version);
        if l.version.pre.is_empty() {
            let next = next_core(m, n, p)?;
            let v = make_version(next.0, next.1, next.2, Vec::new());
            if bounds_match(&v, lo, hi) { return Some(v); }
        } else {
            let v = make_version(m, n, p, Vec::new());
            if bounds_match(&v, lo, hi) { return Some(v); }
            let next = next_core(m, n, p)?;
            let v = make_version(next.0, next.1, next.2, Vec::new());
            if bounds_match(&v, lo, hi) { return Some(v); }
        }
    } else if let Some(h) = hi {
        let (m,n,p) = core_tuple(&h.version);
        let v = make_version(m,n,p,Vec::new());
        if bounds_match(&v, lo, hi) { return Some(v); }
    } else {
        return Some(make_version(0,0,0,Vec::new()));
    }
    None
}

fn next_core(m: u64, n: u64, p: u64) -> Option<(u64,u64,u64)> {
    if p < u64::MAX { Some((m,n,p+1)) }
    else if n < u64::MAX { Some((m,n+1,0)) }
    else if m < u64::MAX { Some((m+1,0,0)) }
    else { None }
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| SelectError::InvalidRange {
            package: package.to_string(), label: label.to_string(), path: path.clone(), error,
        }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn expr_matches(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, include)).unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|c| expr_matches(c, v, include)),
        SetExpr::Intersection(children) => children.iter().all(|c| expr_matches(c, v, include)),
    }
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut parsed: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (index, candidate) in p.candidates.iter().enumerate() {
            match parse_version(candidate) {
                Ok(v) => versions.push(v),
                Err(error) => return Err(SelectError::InvalidCandidate {
                    package: p.name.clone(), index, error,
                }),
            }
        }
        parsed.push(versions);
    }
    for p in packages {
        for c in &p.constraints {
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let qualifies = |i: usize| {
            p.constraints.iter().all(|c| expr_matches(&c.expr, &parsed[pi][i], include_prerelease))
        };
        let mut best: Option<usize> = None;
        for i in 0..p.candidates.len() {
            if !qualifies(i) { continue; }
            let Some(j) = best else { best = Some(i); continue; };
            let a = &parsed[pi][i];
            let b = &parsed[pi][j];
            let choose = match preference {
                Preference::Highest => compare_versions(a,b) == Ordering::Greater,
                Preference::Lowest => compare_versions(a,b) == Ordering::Less,
                Preference::StableFirst => {
                    match (a.pre.is_empty(), b.pre.is_empty()) {
                        (true, false) => true,
                        (false, true) => false,
                        _ => compare_versions(a,b) == Ordering::Greater,
                    }
                }
            };
            if choose { best = Some(i); }
        }
        if let Some(i) = best {
            selections.push(Selection { package: p.name.clone(), version: p.candidates[i].clone() });
        } else {
            let labels = minimum_conflict(p, &parsed[pi], include_prerelease);
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
    }
    Ok(selections)
}

fn minimum_conflict(p: &Package, versions: &[Version], include: bool) -> Vec<String> {
    if versions.is_empty() { return Vec::new(); }
    let n = p.constraints.len();
    for size in 1..=n {
        let mut chosen = Vec::new();
        if let Some(indices) = find_conflict(p, versions, include, size, 0, &mut chosen) {
            let mut labels: Vec<String> = indices.iter().map(|&i| p.constraints[i].label.clone()).collect();
            labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
            return labels;
        }
    }
    Vec::new()
}

fn find_conflict(
    p: &Package,
    versions: &[Version],
    include: bool,
    target: usize,
    start: usize,
    chosen: &mut Vec<usize>,
) -> Option<Vec<usize>> {
    if chosen.len() == target {
        let any = versions.iter().any(|v| chosen.iter().all(|&i| expr_matches(&p.constraints[i].expr, v, include)));
        return if any { None } else { Some(chosen.clone()) };
    }
    for i in start..p.constraints.len() {
        chosen.push(i);
        if let Some(c) = find_conflict(p, versions, include, target, i+1, chosen) {
            return Some(c);
        }
        chosen.pop();
    }
    None
}
