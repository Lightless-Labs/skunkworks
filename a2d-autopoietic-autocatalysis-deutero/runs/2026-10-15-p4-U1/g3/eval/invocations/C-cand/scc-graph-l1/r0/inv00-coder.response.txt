pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: target,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut self_loop = vec![false; n];
    for &(source, target) in edges {
        adjacency[source].push(target);
        if source == target {
            self_loop[source] = true;
        }
    }

    // Iterative Kosaraju pass one: record vertices in finishing order.
    let mut visited = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for start in 0..n {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_edge)) = stack.last_mut() {
            if *next_edge < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_edge];
                *next_edge += 1;
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                order.push(finished);
            }
        }
    }

    let mut reverse = vec![Vec::new(); n];
    for &(source, target) in edges {
        reverse[target].push(source);
    }

    // Pass two assigns provisional SCCs in reverse finishing order.
    let mut provisional_of = vec![usize::MAX; n];
    let mut provisional_components: Vec<Vec<usize>> = Vec::new();
    for &start in order.iter().rev() {
        if provisional_of[start] != usize::MAX {
            continue;
        }

        let id = provisional_components.len();
        let mut component = Vec::new();
        let mut stack = vec![start];
        provisional_of[start] = id;

        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &neighbor in &reverse[vertex] {
                if provisional_of[neighbor] == usize::MAX {
                    provisional_of[neighbor] = id;
                    stack.push(neighbor);
                }
            }
        }
        component.sort_unstable();
        provisional_components.push(component);
    }

    let mut provisional_cyclic = Vec::with_capacity(provisional_components.len());
    for component in &provisional_components {
        provisional_cyclic.push(
            component.len() >= 2 || self_loop[component[0]],
        );
    }

    let mut ids: Vec<usize> = (0..provisional_components.len()).collect();
    ids.sort_unstable_by(|&a, &b| {
        provisional_components[b]
            .len()
            .cmp(&provisional_components[a].len())
            .then_with(|| {
                provisional_components[a][0].cmp(&provisional_components[b][0])
            })
    });

    let mut canonical_id = vec![0usize; provisional_components.len()];
    let mut components = Vec::with_capacity(provisional_components.len());
    let mut cyclic = Vec::with_capacity(provisional_components.len());
    for (new_id, &old_id) in ids.iter().enumerate() {
        canonical_id[old_id] = new_id;
        components.push(provisional_components[old_id].clone());
        cyclic.push(provisional_cyclic[old_id]);
    }

    let component_of = provisional_of
        .into_iter()
        .map(|old_id| canonical_id[old_id])
        .collect();

    Ok(Graph {
        components,
        component_of,
        cyclic,
    })
}

fn main() {}
