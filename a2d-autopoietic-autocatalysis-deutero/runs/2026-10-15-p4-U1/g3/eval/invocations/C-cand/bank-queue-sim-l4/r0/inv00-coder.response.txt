#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    started: bool,
    paused: bool,
    pause_time: u64,
    queue_entry: u64,
    waited: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| b.paused.cmp(&a.paused))
        .then_with(|| {
            if a.paused && b.paused {
                a.pause_time
                    .cmp(&b.pause_time)
                    .then_with(|| a.customer.id.cmp(&b.customer.id))
            } else {
                a.customer
                    .arrival
                    .cmp(&b.customer.arrival)
                    .then_with(|| b.customer.id.cmp(&a.customer.id))
            }
        })
        .then(Ordering::Equal)
}

fn emit(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = Vec::with_capacity(customers.len());
    let mut deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.iter().any(|id| *id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        seen.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index,
                id: c.id,
            })?),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers
        .iter()
        .enumerate()
        .map(|(i, c)| Job {
            customer: c.clone(),
            deadline: deadlines[i],
            remaining: c.service,
            started: false,
            paused: false,
            pause_time: 0,
            queue_entry: c.arrival,
            waited: 0,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();
    let mut last_event_time = 0u64;

    loop {
        if let Some(a) = active.clone() {
            if a.finish == now {
                let id = jobs[a.index].customer.id;
                emit(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                completed += 1;
                completion_order.push(id);
                turnaround_ticks += (now - jobs[a.index].customer.arrival) as u128;
                active = None;
            }
        }

        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            jobs[a]
                .deadline
                .cmp(&jobs[b].deadline)
                .then_with(|| jobs[a].customer.id.cmp(&jobs[b].customer.id))
        });
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_entry;
                jobs[i].waited = jobs[i].waited.saturating_add(segment);
                wait_ticks += segment as u128;
                emit(
                    &mut events,
                    now,
                    jobs[i].customer.id,
                    Kind::Renege { waited: jobs[i].waited },
                );
                last_event_time = now;
                reneged += 1;
            }
        }

        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let i = order[next_arrival];
            next_arrival += 1;
            let id = jobs[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);
            last_event_time = now;

            let capacity = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= capacity {
                emit(&mut events, now, id, Kind::Balk);
                last_event_time = now;
                balked += 1;
            } else {
                accepted += 1;
                jobs[i].queue_entry = now;
                waiting.push(i);
            }
        }

        if let Some(a) = active.clone() {
            if now >= a.work_at {
                let mut best: Option<usize> = None;
                for &i in &waiting {
                    if best.is_none_or(|b| waiting_cmp(&jobs[i], &jobs[b]).is_lt()) {
                        best = Some(i);
                    }
                }
                if let Some(b) = best {
                    if jobs[b].customer.priority < jobs[a.index].customer.priority {
                        let remaining = a.finish - now;
                        let paused_id = jobs[a.index].customer.id;
                        let by = jobs[b].customer.id;
                        emit(&mut events, now, paused_id, Kind::Preempt { remaining, by });
                        last_event_time = now;
                        preemptions += 1;
                        busy_ticks += (now - a.work_at + (a.work_at - (a.finish - jobs[a.index].remaining))) as u128;
                        // Occupied interval is dispatch to preemption.
                        let dispatch_time = a.finish - (a.work_at - (a.finish - jobs[a.index].remaining));
                        let _ = dispatch_time;
                        active = None;
                        jobs[a.index].remaining = remaining;
                        jobs[a.index].paused = true;
                        jobs[a.index].pause_time = now;
                        jobs[a.index].queue_entry = now;
                        jobs[a.index].deadline = match jobs[a.index].customer.patience {
                            Some(_) => Some(now.checked_add(
                                jobs[a.index]
                                    .deadline
                                    .map(|d| d.saturating_sub(a.finish - remaining))
                                    .unwrap_or(0),
                            ).ok_or_else(|| time_overflow(
                                paused_id,
                                now,
                                OverflowStage::Patience,
                            ))?),
                            None => None,
                        };
                        waiting.push(a.index);
                    }
                }
            }
        }

        loop {
            if active.is_none() {
                let mut best_pos: Option<usize> = None;
                for (pos, &i) in waiting.iter().enumerate() {
                    if best_pos.is_none_or(|bp| waiting_cmp(&jobs[i], &jobs[waiting[bp]]).is_lt()) {
                        best_pos = Some(pos);
                    }
                }
                let Some(pos) = best_pos else { break };
                let i = waiting.remove(pos);
                let id = jobs[i].customer.id;
                let segment = now - jobs[i].queue_entry;
                jobs[i].waited = jobs[i].waited.saturating_add(segment);
                wait_ticks += segment as u128;

                let cost = if last_dispatched.is_some_and(|last| last != id) {
                    config.switch_cost
                } else {
                    0
                };
                let work_at = now.checked_add(cost).ok_or_else(|| {
                    time_overflow(id, now, OverflowStage::WorkAt)
                })?;
                let finish = work_at.checked_add(jobs[i].remaining).ok_or_else(|| {
                    time_overflow(id, now, OverflowStage::Finish)
                })?;
                let kind = if jobs[i].started {
                    Kind::Resume { remaining: jobs[i].remaining, work_at, finish }
                } else {
                    jobs[i].started = true;
                    started += 1;
                    Kind::Start { remaining: jobs[i].remaining, work_at, finish }
                };
                emit(&mut events, now, id, kind);
                last_event_time = now;
                last_dispatched = Some(id);
                active = Some(Active { index: i, work_at, finish });

                if finish == now {
                    emit(&mut events, now, id, Kind::Finish);
                    last_event_time = now;
                    completed += 1;
                    completion_order.push(id);
                    turnaround_ticks += (now - jobs[i].customer.arrival) as u128;
                    active = None;
                }
            } else {
                break;
            }
        }

        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            jobs[a]
                .deadline
                .cmp(&jobs[b].deadline)
                .then_with(|| jobs[a].customer.id.cmp(&jobs[b].customer.id))
        });
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_entry;
                jobs[i].waited = jobs[i].waited.saturating_add(segment);
                wait_ticks += segment as u128;
                emit(
                    &mut events,
                    now,
                    jobs[i].customer.id,
                    Kind::Renege { waited: jobs[i].waited },
                );
                last_event_time = now;
                reneged += 1;
            }
        }

        if next_arrival == order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next_time.is_none_or(|old| t < old) {
                next_time = Some(t);
            }
        };
        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].deadline {
                consider(d);
            }
        }
        if let Some(t) = next_time {
            if let Some(a) = &active {
                let dispatch_start = a.finish - jobs[a.index].remaining;
                let end = t.min(a.finish);
                if end > dispatch_start {
                    busy_ticks += (end - dispatch_start) as u128;
                }
            }
            now = t;
        } else {
            break;
        }
    }

    let elapsed = last_event_time;
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap_or(0)
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap_or(0)
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128)
            .unwrap_or(0)
            .min(u32::MAX as u128) as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

fn main() {}
