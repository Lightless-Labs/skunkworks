#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;

    let round_up = remainder > complement
        || (remainder == complement && quotient % 2 == 1);

    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    started: bool,
    paused: bool,
    pause_time: u64,
    queue_entry: u64,
    waited: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,
}

fn waiting_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| b.paused.cmp(&a.paused))
        .then_with(|| {
            if a.paused && b.paused {
                a.pause_time
                    .cmp(&b.pause_time)
                    .then_with(|| a.customer.id.cmp(&b.customer.id))
            } else {
                a.customer
                    .arrival
                    .cmp(&b.customer.arrival)
                    .then_with(|| b.customer.id.cmp(&a.customer.id))
            }
        })
}

fn emit(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event {
        time,
        customer: id,
        kind,
    });
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow {
        customer,
        time,
        stage,
    }
}

fn expire_waiters(
    now: u64,
    waiting: &mut Vec<usize>,
    jobs: &mut [Job],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<usize> = waiting
        .iter()
        .copied()
        .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
        .collect();

    expired.sort_by(|&a, &b| {
        jobs[a]
            .deadline
            .cmp(&jobs[b].deadline)
            .then_with(|| jobs[a].customer.id.cmp(&jobs[b].customer.id))
    });

    for i in expired {
        if let Some(pos) = waiting.iter().position(|&x| x == i) {
            waiting.remove(pos);
            let segment = now - jobs[i].queue_entry;
            jobs[i].waited += segment;
            stats.wait_ticks += segment as u128;
            emit(
                events,
                now,
                jobs[i].customer.id,
                Kind::Renege {
                    waited: jobs[i].waited,
                },
            );
            stats.reneged += 1;
        }
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut seen = Vec::with_capacity(customers.len());
    let mut deadlines = Vec::with_capacity(customers.len());

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        seen.push(customer.id);

        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }

        let deadline = match customer.patience {
            Some(patience) => Some(
                customer
                    .arrival
                    .checked_add(patience)
                    .ok_or(Error::DeadlineOverflow {
                        index,
                        id: customer.id,
                    })?,
            ),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers
        .iter()
        .enumerate()
        .map(|(i, customer)| Job {
            customer: customer.clone(),
            deadline: deadlines[i],
            remaining: customer.service,
            started: false,
            paused: false,
            pause_time: 0,
            queue_entry: customer.arrival,
            waited: 0,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_event_time = 0u64;

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    loop {
        // Phase A: a completion precedes every other action at this tick.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let id = jobs[a.index].customer.id;
            emit(&mut events, now, id, Kind::Finish);
            last_event_time = now;
            stats.completed += 1;
            stats.completion_order.push(id);
            stats.turnaround_ticks += (now - jobs[a.index].customer.arrival) as u128;
            stats.busy_ticks += (now - a.dispatch_time) as u128;
        }

        // Phase B: expire those already in the queue.
        expire_waiters(now, &mut waiting, &mut jobs, &mut events, &mut stats);

        // Phase C: process the entire arrival cohort without dispatching.
        while next_arrival < order.len()
            && customers[order[next_arrival]].arrival == now
        {
            let i = order[next_arrival];
            next_arrival += 1;
            let id = jobs[i].customer.id;

            emit(&mut events, now, id, Kind::Arrive);
            last_event_time = now;

            let capacity = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= capacity {
                emit(&mut events, now, id, Kind::Balk);
                last_event_time = now;
                stats.balked += 1;
            } else {
                stats.accepted += 1;
                jobs[i].queue_entry = now;
                waiting.push(i);
            }
        }

        // Phase D: after protection ends, compare before re-enqueuing.
        if let Some(a) = active.clone() {
            if now >= a.work_at {
                let best = waiting
                    .iter()
                    .copied()
                    .min_by(|&x, &y| waiting_cmp(&jobs[x], &jobs[y]));

                if let Some(b) = best {
                    if jobs[b].customer.priority < jobs[a.index].customer.priority {
                        let remaining = a.finish - now;
                        let paused_id = jobs[a.index].customer.id;
                        let by = jobs[b].customer.id;

                        emit(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt { remaining, by },
                        );
                        last_event_time = now;
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch_time) as u128;

                        active = None;
                        jobs[a.index].remaining = remaining;
                        jobs[a.index].paused = true;
                        jobs[a.index].pause_time = now;
                        jobs[a.index].queue_entry = now;

                        jobs[a.index].deadline = match jobs[a.index].customer.patience {
                            Some(budget) => Some(now.checked_add(budget).ok_or_else(|| {
                                time_overflow(paused_id, now, OverflowStage::Patience)
                            })?),
                            None => None,
                        };

                        waiting.push(a.index);
                    }
                }
            }
        }

        // Phase E: dispatch until occupied, including inline zero-duration work.
        loop {
            if active.is_some() {
                break;
            }

            let best_pos = waiting
                .iter()
                .enumerate()
                .min_by(|(_, &a), (_, &b)| waiting_cmp(&jobs[a], &jobs[b]))
                .map(|(pos, _)| pos);

            let Some(pos) = best_pos else {
                break;
            };

            let i = waiting.remove(pos);
            let id = jobs[i].customer.id;
            let segment = now - jobs[i].queue_entry;
            jobs[i].waited += segment;
            stats.wait_ticks += segment as u128;

            let switch = if last_dispatched.is_some_and(|last| last != id) {
                config.switch_cost
            } else {
                0
            };

            let work_at = now
                .checked_add(switch)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::WorkAt))?;
            let finish = work_at
                .checked_add(jobs[i].remaining)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::Finish))?;

            let remaining = jobs[i].remaining;
            let kind = if jobs[i].started {
                Kind::Resume {
                    remaining,
                    work_at,
                    finish,
                }
            } else {
                jobs[i].started = true;
                stats.started += 1;
                Kind::Start {
                    remaining,
                    work_at,
                    finish,
                }
            };

            emit(&mut events, now, id, kind);
            last_event_time = now;
            last_dispatched = Some(id);
            active = Some(Active {
                index: i,
                work_at,
                finish,
                dispatch_time: now,
            });

            if finish == now {
                let a = active.take().unwrap();
                emit(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                stats.completed += 1;
                stats.completion_order.push(id);
                stats.turnaround_ticks += (now - jobs[i].customer.arrival) as u128;
                stats.busy_ticks += (now - a.dispatch_time) as u128;
                continue;
            }

            break;
        }

        // Phase F: include newly queued zero-budget customers.
        expire_waiters(now, &mut waiting, &mut jobs, &mut events, &mut stats);

        if next_arrival == order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next_time.is_none_or(|old| t < old) {
                next_time = Some(t);
            }
        };

        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            consider(a.work_at);
        }
        for &i in &waiting {
            if let Some(deadline) = jobs[i].deadline {
                consider(deadline);
            }
        }

        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    stats.elapsed = last_event_time;
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(
            stats.turnaround_ticks * 1000,
            stats.completed as u128,
        )
        .unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10_000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}

fn main() {}
