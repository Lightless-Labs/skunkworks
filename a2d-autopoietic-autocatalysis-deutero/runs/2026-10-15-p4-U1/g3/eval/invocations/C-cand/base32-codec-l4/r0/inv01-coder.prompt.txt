pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = Vec::with_capacity(self.pending.len() + input.len());
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 5 * 5;
        let output = encode_bytes(&combined[..complete_len], self.config.alphabet);
        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let output = encode_final(&self.pending, self.config);
        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_byte(alphabet: Alphabet, value: u8) -> u8 {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[value as usize],
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV"[value as usize],
    }
}

fn encode_bytes(input: &[u8], alphabet: Alphabet) -> String {
    let mut output = String::with_capacity(input.len() / 5 * 8);
    for chunk in input.chunks_exact(5) {
        let bits = ((chunk[0] as u64) << 32)
            | ((chunk[1] as u64) << 24)
            | ((chunk[2] as u64) << 16)
            | ((chunk[3] as u64) << 8)
            | chunk[4] as u64;

        for shift in (0..8).rev() {
            output.push(alphabet_byte(alphabet, ((bits >> (shift * 5)) & 31) as u8) as char);
        }
    }
    output
}

fn encode_final(input: &[u8], config: Config) -> String {
    if input.is_empty() {
        return String::new();
    }

    let total_bits = input.len() * 8;
    let symbol_count = (total_bits + 4) / 5;
    let mut output = String::with_capacity(8);
    for symbol_index in 0..symbol_count {
        let bit_start = symbol_index * 5;
        let mut value = 0u8;
        for bit_offset in 0..5 {
            value <<= 1;
            let bit_index = bit_start + bit_offset;
            if bit_index < total_bits {
                let byte = input[bit_index / 8];
                let bit = (byte >> (7 - (bit_index % 8))) & 1;
                value |= bit;
            }
        }
        output.push(alphabet_byte(config.alphabet, value) as char);
    }

    if config.padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }
    output
}

pub struct Decoder {
    config: Config,
    bit_buffer: u8,
    bit_count: u8,
    data_count: usize,
    consumed: usize,
    last_data_offset: Option<usize>,
    last_data_value: u8,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            bit_buffer: 0,
            bit_count: 0,
            data_count: 0,
            consumed: 0,
            last_data_offset: None,
            last_data_value: 0,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.consumed;
            let recognized = recognize(next.config.alphabet, byte);

            match recognized {
                Recognized::Invalid => {
                    return Err(StreamError::InvalidByte { offset, byte });
                }
                Recognized::Data(value, uppercase) => {
                    if next.config.canonical && byte != uppercase {
                        return Err(StreamError::NonCanonical {
                            offset,
                            expected: uppercase,
                        });
                    }
                    if next.first_padding_offset.is_some() {
                        return Err(StreamError::Padding { offset });
                    }

                    next.data_count += 1;
                    next.last_data_offset = Some(offset);
                    next.last_data_value = value;
                    next.bit_buffer = (next.bit_buffer << 5) | value;
                    next.bit_count += 5;
                    while next.bit_count >= 8 {
                        next.bit_count -= 8;
                        output.push((next.bit_buffer >> next.bit_count) as u8);
                        if next.bit_count == 0 {
                            next.bit_buffer = 0;
                        } else {
                            next.bit_buffer &= (1u8 << next.bit_count) - 1;
                        }
                    }
                }
                Recognized::Padding => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }

                    if next.first_padding_offset.is_none() {
                        let residue = next.data_count % 8;
                        let expected = match residue {
                            2 => 6,
                            4 => 4,
                            5 => 3,
                            7 => 1,
                            _ => return Err(StreamError::Padding { offset }),
                        };
                        next.first_padding_offset = Some(offset);
                        next.expected_padding = expected;
                    }

                    if next.padding_count >= next.expected_padding {
                        return Err(StreamError::Padding { offset });
                    }
                    next.padding_count += 1;
                }
            }

            next.consumed += 1;
        }

        *self = next;
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => return Err(StreamError::InvalidLength { offset: self.consumed }),
        };

        if self.config.padding == Padding::Required && self.padding_count != expected_padding {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(self.consumed),
            });
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits != 0 {
            let mask = (1u8 << unused_bits) - 1;
            if self.last_data_value & mask != 0 {
                return Err(StreamError::TrailingBits {
                    offset: self.last_data_offset.unwrap_or(self.consumed),
                });
            }
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            data_count: self.data_count,
            consumed: self.consumed,
            last_data_offset: self.last_data_offset,
            last_data_value: self.last_data_value,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }
}

enum Recognized {
    Data(u8, u8),
    Padding,
    Invalid,
}

fn recognize(alphabet: Alphabet, byte: u8) -> Recognized {
    if byte == b'=' {
        return Recognized::Padding;
    }

    let upper = byte.to_ascii_uppercase();
    let table = match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567".as_slice(),
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV".as_slice(),
    };

    match table.iter().position(|&candidate| candidate == upper) {
        Some(value) => Recognized::Data(value as u8, upper),
        None => Recognized::Invalid,
    }
}
