#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let mut digits = Vec::new();

    // Validate all non-separator bytes before checking separator placement.
    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => return Err(BaseError::InvalidDigit { operand, index }),
        }
    }

    for index in payload_start..bytes.len() {
        if bytes[index] == b'_'
            && (index == payload_start
                || index + 1 == bytes.len()
                || bytes[index - 1] == b'_'
                || bytes[index + 1] == b'_')
        {
            return Err(BaseError::Separator { operand, index });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Little-endian limbs, each storing one digit in the given radix.
fn convert_digits(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value: Vec<u8> = Vec::new();

    for &digit in digits {
        let mut carry = digit as u16;
        for limb in &mut value {
            let n = (*limb as u16) * (from as u16) + carry;
            *limb = (n % to as u16) as u8;
            carry = n / to as u16;
        }
        while carry != 0 {
            value.push((carry % to as u16) as u8);
            carry /= to as u16;
        }
    }

    if value.is_empty() {
        value.push(0);
    }
    value
}

fn render(mut digits: Vec<u8>, style: BaseStyle) -> String {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }

    let width = digits.len().max(style.min_width);
    let padding = width - digits.len();
    let mut chars = Vec::with_capacity(width);
    chars.extend(std::iter::repeat_n(b'0', padding));

    for digit in digits.iter().rev() {
        let byte = if *digit < 10 {
            b'0' + *digit
        } else if style.upper {
            b'A' + (*digit - 10)
        } else {
            b'a' + (*digit - 10)
        };
        chars.push(byte);
    }

    if style.group == 0 {
        return String::from_utf8(chars).expect("ASCII output");
    }

    let separators = (chars.len().saturating_sub(1)) / style.group;
    let mut output = String::with_capacity(chars.len() + separators);
    for (i, byte) in chars.iter().enumerate() {
        if i > 0 && (chars.len() - i) % style.group == 0 {
            output.push('_');
        }
        output.push(*byte as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    Ok(render(convert_digits(&digits, from, to), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_value = convert_digits(&left_digits, base, base);
    let right_value = convert_digits(&right_digits, base, base);
    let mut carry = 0u16;
    let max_len = left_value.len().max(right_value.len());

    left_value.resize(max_len, 0);
    for i in 0..max_len {
        let sum = left_value[i] as u16
            + right_value.get(i).copied().unwrap_or(0) as u16
            + carry;
        left_value[i] = (sum % base as u16) as u8;
        carry = sum / base as u16;
    }
    if carry != 0 {
        left_value.push(carry as u8);
    }

    Ok(render(left_value, style))
}

fn main() {}
