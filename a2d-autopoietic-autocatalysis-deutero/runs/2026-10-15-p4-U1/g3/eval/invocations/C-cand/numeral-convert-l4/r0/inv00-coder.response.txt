#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "", "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn spell_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }

    if n < 100 {
        let tens = TENS[(n / 10) as usize];
        let units = n % 10;
        return if units == 0 {
            tens.to_string()
        } else {
            format!("{tens}-{}", SMALL[units as usize])
        };
    }

    let hundreds = n / 100;
    let remainder = n % 100;
    if remainder == 0 {
        format!("{} hundred", SMALL[hundreds as usize])
    } else {
        format!(
            "{} hundred and {}",
            SMALL[hundreds as usize],
            spell_under_thousand(remainder)
        )
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    for scale in 0..=6 {
        let group = (remaining % 1000) as u16;
        remaining /= 1000;
        if group != 0 {
            groups.push((scale, group));
        }
    }
    groups.reverse();

    let last_index = groups.len() - 1;
    let mut pieces = Vec::with_capacity(groups.len());

    for (index, &(scale, group)) in groups.iter().enumerate() {
        let mut piece = spell_under_thousand(group);
        if scale != 0 {
            piece.push(' ');
            piece.push_str(SCALES[scale]);
        }
        if index == last_index
            && scale == 0
            && group < 100
            && groups.len() > 1
        {
            piece.insert_str(0, "and ");
        }
        pieces.push(piece);
    }

    let mut output = pieces.join(" ");
    if value < 0 {
        output.insert_str(0, "minus ");
    }
    Ok(output)
}

fn dictionary_value(token: &str) -> Option<TokenValue> {
    if token == "zero" {
        return Some(TokenValue::Number(0));
    }

    if let Some(index) = SMALL.iter().position(|&word| word == token) {
        if index != 0 {
            return Some(TokenValue::Number(index as u128));
        }
    }

    if let Some(index) = TENS.iter().position(|&word| word == token) {
        if index >= 2 {
            return Some(TokenValue::Number((index * 10) as u128));
        }
    }

    match token {
        "hundred" => Some(TokenValue::Hundred),
        "thousand" => Some(TokenValue::Scale(1_000)),
        "million" => Some(TokenValue::Scale(1_000_000)),
        "billion" => Some(TokenValue::Scale(1_000_000_000)),
        "trillion" => Some(TokenValue::Scale(1_000_000_000_000)),
        "quadrillion" => Some(TokenValue::Scale(1_000_000_000_000_000)),
        "quintillion" => Some(TokenValue::Scale(LIMIT)),
        "and" => Some(TokenValue::And),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum TokenValue {
    Number(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if start < index {
                tokens.push((&input[start..index], start));
            }
            start = index + 1;
        }
    }
    if start < bytes.len() {
        tokens.push((&input[start..], start));
    }

    let mut values = Vec::with_capacity(tokens.len());
    for (token, index) in &tokens {
        if *token == "minus" {
            values.push(TokenValue::Minus);
        } else if let Some(value) = dictionary_value(token) {
            values.push(value);
        } else {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = matches!(values.first(), Some(TokenValue::Minus));
    for (index, value) in values.iter().enumerate() {
        if matches!(value, TokenValue::Minus) && index != 0 {
            return Err(WordsError::InvalidWord { index: tokens[index].1 });
        }
    }

    let arithmetic_start = usize::from(negative);
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for value in values.iter().skip(arithmetic_start) {
        match *value {
            TokenValue::Number(n) => group = group.checked_add(n).ok_or(WordsError::Range)?,
            TokenValue::Hundred => group = group.checked_mul(100).ok_or(WordsError::Range)?,
            TokenValue::Scale(scale) => {
                let scaled = group.checked_mul(scale).ok_or(WordsError::Range)?;
                total = total.checked_add(scaled).ok_or(WordsError::Range)?;
                group = 0;
            }
            TokenValue::And => {}
            TokenValue::Minus => unreachable!(),
        }

        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical != input {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(input.as_bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: common });
    }

    Ok(signed)
}

fn main() {}
