    pub id: u32,
    pub arrival: u64,
    pub service: u64,

    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy: Option<(u64, u32)>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_arrival = 0;
    let mut completed_customers = 0;

    while completed_customers < customers.len() || states.iter().any(|s| !s.opened) {
        let mut next_time = u64::MAX;
        let mut have_next = false;

        if next_arrival < customer_order.len() {
            next_time = customers[customer_order[next_arrival]].arrival;
            have_next = true;
        }
        for state in &states {
            if !state.opened && (!have_next || state.server.opens < next_time) {
                next_time = state.server.opens;
                have_next = true;
            }
            if let Some((finish, _)) = state.busy {
                if !have_next || finish < next_time {
                    next_time = finish;
                    have_next = true;
                }
            }
        }

        if !have_next {
            break;
        }

        let time = next_time;

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                state
                    .busy
                    .filter(|(finish, _)| *finish == time)
                    .map(|_| i)
            })
            .collect();
        finishing.sort_by_key(|&i| states[i].server.id);
        for i in finishing {
            let (finish, customer) = states[i].busy.take().unwrap();
            events.push(Event::Finish {
                time: finish,
                customer,
                server: states[i].server.id,
            });
            states[i].idle_since = time;
            completed_customers += 1;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.server.opens == time).then_some(i)
            })
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time,
                server: states[i].server.id,
            });
        }

        while next_arrival < customer_order.len()
            && customers[customer_order[next_arrival]].arrival == time
        {
            let customer_index = customer_order[next_arrival];
            waiting.push(customer_index);
            events.push(Event::Arrive {
                time,
                customer: customers[customer_index].id,
            });
            next_arrival += 1;
        }

        while !waiting.is_empty() {
            let Some(server_index) = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i)
            else {
                break;

            let finish = time
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(server.overhead))

                    customer: customer.id,
                    server: server.id,
                    time,

                time,
                customer: customer.id,
                server: server.id,
                finish,

                    time,
                    customer: customer.id,
                    server: server.id,
                });

                completed_customers += 1;
            } else {
                states[server_index].busy = Some((finish, customer.id));
            }
        }
    }

    Ok(events)
}
