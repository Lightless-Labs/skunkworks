```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone)]
struct State {
    burst_position: usize,
    remaining: u32,
    level: usize,
}

#[derive(Clone, Copy)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    kind_order: u8,
    index: usize,
    kind: EventKind,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let n = jobs.len();
    if n == 0 {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(n * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind_order: 0,
            index,
            kind: EventKind::Arrival,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind_order, event.index));

    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut completions = vec![0u64; n];
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let mut now = 0u64;
    let mut event_cursor = 0usize;
    let mut last_dispatch: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < n {
        while event_cursor < events.len() && events[event_cursor].time <= now {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    queues[jobs[event.index].level].push_back(event.index);
                }
                EventKind::Wake { level } => {
                    states[event.index].level = level;
                    queues[level].push_back(event.index);
                }
            }
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = jobs[index].id;

        if last_dispatch.is_some_and(|last_id| last_id != id) {
            now += switch_cost as u64;
        }

        while event_cursor < events.len() && events[event_cursor].time <= now {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    queues[jobs[event.index].level].push_back(event.index);
                }
                EventKind::Wake { level } => {
                    states[event.index].level = level;
                    queues[level].push_back(event.index);
                }
            }
            event_cursor += 1;
        }

        let start = now;
        let consumed = states[index].remaining.min(quanta[level]);
        now += consumed as u64;
        let end = now;
        states[index].remaining -= consumed;

        let burst_finished = states[index].remaining == 0;
        let job_finished = burst_finished && states[index].burst_position + 1 == jobs[index].bursts.len();

        runs.push(Run {
            id,
            start,
            end,
            finished: job_finished,
            level,
        });

        while event_cursor < events.len() && events[event_cursor].time <= end {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    queues[jobs[event.index].level].push_back(event.index);
                }
                EventKind::Wake { level } => {
                    states[event.index].level = level;
                    queues[level].push_back(event.index);
                }
            }
            event_cursor += 1;
        }

        if job_finished {
            completions[index] = end;
            completed_count += 1;
        } else if burst_finished {
            let io_position = states[index].burst_position + 1;
            let wake_time = end + jobs[index].bursts[io_position] as u64;
            states[index].burst_position += 2;
            states[index].remaining = jobs[index].bursts[states[index].burst_position];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: wake_time,
                kind_order: 1,
                index,
                kind: EventKind::Wake { level: wake_level },
            });
            events[event_cursor..].sort_by_key(|event| (event.time, event.kind_order, event.index));
        } else {
            let new_level = (level + 1).min(quanta.len() - 1);
            states[index].level = new_level;
            queues[new_level].push_back(index);
        }

        last_dispatch = Some(id);
    }

    let mut stats = Vec::with_capacity(n);
    let mut waiting_total = 0u64;
    let mut turnaround_total = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completions[index];
        let turnaround = completion - job.arrival as u64;
        let work_total: u64 = job.bursts.iter().map(|&value| value as u64).sum();
        let waiting = turnaround - work_total;
        waiting_total += waiting;
        turnaround_total += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = n as u64;
    let mean_waiting_x100 = (waiting_total * 100 + count / 2) / count;
    let mean_turnaround_x100 = (turnaround_total * 100 + count / 2) / count;
    let end = completions.iter().copied().max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}

fn main() {}
```
