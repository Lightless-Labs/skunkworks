#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy)]
    enum Quote {
        None,
        Single(usize),
        Double(usize),
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match quote {
            Quote::None => match ch {
                '\'' => {
                    active = true;
                    quote = Quote::Single(pos);
                }
                '"' => {
                    active = true;
                    quote = Quote::Double(pos);
                }
                '\\' => {
                    if let Some(&(next_pos, next)) = chars.peek() {
                        if next == '\n' {
                            chars.next();
                        } else {
                            chars.next();
                            word.push(next);
                            active = true;
                        }
                        let _ = next_pos;
                    } else {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                }
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                _ => {
                    word.push(ch);
                    active = true;
                }
            },
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
            }
            Quote::Double(_) => match ch {
                '"' => quote = Quote::None,
                '\\' => {
                    if let Some(&(next_pos, next)) = chars.peek() {
                        match next {
                            '"' | '\\' | '$' => {
                                chars.next();
                                word.push(next);
                            }
                            '\n' => {
                                chars.next();
                            }
                            _ => word.push('\\'),
                        }
                        let _ = next_pos;
                    } else {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                }
                _ => word.push(ch),
            },
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn main() {}
