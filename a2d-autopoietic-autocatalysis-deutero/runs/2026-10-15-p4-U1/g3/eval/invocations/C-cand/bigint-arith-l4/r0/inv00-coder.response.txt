use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 1_000_000_000 limbs. Zero is [0].
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit_value(bytes[i]).is_none_or(|d| d >= radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator_error = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                separator_error = Some(i);
                break;
            }
        }
        if let Some(index) = separator_error {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        for i in start..bytes.len() {
            if bytes[i] != b'_' {
                if first_digit.is_none() {
                    first_digit = Some((i, digit_value(bytes[i]).unwrap()));
                }
                digit_count += 1;
            }
        }

        if digit_count > 1 && first_digit.unwrap().1 == 0 {
            return Err(Error::LeadingZero {
                index: first_digit.unwrap().0,
            });
        }

        if digit_count > max_digits {
            let target = max_digits + 1;
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == target {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index: i,
                        });
                    }
                }
            }
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte != b'_' {
                value.mul_small(radix);
                value.add_small(digit_value(byte).unwrap());
            }
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0);
        } else {
            while !n.is_zero() {
                digits.push(n.div_small(radix) as u8);
            }
            digits.reverse();
        }

        if digits.len() < min_digits {
            let mut padded = vec![0; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(digit_char(*digit));
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self {
                negative: self.negative,
                limbs: Self::add_abs(&self.limbs, &rhs.limbs),
            };
            out.normalize();
            out
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self {
                        negative: self.negative,
                        limbs: Self::sub_abs(&self.limbs, &rhs.limbs),
                    };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self {
                        negative: rhs.negative,
                        limbs: Self::sub_abs(&rhs.limbs, &self.limbs),
                    };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let total = result[index] + a as u64 * b as u64 + carry;
                result[index] = total % Self::BASE;
                carry = total / Self::BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry > 0 {
                if index == result.len() {
                    result.push(0);
                }
                let total = result[index] + carry;
                result[index] = total % Self::BASE;
                carry = total / Self::BASE;
                index += 1;
            }
        }

        let mut out = Self {
            negative: self.negative ^ rhs.negative,
            limbs: result.into_iter().map(|x| x as u32).collect(),
        };
        out.normalize();
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (q_abs, r_abs) = Self::div_rem_abs(&self.abs(), &rhs.abs());
        let mut quotient = q_abs;
        quotient.negative = (self.negative ^ rhs.negative) && !quotient.is_zero();

        let mut remainder = r_abs;
        remainder.negative = self.negative && !remainder.is_zero();

        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: LIMIT,
            });
        }

        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power > 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = remainder.abs();
        }
        a
    }

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs(&self) -> Self {
        let mut out = self.clone();
        out.negative = false;
        out
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        order => return order,
                    }
                }
                Ordering::Equal
            }
            order => order,
        }
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let len = a.len().max(b.len());
        let mut result = Vec::with_capacity(len + 1);
        let mut carry = 0u64;
        for i in 0..len {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64
                + carry;
            result.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry > 0 {
            result.push(carry as u32);
        }
        result
    }

    // Requires a >= b.
    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut result = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut value = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(value as u32);
        }
        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result
    }

    fn mul_small(&mut self, multiplier: u32) {
        if multiplier == 0 || self.is_zero() {
            *self = Self::zero();
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * multiplier as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry > 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut i = 0;
        while carry > 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[i] as u64 + carry;
            self.limbs[i] = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
            i += 1;
        }
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut remainder = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let value = remainder * Self::BASE + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        self.normalize();
        remainder as u32
    }

    fn mul_small_copy(&self, multiplier: u32) -> Self {
        let mut out = self.clone();
        out.mul_small(multiplier);
        out
    }

    fn add_small_copy(&self, addend: u32) -> Self {
        let mut out = self.clone();
        out.add_small(addend);
        out
    }

    fn div_rem_abs(dividend: &Self, divisor: &Self) -> (Self, Self) {
        if dividend.cmp_abs(divisor) == Ordering::Less {
            return (Self::zero(), dividend.clone());
        }

        let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
        let mut remainder = Self::zero();

        for &limb in dividend.limbs.iter().rev() {
            remainder.limbs.insert(0, limb);
            remainder.normalize();

            let mut low = 0u32;
            let mut high = (Self::BASE - 1) as u32;
            let mut best = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = divisor.mul_small_copy(mid);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        if mid == (Self::BASE - 1) as u32 {
                            break;
                        }
                        low = mid + 1;
                    }
                }
            }

            if best != 0 {
                remainder = Self::sub_abs_value(&remainder, &divisor.mul_small_copy(best));
            }
            quotient_be.push(best);
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn sub_abs_value(a: &Self, b: &Self) -> Self {
        Self {
            negative: false,
            limbs: Self::sub_abs(&a.limbs, &b.limbs),
        }
    }
}

fn digit_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
        b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
        _ => None,
    }
}

fn digit_char(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'a' + value - 10) as char,
    }
}

fn main() {}
