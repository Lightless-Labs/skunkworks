#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }

    let expected = board.width * board.height;
    if board.cells.len() != expected {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

fn process_line(
    board: &Board,
    output: &mut [u32],
    direction: Direction,
    line: usize,
    merges: &mut Vec<Merge>,
    score: &mut u64,
) -> Result<(), Error> {
    let length = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };

    let mut indices = Vec::with_capacity(length);
    for offset in 0..length {
        let position = match direction {
            Direction::Left => line * board.width + offset,
            Direction::Right => line * board.width + (board.width - 1 - offset),
            Direction::Up => offset * board.width + line,
            Direction::Down => (board.height - 1 - offset) * board.width + line,
        };
        indices.push(position);
    }

    let tiles: Vec<(usize, u32)> = indices
        .iter()
        .filter_map(|&index| {
            let value = board.cells[index];
            (value != 0).then_some((index, value))
        })
        .collect();

    let mut packed = Vec::with_capacity(length);
    let mut i = 0;
    while i < tiles.len() {
        let (first_index, value) = tiles[i];
        if i + 1 < tiles.len() && tiles[i + 1].1 == value {
            let second_index = tiles[i + 1].0;
            if value == 2_147_483_648 {
                return Err(Error::MergeOverflow {
                    first: first_index,
                    second: second_index,
                });
            }
            let merged_value = value * 2;
            let destination = indices[packed.len()];
            packed.push(merged_value);
            merges.push(Merge {
                from: [first_index, second_index],
                to: destination,
                value: merged_value,
            });
            *score += u64::from(merged_value);
            i += 2;
        } else {
            packed.push(value);
            i += 1;
        }
    }

    for (offset, &index) in indices.iter().enumerate() {
        output[index] = packed.get(offset).copied().unwrap_or(0);
    }

    Ok(())
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    match direction {
        Direction::Left | Direction::Right => {
            for row in 0..board.height {
                process_line(board, &mut cells, direction, row, &mut merges, &mut score)?;
            }
        }
        Direction::Up | Direction::Down => {
            for column in 0..board.width {
                process_line(board, &mut cells, direction, column, &mut merges, &mut score)?;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let mut legal = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        if let Ok(result) = move_board(board, direction) {
            if result.changed {
                legal.push(direction);
            }
        }
    }
    Ok(legal)
}

fn main() {}
