#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        let (segments, trailing) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path(head, 0).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (i, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((i, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            if candidates
                .iter()
                .any(|(i, _)| self.routes[*i].method == "HEAD")
            {
                candidates
                    .iter()
                    .filter(|(i, _)| self.routes[*i].method == "HEAD")
                    .map(|(i, _)| *i)
                    .collect()
            } else {
                candidates
                    .iter()
                    .filter(|(i, _)| self.routes[*i].method == "GET")
                    .map(|(i, _)| *i)
                    .collect()
            }
        } else {
            candidates
                .iter()
                .filter(|(i, _)| self.routes[*i].method == method)
                .map(|(i, _)| *i)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(i, _)| self.routes[*i].method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut winner = eligible[0];
        for &candidate in eligible.iter().skip(1) {
            if self.routes[candidate].rank > self.routes[winner].rank {
                winner = candidate;
            }
        }

        let params = candidates
            .into_iter()
            .find(|(i, _)| *i == winner)
            .map(|(_, p)| p)
            .unwrap_or_default();

        let redirect = if self.policy == SlashPolicy::Redirect {
            let route = &self.routes[winner];
            if route.trailing != parsed.trailing && !parsed.segments.is_empty() {
                let mut target = head.to_owned();
                if route.trailing {
                    target.push('/');
                } else {
                    target.pop();
                }
                Some(target)
            } else {
                None
            }
        } else {
            None
        };

        Ok(Match {
            route: winner,
            params,
            redirect,
        })
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let parsed = parse_path(pattern, 0).map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::with_capacity(parsed.segments.len());
    let mut names = Vec::<String>::new();

    for (index, value) in parsed.segments.iter().enumerate() {
        let start = segment_start(pattern, index);
        let bytes = value.as_bytes();

        if bytes.first() == Some(&b':') || bytes.first() == Some(&b'*') {
            let wildcard = bytes[0] == b'*';
            let name = &value[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && index + 1 != parsed.segments.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            segments.push(Segment::Static(value.clone()));
        }
    }

    Ok((segments, parsed.trailing))
}

fn segment_start(input: &str, index: usize) -> usize {
    let mut starts = input
        .match_indices('/')
        .map(|(i, _)| i + 1)
        .filter(|&i| i < input.len())
        .collect::<Vec<_>>();
    starts.insert(0, 1);
    starts.get(index).copied().unwrap_or(input.len())
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_path(input: &str, base: usize) -> Result<ParsedPath, usize> {
    if !input.starts_with('/') {
        return Err(base);
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let trailing = input.ends_with('/');
    let content_end = if trailing { input.len() - 1 } else { input.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < content_end {
        let end = input[start..content_end]
            .find('/')
            .map(|i| start + i)
            .unwrap_or(content_end);
        if end == start {
            return Err(base + start);
        }
        segments.push(decode_segment(&input[start..end], base + start)?);
        start = end + 1;
    }

    if segments.is_empty() {
        return Err(base + 1);
    }

    Ok(ParsedPath { segments, trailing })
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(base + i);
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            decoded.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_position = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));

    if wildcard_position.is_none() && route.segments.len() != request.len() {
        return None;
    }
    if let Some(pos) = wildcard_position {
        if request.len() < pos {
            return None;
        }
    }

    let mut params = Vec::new();
    for (i, pattern_segment) in route.segments.iter().enumerate() {
        match pattern_segment {
            Segment::Static(value) => {
                if request.get(i) != Some(value) {
                    return None;
                }
            }
            Segment::Param(name) => {
                let value = request.get(i)?;
                params.push((name.clone(), value.clone()));
            }
            Segment::Wildcard(name) => {
                let value = request[i..].join("/");
                params.push((name.clone(), value));
                break;
            }
        }
    }
    Some(params)
}

fn main() {}
