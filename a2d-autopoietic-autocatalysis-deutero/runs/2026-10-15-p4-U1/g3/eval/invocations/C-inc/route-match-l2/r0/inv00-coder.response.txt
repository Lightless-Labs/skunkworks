#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, trailing) = parse_pattern(pattern)?;
        if trailing {
            return Err(Error::InvalidPattern {
                at: pattern.len() - 1,
            });
        }

        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(usize, Match)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &request) {
                let candidate = Match {
                    route: route.id,
                    params,
                };
                let replace = match &best {
                    None => true,
                    Some((best_index, _)) => {
                        route.rank > self.routes[*best_index].rank
                    }
                };
                if replace {
                    best = Some((route.id, candidate));
                }
            }
        }

        best.map(|(_, found)| found).ok_or(Error::NotFound)
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = pattern.ends_with('/');
    let content_end = if trailing { pattern.len() - 1 } else { pattern.len() };
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();

    let mut start = 1;
    while start < content_end {
        let end = pattern[start..content_end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(content_end);

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..end];
        if raw.starts_with(':') || raw.starts_with('*') {
            let is_wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];
            if !valid_capture_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if is_wildcard && end != content_end {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            segments.push(if is_wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }

        start = end + 1;
    }

    Ok((segments, trailing))
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let trailing = path.ends_with('/');
    let content_end = if trailing { path.len() - 1 } else { path.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < content_end {
        let end = path[start..content_end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(content_end);

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }

        segments.push(decode_segment(&path[start..end], start, false)?);
        start = end + 1;
    }

    if trailing {
        return Err(Error::InvalidPath { at: path.len() - 1 });
    }
    Ok(segments)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return invalid_segment(pattern, offset + i);
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return invalid_segment(pattern, offset + i);
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return invalid_segment(pattern, offset + i);
            };
            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return invalid_segment(pattern, offset + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return invalid_segment(pattern, offset + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn invalid_segment<T>(pattern: bool, at: usize) -> Result<T, Error> {
    if pattern {
        Err(Error::InvalidPattern { at })
    } else {
        Err(Error::InvalidPath { at })
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn match_route(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut request_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if request.get(request_index)? != expected {
                    return None;
                }
                request_index += 1;
            }
            Segment::Param(name) => {
                let value = request.get(request_index)?;
                params.push((name.clone(), value.clone()));
                request_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = request[request_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    (request_index == request.len()).then_some(params)
}

fn main() {}
