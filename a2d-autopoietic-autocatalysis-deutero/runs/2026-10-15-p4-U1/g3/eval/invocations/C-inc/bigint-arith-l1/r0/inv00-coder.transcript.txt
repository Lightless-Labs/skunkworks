use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Base-10 digits, most significant first. Zero is represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let magnitude_start = if bytes[0] == b'+' { 1 } else { 0 };
        if magnitude_start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan all magnitude bytes before considering separator errors.
        for (index, &byte) in bytes.iter().enumerate().skip(magnitude_start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_separator_error = None;

        for index in magnitude_start..bytes.len() {
            if bytes[index] == b'_' {
                let prev_is_digit =
                    index > magnitude_start && bytes[index - 1].is_ascii_digit();
                let next_is_digit =
                    index + 1 < bytes.len() && bytes[index + 1].is_ascii_digit();
                if !(prev_is_digit && next_is_digit) && first_separator_error.is_none() {
                    first_separator_error = Some(index);
                }
            } else {
                digit_count += 1;
                if first_digit.is_none() {
                    first_digit = Some(index);
                }
            }
        }

        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let first_digit = first_digit.expect("nonempty magnitude has a digit");
        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for index in magnitude_start..bytes.len() {
                if bytes[index].is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let digits = bytes[magnitude_start..]
            .iter()
            .filter(|&&byte| byte != b'_')
            .map(|&byte| byte - b'0')
            .collect();

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let padding = width - self.digits.len();
        let mut output = String::with_capacity(width + if group > 0 { width / group } else { 0 });

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                output.push('_');
            }

            if position < padding {
                output.push('0');
            } else {
                output.push((b'0' + self.digits[position - padding]) as char);
            }
        }

        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        self.digits
            .len()
            .cmp(&rhs.digits.len())
            .then_with(|| self.digits.cmp(&rhs.digits))
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = Vec::with_capacity(self.digits.len().max(rhs.digits.len()) + 1);
        let mut carry = 0u8;
        let mut left = self.digits.len();
        let mut right = rhs.digits.len();

        while left > 0 || right > 0 || carry > 0 {
            let a = if left > 0 {
                left -= 1;
                self.digits[left]
            } else {
                0
            };
            let b = if right > 0 {
                right -= 1;
                rhs.digits[right]
            } else {
                0
            };

            let sum = a + b + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        result.reverse();
        Self { digits: result }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;
        let mut left = self.digits.len();
        let mut right = rhs.digits.len();

        while left > 0 {
            left -= 1;
            let a = self.digits[left] as i16 - borrow;
            let b = if right > 0 {
                right -= 1;
                rhs.digits[right] as i16
            } else {
                0
            };

            if a < b {
                result.push((a + 10 - b) as u8);
                borrow = 1;
            } else {
                result.push((a - b) as u8);
                borrow = 0;
            }
        }

        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result.reverse();

        Ok(Self { digits: result })
    }
}
