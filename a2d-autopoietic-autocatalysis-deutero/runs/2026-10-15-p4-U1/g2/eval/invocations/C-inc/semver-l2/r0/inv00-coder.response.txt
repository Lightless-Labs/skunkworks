use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for (idx, slot) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        let field = &bytes[start..pos];
        if field.len() > 1 && field[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut value = 0u64;
        for &digit in field {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *slot = value;

        if idx < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pos + offset)
            .unwrap_or(bytes.len());
        parse_identifiers(s, pre_start, end, true, &mut pre)?;
        pos = end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    check_pre_numeric: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let mut ident_start = start;

    loop {
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let mut cursor = ident_start;
        while cursor < end && bytes[cursor] != b'.' {
            let b = bytes[cursor];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: cursor });
            }
            cursor += 1;
        }

        let ident = &s[ident_start..cursor];
        if check_pre_numeric
            && ident.len() > 1
            && ident.as_bytes()[0] == b'0'
            && ident.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        out.push(ident.to_string());

        if cursor == end {
            break;
        }
        ident_start = cursor + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }
    Ok(())
}

fn is_numeric_ident(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_ident(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let order = match (is_numeric_ident(x), is_numeric_ident(y)) {
                        (true, true) => cmp_numeric_ident(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if order != Ordering::Equal {
                        return order;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    conditions: Vec<Condition>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Condition {
    version: Version,
    op: CompareOp,
}

#[derive(Clone, Copy, Debug)]
enum CompareOp {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

impl Condition {
    fn accepts(&self, candidate: &Version) -> bool {
        let order = compare_versions(candidate, &self.version);
        match self.op {
            CompareOp::Eq => order == Ordering::Equal,
            CompareOp::Lt => order == Ordering::Less,
            CompareOp::Le => order != Ordering::Greater,
            CompareOp::Gt => order == Ordering::Greater,
            CompareOp::Ge => order != Ordering::Less,
        }
    }
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<(usize, usize)> {
    let b = s.as_bytes();
    let mut result = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && ws(b[i]) {
            i += 1;
        }
        if i == end {
            break;
        }
        let a = i;
        while i < end && !ws(b[i]) {
            i += 1;
        }
        result.push((a, i));
    }
    result
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn version_at(s: &str, operand_start: usize, operand_end: usize) -> Result<Version, RangeError> {
    parse_version(&s[operand_start..operand_end]).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })
}

#[derive(Clone, Debug)]
struct Partial {
    values: [u64; 3],
    numeric: usize,
    full: Option<Version>,
}

fn parse_partial(
    s: &str,
    start: usize,
    end: usize,
) -> Result<Partial, RangeError> {
    let operand = &s[start..end];
    let bytes = operand.as_bytes();
    let has_sign = bytes.iter().any(|&b| matches!(b, b'-' | b'+'));
    let fields: Vec<&str> = operand.split('.').collect();
    let all_numeric_full_shape = fields.len() >= 3
        && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_sign || all_numeric_full_shape {
        let v = version_at(s, start, end)?;
        return Ok(Partial {
            values: [v.major, v.minor, v.patch],
            numeric: 3,
            full: Some(v),
        });
    }

    if fields.len() > 3 || fields.is_empty() {
        return Err(RangeError::Syntax { at: start });
    }

    let mut values = [0u64; 3];
    let mut numeric = 0usize;
    let mut wildcard_seen = false;

    for (i, field) in fields.iter().enumerate() {
        if field.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if matches!(*field, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen || !field.bytes().all(|b| b.is_ascii_digit()) {
            return Err(RangeError::Syntax { at: start });
        }
        if field.len() > 1 && field.as_bytes()[0] == b'0' {
            return Err(RangeError::Syntax { at: start });
        }
        let mut n = 0u64;
        for &b in field.as_bytes() {
            n = n
                .checked_mul(10)
                .and_then(|x| x.checked_add((b - b'0') as u64))
                .ok_or(RangeError::Syntax { at: start })?;
        }
        values[i] = n;
        numeric += 1;
    }

    Ok(Partial {
        values,
        numeric,
        full: None,
    })
}

fn endpoint(values: [u64; 3], pre: Vec<String>) -> Version {
    Version {
        major: values[0],
        minor: values[1],
        patch: values[2],
        pre,
        build: Vec::new(),
    }
}

fn lower_partial(partial: &Partial) -> Version {
    partial
        .full
        .clone()
        .unwrap_or_else(|| endpoint(partial.values, Vec::new()))
}

fn increment_bound(
    values: [u64; 3],
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut next = values;
    next[field] = next[field]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at })?;
    for item in next.iter_mut().skip(field + 1) {
        *item = 0;
    }
    Ok(endpoint(next, vec!["0".to_string()]))
}

fn add_condition(arm: &mut Arm, version: Version, op: CompareOp, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core_tuple(&version));
    }
    arm.conditions.push(Condition { version, op });
}

fn parse_expansion(
    s: &str,
    arm: &mut Arm,
    token_start: usize,
    operand_start: usize,
    token_end: usize,
    kind: u8,
) -> Result<(), RangeError> {
    let partial = parse_partial(s, operand_start, token_end)?;
    if partial.numeric == 0 {
        return Err(RangeError::Syntax { at: token_start });
    }
    let lower = lower_partial(&partial);
    if !lower.pre.is_empty() {
        arm.anchors.push(core_tuple(&lower));
    }
    add_condition(arm, lower, CompareOp::Ge, false);

    let upper = match kind {
        b'^' => {
            let field = if partial.values[0] > 0 || partial.numeric == 1 {
                0
            } else if partial.values[1] > 0 || partial.numeric == 2 {
                1
            } else {
                2
            };
            increment_bound(partial.values, field, operand_start)?
        }
        b'~' => {
            let field = if partial.numeric == 1 { 0 } else { 1 };
            increment_bound(partial.values, field, operand_start)?
        }
        _ => {
            if partial.numeric == 1 {
                increment_bound(partial.values, 0, operand_start)?
            } else {
                increment_bound(partial.values, 1, operand_start)?
            }
        }
    };
    arm.conditions.push(Condition {
        version: upper,
        op: CompareOp::Lt,
    });
    Ok(())
}

fn parse_bare(s: &str, arm: &mut Arm, start: usize, end: usize) -> Result<(), RangeError> {
    let partial = parse_partial(s, start, end)?;
    if let Some(full) = partial.full {
        add_condition(arm, full, CompareOp::Eq, true);
        return Ok(());
    }

    if partial.numeric == 0 {
        return Ok(());
    }

    let lower = lower_partial(&partial);
    add_condition(arm, lower, CompareOp::Ge, false);
    let upper = if partial.numeric == 1 {
        increment_bound(partial.values, 0, start)?
    } else {
        increment_bound(partial.values, 1, start)?
    };
    arm.conditions.push(Condition {
        version: upper,
        op: CompareOp::Lt,
    });
    Ok(())
}

fn parse_comparator(
    s: &str,
    arm: &mut Arm,
    start: usize,
    end: usize,
) -> Result<(), RangeError> {
    let bytes = s.as_bytes();
    let (op, op_len) = if bytes[start] == b'<' {
        if start + 1 < end && bytes[start + 1] == b'=' {
            (CompareOp::Le, 2)
        } else {
            (CompareOp::Lt, 1)
        }
    } else if bytes[start] == b'>' {
        if start + 1 < end && bytes[start + 1] == b'=' {
            (CompareOp::Ge, 2)
        } else {
            (CompareOp::Gt, 1)
        }
    } else if bytes[start] == b'=' {
        (CompareOp::Eq, 1)
    } else {
        return Err(RangeError::Syntax { at: start });
    };

    let operand_start = start + op_len;
    if operand_start == end || !bytes[operand_start].is_ascii_digit() {
        return Err(RangeError::Syntax { at: start });
    }
    let version = version_at(s, operand_start, end)?;
    add_condition(arm, version, op, true);
    Ok(())
}

fn parse_arm(s: &str, range_start: usize, range_end: usize) -> Result<Arm, RangeError> {
    let toks = tokens(s, range_start, range_end);
    let mut arm = Arm {
        conditions: Vec::new(),
        anchors: Vec::new(),
    };
    if toks.is_empty() {
        return Err(RangeError::Empty { at: range_start });
    }

    let standalone_minus = toks.iter().find(|&&(a, b)| &s[a..b] == "-").copied();
    if let Some((minus_start, _)) = standalone_minus {
        if toks.len() != 3 || &s[toks[1].0..toks[1].1] != "-" {
            return Err(RangeError::Syntax { at: minus_start });
        }
        let (lo_start, lo_end) = toks[0];
        let (hi_start, hi_end) = toks[2];
        let lo = parse_partial(s, lo_start, lo_end)?;
        let hi = parse_partial(s, hi_start, hi_end)?;
        if lo.numeric == 0 {
            return Err(RangeError::Syntax { at: lo_start });
        }
        if hi.numeric == 0 {
            return Err(RangeError::Syntax { at: hi_start });
        }
        let lower = lower_partial(&lo);
        if !lower.pre.is_empty() {
            arm.anchors.push(core_tuple(&lower));
        }
        add_condition(&mut arm, lower, CompareOp::Ge, false);

        if let Some(full) = hi.full {
            add_condition(&mut arm, full, CompareOp::Le, true);
        } else {
            let upper = if hi.numeric == 1 {
                increment_bound(hi.values, 0, hi_start)?
            } else {
                increment_bound(hi.values, 1, hi_start)?
            };
            arm.conditions.push(Condition {
                version: upper,
                op: CompareOp::Lt,
            });
        }
        return Ok(arm);
    }

    for (start, end) in toks {
        let bytes = s.as_bytes();
        match bytes[start] {
            b'<' | b'>' | b'=' => parse_comparator(s, &mut arm, start, end)?,
            b'^' | b'~' => {
                if start + 1 == end {
                    return Err(RangeError::Syntax { at: start });
                }
                parse_expansion(s, &mut arm, start, start + 1, end, bytes[start])?;
            }
            b'*' | b'x' | b'X' | b'0'..=b'9' => parse_bare(s, &mut arm, start, end)?,
            _ => return Err(RangeError::Syntax { at: start }),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut i = 0usize;

    while i < bytes.len() {
        if ws(bytes[i]) {
            i += 1;
            continue;
        }
        if bytes[i] == b'|' && i + 1 < bytes.len() && bytes[i + 1] == b'|' {
            let sep_start = i;
            let before = tokens(s, arm_start, i);
            if before.is_empty() {
                return Err(RangeError::Empty { at: sep_start });
            }
            arms.push(parse_arm(s, arm_start, i)?);

            i += 2;
            if i < bytes.len() && !ws(bytes[i]) {
                // A glued || is part of an operand, whose parser supplies the error.
                let next_end = bytes[i..]
                    .iter()
                    .position(|&b| ws(b))
                    .map(|offset| i + offset)
                    .unwrap_or(bytes.len());
                let start = arm_start;
                let mut temporary = Arm {
                    conditions: Vec::new(),
                    anchors: Vec::new(),
                };
                let token_start = tokens(s, arm_start, i)
                    .last()
                    .map(|x| x.0)
                    .unwrap_or(i);
                let _ = (&mut temporary, next_end, start, token_start);
                return Err(RangeError::Syntax { at: i });
            }
            arm_start = i;
            continue;
        }
        i += 1;
    }

    let trailing_ws = arm_start;
    let final_tokens = tokens(s, arm_start, bytes.len());
    if final_tokens.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: bytes.len() });
    }
    arms.push(parse_arm(s, arm_start, bytes.len()).map_err(|e| {
        if matches!(e, RangeError::Empty { .. }) {
            RangeError::Empty { at: trailing_ws }
        } else {
            e
        }
    })?);

    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        (!v.pre.is_empty() && arm.anchors.iter().any(|&tuple| tuple == core_tuple(v)))
            .then_some(())
            .is_some()
            .or(v.pre.is_empty())
            && arm.conditions.iter().all(|condition| condition.accepts(v))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate)
            .map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(current) => {
                let order = compare_versions(version, &parsed[current]);
                if (want_max && order == Ordering::Greater)
                    || (!want_max && order == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }
    Ok(best.map(|index| candidates[index].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

fn main() {}
