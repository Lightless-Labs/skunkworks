use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }

    if *pos == start {
        return match bytes.get(*pos).copied() {
            None | Some(b'.' | b'-' | b'+') => {
                Err(VersionError::MissingCore { at: *pos })
            }
            Some(_) => Err(VersionError::InvalidChar { at: *pos }),
        };
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }

    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    bytes: &[u8],
    pos: &mut usize,
    stop_at_plus: bool,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut identifiers = Vec::new();

    loop {
        let start = *pos;
        while *pos < bytes.len()
            && bytes[*pos] != b'.'
            && !(stop_at_plus && bytes[*pos] == b'+')
        {
            let b = bytes[*pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let ident = &s[start..*pos];
        if prerelease
            && ident.len() > 1
            && ident.bytes().all(is_digit)
            && ident.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: start });
        }

        identifiers.push(ident.to_owned());

        if *pos >= bytes.len() {
            break;
        }

        if bytes[*pos] == b'.' {
            *pos += 1;
            continue;
        }

        if stop_at_plus && bytes[*pos] == b'+' {
            break;
        }

        break;
    }

    Ok(identifiers)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;

    let major = parse_core(s, bytes, &mut pos)?;
    if bytes.get(pos) != Some(&b'.') {
        return match bytes.get(pos).copied() {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            Some(_) => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;

    let minor = parse_core(s, bytes, &mut pos)?;
    if bytes.get(pos) != Some(&b'.') {
        return match bytes.get(pos).copied() {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            Some(_) => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;

    let patch = parse_core(s, bytes, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if let Some(b'-') = bytes.get(pos).copied() {
        pos += 1;
        pre = parse_identifiers(s, bytes, &mut pos, true, true)?;
    }

    if let Some(b'+') = bytes.get(pos).copied() {
        pos += 1;
        build = parse_identifiers(s, bytes, &mut pos, false, false)?;
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    match (numeric_identifier(a), numeric_identifier(b)) {
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let order = compare_identifier(x, y);
                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, op_len) = if token.starts_with(">=") {
            (Op::Ge, 2)
        } else if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with('>') {
            (Op::Gt, 1)
        } else if token.starts_with('<') {
            (Op::Lt, 1)
        } else if token.starts_with('=') {
            (Op::Eq, 1)
        } else {
            (Op::Eq, 0)
        };

        let operand_start = token_start + op_len;
        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { comparators })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let admitted = v.pre.is_empty()
        || r.comparators.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && (
                    comparator.version.major,
                    comparator.version.minor,
                    comparator.version.patch,
                ) == (v.major, v.minor, v.patch)
        });

    admitted
        && r.comparators.iter().all(|comparator| {
            let order = compare_versions(v, &comparator.version);
            match comparator.op {
                Op::Eq => order == Ordering::Equal,
                Op::Lt => order == Ordering::Less,
                Op::Le => order != Ordering::Greater,
                Op::Gt => order == Ordering::Greater,
                Op::Ge => order != Ordering::Less,
            }
        })
}

fn main() {}
