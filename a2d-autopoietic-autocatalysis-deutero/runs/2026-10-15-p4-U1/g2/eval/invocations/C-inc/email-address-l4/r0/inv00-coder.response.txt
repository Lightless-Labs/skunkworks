#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn pos(positions: &[usize], i: usize, end: usize) -> usize {
    positions.get(i).copied().unwrap_or(end)
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_comment = 0usize;
    let mut protected_quote = false;
    let mut escaped_quote = false;

    while i < input.len() {
        let b = input[i];

        if protected_quote {
            out.push(b);
            positions.push(i);
            if escaped_quote {
                escaped_quote = false;
            } else if b == b'\\' {
                if i + 1 < input.len() {
                    escaped_quote = true;
                }
            } else if b == b'"' {
                protected_quote = false;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                let Some(&next) = input.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !(0x20..=0x7e).contains(&next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                let Some(end) = fold_end(input, i) else {
                    return Err(EmailError::BadFold { at: i });
                };
                i = end;
                continue;
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if b == b'"' && out.is_empty() {
            protected_quote = true;
            out.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if b == b'(' || is_hws(b) || b == b'\r' || b == b'\n' {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if c == b'(' {
                    depth = 1;
                    outer_comment = i;
                    i += 1;
                    break;
                }
                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                break;
            }
            while depth > 0 {
                if i >= input.len() {
                    return Err(EmailError::UnclosedComment { at: outer_comment });
                }
                let c = input[i];
                if c == b'\\' {
                    let Some(&next) = input.get(i + 1) else {
                        return Err(EmailError::CommentEscape { at: i });
                    };
                    if !(0x20..=0x7e).contains(&next) {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                } else if c == b'\r' || c == b'\n' {
                    let Some(end) = fold_end(input, i) else {
                        return Err(EmailError::BadFold { at: i });
                    };
                    i = end;
                } else if c == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                } else if c == b')' {
                    depth -= 1;
                    i += 1;
                } else if c == b'\t' || (0x20..=0x7e).contains(&c) {
                    i += 1;
                } else {
                    return Err(EmailError::CommentChar { at: i });
                }
            }
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        out.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    Ok(Cleaned {
        bytes: out,
        positions,
        runs,
    })
}

fn is_atext(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else if is_atext(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    !previous_dot
}

fn parse_core(
    input: &str,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;
    let positions = &cleaned.positions;
    for run in &cleaned.runs {
        let left_at = run.start > 0 && bytes.get(run.start - 1) == Some(&b'@');
        let right_at = run.start < bytes.len() && bytes.get(run.start) == Some(&b'@');
        if run.start != 0 && run.start != bytes.len() && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary {
                at: run.start.min(raw.len()),
            });
        }
    }
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(positions, 0, raw.len()),
            });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !(quote_end.is_some() && i <= protected_end) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(positions, ats[1], raw.len()),
        });
    }
    let separator = ats[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(positions, separator, raw.len()),
        });
    }

    let mut decoded = Vec::new();
    let mut origins = Vec::new();
    let quoted = bytes[0] == b'"';
    if quoted {
        let close = quote_end.unwrap();
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal {
                at: pos(positions, close + 1, raw.len()),
            });
        }
        let mut i = 1;
        while i < close {
            if bytes[i] == b'\\' {
                let Some(&next) = bytes.get(i + 1) else {
                    return Err(EmailError::BadEscape {
                        at: pos(positions, i, raw.len()),
                    });
                };
                if !matches!(next, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: pos(positions, i, raw.len()),
                    });
                }
                decoded.push(next);
                origins.push(pos(positions, i, raw.len()));
                i += 2;
            } else {
                let b = bytes[i];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(positions, i, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(pos(positions, i, raw.len()));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(positions, close, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if separator > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(positions, 64, raw.len()),
            });
        }
        let mut previous_dot = false;
        for i in 0..separator {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == separator || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: pos(positions, i, raw.len()),
                    });
                }
                previous_dot = true;
            } else {
                if !is_atext(b) {
                    return Err(EmailError::LocalChar {
                        at: pos(positions, i, raw.len()),
                    });
                }
                previous_dot = false;
            }
            decoded.push(b);
            origins.push(pos(positions, i, raw.len()));
        }
    }

    let domain_start = separator + 1;
    let domain_raw = &bytes[domain_start..];
    let mut domain_len = domain_raw.len();
    let root_pos;
    if domain_len > 0 && domain_raw[domain_len - 1] == b'.' {
        domain_len -= 1;
        root_pos = pos(positions, domain_start + domain_len, raw.len());
    } else {
        root_pos = raw.len();
    }
    if domain_len == 0 {
        return Err(EmailError::EmptyDomain {
            at: pos(positions, domain_start, raw.len()),
        });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(positions, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;
    for i in 0..=domain_len {
        if i == domain_len || domain_raw[i] == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }
    for (start, end) in &labels {
        if start == end {
            let at = if *start == domain_len && root_pos != raw.len() {
                root_pos
            } else if *start < domain_len {
                pos(positions, domain_start + *start, raw.len())
            } else {
                pos(positions, domain_start + *start, raw.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(positions, domain_start + start + 63, raw.len()),
            });
        }
        let label = &domain_raw[*start..*end];
        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(positions, domain_start + *start, raw.len()),
            });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(positions, domain_start + end - 1, raw.len()),
            });
        }
        for (j, &b) in label.iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar {
                    at: pos(positions, domain_start + start + j, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: pos(positions, domain_start, raw.len()),
        });
    }
    let (last_start, last_end) = labels[labels.len() - 1];
    if domain_raw[last_start..last_end]
        .iter()
        .all(|b| b.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: pos(positions, domain_start + last_start, raw.len()),
        });
    }

    let local = String::from_utf8(decoded).unwrap();
    let domain = domain_raw[..domain_len]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();
    Ok((
        Address {
            local,
            domain,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(index) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if index == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(index);
        }
    }

    let safe = valid_dot_atom(address.local.as_bytes());
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    addresses.push(address);
                    source_indices.push(index);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

fn main() {}
