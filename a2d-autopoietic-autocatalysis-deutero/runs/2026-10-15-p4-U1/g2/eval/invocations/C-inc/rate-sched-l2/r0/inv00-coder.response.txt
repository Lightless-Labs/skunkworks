#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        for (i, b) in text.bytes().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = text
            .bytes()
            .enumerate()
            .filter_map(|(i, b)| if b == b' ' { Some(i) } else { None })
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == text.len() {
            return Err(CronError::Layout(split));
        }

        let minute_text = &text[..split];
        let hour_text = &text[split + 1..];
        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(minute_text, 0, 59, &mut minutes)?;
        parse_field(hour_text, split + 1, 23, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .find(',')
            .map(|i| start + i)
            .unwrap_or(field.len());
        parse_item(&field[start..end], offset + start, max, selected)?;
        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    if item.is_empty() {
        return Err(CronError::Syntax(offset));
    }

    let bytes = item.as_bytes();
    if let Some(slash) = bytes.iter().position(|&b| b == b'/') {
        if let Some(second) = bytes[slash + 1..].iter().position(|&b| b == b'/') {
            return Err(CronError::Syntax(offset + slash + 1 + second));
        }
    }

    let slash = bytes.iter().position(|&b| b == b'/');
    let base_end = slash.unwrap_or(bytes.len());
    let base = &item[..base_end];
    let step_text = slash.map(|i| &item[i + 1..]);

    let is_star = base == "*";
    let dash = base.as_bytes().iter().position(|&b| b == b'-');

    if step_text.is_some() && !is_star && dash.is_none() {
        return Err(CronError::Syntax(offset + slash.unwrap()));
    }

    let (range_start, range_end, base_start_offset, dash_offset) = if is_star {
        (0usize, max, offset, None)
    } else if let Some(dash) = dash {
        let left = &base[..dash];
        let right = &base[dash + 1..];
        let left_value = parse_number(left, offset, max)?;
        let right_value = parse_number(right, offset + dash + 1, max)?;
        if left_value > right_value {
            return Err(CronError::Descending(offset + dash));
        }
        (left_value, right_value, offset, Some(dash))
    } else {
        let n = parse_number(base, offset, max)?;
        (n, n, offset, None)
    };

    let step = if let Some(step_text) = step_text {
        let step_offset = offset + slash.unwrap() + 1;
        let value = parse_number(step_text, step_offset, usize::MAX)?;
        let step_max = if max == 59 { 60 } else { 24 };
        if value == 0 || value > step_max {
            return Err(CronError::Step(step_offset));
        }
        value
    } else {
        1
    };

    let _ = (base_start_offset, dash_offset);
    let mut value = range_start;
    while value <= range_end {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_number(fragment: &str, offset: usize, max: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();
    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in bytes {
        let digit = (b - b'0') as usize;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(offset)),
        };
        if value > max {
            return Err(CronError::Bounds(offset));
        }
    }
    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    added_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
            added_after: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let interval_start = match (lower_exclusive, job.added_after) {
                (Some(last), _) => last,
                (None, Some(added)) => added,
                (None, None) => 0,
            };
            let first_minute = if lower_exclusive.is_none() && job.added_after.is_none() {
                0
            } else {
                interval_start / 60_000 + 1
            };
            let last_minute = now / 60_000;
            if first_minute > last_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    for minute in first_minute..=last_minute {
                        if job.cron.matches(minute) {
                            let at = minute * 60_000;
                            if lower_exclusive.map_or(true, |last| at > last)
                                && job.added_after.map_or(true, |added| at > added)
                            {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
                Policy::Once => {
                    let mut latest = None;
                    for minute in first_minute..=last_minute {
                        if job.cron.matches(minute) {
                            let at = minute * 60_000;
                            if lower_exclusive.map_or(true, |last| at > last)
                                && job.added_after.map_or(true, |added| at > added)
                            {
                                latest = Some(at);
                            }
                        }
                    }
                    if let Some(at) = latest {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0
                        && lower_exclusive.map_or(true, |last| now > last)
                        && job.added_after.map_or(true, |added| now > added)
                        && job.cron.matches(now / 60_000)
                    {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|item| item.0).collect())
    }
}

fn main() {}
