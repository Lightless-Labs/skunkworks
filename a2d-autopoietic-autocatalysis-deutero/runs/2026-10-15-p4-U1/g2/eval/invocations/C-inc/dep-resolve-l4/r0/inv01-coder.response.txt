fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
        && bytes.last() != Some(&b'-')
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut result = cs.to_vec();
    result.sort_by_key(|c| constraint_key(*c));
    result.dedup();
    result
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            for e in sorted_evidence(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{base}.gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{base}.feature[{f}]"),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    feature_requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    evidence: BTreeMap<String, Vec<Evidence>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    let mut features = dep.features.clone();
    features.sort();
    features.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features.join(","),
        gate,
        platform
    )
}

fn root_token(index: usize, root: &Requirement) -> String {
    let mut features = root.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        features.join(",")
    )
}

fn compute_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut c = Closure::default();

    for (i, root) in roots.iter().enumerate() {
        let token = root_token(i, root);
        c.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        c.feature_requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        c.evidence
            .entry(root.package.clone())
            .or_default()
            .push(Evidence {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        let path = vec![token];
        let entry = c
            .paths
            .entry(root.package.clone())
            .or_insert_with(|| path.clone());
        if (path.len(), &path) < (entry.len(), entry) {
            *entry = path;
        }
    }

    loop {
        let mut changed = false;
        let package_names: Vec<String> = selected.keys().cloned().collect();

        for package_name in package_names {
            let Some(package) = catalog.get(&package_name) else {
                continue;
            };
            let version = selected[&package_name];
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requests = c
                .feature_requests
                .get(&package_name)
                .cloned()
                .unwrap_or_default();

            for feature_name in requests {
                if c.active_features
                    .entry(package_name.clone())
                    .or_default()
                    .insert(feature_name)
                {
                    changed = true;
                }
            }

            let active_names = c
                .active_features
                .get(&package_name)
                .cloned()
                .unwrap_or_default();
            for feature_name in active_names {
                if let Some(decl) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &decl.enables {
                        if c.feature_requests
                            .entry(package_name.clone())
                            .or_default()
                            .insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
            }

            for (index, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|gate| {
                        !c.active_features
                            .get(&package_name)
                            .is_some_and(|set| set.contains(gate))
                    })
                {
                    continue;
                }

                c.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                c.evidence
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Evidence {
                        constraint: dep.constraint,
                        path: {
                            let mut path =
                                c.paths.get(&package_name).cloned().unwrap_or_default();
                            path.push(dep_token(&package_name, version, index, dep));
                            path
                        },
                    });

                let mut path = c.paths.get(&package_name).cloned().unwrap_or_default();
                path.push(dep_token(&package_name, version, index, dep));
                match c.paths.get(&dep.package) {
                    None => {
                        c.paths.insert(dep.package.clone(), path);
                        changed = true;
                    }
                    Some(old) if (path.len(), &path) < (old.len(), old) => {
                        c.paths.insert(dep.package.clone(), path);
                        changed = true;
                    }
                    _ => {}
                }

                let requests = c
                    .feature_requests
                    .entry(dep.package.clone())
                    .or_default();
                for feature in &dep.features {
                    if requests.insert(feature.clone()) {
                        changed = true;
                    }
                }
            }
        }

        if !changed {
            break;
        }
    }

    for constraints in c.constraints.values_mut() {
        *constraints = canonical_constraints(constraints);
    }
    c
}

fn matching_releases<'a>(
    package: &'a Package,
    constraints: &[Constraint],
) -> Vec<&'a Release> {
    let mut releases: Vec<&Release> = package
        .releases
        .iter()
        .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
        .collect();
    releases.sort_by(|a, b| b.version.cmp(&a.version));
    releases
}

struct Search<'a> {
    catalog: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn inspect(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = compute_closure(&self.catalog, self.roots, selected, self.target);

        for package in closure.constraints.keys() {
            if !self.catalog.contains_key(package) {
                return Err(ResolveError::Missing {
                    package: package.clone(),
                    evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
                });
            }
        }

        for (package, version) in selected.iter() {
            let constraints = closure.constraints.get(package).cloned().unwrap_or_default();
            if !constraints.iter().all(|c| matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
                });
            }
        }

        for (package, version) in selected.iter() {
            let release = self.catalog[package]
                .releases
                .iter()
                .find(|r| r.version == *version)
                .unwrap();
            if let Some(features) = closure.active_features.get(package) {
                for feature in features {
                    if !release.features.iter().any(|decl| decl.name == *feature) {
                        return Err(ResolveError::MissingFeature {
                            package: package.clone(),
                            selected: *version,
                            feature: feature.clone(),
                            path: closure.paths.get(package).cloned().unwrap_or_default(),
                        });
                    }
                }
            }
        }

        for package in closure.constraints.keys() {
            if selected.contains_key(package) {
                continue;
            }
            let constraints = &closure.constraints[package];
            if matching_releases(self.catalog[package], constraints).is_empty() {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                    evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
                });
            }
        }

        let next = closure
            .constraints
            .keys()
            .find(|package| !selected.contains_key(*package))
            .cloned();

        let Some(package) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let vector = self
                .catalog
                .keys()
                .map(|name| {
                    (
                        name.clone(),
                        selected.get(name).copied().unwrap_or(Version {
                            major: 0,
                            minor: 0,
                            patch: 0,
                        }),
                    )
                })
                .collect::<Vec<_>>();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_vector, _)) => {
                    cost < *best_cost || (cost == *best_cost && vector > *best_vector)
                }
            };
            if replace {
                self.best = Some((cost, vector, selected.clone()));
            }
            return Ok(());
        };

        let constraints = &closure.constraints[&package];
        let mut candidates = matching_releases(self.catalog[&package], constraints);
        if let Some(lock_version) = self.locks.get(&package) {
            if let Some(pos) = candidates.iter().position(|r| r.version == *lock_version) {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        let mut first_child_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate.version);
            match self.inspect(selected) {
                Ok(()) => {}
                Err(ResolveError::Limit { .. }) => {
                    selected.remove(&package);
                    return Err(ResolveError::Limit {
                        attempts: self.max_attempts,
                    });
                }
                Err(err) => {
                    if first_child_error.is_none() {
                        first_child_error = Some(err.clone());
                    }
                }
            }
            selected.remove(&package);
        }

        if let Some(err) = first_child_error {
            if self.first_error.is_none() {
                self.first_error = Some(err.clone());
            }
            Err(err)
        } else {
            Ok(())
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();

    let mut search = Search {
        catalog: catalog_map,
        roots,
        locks: lock_map.clone(),
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    let mut selected = BTreeMap::new();
    match search.inspect(&mut selected) {
        Err(err @ ResolveError::Limit { .. }) => return Err(err),
        Err(err) => {
            return Err(search.first_error.unwrap_or(err));
        }
        Ok(()) => {}
    }

    let Some((_, _, selected)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_closure = compute_closure(&search.catalog, roots, &selected, target);
    let selected_vec = selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut names = BTreeSet::new();
    names.extend(lock_map.keys().cloned());
    names.extend(selected.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .catalog
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_closure
                    .constraints
                    .get(&package)
                    .cloned()
                    .unwrap_or_default()
                    .iter()
                    .all(|c| matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}

fn main() {}
