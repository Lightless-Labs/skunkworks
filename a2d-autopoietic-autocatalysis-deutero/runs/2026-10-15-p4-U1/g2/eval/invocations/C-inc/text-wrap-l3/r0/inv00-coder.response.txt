#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn validate_markers(input: &str) -> Result<(), WrapError> {
    let chars: Vec<(usize, char)> = input.char_indices().collect();

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }

        let valid_neighbor = |j: usize| {
            j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
        };

        if i == 0 || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
}

impl Line {
    fn content_len(&self) -> usize {
        self.pieces.iter().map(|s| s.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn render(&self, extra_spaces: Option<&[usize]>) -> String {
        let mut out = " ".repeat(self.indent);
        for (i, piece) in self.pieces.iter().enumerate() {
            if i > 0 {
                let n = extra_spaces.map_or(1, |spaces| spaces[i - 1]);
                out.push_str(&" ".repeat(n));
            }
            out.push_str(piece);
        }
        out
    }
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut breaks = Vec::new();

    let finish = |chars: &mut Vec<char>, breaks: &mut Vec<usize>, words: &mut Vec<Word>| {
        if !chars.is_empty() {
            words.push(Word {
                chars: std::mem::take(chars),
                breaks: std::mem::take(breaks),
            });
        }
    };

    for &ch in segment {
        if is_separator(ch) {
            finish(&mut chars, &mut breaks, &mut words);
        } else if ch == '~' {
            breaks.push(chars.len());
        } else {
            chars.push(ch);
        }
    }
    finish(&mut chars, &mut breaks, &mut words);
    words
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    let mut lines = Vec::new();
    let mut line = Line {
        indent: options.first_indent,
        pieces: Vec::new(),
    };
    let mut first_line = true;

    let emit = |line: &mut Line, lines: &mut Vec<Line>, first_line: &mut bool| {
        if line.content_len() > 0 {
            lines.push(line.clone());
            *first_line = false;
            line.indent = options.rest_indent;
            line.pieces.clear();
        }
    };

    for word in words {
        let mut offset = 0usize;
        while offset < word.chars.len() {
            let capacity = options.width - line.indent;
            let current_len = line.content_len();
            let room = capacity.saturating_sub(current_len + usize::from(current_len > 0));
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room {
                line.pieces.push(word.chars[offset..].iter().collect());
                offset = word.chars.len();
                continue;
            }

            let available = if current_len == 0 {
                capacity
            } else {
                capacity.saturating_sub(current_len + 1)
            };

            let usable: Vec<usize> = word
                .breaks
                .iter()
                .copied()
                .filter(|&p| p > offset && p < word.chars.len() && p - offset + 1 <= available)
                .collect();

            let selected = usable.last().copied();
            if let Some(mark) = selected {
                let prefix_len = mark - offset;
                let used = if current_len == 0 {
                    prefix_len + 1
                } else {
                    current_len + 1 + prefix_len + 1
                };
                let unused = capacity.saturating_sub(used);
                let accept = current_len == 0
                    || unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                        < capacity.saturating_sub(current_len).saturating_mul(
                            capacity.saturating_sub(current_len),
                        );

                if accept {
                    line.pieces.push(
                        word.chars[offset..mark]
                            .iter()
                            .chain(std::iter::once(&'-'))
                            .collect(),
                    );
                    emit(&mut line, &mut lines, &mut first_line);
                    offset = mark + 1;
                    continue;
                }
            }

            if current_len > 0 {
                emit(&mut line, &mut lines, &mut first_line);
                continue;
            }

            if let Some(mark) = selected {
                line.pieces.push(
                    word.chars[offset..mark]
                        .iter()
                        .chain(std::iter::once(&'-'))
                        .collect(),
                );
                emit(&mut line, &mut lines, &mut first_line);
                offset = mark + 1;
                continue;
            }

            let take = capacity.min(suffix_len);
            line.pieces.push(word.chars[offset..offset + take].iter().collect());
            emit(&mut line, &mut lines, &mut first_line);
            offset += take;
        }
    }

    if line.content_len() > 0 {
        lines.push(line);
    }

    lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    validate_markers(input)?;

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let count = i - start;
            if count >= 2 {
                let words = parse_words(&segment);
                let lines = layout_paragraph(&words, options);
                append_rendered(&mut output, lines, options);
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(count - 1));
                continue;
            }
            segment.push('\n');
        } else {
            segment.push(normalized[i]);
            i += 1;
        }
    }

    let words = parse_words(&segment);
    let lines = layout_paragraph(&words, options);
    append_rendered(&mut output, lines, options);
    Ok(output)
}

fn append_rendered(output: &mut Vec<String>, lines: Vec<Line>, options: &WrapOptions) {
    let final_index = lines.len().saturating_sub(1);
    for (line_index, line) in lines.iter().enumerate() {
        let gaps = line.pieces.len().saturating_sub(1);
        if options.justify && line_index < final_index && gaps > 0 {
            let extra = options.width.saturating_sub(line.indent + line.content_len());
            let each = extra / gaps;
            let remainder = extra % gaps;
            let mut spaces = vec![1 + each; gaps];
            for k in 0..remainder {
                spaces[(line_index + k) % gaps] += 1;
            }
            output.push(line.render(Some(&spaces)));
        } else {
            output.push(line.render(None));
        }
    }
}

fn main() {}
