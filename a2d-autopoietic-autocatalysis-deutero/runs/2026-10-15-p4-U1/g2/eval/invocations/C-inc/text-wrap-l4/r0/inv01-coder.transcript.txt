pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let char_count = input.chars().count();
    if char_count > 512 {
        return Err(WrapError::InputTooLong { chars: char_count });
    }

    let mut byte_index = 0usize;
    for c in input.chars() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        byte_index += c.len_utf8();
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for (i, &(byte, c)) in original.iter().enumerate() {
        if c == '~' {
            let valid_neighbor = |j: isize| -> bool {
                if j < 0 || j as usize >= original.len() {
                    return false;
                }
                let x = original[j as usize].1;
                x != '~' && !is_separator(x)
            };
            if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: byte });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let words = parse_segment(&segment);
                output.extend(
                    solve_paragraph(&words, options)
                        .into_iter()
                        .map(|l| l.text),
                );
                segment.clear();
                output.extend(std::iter::repeat_n(String::new(), j - i - 1));
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    let words = parse_segment(&segment);
    output.extend(
        solve_paragraph(&words, options)
            .into_iter()
            .map(|l| l.text),
    );
    Ok(output)
}

        while i < chars.len() && !is_separator(chars[i]) {
            if chars[i] == '~' {
                markers.push(spelling.len());
            } else {
                spelling.push(chars[i]);
            }
            i += 1;
        }
        words.push(Word {
            chars: spelling,
            markers,
        });
    }
    words
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_count != b.marker_count {
        return a.marker_count < b.marker_count;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed.iter().zip(&b.consumed) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.strings.iter().zip(&b.strings) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn solve_paragraph(words: &[Word], opts: &WrapOptions) -> Vec<Line> {
    if words.is_empty() {
        return Vec::new();
    }

    fn recurse(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        opts: &WrapOptions,
        memo: &mut std::collections::HashMap<(usize, usize, bool), Solution>,
    ) -> Solution {
        if wi == words.len() {
            return Solution {
                cost: 0,
                marker_count: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                strings: Vec::new(),
            };
        }

        let key = (wi, off, first);
        if let Some(solution) = memo.get(&key) {
            return solution.clone();
        }

        let indent = if first {
            opts.first_indent
        } else {
            opts.rest_indent
        };
        let cap = opts.width - indent;
        let mut best: Option<Solution> = None;

        let mut consider = |end_wi: usize,
                            end_off: usize,
                            display: Vec<char>,
                            pieces: usize,
                            consumed: usize,
                            marked: bool,
                            raw: bool| {
            let final_line = end_wi == words.len();
            let display_count = display.len() + indent;
            if display_count == indent {
                return;
            }

            let gap = opts.width - display_count;
            let mut line_text = " ".repeat(indent);
            line_text.extend(display);

            let tail = recurse(end_wi, end_off, false, words, opts, memo);
            let mut cost = tail.cost
                + if final_line {
                    (gap * gap / 4) as u64
                } else {
                    (gap * gap) as u64
                };
            if marked {
                cost += opts.hyphen_cost as u64;
            }
            if raw {
                cost += (2 * opts.width + 1) as u64;
            }
            if !final_line && pieces == 1 {
                cost += 3;
            }

            let mut candidate = Solution {
                cost,
                marker_count: tail.marker_count + usize::from(marked),
                lines: Vec::with_capacity(tail.lines.len() + 1),
                consumed: Vec::with_capacity(tail.consumed.len() + 1),
                strings: Vec::with_capacity(tail.strings.len() + 1),
            };
            candidate.lines.push(Line {
                text: line_text.clone(),
                pieces,
                gap,
            });
            candidate.lines.extend(tail.lines);
            candidate.consumed.push(consumed);
            candidate.consumed.extend(tail.consumed);
            candidate.strings.push(line_text);
            candidate.strings.extend(tail.strings);

            if best.as_ref().map_or(true, |current| better(&candidate, current)) {
                best = Some(candidate);
            }
        };

        let mut display = Vec::new();
        let mut cur_wi = wi;
        let mut piece_count = 0usize;
        let mut consumed_count = 0usize;

        loop {
            let word = &words[cur_wi];
            let remain = word.chars.len() - if cur_wi == wi { off } else { 0 };
            let word_off = if cur_wi == wi { off } else { 0 };
            let sep = usize::from(piece_count > 0);

            if display.len() + sep + remain > cap {
                break;
            }
            if sep == 1 {
                display.push(' ');
            }
            display.extend_from_slice(&word.chars[word_off..]);
            piece_count += 1;
            consumed_count += remain;

            consider(
                cur_wi + 1,
                0,
                display.clone(),
                piece_count,
                consumed_count,
                false,
                false,
            );

            cur_wi += 1;
            if cur_wi == words.len() || display.len() + 1 >= cap {
                break;
            }
        }

        let word = &words[wi];
        let remaining = word.chars.len() - off;
        let marker_fits: Vec<usize> = word
            .markers
            .iter()
            .copied()
            .filter(|&marker| marker > off && marker < word.chars.len())
            .filter(|&marker| marker - off + 1 <= cap)
            .collect();

        for &marker in &marker_fits {
            let n = marker - off;
            if n == 0 || n + 1 > cap {
                continue;
            }
            let mut d = word.chars[off..marker].to_vec();
            d.push('-');
            consider(wi, marker, d, 1, n, true, false);
        }

        if piece_count == 0 && remaining > cap && marker_fits.is_empty() {
            let end = off + cap;
            consider(
                if end == word.chars.len() { wi + 1 } else { wi },
                if end == word.chars.len() { 0 } else { end },
                word.chars[off..end].to_vec(),
                1,
                cap,
                false,
                true,
            );
        }

        let result = best.unwrap_or_else(|| {
            let end = (off + cap).min(words[wi].chars.len());
            let complete = end == words[wi].chars.len();
            recurse(
                if complete { wi + 1 } else { wi },
                if complete { 0 } else { end },
                false,
                words,
                opts,
                memo,
            )
        });

        memo.insert(key, result.clone());
        result
    }

    let mut memo = std::collections::HashMap::new();
    let mut solution = recurse(0, 0, true, words, opts, &mut memo);

    if opts.justify {
        let line_count = solution.lines.len();
        for (idx, line) in solution.lines.iter_mut().enumerate() {
            if idx + 1 == line_count || line.pieces < 2 {
                continue;
            }

            let extra = line.gap;
            let gaps = line.pieces - 1;
            let base = extra / gaps;
            let rem = extra % gaps;
            let indent = line.text.chars().take_while(|&c| c == ' ').count();
            let body: String = line.text.chars().skip(indent).collect();
            let parts: Vec<&str> = body.split(' ').collect();
            let start = idx % gaps;

            let mut rebuilt = " ".repeat(indent);
            for p in 0..parts.len() {
                if p > 0 {
                    let gap_index = p - 1;
                    let add = base + usize::from((gap_index + gaps - start) % gaps < rem);
                    rebuilt.push_str(&" ".repeat(1 + add));
                }
                rebuilt.push_str(parts[p]);
            }
            line.text = rebuilt;
        }
    }

    solution.lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let char_count = input.chars().count();
    if char_count > 512 {
        return Err(WrapError::InputTooLong { chars: char_count });
    }

    let mut byte_index = 0usize;
    for c in input.chars() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        byte_index += c.len_utf8();
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for (i, &(byte, c)) in original.iter().enumerate() {
        if c == '~' {
            let valid_neighbor = |j: isize| -> bool {
                if j < 0 || j as usize >= original.len() {
                    return false;
                }
                let x = original[j as usize].1;
                x != '~' && !is_separator(x)
            };
            if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: byte });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let words = parse_segment(&segment);
                output.extend(
                    solve_paragraph(&words, options)
                        .into_iter()
                        .map(|line| line.text),
                );
                segment.clear();
                output.extend(std::iter::repeat_n(String::new(), j - i - 1));
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }

    let words = parse_segment(&segment);
    output.extend(
        solve_paragraph(&words, options)
            .into_iter()
            .map(|line| line.text),
    );
    Ok(output)
}
