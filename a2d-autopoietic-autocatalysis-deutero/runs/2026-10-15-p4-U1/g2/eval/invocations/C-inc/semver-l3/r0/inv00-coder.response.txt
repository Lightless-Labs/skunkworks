use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    allow_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < bytes.len() && bytes[p] != b'.' && bytes[p] != b'+' {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if !allow_numeric_leading_zero
            && id.len() > 1
            && id.bytes().all(is_digit)
            && id.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if p >= bytes.len() || bytes[p] == b'+' {
            break;
        }
        p += 1;
        if p == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core_number(s, &mut p)?;
    for _ in 0..2 {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        if s.as_bytes()[p] != b'.' {
            return Err(VersionError::InvalidChar { at: p });
        }
        p += 1;
        if _ == 0 {
            break;
        }
    }
    let minor = parse_core_number(s, &mut p)?;
    if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
        return Err(VersionError::MissingCore { at: p });
    }
    if s.as_bytes()[p] != b'.' {
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let patch = parse_core_number(s, &mut p)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() {
        match s.as_bytes()[p] {
            b'-' => {
                p += 1;
                let pre_start = p;
                let end = s[p..].find('+').map(|i| p + i).unwrap_or(s.len());
                pre = parse_identifiers(&s[..end], pre_start, false)?;
                p = end;
                if p < s.len() {
                    p += 1;
                    build = parse_identifiers(s, p, true)?;
                }
            }
            b'+' => {
                p += 1;
                build = parse_identifiers(s, p, true)?;
            }
            _ => return Err(VersionError::InvalidChar { at: p }),
        }
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric_id(x);
        let yn = is_numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    generated: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let b = s.as_bytes();
    let mut p = 0;
    while p < b.len() {
        while p < b.len() && whitespace(b[p]) {
            p += 1;
        }
        if p == b.len() {
            break;
        }
        let start = p;
        while p < b.len() && !whitespace(b[p]) {
            p += 1;
        }
        out.push((start, &s[start..p]));
    }
    out
}

fn next_core(v: &Version, k: usize) -> Option<(u64, u64, u64)> {
    match k {
        1 => Some((v.major.checked_add(1)?, 0, 0)),
        2 => Some((v.major, v.minor.checked_add(1)?, 0)),
        _ => Some((v.major, v.minor, v.patch.checked_add(1)?)),
    }
}

fn synthetic(core: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: core.0,
        minor: core.1,
        patch: core.2,
        pre,
        build: Vec::new(),
    }
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, generated: bool) {
    if !generated && !version.pre.is_empty() {
        arm.anchors
            .push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator {
        op,
        version,
        generated,
    });
}

fn version_operand(s: &str, offset: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at: offset, error })
}

#[derive(Clone)]
struct Partial {
    version: Version,
    k: usize,
    full: bool,
}

fn parse_operand(s: &str, offset: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|x| matches!(*x, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        let v = version_operand(s, offset)?;
        return Ok(Partial {
            version: v,
            k: 3,
            full: true,
        });
    }
    if fields.len() > 3 {
        return Err(RangeError::Syntax { at: offset });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in &fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at: offset });
        }
        if matches!(*f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at: offset });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: offset })?;
            nums.push(n);
        }
    }
    let k = nums.len();
    let version = synthetic(
        (
            *nums.get(0).unwrap_or(&0),
            *nums.get(1).unwrap_or(&0),
            *nums.get(2).unwrap_or(&0),
        ),
        Vec::new(),
    );
    Ok(Partial {
        version,
        k,
        full: false,
    })
}

fn expand_operand(
    arm: &mut Arm,
    token: &str,
    token_start: usize,
    operand: &str,
    operand_start: usize,
    kind: char,
) -> Result<(), RangeError> {
    let p = parse_operand(operand, operand_start)?;
    if kind == '^' || kind == '~' {
        if p.k == 0 {
            return Err(RangeError::Syntax { at: token_start });
        }
        add_cmp(arm, Op::Ge, p.version.clone(), false);
        let (ma, mi, pa) = (p.version.major, p.version.minor, p.version.patch);
        let core = if kind == '^' {
            if ma > 0 || p.k == 1 {
                ma.checked_add(1).map(|m| (m, 0, 0))
            } else if mi > 0 || p.k == 2 {
                mi.checked_add(1).map(|m| (ma, m, 0))
            } else {
                pa.checked_add(1).map(|p| (ma, mi, p))
            }
        } else if p.k == 1 {
            ma.checked_add(1).map(|m| (m, 0, 0))
        } else {
            mi.checked_add(1).map(|m| (ma, m, 0))
        }
        .ok_or(RangeError::Syntax { at: operand_start })?;
        add_cmp(arm, Op::Lt, synthetic(core, vec!["0".into()]), true);
    } else if p.full {
        add_cmp(arm, Op::Eq, p.version, false);
    } else if p.k == 0 {
        // All wildcards match all releases.
    } else {
        add_cmp(arm, Op::Ge, p.version.clone(), false);
        let core = next_core(&p.version, p.k)
            .ok_or(RangeError::Syntax { at: operand_start })?;
        add_cmp(arm, Op::Lt, synthetic(core, vec!["0".into()]), true);
    }
    let _ = token;
    Ok(())
}

fn parse_comparator(
    arm: &mut Arm,
    token: &str,
    start: usize,
) -> Result<(), RangeError> {
    let (op, n) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else {
        (Op::Eq, usize::from(token.starts_with('=')))
    };
    let operand = &token[n..];
    let operand_start = start + n;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: start });
    }
    let v = version_operand(operand, operand_start)?;
    add_cmp(arm, op, v, false);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[(usize, &str)], dash_index: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if tokens.len() != 3 || dash_index != 1 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: tokens[dash_index].0 });
    }
    let (ls, lower) = tokens[0];
    let (us, upper) = tokens[2];
    let lo = parse_operand(lower, ls)?;
    let hi = parse_operand(upper, us)?;
    if lo.k == 0 {
        return Err(RangeError::Syntax { at: ls });
    }
    if hi.k == 0 {
        return Err(RangeError::Syntax { at: us });
    }
    add_cmp(arm, Op::Ge, lo.version.clone(), false);
    if hi.full {
        add_cmp(arm, Op::Le, hi.version, false);
    } else {
        let core = next_core(&hi.version, hi.k)
            .ok_or(RangeError::Syntax { at: us })?;
        add_cmp(arm, Op::Lt, synthetic(core, vec!["0".into()]), true);
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = split_tokens(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].1 == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].0 });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
        } else {
            current.push(tokens[i]);
        }
        i += 1;
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn parse_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    if let Some(idx) = tokens.iter().position(|(_, t)| *t == "-") {
        parse_hyphen_arm(tokens, idx, &mut arm)?;
        return Ok(arm);
    }
    for &(start, token) in tokens {
        if token.starts_with('^') || token.starts_with('~') {
            let kind = token.as_bytes()[0] as char;
            expand_operand(&mut arm, token, start, &token[1..], start + 1, kind)?;
        } else if token.starts_with('<')
            || token.starts_with('>')
            || token.starts_with('=')
        {
            parse_comparator(&mut arm, token, start)?;
        } else if token.starts_with('*') || token.starts_with('x') || token.starts_with('X') {
            expand_operand(&mut arm, token, start, token, start, 'b')?;
        } else if token.as_bytes().first().is_some_and(u8::is_ascii_digit) {
            expand_operand(&mut arm, token, start, token, start, 'b')?;
        } else {
            return Err(RangeError::Syntax { at: start });
        }
    }
    Ok(arm)
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn admitted(v: &Version, arm: &Arm) -> bool {
    arm.anchors.iter().any(|t| *t == tuple(v))
}

fn cmp_match(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn matches_arm(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty() && !include_prerelease && !admitted(v, arm) {
        return false;
    }
    arm.comparators.iter().all(|c| cmp_match(v, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| matches_arm(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (want_max && ord == Ordering::Greater)
                    || (!want_max && ord == Ordering::Less)
                {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn arm_bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                set_lower(&mut lower, c.version.clone(), true);
                set_upper(&mut upper, c.version.clone(), true);
            }
            Op::Gt | Op::Ge => set_lower(&mut lower, c.version.clone(), matches!(c.op, Op::Ge)),
            Op::Lt | Op::Le => set_upper(&mut upper, c.version.clone(), matches!(c.op, Op::Le)),
        }
    }
    (lower, upper)
}

fn set_lower(dst: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match dst {
        None => *dst = Some((v, inclusive)),
        Some((old, old_inc)) => match compare_versions(&v, old) {
            Ordering::Greater => *dst = Some((v, inclusive)),
            Ordering::Equal if !inclusive && *old_inc => *dst = Some((v, inclusive)),
            _ => {}
        },
    }
}

fn set_upper(dst: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match dst {
        None => *dst = Some((v, inclusive)),
        Some((old, old_inc)) => match compare_versions(&v, old) {
            Ordering::Less => *dst = Some((v, inclusive)),
            Ordering::Equal if !inclusive && *old_inc => *dst = Some((v, inclusive)),
            _ => {}
        },
    }
}

fn fmt_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn arm_string(arm: &Arm) -> String {
    let (lo, hi) = arm_bounds(arm);
    let lower = lo.map_or("*".to_string(), |(v, inc)| {
        format!("{}{}", if inc { ">=" } else { ">" }, fmt_version(&v))
    });
    let upper = hi.map_or("*".to_string(), |(v, inc)| {
        format!("{}{}", if inc { "<=" } else { "<" }, fmt_version(&v))
    });
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let a = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors
            .iter()
            .map(|(x, y, z)| format!("{x}.{y}.{z}"))
            .collect::<Vec<_>>()
            .join(",")
    };
    format!("[{lower};{upper};pre={a}]")
}

pub fn canonical(r: &Range) -> String {
    let mut strings: Vec<String> = r.arms.iter().map(arm_string).collect();
    strings.sort();
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".into()
    } else {
        strings.join(" || ")
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut combined = Arm {
                comparators: aa.comparators.clone(),
                anchors: aa.anchors.clone(),
            };
            combined.comparators.extend(bb.comparators.clone());
            combined.anchors.extend(bb.anchors.clone());
            let (lo, hi) = arm_bounds(&combined);
            if let (Some((l, li)), Some((u, ui))) = (&lo, &hi) {
                match compare_versions(l, u) {
                    Ordering::Greater => continue,
                    Ordering::Equal if !(*li && *ui) => continue,
                    _ => {}
                }
            }
            if include_prerelease {
                return true;
            }
            if stable_witness(&lo, &hi) {
                return true;
            }
            let mut anchors = aa.anchors.clone();
            anchors.retain(|t| bb.anchors.contains(t));
            for t in anchors {
                // The least prerelease at an admitted tuple is sufficient to test
                // the lower side; upper-side exact constraints can exclude it.
                let w = synthetic(t, vec!["0".into()]);
                if combined.comparators.iter().all(|c| cmp_match(&w, c)) {
                    return true;
                }
            }
        }
    }
    false
}

fn stable_witness(
    lo: &Option<(Version, bool)>,
    hi: &Option<(Version, bool)>,
) -> bool {
    let start = match lo {
        None => (0, 0, 0),
        Some((v, inc)) => {
            if !v.pre.is_empty() {
                tuple(v)
            } else if *inc {
                tuple(v)
            } else {
                match next_core(v, 3) {
                    Some(t) => t,
                    None => return false,
                }
            }
        }
    };
    let w = synthetic(start, Vec::new());
    match hi {
        None => true,
        Some((u, inc)) => match compare_versions(&w, u) {
            Ordering::Less => true,
            Ordering::Equal => *inc,
            Ordering::Greater => false,
        },
    }
}
