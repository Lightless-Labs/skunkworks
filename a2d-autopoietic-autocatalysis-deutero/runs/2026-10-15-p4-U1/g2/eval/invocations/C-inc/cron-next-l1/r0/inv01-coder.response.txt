#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<u8>; 5],
    day_dom_is_exact_star: bool,
    day_dow_is_exact_star: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.values[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_spec(field: Field) -> (u32, u32, u32) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    let offset = start;
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if step {
        let (_, _, width) = field_spec(field);
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max, _) = field_spec(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }

    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let (base_end, step_part) = if let Some(&slash) = slash_positions.first() {
        if slash == 0 || slash + 1 == item.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        (&item[..slash], Some((slash + 1, item.len())))
    } else {
        (item, None)
    };

    enum Base {
        Star,
        Number(usize, usize),
        Range(usize, usize, usize, usize),
    }

    let base = if base_end == b"*" {
        Base::Star
    } else {
        let hyphens: Vec<usize> = base_end
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        match hyphens.as_slice() {
            [] if !base_end.is_empty() && base_end.iter().all(u8::is_ascii_digit) => {
                Base::Number(0, base_end.len())
            }
            [hyphen]
                if *hyphen > 0
                    && *hyphen + 1 < base_end.len()
                    && base_end[..*hyphen].iter().all(u8::is_ascii_digit)
                    && base_end[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
            {
                Base::Range(0, *hyphen, *hyphen + 1, base_end.len())
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }
    };

    let (min, max, _) = field_spec(field);
    let parsed_base = match base {
        Base::Star => None,
        Base::Number(start, end) => {
            let n = parse_number(base_end, start, end, field, false)?;
            Some((n, n, false))
        }
        Base::Range(start_start, start_end, end_start, end_end) => {
            let start = parse_number(base_end, start_start, start_end, field, false)?;
            let end = parse_number(base_end, end_start, end_end, field, false)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + end_start,
                });
            }
            Some((start, end, true))
        }
    };

    let step = if let Some((step_start, step_end)) = step_part {
        Some(parse_number(item, step_start, step_end, field, true)?)
    } else {
        None
    };

    let (start, end) = match (parsed_base, step) {
        (None, _) => (min, max),
        (Some((n, _, _)), None) => (n, n),
        (Some((start, end, true)), Some(_)) => (start, end),
        (Some((start, _, false)), Some(_)) => (start, max),
    };

    if let Some(step) = step {
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
    } else {
        for value in start..=end {
            selected[value as usize] = true;
        }
    }

    Ok(())
}

fn parse_field(input: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = [false; 60];
    let mut start = 0;

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(bytes, start, index, field, &mut selected)?;
            start = index + 1;
        }
    }
    parse_item(bytes, start, bytes.len(), field, &mut selected)?;

    let (min, max, _) = field_spec(field);
    Ok((min..=max)
        .filter(|&value| selected[value as usize])
        .map(|value| value as u8)
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut values: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for (index, field) in fields.into_iter().enumerate() {
        values[index] = parse_field(parts[index], field)?;
    }

    Ok(Cron {
        values,
        day_dom_is_exact_star: parts[2] == "*",
        day_dow_is_exact_star: parts[4] == "*",
    })
}

fn main() {}
