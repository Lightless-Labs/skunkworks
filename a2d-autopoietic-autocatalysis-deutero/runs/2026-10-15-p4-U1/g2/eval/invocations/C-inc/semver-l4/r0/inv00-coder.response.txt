use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_num(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    check_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut pos = start;

    loop {
        let ident_start = pos;
        while pos < bytes.len() && bytes[pos] != b'.' && bytes[pos] != b'+' {
            let b = bytes[pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
        if ident_start == pos {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..pos];
        let numeric = ident.bytes().all(is_digit);
        if check_numeric_leading_zero && numeric && ident.len() > 1 && ident.starts_with('0') {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        out.push(ident.to_string());

        if pos == bytes.len() || bytes[pos] == b'+' {
            break;
        }
        pos += 1;
        if pos == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_num(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let minor = parse_core_num(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let patch = parse_core_num(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        pos += 1;
        let plus = s[pos..].find('+').map(|i| pos + i).unwrap_or(s.len());
        let pre_part = &s[..plus];
        // Parse pre with an explicit boundary so '+' ends the sequence.
        let mut q = pos;
        loop {
            let st = q;
            while q < plus {
                let b = s.as_bytes()[q];
                if !(b.is_ascii_alphanumeric() || b == b'-' || b == b'.') {
                    return Err(VersionError::InvalidChar { at: q });
                }
                if b == b'.' {
                    break;
                }
                q += 1;
            }
            if st == q {
                return Err(VersionError::EmptyIdentifier { at: st });
            }
            let id = &s[st..q];
            let numeric = id.bytes().all(is_digit);
            if numeric && id.len() > 1 && id.starts_with('0') {
                return Err(VersionError::LeadingZero { at: st });
            }
            pre.push(id.to_string());
            if q == plus {
                break;
            }
            q += 1;
            if q == plus {
                return Err(VersionError::EmptyIdentifier { at: q });
            }
        }
        let _ = pre_part;
        pos = plus;
    }

    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        pos += 1;
        if pos == s.len() {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
        build = parse_identifiers(s, pos, false)?;
        pos = s.len();
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version { major, minor, patch, pre, build })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() {
        return if b.is_empty() { Ordering::Equal } else { Ordering::Greater };
    }
    if b.is_empty() {
        return Ordering::Less;
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => cmp_numeric_id(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| cmp_pre(&a.pre, &b.pre))
}

fn split_fields(s: &str) -> Vec<&str> {
    s.split('.').collect()
}

fn parse_partial_or_full(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parse_version(s)
            .map(Partial::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    let parts = split_fields(s);
    if parts.len() >= 3 && !parts.iter().any(|p| matches!(*p, "*" | "x" | "X")) {
        return parse_version(s)
            .map(Partial::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    if parts.is_empty() || parts.len() > 3 || parts.iter().any(|p| p.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut saw_wild = false;
    for p in parts {
        if matches!(p, "*" | "x" | "X") {
            saw_wild = true;
        } else {
            if saw_wild || !p.bytes().all(is_digit) || (p.len() > 1 && p.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            fields.push(n);
        }
    }
    Ok(Partial::Fields(fields))
}

fn version_from_tuple(t: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version { major: t.0, minor: t.1, patch: t.2, pre, build: Vec::new() }
}

fn padded(fields: &[u64]) -> Version {
    version_from_tuple(
        (fields.get(0).copied().unwrap_or(0), fields.get(1).copied().unwrap_or(0), fields.get(2).copied().unwrap_or(0)),
        Vec::new(),
    )
}

fn bump_upper(fields: &[u64], mode: char) -> Option<Version> {
    let k = fields.len();
    let mut a = fields.get(0).copied().unwrap_or(0);
    let mut b = fields.get(1).copied().unwrap_or(0);
    let mut c = fields.get(2).copied().unwrap_or(0);
    if mode == '^' {
        if a > 0 || k == 1 {
            a = a.checked_add(1)?;
            b = 0;
            c = 0;
        } else if b > 0 || k == 2 {
            b = b.checked_add(1)?;
            c = 0;
        } else {
            c = c.checked_add(1)?;
        }
    } else if k == 1 {
        a = a.checked_add(1)?;
        b = 0;
        c = 0;
    } else {
        b = b.checked_add(1)?;
        c = 0;
    }
    Some(version_from_tuple((a, b, c), vec!["0".to_string()]))
}

fn partial_upper(fields: &[u64]) -> Option<Version> {
    match fields.len() {
        0 => None,
        1 => {
            let a = fields[0].checked_add(1)?;
            Some(version_from_tuple((a, 0, 0), vec!["0".into()]))
        }
        2 => {
            let b = fields[1].checked_add(1)?;
            Some(version_from_tuple((fields[0], b, 0), vec!["0".into()]))
        }
        _ => None,
    }
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, explicit_anchor: bool) {
    if explicit_anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.comparators.push(Comparator { op, version, explicit_anchor });
}

fn op_matches(v: &Version, c: &Comparator) -> bool {
    let cmp = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => cmp == Ordering::Equal,
        Op::Lt => cmp == Ordering::Less,
        Op::Le => cmp != Ordering::Greater,
        Op::Gt => cmp == Ordering::Greater,
        Op::Ge => cmp != Ordering::Less,
    }
}

fn token_error(s: &str, start: usize) -> RangeError {
    RangeError::Syntax { at: start + s.len() }
}

fn parse_bare_token(tok: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (mode, operand, operand_at) = if let Some(rest) = tok.strip_prefix('^') {
        (Some('^'), rest, at + 1)
    } else if let Some(rest) = tok.strip_prefix('~') {
        (Some('~'), rest, at + 1)
    } else {
        (None, tok, at)
    };
    let p = parse_partial_or_full(operand, operand_at)?;
    match (mode, p) {
        (None, Partial::Full(v)) => add_cmp(arm, Op::Eq, v, true),
        (None, Partial::Fields(fields)) => {
            if fields.is_empty() {
                return Ok(());
            }
            let low = padded(&fields);
            add_cmp(arm, Op::Ge, low, true);
            if let Some(up) = partial_upper(&fields) {
                add_cmp(arm, Op::Lt, up, false);
            }
        }
        (Some(m), Partial::Full(v)) => {
            let fields = vec![v.major, v.minor, v.patch];
            let up = bump_upper(&fields, m).ok_or(RangeError::Syntax { at: operand_at })?;
            add_cmp(arm, Op::Ge, v, true);
            add_cmp(arm, Op::Lt, up, false);
        }
        (Some(m), Partial::Fields(fields)) => {
            if fields.is_empty() {
                return Err(RangeError::Syntax { at });
            }
            let up = bump_upper(&fields, m).ok_or(RangeError::Syntax { at: operand_at })?;
            let low = padded(&fields);
            add_cmp(arm, Op::Ge, low, true);
            add_cmp(arm, Op::Lt, up, false);
        }
    }
    Ok(())
}

fn parse_comparator(tok: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, n) = if tok.starts_with(">=") { (Op::Ge, 2) }
    else if tok.starts_with("<=") { (Op::Le, 2) }
    else if tok.starts_with('>') { (Op::Gt, 1) }
    else if tok.starts_with('<') { (Op::Lt, 1) }
    else if tok.starts_with('=') { (Op::Eq, 1) }
    else { return Err(RangeError::Syntax { at }); };
    let operand = &tok[n..];
    let operand_at = at + n;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let v = parse_version(operand).map_err(|error| RangeError::Version { at: operand_at, error })?;
    add_cmp(arm, op, v, true);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[(String, usize)], arm: &mut Arm) -> Result<(), RangeError> {
    if tokens.len() != 3 || tokens[1].0 != "-" {
        let at = tokens.iter().find(|t| t.0 == "-").map(|t| t.1).unwrap_or(0);
        return Err(RangeError::Syntax { at });
    }
    let lo = parse_partial_or_full(&tokens[0].0, tokens[0].1)?;
    let hi = parse_partial_or_full(&tokens[2].0, tokens[2].1)?;
    let (lf, lv) = match lo {
        Partial::Full(v) => (vec![v.major, v.minor, v.patch], Some(v)),
        Partial::Fields(f) if !f.is_empty() => (f, None),
        Partial::Fields(_) => return Err(RangeError::Syntax { at: tokens[0].1 }),
    };
    let (hf, hv) = match hi {
        Partial::Full(v) => (vec![v.major, v.minor, v.patch], Some(v)),
        Partial::Fields(f) if !f.is_empty() => (f, None),
        Partial::Fields(_) => return Err(RangeError::Syntax { at: tokens[2].1 }),
    };
    let low = lv.unwrap_or_else(|| padded(&lf));
    add_cmp(arm, Op::Ge, low, true);
    if let Some(v) = hv {
        add_cmp(arm, Op::Le, v, true);
    } else if let Some(up) = partial_upper(&hf) {
        add_cmp(arm, Op::Lt, up, false);
    } else {
        add_cmp(arm, Op::Le, padded(&hf), false);
    }
    Ok(())
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut tokens: Vec<(String, usize)> = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) { i += 1; }
        if i == bytes.len() { break; }
        let st = i;
        while i < bytes.len() && !is_ws(bytes[i]) { i += 1; }
        tokens.push((s[st..i].to_string(), st));
    }
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current: Vec<(String, usize)> = Vec::new();
    let mut idx = 0;
    while idx < tokens.len() {
        if tokens[idx].0 == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[idx].1 });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            idx += 1;
            if idx == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(tokens[idx].clone());
            idx += 1;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    if tokens.iter().any(|t| t.0 == "-") {
        parse_hyphen_arm(tokens, &mut arm)?;
        return Ok(arm);
    }
    for (tok, at) in tokens {
        if tok.starts_with('=') || tok.starts_with('<') || tok.starts_with('>') {
            parse_comparator(tok, *at, &mut arm)?;
        } else if tok.starts_with('^') || tok.starts_with('~') || tok.starts_with_ascii_digit() || matches!(tok.as_str(), "*" | "x" | "X") {
            parse_bare_token(tok, *at, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: *at });
        }
    }
    Ok(arm)
}

impl StartsAsciiDigit for str {

        self.as_bytes().first().copied().map_or(false, |b| b.is_ascii_digit())

fn arm_admits(v: &Version, arm: &Arm) -> bool {
    arm.anchors.iter().any(|t| *t == core(v))
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparators.iter().all(|c| op_matches(v, c)) {
        return false;
    }
    v.pre.is_empty() || include_prerelease || arm_admits(v, arm)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

fn candidates_impl(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(j) = best {
            let c = compare_versions(&parsed[i], &parsed[j]);
            if (greatest && c == Ordering::Greater) || (!greatest && c == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    candidates_impl(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    candidates_impl(candidates, r, false)
}

pub fn intersects(_a: &Range, _b: &Range, _include_prerelease: bool) -> bool {
    false
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        for c in &arm.comparators {
            match c.op {
                Op::Eq => {
                    lower = Some((c.version.clone(), true));
                    upper = Some((c.version.clone(), true));
                }
                Op::Ge | Op::Gt => {
                    let inc = matches!(c.op, Op::Ge);
                    let replace = lower.as_ref().map_or(true, |(v, old_inc)| {
                        let cmp = compare_versions(&c.version, v);
                        cmp == Ordering::Greater || (cmp == Ordering::Equal && !inc && *old_inc)
                    });
                    if replace { lower = Some((c.version.clone(), inc)); }
                }
                Op::Le | Op::Lt => {
                    let inc = matches!(c.op, Op::Le);
                    let replace = upper.as_ref().map_or(true, |(v, old_inc)| {
                        let cmp = compare_versions(&c.version, v);
                        cmp == Ordering::Less || (cmp == Ordering::Equal && !inc && *old_inc)
                    });
                    if replace { upper = Some((c.version.clone(), inc)); }
                }
            }
        }
        let lo = lower.map(|(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, display_version(&v))).unwrap_or("*".into());
        let hi = upper.map(|(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, display_version(&v))).unwrap_or("*".into());
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anc = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{lo};{hi};pre={anc}]"));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    parsed: &mut Vec<ParsedLeaf>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let range = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            parsed.push(ParsedLeaf { path: path.clone(), range });
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, parsed)?;
                path.pop();
            }
        }
    }
    Ok(())
}

#[derive(Clone)]
struct ParsedLeaf {
    path: Vec<usize>,
    range: Range,
}

fn eval_expr(expr: &SetExpr, parsed: &mut std::slice::Iter<'_, ParsedLeaf>, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let leaf = parsed.next().unwrap();
            matches_with(v, &leaf.range, include)
        }
        SetExpr::Union(children) => children.iter().any(|c| eval_expr(c, parsed, v, include)),
        SetExpr::Intersection(children) => children.iter().all(|c| eval_expr(c, parsed, v, include)),
    }
}

fn qualifies(
    package: &Package,
    leaves: &[Vec<ParsedLeaf>],
    v: &Version,
    include: bool,
    constraint_indices: &[usize],
) -> bool {
    for &ci in constraint_indices {
        let mut iter = leaves[ci].iter();
        if !eval_expr(&package.constraints[ci].expr, &mut iter, v, include) {
            return false;
        }
    }
    true
}

fn conflict_labels(
    package: &Package,
    parsed: &[Vec<ParsedLeaf>],
    versions: &[Version],
    include: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if versions.is_empty() { return Vec::new(); }
    for size in 1..=n {
        let mut chosen = Vec::new();
        let mut found: Vec<Vec<String>> = Vec::new();
        fn search(
            start: usize,
            size: usize,
            chosen: &mut Vec<usize>,
            found: &mut Vec<Vec<String>>,
            package: &Package,
            parsed: &[Vec<ParsedLeaf>],
            versions: &[Version],
            include: bool,
        ) {
            if chosen.len() == size {
                if !versions.iter().any(|v| qualifies(package, parsed, v, include, chosen)) {
                    let mut labels = chosen.iter().map(|&i| package.constraints[i].label.clone()).collect::<Vec<_>>();
                    labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                    found.push(labels);
                }
                return;
            }
            for i in start..package.constraints.len() {
                chosen.push(i);
                search(i + 1, size, chosen, found, package, parsed, versions, include);
                chosen.pop();
            }
        }
        search(0, size, &mut chosen, &mut found, package, parsed, versions, include);
        if !found.is_empty() {
            found.sort_by(|a,b| {
                for (x,y) in a.iter().zip(b.iter()) {
                    let c = x.as_bytes().cmp(y.as_bytes());
                    if c != Ordering::Equal { return c; }
                }
                a.len().cmp(&b.len())
            });
            return found.remove(0);
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut all_versions = Vec::new();
    for p in packages {
        let mut versions = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            versions.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index, error
            })?);
        }
        all_versions.push(versions);
    }
    let mut all_leaves = Vec::new();
    for p in packages {
        let mut per = Vec::new();
        for c in &p.constraints {
            let mut leaves = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut leaves)?;
            per.push(leaves);
        }
        all_leaves.push(per);
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &all_versions[pi];
        let leaves = &all_leaves[pi];
        let all_indices = (0..p.constraints.len()).collect::<Vec<_>>();
        let mut best: Option<usize> = None;
        for i in 0..versions.len() {
            if !qualifies(p, leaves, &versions[i], include_prerelease, &all_indices) { continue; }
            let Some(j) = best else { best = Some(i); continue; };
            let c = compare_versions(&versions[i], &versions[j]);
            let take = match preference {
                Preference::Highest => c == Ordering::Greater,
                Preference::Lowest => c == Ordering::Less,
                Preference::StableFirst => {
                    let si = versions[i].pre.is_empty();
                    let sj = versions[j].pre.is_empty();
                    if si != sj { si } else { c == Ordering::Greater }
                }
            };
            if take { best = Some(i); }
        }
        if let Some(i) = best {
            selections.push(Selection { package: p.name.clone(), version: p.candidates[i].clone() });
        } else {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflict_labels(p, leaves, versions, include_prerelease),
            });
        }
    }
    Ok(selections)
}

fn main() {}
