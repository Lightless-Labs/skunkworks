#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
enum Phase {
    Fresh,
    Resumed { remaining: u64 },
}

struct Entry {
    job: Job,
    state: State,
    phase: Phase,
    waiting_since: u64,
    started_at: u64,
}

pub struct Pool {
    entries: Vec<Entry>,
    by_id: std::collections::HashMap<String, usize>,
    deps: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    first_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        fn valid_id(s: &str) -> bool {
            !s.is_empty()
                && s.len() <= 16
                && s.bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        }

        fn valid_tag(s: &str) -> bool {
            !s.is_empty()
                && s.len() <= 8
                && s.bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_deref().is_some_and(|tag| !valid_tag(tag)) {
                return Err(Error::BadTag(i));
            }
            if by_id.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut deps = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&d) = by_id.get(dep) else {
                    return Err(Error::MissingDependency(i, j));
                };
                deps[i].push(d);
            }
        }

        fn visit(i: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[i] == 1 {
                return false;
            }
            if marks[i] == 2 {
                return true;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[i] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &deps, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .map(|job| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                waiting_since: job.release_at,
                started_at: 0,
                phase: Phase::Fresh,
                job,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            deps,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.by_id
            .get(id)
            .map(|&i| self.entries[i].state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut trace = Vec::new();
        let mut time = if self.first_run {
            0
        } else {
            self.horizon.saturating_add(1)
        };

        while time <= until {
            let mut completed: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| match e.state {
                    State::Running { finish_at, .. } if finish_at == time => Some(i),
                    _ => None,
                })
                .collect();
            completed.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

            for i in completed {
                let attempt = match self.entries[i].state {
                    State::Running { attempt, .. } => attempt,
                    _ => continue,
                };
                let id = self.entries[i].job.id.clone();
                let success = attempt > self.entries[i].job.failures;
                trace.push(format!(
                    "{} finish {} {} {}",
                    time,
                    id,
                    attempt,
                    if success { "ok" } else { "fail" }
                ));
                if success {
                    self.entries[i].state = State::Succeeded;
                } else if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self
                        .retry_base
                        .checked_mul(1u64.checked_shl(shift).unwrap_or(u64::MAX))
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready = time.saturating_add(delay);
                    self.entries[i].state = State::Waiting {
                        ready_at: ready,
                        attempt,
                    };
                    self.entries[i].phase = Phase::Fresh;
                    self.entries[i].waiting_since = ready;
                    trace.push(format!("{} retry {} {} {}", time, id, attempt, ready));
                }
            }

            loop {
                let mut newly_blocked = Vec::new();
                loop {
                    let mut changed = false;
                    for i in 0..self.entries.len() {
                        if matches!(self.entries[i].state, State::Waiting { .. })
                            && self.deps[i].iter().any(|&d| {
                                matches!(
                                    self.entries[d].state,
                                    State::Failed | State::Blocked
                                )
                            })
                        {
                            self.entries[i].state = State::Blocked;
                            newly_blocked.push(i);
                            changed = true;
                        }
                    }
                    if !changed {
                        break;
                    }
                }
                newly_blocked.sort_by(|&a, &b| {
                    self.entries[a].job.id.cmp(&self.entries[b].job.id)
                });
                for i in newly_blocked {
                    let dep = self.deps[i]
                        .iter()
                        .filter(|&&d| {
                            matches!(
                                self.entries[d].state,
                                State::Failed | State::Blocked
                            )
                        })
                        .min_by_key(|&&d| self.entries[d].job.id.as_str())
                        .copied();
                    if let Some(d) = dep {
                        trace.push(format!(
                            "{} skip {} {}",
                            time, self.entries[i].job.id, self.entries[d].job.id
                        ));
                    }
                }

                let mut candidates: Vec<usize> = (0..self.entries.len())
                    .filter(|&i| {
                        matches!(
                            self.entries[i].state,
                            State::Waiting { ready_at, .. } if ready_at <= time
                        ) && self.deps[i]
                            .iter()
                            .all(|&d| matches!(self.entries[d].state, State::Succeeded))
                    })
                    .collect();

                candidates.sort_by(|&a, &b| {
                    let pa = self.effective_priority(a, time);
                    let pb = self.effective_priority(b, time);
                    pb.cmp(&pa).then(a.cmp(&b))
                });

                let mut did_action = false;
                for candidate in candidates {
                    let tag = self.entries[candidate].job.tag.as_deref();
                    let running: Vec<usize> = self
                        .entries
                        .iter()
                        .enumerate()
                        .filter_map(|(i, e)| {
                            matches!(e.state, State::Running { .. }).then_some(i)
                        })
                        .collect();
                    let same_tag_running = tag.is_some_and(|t| {
                        running.iter().any(|&i| {
                            self.entries[i].job.tag.as_deref() == Some(t)
                        })
                    });

                    if running.len() < self.capacity && !same_tag_running {
                        self.start(candidate, time, &mut trace);
                        did_action = true;
                        break;
                    }

                    let mut victims: Vec<usize> = running
                        .iter()
                        .copied()
                        .filter(|&v| {
                            self.entries[candidate].job.priority
                                >= self.entries[v].job.priority.saturating_add(2)
                                && (tag.is_none()
                                    || self.entries[v].job.tag.as_deref() != tag)
                        })
                        .collect();

                    victims.sort_by(|&a, &b| {
                        self.entries[a]
                            .job
                            .priority
                            .cmp(&self.entries[b].job.priority)
                            .then_with(|| {
                                self.entries[b].started_at.cmp(&self.entries[a].started_at)
                            })
                            .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                    });

                    for victim in victims {
                        let after_count = running.len() - 1;
                        let victim_tag = self.entries[victim].job.tag.as_deref();
                        let slot_free = after_count < self.capacity;
                        let tag_free = tag.is_none()
                            || (victim_tag == tag)
                            || !running.iter().any(|&i| {
                                i != victim && self.entries[i].job.tag.as_deref() == tag
                            });

                        if slot_free && tag_free {
                            let (attempt, finish_at) = match self.entries[victim].state {
                                State::Running { attempt, finish_at } => (attempt, finish_at),
                                _ => continue,
                            };
                            let remaining = finish_at - time;
                            let victim_id = self.entries[victim].job.id.clone();
                            let candidate_id = self.entries[candidate].job.id.clone();
                            trace.push(format!(
                                "{} preempt {} by {}",
                                time, victim_id, candidate_id
                            ));
                            self.entries[victim].state = State::Waiting {
                                ready_at: time,
                                attempt,
                            };
                            self.entries[victim].phase = Phase::Resumed { remaining };
                            self.entries[victim].waiting_since = time;
                            self.start(candidate, time, &mut trace);
                            did_action = true;
                            break;
                        }
                    }

                    if did_action {
                        break;
                    }
                }

                if !did_action {
                    break;
                }
            }

            let next_running = self
                .entries
                .iter()
                .filter_map(|e| match e.state {
                    State::Running { finish_at, .. } if finish_at > time => Some(finish_at),
                    _ => None,
                })
                .min();
            let next_waiting = self
                .entries
                .iter()
                .filter_map(|e| match e.state {
                    State::Waiting { ready_at, .. } if ready_at > time => Some(ready_at),
                    _ => None,
                })
                .min();

            let next = match (next_running, next_waiting) {
                (Some(a), Some(b)) => Some(a.min(b)),
                (Some(a), None) => Some(a),
                (None, Some(b)) => Some(b),
                (None, None) => None,
            };

            let Some(next) = next else { break };
            if next > until {
                break;
            }
            time = next;
        }

        self.horizon = until;
        self.first_run = false;
        Ok(trace)
    }

    fn effective_priority(&self, i: usize, time: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.entries[i].job.priority;
        }
        let age = time.saturating_sub(self.entries[i].waiting_since) / self.aging_ms;
        (u64::from(self.entries[i].job.priority) + age).min(9) as u8
    }

    fn start(&mut self, i: usize, time: u64, trace: &mut Vec<String>) {
        let (attempt, remaining) = match self.entries[i].phase {
            Phase::Fresh => {
                let attempt = match self.entries[i].state {
                    State::Waiting { attempt, .. } => attempt.saturating_add(1),
                    _ => 1,
                };
                (attempt, self.entries[i].job.duration)
            }
            Phase::Resumed { remaining } => {
                let attempt = match self.entries[i].state {
                    State::Waiting { attempt, .. } => attempt,
                    _ => 1,
                };
                (attempt, remaining)
            }
        };
        let finish_at = time.saturating_add(remaining);
        let id = self.entries[i].job.id.clone();
        trace.push(format!("{} start {} {} {}", time, id, attempt, remaining));
        self.entries[i].state = State::Running { attempt, finish_at };
        self.entries[i].phase = Phase::Fresh;
        self.entries[i].started_at = time;
    }
}

fn main() {}
