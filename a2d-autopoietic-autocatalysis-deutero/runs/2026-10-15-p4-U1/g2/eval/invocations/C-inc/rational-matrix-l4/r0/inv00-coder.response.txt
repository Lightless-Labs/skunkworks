use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128
        || n > i64::MAX as i128
        || d < 1
        || d > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        rational_from_i128(a + b, self.den as i128 * rhs.den as i128)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        rational_from_i128(a - b, self.den as i128 * rhs.den as i128)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse {
                        at: i,
                        kind: ParseKind::InvalidByte,
                    });
                }
                slash = Some(i);
                continue;
            }
            if b.is_ascii_digit() || (b == b'-' && i == 0) {
                continue;
            }
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');
        if num_start == num_end {
            return Err(RationalError::Parse {
                at: num_end,
                kind: ParseKind::InvalidByte,
            });
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse {
                    at: bytes.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if bytes[0] == b'-' && num_digits == b"0" {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        let den_digits = slash.map(|s| &bytes[s + 1..]);
        if let Some(digits) = den_digits {
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 2,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let mut nmag: u128 = 0;
        for &b in num_digits {
            nmag = nmag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
        }
        let negative = bytes[0] == b'-';
        let nlimit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        if nmag > nlimit {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::OutOfRange,
            });
        }

        let mut dmag: u128 = 1;
        if let Some(digits) = den_digits {
            dmag = 0;
            for &b in digits {
                dmag = dmag
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
            }
            if dmag > i64::MAX as u128 {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: ParseKind::OutOfRange,
                });
            }
            if dmag == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: ParseKind::ZeroDenominator,
                });
            }
        }

        if let Some(s) = slash {
            if dmag == 1 {
                return Err(RationalError::Parse {
                    at: s,
                    kind: ParseKind::NonCanonical,
                });
            }
            if nmag == 0 {
                return Err(RationalError::Parse {
                    at: s,
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(nmag, dmag) > 1 {
                return Err(RationalError::Parse {
                    at: s,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let n = if negative {
            if nmag == (1u128 << 63) {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        Ok(Rational {
            num: n,
            den: dmag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

fn checked_len(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

fn eliminate(
    mut matrix: Matrix,
    candidate_cols: usize,
) -> Result<Reduction, MatrixError> {
    let mut p = 0usize;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();

    for col in 0..candidate_cols {
        if p == matrix.rows {
            break;
        }

        let mut chosen: Option<(usize, i64, u64)> = None;
        for row in p..matrix.rows {
            let value = matrix.data[row * matrix.cols + col];
            if value.num == 0 {
                continue;
            }
            let key = (value.den, value.num.unsigned_abs());
            if chosen
                .map(|(best_row, den, mag)| {
                    key < (den, mag) || (key == (den, mag) && row < best_row)
                })
                .unwrap_or(true)
            {
                chosen = Some((row, key.0, key.1));
            }
        }

        let Some((q, _, _)) = chosen else {
            continue;
        };

        if q != p {
            for c in 0..matrix.cols {
                matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * matrix.cols + col];
        for c in 0..matrix.cols {
            let idx = p * matrix.cols + c;
            matrix.data[idx] = matrix.data[idx].div(&pivot).map_err(|source| {
                MatrixError::Arithmetic {
                    row: p,
                    col: c,
                    step: p,
                    source,
                }
            })?;
        }

        for row in 0..matrix.rows {
            if row == p {
                continue;
            }
            let factor = matrix.data[row * matrix.cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..matrix.cols {
                let product = factor
                    .mul(&matrix.data[p * matrix.cols + c])
                    .map_err(|source| MatrixError::Arithmetic {
                        row,
                        col: c,
                        step: p,
                        source,
                    })?;
                let idx = row * matrix.cols + c;
                matrix.data[idx] = matrix.data[idx].sub(&product).map_err(|source| {
                    MatrixError::Arithmetic {
                        row,
                        col: c,
                        step: p,
                        source,
                    }
                })?;
            }
        }

        pivots.push(col);
        p += 1;
    }

    Ok(Reduction {
        matrix,
        pivots,
        swaps,
    })
}

impl Matrix {
    pub fn new(
        rows: usize,
        cols: usize,
        data: Vec<Rational>,
    ) -> Result<Self, MatrixError> {
        let expected = checked_len(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Matrix {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = checked_len(self.rows, rhs.cols)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[row * self.cols + k];
                    let right = rhs.data[k * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step: k,
                            source,
                        }
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Matrix {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        self.rref().map(|reduction| reduction.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = checked_len(n, width)?;
        let mut data = Vec::with_capacity(len);

        for row in 0..n {
            for col in 0..n {
                data.push(self.data[row * n + col]);
            }
            for col in 0..n {
                data.push(if row == col {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }

        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduction = eliminate(augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for row in 0..n {
            for col in 0..n {
                inverse_data.push(reduction.matrix.data[row * width + n + col]);
            }
        }

        Ok(Matrix {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let n = self.cols;
        let k = rhs.cols;
        let width = n
            .checked_add(k)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let augmented_len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented_data = Vec::with_capacity(augmented_len);
        for row in 0..self.rows {
            let a_start = row * n;
            let b_start = row * k;
            augmented_data.extend_from_slice(&self.data[a_start..a_start + n]);
            augmented_data.extend_from_slice(&rhs.data[b_start..b_start + k]);
        }

        let augmented = Matrix {
            rows: self.rows,
            cols: width,
            data: augmented_data,
        };
        let reduction = eliminate(augmented, n).map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let coefficients_zero = (0..n)
                .all(|col| reduction.matrix.data[row * width + col].num == 0);
            if coefficients_zero {
                if let Some(rhs_col) =
                    (0..k).find(|&col| reduction.matrix.data[row * width + n + col].num != 0)
                {
                    let witness =
                        reduction.matrix.data[row * width..(row + 1) * width].to_vec();
                    return Ok(Solution::None {
                        row,
                        rhs_col,
                        witness,
                    });
                }
            }
        }

        let mut free = Vec::new();
        let mut pivoted = vec![false; n];
        for &col in &reduction.pivots {
            pivoted[col] = true;
        }
        for col in 0..n {
            if !pivoted[col] {
                free.push(col);
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(
            MatrixError::ShapeOverflow,
        ))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular_data = vec![zero; particular_len];

        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular_data[pivot_col * k + rhs_col] =
                    reduction.matrix.data[pivot_row * width + n + rhs_col];
            }
        }

        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        if reduction.pivots.len() == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: reduction.pivots,
            });
        }

        let basis_len = n.checked_mul(free.len()).ok_or(SolveError::Matrix(
            MatrixError::ShapeOverflow,
        ))?;
        let mut basis_data = vec![zero; basis_len];

        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + basis_col] =
                Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
                let coefficient =
                    reduction.matrix.data[pivot_row * width + free_col];
                basis_data[pivot_col * free.len() + basis_col] =
                    coefficient.neg().map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: basis_col,
                            step: reduction.pivots.len(),
                            source,
                        })
                    })?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: basis_len.checked_div(n.max(1)).unwrap_or(0),
                data: basis_data,
            },
            pivots: reduction.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn main() {}
