#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_day: bool,
    unrestricted_weekday: bool,
}

#[derive(Clone, Copy)]
struct Domain {
    min: u8,
    max: u8,
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u8> {
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => match upper.as_slice() {
            b"JAN" => Some(1),
            b"FEB" => Some(2),
            b"MAR" => Some(3),
            b"APR" => Some(4),
            b"MAY" => Some(5),
            b"JUN" => Some(6),
            b"JUL" => Some(7),
            b"AUG" => Some(8),
            b"SEP" => Some(9),
            b"OCT" => Some(10),
            b"NOV" => Some(11),
            b"DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_slice() {
            b"SUN" => Some(0),
            b"MON" => Some(1),
            b"TUE" => Some(2),
            b"WED" => Some(3),
            b"THU" => Some(4),
            b"FRI" => Some(5),
            b"SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn is_atom(field: Field, bytes: &[u8]) -> bool {
    !bytes.is_empty()
        && (bytes.iter().all(u8::is_ascii_digit) || name_value(field, bytes).is_some())
}

fn parse_number(field: Field, bytes: &[u8], offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    Ok(value)
}

fn parse_endpoint(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if let Some(v) = name_value(field, bytes) {
        return Ok((v as u32, false));
    }
    let n = parse_number(field, bytes, offset)?;
    let d = domain(field);
    if n < d.min as u32 || n > d.max as u32 {
        return Err(parse_error(field, offset, 3));
    }
    Ok((n, true))
}

fn parse_field(field: Field, text: &str) -> Result<(Vec<bool>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let d = domain(field);
    let mut selected = vec![false; d.max as usize + 1];

    let mut item_start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item_bytes = item.as_bytes();
        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, b)| if *b == b'/' { Some(i) } else { None })
            .collect();

        let (base, step_part) = match slash_positions.as_slice() {
            [] => (item_bytes, None),
            [slash] => (&item_bytes[..*slash], Some((*slash + 1, &item_bytes[*slash + 1..]))),
            _ => return Err(parse_error(field, item_start, 0)),
        };

        let (is_star, start_part, end_part) = if base == b"*" {
            (true, None, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, b)| if *b == b'-' { Some(i) } else { None })
                .collect();
            match hyphens.as_slice() {
                [] if is_atom(field, base) => (false, Some((0usize, base)), None),
                [hyphen]
                    if is_atom(field, &base[..*hyphen])
                        && is_atom(field, &base[*hyphen + 1..]) =>
                {
                    (
                        false,
                        Some((0usize, &base[..*hyphen])),
                        Some((*hyphen + 1, &base[*hyphen + 1..])),
                    )
                }
                _ => return Err(parse_error(field, item_start, 0)),
            }
        };

        let mut start = d.min as u32;
        let mut end = d.max as u32;

        if !is_star {
            let (start_offset_in_item, start_bytes) = start_part.unwrap();
            let (start_value, start_numeric) =
                parse_endpoint(field, start_bytes, item_start + start_offset_in_item)?;
            start = start_value;

            if let Some((end_offset_in_item, end_bytes)) = end_part {
                let (end_value, end_numeric) =
                    parse_endpoint(field, end_bytes, item_start + end_offset_in_item)?;
                end = end_value;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + end_offset_in_item,
                    });
                }
                let _ = (start_numeric, end_numeric);
            } else {
                end = if step_part.is_some() {
                    d.max as u32
                } else {
                    start
                };
            }
        }

        let step = if let Some((step_offset, step_bytes)) = step_part {
            if step_bytes.is_empty() || !step_bytes.iter().all(u8::is_ascii_digit) {
                return Err(parse_error(field, item_start, 0));
            }
            let step_value = parse_number(field, step_bytes, item_start + step_offset)?;
            let width = (d.max - d.min + 1) as u32;
            if step_value == 0 || step_value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + step_offset,
                });
            }
            step_value
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }

    Ok((selected, text == "*"))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];
    let mut unrestricted_day = false;
    let mut unrestricted_weekday = false;

    for (i, field) in fields.iter().copied().enumerate() {
        let (values, unrestricted) = parse_field(field, pieces[i])?;
        selected[i] = values;
        if field == Field::DayOfMonth {
            unrestricted_day = unrestricted;
        } else if field == Field::DayOfWeek {
            unrestricted_weekday = unrestricted;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_day,
        unrestricted_weekday,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut result = 6u16;
    for y in 2000..year {
        result = (result + if leap_year(y) { 366 } else { 365 }) % 7;
    }
    for m in 1..month {
        result = (result + month_length(year, m) as u16) % 7;
    }
    (result + day as u16 - 1) as u8 % 7
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let d = domain(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, yes)| {
                if *yes && value >= d.min as usize && value <= d.max as usize {
                    Some(value as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];

        match (self.unrestricted_day, self.unrestricted_weekday) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if at.day == month_length(at.year, at.month) => dom && dow,
            (false, false) => dom || dow,
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    if current.day < month_length(current.year, current.month) {
                        current.day += 1;
                    } else {
                        current.day = 1;
                        if current.month < 12 {
                            current.month += 1;
                        } else {
                            current.month = 1;
                            if current.year == 2399 {
                                return Ok(None);
                            }
                            current.year += 1;
                        }
                    }
                }
            }

            if self.matches_valid(current) {
                return Ok(Some(current));
            }
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
        }
    }
}
