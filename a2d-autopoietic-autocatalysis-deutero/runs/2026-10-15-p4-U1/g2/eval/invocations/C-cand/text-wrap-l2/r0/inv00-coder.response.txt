#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized: String = input
        .replace("\r\n", "\n")
        .replace('\r', "\n");

    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                output.extend(std::iter::repeat_n(String::new(), i - run_start - 1));
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00A0}'
}

fn wrap_segment(chars: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in chars {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }
    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut line = String::new();
    let mut content_len = 0usize;
    let mut word_index = 0usize;
    let mut word_offset = 0usize;

    while word_index < words.len() {
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let suffix_len = words[word_index].len() - word_offset;

        if suffix_len <= capacity {
            let added_len = suffix_len + usize::from(content_len > 0);
            if content_len + added_len <= capacity {
                if content_len == 0 {
                    line = " ".repeat(indent);
                } else {
                    line.push(' ');
                }
                for &ch in &words[word_index][word_offset..] {
                    line.push(ch);
                }
                content_len += added_len;
                word_index += 1;
                word_offset = 0;
            } else {
                emit_line(&mut line, output);
                content_len = 0;
                first_line = false;
            }
            continue;
        }

        if content_len == 0 {
            line = " ".repeat(indent);
            for &ch in &words[word_index][word_offset..word_offset + capacity] {
                line.push(ch);
            }
            emit_line(&mut line, output);
            content_len = 0;
            first_line = false;
            word_offset += capacity;
            if word_offset == words[word_index].len() {
                word_index += 1;
                word_offset = 0;
            }
        } else {
            let available = capacity - content_len - 1;
            if available >= 2 {
                line.push(' ');
                for &ch in &words[word_index][word_offset..word_offset + available] {
                    line.push(ch);
                }
                emit_line(&mut line, output);
                content_len = 0;
                first_line = false;
                word_offset += available;
                if word_offset == words[word_index].len() {
                    word_index += 1;
                    word_offset = 0;
                }
            } else {
                emit_line(&mut line, output);
                content_len = 0;
                first_line = false;
            }
        }
    }

    if content_len > 0 {
        output.push(line);
    }
}

fn emit_line(line: &mut String, output: &mut Vec<String>) {
    output.push(std::mem::take(line));
}

fn main() {}
