#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }

    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn reduce(
    path: &str,
    start: Vec<String>,
    absolute: bool,
    offset_base: usize,
) -> Result<Vec<String>, PathError> {
    let mut components = start;
    let mut cursor = 0;

    while cursor < path.len() {
        let bytes = path.as_bytes();
        while cursor < bytes.len() && bytes[cursor] == b'/' {
            cursor += 1;
        }
        if cursor >= bytes.len() {
            break;
        }

        let component_start = cursor;
        while cursor < bytes.len() && bytes[cursor] != b'/' {
            cursor += 1;
        }
        let component = &path[component_start..cursor];

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|last| last != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: offset_base + component_start,
                    });
                } else {
                    components.push("..".to_owned());
                }
            }
            _ => components.push(component.to_owned()),
        }
    }

    Ok(components)
}

fn compose(components: &[String], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    } else if components.is_empty() {
        ".".to_owned()
    } else {
        components.join("/")
    };

    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let directory = directory_intent(path);
    let components = reduce(path, Vec::new(), absolute, 0)?;

    Ok(compose(&components, absolute, directory))
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;

    let home_components = match home {
        Some(home_path) => Some(
            normalized_context(home_path).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let directory = directory_intent(path);
    let bytes = path.as_bytes();

    let (remainder, start, offset_base) = if bytes[0] == b'/' {
        (path, Vec::new(), 0)
    } else if bytes[0] == b'~' {
        if bytes.len() == 1 || bytes[1] == b'/' {
            let start = home_components.ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            (&path[consumed..], start, consumed)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (path, cwd_components, 0)
    };

    let components = reduce(remainder, start, true, offset_base)?;
    Ok(compose(&components, true, directory))
}

fn main() {}
