#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, ch) in &chars {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let is_separator = |ch: char| {
        matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
    };
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_side = |j: usize| {
                j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
            };
            if i == 0 || !valid_side(i - 1) || !valid_side(i + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    #[derive(Clone)]
    struct Word {
        chars: Vec<char>,
        markers_after: Vec<bool>,
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i].1 == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 2;
            } else {
                i += 1;
            }
        } else {
            normalized.push(chars[i].1);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut pos = 0;

    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let start = pos;
            while pos < normalized.len() && normalized[pos] == '\n' {
                pos += 1;
            }
            if pos - start >= 2 {
                append_paragraph(&segment, options, &mut output, &is_separator);
                segment.clear();
                for _ in 1..(pos - start) {
                    output.push(String::new());
                }
                continue;
            }
            segment.push('\n');
        } else {
            segment.push(normalized[pos]);
            pos += 1;
        }
    }
    append_paragraph(&segment, options, &mut output, &is_separator);
    Ok(output)
}

#[derive(Clone)]
struct Line {
    text: String,
    pieces: usize,
    gap: usize,
    consumed: usize,
    hyphen: bool,
    raw: bool,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    hyphens: usize,
    lines: Vec<Line>,
}

fn append_paragraph(
    chars: &[char],
    options: &WrapOptions,
    output: &mut Vec<String>,
    is_separator: &impl Fn(char) -> bool,
) {
    let mut words = Vec::<(Vec<char>, Vec<usize>)>::new();
    let mut at = 0;
    while at < chars.len() {
        while at < chars.len() && is_separator(chars[at]) {
            at += 1;
        }
        if at == chars.len() {
            break;
        }

        let mut spelling = Vec::new();
        let mut markers = Vec::new();
        while at < chars.len() && !is_separator(chars[at]) {
            if chars[at] == '~' {
                markers.push(spelling.len());
            } else {
                spelling.push(chars[at]);
            }
            at += 1;
        }
        words.push((spelling, markers));
    }

    if words.is_empty() {
        return;
    }

    let mut memo: Vec<Vec<[Option<Solution>; 2]>> = words
        .iter()
        .map(|(w, _)| vec![[None, None]; w.len() + 1])
        .collect();

    fn solve(
        word_idx: usize,
        offset: usize,
        first: bool,
        words: &[(Vec<char>, Vec<usize>)],
        options: &WrapOptions,
        memo: &mut [Vec<[Option<Solution>; 2]>],
    ) -> Solution {
        if word_idx == words.len() {
            return Solution {
                cost: 0,
                hyphens: 0,
                lines: Vec::new(),
            };
        }
        let fi = usize::from(first);
        if let Some(cached) = &memo[word_idx][offset][fi] {
            return cached.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let (word, markers) = &words[word_idx];
        let remaining = word.len() - offset;
        let mut candidates = Vec::<(Line, usize, usize, bool)>::new();

        let mut used = 0usize;
        let mut wi = word_idx;
        let mut off = offset;
        let mut pieces = 0usize;

        while wi < words.len() {
            let (current, current_markers) = &words[wi];
            let start = if wi == word_idx { off } else { 0 };
            let available = current.len() - start;
            let separator_cost = usize::from(pieces > 0);

            if used + separator_cost + available <= capacity {
                used += separator_cost + available;
                pieces += 1;

                let consumed = used;
                candidates.push((
                    Line {
                        text: String::new(),
                        pieces,
                        gap: options.width - indent - used,
                        consumed,
                        hyphen: false,
                        raw: false,
                    },
                    wi + 1,
                    0,
                    false,
                ));

                for &marker_pos in current_markers {
                    if marker_pos <= start || marker_pos >= current.len() {
                        continue;
                    }
                    let prefix = marker_pos - start;
                    let display = used - available + prefix + 1;
                    if display <= capacity {
                        candidates.push((
                            Line {
                                text: String::new(),
                                pieces,
                                gap: options.width - indent - display,
                                consumed: display - 1,
                                hyphen: true,
                                raw: false,
                            },
                            wi,
                            marker_pos,
                            true,
                        ));
                    }
                }
                break;
            }

            let room = capacity.saturating_sub(used + separator_cost);
            let fitting_marker = current_markers
                .iter()
                .copied()
                .filter(|&m| m > start && m < current.len() && m - start + 1 <= room)
                .max();

            if let Some(marker_pos) = fitting_marker {
                let prefix = marker_pos - start;
                let display = used + separator_cost + prefix + 1;
                candidates.push((
                    Line {
                        text: String::new(),
                        pieces: pieces + 1,
                        gap: options.width - indent - display,
                        consumed: prefix,
                        hyphen: true,
                        raw: false,
                    },
                    wi,
                    marker_pos,
                    true,
                ));
            } else if pieces == 0 && available > capacity {
                candidates.push((
                    Line {
                        text: String::new(),
                        pieces: 1,
                        gap: options.width - indent - capacity,
                        consumed: capacity,
                        hyphen: false,
                        raw: true,
                    },
                    wi,
                    start + capacity,
                    false,
                ));
            }
            break;
        }

        let mut best: Option<Solution> = None;
        for (line, next_word, next_offset, ended_at_marker) in candidates {
            let final_line = next_word == words.len();
            let line_cost = (line.gap as u64) * (line.gap as u64)
                / if final_line { 4 } else { 1 };
            let extra = if ended_at_marker {
                options.hyphen_cost as u64
            } else if line.raw {
                (2 * options.width + 1) as u64
            } else {
                0
            } + if !final_line && line.pieces == 1 { 3 } else { 0 };

            let tail = solve(
                next_word,
                next_offset,
                false,
                words,
                options,
                memo,
            );
            let mut candidate = Solution {
                cost: line_cost + extra + tail.cost,
                hyphens: usize::from(ended_at_marker) + tail.hyphens,
                lines: {
                    let mut all = Vec::with_capacity(1 + tail.lines.len());
                    all.push(line.clone());
                    all.extend(tail.lines);
                    all
                },
            };

            if best
                .as_ref()
                .map(|b| better(&candidate, b))
                .unwrap_or(true)
            {
                best = Some(candidate);
            }
        }

        let best = best.unwrap_or(Solution {
            cost: u64::MAX,
            hyphens: usize::MAX,
            lines: Vec::new(),
        });
        memo[word_idx][offset][fi] = Some(best.clone());
        best
    }

    fn better(a: &Solution, b: &Solution) -> bool {
        if a.cost != b.cost {
            return a.cost < b.cost;
        }
        if a.hyphens != b.hyphens {
            return a.hyphens < b.hyphens;
        }
        if a.lines.len() != b.lines.len() {
            return a.lines.len() < b.lines.len();
        }
        for (x, y) in a.lines.iter().zip(&b.lines) {
            if x.consumed != y.consumed {
                return x.consumed > y.consumed;
            }
        }
        false
    }

    let mut solution = solve(0, 0, true, &words, options, &mut memo);

    for (line_no, line) in solution.lines.iter_mut().enumerate() {
        let indent = if line_no == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let mut rendered = " ".repeat(indent);
        let mut remaining_pieces = line.pieces;
        let mut wi = 0usize;
        let mut off = 0usize;
        let mut used = 0usize;
        let mut piece_texts = Vec::<String>::new();

        while remaining_pieces > 0 && wi < words.len() {
            let (word, _) = &words[wi];
            let start = if wi == 0 { off } else { 0 };
            let take = if line.hyphen && wi == solution_word_for_line(line, &words) {
                line.consumed
            } else {
                word.len() - start
            };
            if take == 0 {
                wi += 1;
                continue;
            }
            let text: String = word[start..start + take].iter().collect();
            piece_texts.push(text);
            used += take;
            remaining_pieces -= 1;
            wi += 1;
            off = 0;
        }

        if piece_texts.is_empty() {
            continue;
        }
        for (i, piece) in piece_texts.iter().enumerate() {
            if i > 0 {
                rendered.push(' ');
            }
            rendered.push_str(piece);
        }
        if line.hyphen {
            rendered.push('-');
        }

        if options.justify && line_no + 1 < solution.lines.len() && line.pieces >= 2 {
            let gaps = line.pieces - 1;
            let pre_width = rendered.chars().count();
            let extra = options.width.saturating_sub(pre_width);
            let each = extra / gaps;
            let remainder = extra % gaps;
            let start_gap = line_no % gaps;
            let mut justified = " ".repeat(indent);
            for (i, piece) in piece_texts.iter().enumerate() {
                if i > 0 {
                    let gap = i - 1;
                    justified.push_str(&" ".repeat(1 + each + usize::from((gap + gaps - start_gap) % gaps < remainder)));
                }
                justified.push_str(piece);
            }
            if line.hyphen {
                justified.push('-');
            }
            rendered = justified;
        }
        output.push(rendered);
    }

    fn solution_word_for_line(_line: &Line, _words: &[(Vec<char>, Vec<usize>)]) -> usize {
        0
    }
}

fn main() {}
