#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy)]
    enum State {
        StartField,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::StartField;
    let mut record_index = 0usize;
    let mut ended_with_terminator = false;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        match state {
            State::Quoted { .. } => {
                if b == b'"' {
                    if pos + 1 < bytes.len() && bytes[pos + 1] == b'"' {
                        field.push('"');
                        pos += 2;
                    } else {
                        state = State::AfterQuote;
                        pos += 1;
                    }
                } else {
                    let ch = input[pos..].chars().next().unwrap();
                    field.push(ch);
                    pos += ch.len_utf8();
                }
                ended_with_terminator = false;
            }
            State::StartField | State::Unquoted | State::AfterQuote => {
                if b == b'\r' && (pos + 1 >= bytes.len() || bytes[pos + 1] != b'\n') {
                    return Err(CsvError::BareCr {
                        record: record_index,
                        byte: pos,
                    });
                }

                match state {
                    State::AfterQuote => match b {
                        b',' => {
                            record.push(std::mem::take(&mut field));
                            state = State::StartField;
                            pos += 1;
                            ended_with_terminator = false;
                        }
                        b'\n' => {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            state = State::StartField;
                            pos += 1;
                            ended_with_terminator = true;
                        }
                        b'\r' => {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            state = State::StartField;
                            pos += 2;
                            ended_with_terminator = true;
                        }
                        _ => {
                            return Err(CsvError::TrailingByte {
                                record: record_index,
                                byte: pos,
                            });
                        }
                    },
                    State::StartField => match b {
                        b'"' => {
                            state = State::Quoted { opening: pos };
                            pos += 1;
                            ended_with_terminator = false;
                        }
                        b',' => {
                            record.push(std::mem::take(&mut field));
                            pos += 1;
                            ended_with_terminator = false;
                        }
                        b'\n' => {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            pos += 1;
                            ended_with_terminator = true;
                        }
                        b'\r' => {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            pos += 2;
                            ended_with_terminator = true;
                        }
                        _ => {
                            let ch = input[pos..].chars().next().unwrap();
                            field.push(ch);
                            state = State::Unquoted;
                            pos += ch.len_utf8();
                            ended_with_terminator = false;
                        }
                    },
                    State::Unquoted => match b {
                        b'"' => {
                            return Err(CsvError::UnexpectedQuote {
                                record: record_index,
                                byte: pos,
                            });
                        }
                        b',' => {
                            record.push(std::mem::take(&mut field));
                            state = State::StartField;
                            pos += 1;
                            ended_with_terminator = false;
                        }
                        b'\n' => {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            state = State::StartField;
                            pos += 1;
                            ended_with_terminator = true;
                        }
                        b'\r' => {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            state = State::StartField;
                            pos += 2;
                            ended_with_terminator = true;
                        }
                        _ => {
                            let ch = input[pos..].chars().next().unwrap();
                            field.push(ch);
                            pos += ch.len_utf8();
                            ended_with_terminator = false;
                        }
                    },
                    State::Quoted { .. } => unreachable!(),
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        }),
        State::StartField | State::Unquoted | State::AfterQuote => {
            if !ended_with_terminator {
                record.push(field);
                records.push(record);
            }
            Ok(records)
        }
    }
}

fn main() {}
