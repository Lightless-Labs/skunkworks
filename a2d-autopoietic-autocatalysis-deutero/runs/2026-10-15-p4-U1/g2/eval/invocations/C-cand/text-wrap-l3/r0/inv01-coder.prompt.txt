#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Scalar {
    ch: char,
    byte_index: usize,
}

#[derive(Default)]
struct Line {
    indent: usize,
    text: String,
    pieces: Vec<String>,
}

impl Line {
    fn content_len(&self) -> usize {
        self.text.chars().count()
    }

    fn is_empty(&self) -> bool {
        self.pieces.is_empty()
    }

    fn append_piece(&mut self, piece: &str) {
        if !self.pieces.is_empty() {
            self.text.push(' ');
        }
        self.text.push_str(piece);
        self.pieces.push(piece.to_string());
    }

    fn append_continuation(&mut self, piece: &str) {
        self.text.push_str(piece);
        self.pieces.push(piece.to_string());
    }
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn render_paragraph(mut lines: Vec<Line>, options: &WrapOptions, output: &mut Vec<String>) {
    let final_index = lines.len().saturating_sub(1);
    for (line_index, line) in lines.drain(..).enumerate() {
        let mut rendered = " ".repeat(line.indent);
        if options.justify
            && line_index != final_index
            && line.pieces.len() >= 2
        {
            let gaps = line.pieces.len() - 1;
            let pre_width = line.indent + line.content_len();
            let extra = options.width.saturating_sub(pre_width);
            let each = extra / gaps;
            let remainder = extra % gaps;

            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    rendered.push(' ');
                    rendered.push_str(&" ".repeat(each));
                    if (i - 1 + gaps - (line_index % gaps)) % gaps < remainder {
                        rendered.push(' ');
                    }
                }
                rendered.push_str(piece);
            }
        } else {
            rendered.push_str(&line.text);
        }
        output.push(rendered);
    }
}

fn layout_paragraph(words: &[Vec<char>], options: &WrapOptions, output: &mut Vec<String>) {
    if words.is_empty() {
        return;
    }

    let mut lines = Vec::<Line>::new();
    let mut first_line = true;
    let mut current = Line {
        indent: options.first_indent,
        ..Line::default()
    };

    let new_line = |first: bool| Line {
        indent: if first {
            options.first_indent
        } else {
            options.rest_indent
        },
        ..Line::default()
    };

    for word in words {
        let mut offset = 0;
        let mut continuation = false;

        while offset < word.len() {
            let capacity = options.width - current.indent;
            let current_len = current.content_len();
            let gap = if current.is_empty() { 0 } else { 1 };
            let room = capacity.saturating_sub(current_len + gap);
            let suffix_len = word.len() - offset;

            if suffix_len <= room {
                let piece: String = word[offset..].iter().collect();
                if continuation {
                    current.append_continuation(&piece);
                } else {
                    current.append_piece(&piece);
                }
                offset = word.len();
                continue;
            }

            let available = room;
            let mut usable = None;
            for marker in offset + 1..word.len() {
                if word[marker] == '~' {
                    let prefix_len = marker - offset;
                    if prefix_len + 1 <= available {
                        usable = Some(marker);
                    }
                }
            }

            let accepted = if let Some(marker) = usable {
                let prefix_len = marker - offset;
                let unused = available - (prefix_len + 1);
                if current.is_empty() {
                    Some(marker)
                } else {
                    let baseline = capacity.saturating_sub(current_len);
                    let lhs = unused.saturating_mul(unused).saturating_add(options.hyphen_cost);
                    let rhs = baseline.saturating_mul(baseline);
                    if lhs < rhs {
                        Some(marker)
                    } else {
                        None
                    }
                }
            } else {
                None
            };

            if let Some(marker) = accepted {
                let mut piece: String = word[offset..marker].iter().collect();
                piece.push('-');
                if continuation {
                    current.append_continuation(&piece);
                } else {
                    current.append_piece(&piece);
                }
                lines.push(current);
                first_line = false;
                current = new_line(first_line);
                offset = marker + 1;
                continuation = true;
                continue;
            }

            if !current.is_empty() {
                lines.push(current);
                first_line = false;
                current = new_line(first_line);
                continue;
            }

            if suffix_len > capacity {
                let end = offset + capacity;
                let piece: String = word[offset..end].iter().filter(|&&ch| ch != '~').collect();
                current.append_continuation(&piece);
                lines.push(current);
                first_line = false;
                current = new_line(first_line);
                offset = end;
                continuation = true;
            } else {
                let piece: String = word[offset..].iter().filter(|&&ch| ch != '~').collect();
                current.append_continuation(&piece);
                offset = word.len();
            }
        }
    }

    if !current.is_empty() {
        lines.push(current);
    }
    render_paragraph(lines, options, output);
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let scalars: Vec<Scalar> = input
        .char_indices()
        .map(|(byte_index, ch)| Scalar { ch, byte_index })
        .collect();

    for scalar in &scalars {
        if scalar.ch.is_control() && !matches!(scalar.ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl {
                index: scalar.byte_index,
            });
        }
    }

    for (i, scalar) in scalars.iter().enumerate() {
        if scalar.ch == '~'
            && (i == 0
                || i + 1 == scalars.len()
                || scalars[i - 1].ch == '~'
                || scalars[i + 1].ch == '~'
                || is_separator(scalars[i - 1].ch)
                || is_separator(scalars[i + 1].ch))
        {
            return Err(WrapError::InvalidMarker {
                index: scalar.byte_index,
            });
        }
    }

    let mut normalized = Vec::<char>::new();
    let mut i = 0;
    while i < scalars.len() {
        match scalars[i].ch {
            '\r' if i + 1 < scalars.len() && scalars[i + 1].ch == '\n' => {
                normalized.push('\n');
                i += 2;
            }
            '\r' | '\n' => {
                normalized.push('\n');
                i += 1;
            }
            ch => {
                normalized.push(ch);
                i += 1;
            }
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::<char>::new();
    let mut pos = 0;

    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let mut end = pos;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - pos >= 2 {
                let words = parse_words(&segment);
                layout_paragraph(&words, options, &mut output);
                segment.clear();
                for _ in 1..(end - pos) {
                    output.push(String::new());
                }
                pos = end;
                continue;
            }
        }
        segment.push(normalized[pos]);
        pos += 1;
    }

    let words = parse_words(&segment);
    layout_paragraph(&words, options, &mut output);
    Ok(output)
}

fn parse_words(segment: &[char]) -> Vec<Vec<char>> {
    let mut words = Vec::<Vec<char>>::new();
    let mut word = Vec::<char>::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else if ch != '~' {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }
    words
}
