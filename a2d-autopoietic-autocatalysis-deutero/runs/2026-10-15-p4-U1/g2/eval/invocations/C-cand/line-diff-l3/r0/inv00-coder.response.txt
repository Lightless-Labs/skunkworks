#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Debug, Clone)]
enum Row {
    Context(String),
    Remove(String),
    Add(String),
}

#[derive(Debug, Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty()
        || (s.len() > 1 && s.starts_with('0'))
        || !s.bytes().all(|b| b.is_ascii_digit())
    {
        return None;
    }
    let mut n = 0usize;
    for b in s.bytes() {
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, suffix) = rest.split_once(" @@")?;
    if !suffix.is_empty() {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if b.contains(',') || d.contains(',') {
        return None;
    }
    let a = parse_decimal(a)?;
    let b = parse_decimal(b)?;
    let c = parse_decimal(c)?;
    let d = parse_decimal(d)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn physical_lines(input: &str) -> Vec<&str> {
    input.split_terminator('\n').collect()
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|b| *b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for (idx, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(idx) != Some(expected) {
            return Err(ApplyError::BadHeader { line: idx + 1 });
        }
    }
    if lines.len() < 3 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_old_end = 0usize;
    let mut cumulative_delta: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        let mut has_change = false;

        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let text = lines[i];
            if text.contains('\r') || text.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &text[1..];
            if payload.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            match text.as_bytes()[0] {
                b' ' => {
                    rows.push(Row::Context(payload.to_owned()));
                    old_seen += 1;
                    new_seen += 1;
                }
                b'-' => {
                    rows.push(Row::Remove(payload.to_owned()));
                    old_seen += 1;
                    has_change = true;
                }
                b'+' => {
                    rows.push(Row::Add(payload.to_owned()));
                    new_seen += 1;
                    has_change = true;
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            }
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !has_change {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_end = old_start
            .checked_add(b)
            .ok_or(ApplyError::Misordered { line: header_line })?;
        let required_new = old_start as i128 + cumulative_delta;
        if old_start < previous_old_end
            || required_new < 0
            || new_start as i128 != required_new
        {
            return Err(ApplyError::Misordered { line: header_line });
        }

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        previous_old_end = old_end;
        cumulative_delta += d as i128 - b as i128;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|b| *b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source
        .strip_suffix('\n')
        .unwrap()
        .split('\n')
        .map(str::to_owned)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_distance(a: i128, b: i128) -> u128 {
    (a - b).unsigned_abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_idx, hunk) in hunks.iter().enumerate() {
        let lead = hunk
            .rows
            .iter()
            .take_while(|row| matches!(row, Row::Context(_)))
            .count();
        let trail = hunk
            .rows
            .iter()
            .rev()
            .take_while(|row| matches!(row, Row::Context(_)))
            .count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut best_starts = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > lead + trail {
                    continue;
                }

                let mut old_payloads = Vec::new();
                let mut new_payloads = Vec::new();
                let mut context_index = 0usize;
                for row in &hunk.rows {
                    match row {
                        Row::Context(payload) => {
                            let dropped = (context_index < l)
                                || (context_index >= hunk.rows.len().saturating_sub(trail)
                                    && context_index >= lead + hunk.rows.len().saturating_sub(trail));
                            context_index += 1;
                            if !dropped {
                                old_payloads.push(payload.as_str());
                                new_payloads.push(payload.as_str());
                            }
                        }
                        Row::Remove(payload) => old_payloads.push(payload.as_str()),
                        Row::Add(payload) => new_payloads.push(payload.as_str()),
                    }
                }

                let max_start = working.len().saturating_sub(hunk.old_count);
                if hunk.old_count > working.len() {
                    continue;
                }
                for s in 0..=max_start {
                    let distance = abs_distance(s as i128, expected);
                    if distance > options.max_offset as u128 {
                        continue;
                    }
                    let match_start = s + l;
                    let match_end = s + hunk.old_count - r;
                    if match_end > working.len() || match_end < match_start {
                        continue;
                    }
                    let candidate = &working[match_start..match_end];
                    if candidate.len() != old_payloads.len()
                        || !candidate
                            .iter()
                            .zip(old_payloads.iter())
                            .all(|(a, b)| a == b)
                    {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(existing) if rank < existing => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(existing) if rank == existing => {
                            best_starts.push((s, l, r));
                        }
                        _ => {}
                    }
                }
            }
        }

        let Some(_) = best_rank else {
            return Err(ApplyError::NoMatch {
                hunk: hunk_idx + 1,
            });
        };

        let mut starts: Vec<usize> = best_starts.iter().map(|entry| entry.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() >= 2 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_idx + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        let start = starts[0];
        let mut choices: Vec<(usize, usize)> = best_starts
            .iter()
            .filter(|entry| entry.0 == start)
            .map(|entry| (entry.1, entry.2))
            .collect();
        choices.sort_unstable();
        let (l, r) = choices[0];

        let mut replacement = Vec::new();
        let mut context_index = 0usize;
        let trailing_context_start = hunk.rows.len().saturating_sub(trail);
        for (idx, row) in hunk.rows.iter().enumerate() {
            match row {
                Row::Context(payload) => {
                    let is_dropped = context_index < l
                        || (idx >= trailing_context_start && context_index >= lead);
                    context_index += 1;
                    if !is_dropped {
                        replacement.push(payload.clone());
                    }
                }
                Row::Add(payload) => replacement.push(payload.clone()),
                Row::Remove(_) => {}
            }
        }

        let replace_start = start + l;
        let replace_end = start + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        drift = start as i128 - (hunk.old_start as i128 + net);
        net += hunk.new_count as i128 - hunk.old_count as i128;
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

fn main() {}
