fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_version(c: Constraint) -> Version {
    match c {
        Constraint::Exact(v)
        | Constraint::AtLeast(v)
        | Constraint::Before(v)
        | Constraint::Caret(v) => v,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| constraint_version(*a).cmp(&constraint_version(*b)))
}

fn constraint_text(c: Constraint) -> String {
    let (prefix, v) = match c {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{}{}", prefix, version_text(v))
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by(constraint_cmp);
    cs.dedup();
    cs
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
}

fn evidence_lines(first: String, evidence: &[Evidence]) -> String {
    let mut ev = evidence.to_vec();
    ev.sort_by(evidence_cmp);
    let mut out = first;
    for e in ev {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => evidence_lines(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            evidence_lines(
                format!("no-match|package={}|constraints={}", encode(package), cs),
                evidence,
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            evidence_lines(
                format!(
                    "conflict|package={}|selected={}|constraints={}",
                    encode(package),
                    version_text(*selected),
                    cs
                ),
                evidence,
            )
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Active {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    features: BTreeMap<String, BTreeSet<String>>,
    edges: Vec<(String, String, Constraint, String)>,
}

fn feature_list(fs: &[String]) -> String {
    let set: BTreeSet<&str> = fs.iter().map(String::as_str).collect();
    set.into_iter().collect::<Vec<_>>().join(",")
}

fn target_name(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn platform_active(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn build_active(
    catalog: &[Package],
    by_name: &BTreeMap<String, usize>,
    roots: &[Requirement],
    assignment: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut active = Active {
        constraints: BTreeMap::new(),
        origins: BTreeMap::new(),
        features: BTreeMap::new(),
        edges: Vec::new(),
    };
    let mut root_origins: Vec<(String, Constraint, String)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let token = format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            root.package,
            constraint_text(root.constraint),
            feature_list(&root.features)
        );
        active
            .constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        active
            .features
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        root_origins.push((root.package.clone(), root.constraint, token));
    }

    loop {
        let before_constraints = active.constraints.clone();
        let before_features = active.features.clone();
        let before_edges_len = active.edges.len();

        let names: Vec<String> = assignment.keys().cloned().collect();
        for source in names {
            let Some(&pi) = by_name.get(&source) else {
                continue;
            };
            let version = assignment[&source];
            let Some(release) = catalog[pi].releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = active.features.get(&source).cloned().unwrap_or_default();
            for feature in &release.features {
                if requested.contains(&feature.name) {
                    active
                        .features
                        .entry(source.clone())
                        .or_default()
                        .extend(feature.enables.iter().cloned());
                }
            }

            let current_features = active.features.get(&source).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !current_features.contains(g))
                {
                    continue;
                }
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    source,
                    version_text(version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    feature_list(&dep.features),
                    dep.gate.as_deref().unwrap_or("*"),
                    target_name(dep.platform)
                );
                active
                    .constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                active
                    .features
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                active.edges.push((
                    source.clone(),
                    dep.package.clone(),
                    dep.constraint,
                    token,
                ));
            }
        }

        let constraints_same = before_constraints == active.constraints;
        let features_same = before_features == active.features;
        // Edges are rebuilt on each pass, so retain only unique semantic edges.
        active.edges.sort();
        active.edges.dedup();
        if constraints_same && features_same && active.edges.len() == before_edges_len {
            break;
        }
        // The edge count comparison above can differ on the first pass; state is monotone.
        if constraints_same && features_same {
            break;
        }
    }

    let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
    for (pkg, _, token) in &root_origins {
        let candidate = vec![token.clone()];
        match paths.get(pkg) {
            None => {
                paths.insert(pkg.clone(), candidate);
            }
            Some(old) if path_cmp(&candidate, old).is_lt() => {
                paths.insert(pkg.clone(), candidate);
            }
            _ => {}
        }
    }

    loop {
        let mut changed = false;
        let known = paths.clone();
        for (source, dest, _, token) in &active.edges {
            let Some(src_path) = known.get(source) else {
                continue;
            };
            let mut candidate = src_path.clone();
            candidate.push(token.clone());
            match paths.get(dest) {
                None => {
                    paths.insert(dest.clone(), candidate);
                    changed = true;
                }
                Some(old) if path_cmp(&candidate, old).is_lt() => {
                    paths.insert(dest.clone(), candidate);
                    changed = true;
                }
                _ => {}
            }
        }
        if !changed {
            break;
        }
    }

    for (pkg, constraint, token) in root_origins {
        active
            .origins
            .entry(pkg)
            .or_default()
            .push(Origin {
                constraint,
                path: vec![token],
            });
    }
    for (source, dest, constraint, token) in &active.edges {
        if let Some(path) = paths.get(source) {
            let mut full = path.clone();
            full.push(token.clone());
            active
                .origins
                .entry(dest.clone())
                .or_default()
                .push(Origin {
                    constraint: *constraint,
                    path: full,
                });
        }
    }
    active
}

fn path_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

fn canonical_evidence(origins: &[Origin]) -> Vec<Evidence> {
    let mut out: Vec<Evidence> = origins
        .iter()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    out.sort_by(evidence_cmp);
    out
}

fn active_error(
    catalog: &[Package],
    by_name: &BTreeMap<String, usize>,
    assignment: &BTreeMap<String, Version>,
    active: &Active,
) -> Option<ResolveError> {
    for pkg in active.constraints.keys() {
        if !by_name.contains_key(pkg) {
            return Some(ResolveError::Missing {
                package: pkg.clone(),
                evidence: canonical_evidence(active.origins.get(pkg).map(Vec::as_slice).unwrap_or(&[])),
            });
        }
    }

    for (pkg, version) in assignment {
        let Some(cs) = active.constraints.get(pkg) else {
            continue;
        };
        if cs.iter().any(|c| !matches(*version, *c)) {
            return Some(ResolveError::Conflict {
                package: pkg.clone(),
                selected: *version,
                requirements: canonical_constraints(cs.clone()),
                evidence: canonical_evidence(active.origins.get(pkg).map(Vec::as_slice).unwrap_or(&[])),
            });
        }
    }

    for (pkg, cs) in &active.constraints {
        if assignment.contains_key(pkg) {
            continue;
        }
        let pi = by_name[pkg];
        let has_match = catalog[pi]
            .releases
            .iter()
            .any(|r| cs.iter().all(|c| matches(r.version, *c)));
        if !has_match {
            return Some(ResolveError::NoMatch {
                package: pkg.clone(),
                requirements: canonical_constraints(cs.clone()),
                evidence: canonical_evidence(active.origins.get(pkg).map(Vec::as_slice).unwrap_or(&[])),
            });
        }
    }

    for (pkg, version) in assignment {
        let Some(&pi) = by_name.get(pkg) else {
            continue;
        };
        let release = catalog[pi].releases.iter().find(|r| r.version == *version)?;
        if let Some(fs) = active.features.get(pkg) {
            for feature in fs {
                if !release.features.iter().any(|f| &f.name == feature) {
                    let path = canonical_path(active, pkg);
                    return Some(ResolveError::MissingFeature {
                        package: pkg.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path,
                    });
                }
            }
        }
    }
    None
}

fn canonical_path(active: &Active, pkg: &str) -> Vec<String> {
    let mut best: Option<Vec<String>> = None;
    for origin in active.origins.get(pkg).into_iter().flatten() {
        let candidate = origin.path.clone();
        if best.as_ref().is_none_or(|b| path_cmp(&candidate, b).is_lt()) {
            best = Some(candidate);
        }
    }
    best.unwrap_or_default()
}

struct Search<'a> {
    catalog: &'a [Package],
    by_name: &'a BTreeMap<String, usize>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best_cost: Option<usize>,
    best_vector: Vec<Option<Version>>,
    best_assignment: Option<BTreeMap<String, Version>>,
    catalog_names: Vec<String>,
}

impl Search<'_> {
    fn visit(&mut self, assignment: &mut BTreeMap<String, Version>) -> bool {
        let active = build_active(
            self.catalog,
            self.by_name,
            self.roots,
            assignment,
            self.target,
        );
        if let Some(error) = active_error(self.catalog, self.by_name, assignment, &active) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return false;
        }

        let next = active
            .constraints
            .keys()
            .find(|name| !assignment.contains_key(*name))
            .cloned();

        let Some(pkg) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|l| assignment.get(&l.package) != Some(&l.version))
                .count();
            let vector: Vec<Option<Version>> = self
                .catalog_names
                .iter()
                .map(|n| assignment.get(n).copied())
                .collect();
            let better = match self.best_cost {
                None => true,
                Some(old_cost) if cost < old_cost => true,
                Some(old_cost) if cost == old_cost => {
                    vector_lex_cmp(&vector, &self.best_vector).is_gt()
                }
                _ => false,
            };
            if better {
                self.best_cost = Some(cost);
                self.best_vector = vector;
                self.best_assignment = Some(assignment.clone());
            }
            return false;
        };

        let pi = self.by_name[&pkg];
        let cs = active.constraints.get(&pkg).cloned().unwrap_or_default();
        let mut candidates: Vec<Version> = self.catalog[pi]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| cs.iter().all(|c| matches(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == pkg) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return true;
            }
            self.attempts += 1;
            assignment.insert(pkg.clone(), version);
            let limit = self.visit(assignment);
            assignment.remove(&pkg);
            if limit {
                return true;
            }
        }
        false
    }
}

fn vector_lex_cmp(a: &[Option<Version>], b: &[Option<Version>]) -> std::cmp::Ordering {
    a.iter().zip(b).find_map(|(x, y)| {
        let c = x.cmp(y);
        if c.is_eq() { None } else { Some(c) }
    }).unwrap_or(std::cmp::Ordering::Equal)
}

fn validate_name(name: &str, site: String) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        validate_name(&root.package, format!("root[{}]", i))?;
        for (f, name) in root.features.iter().enumerate() {
            validate_name(name, format!("root[{}].feature[{}]", i, f))?;
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        validate_name(&package.name, format!("catalog[{}]", i))?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                validate_name(
                    &dep.package,
                    format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                )?;
                if let Some(gate) = &dep.gate {
                    validate_name(
                        gate,
                        format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                    )?;
                }
                for (f, name) in dep.features.iter().enumerate() {
                    validate_name(
                        name,
                        format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f),
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                validate_name(
                    &feature.name,
                    format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                )?;
                for (e, name) in feature.enables.iter().enumerate() {
                    validate_name(
                        name,
                        format!(
                            "catalog[{}].release[{}].feature[{}].enable[{}]",
                            i, j, f, e
                        ),
                    )?;
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        validate_name(&lock.package, format!("lock[{}]", i))?;
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let by_name: BTreeMap<String, usize> = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect();
    let mut catalog_names: Vec<String> = by_name.keys().cloned().collect();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        by_name: &by_name,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best_cost: None,
        best_vector: Vec::new(),
        best_assignment: None,
        catalog_names,
    };
    let mut assignment = BTreeMap::new();
    let limited = search.visit(&mut assignment);
    if limited {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some(final_assignment) = search.best_assignment else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_active = build_active(catalog, &by_name, roots, &final_assignment, target);
    let selected = final_assignment
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let lock_map: BTreeMap<&str, Version> = locks
        .iter()
        .map(|l| (l.package.as_str(), l.version))
        .collect();
    let mut all_names: BTreeSet<String> = lock_map.keys().map(|s| (*s).to_string()).collect();
    all_names.extend(final_assignment.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(package.as_str()).copied();
        let after = final_assignment.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let available = by_name
                    .get(&package)
                    .is_some_and(|pi| catalog[*pi].releases.iter().any(|r| r.version == old));
                if !available {
                    ChangeReason::Unavailable
                } else if final_active
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| cs.iter().any(|c| !matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

fn main() {}
