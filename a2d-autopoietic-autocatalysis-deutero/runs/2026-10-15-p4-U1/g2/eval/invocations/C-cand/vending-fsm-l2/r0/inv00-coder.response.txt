#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }

            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }

        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(_) = self.products.iter().find(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_string() });
        };

        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_string());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };

        let product = self
            .products
            .iter()
            .find(|product| product.id == *selected_id)
            .expect("selected products are always in the catalog");
        let price = product.price;
        let id = product.id.clone();

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for (index, denomination) in DENOMINATIONS.iter().enumerate() {
            let removed = change
                .iter()
                .filter(|coin| *coin == denomination)
                .count() as u16;
            proposed_tubes[index] -= removed;
        }

        for coin in &self.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            proposed_tubes[index] += 1;
        }

        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt {
            id,
            price,
            change,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(refunded)
    }
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];

    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index];
        let count = (remaining / denomination).min(tubes[index]);
        greedy_counts[index] = count;
        remaining -= count * denomination;
    }

    if remaining == 0 {
        return Some(counts_to_descending_coins(&greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    search_combinations(
        3,
        amount,
        tubes,
        &mut [0u16; 4],
        &mut best,
    );

    best.map(|counts| counts_to_descending_coins(&counts))
}

fn search_combinations(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index < 0 {
        if remaining == 0 && is_better_combination(current, best.as_ref()) {
            *best = Some(*current);
        }
        return;
    }

    let index = index as usize;
    let denomination = DENOMINATIONS[index];
    let max_count = (remaining / denomination).min(tubes[index]);

    for count in 0..=max_count {
        current[index] = count;
        let next_remaining = remaining - count * denomination;

        if let Some(best_counts) = best.as_ref() {
            let current_coins: u16 = current.iter().copied().sum();
            let remaining_denominations = index;
            let min_additional = if next_remaining == 0 {
                0
            } else if remaining_denominations == 0 {
                u16::MAX
            } else {
                (next_remaining + DENOMINATIONS[remaining_denominations - 1] - 1)
                    / DENOMINATIONS[remaining_denominations - 1]
            };
            let best_coins: u16 = best_counts.iter().copied().sum();
            if current_coins.saturating_add(min_additional) > best_coins {
                continue;
            }
        }

        search_combinations(
            index as isize - 1,
            next_remaining,
            tubes,
            current,
            best,
        );
    }

    current[index] = 0;
}

fn is_better_combination(candidate: &[u16; 4], best: Option<&[u16; 4]>) -> bool {
    let Some(best) = best else {
        return true;
    };

    let candidate_total: u16 = candidate.iter().copied().sum();
    let best_total: u16 = best.iter().copied().sum();

    if candidate_total != best_total {
        return candidate_total < best_total;
    }

    for index in (0..4).rev() {
        if candidate[index] != best[index] {
            return candidate[index] > best[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: &[u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

fn main() {}
