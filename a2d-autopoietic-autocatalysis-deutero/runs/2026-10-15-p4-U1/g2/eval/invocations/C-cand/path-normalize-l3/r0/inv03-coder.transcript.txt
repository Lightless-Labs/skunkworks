    links: &[Link],

    active: &mut Vec<usize>,

) -> Result<(), PathError> {

        .into_iter()

        .collect();

        match item {
            Work::EndTarget(index) => {
                if let Some(position) = active.iter().rposition(|&entry| entry == index) {
                    active.remove(position);
                }
            }

                "." => {}
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at });
                    }
                }
                _ => {
                    stack.push(component);
                    let key = absolute_key(stack);
                    if let Some(&index) = indices.get(&key) {
                        if active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if *expansions >= max_expansions {
                            return Err(PathError::Limit { index });
                        }
                        *expansions += 1;
                        active.push(index);

                        let target = &links[index].target;
                        let target_absolute = target.starts_with('/');
                        if target_absolute {
                            stack.clear();
                        } else {
                            stack.pop();
                        }

                        let remaining = std::mem::take(&mut work);
                        work.push(Work::EndTarget(index));
                        for (target_component, target_at) in

                            work.push(Work::Component(target_component, target_at));
                        }
                        work.extend(remaining);
                    }
                }
            },
        }
    }
    Ok(())
}

    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {

    let mut indices = std::collections::HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && link.path != "/"
            && !link.path.ends_with('/')
            && normalize(&link.path)

            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

        prepare_request(path, home_parts.as_deref())?;

    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        tilde_home
    } else {
        cwd_parts
    };

        items.extend(stack.iter().cloned().map(|component| (component, 0)));
    }

    let mut expansions = 0;

        items,
        &mut stack,
        links,
        &indices,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

}

        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home.ok_or(PathError::MissingHome)?.to_vec();
        let (remainder, offset) = if path.len() > 1 {
            (&path[2..], 2)
        } else {
            ("", path.len())
        };
        return Ok((false, intent, remainder, offset, home));
    }
    Ok((absolute, intent, path, 0, Vec::new()))
}

fn validate_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    let (absolute, intent, remainder, offset, tilde_home) =
        prepare_request(path, home_parts.as_deref())?;

    let mut parts = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        tilde_home
    } else {
        cwd_parts
    };

    for (component, at) in components(remainder, offset) {
        reduce_component(&mut parts, &component, true, at)?;
    }
    Ok(render_absolute(&parts, intent))
}

fn absolute_key(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

enum Work {
    Component(String, usize),
    EndTarget(usize),
}

fn process_components(
    items: Vec<(String, usize)>,
    stack: &mut Vec<String>,
    links: &[Link],
    indices: &std::collections::HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    let mut work: Vec<Work> = items
        .into_iter()
        .rev()
        .map(|(component, at)| Work::Component(component, at))
        .collect();

    while let Some(item) = work.pop() {
        match item {
            Work::EndTarget(index) => {
                if let Some(position) = active.iter().rposition(|&entry| entry == index) {
                    active.remove(position);
                }
            }
            Work::Component(component, at) => match component.as_str() {
                "." => {}
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at });
                    }
                }
                _ => {
                    stack.push(component);
                    let key = absolute_key(stack);
                    if let Some(&index) = indices.get(&key) {
                        if active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if *expansions >= max_expansions {
                            return Err(PathError::Limit { index });
                        }
                        *expansions += 1;
                        active.push(index);

                        let target = &links[index].target;
                        let target_absolute = target.starts_with('/');
                        if target_absolute {
                            stack.clear();
                        } else {
                            stack.pop();
                        }

                        let remaining = std::mem::take(&mut work);
                        work.push(Work::EndTarget(index));
                        for (target_component, target_at) in
                            components(target, 0).into_iter().rev()
                        {
                            work.push(Work::Component(target_component, target_at));
                        }
                        work.extend(remaining);
                    }
                }
            },
        }
    }
    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;

    let mut indices = std::collections::HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && link.path != "/"
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .map(|normalized| normalized == link.path)
                .unwrap_or(false);
        let target_valid = validate_raw(&link.target).is_ok();

        if !path_valid || !target_valid {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

    let (absolute, intent, remainder, offset, tilde_home) =
        prepare_request(path, home_parts.as_deref())?;

    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        tilde_home
    } else {
        cwd_parts
    };

    let mut items = Vec::new();
    items.extend(components(remainder, offset));

    let mut active = Vec::new();
    let mut expansions = 0;
    process_components(
        items,
        &mut stack,
        links,
        &indices,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

    Ok(render_absolute(&stack, intent))
}
