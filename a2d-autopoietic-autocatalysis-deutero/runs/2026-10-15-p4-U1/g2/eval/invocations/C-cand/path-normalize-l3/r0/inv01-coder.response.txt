#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.as_bytes().last() == Some(&b'/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str, base_offset: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_owned(), base_offset + start));
    }

    result
}

fn render_absolute(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return "/".to_owned();
    }

    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn render_normalized(parts: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return render_absolute(parts, intent);
    }
    if parts.is_empty() {
        return ".".to_owned();
    }

    let mut result = parts.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce_component(
    parts: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if parts.last().is_some_and(|part| part != "..") {
                parts.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                parts.push("..".to_owned());
            }
        }
        _ => parts.push(component.to_owned()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut parts = Vec::new();

    for (component, at) in components(path, 0) {
        reduce_component(&mut parts, &component, absolute, at)?;
    }

    Ok(render_normalized(&parts, absolute, intent))
}

fn validated_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(components(&normalized, 0)
        .into_iter()
        .map(|(component, _)| component)
        .collect())
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, bool, &'a str, usize, Vec<String>), PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);

    if path.as_bytes()[0] == b'~' {
        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home.ok_or(PathError::MissingHome)?.to_vec();
        let (remainder, offset) = if path.len() > 1 {
            (&path[2..], 2)
        } else {
            ("", path.len())
        };
        return Ok((false, intent, remainder, offset, home));
    }

    Ok((absolute, intent, path, 0, Vec::new()))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    let (absolute, intent, remainder, offset, tilde_home) =
        prepare_request(path, home_parts.as_deref())?;

    let mut parts = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        tilde_home
    } else {
        cwd_parts
    };

    for (component, at) in components(remainder, offset) {
        reduce_component(&mut parts, &component, true, at)?;
    }

    Ok(render_absolute(&parts, intent))
}

fn absolute_key(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn process_link_components(
    items: Vec<(String, usize)>,
    stack: &mut Vec<String>,
    links: &[Link],
    indices: &std::collections::HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    enum Work {
        Component(String, usize),
        EndTarget(usize),
    }

    let mut work: Vec<Work> = items
        .into_iter()
        .rev()
        .map(|(component, at)| Work::Component(component, at))
        .collect();

    while let Some(item) = work.pop() {
        match item {
            Work::EndTarget(index) => {
                if let Some(position) = active.iter().rposition(|&active_index| active_index == index) {
                    active.remove(position);
                }
            }
            Work::Component(component, at) => {
                match component.as_str() {
                    "." => continue,
                    ".." => {
                        if stack.pop().is_none() {
                            return Err(PathError::Escape { at });
                        }
                        continue;
                    }
                    _ => stack.push(component),
                }

                let key = absolute_key(stack);
                let Some(&index) = indices.get(&key) else {
                    continue;
                };

                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if *expansions >= max_expansions {
                    return Err(PathError::Limit { index });
                }
                *expansions += 1;
                active.push(index);

                let target = &links[index].target;
                let remaining = std::mem::take(&mut work);
                let target_absolute = target.starts_with('/');
                if target_absolute {
                    stack.clear();
                } else {
                    stack.pop();
                }

                work.push(Work::EndTarget(index));
                for (target_component, target_at) in components(target, 0).into_iter().rev() {
                    work.push(Work::Component(target_component, target_at));
                }
                work.extend(remaining);
            }
        }
    }

    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    let mut indices = std::collections::HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && link.path != "/"
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .map(|normalized| normalized == link.path)
                .unwrap_or(false);
        let target_valid = validate_raw(&link.target).is_ok();

        if !path_valid || !target_valid {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

    let (absolute, intent, remainder, offset, tilde_home) =
        prepare_request(path, home_parts.as_deref())?;

    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        tilde_home
    } else {
        cwd_parts
    };

    let mut items = Vec::new();
    if !absolute {
        let starting = if path.starts_with('~') {
            &stack
        } else {
            &stack
        };
        let mut context_items: Vec<(String, usize)> = starting
            .iter()
            .map(|component| (component.clone(), 0))
            .collect();
        context_items.extend(components(remainder, offset));
        items = context_items;
    } else {
        items = components(remainder, offset);
    }

    let mut active = Vec::new();
    let mut expansions = 0;
    process_link_components(
        items,
        &mut stack,
        links,
        &indices,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

    Ok(render_absolute(&stack, intent))
}

fn main() {}
