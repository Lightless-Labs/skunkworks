fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(match bytes.get(*pos) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: *pos },
            _ => VersionError::InvalidChar { at: *pos },
        });
    }

        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]

        .map_err(|_| VersionError::Overflow { at: start })
}

    end: usize,

        let start = i;
        while i < b.len() && !whitespace(b[i]) {
            i += 1;
        }
        out.push((start, &s[start..i]));
    }
    out
}

#[derive(Clone, Debug)]
struct Operand {
    lower: Version,
    full: bool,
    numeric: usize,
}

fn parse_full_operand(s: &str, offset: usize) -> Result<Operand, RangeError> {
    let v = parse_version(s).map_err(|error| RangeError::Version { at: offset, error })?;
    Ok(Operand {
        lower: v,
        full: true,
        numeric: 3,
    })
}

fn parse_partial(s: &str, offset: usize) -> Result<Operand, RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let has_wildcard = fields
        .iter()
        .any(|f| matches!(*f, "*" | "x" | "X"));

    if s.contains('-') || s.contains('+') || (fields.len() >= 3 && !has_wildcard) {
        return parse_full_operand(s, offset);
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: offset });
    }

    let mut nums = Vec::new();
    let mut seen_wild = false;
    let mut numeric = 0;

    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at: offset });
        }
        if matches!(f, "*" | "x" | "X") {
            seen_wild = true;
        } else {
            if seen_wild
                || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: offset });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: offset })?;
            nums.push(n);
            numeric += 1;
        }
    }

    let (major, minor, patch) = match nums.as_slice() {
        [] => (0, 0, 0),
        [a] => (*a, 0, 0),
        [a, b] => (*a, *b, 0),
        [a, b, c] => (*a, *b, *c),
        _ => return Err(RangeError::Syntax { at: offset }),
    };

    Ok(Operand {
        lower: Version {
            major,
            minor,
            patch,
            pre: Vec::new(),
            build: Vec::new(),
        },
        full: false,
        numeric,
    })
}

fn next_tuple(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (major, minor, patch) = match field {
        0 => (
            v.major.checked_add(1).ok_or(RangeError::Syntax { at })?,
            0,
            0,
        ),
        1 => (
            v.major,
            v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?,
            0,
        ),
        _ => (
            v.major,
            v.minor,
            v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?,
        ),
    };
    Ok(Version {
        major,
        minor,
        patch,
        pre: vec!["0".into()],
        build: Vec::new(),
    })
}

fn add_test(arm: &mut Arm, op: Op, version: Version) {
    if !version.pre.is_empty() {
        arm.anchors
            .push((version.major, version.minor, version.patch));
    }
    arm.tests.push(Test { op, version });
}

fn parse_token_operand(token: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if token.starts_with('^') || token.starts_with('~') {
        let caret = token.starts_with('^');
        let operand_start = start + 1;
        let operand = parse_partial(&token[1..], operand_start)?;
        if operand.numeric == 0 {
            return Err(RangeError::Syntax { at: start });
        }

        add_test(arm, Op::Ge, operand.lower.clone());
        let v = &operand.lower;
        let field = if caret {
            if v.major > 0 || operand.numeric == 1 {
                0
            } else if v.minor > 0 || operand.numeric == 2 {
                1
            } else {
                2
            }
        } else if operand.numeric == 1 {
            0
        } else {
            1
        };
        add_test(arm, Op::Lt, next_tuple(v, field, operand_start)?);
        return Ok(());
    }

    let (op, body, body_start) = if token.starts_with(">=") {
        (Some(Op::Ge), &token[2..], start + 2)
    } else if token.starts_with("<=") {
        (Some(Op::Le), &token[2..], start + 2)
    } else if token.starts_with('>') {
        (Some(Op::Gt), &token[1..], start + 1)
    } else if token.starts_with('<') {
        (Some(Op::Lt), &token[1..], start + 1)
    } else if token.starts_with('=') {
        (Some(Op::Eq), &token[1..], start + 1)
    } else {
        (None, token, start)
    };

    if let Some(op) = op {
        if body.is_empty() || !body.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }
        let operand = parse_full_operand(body, body_start)?;
        add_test(arm, op, operand.lower);
        return Ok(());
    }

    let operand = parse_partial(body, body_start)?;
    if operand.full {
        add_test(arm, Op::Eq, operand.lower);
    } else if operand.numeric == 1 {
        add_test(arm, Op::Ge, operand.lower.clone());
        add_test(arm, Op::Lt, next_tuple(&operand.lower, 0, body_start)?);
    } else if operand.numeric == 2 {
        add_test(arm, Op::Ge, operand.lower.clone());
        add_test(arm, Op::Lt, next_tuple(&operand.lower, 1, body_start)?);
    }
    Ok(())
}

fn compare_test(v: &Version, t: &Test) -> bool {
    let c = compare_versions(v, &t.version);
    match t.op {
        Op::Eq => c == Ordering::Equal,
        Op::Lt => c == Ordering::Less,
        Op::Le => c != Ordering::Greater,
        Op::Gt => c == Ordering::Greater,
        Op::Ge => c != Ordering::Less,
    }
}

fn split_arms(s: &str) -> Result<Vec<(usize, &str)>, RangeError> {
    let ts = tokens(s);
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut arm_start: Option<usize> = None;
    let mut saw_separator = false;

    for &(at, token) in &ts {
        if token == "||" {
            if arm_start.is_none() {
                return Err(RangeError::Empty { at });
            }
            arms.push((arm_start.take().unwrap(), &s[arm_start.unwrap_or(0)..at]));
            saw_separator = true;
        } else if arm_start.is_none() {
            arm_start = Some(at);
            saw_separator = false;
        }
    }

    if arm_start.is_none() {
        if saw_separator {
            return Err(RangeError::Empty { at: s.len() });
        }
        return Err(RangeError::Empty { at: 0 });
    }

    let start = arm_start.unwrap();
    arms.push((start, &s[start..]));
    Ok(arms)
}

fn parse_arm(text: &str, base: usize) -> Result<Arm, RangeError> {
    let ts = tokens(text);
    let mut arm = Arm {
        tests: Vec::new(),
        anchors: Vec::new(),
    };
    if ts.is_empty() {
        return Err(RangeError::Empty { at: base });
    }

    if let Some((hyphen_at, _)) = ts.iter().find(|(_, t)| *t == "-") {
        if ts.len() != 3 || ts[1].1 != "-" {
            return Err(RangeError::Syntax {
                at: base + *hyphen_at,
            });
        }

        let (lo_off, lo_s) = ts[0];
        let (hi_off, hi_s) = ts[2];
        let lo_at = base + lo_off;
        let hi_at = base + hi_off;
        let lo = parse_partial(lo_s, lo_at)?;
        let hi = parse_partial(hi_s, hi_at)?;
        if lo.numeric == 0 {
            return Err(RangeError::Syntax { at: lo_at });
        }
        if hi.numeric == 0 {
            return Err(RangeError::Syntax { at: hi_at });
        }

        add_test(&mut arm, Op::Ge, lo.lower);
        if hi.full {
            add_test(&mut arm, Op::Le, hi.lower);
        } else {
            let field = if hi.numeric == 1 { 0 } else { 1 };
            add_test(&mut arm, Op::Lt, next_tuple(&hi.lower, field, hi_at)?);
        }
        return Ok(arm);
    }

    for (at, token) in ts {
        let absolute = base + at;
        let first = token.as_bytes()[0];
        if !first.is_ascii_digit()
            && !matches!(first, b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'=')
        {
            return Err(RangeError::Syntax { at: absolute });
        }
        parse_token_operand(token, absolute, &mut arm)?;
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let pieces = split_arms(s)?;
    let mut arms = Vec::new();
    for (base, text) in pieces {
        arms.push(parse_arm(text, base)?);
    }
    Ok(Range { arms })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| {
        arm.tests.iter().all(|t| compare_test(v, t))
            && (include_prerelease
                || v.pre.is_empty()
                || arm
                    .anchors
                    .contains(&(v.major, v.minor, v.patch)))
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i, s) in candidates.iter().enumerate() {
        parsed.push(
            parse_version(s).map_err(|error| CandidateError { index: i, error })?,
        );
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let cmp = compare_versions(v, &parsed[j]);
            if (greatest && cmp == Ordering::Greater)
                || (!greatest && cmp == Ordering::Less)
            {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn version_string(v: &Version, include_build: bool) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    if include_build && !v.build.is_empty() {
        s.push('+');
        s.push_str(&v.build.join("."));
    }
    s
}

fn stronger_lower(
    old: Option<(Version, bool)>,
    new: (Version, bool),
) -> Option<(Version, bool)> {
    match old {
        None => Some(new),
        Some(o) => {
            let c = compare_versions(&new.0, &o.0);
            if c == Ordering::Greater || (c == Ordering::Equal && !new.1 && o.1) {
                Some(new)
            } else {
                Some(o)
            }
        }
    }
}

fn stronger_upper(
    old: Option<(Version, bool)>,
    new: (Version, bool),
) -> Option<(Version, bool)> {
    match old {
        None => Some(new),
        Some(o) => {
            let c = compare_versions(&new.0, &o.0);
            if c == Ordering::Less || (c == Ordering::Equal && !new.1 && o.1) {
                Some(new)
            } else {
                Some(o)
            }
        }
    }
}

fn arm_bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lower = None;
    let mut upper = None;
    for t in &arm.tests {
        match t.op {
            Op::Eq => {
                lower = stronger_lower(lower, (t.version.clone(), true));
                upper = stronger_upper(upper, (t.version.clone(), true));
            }
            Op::Gt | Op::Ge => {
                lower = stronger_lower(lower, (t.version.clone(), t.op == Op::Ge));
            }
            Op::Lt | Op::Le => {
                upper = stronger_upper(upper, (t.version.clone(), t.op == Op::Le));
            }
        }
    }
    (lower, upper)
}

fn cmp_core(a: (u64, u64, u64), b: (u64, u64, u64)) -> Ordering {
    a.cmp(&b)
}

fn tuple_version(t: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: t.0,
        minor: t.1,
        patch: t.2,
        pre,
        build: Vec::new(),
    }
}

fn pre_zero(t: (u64, u64, u64)) -> Version {
    tuple_version(t, vec!["0".to_string()])
}

fn stable(t: (u64, u64, u64)) -> Version {
    tuple_version(t, Vec::new())
}

fn next_core(t: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if t.2 < u64::MAX {
        Some((t.0, t.1, t.2 + 1))
    } else if t.1 < u64::MAX {
        Some((t.0, t.1 + 1, 0))
    } else if t.0 < u64::MAX {
        Some((t.0 + 1, 0, 0))
    } else {
        None
    }
}

fn least_accepted_in_interval(
    lower: &Option<(Version, bool)>,
    upper: &Option<(Version, bool)>,
    allow_stable: bool,
    anchors: &[(u64, u64, u64)],
) -> bool {
    let cmp_bounds = match (lower, upper) {
        (Some((l, li)), Some((u, ui))) => {
            let c = compare_versions(l, u);
            if c == Ordering::Greater || (c == Ordering::Equal && !(*li && *ui)) {
                return false;
            }
            Some(c)
        }
        _ => None,
    };

    if let (Some((l, li)), Some((u, ui))) = (lower, upper) {
        if cmp_bounds == Some(Ordering::Equal) {
            return *li && *ui && l.pre.is_empty() && allow_stable;
        }
    }

    let candidate = match lower {
        None => stable((0, 0, 0)),
        Some((l, inclusive)) if *inclusive => l.clone(),
        Some((l, false)) if l.pre.is_empty() => match next_core((l.major, l.minor, l.patch)) {
            Some(t) => stable(t),
            None => return false,
        },
        Some((l, false)) => {
            let mut p = l.pre.clone();
            p.push("0".to_string());
            tuple_version((l.major, l.minor, l.patch), p)
        }
    };

    if !candidate.pre.is_empty() {
        if !allow_stable
            && !anchors.contains(&(candidate.major, candidate.minor, candidate.patch))
        {
            if let Some((u, ui)) = upper {
                let uc = compare_versions(&candidate, u);
                if uc == Ordering::Greater || (uc == Ordering::Equal && !*ui) {
                    return false;
                }
            }

            for anchor in anchors {
                let lo_cmp = compare_versions(&candidate, &pre_zero(*anchor));
                let hi_cmp = compare_versions(&candidate, &stable(*anchor));
                if lo_cmp != Ordering::Greater || hi_cmp == Ordering::Greater {
                    continue;
                }

                let witness = match lower {
                    Some((l, _)) if (l.major, l.minor, l.patch) == *anchor => {
                        let c = compare_versions(l, &pre_zero(*anchor));
                        if c == Ordering::Greater {
                            l.clone()
                        } else {
                            pre_zero(*anchor)
                        }
                    }
                    _ => pre_zero(*anchor),
                };
                if compare_versions(&witness, &candidate) != Ordering::Less {
                    if let Some((u, ui)) = upper {
                        let uc = compare_versions(&witness, u);
                        if uc == Ordering::Greater || (uc == Ordering::Equal && !*ui) {
                            continue;
                        }
                    }
                    return true;
                }
            }
            return false;
        }
    }

    if let Some((u, ui)) = upper {
        let c = compare_versions(&candidate, u);
        if c == Ordering::Greater || (c == Ordering::Equal && !*ui) {
            return false;
        }
    }

    candidate.pre.is_empty()
        || allow_stable
        || anchors.contains(&(candidate.major, candidate.minor, candidate.patch))
}

fn arm_has_witness(arm: &Arm) -> bool {
    let (lower, upper) = arm_bounds(arm);
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    least_accepted_in_interval(&lower, &upper, true, &anchors)
        || least_accepted_in_interval(&lower, &upper, false, &anchors)
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();

    for arm in &r.arms {
        let (lower, upper) = arm_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();

        if !least_accepted_in_interval(&lower, &upper, true, &anchors)
            && !least_accepted_in_interval(&lower, &upper, false, &anchors)
        {
            continue;
        }

        let low = lower
            .map(|(v, inclusive)| {
                format!(
                    "{}{}",
                    if inclusive { ">=" } else { ">" },
                    version_string(&v, false)
                )
            })
            .unwrap_or_else(|| "*".to_string());
        let high = upper
            .map(|(v, inclusive)| {
                format!(
                    "{}{}",
                    if inclusive { "<=" } else { "<" },
                    version_string(&v, false)
                )
            })
            .unwrap_or_else(|| "*".to_string());
        let pre = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(a, b, c)| format!("{a}.{b}.{c}"))
                .collect::<Vec<_>>()
                .join(",")
        };

        strings.push(format!("[{low};{high};pre={pre}]"));
    }

    strings.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".to_string()
    } else {
        strings.join(" || ")
    }
}

fn arm_intersects(a: &Arm, b: &Arm, include_prerelease: bool) -> bool {
    let (al, au) = arm_bounds(a);
    let (bl, bu) = arm_bounds(b);

    let lower = match (al, bl) {
        (None, x) | (x, None) => x,
        (Some(x), Some(y)) => stronger_lower(Some(x), y),
    };
    let upper = match (au, bu) {
        (None, x) | (x, None) => x,
        (Some(x), Some(y)) => stronger_upper(Some(x), y),
    };

    let mut anchors = a.anchors.clone();
    anchors.extend_from_slice(&b.anchors);
    anchors.sort();
    anchors.dedup();

    if include_prerelease {
        least_accepted_in_interval(&lower, &upper, true, &anchors)
    } else {
        let common: Vec<_> = a
            .anchors
            .iter()
            .copied()
            .filter(|t| b.anchors.contains(t))
            .collect();
        least_accepted_in_interval(&lower, &upper, true, &[])
            || least_accepted_in_interval(&lower, &upper, false, &common)
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    a.arms.iter().any(|x| {
        b.arms
            .iter()
            .any(|y| arm_intersects(x, y, include_prerelease))
    })
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(
    expr: &SetExpr,
    v: &Version,
    include: bool,
    parsed_ranges: &[Vec<Range>],
    cursor: &mut usize,
) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let result = matches_with(v, &parsed_ranges[*cursor], include);
            *cursor += 1;
            result
        }
        SetExpr::Union(xs) => xs
            .iter()
            .any(|x| eval_expr(x, v, include, parsed_ranges, cursor)),
        SetExpr::Intersection(xs) => xs
            .iter()
            .all(|x| eval_expr(x, v, include, parsed_ranges, cursor)),
    }
}

fn collect_ranges(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    out: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            out.push(r);
            Ok(())
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                collect_ranges(x, package, label, path, out)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn expression_matches(
    expr: &SetExpr,
    v: &Version,
    include: bool,
    ranges: &[Range],
) -> bool {
    fn go(expr: &SetExpr, v: &Version, include: bool, ranges: &[Range], cursor: &mut usize) -> bool {
        match expr {
            SetExpr::Range(_) => {
                let result = matches_with(v, &ranges[*cursor], include);
                *cursor += 1;
                result
            }
            SetExpr::Union(xs) => xs.iter().any(|x| go(x, v, include, ranges, cursor)),
            SetExpr::Intersection(xs) => {
                xs.iter().all(|x| go(x, v, include, ranges, cursor))
            }
        }
    }
    go(expr, v, include, ranges, &mut 0)
}

fn conflict_labels(
    p: &Package,
    versions: &[Version],
    include: bool,
    parsed: &[Vec<Range>],
) -> Vec<String> {
    if versions.is_empty() || p.constraints.is_empty() {
        return Vec::new();
    }

    fn search(
        p: &Package,
        versions: &[Version],
        include: bool,
        parsed: &[Vec<Range>],
        size: usize,
        start: usize,
        choice: &mut Vec<usize>,
    ) -> Option<Vec<String>> {
        if size == 0 {
            let satisfiable = versions.iter().any(|v| {
                choice.iter().all(|&i| {
                    expression_matches(
                        &p.constraints[i].expr,
                        v,
                        include,
                        &parsed[i],
                    )
                })
            });
            if !satisfiable {
                let mut labels: Vec<String> = choice
                    .iter()
                    .map(|&i| p.constraints[i].label.clone())
                    .collect();

            );

                package: p.name.clone(),
                labels,
            });
        }

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    }

pub fn parse_version(s: &str) -> Result<Version, VersionError> {

        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();

    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(match bytes.get(pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(match bytes.get(pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();

            b'-' => {

    match (numeric_id(a), numeric_id(b)) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {

    if core != Ordering::Equal {

fn parse_partial(s: &str, at: usize) -> Result<Operand, RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let has_wildcard = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));

    if s.contains('-') || s.contains('+') || (fields.len() >= 3 && !has_wildcard) {
        return parse_full_operand(s, at);
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }

    let mut values = Vec::new();
    let mut seen_wildcard = false;
    for field in &fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(*field, "*" | "x" | "X") {
            seen_wildcard = true;
        } else {
            if seen_wildcard
                || !field.bytes().all(is_digit)
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            values.push(
                field
                    .parse::<u64>()
                    .map_err(|_| RangeError::Syntax { at })?,
            );
        }
    }

    let numeric = values.len();
    let (major, minor, patch) = match values.as_slice() {
        [] => (0, 0, 0),
        [a] => (*a, 0, 0),
        [a, b] => (*a, *b, 0),
        [a, b, c] => (*a, *b, *c),
        _ => return Err(RangeError::Syntax { at }),
    };

    Ok(Operand {
        version: Version {
            major,
            minor,
            patch,
            pre: Vec::new(),
            build: Vec::new(),
        },
        full: false,
        numeric,
    })
}

fn parse_token(token: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if token.starts_with('^') || token.starts_with('~') {
        let caret = token.starts_with('^');
        let operand_at = start + 1;
        let operand = parse_partial(&token[1..], operand_at)?;
        if operand.numeric == 0 {
            return Err(RangeError::Syntax { at: start });
        }
        add_test(arm, Op::Ge, operand.version.clone());

        let v = &operand.version;
        let field = if caret {
            if v.major > 0 || operand.numeric == 1 {
                0
            } else if v.minor > 0 || operand.numeric == 2 {
                1
            } else {
                2
            }
        } else if operand.numeric == 1 {
            0
        } else {
            1
        };
        add_test(arm, Op::Lt, increment(v, field, operand_at)?);
        return Ok(());
    }

    let (op, body, body_at) = if token.starts_with(">=") {
        (Some(Op::Ge), &token[2..], start + 2)
    } else if token.starts_with("<=") {
        (Some(Op::Le), &token[2..], start + 2)
    } else if token.starts_with('>') {
        (Some(Op::Gt), &token[1..], start + 1)
    } else if token.starts_with('<') {
        (Some(Op::Lt), &token[1..], start + 1)
    } else if token.starts_with('=') {
        (Some(Op::Eq), &token[1..], start + 1)
    } else {
        (None, token, start)
    };

    if let Some(op) = op {
        if body.is_empty() || !body.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }
        let operand = parse_full_operand(body, body_at)?;
        add_test(arm, op, operand.version);
        return Ok(());
    }

    let operand = parse_partial(body, body_at)?;
    if operand.full {
        add_test(arm, Op::Eq, operand.version);
    } else if operand.numeric == 1 {
        add_test(arm, Op::Ge, operand.version.clone());
        add_test(arm, Op::Lt, increment(&operand.version, 0, body_at)?);
    } else if operand.numeric == 2 {
        add_test(arm, Op::Ge, operand.version.clone());
        add_test(arm, Op::Lt, increment(&operand.version, 1, body_at)?);
    }
    Ok(())
}

            add_test(&mut arm, Op::Le, hi.version);

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let ts = tokens(s);
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut arm_start: Option<usize> = None;
    let mut arm_tokens = Vec::new();

    for (at, token) in ts {
        if token == "||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at });
            }
            let start = arm_start.unwrap();
            let end = at;
            arms.push(parse_arm(&s[start..end], start)?);
            arm_start = None;
            arm_tokens.clear();
        } else {
            if arm_start.is_none() {
                arm_start = Some(at);
            }
            arm_tokens.push((at, token));
        }
    }

    if arm_tokens.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let start = arm_start.unwrap();
    arms.push(parse_arm(&s[start..], start)?);
    Ok(Range { arms })
}

fn test_matches(v: &Version, test: &Test) -> bool {
    match (compare_versions(v, &test.version), test.op) {
        (Ordering::Equal, Op::Eq) => true,
        (Ordering::Less, Op::Lt | Op::Le | Op::Ge) => test.op != Op::Ge,
        (Ordering::Greater, Op::Gt | Op::Ge | Op::Le) => test.op != Op::Le,
        (Ordering::Equal, Op::Lt | Op::Le | Op::Gt | Op::Ge) => {
            matches!(test.op, Op::Le | Op::Ge)
        }
        _ => false,
    }
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| {
        arm.tests.iter().all(|t| test_matches(v, t))
            && (include_prerelease
                || v.pre.is_empty()
                || arm.anchors.contains(&(v.major, v.minor, v.patch)))
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)

fn extrema(
    candidates: &[String],
    range: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| {
            parse_version(s).map_err(|error| CandidateError { index, error })
        })
        .collect::<Result<_, _>>()?;

    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, range) {
            continue;
        }
        if let Some(j) = best {
            let ordering = compare_versions(version, &parsed[j]);
            if (greatest && ordering == Ordering::Greater)
                || (!greatest && ordering == Ordering::Less)
            {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn version_string(v: &Version, include_build: bool) -> String {
    let mut result = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        result.push('-');
        result.push_str(&v.pre.join("."));
    }
    if include_build && !v.build.is_empty() {
        result.push('+');
        result.push_str(&v.build.join("."));
    }
    result
}

type Bound = (Version, bool);

fn stronger_lower(old: Option<Bound>, new: Bound) -> Option<Bound> {
    match old {
        None => Some(new),
        Some(o) => {
            let ordering = compare_versions(&new.0, &o.0);
            if ordering == Ordering::Greater
                || (ordering == Ordering::Equal && !new.1 && o.1)
            {
                Some(new)
            } else {
                Some(o)
            }
        }
    }
}

fn stronger_upper(old: Option<Bound>, new: Bound) -> Option<Bound> {
    match old {
        None => Some(new),
        Some(o) => {
            let ordering = compare_versions(&new.0, &o.0);
            if ordering == Ordering::Less
                || (ordering == Ordering::Equal && !new.1 && o.1)
            {
                Some(new)
            } else {
                Some(o)
            }
        }
    }
}

fn arm_bounds(arm: &Arm) -> (Option<Bound>, Option<Bound>) {
    let mut lower = None;
    let mut upper = None;
    for test in &arm.tests {
        match test.op {
            Op::Eq => {
                lower = stronger_lower(lower, (test.version.clone(), true));
                upper = stronger_upper(upper, (test.version.clone(), true));
            }
            Op::Gt | Op::Ge => {
                lower = stronger_lower(
                    lower,
                    (test.version.clone(), test.op == Op::Ge),
                );
            }
            Op::Lt | Op::Le => {
                upper = stronger_upper(
                    upper,
                    (test.version.clone(), test.op == Op::Le),
                );
            }
        }
    }
    (lower, upper)
}

fn next_core(core: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if core.2 < u64::MAX {
        Some((core.0, core.1, core.2 + 1))
    } else if core.1 < u64::MAX {
        Some((core.0, core.1 + 1, 0))
    } else if core.0 < u64::MAX {
        Some((core.0 + 1, 0, 0))
    } else {
        None
    }
}

fn stable(core: (u64, u64, u64)) -> Version {
    endpoint(core, Vec::new())
}

fn pre_zero(core: (u64, u64, u64)) -> Version {
    endpoint(core, vec!["0".to_string()])
}

fn lower_candidate(lower: &Option<Bound>) -> Option<Version> {
    match lower {
        None => Some(stable((0, 0, 0))),
        Some((version, true)) => Some(version.clone()),
        Some((version, false)) if version.pre.is_empty() => {
            next_core((version.major, version.minor, version.patch)).map(stable)
        }
        Some((version, false)) => {
            let mut pre = version.pre.clone();
            pre.push("0".to_string());
            Some(endpoint(
                (version.major, version.minor, version.patch),
                pre,
            ))
        }
    }
}

fn within_upper(candidate: &Version, upper: &Option<Bound>) -> bool {
    match upper {
        None => true,
        Some((version, inclusive)) => {
            let ordering = compare_versions(candidate, version);
            ordering == Ordering::Less || (*inclusive && ordering == Ordering::Equal)
        }
    }
}

fn interval_has_stable(lower: &Option<Bound>, upper: &Option<Bound>) -> bool {
    let candidate = match lower {
        None => stable((0, 0, 0)),
        Some((v, inclusive)) if v.pre.is_empty() && *inclusive => v.clone(),
        Some((v, _)) if v.pre.is_empty() => {
            match next_core((v.major, v.minor, v.patch)) {
                Some(core) => stable(core),
                None => return false,
            }
        }
        Some((v, _)) => stable((v.major, v.minor, v.patch)),
    };
    within_upper(&candidate, upper)
        && match lower {
            None => true,
            Some((v, inclusive)) => {
                let c = compare_versions(&candidate, v);
                c == Ordering::Greater || (*inclusive && c == Ordering::Equal)
            }
        }
}

fn interval_has_prerelease(
    lower: &Option<Bound>,
    upper: &Option<Bound>,
    anchors: &[(u64, u64, u64)],
) -> bool {
    for &core in anchors {
        let mut candidate = pre_zero(core);
        if let Some((low, inclusive)) = lower {
            let c = compare_versions(&candidate, low);
            if c == Ordering::Less || (c == Ordering::Equal && !inclusive) {
                if (low.major, low.minor, low.patch) == core {
                    if low.pre.is_empty() {
                        candidate = pre_zero(core);
                    } else {
                        candidate = low.clone();
                        if !inclusive {
                            let mut ids = low.pre.clone();
                            ids.push("0".to_string());
                            candidate = endpoint(core, ids);
                        }
                    }
                }
            }
        }

        let low_ok = match lower {
            None => true,
            Some((low, inclusive)) => {
                let c = compare_versions(&candidate, low);
                c == Ordering::Greater || (*inclusive && c == Ordering::Equal)
            }
        };
        if low_ok && within_upper(&candidate, upper) {
            return true;
        }

        let stable_at_core = stable(core);
        let after_lower = match lower {
            None => true,
            Some((low, inclusive)) => {
                let c = compare_versions(&stable_at_core, low);
                c == Ordering::Greater || (*inclusive && c == Ordering::Equal)
            }
        };
        if after_lower && within_upper(&pre_zero(core), upper) {
            return true;
        }
    }
    false
}

fn arm_has_default_witness(arm: &Arm) -> bool {
    let (lower, upper) = arm_bounds(arm);
    let anchors = &arm.anchors;

    if let (Some((l, li)), Some((u, ui))) = (&lower, &upper) {
        let c = compare_versions(l, u);
        if c == Ordering::Greater || (c == Ordering::Equal && !(*li && *ui)) {
            return false;
        }
    }

    interval_has_stable(&lower, &upper)
        || interval_has_prerelease(&lower, &upper, anchors)
}

pub fn canonical(r: &Range) -> String {
    let mut result = Vec::new();
    for arm in &r.arms {
        if !arm_has_default_witness(arm) {
            continue;
        }
        let (lower, upper) = arm_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();

        let low = lower
            .map(|(v, inclusive)| {
                format!(
                    "{}{}",
                    if inclusive { ">=" } else { ">" },
                    version_string(&v, false)
                )
            })
            .unwrap_or_else(|| "*".to_string());
        let high = upper
            .map(|(v, inclusive)| {
                format!(
                    "{}{}",
                    if inclusive { "<=" } else { "<" },
                    version_string(&v, false)
                )
            })
            .unwrap_or_else(|| "*".to_string());
        let pre = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(a, b, c)| format!("{a}.{b}.{c}"))
                .collect::<Vec<_>>()
                .join(",")
        };
        result.push(format!("[{low};{high};pre={pre}]"));
    }

    result.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    result.dedup();
    if result.is_empty() {
        "EMPTY".to_string()
    } else {
        result.join(" || ")
    }
}

fn arm_intersects(a: &Arm, b: &Arm, include_prerelease: bool) -> bool {
    let (al, au) = arm_bounds(a);
    let (bl, bu) = arm_bounds(b);

    let lower = match (al, bl) {
        (None, other) | (other, None) => other,
        (Some(x), Some(y)) => stronger_lower(Some(x), y),
    };
    let upper = match (au, bu) {
        (None, other) | (other, None) => other,
        (Some(x), Some(y)) => stronger_upper(Some(x), y),
    };

    if let (Some((l, li)), Some((u, ui))) = (&lower, &upper) {
        let c = compare_versions(l, u);
        if c == Ordering::Greater || (c == Ordering::Equal && !(*li && *ui)) {
            return false;
        }
    }

    if include_prerelease {
        return lower_candidate(&lower)
            .is_some_and(|v| within_upper(&v, &upper))
            || interval_has_stable(&lower, &upper);
    }

    let common: Vec<_> = a
        .anchors
        .iter()
        .copied()
        .filter(|core| b.anchors.contains(core))
        .collect();
    interval_has_stable(&lower, &upper)
        || interval_has_prerelease(&lower, &upper, &common)
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    a.arms
        .iter()
        .any(|x| b.arms.iter().any(|y| arm_intersects(x, y, include_prerelease)))
}

pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn collect_ranges(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    out: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let range = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            out.push(range);
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                collect_ranges(child, package, label, path, out)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn expression_matches(
    expr: &SetExpr,
    version: &Version,
    include: bool,
    ranges: &[Range],
    cursor: &mut usize,
) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let result = matches_with(version, &ranges[*cursor], include);
            *cursor += 1;
            result
        }
        SetExpr::Union(children) => children.iter().any(|child| {
            expression_matches(child, version, include, ranges, cursor)
        }),
        SetExpr::Intersection(children) => children.iter().all(|child| {
            expression_matches(child, version, include, ranges, cursor)
        }),
    }
}

fn constraint_matches(
    constraint: &Constraint,
    version: &Version,
    include: bool,
    ranges: &[Range],
) -> bool {
    expression_matches(&constraint.expr, version, include, ranges, &mut 0)
}

fn diagnostic_labels(
    package: &Package,
    versions: &[Version],
    include: bool,
    parsed: &[Vec<Range>],
) -> Vec<String> {
    if versions.is_empty() || package.constraints.is_empty() {
        return Vec::new();
    }

    fn search(
        package: &Package,
        versions: &[Version],
        include: bool,
        parsed: &[Vec<Range>],
        target_size: usize,
        start: usize,
        chosen: &mut Vec<usize>,
    ) -> Option<Vec<String>> {
        if chosen.len() == target_size {
            let has_candidate = versions.iter().any(|version| {
                chosen.iter().all(|&i| {
                    constraint_matches(
                        &package.constraints[i],
                        version,
                        include,
                        &parsed[i],
                    )
                })
            });
            if !has_candidate {
                let mut labels: Vec<String> = chosen
                    .iter()
                    .map(|&i| package.constraints[i].label.clone())
                    .collect();
                labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                return Some(labels);
            }
            return None;
        }

        let remaining = target_size - chosen.len();
        if package.constraints.len().saturating_sub(start) < remaining {
            return None;
        }
        for i in start..=package.constraints.len() - remaining {
            chosen.push(i);
            if let Some(found) = search(
                package,
                versions,
                include,
                parsed,
                target_size,
                i + 1,
                chosen,
            ) {
                return Some(found);
            }
            chosen.pop();
        }
        None
    }

    for size in 1..=package.constraints.len() {
        if let Some(labels) = search(
            package,
            versions,
            include,
            parsed,
            size,
            0,
            &mut Vec::new(),
        ) {
            return labels;
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut package_names = HashSet::new();
    for package in packages {
        if !package_names.insert(package.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: package.name.clone(),
            });
        }
    }

    for package in packages {
        let mut labels = HashSet::new();
        for constraint in &package.constraints {
            if !labels.insert(constraint.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: package.name.clone(),
                    label: constraint.label.clone(),
                });
            }
        }
    }

    let mut all_versions = Vec::new();
    for package in packages {
        let mut versions = Vec::new();
        for (index, candidate) in package.candidates.iter().enumerate() {
            versions.push(parse_version(candidate).map_err(|error| {
                SelectError::InvalidCandidate {
                    package: package.name.clone(),
                    index,
                    error,
                }
            })?);
        }
        all_versions.push(versions);
    }

    let mut all_ranges: Vec<Vec<Vec<Range>>> = Vec::new();
    for package in packages {
        let mut package_ranges = Vec::new();
        for constraint in &package.constraints {
            let mut ranges = Vec::new();
            collect_ranges(
                &constraint.expr,
                &package.name,
                &constraint.label,
                &mut Vec::new(),
                &mut ranges,
            )?;
            package_ranges.push(ranges);
        }
        all_ranges.push(package_ranges);
    }

    let mut selections = Vec::new();
    for (pi, package) in packages.iter().enumerate() {
        let versions = &all_versions[pi];
        let mut chosen: Option<usize> = None;

        for (i, version) in versions.iter().enumerate() {
            let accepted = package.constraints.iter().enumerate().all(|(ci, constraint)| {
                constraint_matches(
                    constraint,
                    version,
                    include_prerelease,
                    &all_ranges[pi][ci],
                )
            });
            if !accepted {
                continue;
            }

            if let Some(j) = chosen {
                let ordering = compare_versions(version, &versions[j]);
                let better = match preference {
                    Preference::Highest => ordering == Ordering::Greater,
                    Preference::Lowest => ordering == Ordering::Less,
                    Preference::StableFirst => {
                        (version.pre.is_empty() && !versions[j].pre.is_empty())
                            || (version.pre.is_empty() == versions[j].pre.is_empty()
                                && ordering == Ordering::Greater)
                    }
                };
                if better {
                    chosen = Some(i);
                }
            } else {
                chosen = Some(i);
            }
        }

        if let Some(i) = chosen {
            selections.push(Selection {
                package: package.name.clone(),
                version: package.candidates[i].clone(),
            });
        } else {
            let labels = diagnostic_labels(
                package,
                versions,
                include_prerelease,
                &all_ranges[pi],
            );
            return Err(SelectError::NoSolution {
                package: package.name.clone(),
                labels,
            });
        }
    }

    Ok(selections)
}
