use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(match bytes.get(*pos) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: *pos },
            _ => VersionError::InvalidChar { at: *pos },
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut id_start = start;
    loop {
        let mut i = id_start;
        while i < end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if id_start == end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let part = &s[id_start..end];
        if prerelease
            && part.len() > 1
            && part.bytes().all(is_digit)
            && part.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        ids.push(part.to_string());
        if end >= bytes.len() || bytes[end] != b'.' {
            break;
        }
        id_start = end + 1;
        let mut next_end = id_start;
        while next_end < bytes.len()
            && bytes[next_end] != b'.'
            && (!prerelease || bytes[next_end] != b'+')
        {
            next_end += 1;
        }
        // For a build section, dots are always separators; for prerelease, plus ends it.
        let next_end = if prerelease {
            next_end
        } else {
            let mut n = id_start;
            while n < bytes.len() && bytes[n] != b'.' {
                n += 1;
            }
            n
        };
        id_start = end + 1;
        if id_start > bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        // Continue with a separately located identifier boundary.
        let mut cursor = id_start;
        while cursor < bytes.len()
            && bytes[cursor] != b'.'
            && (!prerelease || bytes[cursor] != b'+')
        {
            cursor += 1;
        }
        // Replace loop state by processing the next segment directly.
        let _ = next_end;
        let part_end = cursor.min(end.max(cursor));
        let segment = &s[id_start..part_end];
        if segment.is_empty() {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        for (off, b) in segment.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar {
                    at: id_start + off,
                });
            }
        }
        if prerelease
            && segment.len() > 1
            && segment.bytes().all(is_digit)
            && segment.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        ids.push(segment.to_string());
        if cursor >= end || (prerelease && bytes[cursor] == b'+') {
            break;
        }
        id_start = cursor + 1;
        let mut next = id_start;
        while next < bytes.len() && bytes[next] != b'.' && (!prerelease || bytes[next] != b'+') {
            next += 1;
        }
        if next == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        // The remaining segments are handled by the outer state below.
        let mut p = id_start;
        loop {
            let mut q = p;
            while q < bytes.len() && bytes[q] != b'.' && (!prerelease || bytes[q] != b'+') {
                q += 1;
            }
            if p == q {
                return Err(VersionError::EmptyIdentifier { at: p });
            }
            let seg = &s[p..q];
            for (off, b) in seg.bytes().enumerate() {
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(VersionError::InvalidChar { at: p + off });
                }
            }
            if prerelease
                && seg.len() > 1
                && seg.bytes().all(is_digit)
                && seg.as_bytes()[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: p });
            }
            ids.push(seg.to_string());
            if q >= bytes.len() || (prerelease && bytes[q] == b'+') {
                return Ok(ids);
            }
            p = q + 1;
        }
    }
    Ok(ids)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(match s.as_bytes().get(p) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p },
            _ => VersionError::InvalidChar { at: p },
        });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(match s.as_bytes().get(p) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p },
            _ => VersionError::InvalidChar { at: p },
        });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    if p == s.len() {
        return Ok(Version {
            major,
            minor,
            patch,
            pre: Vec::new(),
            build: Vec::new(),
        });
    }
    match s.as_bytes()[p] {
        b'-' | b'+' => {}
        _ => return Err(VersionError::InvalidChar { at: p }),
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if s.as_bytes()[p] == b'-' {
        let pre_start = p + 1;
        let plus = s[pre_start..].find('+').map(|n| pre_start + n);
        let pre_end = plus.unwrap_or(s.len());
        if pre_start == pre_end {
            return Err(VersionError::EmptyIdentifier { at: pre_start });
        }
        let mut seg_start = pre_start;
        loop {
            let seg_end = s[seg_start..pre_end]
                .find('.')
                .map(|n| seg_start + n)
                .unwrap_or(pre_end);
            if seg_start == seg_end {
                return Err(VersionError::EmptyIdentifier { at: seg_start });
            }
            for (off, b) in s.as_bytes()[seg_start..seg_end].iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || *b == b'-') {
                    return Err(VersionError::InvalidChar {
                        at: seg_start + off,
                    });
                }
            }
            let seg = &s[seg_start..seg_end];
            if seg.len() > 1 && seg.bytes().all(is_digit) && seg.as_bytes()[0] == b'0' {
                return Err(VersionError::LeadingZero { at: seg_start });
            }
            pre.push(seg.to_string());
            if seg_end == pre_end {
                break;
            }
            seg_start = seg_end + 1;
            if seg_start == pre_end {
                return Err(VersionError::EmptyIdentifier { at: seg_start });
            }
        }
        if let Some(plus_at) = plus {
            p = plus_at;
        } else {
            p = s.len();
        }
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let build_start = p + 1;
        if build_start == s.len() {
            return Err(VersionError::EmptyIdentifier { at: build_start });
        }
        let mut seg_start = build_start;
        loop {
            let seg_end = s[seg_start..]
                .find('.')
                .map(|n| seg_start + n)
                .unwrap_or(s.len());
            if seg_start == seg_end {
                return Err(VersionError::EmptyIdentifier { at: seg_start });
            }
            for (off, b) in s.as_bytes()[seg_start..seg_end].iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || *b == b'-') {
                    return Err(VersionError::InvalidChar {
                        at: seg_start + off,
                    });
                }
            }
            build.push(s[seg_start..seg_end].to_string());
            if seg_end == s.len() {
                break;
            }
            seg_start = seg_end + 1;
            if seg_start == s.len() {
                return Err(VersionError::EmptyIdentifier { at: seg_start });
            }
        }
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_identifier(a: &str, b: &str) -> Ordering {
    match (numeric_id(a), numeric_id(b)) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let c = cmp_identifier(x, y);
                if c != Ordering::Equal {
                    return c;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    tests: Vec<Test>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Test {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let b = s.as_bytes();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && whitespace(b[i]) {
            i += 1;
        }
        if i == b.len() {
            break;
        }
        let start = i;
        while i < b.len() && !whitespace(b[i]) {
            i += 1;
        }
        out.push((start, &s[start..i]));
    }
    out
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum PartialField {
    Num(u64),
    Wild,
}

#[derive(Clone, Debug)]
struct Operand {
    lower: Version,
    full: bool,
    numeric: usize,
}

fn parse_partial(s: &str, offset: usize) -> Result<Operand, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parse_full_operand(s, offset);
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X")) {
        return parse_full_operand(s, offset);
    }
    if fields.len() > 3 || fields.is_empty() {
        return Err(RangeError::Syntax { at: offset });
    }
    let mut nums = Vec::new();
    let mut seen_wild = false;
    let mut numeric = 0;
    for f in &fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at: offset });
        }
        if matches!(*f, "*" | "x" | "X") {
            seen_wild = true;
        } else {
            if seen_wild
                || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: offset });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: offset })?;
            nums.push(n);
            numeric += 1;
        }
    }
    let (major, minor, patch) = match nums.as_slice() {
        [] => (0, 0, 0),
        [a] => (*a, 0, 0),
        [a, b] => (*a, *b, 0),
        [a, b, c] => (*a, *b, *c),
        _ => return Err(RangeError::Syntax { at: offset }),
    };
    Ok(Operand {
        lower: Version {
            major,
            minor,
            patch,
            pre: Vec::new(),
            build: Vec::new(),
        },
        full: false,
        numeric,
    })
}

fn parse_full_operand(s: &str, offset: usize) -> Result<Operand, RangeError> {
    let v = parse_version(s).map_err(|error| RangeError::Version { at: offset, error })?;
    Ok(Operand {
        lower: v,
        full: true,
        numeric: 3,
    })
}

fn next_tuple(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (major, minor, patch) = match field {
        0 => (v.major.checked_add(1).ok_or(RangeError::Syntax { at })?, 0, 0),
        1 => (v.major, v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?, 0),
        _ => (v.major, v.minor, v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?),
    };
    Ok(Version {
        major,
        minor,
        patch,
        pre: vec!["0".into()],
        build: Vec::new(),
    })
}

fn add_test(arm: &mut Arm, op: Op, version: Version) {
    if !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
    arm.tests.push(Test { op, version });
}

fn parse_token_operand(token: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if token.starts_with('^') || token.starts_with('~') {
        let caret = token.starts_with('^');
        let operand_start = start + 1;
        let operand = parse_partial(&token[1..], operand_start)?;
        if operand.numeric == 0 {
            return Err(RangeError::Syntax { at: start });
        }
        add_test(arm, Op::Ge, operand.lower.clone());
        let v = &operand.lower;
        let field = if caret {
            if v.major > 0 || (operand.numeric == 1 && !operand.full) {
                0
            } else if v.minor > 0 || (operand.numeric == 2 && !operand.full) {
                1
            } else {
                2
            }
        } else if operand.numeric == 1 && !operand.full {
            0
        } else {
            1
        };
        let upper = next_tuple(v, field, operand_start)?;
        add_test(arm, Op::Lt, upper);
        return Ok(());
    }

    let (op, body, body_start) = if token.starts_with(">=") {
        (Some(Op::Ge), &token[2..], start + 2)
    } else if token.starts_with("<=") {
        (Some(Op::Le), &token[2..], start + 2)
    } else if token.starts_with('>') {
        (Some(Op::Gt), &token[1..], start + 1)
    } else if token.starts_with('<') {
        (Some(Op::Lt), &token[1..], start + 1)
    } else if token.starts_with('=') {
        (Some(Op::Eq), &token[1..], start + 1)
    } else {
        (None, token, start)
    };

    if let Some(op) = op {
        if body.is_empty() || !body.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }
        let operand = parse_full_operand(body, body_start)?;
        add_test(arm, op, operand.lower);
        return Ok(());
    }

    let operand = parse_partial(body, body_start)?;
    if operand.full {
        add_test(arm, Op::Eq, operand.lower);
    } else if operand.numeric == 1 {
        add_test(arm, Op::Ge, operand.lower.clone());
        add_test(arm, Op::Lt, next_tuple(&operand.lower, 0, body_start)?);
    } else if operand.numeric == 2 {
        add_test(arm, Op::Ge, operand.lower.clone());
        add_test(arm, Op::Lt, next_tuple(&operand.lower, 1, body_start)?);
    }
    Ok(())
}

fn compare_test(v: &Version, t: &Test) -> bool {
    let c = compare_versions(v, &t.version);
    match t.op {
        Op::Eq => c == Ordering::Equal,
        Op::Lt => c == Ordering::Less,
        Op::Le => c != Ordering::Greater,
        Op::Gt => c == Ordering::Greater,
        Op::Ge => c != Ordering::Less,
    }
}

fn split_arms(s: &str) -> Result<Vec<(usize, &str)>, RangeError> {
    let ts = tokens(s);
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut start = 0usize;
    let mut first = true;
    let mut pending = false;
    let mut i = 0;
    while i < ts.len() {
        let (at, t) = ts[i];
        if t == "||" {
            if first || pending {
                return Err(RangeError::Empty { at });
            }
            let end = at;
            arms.push((start, &s[start..end]));
            pending = true;
            first = true;
        } else {
            if first {
                start = at;
                first = false;
                pending = false;
            }
        }
        i += 1;
    }
    if pending {
        return Err(RangeError::Empty { at: s.len() });
    }
    if !first {
        let end = s.len();
        arms.push((start, &s[start..end]));
    }
    Ok(arms)
}

fn parse_arm(text: &str, base: usize) -> Result<Arm, RangeError> {
    let ts = tokens(text);
    let mut arm = Arm {
        tests: Vec::new(),
        anchors: Vec::new(),
    };
    if ts.is_empty() {
        return Err(RangeError::Empty { at: base });
    }
    if ts.iter().any(|(_, t)| *t == "-") {
        let hyphen = ts.iter().find(|(_, t)| *t == "-").unwrap().0 + base;
        if ts.len() != 3 || ts[1].1 != "-" {
            return Err(RangeError::Syntax { at: hyphen });
        }
        let (lo_at, lo_s) = (ts[0].0 + base, ts[0].1);
        let (hi_at, hi_s) = (ts[2].0 + base, ts[2].1);
        let lo = parse_partial(lo_s, lo_at)?;
        let hi = parse_partial(hi_s, hi_at)?;
        if lo.numeric == 0 {
            return Err(RangeError::Syntax { at: lo_at });
        }
        if hi.numeric == 0 {
            return Err(RangeError::Syntax { at: hi_at });
        }
        add_test(&mut arm, Op::Ge, lo.lower.clone());
        if hi.full {
            add_test(&mut arm, Op::Le, hi.lower);
        } else {
            let field = if hi.numeric == 1 { 0 } else { 1 };
            add_test(&mut arm, Op::Lt, next_tuple(&hi.lower, field, hi_at)?);
        }
        return Ok(arm);
    }
    for (at, token) in ts {
        let absolute = base + at;
        if !token.as_bytes()[0].is_ascii_digit()
            && !matches!(token.as_bytes()[0], b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'=')
        {
            return Err(RangeError::Syntax { at: absolute });
        }
        parse_token_operand(token, absolute, &mut arm)?;
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let pieces = split_arms(s)?;
    let mut arms = Vec::new();
    for (base, text) in pieces {
        arms.push(parse_arm(text, base)?);
    }
    Ok(Range { arms })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| {
        arm.tests.iter().all(|t| compare_test(v, t))
            && (include_prerelease
                || v.pre.is_empty()
                || arm.anchors.contains(&(v.major, v.minor, v.patch)))
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index: i, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let cmp = compare_versions(v, &parsed[j]);
            if (greatest && cmp == Ordering::Greater) || (!greatest && cmp == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn version_string(v: &Version, build: bool) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    if build && !v.build.is_empty() {
        s.push('+');
        s.push_str(&v.build.join("."));
    }
    s
}

fn arm_bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>, bool) {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    let mut impossible = false;
    for t in &arm.tests {
        match t.op {
            Op::Eq => {
                lower = stronger_lower(lower, (t.version.clone(), true));
                upper = stronger_upper(upper, (t.version.clone(), true));
            }
            Op::Gt | Op::Ge => {
                lower = stronger_lower(lower, (t.version.clone(), t.op == Op::Ge));
            }
            Op::Lt | Op::Le => {
                upper = stronger_upper(upper, (t.version.clone(), t.op == Op::Le));
            }
        }
    }
    if let (Some((l, li)), Some((u, ui))) = (&lower, &upper) {
        let c = compare_versions(l, u);
        impossible = c == Ordering::Greater || (c == Ordering::Equal && !(*li && *ui));
    }
    (lower, upper, impossible)
}

fn stronger_lower(
    old: Option<(Version, bool)>,
    new: (Version, bool),
) -> Option<(Version, bool)> {
    match old {
        None => Some(new),
        Some(o) => {
            let c = compare_versions(&new.0, &o.0);
            if c == Ordering::Greater || (c == Ordering::Equal && !new.1 && o.1) {
                Some(new)
            } else {
                Some(o)
            }
        }
    }
}

fn stronger_upper(
    old: Option<(Version, bool)>,
    new: (Version, bool),
) -> Option<(Version, bool)> {
    match old {
        None => Some(new),
        Some(o) => {
            let c = compare_versions(&new.0, &o.0);
            if c == Ordering::Less || (c == Ordering::Equal && !new.1 && o.1) {
                Some(new)
            } else {
                Some(o)
            }
        }
    }
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        let (lower, upper, impossible) = arm_bounds(arm);
        if impossible {
            continue;
        }
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let low = lower
            .map(|(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, version_string(&v, false)))
            .unwrap_or_else(|| "*".to_string());
        let high = upper
            .map(|(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, version_string(&v, false)))
            .unwrap_or_else(|| "*".to_string());
        let pre = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(a, b, c)| format!("{a}.{b}.{c}"))
                .collect::<Vec<_>>()
                .join(",")
        };
        strings.push(format!("[{low};{high};pre={pre}]"));
    }
    strings.sort();
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".to_string()
    } else {
        strings.join(" || ")
    }
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s)
            .map(|r| matches_with(v, &r, include))
            .unwrap_or(false),
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, v, include)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, v, include)),
    }
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| {
            SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            }
        }),
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                validate_expr(x, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn conflict_labels(
    package: &Package,
    versions: &[Version],
    include: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if versions.is_empty() || n == 0 {
        return Vec::new();
    }
    for size in 1..=n {
        let mut choice = Vec::new();
        if let Some(result) = find_conflict(package, versions, include, size, 0, &mut choice) {
            return result;
        }
    }
    Vec::new()
}

fn find_conflict(
    package: &Package,
    versions: &[Version],
    include: bool,
    remaining: usize,
    start: usize,
    choice: &mut Vec<usize>,
) -> Option<Vec<String>> {
    if remaining == 0 {
        let satisfiable = versions.iter().any(|v| {
            choice.iter().all(|&i| eval_expr(&package.constraints[i].expr, v, include))
        });
        if !satisfiable {
            let mut labels: Vec<String> = choice
                .iter()
                .map(|&i| package.constraints[i].label.clone())
                .collect();
            labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            return Some(labels);
        }
        return None;
    }
    for i in start..=package.constraints.len().saturating_sub(remaining) {
        choice.push(i);
        if let Some(v) = find_conflict(package, versions, include, remaining - 1, i + 1, choice) {
            return Some(v);
        }
        choice.pop();
    }
    None
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: p.name.clone(),
            });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: p.name.clone(),
                    label: c.label.clone(),
                });
            }
        }
    }
    let mut all_versions = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(),
                index: i,
                error,
            })?);
        }
        all_versions.push(vs);
    }
    for p in packages {
        for c in &p.constraints {
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut out = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &all_versions[pi];
        let mut chosen: Option<usize> = None;
        for (i, v) in versions.iter().enumerate() {
            if !p
                .constraints
                .iter()
                .all(|c| eval_expr(&c.expr, v, include_prerelease))
            {
                continue;
            }
            if let Some(j) = chosen {
                let cmp = compare_versions(v, &versions[j]);
                let better = match preference {
                    Preference::Highest => cmp == Ordering::Greater,
                    Preference::Lowest => cmp == Ordering::Less,
                    Preference::StableFirst => {
                        (v.pre.is_empty() && !versions[j].pre.is_empty())
                            || (v.pre.is_empty() == versions[j].pre.is_empty()
                                && cmp == Ordering::Greater)
                    }
                };
                if better {
                    chosen = Some(i);
                }
            } else {
                chosen = Some(i);
            }
        }
        if let Some(i) = chosen {
            out.push(Selection {
                package: p.name.clone(),
                version: p.candidates[i].clone(),
            });
        } else {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflict_labels(p, versions, include_prerelease),
            });
        }
    }
    Ok(out)
}

fn main() {}
