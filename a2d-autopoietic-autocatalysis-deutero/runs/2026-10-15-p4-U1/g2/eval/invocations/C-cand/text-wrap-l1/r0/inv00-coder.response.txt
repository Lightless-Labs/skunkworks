#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(matches!(byte, b' '..=b'~' | b'\t' | b'\n' | b'\r')) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();
    let mut position = 0;

    while position < bytes.len() {
        while position < bytes.len() && matches!(bytes[position], b' ' | b'\t' | b'\n' | b'\r') {
            position += 1;
        }
        if position == bytes.len() {
            break;
        }

        let start = position;
        while position < bytes.len() && !matches!(bytes[position], b' ' | b'\t' | b'\n' | b'\r') {
            position += 1;
        }
        let mut suffix = &bytes[start..position];

        while !suffix.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if suffix.len() <= width {
                let needed = suffix.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(suffix).expect("validated ASCII"));
                    suffix = &[];
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                let (part, rest) = suffix.split_at(width);
                lines.push(std::str::from_utf8(part).expect("validated ASCII").to_string());
                suffix = rest;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(
                        std::str::from_utf8(&suffix[..available]).expect("validated ASCII"),
                    );
                    lines.push(std::mem::take(&mut line));
                    suffix = &suffix[available..];
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}

fn main() {}
