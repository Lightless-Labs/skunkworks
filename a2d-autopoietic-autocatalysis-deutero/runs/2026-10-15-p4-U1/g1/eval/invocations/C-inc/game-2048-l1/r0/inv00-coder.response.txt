```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Vec<usize> = match side {
        Side::Left => (0..row.len()).collect(),
        Side::Right => (0..row.len()).rev().collect(),
    };

    let mut tiles = Vec::new();
    for index in indices {
        if row[index] != 0 {
            tiles.push((index, row[index]));
        }
    }

    let mut outputs: Vec<(u32, Option<Merge>)> = Vec::new();
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;

    while i < tiles.len() {
        if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
            let (first, value) = tiles[i];
            let (second, _) = tiles[i + 1];

            let Some(new_value) = value.checked_mul(2) else {
                return Err(Error::MergeOverflow { first, second });
            };

            let to = match side {
                Side::Left => outputs.len(),
                Side::Right => row.len() - 1 - outputs.len(),
            };
            let merge = Merge {
                from: [first, second],
                to,
                value: new_value,
            };
            score += u64::from(new_value);
            merges.push(merge.clone());
            outputs.push((new_value, Some(merge)));
            i += 2;
        } else {
            outputs.push((tiles[i].1, None));
            i += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    for (position, (value, _)) in outputs.into_iter().enumerate() {
        let index = match side {
            Side::Left => position,
            Side::Right => row.len() - 1 - position,
        };
        cells[index] = value;
    }

    let changed = cells != row;

    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}

fn main() {}
```
