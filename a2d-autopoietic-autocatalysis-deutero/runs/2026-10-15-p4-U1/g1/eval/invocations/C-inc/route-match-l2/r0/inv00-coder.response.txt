#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parts = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        for route in &self.routes {
            let Some(params) = match_route(&route.segments, &parts) else {
                continue;
            };

            let replace = match &best {
                None => true,
                Some((current, _)) => {
                    route.rank > current.rank
                        || (route.rank == current.rank && route.id < current.id)
                }
            };
            if replace {
                best = Some((route, params));
            }
        }

        best.map(|(route, params)| Match {
            route: route.id,
            params,
        })
        .ok_or(Error::NotFound)
    }
}

#[derive(Clone)]
struct PathPart {
    decoded: String,
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    let raw_parts = split_path(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::with_capacity(raw_parts.len());
    let mut names = std::collections::HashSet::new();

    for (index, (part, start)) in raw_parts.iter().enumerate() {
        if part.starts_with(':') || part.starts_with('*') {
            let wildcard = part.starts_with('*');
            let name = &part[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard && index + 1 != raw_parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = decode_static_segment(part, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<PathPart>, Error> {
    let raw_parts = split_path(path).map_err(|at| Error::InvalidPath { at })?;
    raw_parts
        .iter()
        .map(|(part, start)| {
            decode_static_segment(part, *start)
                .map(|decoded| PathPart { decoded })
                .map_err(|at| Error::InvalidPath { at })
        })
        .collect()
}

fn split_path(path: &str) -> Result<Vec<(&str, usize)>, usize> {
    if !path.starts_with('/') {
        return Err(0);
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == bytes.len() && i > 0 && bytes[i - 1] == b'/' {
                return Err(i - 1);
            }
            if i == start {
                return Err(start);
            }
            parts.push((&path[start..i], start));
            start = i + 1;
        }
        i += 1;
    }

    Ok(parts)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_static_segment(segment: &str, start: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(start + i);
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(start + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(byte) {
                return Err(start + i);
            }
            decoded.push(byte);
            i += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(
    pattern: &[Segment],
    path: &[PathPart],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                let actual = path.get(path_index)?;
                if actual.decoded != *expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let actual = path.get(path_index)?;
                params.push((name.clone(), actual.decoded.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..]
                    .iter()
                    .map(|part| part.decoded.as_str())
                    .collect::<Vec<_>>()
                    .join("/");
                params.push((name.clone(), value));
                path_index = path.len();
            }
        }
    }

    (path_index == path.len()).then_some(params)
}

fn main() {}
