#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum PatternSegment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    method: String,
    segments: Vec<PatternSegment>,
    trailing_slash: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (segments, trailing_slash) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments.iter().any(|s| matches!(s, PatternSegment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                PatternSegment::Static(_) => 2,
                PatternSegment::Param(_) => 1,
                PatternSegment::Wildcard(_) => 0,
            });
        }

        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing_slash) = parse_path(head)?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (index, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != request_trailing_slash
            {
                continue;
            }

            if let Some(params) = match_route(route, &request_segments) {
                candidates.push((index, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && !candidates
                .iter()
                .any(|(index, _)| self.routes[*index].method == "HEAD")
        {
            "GET"
        } else {
            method
        };

        let eligible: Vec<(usize, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(index, _)| self.routes[*index].method == eligible_method)
            .cloned()
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(index, _)| self.routes[*index].method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (winner_index, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                self.routes[*a]
                    .rank
                    .cmp(&self.routes[*b].rank)
                    .then_with(|| b.cmp(a))
            })
            .expect("eligible candidates are nonempty");

        let route = &self.routes[winner_index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != request_trailing_slash
            && !request_segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing_slash {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner_index,
            params,
            redirect,
        })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn parse_pattern(input: &str) -> Result<(Vec<PatternSegment>, bool), Error> {
    let raw_segments = split_head(input).map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();
    let mut wildcard_seen = false;

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        let is_final = index + 1 == raw_segments.len();

        if let Some(name) = raw.strip_prefix(':') {
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(PatternSegment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_capture_name(name) || !names.insert(name.to_owned()) || !is_final {
                return Err(Error::InvalidPattern { at: *start });
            }
            wildcard_seen = true;
            segments.push(PatternSegment::Wildcard(name.to_owned()));
        } else {
            let decoded = decode_static(raw, *start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(PatternSegment::Static(decoded));
        }
    }

    let _ = wildcard_seen;
    Ok((segments, input.len() > 1 && input.ends_with('/')))
}

fn parse_path(input: &str) -> Result<(Vec<String>, bool), Error> {
    let raw_segments = split_head(input).map_err(|at| Error::InvalidPath { at })?;
    let mut segments = Vec::with_capacity(raw_segments.len());
    for (raw, start) in raw_segments {
        segments.push(decode_static(&raw, start).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((segments, input.len() > 1 && input.ends_with('/')))
}

fn split_head(input: &str) -> Result<Vec<(String, usize)>, usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                if i == bytes.len() && start == bytes.len() {
                    break;
                }
                return Err(start);
            }
            if i == bytes.len() && start == bytes.len() {
                break;
            }
            if i == bytes.len() && bytes[i - 1] == b'/' {
                break;
            }
            result.push((input[start..i].to_owned(), start));
            start = i + 1;
        }
        i += 1;
    }

    if input.ends_with('/') {
        if result.is_empty() {
            return Ok(Vec::new());
        }
    } else if start < bytes.len() {
        result.push((input[start..].to_owned(), start));
    }

    Ok(result)
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn decode_static(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut output = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return Err(base + i);
            };
            let decoded = high * 16 + low;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            output.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            output.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(output).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut request_index = 0;

    for segment in &route.segments {
        match segment {
            PatternSegment::Static(value) => {
                if request.get(request_index)? != value {
                    return None;
                }
                request_index += 1;
            }
            PatternSegment::Param(name) => {
                let value = request.get(request_index)?;
                params.push((name.clone(), value.clone()));
                request_index += 1;
            }
            PatternSegment::Wildcard(name) => {
                let value = request[request_index..].join("/");
                params.push((name.clone(), value));
                request_index = request.len();
                break;
            }
        }
    }

    if request_index == request.len() {
        Some(params)
    } else {
        None
    }
}

fn main() {}
