```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Debug)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

#[derive(Clone, Debug)]
struct State {
    burst_index: usize,
    remaining: u32,
    level: usize,
}

fn mean_x100(total: u64, count: usize) -> u64 {
    let denominator = count as u64;
    (total * 100 + denominator / 2) / denominator
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::new();
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
        });
    }
    events.sort_by_key(|event| {
        (
            event.time,
            match event.kind {
                EventKind::Arrival => 0usize,
                EventKind::Wake { .. } => 1usize,
            },
            event.index,
        )
    });

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            burst_index: 0,
            remaining: job.bursts[0],
            level: job.level,
        })
        .collect();
    let mut completions = vec![0u64; jobs.len()];
    let mut event_cursor = 0usize;
    let mut time = 0u64;
    let mut completed = 0usize;
    let mut last_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    while completed < jobs.len() {
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor].clone();
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake { level } => queues[level].push_back(event.index),
            }
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let job_id = jobs[index].id;

        if last_id.is_some_and(|previous| previous != job_id) {
            time += switch_cost as u64;
        }
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor].clone();
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake { level } => queues[level].push_back(event.index),
            }
            event_cursor += 1;
        }

        let quantum = quanta[level];
        let consumed = states[index].remaining.min(quantum);
        let start = time;
        time += consumed as u64;
        states[index].remaining -= consumed;

        let burst_ended = states[index].remaining == 0;
        let finished = burst_ended && states[index].burst_index + 1 == jobs[index].bursts.len();
        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished,
            level,
        });

        if burst_ended && !finished {
            let io_position = states[index].burst_index + 1;
            let io_delay = jobs[index].bursts[io_position] as u64;
            states[index].burst_index += 2;
            states[index].remaining = jobs[index].bursts[states[index].burst_index];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: time + io_delay,
                kind: EventKind::Wake { level: wake_level },
                index,
            });
            events[event_cursor..].sort_by_key(|event| {
                (
                    event.time,
                    match event.kind {
                        EventKind::Arrival => 0usize,
                        EventKind::Wake { .. } => 1usize,
                    },
                    event.index,
                )
            });
        }

        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor].clone();
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake { level } => queues[level].push_back(event.index),
            }
            event_cursor += 1;
        }

        if finished {
            completions[index] = time;
            completed += 1;
        } else if burst_ended {
            // The wake event at the endpoint has already been admitted.
        } else {
            states[index].level = (level + 1).min(quanta.len() - 1);
            queues[states[index].level].push_back(index);
        }

        last_id = Some(job_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completions[index];
        let turnaround = completion - job.arrival as u64;
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| burst as u64)
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&delay| delay as u64)
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: mean_x100(total_turnaround, jobs.len()),
        end: time,
    })
}

fn main() {}
```
