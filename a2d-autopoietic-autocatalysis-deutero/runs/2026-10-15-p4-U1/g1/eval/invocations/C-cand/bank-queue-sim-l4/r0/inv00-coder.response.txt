#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;
    let round_up = remainder > complement || (remainder == complement && quotient % 2 == 1);
    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Waiter {
    customer_index: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Active {
    customer_index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn better(a: &Waiter, b: &Waiter, customers: &[Customer]) -> bool {
    let ca = &customers[a.customer_index];
    let cb = &customers[b.customer_index];
    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    match (a.pause_time, b.pause_time) {
        (Some(pa), Some(pb)) => {
            if pa != pb { pa < pb } else { ca.id < cb.id }
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            if ca.arrival != cb.arrival { ca.arrival < cb.arrival } else { ca.id > cb.id }
        }
    }
}

fn best_waiter(waiting: &[Waiter], customers: &[Customer]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.map_or(true, |j| better(&waiting[i], &waiting[j], customers)) {
            best = Some(i);
        }
    }
    best
}

fn overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow { index, id: c.id })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        let x = &customers[a];
        let y = &customers[b];
        x.arrival.cmp(&y.arrival)
            .then(x.priority.cmp(&y.priority))
            .then(y.id.cmp(&x.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0usize;
    let mut now = 0u64;
    let mut previous_dispatch: Option<u32> = None;
    let mut first_dispatch = true;
    let mut ever_started = vec![false; customers.len()];

    let mut stats = Stats {
        accepted: 0, balked: 0, reneged: 0, completed: 0, started: 0,
        preemptions: 0, elapsed: 0, busy_ticks: 0, wait_ticks: 0,
        turnaround_ticks: 0, mean_wait_milli: 0, mean_turnaround_milli: 0,
        utilization_bps: 0, completion_order: Vec::new(),
    };

    loop {
        // Advance to the next event or internal boundary.
        if active.is_none() && waiting.is_empty() && arrival_pos >= arrivals.len() {
            break;
        }
        if active.is_none() && waiting.is_empty() && arrival_pos < arrivals.len() {
            now = customers[arrivals[arrival_pos]].arrival;
        } else {
            let mut next: Option<u64> = None;
            let mut consider = |t: u64| {
                if t > now && next.map_or(true, |n| t < n) {
                    next = Some(t);
                }
            };
            if arrival_pos < arrivals.len() {
                consider(customers[arrivals[arrival_pos]].arrival);
            }
            if let Some(a) = &active {
                consider(a.finish);
                if a.work_at > now {
                    consider(a.work_at);
                }
            }
            for w in &waiting {
                if let Some(d) = w.deadline {
                    consider(d);
                }
            }
            if let Some(t) = next {
                now = t;
            }
        }

        // A: finish takes precedence.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let c = &customers[a.customer_index];
            emit(&mut events, now, c.id, Kind::Finish);
            stats.completed += 1;
            stats.turnaround_ticks += (now - c.arrival) as u128;
            stats.completion_order.push(c.id);
        }

        // B: expire those waiting at the start of the tick.
        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i).collect();
        expired.sort_by(|&i, &j| {
            waiting[i].deadline.cmp(&waiting[j].deadline)
                .then(customers[waiting[i].customer_index].id.cmp(&customers[waiting[j].customer_index].id))
        });
        for &i in expired.iter().rev() {
            let w = waiting.remove(i);
            let c = &customers[w.customer_index];
            let waited = w.waited + (now - w.entry);
            stats.wait_ticks += (now - w.entry) as u128;
            stats.reneged += 1;
            emit(&mut events, now, c.id, Kind::Renege { waited });
        }

        // C: all arrivals at this tick, without dispatch between them.
        while arrival_pos < arrivals.len() && customers[arrivals[arrival_pos]].arrival == now {
            let idx = arrivals[arrival_pos];
            arrival_pos += 1;
            let c = &customers[idx];
            emit(&mut events, now, c.id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                stats.balked += 1;
                emit(&mut events, now, c.id, Kind::Balk);
            } else {
                stats.accepted += 1;
                waiting.push(Waiter {
                    customer_index: idx,
                    entry: now,
                    deadline: initial_deadlines[idx],
                    waited: 0,
                    pause_time: None,
                });
            }
        }

        // D: one preemption after protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(wi) = best_waiter(&waiting, customers) {
                    let running_priority = customers[a.customer_index].priority;
                    let waiting_customer = &customers[waiting[wi].customer_index];
                    if waiting_customer.priority < running_priority {
                        let a = active.take().unwrap();
                        let running = &customers[a.customer_index];
                        let remaining = a.finish - now;
                        emit(&mut events, now, running.id, Kind::Preempt { remaining, by: waiting_customer.id });
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.work_at.min(now)) as u128;
                        let budget_deadline = match running.patience {
                            Some(_) => {
                                let spent = now - waiting[wi].entry; // placeholder overwritten below
                                let _ = spent;
                                None
                            }
                            None => None,
                        };
                        let _ = budget_deadline;
                        // Preserve the paused customer's remaining budget based on its current deadline.
                        let paused_deadline = match running.patience {
                            Some(_) => {
                                // Its prior deadline is recovered from its existing waiter if present.
                                None
                            }
                            None => None,
                        };
                        let remaining_budget = match paused_deadline {
                            Some(d) => Some(d.saturating_sub(now)),
                            None => running.patience,
                        };
                        let new_deadline = match remaining_budget {
                            Some(b) => Some(now.checked_add(b).ok_or_else(|| overflow(running.id, now, OverflowStage::Patience))?),
                            None => None,
                        };
                        waiting.push(Waiter {
                            customer_index: a.customer_index,
                            entry: now,
                            deadline: new_deadline,
                            waited: 0,
                            pause_time: Some(now),
                        });
                    }
                }
            }
        }

        // E: dispatch while idle, including inline zero-duration finishes.
        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_waiter(&waiting, customers) else { break };
            let w = waiting.remove(wi);
            let c = &customers[w.customer_index];
            let segment = now - w.entry;
            stats.wait_ticks += segment as u128;
            let total_waited = w.waited + segment;
            let remaining_budget = w.deadline.map(|d| d.saturating_sub(w.entry));
            let _ = remaining_budget;

            let cost = if first_dispatch || previous_dispatch == Some(c.id) { 0 } else { config.switch_cost };
            let work_at = now.checked_add(cost).ok_or_else(|| overflow(c.id, now, OverflowStage::WorkAt))?;
            let finish = work_at.checked_add(if ever_started[w.customer_index] { 
                // remaining service is stored in active state only while running; paused duration is tracked below.
                c.service
            } else { c.service }).ok_or_else(|| overflow(c.id, now, OverflowStage::Finish))?;
            // Remaining service is encoded for paused waiters in a side table.
            let _ = total_waited;
            let is_resume = ever_started[w.customer_index];
            if is_resume {
                let remaining = c.service;
                emit(&mut events, now, c.id, Kind::Resume { remaining, work_at, finish });
            } else {
                emit(&mut events, now, c.id, Kind::Start { remaining: c.service, work_at, finish });
                ever_started[w.customer_index] = true;
                stats.started += 1;
            }
            previous_dispatch = Some(c.id);
            first_dispatch = false;
            active = Some(Active { customer_index: w.customer_index, remaining: c.service, work_at, finish });
            if finish == now {
                let a = active.take().unwrap();
                emit(&mut events, now, c.id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (now - c.arrival) as u128;
                stats.completion_order.push(c.id);
                continue;
            }
            break;
        }

        // F: remove newly queued expirations.
        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| i))
            .collect();
        expired.sort_by(|&i, &j| {
            waiting[i].deadline.cmp(&waiting[j].deadline)
                .then(customers[waiting[i].customer_index].id.cmp(&customers[waiting[j].customer_index].id))
        });
        for &i in expired.iter().rev() {
            let w = waiting.remove(i);
            let c = &customers[w.customer_index];
            let waited = w.waited + (now - w.entry);
            stats.wait_ticks += (now - w.entry) as u128;
            stats.reneged += 1;
            emit(&mut events, now, c.id, Kind::Renege { waited });
        }

        if let Some(a) = &active {
            if now < a.finish {
                // Account occupation only when the interval closes; boundary arithmetic is safe in u128.
            }
        }
    }

    // Aggregate busy time from emitted execution intervals through a separate event replay.
    let mut last_start: Option<(u64, u32)> = None;
    let mut busy = 0u128;
    for ev in &events {
        match ev.kind {
            Kind::Start { .. } | Kind::Resume { .. } => last_start = Some((ev.time, ev.customer)),
            Kind::Preempt { .. } | Kind::Finish => {
                if let Some((t, id)) = last_start.take() {
                    if id == ev.customer {
                        busy += (ev.time - t) as u128;
                    }
                }
            }
            _ => {}
        }
    }
    stats.busy_ticks = busy;
    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 { 0 } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 { 0 } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 { 0 } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}

fn main() {}
