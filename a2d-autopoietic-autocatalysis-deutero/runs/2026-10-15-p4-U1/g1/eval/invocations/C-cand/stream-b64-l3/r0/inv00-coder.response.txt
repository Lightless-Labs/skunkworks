#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    const STANDARD: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const URL_SAFE: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
    match alphabet {
        Alphabet::Standard => STANDARD,
        Alphabet::UrlSafe => URL_SAFE,
    }
}

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    alphabet_bytes(alphabet)
        .iter()
        .position(|&b| b == byte)
        .map(|n| n as u8)
}

fn encode_symbols(input: &[u8], options: Options) -> Vec<u8> {
    let table = alphabet_bytes(options.alphabet);
    let emit_padding = options.padding != Padding::Forbidden;
    let mut out = Vec::with_capacity(input.len().saturating_add(2) / 3 * 4);

    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        let n = ((chunk[0] as u32) << 16)
            | ((chunk[1] as u32) << 8)
            | chunk[2] as u32;
        out.push(table[((n >> 18) & 63) as usize]);
        out.push(table[((n >> 12) & 63) as usize]);
        out.push(table[((n >> 6) & 63) as usize]);
        out.push(table[(n & 63) as usize]);
    }

    let rem = chunks.remainder();
    if rem.len() == 1 {
        let b = rem[0];
        out.push(table[(b >> 2) as usize]);
        out.push(table[((b & 3) << 4) as usize]);
        if emit_padding {
            out.extend_from_slice(b"==");
        }
    } else if rem.len() == 2 {
        let n = ((rem[0] as u16) << 8) | rem[1] as u16;
        out.push(table[((n >> 10) & 63) as usize]);
        out.push(table[((n >> 4) & 63) as usize]);
        out.push(table[((n & 15) << 2) as usize]);
        if emit_padding {
            out.push(b'=');
        }
    }
    out
}

fn wrap_symbols(symbols: &[u8], whitespace: Whitespace) -> String {
    match whitespace {
        Whitespace::Wrapped { width, ending } if symbols.len() > width => {
            let separator: &[u8] = match ending {
                LineEnding::Lf => b"\n",
                LineEnding::CrLf => b"\r\n",
            };
            let mut out = Vec::with_capacity(
                symbols.len() + ((symbols.len() - 1) / width) * separator.len(),
            );
            for (i, chunk) in symbols.chunks(width).enumerate() {
                if i != 0 {
                    out.extend_from_slice(separator);
                }
                out.extend_from_slice(chunk);
            }
            String::from_utf8(out).expect("base64 symbols are ASCII")
        }
        _ => String::from_utf8(symbols.to_vec()).expect("base64 symbols are ASCII"),
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;
    Ok(wrap_symbols(&encode_symbols(input, options), options.whitespace))
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            buffered: Vec::new(),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        if input.is_empty() {
            return Ok(String::new());
        }

        let mut combined = Vec::with_capacity(self.buffered.len() + input.len());
        combined.extend_from_slice(&self.buffered);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let symbols = encode_symbols(&combined[..complete_len], self.options);
        self.buffered.clear();
        self.buffered.extend_from_slice(&combined[complete_len..]);
        Ok(wrap_symbols(&symbols, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let symbols = encode_symbols(&self.buffered, self.options);
        self.buffered.clear();
        self.finished = true;
        Ok(wrap_symbols(&symbols, self.options.whitespace))
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

struct DecoderCore {
    options: Options,
    offset: usize,
    frame: FrameState,
    row_count: usize,
    group: Vec<Symbol>,
    pending_second_pad: Option<Symbol>,
    padded_group: bool,
    finished: bool,
    sticky: Option<Error>,
}

impl DecoderCore {
    fn new(options: Options) -> Self {
        Self {
            options,
            offset: 0,
            frame: FrameState::Normal,
            row_count: 0,
            group: Vec::with_capacity(4),
            pending_second_pad: None,
            padded_group: false,
            finished: false,
            sticky: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky = Some(error.clone());
        error
    }

    fn is_ignored_whitespace(byte: u8) -> bool {
        matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let here = self.offset;
            self.offset = self.offset.saturating_add(1);

            if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
                match self.frame {
                    FrameState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(self.fail(Error::BadWrap { offset: here }));
                        }
                        self.frame = FrameState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        continue;
                    }
                    FrameState::NeedSymbol { .. } => {
                        self.frame = FrameState::Normal;
                        self.row_count = 0;
                    }
                    FrameState::Normal => {}
                }

                if self.row_count == width {
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.frame = FrameState::NeedSymbol {
                                separator_offset: here,
                            };
                            continue;
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.frame = FrameState::NeedLf { cr_offset: here };
                            continue;
                        }
                        _ => return Err(self.fail(Error::BadWrap { offset: here })),
                    }
                } else if matches!(byte, b'\r' | b'\n') {
                    return Err(self.fail(Error::BadWrap { offset: here }));
                }

                self.row_count += 1;
            } else if matches!(self.options.whitespace, Whitespace::IgnoreAscii)
                && Self::is_ignored_whitespace(byte)
            {
                continue;
            }

            if let Some(pad) = self.pending_second_pad {
                if byte != b'=' {
                    return Err(self.fail(Error::BadPadding { offset: here }));
                }
                self.pending_second_pad = None;
                self.group.push(Symbol { byte, offset: here });
                self.complete_group(&mut out)?;
                self.padded_group = true;
                continue;
            }

            if self.padded_group {
                return Err(self.fail(Error::BadPadding { offset: here }));
            }

            let position = self.group.len();
            if byte == b'=' {
                match position {
                    0 | 1 => {
                        return Err(self.fail(Error::BadPadding { offset: here }));
                    }
                    2 => {
                        if self.options.padding == Padding::Forbidden {
                            return Err(self.fail(Error::BadPadding { offset: here }));
                        }
                        self.pending_second_pad = Some(Symbol { byte, offset: here });
                        self.group.push(Symbol { byte, offset: here });
                        continue;
                    }
                    3 => {
                        if self.options.padding == Padding::Forbidden {
                            return Err(self.fail(Error::BadPadding { offset: here }));
                        }
                        self.group.push(Symbol { byte, offset: here });
                        self.complete_group(&mut out)?;
                        self.padded_group = true;
                        continue;
                    }
                    _ => unreachable!(),
                }
            }

            let Some(value) = digit(byte, self.options.alphabet) else {
                return Err(self.fail(Error::InvalidByte { offset: here, byte }));
            };
            let _ = value;
            self.group.push(Symbol { byte, offset: here });
            if self.group.len() == 4 {
                self.complete_group(&mut out)?;
            }
        }
        Ok(out)
    }

    fn complete_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.group.len() != 4 {
            return Ok(());
        }

        let a = digit(self.group[0].byte, self.options.alphabet)
            .ok_or_else(|| Error::BadPadding { offset: self.group[0].offset })?;
        let b = digit(self.group[1].byte, self.options.alphabet)
            .ok_or_else(|| Error::BadPadding { offset: self.group[1].offset })?;

        if self.group[2].byte == b'=' {
            if self.group[3].byte != b'=' {
                return Err(Error::BadPadding {
                    offset: self.group[3].offset,
                });
            }
            out.push((a << 2) | (b >> 4));
        } else {
            let c = digit(self.group[2].byte, self.options.alphabet)
                .ok_or_else(|| Error::BadPadding { offset: self.group[2].offset })?;
            if self.group[3].byte == b'=' {
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
            } else {
                let d = digit(self.group[3].byte, self.options.alphabet)
                    .ok_or_else(|| Error::BadPadding { offset: self.group[3].offset })?;
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                out.push((c << 6) | d);
            }
        }

        self.group.clear();
        Ok(())
    }

    fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame {
                FrameState::NeedLf { cr_offset } => {
                    return Err(self.fail(Error::BadWrap { offset: cr_offset }));
                }
                FrameState::NeedSymbol { separator_offset } => {
                    return Err(self.fail(Error::BadWrap {
                        offset: separator_offset,
                    }));
                }
                FrameState::Normal => {}
            }
        }

        if let Some(pad) = self.pending_second_pad {
            return Err(self.fail(Error::BadPadding { offset: pad.offset }));
        }

        let mut out = Vec::new();
        match self.group.len() {
            0 => {}
            1 => {
                return Err(self.fail(Error::Truncated {
                    offset: self.group[0].offset,
                }));
            }
            2 if self.options.padding != Padding::Required => {
                let a = digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = digit(self.group[1].byte, self.options.alphabet).unwrap();
                out.push((a << 2) | (b >> 4));
            }
            3 if self.options.padding != Padding::Required => {
                let a = digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = digit(self.group[1].byte, self.options.alphabet).unwrap();
                let c = digit(self.group[2].byte, self.options.alphabet).unwrap();
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
            }
            _ => {
                return Err(self.fail(Error::Truncated {
                    offset: self.group[0].offset,
                }));
            }
        }

        self.group.clear();
        self.finished = true;
        Ok(out)
    }
}

pub struct Decoder {
    core: DecoderCore,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            core: DecoderCore::new(options),
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        self.core.push(input)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        self.core.finish()
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate_options(options)?;
    let mut decoder = DecoderCore::new(options);
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

fn main() {}
