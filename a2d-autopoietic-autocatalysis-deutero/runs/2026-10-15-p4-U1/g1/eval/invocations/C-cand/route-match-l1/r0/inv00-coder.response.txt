#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum PatternSegment {
    Static(String),
    Param(String),
}

struct Route {
    id: usize,
    segments: Vec<PatternSegment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                PatternSegment::Static(_) => 2,
                PatternSegment::Param(_) => 1,
            })
            .collect();
        let id = self.routes.len();

        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .as_bytes()
            .iter()
            .position(|&byte| byte == b'?' || byte == b'#')
            .unwrap_or(path.len());
        let segments = parse_path(&path[..head_end])?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if route.segments.len() != segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_segment, path_segment) in route.segments.iter().zip(&segments) {
                match pattern_segment {
                    PatternSegment::Static(expected) if expected == path_segment => {}
                    PatternSegment::Static(_) => {
                        matches = false;
                        break;
                    }
                    PatternSegment::Param(name) => {
                        params.push((name.clone(), path_segment.clone()));
                    }
                }
            }

            if matches {
                let replace = match best.as_ref() {
                    None => true,
                    Some((previous, _)) => route.rank > previous.rank,
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<PatternSegment>, Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    for end in 1..=bytes.len() {
        if end == bytes.len() || bytes[end] == b'/' {
            if end == start {
                return Err(Error::InvalidPattern { at: start });
            }

            let raw = &pattern[start..end];
            if raw.starts_with(':') {
                let name = &raw[1..];
                let valid = name
                    .as_bytes()
                    .first()
                    .is_some_and(|b| b.is_ascii_lowercase())
                    && name.as_bytes()[1..]
                        .iter()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
                if !valid || !names.insert(name.to_owned()) {
                    return Err(Error::InvalidPattern { at: start });
                }
                segments.push(PatternSegment::Param(name.to_owned()));
            } else if raw.starts_with('*') {
                return Err(Error::InvalidPattern { at: start });
            } else {
                let decoded = decode_segment(raw, start)
                    .map_err(|at| Error::InvalidPattern { at })?;
                segments.push(PatternSegment::Static(decoded));
            }

            if end == bytes.len() {
                break;
            }
            start = end + 1;
        }
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let mut segments = Vec::new();
    let mut start = 1;

    for end in 1..=bytes.len() {
        if end == bytes.len() || bytes[end] == b'/' {
            if end == start {
                return Err(Error::InvalidPath { at: start });
            }

            let raw = &path[start..end];
            let decoded =
                decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })?;
            segments.push(decoded);

            if end == bytes.len() {
                break;
            }
            start = end + 1;
        }
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' {
            if index + 2 >= bytes.len() {
                return Err(base + index);
            }

            let high = uppercase_hex(bytes[index + 1]).ok_or(base + index)?;
            let low = uppercase_hex(bytes[index + 2]).ok_or(base + index)?;
            let value = (high << 4) | low;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + index);
            }

            decoded.push(value);
            index += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            index += 1;
        } else {
            return Err(base + index);
        }
    }

    // All accepted decoded bytes are ASCII, so this conversion cannot fail.
    Ok(String::from_utf8(decoded).expect("decoded segment is ASCII"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn main() {}
