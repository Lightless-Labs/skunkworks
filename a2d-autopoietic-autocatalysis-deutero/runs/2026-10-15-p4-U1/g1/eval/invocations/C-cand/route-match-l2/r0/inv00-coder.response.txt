#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let decoded = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(route, &decoded) {
                let replace = best
                    .as_ref()
                    .map(|(current, _)| route.rank > current.rank)
                    .unwrap_or(true);
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let parts = split_segments(input);
    let count = parts.len();
    let mut segments = Vec::with_capacity(count);
    let mut names = Vec::<String>::new();

    for (index, (start, end)) in parts.iter().copied().enumerate() {
        if start == end {
            if index + 1 == count && end == input.len() {
                return Err(Error::InvalidPattern { at: end - 1 });
            }
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &input[start..end];
        let first = raw.as_bytes()[0];
        if first == b':' || first == b'*' {
            let name_bytes = &raw.as_bytes()[1..];
            if !valid_name(name_bytes) {
                return Err(Error::InvalidPattern { at: start });
            }
            let name = std::str::from_utf8(name_bytes).unwrap().to_owned();
            if names.iter().any(|existing| existing == &name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.clone());

            if first == b'*' && index + 1 != count {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if first == b':' {
                Segment::Param(name)
            } else {
                Segment::Wildcard(name)
            });
        } else {
            let value = decode_segment(raw.as_bytes(), start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(value));
        }
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }
    Ok(segments)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let parts = split_segments(input);
    let count = parts.len();
    let mut segments = Vec::with_capacity(count);

    for (index, (start, end)) in parts.iter().copied().enumerate() {
        if start == end {
            if index + 1 == count && end == input.len() {
                return Err(Error::InvalidPath { at: end - 1 });
            }
            return Err(Error::InvalidPath { at: start });
        }
        let decoded = decode_segment(input[start..end].as_bytes(), start)
            .map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }
    Ok(segments)
}

// Returns segment byte ranges after the leading slash. A trailing slash is
// represented by a final empty range.
fn split_segments(input: &str) -> Vec<(usize, usize)> {
    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            result.push((start, i));
            start = i + 1;
        }
    }
    result.push((start, bytes.len()));
    result
}

fn valid_name(bytes: &[u8]) -> bool {
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..].iter().all(|&b| {
        b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'
    })
}

fn decode_segment(bytes: &[u8], base: usize) -> Result<String, usize> {
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            decoded.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = path.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}

fn main() {}
