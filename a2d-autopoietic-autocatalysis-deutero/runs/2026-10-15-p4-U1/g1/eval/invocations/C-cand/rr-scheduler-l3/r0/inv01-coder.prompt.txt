pub struct Job {
    pub id: u32,
    pub arrival: u32,

pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completion_times = vec![None; jobs.len()];
    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let mut event_cursor = 0;
    let mut completed = 0;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;

    while completed < jobs.len() {
        while event_cursor < events.len()
            && u64::from(jobs[events[event_cursor]].arrival) <= time
        {
            let index = events[event_cursor];
            if jobs[index].ticks == 0 {
                completion_times[index] = Some(u64::from(jobs[index].arrival));
                completed += 1;
            } else {
                queues[jobs[index].level].push_back(index);
            }
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());

        let Some(queue_level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }

            let next_time = u64::from(jobs[events[event_cursor]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let index = queues[queue_level].pop_front().unwrap();
        let job_id = jobs[index].id;

        if last_dispatched_id.is_some_and(|last_id| last_id != job_id) {
            time += u64::from(switch_cost);

        while event_cursor < events.len()

        {
            let arrived = events[event_cursor];
            if jobs[arrived].ticks == 0 {
                completion_times[arrived] = Some(u64::from(jobs[arrived].arrival));
                completed += 1;
            } else {
                queues[jobs[arrived].level].push_back(arrived);
            }
            event_cursor += 1;
        }

        let run_level = levels[index];
        let consumed = remaining[index].min(quanta[run_level]);
        let start = time;
        time += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished,
            level: run_level,
        });

            let arrived = events[event_cursor];
            if jobs[arrived].ticks == 0 {

                completed += 1;
            } else {
                queues[jobs[arrived].level].push_back(arrived);
            }
            event_cursor += 1;

            completed += 1;
        } else {

            queues[levels[index]].push_back(index);

        .iter()
        .enumerate()

        .collect();

        runs,
        idle,
        completion,
        end: time,
    })
