#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { opening: usize },
    Double { opening: usize },
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    after_dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let bytes = input.as_bytes();

    if bytes.get(after_dollar) != Some(&b'{') {
        let mut end = after_dollar;
        let first = input[after_dollar..].chars().next().unwrap();
        if !is_name_start(first) {
            return Ok(("$".to_string(), after_dollar));
        }
        end += first.len_utf8();
        while end < input.len() {
            let ch = input[end..].chars().next().unwrap();
            if !is_name_continue(ch) {
                break;
            }
            end += ch.len_utf8();
        }
        let name = &input[after_dollar..end];
        return Ok((lookup(name, env).unwrap_or("").to_string(), end));
    }

    let contents_start = after_dollar + 1;
    let mut cursor = contents_start;
    let mut closing = None;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        if ch == '\\' {
            cursor += ch.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if ch == '}' {
            closing = Some(cursor);
            break;
        } else {
            cursor += ch.len_utf8();
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[contents_start..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, ch)| (!is_name_continue(ch)).then_some(i))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    if name.is_empty()
        || !name.chars().next().is_some_and(is_name_start)
        || !name.chars().all(is_name_continue)
    {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let suffix = &contents[name_end..];
    let value = if suffix.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = suffix.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut cursor = 0;
    while cursor < text.len() {
        let ch = text[cursor..].chars().next().unwrap();
        cursor += ch.len_utf8();
        if ch == '\\' && cursor < text.len() {
            let next = text[cursor..].chars().next().unwrap();
            result.push(next);
            cursor += next.len_utf8();
        } else {
            result.push(ch);
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.bytes().position(|byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + ch.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                cursor = next_cursor;
            }
            Quote::Double { .. } => {
                if ch == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    let after_next = next_cursor + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.push(next);
                        }
                        cursor = after_next;
                    } else {
                        word.push('\\');
                        cursor = next_cursor;
                    }
                } else if ch == '$' {
                    let (value, end) = parse_expansion(input, cursor, next_cursor, env)?;
                    word.push_str(&value);
                    active = true;
                    cursor = end;
                } else {
                    word.push(ch);
                    cursor = next_cursor;
                }
            }
            Quote::None => {
                if is_separator(ch) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next_cursor;
                } else if ch == '\'' {
                    quote = Quote::Single { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                } else if ch == '"' {
                    quote = Quote::Double { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    let after_next = next_cursor + next.len_utf8();
                    if next == '\n' {
                        cursor = after_next;
                    } else {
                        word.push(next);
                        active = true;
                        cursor = after_next;
                    }
                } else if ch == '$' {
                    let (value, end) = parse_expansion(input, cursor, next_cursor, env)?;
                    word.push_str(&value);
                    active = true;
                    cursor = end;
                } else {
                    word.push(ch);
                    active = true;
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single { opening } => {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        Quote::Double { opening } => {
            return Err(SplitError::UnterminatedDouble { pos: opening });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn main() {}
