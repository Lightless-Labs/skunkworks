#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width)
        || !(1..=8).contains(&board.height)
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (!(2..=2_147_483_648).contains(&value) || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

fn lines(board: &Board, direction: Direction) -> Vec<Vec<usize>> {
    let mut result = Vec::new();

    match direction {
        Direction::Left => {
            for row in 0..board.height {
                result.push((0..board.width).map(|col| row * board.width + col).collect());
            }
        }
        Direction::Right => {
            for row in 0..board.height {
                result.push((0..board.width).rev().map(|col| row * board.width + col).collect());
            }
        }
        Direction::Up => {
            for col in 0..board.width {
                result.push((0..board.height).map(|row| row * board.width + col).collect());
            }
        }
        Direction::Down => {
            for col in 0..board.width {
                result.push((0..board.height).rev().map(|row| row * board.width + col).collect());
            }
        }
    }

    result
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    for line in lines(board, direction) {
        let tiles: Vec<(usize, u32)> = line
            .iter()
            .filter_map(|&index| {
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut output_position = 0;
        let mut i = 0;

        while i < tiles.len() {
            let (first_index, first_value) = tiles[i];

            if i + 1 < tiles.len() && tiles[i + 1].1 == first_value {
                let (second_index, _) = tiles[i + 1];
                if first_value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: second_index,
                    });
                }

                let value = first_value * 2;
                let to = line[output_position];
                cells[to] = value;
                merges.push(Merge {
                    from: [first_index, second_index],
                    to,
                    value,
                });
                score += u64::from(value);
                output_position += 1;
                i += 2;
            } else {
                cells[line[output_position]] = first_value;
                output_position += 1;
                i += 1;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let mut legal = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(result) if result.changed => legal.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }

    Ok(legal)
}

fn main() {}
