#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    raw: Vec<String>,
}

#[derive(Clone)]
struct Line {
    pieces: Vec<String>,
    displayed: String,
    consumed: usize,
    marker: bool,
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.raw < b.raw
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut std::collections::HashMap<(usize, usize, bool), Candidate>,
) -> Candidate {
    if wi >= words.len() {
        return Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
            consumed: Vec::new(),
            raw: Vec::new(),
        };
    }
    let key = (wi, offset, first);
    if let Some(c) = memo.get(&key) {
        return c.clone();
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let cap = options.width - indent;
    let mut best: Option<Candidate> = None;

    let mut pieces = Vec::<String>::new();
    let mut display = String::new();
    let mut display_count = 0usize;
    let mut consumed_count = 0usize;

    for idx in wi..words.len() {
        let start = if idx == wi { offset } else { 0 };
        let word = &words[idx];
        if start >= word.chars.len() {
            continue;
        }
        let gap = if pieces.is_empty() { 0 } else { 1 };
        let available = cap.saturating_sub(display_count + gap);
        if available == 0 {
            break;
        }

        let remain = word.chars.len() - start;
        let fit = remain.min(available);

        // Completed-word ending.
        if fit == remain {
            let spelling: String = word.chars[start..].iter().collect();
            let mut next_pieces = pieces.clone();
            next_pieces.push(spelling.clone());
            let mut next_display = display.clone();
            if !next_display.is_empty() {
                next_display.push(' ');
            }
            next_display.push_str(&spelling);

            let mut line = String::new();
            line.extend(std::iter::repeat_n(' ', indent));
            line.push_str(&next_display);
            let used = indent + next_display.chars().count();
            let gap_cost = (options.width - used) as u64;
            let rest = solve(words, idx + 1, 0, false, options, memo);
            let final_line = rest.lines.is_empty();
            let line_cost = if final_line {
                gap_cost * gap_cost / 4
            } else {
                gap_cost * gap_cost
            } + if !final_line && next_pieces.len() == 1 { 3 } else { 0 };
            let mut candidate = rest.clone();
            candidate.cost += line_cost;
            candidate.lines.insert(
                0,
                Line {
                    pieces: next_pieces,
                    displayed: line.clone(),
                    consumed: consumed_count + remain,
                    marker: false,
                },
            );
            candidate.consumed.insert(0, consumed_count + remain);
            candidate.raw.insert(0, line);
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        // Marker endings within this word.
        for &m in &word.markers {
            if m < start || m >= word.chars.len() || m - start + 1 > available {
                continue;
            }
            let prefix: String = word.chars[start..m + 1].iter().collect();
            let mut next_display = display.clone();
            if !next_display.is_empty() {
                next_display.push(' ');
            }
            next_display.push_str(&prefix);
            next_display.push('-');

            let mut line = String::new();
            line.extend(std::iter::repeat_n(' ', indent));
            line.push_str(&next_display);
            let used = indent + next_display.chars().count();
            let gap_cost = (options.width - used) as u64;
            let rest = solve(words, idx, m + 1, false, options, memo);
            let final_line = rest.lines.is_empty();
            let line_cost = if final_line {
                gap_cost * gap_cost / 4
            } else {
                gap_cost * gap_cost
            } + options.hyphen_cost as u64
                + if !final_line && pieces.len() + 1 == 1 { 3 } else { 0 };
            let mut candidate = rest.clone();
            candidate.cost += line_cost;
            candidate.marker_lines += 1;
            candidate.lines.insert(
                0,
                Line {
                    pieces: {
                        let mut p = pieces.clone();
                        p.push(prefix);
                        p
                    },
                    displayed: line.clone(),
                    consumed: consumed_count + m + 1 - start,
                    marker: true,
                },
            );
            candidate.consumed.insert(0, consumed_count + m + 1 - start);
            candidate.raw.insert(0, line);
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        // Continue to the next word if this one fits completely.
        if fit < remain {
            // Raw split is available only on an otherwise empty line and only
            // when no marker prefix can fit.
            if pieces.is_empty() && start == offset && remain > cap {
                let any_marker_fits = word
                    .markers
                    .iter()
                    .any(|&m| m >= start && m < word.chars.len() && m - start + 2 <= cap);
                if !any_marker_fits {
                    let end = start + cap;
                    let prefix: String = word.chars[start..end].iter().collect();
                    let mut line = String::new();
                    line.extend(std::iter::repeat_n(' ', indent));
                    line.push_str(&prefix);
                    let rest = solve(words, idx, end, false, options, memo);
                    let final_line = rest.lines.is_empty();
                    let gap_cost = (options.width - (indent + prefix.chars().count())) as u64;
                    let line_cost = if final_line {
                        gap_cost * gap_cost / 4
                    } else {
                        gap_cost * gap_cost
                    } + (2 * options.width + 1) as u64
                        + if !final_line { 3 } else { 0 };
                    let mut candidate = rest.clone();
                    candidate.cost += line_cost;
                    candidate.lines.insert(
                        0,
                        Line {
                            pieces: vec![prefix],
                            displayed: line.clone(),
                            consumed: cap,
                            marker: false,
                        },
                    );
                    candidate.consumed.insert(0, cap);
                    candidate.raw.insert(0, line);
                    if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }
            }
            break;
        }

        let spelling: String = word.chars[start..].iter().collect();
        if !display.is_empty() {
            display.push(' ');
            display_count += 1;
        }
        display.push_str(&spelling);
        display_count += spelling.chars().count();
        consumed_count += remain;
        pieces.push(spelling);
        if display_count >= cap {
            break;
        }
    }

    let answer = best.unwrap_or_else(|| Candidate {
        cost: u64::MAX / 4,
        marker_lines: usize::MAX / 4,
        lines: Vec::new(),
        consumed: Vec::new(),
        raw: Vec::new(),
    });
    memo.insert(key, answer.clone());
    answer
}

fn wrap_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut memo = std::collections::HashMap::new();
    let mut candidate = solve(words, 0, 0, true, options, &mut memo);
    let line_count = candidate.lines.len();
    let width = options.width;
    for (i, line) in candidate.lines.iter_mut().enumerate() {
        let indent = if i == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let mut rendered = String::new();
        rendered.extend(std::iter::repeat_n(' ', indent));
        if options.justify && i + 1 < line_count && line.pieces.len() >= 2 {
            let gaps = line.pieces.len() - 1;
            let used = indent + line.displayed.chars().count();
            let extra = width - used;
            let each = extra / gaps;
            let remainder = extra % gaps;
            for (j, piece) in line.pieces.iter().enumerate() {
                if j > 0 {
                    let add = each + usize::from((j - 1 + i % gaps) % gaps < remainder);
                    rendered.extend(std::iter::repeat_n(' ', 1 + add));
                }
                rendered.push_str(piece);
                if line.marker && j + 1 == line.pieces.len() {
                    rendered.push('-');
                }
            }
        } else {
            rendered.push_str(&line.displayed);
        }
        line.displayed = rendered;
    }
    candidate.lines.into_iter().map(|l| l.displayed).collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, ch) in &chars {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= chars.len() {
                return false;
            }
            let c = chars[j as usize].1;
            c != '~' && !is_sep(c)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let mut q = p;
            while q < normalized.len() && normalized[q] == '\n' {
                q += 1;
            }
            if q - p >= 2 {
                let words = parse_words(&segment);
                result.extend(wrap_paragraph(&words, options));
                for _ in 1..(q - p) {
                    result.push(String::new());
                }
                segment.clear();
                p = q;
                continue;
            }
        }
        segment.push(normalized[p]);
        p += 1;
    }
    let words = parse_words(&segment);
    result.extend(wrap_paragraph(&words, options));
    Ok(result)
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current_chars = Vec::new();
    let mut markers = Vec::new();
    for &c in chars {
        if is_sep(c) {
            if !current_chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current_chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(current_chars.len());
        } else {
            current_chars.push(c);
        }
    }
    if !current_chars.is_empty() {
        words.push(Word {
            chars: current_chars,
            markers,
        });
    }
    words
}

fn main() {}
