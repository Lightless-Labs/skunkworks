#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    pieces: Vec<String>,
    consumed: usize,
    marker: bool,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    raw: Vec<String>,
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.raw < b.raw
}

fn line_text(line: &Line, indent: usize, justify: bool, width: usize, line_index: usize, final_line: bool) -> String {
    let mut rendered = " ".repeat(indent);
    if line.pieces.is_empty() {
        return rendered;
    }

    let gaps = line.pieces.len().saturating_sub(1);
    let base_gap = 1usize;
    let piece_width: usize = line.pieces.iter().map(|p| p.chars().count()).sum();
    let mut remaining = piece_width + gaps * base_gap + usize::from(line.marker);
    if justify && !final_line && gaps > 0 {
        let extra = width.saturating_sub(indent + remaining);
        let each = extra / gaps;
        let rem = extra % gaps;
        let start = line_index % gaps;
        for (i, piece) in line.pieces.iter().enumerate() {
            if i > 0 {
                let gap_index = i - 1;
                let bonus = usize::from((gap_index + gaps - start) % gaps < rem);
                rendered.push_str(&" ".repeat(base_gap + each + bonus));
            }
            rendered.push_str(piece);
            if line.marker && i + 1 == line.pieces.len() {
                rendered.push('-');
            }
        }
        return rendered;
    }

    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            rendered.push(' ');
        }
        rendered.push_str(piece);
    }
    if line.marker {
        rendered.push('-');
    }
    remaining = 0;
    let _ = remaining;
    rendered
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut std::collections::HashMap<(usize, usize, bool), Candidate>,
) -> Candidate {
    if wi >= words.len() {
        return Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
            consumed: Vec::new(),
            raw: Vec::new(),
        };
    }

    let key = (wi, offset, first);
    if let Some(c) = memo.get(&key) {
        return c.clone();
    }

    let indent = if first { options.first_indent } else { options.rest_indent };
    let cap = options.width - indent;
    let mut best: Option<Candidate> = None;
    let mut pieces: Vec<String> = Vec::new();
    let mut displayed_count = 0usize;
    let mut consumed_count = 0usize;

    for idx in wi..words.len() {
        let start = if idx == wi { offset } else { 0 };
        let word = &words[idx];
        if start >= word.chars.len() {
            continue;
        }

        let gap = usize::from(!pieces.is_empty());
        if displayed_count + gap >= cap {
            break;
        }
        let available = cap - displayed_count - gap;
        let remain = word.chars.len() - start;

        if remain <= available {
            let spelling: String = word.chars[start..].iter().collect();
            let mut line_pieces = pieces.clone();
            line_pieces.push(spelling);
            let used = indent
                + displayed_count
                + gap
                + remain;
            let rest = solve(words, idx + 1, 0, false, options, memo);
            let final_line = rest.lines.is_empty();
            let line_gap = (options.width - used) as u64;
            let cost = if final_line { line_gap * line_gap / 4 } else { line_gap * line_gap }
                + if !final_line && line_pieces.len() == 1 { 3 } else { 0 };

            let mut candidate = rest.clone();
            candidate.cost += cost;
            candidate.lines.insert(0, Line {
                pieces: line_pieces.clone(),
                consumed: consumed_count + remain,
                marker: false,
            });
            candidate.consumed.insert(0, consumed_count + remain);
            candidate.raw.insert(0, make_raw_line(&line_pieces, indent, false));
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        for &marker in &word.markers {
            if marker < start || marker >= word.chars.len() {
                continue;
            }
            let prefix_len = marker + 1 - start;
            if prefix_len + 1 > available {
                continue;
            }
            let prefix: String = word.chars[start..=marker].iter().collect();
            let mut line_pieces = pieces.clone();
            line_pieces.push(prefix);
            let used = indent + displayed_count + gap + prefix_len + 1;
            let rest = solve(words, idx, marker + 1, false, options, memo);
            let final_line = rest.lines.is_empty();
            let line_gap = (options.width - used) as u64;
            let cost = if final_line { line_gap * line_gap / 4 } else { line_gap * line_gap }
                + options.hyphen_cost as u64
                + if !final_line && line_pieces.len() == 1 { 3 } else { 0 };

            let mut candidate = rest.clone();
            candidate.cost += cost;
            candidate.marker_lines += 1;
            candidate.lines.insert(0, Line {
                pieces: line_pieces.clone(),
                consumed: consumed_count + prefix_len,
                marker: true,
            });
            candidate.consumed.insert(0, consumed_count + prefix_len);
            candidate.raw.insert(0, make_raw_line(&line_pieces, indent, true));
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        if remain > available {
            if pieces.is_empty() && remain > cap {
                let any_marker_fits = word.markers.iter().any(|&m| {
                    m >= start && m < word.chars.len() && m + 2 - start <= cap
                });
                if !any_marker_fits {
                    let end = start + cap;
                    let prefix: String = word.chars[start..end].iter().collect();
                    let raw_pieces = vec![prefix];
                    let rest = solve(words, idx, end, false, options, memo);
                    let final_line = rest.lines.is_empty();
                    let line_gap = (options.width - (indent + cap)) as u64;
                    let cost = if final_line { line_gap * line_gap / 4 } else { line_gap * line_gap }
                        + (2 * options.width + 1) as u64
                        + if !final_line { 3 } else { 0 };

                    let mut candidate = rest.clone();
                    candidate.cost += cost;
                    candidate.lines.insert(0, Line {
                        pieces: raw_pieces.clone(),
                        consumed: cap,
                        marker: false,
                    });
                    candidate.consumed.insert(0, cap);
                    candidate.raw.insert(0, make_raw_line(&raw_pieces, indent, false));
                    if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }
            }
            break;
        }

        let spelling: String = word.chars[start..].iter().collect();
        if !pieces.is_empty() {
            displayed_count += 1;
        }
        displayed_count += remain;
        consumed_count += remain;
        pieces.push(spelling);
        if displayed_count >= cap {
            break;
        }
    }

    let answer = best.unwrap_or(Candidate {
        cost: u64::MAX / 4,
        marker_lines: usize::MAX / 4,
        lines: Vec::new(),
        consumed: Vec::new(),
        raw: Vec::new(),
    });
    memo.insert(key, answer.clone());
    answer
}

fn make_raw_line(pieces: &[String], indent: usize, marker: bool) -> String {
    let mut line = " ".repeat(indent);
    for (i, piece) in pieces.iter().enumerate() {
        if i > 0 {
            line.push(' ');
        }
        line.push_str(piece);
    }
    if marker {
        line.push('-');
    }
    line
}

fn wrap_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut memo = std::collections::HashMap::new();
    let candidate = solve(words, 0, 0, true, options, &mut memo);
    let line_count = candidate.lines.len();
    candidate.lines.iter().enumerate().map(|(i, line)| {
        let indent = if i == 0 { options.first_indent } else { options.rest_indent };
        line_text(line, indent, options.justify, options.width, i, i + 1 == line_count)
    }).collect()
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current_chars = Vec::new();
    let mut markers = Vec::new();

    for &c in chars {
        if is_sep(c) {
            if !current_chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current_chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(current_chars.len());
        } else {
            current_chars.push(c);
        }
    }

    if !current_chars.is_empty() {
        words.push(Word {
            chars: current_chars,
            markers,
        });
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, ch) in &chars {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid = |j: isize| -> bool {
            if j < 0 || j as usize >= chars.len() {
                return false;
            }
            let c = chars[j as usize].1;
            c != '~' && !is_sep(c)
        };
        if !valid(i as isize - 1) || !valid(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i].1 == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 2;
            } else {
                i += 1;
            }
        } else {
            normalized.push(chars[i].1);
            i += 1;
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let mut q = p;
            while q < normalized.len() && normalized[q] == '\n' {
                q += 1;
            }
            if q - p >= 2 {
                let words = parse_words(&segment);
                result.extend(wrap_paragraph(&words, options));
                for _ in 1..(q - p) {
                    result.push(String::new());
                }
                segment.clear();
                p = q;
                continue;
            }
        }
        segment.push(normalized[p]);
        p += 1;
    }
    let words = parse_words(&segment);
    result.extend(wrap_paragraph(&words, options));
    Ok(result)
}

fn main() {}
