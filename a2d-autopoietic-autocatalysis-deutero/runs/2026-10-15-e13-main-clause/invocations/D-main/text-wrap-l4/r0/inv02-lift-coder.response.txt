#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    pieces: Vec<String>,
    marker: bool,
    raw_split: bool,
    consumed: usize,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    raw: Vec<String>,
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.raw < b.raw
}

fn raw_line(pieces: &[String], indent: usize, marker: bool) -> String {
    let mut s = " ".repeat(indent);
    for (i, piece) in pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(piece);
    }
    if marker {
        s.push('-');
    }
    s
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut std::collections::HashMap<(usize, usize, bool), Candidate>,
) -> Candidate {
    if wi >= words.len() {
        return Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
            consumed: Vec::new(),
            raw: Vec::new(),
        };
    }
    let key = (wi, offset, first);
    if let Some(c) = memo.get(&key) {
        return c.clone();
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let cap = options.width - indent;
    let mut best = None;
    let mut pieces: Vec<String> = Vec::new();
    let mut displayed = 0usize;
    let mut consumed_before = 0usize;

    for idx in wi..words.len() {
        let start = if idx == wi { offset } else { 0 };
        let word = &words[idx];
        if start >= word.chars.len() {
            continue;
        }

        let gap = usize::from(!pieces.is_empty());
        if displayed + gap >= cap {
            break;
        }
        let available = cap - displayed - gap;
        let remain = word.chars.len() - start;

        if remain <= available {
            let spelling: String = word.chars[start..].iter().collect();
            let mut line_pieces = pieces.clone();
            line_pieces.push(spelling);
            let used = indent + displayed + gap + remain;
            let rest = solve(words, idx + 1, 0, false, options, memo);
            let final_line = rest.lines.is_empty();
            let line_gap = (options.width - used) as u64;
            let cost = if final_line {
                line_gap * line_gap / 4
            } else {
                line_gap * line_gap
            } + if !final_line && line_pieces.len() == 1 {
                3
            } else {
                0
            };

            let mut candidate = rest;
            candidate.cost += cost;
            candidate.lines.insert(
                0,
                Line {
                    pieces: line_pieces.clone(),
                    marker: false,
                    raw_split: false,
                    consumed: consumed_before + remain,
                },
            );
            candidate.consumed.insert(0, consumed_before + remain);
            candidate
                .raw
                .insert(0, raw_line(&line_pieces, indent, false));
            if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        for &marker in &word.markers {
            if marker < start || marker >= word.chars.len() {
                continue;
            }
            let prefix_len = marker + 1 - start;
            if prefix_len + 1 > available {
                continue;
            }
            let prefix: String = word.chars[start..=marker].iter().collect();
            let mut line_pieces = pieces.clone();
            line_pieces.push(prefix);
            let used = indent + displayed + gap + prefix_len + 1;
            let rest = solve(words, idx, marker + 1, false, options, memo);
            let final_line = rest.lines.is_empty();
            let line_gap = (options.width - used) as u64;
            let cost = if final_line {
                line_gap * line_gap / 4
            } else {
                line_gap * line_gap
            } + options.hyphen_cost as u64
                + if !final_line && line_pieces.len() == 1 {
                    3
                } else {
                    0
                };

            let mut candidate = rest;
            candidate.cost += cost;
            candidate.marker_lines += 1;
            candidate.lines.insert(
                0,
                Line {
                    pieces: line_pieces.clone(),
                    marker: true,
                    raw_split: false,
                    consumed: consumed_before + prefix_len,
                },
            );
            candidate
                .consumed
                .insert(0, consumed_before + prefix_len);
            candidate
                .raw
                .insert(0, raw_line(&line_pieces, indent, true));
            if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        if remain > available {
            if pieces.is_empty() && remain > cap {
                let marker_fits = word.markers.iter().any(|&m| {
                    m >= start && m < word.chars.len() && m + 2 - start <= cap
                });
                if !marker_fits {
                    let end = start + cap;
                    let prefix: String = word.chars[start..end].iter().collect();
                    let split_pieces = vec![prefix];
                    let rest = solve(words, idx, end, false, options, memo);
                    let final_line = rest.lines.is_empty();
                    let line_gap = (options.width - (indent + cap)) as u64;
                    let cost = if final_line {
                        line_gap * line_gap / 4
                    } else {
                        line_gap * line_gap
                    } + (2 * options.width + 1) as u64
                        + if !final_line { 3 } else { 0 };

                    let mut candidate = rest;
                    candidate.cost += cost;
                    candidate.lines.insert(
                        0,
                        Line {
                            pieces: split_pieces.clone(),
                            marker: false,
                            raw_split: true,
                            consumed: cap,
                        },
                    );
                    candidate.consumed.insert(0, cap);
                    candidate
                        .raw
                        .insert(0, raw_line(&split_pieces, indent, false));
                    if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }
            }
            break;
        }

        let spelling: String = word.chars[start..].iter().collect();
        if !pieces.is_empty() {
            displayed += 1;
        }
        displayed += remain;
        consumed_before += remain;
        pieces.push(spelling);
        if displayed >= cap {
            break;
        }
    }

    let answer = best.unwrap_or(Candidate {
        cost: u64::MAX / 4,
        marker_lines: usize::MAX / 4,
        lines: Vec::new(),
        consumed: Vec::new(),
        raw: Vec::new(),
    });
    memo.insert(key, answer.clone());
    answer
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Vec::new();
    let mut markers = Vec::new();

    for &c in chars {
        if is_sep(c) {
            if !current.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(current.len());
        } else {
            current.push(c);
        }
    }
    if !current.is_empty() {
        words.push(Word {
            chars: current,
            markers,
        });
    }
    words
}

fn render_line(
    line: &Line,
    indent: usize,
    justify: bool,
    width: usize,
    line_index: usize,
    final_line: bool,
) -> String {
    let mut rendered = " ".repeat(indent);
    let gaps = line.pieces.len().saturating_sub(1);
    if justify && !final_line && gaps > 0 {
        let piece_width: usize = line.pieces.iter().map(|p| p.chars().count()).sum();
        let used = indent + piece_width + gaps + usize::from(line.marker);
        let extra = width - used;
        let each = extra / gaps;
        let rem = extra % gaps;
        let start = line_index % gaps;

        for (i, piece) in line.pieces.iter().enumerate() {
            if i > 0 {
                let gap = i - 1;
                let bonus = usize::from((gap + gaps - start) % gaps < rem);
                rendered.push_str(&" ".repeat(1 + each + bonus));
            }
            rendered.push_str(piece);
            if line.marker && i + 1 == line.pieces.len() {
                rendered.push('-');
            }
        }
    } else {
        for (i, piece) in line.pieces.iter().enumerate() {
            if i > 0 {
                rendered.push(' ');
            }
            rendered.push_str(piece);
        }
        if line.marker {
            rendered.push('-');
        }
    }
    rendered
}

fn wrap_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut memo = std::collections::HashMap::new();
    let candidate = solve(words, 0, 0, true, options, &mut memo);
    let line_count = candidate.lines.len();

    candidate
        .lines
        .iter()
        .enumerate()
        .map(|(i, line)| {
            let indent = if i == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let _ = line.raw_split;
            let _ = line.consumed;
            render_line(line, indent, options.justify, options.width, i, i + 1 == line_count)
        })
        .collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, ch) in &chars {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_neighbor = |j: isize| {
            if j < 0 || j as usize >= chars.len() {
                return false;
            }
            let c = chars[j as usize].1;
            c != '~' && !is_sep(c)
        };
        if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i].1 == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 2;
            } else {
                i += 1;
            }
        } else {
            normalized.push(chars[i].1);
            i += 1;
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let mut q = p;
            while q < normalized.len() && normalized[q] == '\n' {
                q += 1;
            }
            if q - p >= 2 {
                let words = parse_words(&segment);
                result.extend(wrap_paragraph(&words, options));
                for _ in 1..(q - p) {
                    result.push(String::new());
                }
                segment.clear();
                p = q;
                continue;
            }
        }
        segment.push(normalized[p]);
        p += 1;
    }

    let words = parse_words(&segment);
    result.extend(wrap_paragraph(&words, options));
    Ok(result)
}

fn main() {}
