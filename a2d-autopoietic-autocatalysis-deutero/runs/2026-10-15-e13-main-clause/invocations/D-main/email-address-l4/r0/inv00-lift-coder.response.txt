#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn is_horizontal(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold_at(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_horizontal) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_horizontal) {
        j += 1;
    }
    Some(j)
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut protected = false;
    let mut in_run = false;
    let mut run_start = 0usize;
    let mut run_end = 0usize;

    while i < input.len() {
        let b = input[i];

        if protected {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    bytes.push(input[i + 1]);
                    positions.push(i + 1);
                    i += 2;
                } else {
                    i += 1;
                }
            } else {
                if b == b'"' {
                    protected = false;
                }
                i += 1;
            }
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                if !in_run {
                    in_run = true;
                    run_start = i;
                }
                run_end = i + 2;
                i += 2;
                continue;
            }

            if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold_at(input, i) {
                    if !in_run {
                        in_run = true;
                        run_start = i;
                    }
                    run_end = end;
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }

            if !(is_printable(b) || b == b'\t' || b == b'(' || b == b')') {
                return Err(EmailError::CommentChar { at: i });
            }

            if !in_run {
                in_run = true;
                run_start = i;
            }
            run_end = i + 1;

            match b {
                b'(' => {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                }
                b')' => depth -= 1,
                _ => {}
            }
            i += 1;
            continue;
        }

        if b == b'(' || b == b')' || is_horizontal(b) || b == b'\r' || b == b'\n' {
            if !in_run {
                in_run = true;
                run_start = i;
            }

            if b == b'(' {
                if depth == 0 {
                    outer_open = i;
                }
                depth = 1;
                run_end = i + 1;
                i += 1;
                continue;
            }
            if b == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold_at(input, i) {
                    run_end = end;
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }

            run_end = i + 1;
            i += 1;
            continue;
        }

        if in_run {
            runs.push(Run {
                start: run_start,
                end: run_end,
            });
            in_run = false;
        }

        if b >= 0x80 {
            return Err(EmailError::NonAscii { at: i });
        }

        if bytes.is_empty() && b == b'"' {
            protected = true;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }

    if in_run {
        runs.push(Run {
            start: run_start,
            end: run_end,
        });
    }

    // Comment contents can contain non-ASCII only via invalid bytes, diagnosed
    // during scanning. For retained bytes the early raw-input check handles it.
    let _ = outer_open;
    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn original_pos(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn is_dot_atom(local: &[u8]) -> bool {
    if local.is_empty() || local.len() > 64 {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == local.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_core(
    input: &str,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 0x80) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let n = cleaned.bytes.len();

    for run in &cleaned.runs {
        let left = cleaned.positions.iter().position(|&p| p >= run.end);
        let right = cleaned.positions.iter().rposition(|&p| p < run.start);
        let legal_start = left.is_none();
        let legal_end = right.is_none();
        let adjacent_left_at = right
            .and_then(|idx| cleaned.bytes.get(idx))
            .is_some_and(|&b| b == b'@');
        let adjacent_right_at = left
            .and_then(|idx| cleaned.bytes.get(idx))
            .is_some_and(|&b| b == b'@');
        if !(legal_start || legal_end || adjacent_left_at || adjacent_right_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if n == 0 {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if cleaned.bytes[0] == b'"' {
        let mut i = 1;
        while i < n {
            if cleaned.bytes[i] == b'\\' {
                i = i.saturating_add(2);
            } else if cleaned.bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: original_pos(&cleaned, 0, raw.len()),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    while i < n {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if cleaned.bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: original_pos(&cleaned, separators[1], raw.len()),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: original_pos(&cleaned, sep, raw.len()),
        });
    }

    let mut decoded = Vec::new();
    let mut origins = Vec::new();
    let mut quoted = false;
    if cleaned.bytes[0] == b'"' {
        quoted = true;
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: original_pos(&cleaned, end + 1, raw.len()),
            });
        }
        let mut j = 1;
        while j < end {
            let b = cleaned.bytes[j];
            if b == b'\\' {
                if j + 1 >= end
                    || !matches!(cleaned.bytes[j + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: original_pos(&cleaned, j, raw.len()),
                    });
                }
                decoded.push(cleaned.bytes[j + 1]);
                origins.push(original_pos(&cleaned, j, raw.len()));
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: original_pos(&cleaned, j, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(original_pos(&cleaned, j, raw.len()));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: original_pos(&cleaned, end, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: original_pos(&cleaned, 64, raw.len()),
            });
        }
        let mut prev_dot = false;
        for j in 0..sep {
            let b = cleaned.bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || prev_dot {
                    return Err(EmailError::LocalDot {
                        at: original_pos(&cleaned, j, raw.len()),
                    });
                }
                prev_dot = true;
            } else if is_atom_char(b) {
                prev_dot = false;
            } else {
                return Err(EmailError::LocalChar {
                    at: original_pos(&cleaned, j, raw.len()),
                });
            }
            decoded.push(b);
            origins.push(original_pos(&cleaned, j, raw.len()));
        }
    }

    let domain_start = sep + 1;
    let domain_raw_len = n - domain_start;
    let mut domain_end = n;
    let root_position = if domain_raw_len > 0 && cleaned.bytes[n - 1] == b'.' {
        domain_end -= 1;
        Some(original_pos(&cleaned, n - 1, raw.len()))
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: original_pos(&cleaned, domain_start, raw.len()),
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: original_pos(&cleaned, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    let mut j = domain_start;
    while j <= domain_end {
        if j == domain_end || cleaned.bytes[j] == b'.' {
            if label_start == j {
                let at = if j == domain_end && root_position.is_some() {
                    root_position.unwrap()
                } else if j < domain_end {
                    original_pos(&cleaned, j, raw.len())
                } else {
                    original_pos(&cleaned, label_start, raw.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
        j += 1;
    }

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: original_pos(&cleaned, start + 63, raw.len()),
            });
        }
        if cleaned.bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&cleaned, start, raw.len()),
            });
        }
        if cleaned.bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&cleaned, end - 1, raw.len()),
            });
        }
        for k in start..end {
            let b = cleaned.bytes[k];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar {
                    at: original_pos(&cleaned, k, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: original_pos(&cleaned, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if cleaned.bytes[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: original_pos(&cleaned, last_start, raw.len()),
        });
    }

    let local = String::from_utf8(decoded).expect("ASCII validation guarantees UTF-8");
    let domain = String::from_utf8(
        cleaned.bytes[domain_start..domain_end]
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .expect("ASCII validation guarantees UTF-8");

    Ok((
        Address {
            local,
            domain,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }

    if address.local.is_empty() {
        return Err(EmailError::EmptyNormalizedLocal {
            at: origins.first().copied().unwrap_or(input.len()),
        });
    }

    if !is_dot_atom(address.local.as_bytes()) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| {
                        local == &address.local && domain == &address.domain
                    })
                {
                    duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
