```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Vec<usize> = match side {
        Side::Left => (0..row.len()).collect(),
        Side::Right => (0..row.len()).rev().collect(),
    };

    let tiles: Vec<(usize, u32)> = indices
        .into_iter()
        .filter_map(|index| (row[index] != 0).then_some((index, row[index])))
        .collect();

    let mut cells = vec![0; row.len()];
    let mut merges = Vec::new();
    let mut score = 0_u64;
    let mut input_pos = 0;
    let mut output_pos = 0;

    while input_pos < tiles.len() {
        let (index, value) = tiles[input_pos];
        let output_index = match side {
            Side::Left => output_pos,
            Side::Right => row.len() - 1 - output_pos,
        };

        if input_pos + 1 < tiles.len() && tiles[input_pos + 1].1 == value {
            let (second_index, _) = tiles[input_pos + 1];
            let Some(merged_value) = value.checked_mul(2) else {
                return Err(Error::MergeOverflow {
                    first: index,
                    second: second_index,
                });
            };

            cells[output_index] = merged_value;
            merges.push(Merge {
                from: [index, second_index],
                to: output_index,
                value: merged_value,
            });
            score += u64::from(merged_value);
            input_pos += 2;
        } else {
            cells[output_index] = value;
            input_pos += 1;
        }

        output_pos += 1;
    }

    let changed = cells != row;

    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}

fn main() {}
```
