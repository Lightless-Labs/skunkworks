#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct HunkRow {
    prefix: u8,
    payload: String,
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<HunkRow>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty()
        || !s.bytes().all(|b| b.is_ascii_digit())
        || (s.len() > 1 && s.starts_with('0'))
    {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(',')?;
    let (old_count, rest) = rest.split_once(" +")?;
    let (new, rest) = rest.split_once(',')?;
    let new_count = rest.strip_suffix(" @@")?;
    Some((
        parse_number(old)?,
        parse_number(old_count)?,
        parse_number(new)?,
        parse_number(new_count)?,
    ))
}

fn full_start(anchor: usize, count: usize) -> usize {
    if count > 0 { anchor - 1 } else { anchor }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i) != Some(expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() < 3 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end = 0usize;
    let mut net: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        if (b > 0 && a == 0) || (d > 0 && c == 0) {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while i < lines.len() && !lines[i].starts_with("@@") {
            let row = lines[i];
            let bytes = row.as_bytes();
            if bytes.is_empty()
                || !matches!(bytes[0], b' ' | b'-' | b'+')
                || row.as_bytes().contains(&b'\r')
            {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            let payload = &row[1..];
            if payload.as_bytes().contains(&b'\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            match bytes[0] {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                }
                b'-' => old_seen += 1,
                b'+' => new_seen += 1,
                _ => unreachable!(),
            }
            rows.push(HunkRow {
                prefix: bytes[0],
                payload: payload.to_string(),
            });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end
            || (new_start as i128) != (old_start as i128 + net)
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_start + b;
        net += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });

        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_string)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (index, hunk) in hunks.iter().enumerate() {
        let h = index + 1;
        let lead = hunk.rows.iter().take_while(|r| r.prefix == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.prefix == b' ').count();
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_by_start: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=options.fuzz.min(lead) {
            for r in 0..=options.fuzz.min(trail) {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for (ri, row) in hunk.rows.iter().enumerate() {
                    if row.prefix == b' ' || row.prefix == b'-' {
                        let is_dropped_lead = ri < l;
                        let is_dropped_trail = ri >= hunk.rows.len().saturating_sub(r);
                        if !is_dropped_lead && !is_dropped_trail {
                            old_payloads.push(row.payload.as_str());
                        }
                    }
                }

                let b = hunk.old_count;
                if b > working.len() {
                    continue;
                }
                for s in 0..=working.len() - b {
                    if s + l > working.len() || r > working.len().saturating_sub(s + l) {
                        continue;
                    }
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    if s + l + old_payloads.len() != s + b - r {
                        continue;
                    }
                    if old_payloads
                        .iter()
                        .enumerate()
                        .any(|(j, p)| working[s + l + j] != *p)
                    {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, l, r));
                        }
                        Some(old_rank) if rank < old_rank => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, l, r));
                        }
                        Some(old_rank) if rank == old_rank => {
                            best_by_start.push((s, l, r));
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_by_start.is_empty() {
            return Err(ApplyError::NoMatch { hunk: h });
        }
        best_by_start.sort_by_key(|&(s, l, _)| (s, l));
        best_by_start.dedup_by_key(|entry| entry.0);
        if best_by_start.len() > 1 {
            let first = best_by_start[0].0;
            let second = best_by_start[1].0;
            return Err(ApplyError::Ambiguous {
                hunk: h,
                first,
                second,
            });
        }

        let (s, l, r) = best_by_start[0];
        let mut replacement = Vec::new();
        for (ri, row) in hunk.rows.iter().enumerate() {
            let dropped = (ri < l) || (ri >= hunk.rows.len().saturating_sub(r));
            if dropped {
                continue;
            }
            if row.prefix == b' ' || row.prefix == b'+' {
                replacement.push(row.payload.clone());
            }
        }

        let start = s + l;
        let end = s + hunk.old_count - r;
        working.splice(start..end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

fn main() {}
