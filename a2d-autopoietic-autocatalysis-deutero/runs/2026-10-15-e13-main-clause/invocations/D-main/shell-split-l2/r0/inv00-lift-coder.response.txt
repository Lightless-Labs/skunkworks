#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { opening: usize },
    Double { opening: usize },
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_continue(ch: char) -> bool {
    ch.is_ascii_alphanumeric() || ch == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut output = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, ch)) = chars.next() {
        if ch == '\\' {
            if let Some((_, next)) = chars.next() {
                output.push(next);
            }
        } else {
            output.push(ch);
        }
    }

    output
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_open = dollar + 2;
    let mut cursor = after_open;
    let mut closing = None;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        if ch == '\\' {
            cursor += ch.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if ch == '}' {
            closing = Some(cursor);
            break;
        } else {
            cursor += ch.len_utf8();
        }
    }

    let close_pos = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close_pos];

    let name_end = contents
        .char_indices()
        .find_map(|(offset, ch)| {
            if is_name_continue(ch) {
                None
            } else {
                Some(offset)
            }
        })
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    if name.is_empty() || !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &contents[name_end..];
    let selected = if remainder.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((selected, close_pos + 1))
}

fn expand_unbraced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Option<(String, usize)> {
    let start = dollar + 1;
    if start >= input.len() {
        return None;
    }

    let first = input[start..].chars().next()?;
    if !is_name_start(first) {
        return None;
    }

    let mut end = start + first.len_utf8();
    while end < input.len() {
        let ch = input[end..].chars().next().unwrap();
        if !is_name_continue(ch) {
            break;
        }
        end += ch.len_utf8();
    }

    let value = lookup(&input[start..end], env).unwrap_or("").to_string();
    Some((value, end))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + ch.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                cursor = next_cursor;
            }
            Quote::Double { .. } => {
                if ch == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    let after_next = next_cursor + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next);
                        cursor = after_next;
                    } else if next == '\n' {
                        cursor = after_next;
                    } else {
                        word.push('\\');
                        cursor = next_cursor;
                    }
                } else if ch == '$' {
                    active = true;
                    if input[next_cursor..].starts_with('{') {
                        let (value, end) = expand_braced(input, cursor, env)?;
                        word.push_str(&value);
                        cursor = end;
                    } else if let Some((value, end)) = expand_unbraced(input, cursor, env) {
                        word.push_str(&value);
                        cursor = end;
                    } else {
                        word.push('$');
                        cursor = next_cursor;
                    }
                } else {
                    word.push(ch);
                    cursor = next_cursor;
                }
            }
            Quote::None => {
                if is_separator(ch) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next_cursor;
                } else if ch == '\'' {
                    active = true;
                    quote = Quote::Single { opening: cursor };
                    cursor = next_cursor;
                } else if ch == '"' {
                    active = true;
                    quote = Quote::Double { opening: cursor };
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    if next == '\n' {
                        cursor = next_cursor + next.len_utf8();
                    } else {
                        active = true;
                        word.push(next);
                        cursor = next_cursor + next.len_utf8();
                    }
                } else if ch == '$' {
                    active = true;
                    if input[next_cursor..].starts_with('{') {
                        let (value, end) = expand_braced(input, cursor, env)?;
                        word.push_str(&value);
                        cursor = end;
                    } else if let Some((value, end)) = expand_unbraced(input, cursor, env) {
                        word.push_str(&value);
                        cursor = end;
                    } else {
                        word.push('$');
                        cursor = next_cursor;
                    }
                } else {
                    active = true;
                    word.push(ch);
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single { opening } => {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        Quote::Double { opening } => {
            return Err(SplitError::UnterminatedDouble { pos: opening });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn main() {}
