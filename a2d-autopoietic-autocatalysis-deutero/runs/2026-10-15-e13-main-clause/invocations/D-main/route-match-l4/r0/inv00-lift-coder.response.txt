use std::collections::{BTreeSet, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(input: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            let bad = || {
                if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                }
            };
            if i + 2 >= bytes.len() {
                return Err(bad());
            }
            let hi = hex_upper(bytes[i + 1]).ok_or_else(bad)?;
            let lo = hex_upper(bytes[i + 2]).ok_or_else(bad)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(bad());
            }
            decoded.push(value);
            i += 3;
        } else if unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: offset + i }
            } else {
                Error::InvalidPath { at: offset + i }
            });
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = pattern.as_bytes();
    let trailing = pattern.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1;

    while start < end {
        let next = pattern[start..end]
            .find('/')
            .map(|n| start + n)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..next];
        let first = raw.as_bytes()[0];
        let segment = if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' && next != end {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b':' {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            }
        } else {
            Segment::Static(decode_segment(raw, start, true)?)
        };
        segments.push(segment);

        if next == end {
            break;
        }
        start = next + 1;
    }

    Ok((segments, trailing))
}

fn parse_request_path(path: &str) -> Result<(ParsedPath, String), Error> {
    let head_end = path
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(path.len());
    let head = &path[..head_end];

    if !head.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if head == "/" {
        return Ok((
            ParsedPath {
                segments: Vec::new(),
                trailing: false,
            },
            head.to_string(),
        ));
    }

    let bytes = head.as_bytes();
    let trailing = head.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let next = head[start..end]
            .find('/')
            .map(|n| start + n)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(&head[start..next], start, false)?);
        if next == end {
            break;
        }
        start = next + 1;
    }

    Ok((ParsedPath { segments, trailing }, head.to_string()))
}

fn shape_equal(a: &Route, segments: &[Segment]) -> bool {
    a.segments.len() == segments.len()
        && a.segments.iter().zip(segments).all(|(left, right)| match (left, right) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn route_matches(route: &Route, path: &ParsedPath, policy: SlashPolicy) -> Option<Vec<(String, String)>> {
    if policy == SlashPolicy::Strict && route.trailing != path.trailing {
        return None;
    }

    let wildcard = route.has_wildcard;
    let fixed_len = if wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };

    if (!wildcard && path.segments.len() != fixed_len) || (wildcard && path.segments.len() < fixed_len) {
        return None;
    }

    let mut captures = Vec::new();
    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(s) if s != &path.segments[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => captures.push((name.clone(), path.segments[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if wildcard {
        if let Segment::Wildcard(name) = &route.segments[fixed_len] {
            captures.push((
                name.clone(),
                path.segments[fixed_len..].join("/"),
            ));
        }
    }

    Some(captures)
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let has_wildcard = matches!(segments.last(), Some(Segment::Wildcard(_)));
        let mut rank: Vec<u8> = segments
            .iter()
            .map(|s| match s {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            })
            .collect();
        if !has_wildcard {
            rank.push(3);
        }

        for route in &self.routes {
            let slash_conflict = self.policy != SlashPolicy::Strict || route.trailing == trailing;
            if route.method == method && slash_conflict && shape_equal(route, &segments) {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (parsed, head) = parse_request_path(path)?;

        let candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| route_matches(route, &parsed, self.policy).map(|p| (route, p)))
            .collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let selected_method = if method == "HEAD" && candidates.iter().any(|(r, _)| r.method == "HEAD") {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(r, _)| r.method == selected_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let mut winner = eligible[0];
        for candidate in eligible.iter().skip(1) {
            if candidate.0.rank > winner.0.rank {
                winner = candidate;
            }
        }

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.eq("/")
            && winner.0.trailing != parsed.trailing
        {
            let mut target = head.clone();
            if winner.0.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.0.id,
            params: winner.1.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = HashSet::new();
        for (key, _) in params {
            if !supplied.insert(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => capture_names.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &capture_names {
            if !supplied.contains(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let capture_set: HashSet<&str> = capture_names.iter().copied().collect();
        let extra = params
            .iter()
            .map(|(key, _)| *key)
            .filter(|key| !capture_set.contains(key))
            .min();
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: key.to_string(),
            });
        }

        let value_for = |key: &str| -> &str {
            params.iter().find(|(k, _)| *k == key).map(|(_, v)| *v).unwrap()
        };

        for segment in &route.segments {
            match segment {
                Segment::Static(_) => {}
                Segment::Param(name) => {
                    let value = value_for(name);
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = value_for(name);
                    let valid_ascii = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
                    let valid_parts = value.is_empty()
                        || value
                            .split('/')
                            .all(|part| !part.is_empty());
                    if !valid_ascii || !valid_parts {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if !first {
                output.push('/');
            }
            first = false;

            match segment {
                Segment::Static(value) => output.push_str(&encode_segment(value)),
                Segment::Param(name) => output.push_str(&encode_segment(value_for(name))),
                Segment::Wildcard(name) => {
                    let value = value_for(name);
                    if !value.is_empty() {
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(part));
                        }
                    }
                }
            }
        }

        if route.segments.is_empty() {
            return Ok("/".to_string());
        }
        if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn main() {}
