#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|n| n as u8)
}

fn encode_group(bytes: &[u8]) -> String {
    let a = bytes[0];
    let b = bytes.get(1).copied().unwrap_or(0);
    let c = bytes.get(2).copied().unwrap_or(0);

    let mut out = String::with_capacity(4);
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() >= 2 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() >= 3 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    for chunk in input.chunks(3) {
        out.push_str(&encode_group(chunk));
    }
    out
}

#[derive(Clone, Copy)]
struct Symbol {
    value: u8,
    offset: usize,
}

fn decode_group(group: &[Symbol; 4]) -> Vec<u8> {
    let a = group[0].value;
    let b = group[1].value;
    let c = group[2].value;
    let d = group[3].value;

    let mut out = vec![(a << 2) | (b >> 4)];
    if c != 64 {
        out.push((b << 4) | (c >> 2));
    }
    if d != 64 {
        out.push((c << 6) | d);
    }
    out
}

fn decode_buffered_group(group: &[Symbol]) -> Vec<u8> {
    let array = [group[0], group[1], group[2], group[3]];
    decode_group(&array)
}

pub struct Encoder {
    buffer: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            buffer: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.buffer.len() + input.len());
        data.extend_from_slice(&self.buffer);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut out = String::with_capacity(complete_len / 3 * 4);
        for chunk in data[..complete_len].chunks(3) {
            out.push_str(&encode_group(chunk));
        }

        self.buffer.clear();
        self.buffer.extend_from_slice(&data[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        if self.buffer.is_empty() {
            Ok(String::new())
        } else {
            let out = encode_group(&self.buffer);
            self.buffer.clear();
            Ok(out)
        }
    }
}

pub struct Decoder {
    group: Vec<Symbol>,
    absolute_offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            absolute_offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

        for &byte in input {
            let offset = self.absolute_offset;
            self.absolute_offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset });
            }

            if self.pending_second_padding.is_some() {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset });
                }
                self.pending_second_padding = None;
                self.group.push(Symbol { value: 64, offset });
                out.extend(decode_buffered_group(&self.group));
                self.group.clear();
                self.terminal_padding = true;
                continue;
            }

            let position = self.group.len();
            if byte == b'=' {
                match position {
                    0 | 1 => return self.fail(Error::BadPadding { offset }),
                    2 => {
                        self.pending_second_padding = Some(offset);
                        self.group.push(Symbol { value: 64, offset });
                    }
                    3 => {
                        self.group.push(Symbol { value: 64, offset });
                        out.extend(decode_buffered_group(&self.group));
                        self.group.clear();
                        self.terminal_padding = true;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let Some(value) = digit(byte) else {
                return self.fail(Error::InvalidByte { offset, byte });
            };

            self.group.push(Symbol { value, offset });
            if self.group.len() == 4 {
                out.extend(decode_buffered_group(&self.group));
                self.group.clear();
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            return self.fail(Error::Truncated {
                offset: self.group[0].offset,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

fn main() {}
