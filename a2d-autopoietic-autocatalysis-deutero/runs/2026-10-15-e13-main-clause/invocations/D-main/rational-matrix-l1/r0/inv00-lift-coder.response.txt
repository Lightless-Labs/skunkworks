```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }

        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    fn from_wide_parts(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128);
        let n_abs = n.unsigned_abs() / g;
        let d_abs = d as u128 / g;

        if d_abs > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }

        let n_reduced = if n < 0 {
            if n_abs == (1u128 << 63) {
                i64::MIN
            } else if n_abs <= i64::MAX as u128 {
                -(n_abs as i64)
            } else {
                return Err(RationalError::Overflow);
            }
        } else if n_abs <= i64::MAX as u128 {
            n_abs as i64
        } else {
            return Err(RationalError::Overflow);
        };

        Ok(Self {
            num: n_reduced,
            den: d_abs as i64,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let n = num as i128;
        let d = den as i128;
        let (n, d) = if d < 0 { (-n, -d) } else { (n, d) };

        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128);
        let n_abs = n.unsigned_abs() / g;
        let d_abs = d as u128 / g;

        if d_abs > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }

        let reduced_num = if n < 0 {
            if n_abs == (1u128 << 63) {
                i64::MIN
            } else if n_abs <= i64::MAX as u128 {
                -(n_abs as i64)
            } else {
                return Err(RationalError::Overflow);
            }
        } else if n_abs <= i64::MAX as u128 {
            n_abs as i64
        } else {
            return Err(RationalError::Overflow);
        };

        Ok(Self {
            num: reduced_num,
            den: d_abs as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide_parts(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide_parts(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide_parts(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        Self::from_wide_parts(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse {
                        at: i,
                        kind: ParseKind::InvalidByte,
                    });
                }
                slash = Some(i);
            } else if !(b.is_ascii_digit() || (b == b'-' && i == 0)) {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');
        if num_start == num_end {
            return Err(RationalError::Parse {
                at: bytes.len(),
                kind: ParseKind::InvalidByte,
            });
        }

        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse {
                    at: bytes.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }

        let negative = bytes[0] == b'-';
        if negative && num_digits == b"0" {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        if let Some(s) = slash {
            let den_digits = &bytes[s + 1..];
            if den_digits.len() >= 2 && den_digits[0] == b'0' {
                return Err(RationalError::Parse {
                    at: s + 2,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let num_mag = parse_magnitude(num_digits).ok_or(RationalError::Parse {
            at: 0,
            kind: ParseKind::OutOfRange,
        })?;
        let num_limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag > num_limit {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::OutOfRange,
            });
        }

        let den_mag = if let Some(s) = slash {
            let digits = &bytes[s + 1..];
            let magnitude = parse_magnitude(digits).ok_or(RationalError::Parse {
                at: s + 1,
                kind: ParseKind::OutOfRange,
            })?;
            if magnitude > i64::MAX as u128 {
                return Err(RationalError::Parse {
                    at: s + 1,
                    kind: ParseKind::OutOfRange,
                });
            }
            Some(magnitude as u64)
        } else {
            None
        };

        if let Some(d) = den_mag {
            if d == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: ParseKind::ZeroDenominator,
                });
            }
            if d == 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if num_mag == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(num_mag, d as u128) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        match den_mag {
            Some(d) => Ok(Self {
                num,
                den: d as i64,
            }),
            None => Ok(Self { num, den: 1 }),
        }
    }
}

fn parse_magnitude(digits: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in digits {
        let digit = (b - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn main() {}
```
