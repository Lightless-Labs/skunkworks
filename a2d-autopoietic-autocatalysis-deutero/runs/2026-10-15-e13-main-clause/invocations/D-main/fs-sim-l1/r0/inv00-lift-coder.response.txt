```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: std::collections::HashMap<String, Entry>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::HashMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries }
    }

    fn canonicalize(path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                let component = &path[start..i];
                if component.is_empty() || component == "." || component == ".." {
                    return Err(Error::InvalidPath(if component.is_empty() {
                        i.saturating_sub(1)
                    } else {
                        start
                    }));
                }
                components.push(component.to_string());
                start = i + 1;
            }
        }

        Ok(components)
    }

    fn path_for(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn lookup(&self, components: &[String]) -> Result<&Entry, Error> {
        let mut prefix = Vec::new();
        let mut current = self.entries.get("/").expect("root exists");

        if components.is_empty() {
            return Ok(current);
        }

        for (i, component) in components.iter().enumerate() {
            if !matches!(current, Entry::Dir) {
                return Err(Error::NotDir(Self::path_for(&prefix)));
            }

            prefix.push(component.clone());
            let canonical = Self::path_for(&prefix);
            current = self
                .entries
                .get(&canonical)
                .ok_or_else(|| Error::Missing(canonical.clone()))?;

            if i + 1 < components.len() && !matches!(current, Entry::Dir) {
                return Err(Error::NotDir(canonical));
            }
        }

        Ok(current)
    }

    fn parent_components(components: &[String]) -> &[String] {
        &components[..components.len().saturating_sub(1)]
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::canonicalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let parent = Self::parent_components(&components);
        match self.lookup(parent)? {
            Entry::Dir => {}
            Entry::File(_) => return Err(Error::NotDir(Self::path_for(parent))),
        }

        let canonical = Self::path_for(&components);
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::canonicalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let parent = Self::parent_components(&components);
        match self.lookup(parent)? {
            Entry::Dir => {}
            Entry::File(_) => return Err(Error::NotDir(Self::path_for(parent))),
        }

        let canonical = Self::path_for(&components);
        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::canonicalize(path)?;
        match self.lookup(&components)? {
            Entry::Dir => Err(Error::IsDir(Self::path_for(&components))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::canonicalize(path)?;
        match self.lookup(&components)? {
            Entry::File(_) => return Err(Error::NotDir(Self::path_for(&components))),
            Entry::Dir => {}
        }

        let parent = Self::path_for(&components);
        let mut children = Vec::new();
        for (stored_path, entry) in &self.entries {
            if stored_path == "/" {
                continue;
            }
            let Some((stored_parent, basename)) = stored_path.rsplit_once('/') else {
                continue;
            };
            let stored_parent = if stored_parent.is_empty() {
                "/"
            } else {
                stored_parent
            };
            if stored_parent == parent {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                children.push((basename.to_string(), kind));
            }
        }

        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }
}

fn main() {}
```
