use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }

    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    pos: &mut usize,
    terminator: Option<u8>,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();

    loop {
        let start = *pos;
        while *pos < bytes.len() && bytes[*pos] != b'.' && Some(bytes[*pos]) != terminator {
            if !is_ident_char(bytes[*pos]) {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let ident = &s[start..*pos];
        if prerelease
            && ident.len() > 1
            && ident.as_bytes().iter().all(|b| is_digit(*b))
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }
        result.push(ident.to_string());

        if *pos >= bytes.len() || Some(bytes[*pos]) == terminator {
            break;
        }
        *pos += 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    if pos < s.len() && !matches!(s.as_bytes()[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        pos += 1;
        pre = parse_identifiers(s, &mut pos, Some(b'+'), true)?;
    }

    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, &mut pos, None, false)?;
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_ident(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric_ident(x);
        let yn = is_numeric_ident(y);
        let ord = match (xn, yn) {
            (true, true) => x
                .len()
                .cmp(&y.len())
                .then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    a.major
        .cmp(&b.major)
        .then_with(|| a.minor.cmp(&b.minor))
        .then_with(|| a.patch.cmp(&b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Bound {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    bounds: Vec<Bound>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    while p < end {
        while p < end && whitespace(b[p]) {
            p += 1;
        }
        if p == end {
            break;
        }
        let st = p;
        while p < end && !whitespace(b[p]) {
            p += 1;
        }
        out.push((st, &s[st..p]));
    }
    out
}

fn version_operand(operand: &str, start: usize) -> Result<Version, RangeError> {
    parse_version(operand).map_err(|error| RangeError::Version { at: start, error })
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn version_from_core(c: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: c.0,
        minor: c.1,
        patch: c.2,
        pre,
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields(Vec<u64>),
}

fn parse_partial(token: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = token.contains('-') || token.contains('+');
    let fields: Vec<&str> = token.split('.').collect();
    let all_numeric_full_shape = fields.len() >= 3
        && !fields
            .iter()
            .any(|f| matches!(*f, "*" | "x" | "X"));

    if has_suffix || all_numeric_full_shape {
        return version_operand(token, at).map(Partial::Full);
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.as_bytes().iter().all(|b| is_digit(*b))
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let value = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(value);
        }
    }
    Ok(Partial::Fields(nums))
}

fn increment_upper(fields: &[u64], at: usize, mode: &str) -> Result<Version, RangeError> {
    let k = fields.len();
    let mut c = [0u64; 3];
    for (i, v) in fields.iter().enumerate() {
        c[i] = *v;
    }
    let idx = match mode {
        "bare" => {
            if k == 1 {
                Some(0)
            } else if k == 2 {
                Some(1)
            } else {
                None
            }
        }
        "tilde" => {
            if k == 1 {
                Some(0)
            } else {
                Some(1)
            }
        }
        "caret" => {
            if c[0] > 0 || k == 1 {
                Some(0)
            } else if c[1] > 0 || k == 2 {
                Some(1)
            } else {
                Some(2)
            }
        }
        "hyphen" => {
            if k == 1 {
                Some(0)
            } else if k == 2 {
                Some(1)
            } else {
                None
            }
        }
        _ => None,
    };
    if let Some(i) = idx {
        c[i] = c[i].checked_add(1).ok_or(RangeError::Syntax { at })?;
        for x in i + 1..3 {
            c[x] = 0;
        }
    }
    Ok(version_from_core((c[0], c[1], c[2]), vec!["0".into()]))
}

fn padded(fields: &[u64]) -> Version {
    let mut c = [0u64; 3];
    for (i, v) in fields.iter().enumerate() {
        c[i] = *v;
    }
    version_from_core((c[0], c[1], c[2]), Vec::new())
}

fn add_bound(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core_tuple(&version));
    }
    arm.bounds.push(Bound { op, version });
}

fn parse_hyphen_endpoint(
    token: &str,
    at: usize,
) -> Result<(Version, usize, bool), RangeError> {
    match parse_partial(token, at)? {
        Partial::Full(v) => Ok((v, 3, true)),
        Partial::Fields(fields) => {
            let k = fields.len();
            Ok((padded(&fields), k, false))
        }
    }
}

fn parse_arm(ts: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        bounds: Vec::new(),
        anchors: Vec::new(),
    };
    if ts.is_empty() {
        return Ok(arm);
    }

    let hyphen = ts.iter().position(|(_, t)| *t == "-");
    if let Some(i) = hyphen {
        if i != 1 || ts.len() != 3 {
            return Err(RangeError::Syntax { at: ts[i].0 });
        }
        let (lo, lk, _) = parse_hyphen_endpoint(ts[0].1, ts[0].0)?;
        let (up, uk, full_upper) = parse_hyphen_endpoint(ts[2].1, ts[2].0)?;
        if lk == 0 {
            return Err(RangeError::Syntax { at: ts[0].0 });
        }
        if uk == 0 {
            return Err(RangeError::Syntax { at: ts[2].0 });
        }

        add_bound(&mut arm, Op::Ge, lo, true);
        if full_upper {
            add_bound(&mut arm, Op::Le, up, true);
        } else {
            let fields = if uk == 1 {
                vec![up.major]
            } else {
                vec![up.major, up.minor]
            };
            let upper = increment_upper(&fields, ts[2].0, "hyphen")?;
            add_bound(&mut arm, Op::Lt, upper, false);
        }
        return Ok(arm);
    }

    for &(at, token) in ts {
        let (prefix, body, body_at) = if let Some(rest) = token.strip_prefix('^') {
            ("caret", rest, at + 1)
        } else if let Some(rest) = token.strip_prefix('~') {
            ("tilde", rest, at + 1)
        } else {
            ("", token, at)
        };

        if !prefix.is_empty() {
            let parsed = parse_partial(body, body_at)?;
            let (lower, fields) = match &parsed {
                Partial::Full(v) => (v.clone(), vec![v.major, v.minor, v.patch]),
                Partial::Fields(f) => (padded(f), f.clone()),
            };
            if fields.is_empty() {
                return Err(RangeError::Syntax { at });
            }
            let upper = increment_upper(&fields, body_at, prefix)?;
            add_bound(&mut arm, Op::Ge, lower, true);
            add_bound(&mut arm, Op::Lt, upper, false);
            continue;
        }

        let first = token.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            let (op, n) = if token.starts_with(">=") {
                (Op::Ge, 2)
            } else if token.starts_with("<=") {
                (Op::Le, 2)
            } else if token.starts_with('>') {
                (Op::Gt, 1)
            } else if token.starts_with('<') {
                (Op::Lt, 1)
            } else {
                (Op::Eq, 1)
            };
            let operand = &token[n..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            let v = version_operand(operand, at + n)?;
            add_bound(&mut arm, op, v, true);
            continue;
        }

        if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at });
        }

        match parse_partial(token, at)? {
            Partial::Full(v) => add_bound(&mut arm, Op::Eq, v, true),
            Partial::Fields(fields) => {
                let k = fields.len();
                if k == 0 {
                    continue;
                }
                let lower = padded(&fields);
                add_bound(&mut arm, Op::Ge, lower, true);
                if k < 3 {
                    let upper = increment_upper(&fields, at, "bare")?;
                    add_bound(&mut arm, Op::Lt, upper, false);
                }
            }
        }
    }
    Ok(arm)
}

fn split_arms(s: &str) -> Result<Vec<(usize, usize)>, RangeError> {
    let b = s.as_bytes();
    let mut arms = Vec::new();
    let mut p = 0;
    let mut arm_start = 0;
    let mut nonempty = false;

    while p < b.len() {
        if whitespace(b[p]) {
            p += 1;
            continue;
        }
        let token_start = p;
        while p < b.len() && !whitespace(b[p]) {
            p += 1;
        }
        if &s[token_start..p] == "||" {
            if !nonempty {
                return Err(RangeError::Empty { at: token_start });
            }
            arms.push((arm_start, token_start));
            nonempty = false;
            arm_start = p;
        } else {
            nonempty = true;
        }
    }

    if !nonempty {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push((arm_start, s.len()));
    Ok(arms)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let slices = split_arms(s)?;
    let mut arms = Vec::new();

    for (start, end) in slices {
        let ts = tokens(s, start, end);
        arms.push(parse_arm(&ts)?);
    }
    Ok(Range { arms })
}

fn arm_admits(v: &Version, arm: &Arm) -> bool {
    v.pre.is_empty() || arm.anchors.iter().any(|x| *x == core_tuple(v))
}

fn compare_bound(v: &Version, b: &Bound) -> bool {
    match b.op {
        Op::Eq => compare_versions(v, &b.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &b.version) == Ordering::Less,
        Op::Le => compare_versions(v, &b.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &b.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &b.version) != Ordering::Less,
    }
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| {
        (include_prerelease || arm_admits(v, arm))
            && arm.bounds.iter().all(|b| compare_bound(v, b))
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, raw) in candidates.iter().enumerate() {
        match parse_version(raw) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (greatest && ord == Ordering::Greater)
                    || (!greatest && ord == Ordering::Less)
                {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn has_stable_witness(bounds: &[Bound]) -> bool {
    let candidate = Version {
        major: 0,
        minor: 0,
        patch: 0,
        pre: Vec::new(),
        build: Vec::new(),
    };
    bounds.iter().all(|b| compare_bound(&candidate, b))
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut bounds = aa.bounds.clone();
            bounds.extend(bb.bounds.clone());

            if has_stable_witness(&bounds) {
                return true;
            }
            if include_prerelease {
                let mut anchors = aa.anchors.clone();
                anchors.extend(bb.anchors.clone());
                for (major, minor, patch) in anchors {
                    let v = version_from_core((major, minor, patch), vec!["0".into()]);
                    if bounds.iter().all(|x| compare_bound(&v, x)) {
                        return true;
                    }
                }
            } else {
                for tuple in &aa.anchors {
                    if bb.anchors.contains(tuple) {
                        let v = version_from_core(*tuple, vec!["0".into()]);
                        if bounds.iter().all(|x| compare_bound(&v, x)) {
                            return true;
                        }
                    }
                }
            }
        }
    }
    false
}

fn version_text(v: &Version) -> String {
    let mut out = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        out.push('-');
        out.push_str(&v.pre.join("."));
    }
    out
}

fn strongest_bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;

    for b in &arm.bounds {
        match b.op {
            Op::Eq => {
                let candidate = (b.version.clone(), true);
                if lower.as_ref().map_or(true, |x| {
                    compare_versions(&candidate.0, &x.0) == Ordering::Greater
                }) {
                    lower = Some(candidate.clone());
                }
                if upper.as_ref().map_or(true, |x| {
                    compare_versions(&candidate.0, &x.0) == Ordering::Less
                }) {
                    upper = Some(candidate);
                }
                continue;
            }
            Op::Gt | Op::Ge => {
                let inclusive = matches!(b.op, Op::Ge);
                let replace = match &lower {
                    None => true,
                    Some((v, inc)) => {
                        let ord = compare_versions(&b.version, v);
                        ord == Ordering::Greater || (ord == Ordering::Equal && !inclusive && *inc)
                    }
                };
                if replace {
                    lower = Some((b.version.clone(), inclusive));
                }
            }
            Op::Lt | Op::Le => {
                let inclusive = matches!(b.op, Op::Le);
                let replace = match &upper {
                    None => true,
                    Some((v, inc)) => {
                        let ord = compare_versions(&b.version, v);
                        ord == Ordering::Less || (ord == Ordering::Equal && !inclusive && *inc)
                    }
                };
                if replace {
                    upper = Some((b.version.clone(), inclusive));
                }
            }
        }
    }
    (lower, upper)
}

pub fn canonical(r: &Range) -> String {
    let mut arms_out = Vec::new();
    for arm in &r.arms {
        if !has_stable_witness(&arm.bounds) {
            continue;
        }
        let (lower, upper) = strongest_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(x, y, z)| format!("{x}.{y}.{z}"))
                .collect::<Vec<_>>()
                .join(",")
        };
        let lo = lower.map_or("*".to_string(), |(v, inclusive)| {
            format!("{}{}", if inclusive { ">=" } else { ">" }, version_text(&v))
        });
        let hi = upper.map_or("*".to_string(), |(v, inclusive)| {
            format!("{}{}", if inclusive { "<=" } else { "<" }, version_text(&v))
        });
        arms_out.push(format!("[{lo};{hi};pre={anchor_text}]"));
    }
    arms_out.sort();
    arms_out.dedup();
    if arms_out.is_empty() {
        "EMPTY".to_string()
    } else {
        arms_out.join(" || ")
    }
}
