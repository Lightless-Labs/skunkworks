#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

#[derive(Clone, Copy)]
struct Domain {
    min: u8,
    max: u8,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

const DOMAINS: [Domain; 5] = [
    Domain { min: 0, max: 59 },
    Domain { min: 0, max: 23 },
    Domain { min: 1, max: 31 },
    Domain { min: 1, max: 12 },
    Domain { min: 0, max: 6 },
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    domain: Domain,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if value < domain.min as u32 || value > domain.max as u32 {
        return Err(ParseError::OutOfRange {
            field,
            offset: start,
        });
    }
    Ok(value)
}

fn parse_step(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    domain: Domain,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if value == 0 || value > (domain.max - domain.min + 1) as u32 {
        return Err(ParseError::BadStep {
            field,
            offset: start,
        });
    }
    Ok(value)
}

fn parse_field(input: &str, field: Field, domain: Domain) -> Result<Vec<bool>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; domain.max as usize + 1];
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base_end, step_bounds) = if let Some(&slash) = slash_positions.first() {
            (slash, Some((item_start + slash + 1, item_end)))
        } else {
            (item_end, None)
        };

        if base_end == item_start {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let base = &bytes[item_start..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            0
        } else if hyphens.is_empty() {
            if base.iter().all(u8::is_ascii_digit) {
                1
            } else {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen == 0
                || hyphen + 1 == base.len()
                || !base[..hyphen].iter().all(u8::is_ascii_digit)
                || !base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            2
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        let step = if let Some((step_start, step_end)) = step_bounds {
            if step_start == step_end
                || !bytes[step_start..step_end]
                    .iter()
                    .all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Some(parse_step(
                bytes, step_start, step_end, field, domain,
            )?)
        } else {
            None
        };

        let (start_value, end_value) = match base_kind {
            0 => (domain.min as u32, domain.max as u32),
            1 => {
                let value = parse_number(
                    bytes,
                    item_start,
                    base_end,
                    field,
                    domain,
                )?;
                (
                    value,
                    if step.is_some() {
                        domain.max as u32
                    } else {
                        value
                    },
                )
            }
            _ => {
                let hyphen = hyphens[0];
                let start = parse_number(
                    bytes,
                    item_start,
                    item_start + hyphen,
                    field,
                    domain,
                )?;
                let end_start = item_start + hyphen + 1;
                let end = parse_number(bytes, end_start, base_end, field, domain)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_start,
                    });
                }
                (start, end)
            }
        };

        let increment = step.unwrap_or(1);
        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount {
            found: parts.len(),
        });
    }

    let mut selected = Vec::with_capacity(5);
    for i in 0..5 {
        selected.push(parse_field(parts[i], FIELDS[i], DOMAINS[i])?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        (0..self.selected[index].len())
            .filter(|&value| self.selected[index][value])
            .map(|value| value as u8)
            .collect()
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    // 2000-01-01 is Saturday (6).
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if is_leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += at.day as u32 - 1;
    ((6 + days) % 7) as u8
}

impl Cron {
    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let dow = weekday(at);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][dow as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            current = if current.minute < 59 {
                CivilDateTime {
                    minute: current.minute + 1,
                    ..current
                }
            } else if current.hour < 23 {
                CivilDateTime {
                    minute: 0,
                    hour: current.hour + 1,
                    ..current
                }
            } else if current.day < days_in_month(current.year, current.month) {
                CivilDateTime {
                    minute: 0,
                    hour: 0,
                    day: current.day + 1,
                    ..current
                }
            } else if current.month < 12 {
                CivilDateTime {
                    minute: 0,
                    hour: 0,
                    day: 1,
                    month: current.month + 1,
                    ..current
                }
            } else {
                CivilDateTime {
                    year: current.year + 1,
                    month: 1,
                    day: 1,
                    hour: 0,
                    minute: 0,
                }
            };

            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn main() {}
