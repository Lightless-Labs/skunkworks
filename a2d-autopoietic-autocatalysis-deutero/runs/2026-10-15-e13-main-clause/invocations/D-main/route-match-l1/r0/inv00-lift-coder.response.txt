#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
            })
            .collect();

        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, ch)| matches!(ch, '?' | '#').then_some(i))
            .unwrap_or(path.len());
        let segments = parse_path(&path[..head_end])?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if route.segments.len() != segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_segment, path_segment) in route.segments.iter().zip(&segments) {
                match pattern_segment {
                    Segment::Static(value) if value == path_segment => {}
                    Segment::Param(name) => params.push((name.clone(), path_segment.clone())),
                    _ => {
                        matches = false;
                        break;
                    }
                }
            }

            if matches
                && best
                    .as_ref()
                    .is_none_or(|(current, _)| route.rank > current.rank)
            {
                best = Some((route, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    parse_segments(input, true).map_err(|at| Error::InvalidPattern { at })
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    parse_segments(input, false)
        .map(|segments| {
            segments
                .into_iter()
                .map(|segment| match segment {
                    Segment::Static(value) => value,
                    Segment::Param(_) => unreachable!(),
                })
                .collect()
        })
        .map_err(|at| Error::InvalidPath { at })
}

fn parse_segments(input: &str, pattern: bool) -> Result<Vec<Segment>, usize> {
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut parts: Vec<(usize, usize)> = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            parts.push((start, i));
            start = i + 1;
        }
    }

    let trailing_slash = start == bytes.len();
    if !trailing_slash {
        parts.push((start, bytes.len()));
    }

    let mut result = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (start, end) in parts {
        let segment = &input[start..end];
        if pattern && segment.starts_with(':') {
            let name = &segment[1..];
            let name_bytes = name.as_bytes();
            let valid = !name_bytes.is_empty()
                && name_bytes[0].is_ascii_lowercase()
                && name_bytes[1..]
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');

            if !valid || !names.insert(name.to_owned()) {
                return Err(start);
            }
            result.push(Segment::Param(name.to_owned()));
        } else {
            if pattern && segment.starts_with('*') {
                return Err(start);
            }
            let decoded = decode_static(segment, start)?;
            result.push(Segment::Static(decoded));
        }
    }

    if trailing_slash {
        return Err(bytes.len() - 1);
    }

    Ok(result)
}

fn decode_static(segment: &str, offset: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = uppercase_hex(bytes[i + 1]).ok_or(offset + i)?;
            let lo = uppercase_hex(bytes[i + 2]).ok_or(offset + i)?;
            let value = hi * 16 + lo;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }

            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }

    String::from_utf8(decoded).map_err(|_| offset)
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn main() {}
