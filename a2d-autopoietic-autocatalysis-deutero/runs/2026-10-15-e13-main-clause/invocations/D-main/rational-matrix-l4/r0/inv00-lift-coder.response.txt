use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        rational_from_i128(a + b, self.den as i128 * rhs.den as i128)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        rational_from_i128(a - b, self.den as i128 * rhs.den as i128)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        if n < i64::MIN as i128 || n > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Rational { num: n as i64, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }

        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            let slash = i;
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            let _ = slash;
        }

        let num_end = den_start.map_or(bytes.len(), |s| s - 1);
        let num_digits_start = if negative { 1 } else { 0 };
        let num_digits = num_end - num_digits_start;

        if num_digits >= 2 && bytes[num_digits_start] == b'0' {
            return Err(RationalError::Parse {
                at: num_digits_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if negative && num_digits == 1 && bytes[1] == b'0' {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::NonCanonical });
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(RationalError::Parse { at: ds + 1, kind: ParseKind::NonCanonical });
            }
        }

        let mut magnitude: u128 = 0;
        for &b in &bytes[num_digits_start..num_end] {
            magnitude = magnitude.saturating_mul(10).saturating_add((b - b'0') as u128);
        }
        let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if magnitude > limit {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::OutOfRange });
        }
        let num = if negative {
            if magnitude == 1u128 << 63 { i64::MIN } else { -(magnitude as i64) }
        } else {
            magnitude as i64
        };

        let mut den = 1i64;
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            let mut dm: u128 = 0;
            for &b in &bytes[ds..de] {
                dm = dm.saturating_mul(10).saturating_add((b - b'0') as u128);
            }
            if dm > i64::MAX as u128 {
                return Err(RationalError::Parse { at: ds, kind: ParseKind::OutOfRange });
            }
            if dm == 0 {
                return Err(RationalError::Parse { at: ds, kind: ParseKind::ZeroDenominator });
            }
            den = dm as i64;

            let slash = ds - 1;
            if den == 1 || num == 0 {
                return Err(RationalError::Parse { at: slash, kind: ParseKind::NonCanonical });
            }
            if gcd_u128(num.unsigned_abs() as u128, den as u128) > 1 {
                return Err(RationalError::Parse { at: slash, kind: ParseKind::NonCanonical });
            }
        }
        Ok(Rational { num, den })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

fn zero() -> Rational {
    Rational { num: 0, den: 1 }
}

fn one() -> Rational {
    Rational { num: 1, den: 1 }
}

fn checked_area(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = checked_area(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let len = self.data.len();
        let mut data = vec![zero(); len];
        if self.rows != 0 && self.cols != 0 {
            for r in 0..self.rows {
                for c in 0..self.cols {
                    data[c * self.rows + r] = self.data[r * self.cols + c];
                }
            }
        }
        Matrix { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = checked_area(self.rows, rhs.cols)?;
        let mut data = Vec::with_capacity(out_len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero();
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Matrix { rows: self.rows, cols: rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = checked_area(n, width)?;
        let mut data = vec![zero(); total];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
                data[r * width + n + c] = if r == c { one() } else { zero() };
            }
        }
        let augmented = Matrix { rows: n, cols: width, data };
        let reduction = eliminate(augmented, n, width)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular { rank: reduction.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                out.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let total = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = vec![zero(); total];
        for r in 0..self.rows {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            for c in 0..k {
                data[r * width + n + c] = rhs.data[r * k + c];
            }
        }
        let augmented = Matrix { rows: self.rows, cols: width, data };
        let reduction = eliminate(augmented, n, width).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..n).all(|c| reduction.matrix.data[r * width + c].num == 0);
            if coeff_zero {
                for j in 0..k {
                    if reduction.matrix.data[r * width + n + j].num != 0 {
                        let witness = reduction.matrix.data[r * width..(r + 1) * width].to_vec();
                        return Ok(Solution::None { row: r, rhs_col: j, witness });
                    }
                }
            }
        }

        let rank = reduction.pivots.len();
        let mut free = Vec::new();
        let mut is_pivot = vec![false; n];
        for &p in &reduction.pivots {
            is_pivot[p] = true;
        }
        for c in 0..n {
            if !is_pivot[c] {
                free.push(c);
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![zero(); particular_len];
        for (j, &pivot_col) in reduction.pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular_data[pivot_col * k + rhs_col] =
                    reduction.matrix.data[j * width + n + rhs_col];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular_data };

        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: reduction.pivots });
        }

        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![zero(); basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + j] = one();
            for (i, &pivot_col) in reduction.pivots.iter().enumerate() {
                let coeff = reduction.matrix.data[i * width + free_col];
                let neg = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col, col: j, step: rank, source,
                    })
                })?;
                basis_data[pivot_col * f + j] = neg;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: f, data: basis_data },
            pivots: reduction.pivots,
        })
    }
}

fn eliminate(mut matrix: Matrix, pivot_limit: usize, update_cols: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let cols = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..pivot_limit {
        if p == rows {
            break;
        }
        let mut chosen: Option<usize> = None;
        for r in p..rows {
            let value = matrix.data[r * cols + col];
            if value.num == 0 {
                continue;
            }
            let key = (value.den, value.num.unsigned_abs(), r);
            if chosen.map_or(true, |q| {
                let v = matrix.data[q * cols + col];
                key < (v.den, v.num.unsigned_abs(), q)
            }) {
                chosen = Some(r);
            }
        }
        let Some(q) = chosen else { continue };

        if q != p {
            for c in 0..update_cols {
                matrix.data.swap(p * cols + c, q * cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + col];
        for c in 0..update_cols {
            let original = matrix.data[p * cols + c];
            matrix.data[p * cols + c] = original.div(&pivot).map_err(|source| {
                MatrixError::Arithmetic { row: p, col: c, step: p, source }
            })?;
        }

        for r in 0..rows {
            if r == p {
                continue;
            }
            let factor = matrix.data[r * cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..update_cols {
                let pivot_entry = matrix.data[p * cols + c];
                let product = factor.mul(&pivot_entry).map_err(|source| {
                    MatrixError::Arithmetic { row: r, col: c, step: p, source }
                })?;
                let current = matrix.data[r * cols + c];
                matrix.data[r * cols + c] = current.sub(&product).map_err(|source| {
                    MatrixError::Arithmetic { row: r, col: c, step: p, source }
                })?;
            }
        }
        pivots.push(col);
        p += 1;
    }

    Ok(Reduction { matrix, pivots, swaps })
}
