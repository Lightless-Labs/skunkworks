```rust
use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("digits", &self.digits)
            .finish()
    }
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let magnitude = &bytes[start..];

        // Invalid bytes take precedence over malformed separators.
        if let Some(offset) = magnitude
            .iter()
            .position(|&b| b != b'_' && !b.is_ascii_digit())
        {
            return Err(Error::InvalidDigit {
                index: start + offset,
            });
        }

        // Validate underscores after the full invalid-byte scan.
        for (i, &byte) in magnitude.iter().enumerate() {
            if byte == b'_'
                && (i == 0
                    || i + 1 == magnitude.len()
                    || !magnitude[i - 1].is_ascii_digit()
                    || !magnitude[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: start + i });
            }
        }

        let digit_count = magnitude.iter().filter(|&&b| b != b'_').count();
        let first_digit_index = magnitude
            .iter()
            .position(|&b| b != b'_')
            .expect("a nonempty magnitude has a digit after validation");

        if digit_count > 1 && magnitude[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: start + first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess = max_digits;
            let excess_offset = magnitude
                .iter()
                .enumerate()
                .filter(|(_, b)| **b != b'_')
                .nth(excess)
                .map(|(i, _)| i)
                .expect("digit count exceeds limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: start + excess_offset,
            });
        }

        let mut digits: Vec<u8> = magnitude
            .iter()
            .filter(|&&b| b != b'_')
            .rev()
            .map(|&b| b - b'0')
            .collect();

        if digits.is_empty() {
            digits.push(0);
        }
        let is_zero = digits.len() == 1 && digits[0] == 0;

        Ok(Self {
            negative: negative && !is_zero,
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(min_digits).max(1);
        let pad = width - self.digits.len();
        let mut magnitude = String::with_capacity(width + width / group.max(1));

        for _ in 0..pad {
            magnitude.push('0');
        }
        for &digit in self.digits.iter().rev() {
            magnitude.push(char::from(b'0' + digit));
        }

        let mut output = String::with_capacity(
            magnitude.len() + if group > 0 { magnitude.len() / group } else { 0 } + 1,
        );
        if self.negative {
            output.push('-');
        }
        if group == 0 {
            output.push_str(&magnitude);
        } else {
            for (i, ch) in magnitude.chars().enumerate() {
                if i > 0 && (magnitude.len() - i) % group == 0 {
                    output.push('_');
                }
                output.push(ch);
            }
        }
        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_mag(&self.digits, &rhs.digits),
            (true, true) => cmp_mag(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = add_mag(&self.digits, &rhs.digits);
            Self::from_parts(self.negative, digits)
        } else {
            match cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Greater => {
                    Self::from_parts(self.negative, sub_mag(&self.digits, &rhs.digits))
                }
                Ordering::Less => {
                    Self::from_parts(rhs.negative, sub_mag(&rhs.digits, &self.digits))
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let k = i + j;
                let total = result[k] as u16 + (a as u16) * (b as u16) + carry;
                result[k] = (total % 10) as u8;
                carry = total / 10;
            }

            let mut k = i + rhs.digits.len();
            while carry > 0 {
                if k == result.len() {
                    result.push(0);
                }
                let total = result[k] as u16 + carry;
                result[k] = (total % 10) as u8;
                carry = total / 10;
                k += 1;
            }
        }

        Self::from_parts(self.negative != rhs.negative, result)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient, remainder) = div_rem_mag(&self.digits, &rhs.digits);
        Ok((
            Self::from_parts(self.negative != rhs.negative, quotient),
            Self::from_parts(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let digits = normalize(digits);
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }
}

fn normalize(mut digits: Vec<u8>) -> Vec<u8> {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
    digits
}

fn cmp_mag(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for i in (0..a.len()).rev() {
                match a[i].cmp(&b[i]) {
                    Ordering::Equal => {}
                    non_equal => return non_equal,
                }
            }
            Ordering::Equal
        }
        non_equal => non_equal,
    }
}

fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let total = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        result.push(total % 10);
        carry = total / 10;
    }
    if carry != 0 {
        result.push(carry);
    }
    result
}

// Requires a >= b.
fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(digit as u8);
    }
    normalize(result)
}

fn mul_mag_digit(a: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }
    let mut result = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u16;
    for &value in a {
        let total = value as u16 * digit as u16 + carry;
        result.push((total % 10) as u8);
        carry = total / 10;
    }
    while carry > 0 {
        result.push((carry % 10) as u8);
        carry /= 10;
    }
    normalize(result)
}

// Magnitudes are little-endian. Returns nonnegative quotient and remainder.
fn div_rem_mag(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_mag(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0];

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        remainder = normalize(remainder);

        let mut q_digit = 0u8;
        for candidate in (1..=9).rev() {
            let product = mul_mag_digit(divisor, candidate);
            if cmp_mag(&product, &remainder) != Ordering::Greater {
                q_digit = candidate;
                remainder = sub_mag(&remainder, &product);
                break;
            }
        }
        quotient_be.push(q_digit);
    }

    quotient_be.reverse();
    (normalize(quotient_be), normalize(remainder))
}
```
