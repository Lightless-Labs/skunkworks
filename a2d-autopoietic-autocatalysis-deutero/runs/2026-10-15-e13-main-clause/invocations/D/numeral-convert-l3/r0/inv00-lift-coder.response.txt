#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1_i128 << 120) - 1;

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    if !(-LIMIT..=LIMIT).contains(&value) {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut digits = Vec::new();

    if n == 0 {
        digits.push(b'0');
    } else {
        match system {
            SignedSystem::BalancedTernary => {
                while n != 0 {
                    let mut remainder = n % 3;
                    n /= 3;
                    if remainder == 2 {
                        remainder = -1;
                        n += 1;
                    } else if remainder == -2 {
                        remainder = 1;
                        n -= 1;
                    }

                    digits.push(match remainder {
                        -1 => b'T',
                        0 => b'0',
                        1 => b'1',
                        _ => unreachable!(),
                    });
                }
            }
            SignedSystem::Negabinary => {
                while n != 0 {
                    let remainder = n.rem_euclid(2);
                    digits.push(if remainder == 0 { b'0' } else { b'1' });
                    n = (n - remainder) / -2;
                }
            }
        }
    }

    digits.reverse();
    let mut result = String::from_utf8(digits).expect("digits are ASCII");

    if checksum {
        let sum = checksum_value(result.as_bytes());
        result.push(':');
        result.push(char::from(b'0' + sum));
    }

    Ok(result)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(SignedError::Empty);
    }

    let colon = bytes.iter().position(|&b| b == b':');
    let payload_end = colon.unwrap_or(bytes.len());
    if payload_end == 0 {
        return Err(SignedError::Empty);
    }

    let payload = &bytes[..payload_end];

    for (index, &byte) in payload.iter().enumerate() {
        let valid = match system {
            SignedSystem::BalancedTernary => matches!(byte, b'T' | b'0' | b'1'),
            SignedSystem::Negabinary => matches!(byte, b'0' | b'1'),
        };
        if !valid {
            return Err(SignedError::InvalidByte { index });
        }
    }

    let found_checksum = if let Some(colon_index) = colon {
        let suffix = &bytes[colon_index + 1..];
        if suffix.len() != 1 || !matches!(suffix[0], b'0'..=b'6') {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
        Some(suffix[0] - b'0')
    } else {
        None
    };

    if payload.len() > 1 && payload[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let mut value: i128 = 0;
    for (index, &byte) in payload.iter().enumerate() {
        let digit = match byte {
            b'T' => -1_i128,
            b'0' => 0,
            b'1' => 1,
            _ => unreachable!(),
        };
        let base = match system {
            SignedSystem::BalancedTernary => 3_i128,
            SignedSystem::Negabinary => -2_i128,
        };

        value = value
            .checked_mul(base)
            .and_then(|v| v.checked_add(digit))
            .ok_or(SignedError::Range { index })?;

        if !(-LIMIT..=LIMIT).contains(&value) {
            return Err(SignedError::Range { index });
        }
    }

    if let Some(found) = found_checksum {
        let expected = checksum_value(payload);
        if found != expected {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(value)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, checksum)
}

fn checksum_value(payload: &[u8]) -> u8 {
    let sum = payload
        .iter()
        .enumerate()
        .map(|(index, &byte)| {
            let code = match byte {
                b'T' => 2_u32,
                b'0' => 0_u32,
                b'1' => 1_u32,
                _ => 0_u32,
            };
            ((index as u32 + 1) * code) % 7
        })
        .sum::<u32>();
    (sum % 7) as u8
}
