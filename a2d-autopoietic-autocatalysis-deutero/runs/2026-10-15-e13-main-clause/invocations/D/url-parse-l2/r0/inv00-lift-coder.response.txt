#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let Some(high) = hex_value(bytes[i + 1]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            let Some(low) = hex_value(bytes[i + 2]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((high << 4) | low);
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = offsets.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

fn component_byte_is_safe(component: Component, byte: u8) -> bool {
    byte.is_ascii_alphanumeric()
        || matches!(byte, b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = text.as_bytes();
    let mut output = String::with_capacity(bytes.len());

    for &byte in bytes {
        if component_byte_is_safe(component, byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }

    output
}

fn is_scheme_first(byte: u8) -> bool {
    byte.is_ascii_alphabetic()
}

fn is_scheme_rest(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')
}

fn ascii_lowercase(text: &str) -> String {
    text.bytes().map(|b| b.to_ascii_lowercase() as char).collect()
}

fn parse_port(input: &str, start: usize) -> Result<u16, ParseError> {
    let bytes = input.as_bytes();
    if start == bytes.len() {
        return Err(ParseError::InvalidPort { at: start });
    }

    let port_bytes = &bytes[start..];
    if let Some(offset) = port_bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: start + offset });
    }

    if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for &byte in port_bytes {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
        {
            Some(n) => n,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
        if value > u16::MAX as u32 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }

    Ok(value as u16)
}

fn validate_component(
    component: Component,
    text: &str,
    original_start: usize,
) -> Result<(), ParseError> {
    match decode_component(component, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => {
            Err(ParseError::BadEscape { at: original_start + at })
        }
        Err(CodecError::InvalidUtf8 { at }) => {
            Err(ParseError::InvalidUtf8 { at: original_start + at })
        }
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let Some(colon) = bytes.iter().position(|&b| b == b':') else {
        return Err(ParseError::InvalidScheme { at: input.len() });
    };

    if colon == 0 || !is_scheme_first(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !is_scheme_rest(byte) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_in_authority) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(&authority[..at]), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[host_start_in_authority..];
    let hostport_start = authority_start + host_start_in_authority;

    let (host, port, ipv6) = if hostport.as_bytes().first() == Some(&b'[') {
        let open_at = hostport_start;
        let Some(close_relative) = hostport.as_bytes()[1..].iter().position(|&b| b == b']') else {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        };
        let close = close_relative + 1;
        let interior = &hostport[1..close];

        if interior.is_empty() || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }

        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(input, hostport_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };

        (ascii_lowercase(interior), port, true)
    } else {
        let colon_in_hostport = hostport.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, port) = if let Some(offset) = colon_in_hostport {
            (
                &hostport[..offset],
                Some(parse_port(input, hostport_start + offset + 1)?),
            )
        } else {
            (hostport, None)
        };

        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }

        if let Some(offset) = raw_host
            .bytes()
            .position(|b| !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')))
        {
            return Err(ParseError::InvalidHost {
                at: hostport_start + offset,
            });
        }

        (ascii_lowercase(raw_host), port, false)
    };

    let remainder = &input[authority_end..];
    let fragment_offset = remainder.find('#');
    let before_fragment = fragment_offset.map(|i| &remainder[..i]).unwrap_or(remainder);
    let fragment = fragment_offset.map(|i| &remainder[i + 1..]);

    let query_offset = before_fragment.find('?');
    let path = query_offset
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_offset.map(|i| &before_fragment[i + 1..]);

    let path_start = authority_end;
    let query_start = query_offset.map(|i| authority_end + i + 1);
    let fragment_start = fragment_offset.map(|i| authority_end + i + 1);

    if let Some(userinfo_text) = userinfo {
        let userinfo_start = authority_start;
        validate_component(Component::UserInfo, userinfo_text, userinfo_start)?;
    }

    if !path.is_empty() {
        validate_component(Component::Path, path, path_start)?;
    }
    if let (Some(query_text), Some(start)) = (query, query_start) {
        validate_component(Component::Query, query_text, start)?;
    }
    if let (Some(fragment_text), Some(start)) = (fragment, fragment_start) {
        validate_component(Component::Fragment, fragment_text, start)?;
    }

    Ok(Url {
        scheme: ascii_lowercase(&input[..colon]),
        userinfo: userinfo.map(str::to_owned),
        host,
        port,
        path: if path.is_empty() {
            "/".to_owned()
        } else {
            path.to_owned()
        },
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}
