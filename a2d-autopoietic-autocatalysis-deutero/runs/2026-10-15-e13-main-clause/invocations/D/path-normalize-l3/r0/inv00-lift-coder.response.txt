#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str, base: usize) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_owned(), base + start));
    }
    result
}

fn render(stack: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent && result != "/" {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    for (component, at) in components(path, 0) {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(component);
                }
            }
            _ => stack.push(component),
        }
    }

    if absolute {
        Ok(render(&stack, intent))
    } else if stack.is_empty() {
        Ok(".".to_owned())
    } else {
        let mut result = stack.join("/");
        if intent && result != "." {
            result.push('/');
        }
        Ok(result)
    }
}

fn validated_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(components(&normalized, 0)
        .into_iter()
        .map(|(component, _)| component)
        .collect())
}

fn check_tilde<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, usize, Option<&'a [String]>), PathError> {
    if !path.starts_with('~') {
        return Ok((false, 0, None));
    }
    if path.len() == 1 || path.as_bytes().get(1) == Some(&b'/') {
        let home = home.ok_or(PathError::MissingHome)?;
        let consumed = if path.len() > 1 { 2 } else { 1 };
        return Ok((true, consumed, Some(home)));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let (expanded, consumed, _) =
        check_tilde(path, home_components.as_deref())?;
    let intent = directory_intent(path);
    let mut stack = if expanded {
        home_components.unwrap_or_default()
    } else if path.starts_with('/') {
        Vec::new()
    } else {
        cwd_components
    };

    let remainder = &path[consumed..];
    for (component, at) in components(remainder, consumed) {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(component),
        }
    }
    Ok(render(&stack, intent))
}

struct Frame {
    components: Vec<(String, usize)>,
    pos: usize,
    origin: Origin,
}

#[derive(Clone, Copy)]
enum Origin {
    Request,
    Context,
    Target,
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(key, _)| *key == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }

    validate_raw(path)?;
    let (expanded, consumed, _) = check_tilde(path, home_components.as_deref())?;
    let intent = directory_intent(path);

    let (mut stack, request_offset, start_components) = if expanded {
        (
            home_components.unwrap_or_default(),
            consumed,
            Vec::new(),
        )
    } else if path.starts_with('/') {
        (Vec::new(), 0, Vec::new())
    } else {
        (Vec::new(), 0, cwd_components)
    };

    let request = components(&path[request_offset..], request_offset);
    let mut frames = Vec::new();

    if !start_components.is_empty() {
        frames.push(Frame {
            components: start_components
                .into_iter()
                .map(|component| (component, 0))
                .collect(),
            pos: 0,
            origin: Origin::Context,
        });
    }
    frames.push(Frame {
        components: request,
        pos: 0,
        origin: Origin::Request,
    });

    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    while !frames.is_empty() {
        let frame_index = frames.len() - 1;
        if frames[frame_index].pos >= frames[frame_index].components.len() {
            let finished = frames.pop().unwrap();
            if let Origin::Target = finished.origin {
                active.pop();
            }
            continue;
        }

        let (component, at) = {
            let frame = &mut frames[frame_index];
            let item = frame.components[frame.pos].clone();
            frame.pos += 1;
            item
        };

        match component.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(component);
                let current = render(&stack, false);
                if let Some((link_index, link)) = links
                    .iter()
                    .enumerate()
                    .find(|(_, link)| link.path == current)
                {
                    if active.contains(&link_index) {
                        return Err(PathError::Loop { index: link_index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index: link_index });
                    }
                    expansions += 1;

                    // A relative target starts at the link's parent.
                    stack.pop();
                    if link.target.starts_with('/') {
                        stack.clear();
                    }

                    active.push(link_index);
                    frames.push(Frame {
                        components: components(&link.target, 0),
                        pos: 0,
                        origin: Origin::Target,
                    });
                }
            }
        }
    }

    Ok(render(&stack, intent))
}
