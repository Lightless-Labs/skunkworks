#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

fn update_crc32(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb88320;
            }
        }
    }
    crc
}

fn update_adler(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % 65521;
        *b = (*b + *a) % 65521;
    }
}

fn update_fletcher16(mut a: u16, mut b: u16, data: &[u8]) -> (u16, u16) {
    for &byte in data {
        a = (a + byte as u16) % 255;
        b = (b + a) % 255;
    }
    (a, b)
}

fn update_fletcher32(
    mut a: u32,
    mut b: u32,
    mut pending_high: Option<u8>,
    data: &[u8],
) -> (u32, u32, Option<u8>) {
    for &byte in data {
        if let Some(high) = pending_high.take() {
            let word = ((high as u32) << 8) | byte as u32;
            a = (a + word) % 65535;
            b = (b + a) % 65535;
        } else {
            pending_high = Some(byte);
        }
    }
    (a, b, pending_high)
}

fn finish_fletcher32(a: u32, b: u32, pending_high: Option<u8>) -> u32 {
    let (a, b) = if let Some(high) = pending_high {
        let a = (a + ((high as u32) << 8)) % 65535;
        let b = (b + a) % 65535;
        (a, b)
    } else {
        (a, b)
    };
    (b << 16) | a
}

fn update_crc16(mut crc: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc <<= 1;
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
        }
        Crc16::X25 => {
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
        }
    }
    crc
}

fn crc16_initial(variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    }
}

fn crc16_finish(crc: u16, variant: Crc16) -> u16 {
    match variant {
        Crc16::X25 => crc ^ 0xffff,
        Crc16::Xmodem | Crc16::False => crc,
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = None;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if let Some(end) = previous_end {
            if span.start < end {
                return Err(SpanError::Overlap { span: index });
            }
        }
        previous_end = Some(span.end);
    }

    Ok(())
}

pub fn crc32(data: &[u8]) -> u32 {
    update_crc32(0xffff_ffff, data) ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    update_adler(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;

    for span in spans {
        let selected = &data[span.start..span.end];
        crc = update_crc32(crc, selected);
        update_adler(&mut a, &mut b, selected);
    }

    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (a, b) = update_fletcher16(0, 0, data);
    (b << 8) | a
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (a, b, pending) = update_fletcher32(0, 0, None, data);
    finish_fletcher32(a, b, pending)
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut a = 0;
    let mut b = 0;
    let mut pending = None;

    for span in spans {
        (a, b, pending) =
            update_fletcher32(a, b, pending, &data[span.start..span.end]);
    }

    Ok(finish_fletcher32(a, b, pending))
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let crc = update_crc16(crc16_initial(variant), data, variant);
    crc16_finish(crc, variant)
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc32_state = 0xffff_ffff;
    let mut adler_a = 1;
    let mut adler_b = 0;
    let mut f16_a = 0;
    let mut f16_b = 0;
    let mut f32_a = 0;
    let mut f32_b = 0;
    let mut f32_pending = None;
    let mut crc16_state = crc16_initial(variant);

    for span in spans {
        let selected = &data[span.start..span.end];
        crc32_state = update_crc32(crc32_state, selected);
        update_adler(&mut adler_a, &mut adler_b, selected);
        (f16_a, f16_b) = update_fletcher16(f16_a, f16_b, selected);
        (f32_a, f32_b, f32_pending) =
            update_fletcher32(f32_a, f32_b, f32_pending, selected);
        crc16_state = update_crc16(crc16_state, selected, variant);
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc32_state ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: (f16_b << 8) | f16_a,
        fletcher32: finish_fletcher32(f32_a, f32_b, f32_pending),
        crc16: crc16_finish(crc16_state, variant),
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;

        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }

        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = update_crc32(self.crc, data);
        update_adler(&mut self.a, &mut self.b, data);
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;

        let mut crc = self.crc;
        let mut a = self.a;
        let mut b = self.b;

        for span in spans {
            let selected = &data[span.start..span.end];
            crc = update_crc32(crc, selected);
            update_adler(&mut a, &mut b, selected);
        }

        self.crc = crc;
        self.a = a;
        self.b = b;
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksum = self.checksum();
        format!("c{:08x}:a{:08x}", checksum.crc32, checksum.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        for offset in 1..9 {
            let digit = hex_value(bytes[offset])
                .ok_or(SnapshotError::Hex { offset })?;
            crc = (crc << 4) | digit as u32;
        }

        let mut adler = 0u32;
        for offset in 11..19 {
            let digit = hex_value(bytes[offset])
                .ok_or(SnapshotError::Hex { offset })?;
            adler = (adler << 4) | digit as u32;
        }

        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}
