#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

impl Machine {
    const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = std::collections::HashSet::new();
        for product in &products {
            if product.id.is_empty()
                || product.id.len() > 8
                || !product.id.bytes().all(|b| b.is_ascii_lowercase())
            {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = Self::DENOMINATIONS.iter().position(|&d| d == coin);
        let Some(_) = index else {
            return Err(Error::BadCoin { coin });
        };

        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|p| p.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let Some(product_index) = self.products.iter().position(|p| p.id == *id) else {
            return Err(Error::NoSelection);
        };
        let product = &self.products[product_index];

        if product.stock == 0 {
            return Err(Error::SoldOut {
                id: product.id.clone(),
            });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let amount = self.credit - product.price;
        let change = match Self::choose_change(amount, self.tubes) {
            Some(coins) => coins,
            None => return Err(Error::NoChange { amount }),
        };

        let mut proposed_tubes = self.tubes;
        for &coin in &change {
            let i = Self::DENOMINATIONS
                .iter()
                .position(|&d| d == coin)
                .expect("change uses supported denominations");
            proposed_tubes[i] -= 1;
        }
        for &coin in &self.escrow {
            let i = Self::DENOMINATIONS
                .iter()
                .position(|&d| d == coin)
                .expect("escrow uses supported denominations");
            proposed_tubes[i] += 1;
            if proposed_tubes[i] > 100 {
                return Err(Error::TubeLimit { coin });
            }
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price: product.price,
            change: change.clone(),
        };

        self.products[product_index].stock -= 1;
        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Vended {
            id: receipt.id.clone(),
            price: receipt.price,
            change: receipt.change.clone(),
        });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn log(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }

    fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
        if amount == 0 {
            return Some(Vec::new());
        }

        let descending = [3usize, 2, 1, 0];
        let mut remaining = amount;
        let mut greedy_counts = [0u16; 4];

        for &i in &descending {
            let denomination = Self::DENOMINATIONS[i];
            let take = tubes[i].min(remaining / denomination);
            greedy_counts[i] = take;
            remaining -= take * denomination;
        }

        if remaining == 0 {
            return Some(Self::counts_to_descending(greedy_counts));
        }

        let mut best: Option<[u16; 4]> = None;
        Self::search_change(
            3,
            amount,
            &mut [0; 4],
            tubes,
            &mut best,
        );

        best.map(Self::counts_to_descending)
    }

    fn search_change(
        index: isize,
        remaining: u16,
        current: &mut [u16; 4],
        tubes: [u16; 4],
        best: &mut Option<[u16; 4]>,
    ) {
        if index < 0 {
            if remaining == 0 {
                let is_better = match best {
                    None => true,
                    Some(existing) => {
                        let total = |counts: &[u16; 4]| -> u32 {
                            counts.iter().map(|&c| u32::from(c)).sum()
                        };
                        let candidate_total = total(current);
                        let existing_total = total(existing);
                        candidate_total < existing_total
                            || (candidate_total == existing_total
                                && Self::lex_greater_descending(current, existing))
                    }
                };
                if is_better {
                    *best = Some(*current);
                }
            }
            return;
        }

        let i = index as usize;
        let denomination = Self::DENOMINATIONS[i];
        let max_take = tubes[i].min(remaining / denomination);

        for take in 0..=max_take {
            current[i] = take;
            Self::search_change(
                index - 1,
                remaining - take * denomination,
                current,
                tubes,
                best,
            );
        }
        current[i] = 0;
    }

    fn lex_greater_descending(a: &[u16; 4], b: &[u16; 4]) -> bool {
        for &i in &[3usize, 2, 1, 0] {
            if a[i] != b[i] {
                return a[i] > b[i];
            }
        }
        false
    }

    fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
        let mut result = Vec::new();
        for &i in &[3usize, 2, 1, 0] {
            for _ in 0..counts[i] {
                result.push(Self::DENOMINATIONS[i]);
            }
        }
        result
    }
}
