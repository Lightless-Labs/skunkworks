#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn parse_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn lexical_cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<CfwsRun>), EmailError> {
    let bytes = input.as_bytes();
    let mut clean = Vec::with_capacity(bytes.len());
    let mut map = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();

    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut quoted_prefix = false;
    let mut quote_closed = false;
    let mut in_run = false;
    let mut run_start = 0usize;

    while i < bytes.len() {
        if comment_depth == 0 && !quoted_prefix && clean.is_empty() && bytes[i] == b'"' {
            quoted_prefix = true;
            clean.push(bytes[i]);
            map.push(i);
            i += 1;
            continue;
        }

        if quoted_prefix && !quote_closed {
            let b = bytes[i];
            clean.push(b);
            map.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    map.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    quote_closed = true;
                }
                i += 1;
            }
            continue;
        }

        if comment_depth > 0 {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= bytes.len() || !is_printable(bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = parse_fold(bytes, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if !is_printable(b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let b = bytes[i];
        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            if !in_run {
                in_run = true;
                run_start = i;
            }

            if b == b'(' {
                comment_depth = 1;
                outer_comment = i;
                i += 1;
            } else if b == b'\r' || b == b'\n' {
                if let Some(end) = parse_fold(bytes, i) {
                    i = end;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else {
                i += 1;
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if in_run {
            runs.push(CfwsRun { start: run_start, end: i });
            in_run = false;
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if in_run {
        runs.push(CfwsRun {
            start: run_start,
            end: bytes.len(),
        });
    }

    Ok((clean, map, runs))
}

fn pos(map: &[usize], index: usize, input_len: usize) -> usize {
    map.get(index).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, map, runs) = lexical_cleanup(input)?;
    for run in &runs {
        let left_boundary = run.start == 0;
        let right_boundary = run.end == input.len();
        let before = map.iter().position(|&p| p >= run.start);
        let after = map.iter().position(|&p| p >= run.end);
        let adjacent_at = before
            .and_then(|i| i.checked_sub(1).map(|j| clean[j] == b'@'))
            .unwrap_or(false)
            || after.map(|i| clean[i] == b'@').unwrap_or(false);
        if !left_boundary && !right_boundary && !adjacent_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: pos(&map, 0, input.len()) });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    while i < clean.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if clean[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(&map, separators[1], input.len()),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&map, sep, input.len()),
        });
    }

    let mut local = String::new();
    let quoted = clean[0] == b'"';
    if !quoted {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&map, 64, input.len()),
            });
        }
        for j in 0..sep {
            let b = clean[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || clean[j - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: pos(&map, j, input.len()),
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: pos(&map, j, input.len()),
                });
            }
            local.push(b as char);
        }
    } else {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&map, end + 1, input.len()),
            });
        }

        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = clean[j];
            if b == b'\\' {
                let Some(&escaped) = clean.get(j + 1) else {
                    return Err(EmailError::BadEscape {
                        at: pos(&map, j, input.len()),
                    });
                };
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: pos(&map, j, input.len()),
                    });
                }
                local.push(escaped as char);
                origins.push(pos(&map, j, input.len()));
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(&map, j, input.len()),
                    });
                }
                local.push(b as char);
                origins.push(pos(&map, j, input.len()));
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&map, end, input.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    }

    let domain_start = sep + 1;
    let raw_domain = &clean[domain_start..];
    let raw_positions = &map[domain_start..];
    let mut domain_len = raw_domain.len();
    let root_pos = if domain_len > 0 && raw_domain[domain_len - 1] == b'.' {
        domain_len -= 1;
        Some(raw_positions[domain_len])
    } else {
        None
    };

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain {
            at: pos(&map, domain_start, input.len()),
        });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: raw_positions[253],
        });
    }

    let domain = &raw_domain[..domain_len];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0usize;
    for j in 0..=domain_len {
        if j == domain_len || domain[j] == b'.' {
            labels.push((start, j));
            start = j + 1;
        }
    }

    for (label_start, label_end) in &labels {
        if label_start == label_end {
            let at = if *label_start == domain_len {
                root_pos.unwrap_or_else(|| pos(raw_positions, domain_len, input.len()))
            } else {
                raw_positions[*label_start]
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if label_end - label_start > 63 {
            return Err(EmailError::LabelTooLong {
                at: raw_positions[label_start + 63],
            });
        }
        if domain[*label_start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: raw_positions[*label_start],
            });
        }
        if domain[label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: raw_positions[label_end - 1],
            });
        }
        for j in *label_start..*label_end {
            let b = domain[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: raw_positions[j],
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&map, domain_start, input.len()),
        });
    }
    let (last_start, last_end) = labels[labels.len() - 1];
    if domain[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: raw_positions[last_start],
        });
    }

    let lowered_domain = domain
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain: lowered_domain,
        quoted,
    })
}
