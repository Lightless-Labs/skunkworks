#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .collect()
}

fn reduce(
    stack: &mut Vec<String>,
    parts: impl IntoIterator<Item = (String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let mut stack = Vec::new();
    let mut offset = 0usize;

    let parts = path.split('/').filter_map(|part| {
        let start = offset;
        offset += part.len() + 1;
        if part.is_empty() {
            None
        } else {
            Some((part.to_owned(), start))
        }
    });

    reduce(&mut stack, parts, absolute)?;

    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if directory_intent(path) && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(components(&normalized)
        .into_iter()
        .map(str::to_owned)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let (absolute, mut stack, start_at, remainder) = if bytes[0] == b'/' {
        (true, Vec::new(), 0usize, path)
    } else if bytes[0] == b'~' {
        if bytes.len() == 1 || bytes[1] == b'/' {
            let parts = home_parts.ok_or(PathError::MissingHome)?;
            let remainder = if bytes.len() == 1 { "" } else { &path[2..] };
            (true, parts, 0usize, remainder)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (true, cwd_parts, 0usize, path)
    };

    let remainder_base = if bytes[0] == b'~' { start_at + 2 } else { 0 };
    let mut offset = remainder_base;
    let parts = remainder.split('/').filter_map(|part| {
        let start = offset;
        offset += part.len() + 1;
        if part.is_empty() {
            None
        } else {
            Some((part.to_owned(), start))
        }
    });

    reduce(&mut stack, parts, absolute)?;

    let mut result = format!("/{}", stack.join("/"));
    if directory_intent(path) && result != "/" {
        result.push('/');
    }

    Ok(result)
}
