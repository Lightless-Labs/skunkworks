#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    consumed: usize,
    marker_ended: bool,
    raw: bool,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    raw_lines: usize,
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    let astrings: Vec<String> = a.lines.iter().map(render_prejustify).collect();
    let bstrings: Vec<String> = b.lines.iter().map(render_prejustify).collect();
    astrings < bstrings
}

fn render_prejustify(line: &Line) -> String {
    let mut s = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(piece);
    }
    s
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Solution {
    fn rec(
        words: &[Word],
        wi: usize,
        offset: usize,
        first: bool,
        options: &WrapOptions,
        memo: &mut std::collections::HashMap<(usize, usize, bool), Solution>,
    ) -> Solution {
        if wi >= words.len() {
            return Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                raw_lines: 0,
            };
        }
        if let Some(s) = memo.get(&(wi, offset, first)) {
            return s.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut best: Option<Solution> = None;

        let mut pieces = Vec::<String>::new();
        let mut used = 0usize;
        let mut cur_wi = wi;
        let mut cur_off = offset;

        loop {
            if cur_wi >= words.len() {
                let gap = options.width - (indent + used);
                let tail = rec(words, cur_wi, 0, false, options, memo);
                let mut line = Line {
                    indent,
                    pieces: pieces.clone(),
                    consumed: used,
                    marker_ended: false,
                    raw: false,
                };
                let mut candidate = tail;
                let line_cost = (gap * gap / 4) as u64;
                candidate.cost += line_cost;
                if pieces.len() == 1 {
                    // Final lines have no single-piece surcharge.
                }
                candidate.lines.insert(0, line.clone());
                candidate.consumed.insert(0, used);
                if better(&candidate, best.as_ref().unwrap_or(&candidate)) {
                    best = Some(candidate);
                }
                break;
            }

            let word = &words[cur_wi];
            let remaining = word.chars.len() - cur_off;
            let gap_between = usize::from(!pieces.is_empty());
            if used + gap_between >= cap {
                break;
            }
            let avail = cap - used - gap_between;

            if remaining <= avail {
                let text: String = word.chars[cur_off..].iter().collect();
                pieces.push(text);
                let line_used = used + gap_between + remaining;
                let next = rec(words, cur_wi + 1, 0, false, options, memo);
                let is_final = cur_wi + 1 >= words.len();
                let gap = options.width - (indent + line_used);
                let mut candidate = next;
                candidate.cost += if is_final {
                    (gap * gap / 4) as u64
                } else {
                    (gap * gap) as u64
                };
                if !is_final && pieces.len() == 1 {
                    candidate.cost += 3;
                }
                candidate.lines.insert(
                    0,
                    Line {
                        indent,
                        pieces: pieces.clone(),
                        consumed: line_used,
                        marker_ended: false,
                        raw: false,
                    },
                );
                candidate.consumed.insert(0, line_used);
                if better(&candidate, best.as_ref().unwrap_or(&candidate)) {
                    best = Some(candidate);
                }
                pieces.pop();

                used = line_used;
                cur_wi += 1;
                cur_off = 0;
                if used >= cap {
                    break;
                }
                continue;
            }

            for &m in word.markers.iter().filter(|&&m| m > cur_off && m < word.chars.len()) {
                let prefix = m - cur_off;
                let total = used + gap_between + prefix + 1;
                if prefix > 0 && total <= cap {
                    let mut next_pieces = pieces.clone();
                    let mut text: String = word.chars[cur_off..m].iter().collect();
                    text.push('-');
                    next_pieces.push(text);
                    let tail = rec(words, cur_wi, m, false, options, memo);
                    let gap = options.width - (indent + total);
                    let mut candidate = tail;
                    candidate.cost += (gap * gap) as u64 + options.hyphen_cost as u64;
                    if next_pieces.len() == 1 {
                        candidate.cost += 3;
                    }
                    candidate.marker_lines += 1;
                    candidate.lines.insert(
                        0,
                        Line {
                            indent,
                            pieces: next_pieces,
                            consumed: total,
                            marker_ended: true,
                            raw: false,
                        },
                    );
                    candidate.consumed.insert(0, total);
                    if better(&candidate, best.as_ref().unwrap_or(&candidate)) {
                        best = Some(candidate);
                    }
                }
            }

            if pieces.is_empty() && remaining > cap {
                let marker_fits = word
                    .markers
                    .iter()
                    .any(|&m| m > cur_off && m < word.chars.len() && m - cur_off + 1 <= cap);
                if !marker_fits {
                    let text: String = word.chars[cur_off..cur_off + cap].iter().collect();
                    let tail = rec(words, cur_wi, cur_off + cap, false, options, memo);
                    let gap = options.width - (indent + cap);
                    let mut candidate = tail;
                    candidate.cost += (gap * gap) as u64 + (2 * options.width + 1) as u64;
                    candidate.lines.insert(
                        0,
                        Line {
                            indent,
                            pieces: vec![text],
                            consumed: cap,
                            marker_ended: false,
                            raw: true,
                        },
                    );
                    candidate.consumed.insert(0, cap);
                    candidate.raw_lines += 1;
                    if better(&candidate, best.as_ref().unwrap_or(&candidate)) {
                        best = Some(candidate);
                    }
                }
            }
            break;
        }

        let answer = best.unwrap_or_else(|| Solution {
            cost: u64::MAX / 4,
            marker_lines: 0,
            lines: Vec::new(),
            consumed: Vec::new(),
            raw_lines: 0,
        });
        memo.insert((wi, offset, first), answer.clone());
        answer
    }

    rec(
        words,
        0,
        0,
        true,
        options,
        &mut std::collections::HashMap::new(),
    )
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();
    for &c in segment {
        if c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}' {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn justify_line(line: &Line, width: usize, line_index: usize) -> String {
    let mut s = " ".repeat(line.indent);
    let gaps = line.pieces.len().saturating_sub(1);
    let content: usize = line.pieces.iter().map(|p| p.chars().count()).sum();
    if gaps == 0 {
        s.push_str(&line.pieces[0]);
        return s;
    }
    let base_spaces = 1 + (width - line.indent - content - gaps) / gaps;
    let extra = (width - line.indent - content - gaps) % gaps;
    let start = line_index % gaps;
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            let gap_index = i - 1;
            let bonus = if (gap_index + gaps - start) % gaps < extra { 1 } else { 0 };
            s.push_str(&" ".repeat(base_spaces + bonus));
        }
        s.push_str(piece);
    }
    s
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let scalar_count = input.chars().count();
    if scalar_count > 512 {
        return Err(WrapError::InputTooLong {
            chars: scalar_count,
        });
    }

    let mut byte_index = 0usize;
    let mut indexed = Vec::new();
    for c in input.chars() {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        indexed.push((c, byte_index));
        byte_index += c.len_utf8();
    }

    let separators = |c: char| {
        c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}'
    };
    for i in 0..indexed.len() {
        if indexed[i].0 == '~' {
            let valid_side = |j: isize| -> bool {
                if j < 0 || j as usize >= indexed.len() {
                    return false;
                }
                let c = indexed[j as usize].0;
                c != '~' && !separators(c)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker {
                    index: indexed[i].1,
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < indexed.len() {
        let (c, _) = indexed[i];
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < indexed.len() && indexed[i + 1].0 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut pos = 0usize;
    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let run_start = pos;
            while pos < normalized.len() && normalized[pos] == '\n' {
                pos += 1;
            }
            let run_len = pos - run_start;
            if run_len >= 2 {
                let words = parse_words(&normalized[start..run_start]);
                if !words.is_empty() {
                    let solution = solve_paragraph(&words, options);
                    for (line_index, line) in solution.lines.iter().enumerate() {
                        let rendered = if options.justify
                            && line_index + 1 < solution.lines.len()
                            && line.pieces.len() >= 2
                        {
                            justify_line(line, options.width, line_index)
                        } else {
                            render_prejustify(line)
                        };
                        output.push(rendered);
                    }
                }
                for _ in 1..run_len {
                    output.push(String::new());
                }
                start = pos;
            }
        } else {
            pos += 1;
        }
    }

    let words = parse_words(&normalized[start..]);
    if !words.is_empty() {
        let solution = solve_paragraph(&words, options);
        for (line_index, line) in solution.lines.iter().enumerate() {
            let rendered = if options.justify
                && line_index + 1 < solution.lines.len()
                && line.pieces.len() >= 2
            {
                justify_line(line, options.width, line_index)
            } else {
                render_prejustify(line)
            };
            output.push(rendered);
        }
    }

    Ok(output)
}
