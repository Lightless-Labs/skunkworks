fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00a0}'
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn justify_line(line: &Line, width: usize, line_index: usize) -> String {
    let mut rendered = " ".repeat(line.indent);
    let gaps = line.pieces.len() - 1;
    if gaps == 0 {
        rendered.push_str(&line.pieces[0]);
        return rendered;
    }

    let content_width: usize = line.pieces.iter().map(|piece| piece.chars().count()).sum();
    let extra_spaces = width - line.indent - content_width - gaps;
    let base_extra = extra_spaces / gaps;
    let remainder = extra_spaces % gaps;
    let start_gap = line_index % gaps;

    for (index, piece) in line.pieces.iter().enumerate() {
        if index > 0 {
            let gap_index = index - 1;
            let bonus = usize::from((gap_index + gaps - start_gap) % gaps < remainder);
            rendered.push_str(&" ".repeat(1 + base_extra + bonus));
        }
        rendered.push_str(piece);
    }
    rendered
}

fn append_paragraph(output: &mut Vec<String>, segment: &[char], options: &WrapOptions) {
    let words = parse_words(segment);
    if words.is_empty() {
        return;
    }

    let solution = solve_paragraph(&words, options);
    for (line_index, line) in solution.lines.iter().enumerate() {
        if options.justify && line_index + 1 < solution.lines.len() && line.pieces.len() >= 2 {
            output.push(justify_line(line, options.width, line_index));
        } else {
            output.push(render_prejustify(line));
        }
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let scalar_count = input.chars().count();
    if scalar_count > 512 {
        return Err(WrapError::InputTooLong { chars: scalar_count });
    }

    let mut indexed = Vec::new();
    let mut byte_index = 0usize;
    for ch in input.chars() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        indexed.push((ch, byte_index));
        byte_index += ch.len_utf8();
    }

    for index in 0..indexed.len() {
        if indexed[index].0 != '~' {
            continue;
        }
        let side_valid = |side: isize| -> bool {
            if side < 0 || side as usize >= indexed.len() {
                return false;
            }
            let ch = indexed[side as usize].0;
            ch != '~' && !is_separator(ch)
        };
        if !side_valid(index as isize - 1) || !side_valid(index as isize + 1) {
            return Err(WrapError::InvalidMarker {
                index: indexed[index].1,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut index = 0usize;
    while index < indexed.len() {
        let ch = indexed[index].0;
        if ch == '\r' {
            normalized.push('\n');
            if index + 1 < indexed.len() && indexed[index + 1].0 == '\n' {
                index += 1;
            }
        } else {
            normalized.push(ch);
        }
        index += 1;
    }

    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut position = 0usize;

    while position < normalized.len() {
        if normalized[position] != '\n' {
            position += 1;
            continue;
        }

        let run_start = position;
        while position < normalized.len() && normalized[position] == '\n' {
            position += 1;
        }
        let run_length = position - run_start;
        if run_length >= 2 {
            append_paragraph(&mut output, &normalized[segment_start..run_start], options);
            for _ in 1..run_length {
                output.push(String::new());
            }
            segment_start = position;
        }
    }

    append_paragraph(&mut output, &normalized[segment_start..], options);
    Ok(output)
}

    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
}

fn render_prejustify(line: &Line) -> String {
    let mut rendered = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            rendered.push(' ');
        }
        rendered.push_str(piece);
    }
    rendered
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.lines
        .iter()
        .map(render_prejustify)
        .cmp(b.lines.iter().map(render_prejustify))
        .is_lt()
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Solution {
    fn rec(
        words: &[Word],
        word_index: usize,
        offset: usize,
        first: bool,
        options: &WrapOptions,
        memo: &mut std::collections::HashMap<(usize, usize, bool), Option<Solution>>,
    ) -> Option<Solution> {
        if word_index == words.len() {
            return Some(Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
            });
        }

        let key = (word_index, offset, first);
        if let Some(solution) = memo.get(&key) {
            return solution.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let mut best: Option<Solution> = None;
        let mut pieces = Vec::<String>::new();
        let mut used = 0usize;
        let mut current_word = word_index;
        let mut current_offset = offset;
        let mut consumed_on_line = 0usize;

        loop {
            if current_word >= words.len() {
                break;
            }

            let word = &words[current_word];
            let remaining = word.chars.len() - current_offset;
            let gap_between = usize::from(!pieces.is_empty());
            if used + gap_between >= capacity {
                break;
            }
            let available = capacity - used - gap_between;

            if remaining <= available {
                let text: String = word.chars[current_offset..].iter().collect();
                pieces.push(text);
                let line_used = used + gap_between + remaining;
                let line_consumed = consumed_on_line + remaining;
                let is_final = current_word + 1 == words.len();

                if let Some(mut candidate) =
                    rec(words, current_word + 1, 0, false, options, memo)
                {
                    let gap = options.width - (indent + line_used);
                    let gap_cost = (gap * gap) as u64;
                    candidate.cost += if is_final { gap_cost / 4 } else { gap_cost };
                    if !is_final && pieces.len() == 1 {
                        candidate.cost += 3;
                    }
                    candidate.lines.insert(
                        0,
                        Line {
                            indent,
                            pieces: pieces.clone(),
                        },
                    );
                    candidate.consumed.insert(0, line_consumed);
                    if best.as_ref().is_none_or(|old| better(&candidate, old)) {
                        best = Some(candidate);
                    }
                }
                pieces.pop();

                used += gap_between + remaining;
                consumed_on_line += remaining;
                current_word += 1;
                current_offset = 0;

                if current_word == words.len() || used >= capacity {
                    break;
                }
                continue;
            }

            for &marker in &word.markers {
                if marker <= current_offset || marker >= word.chars.len() {
                    continue;
                }
                let prefix = marker - current_offset;
                let line_used = used + gap_between + prefix + 1;
                if line_used > capacity {
                    continue;
                }

                let mut next_pieces = pieces.clone();
                let mut text: String = word.chars[current_offset..marker].iter().collect();
                text.push('-');
                next_pieces.push(text);

                let line_consumed = consumed_on_line + prefix;
                if let Some(mut candidate) =
                    rec(words, current_word, marker, false, options, memo)
                {
                    let gap = options.width - (indent + line_used);
                    candidate.cost +=
                        (gap * gap) as u64 + options.hyphen_cost as u64;
                    if next_pieces.len() == 1 {
                        candidate.cost += 3;
                    }
                    candidate.marker_lines += 1;
                    candidate.lines.insert(
                        0,
                        Line {
                            indent,
                            pieces: next_pieces,
                        },
                    );
                    candidate.consumed.insert(0, line_consumed);
                    if best.as_ref().is_none_or(|old| better(&candidate, old)) {
                        best = Some(candidate);
                    }
                }
            }

            if pieces.is_empty() && remaining > capacity {
                let marker_fits = word.markers.iter().any(|&marker| {
                    marker > current_offset
                        && marker < word.chars.len()
                        && marker - current_offset + 1 <= capacity
                });

                if !marker_fits {
                    let text: String = word.chars[current_offset..current_offset + capacity]
                        .iter()
                        .collect();
                    if let Some(mut candidate) = rec(
                        words,
                        current_word,
                        current_offset + capacity,
                        false,
                        options,
                        memo,
                    ) {
                        let gap = options.width - (indent + capacity);
                        candidate.cost +=
                            (gap * gap) as u64 + (2 * options.width + 1) as u64;
                        candidate.lines.insert(
                            0,
                            Line {
                                indent,
                                pieces: vec![text],
                            },
                        );
                        candidate.consumed.insert(0, capacity);
                        if best.as_ref().is_none_or(|old| better(&candidate, old)) {
                            best = Some(candidate);
                        }
                    }
                }
            }

            break;
        }

        memo.insert(key, best.clone());
        best
    }

    rec(
        words,
        0,
        0,
        true,
        options,
        &mut std::collections::HashMap::new(),
    )
    .expect("raw splitting guarantees a legal layout")
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00a0}'
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn justify_line(line: &Line, width: usize, line_index: usize) -> String {
    let mut rendered = " ".repeat(line.indent);
    let gaps = line.pieces.len() - 1;
    if gaps == 0 {
        rendered.push_str(&line.pieces[0]);
        return rendered;
    }

    let piece_width: usize = line.pieces.iter().map(|piece| piece.chars().count()).sum();
    let extra_spaces = width - line.indent - piece_width - gaps;
    let base_extra = extra_spaces / gaps;
    let remainder = extra_spaces % gaps;
    let start_gap = line_index % gaps;

    for (index, piece) in line.pieces.iter().enumerate() {
        if index > 0 {
            let gap_index = index - 1;
            let bonus = usize::from((gap_index + gaps - start_gap) % gaps < remainder);
            rendered.push_str(&" ".repeat(1 + base_extra + bonus));
        }
        rendered.push_str(piece);
    }
    rendered
}

fn append_paragraph(output: &mut Vec<String>, segment: &[char], options: &WrapOptions) {
    let words = parse_words(segment);
    if words.is_empty() {
        return;
    }

    let solution = solve_paragraph(&words, options);
    let line_count = solution.lines.len();
    for (line_index, line) in solution.lines.iter().enumerate() {
        if options.justify && line_index + 1 < line_count && line.pieces.len() >= 2 {
            output.push(justify_line(line, options.width, line_index));
        } else {
            output.push(render_prejustify(line));
        }
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let scalar_count = input.chars().count();
    if scalar_count > 512 {
        return Err(WrapError::InputTooLong { chars: scalar_count });
    }

    let mut indexed = Vec::new();
    let mut byte_index = 0usize;
    for ch in input.chars() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        indexed.push((ch, byte_index));
        byte_index += ch.len_utf8();
    }

    for index in 0..indexed.len() {
        if indexed[index].0 != '~' {
            continue;
        }
        let side_valid = |side: isize| -> bool {
            if side < 0 || side as usize >= indexed.len() {
                return false;
            }
            let ch = indexed[side as usize].0;
            ch != '~' && !is_separator(ch)
        };
        if !side_valid(index as isize - 1) || !side_valid(index as isize + 1) {
            return Err(WrapError::InvalidMarker {
                index: indexed[index].1,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut index = 0usize;
    while index < indexed.len() {
        let ch = indexed[index].0;
        if ch == '\r' {
            normalized.push('\n');
            if index + 1 < indexed.len() && indexed[index + 1].0 == '\n' {
                index += 1;
            }
        } else {
            normalized.push(ch);
        }
        index += 1;
    }

    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut position = 0usize;

    while position < normalized.len() {
        if normalized[position] != '\n' {
            position += 1;
            continue;
        }

        let run_start = position;
        while position < normalized.len() && normalized[position] == '\n' {
            position += 1;
        }
        let run_length = position - run_start;
        if run_length >= 2 {
            append_paragraph(&mut output, &normalized[segment_start..run_start], options);
            for _ in 1..run_length {
                output.push(String::new());
            }
            segment_start = position;
        }
    }

    append_paragraph(&mut output, &normalized[segment_start..], options);
    Ok(output)
}
