#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Row {
    prefix: u8,
    payload: String,
    line: usize,
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.starts_with('0') {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let inner = line.strip_prefix("@@ -").and_then(|s| s.strip_suffix(" @@")).ok_or_else(bad)?;
    let (old, new) = inner.split_once(" +").ok_or_else(bad)?;
    if new.contains(' ') {
        return Err(bad());
    }
    let (a, b) = old.split_once(',').ok_or_else(bad)?;
    let (c, d) = new.split_once(',').ok_or_else(bad)?;
    let a = parse_number(a).ok_or_else(bad)?;
    let b = parse_number(b).ok_or_else(bad)?;
    let c = parse_number(c).ok_or_else(bad)?;
    let d = parse_number(d).ok_or_else(bad)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    Ok((a, b, c, d))
}

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 { start - 1 } else { start }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() < 3 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut prev_old_end = 0usize;
    let mut accumulated_delta: i128 = 0;

    while i < lines.len() {
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: i + 1 });
        }
        let header_line = i + 1;
        let (a, b, c, d) = parse_hunk_header(lines[i], header_line)?;
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);
        i += 1;

        let mut rows = Vec::new();
        while i < lines.len() && !lines[i].starts_with("@@") {
            let body_line = i + 1;
            let bytes = lines[i].as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: body_line });
            }
            if lines[i].contains('\r') {
                return Err(ApplyError::BadBody { line: body_line });
            }
            rows.push(Row {
                prefix: bytes[0],
                payload: lines[i][1..].to_owned(),
                line: body_line,
            });
            i += 1;
        }

        let old_seen = rows.iter().filter(|r| r.prefix != b'+').count();
        let new_seen = rows.iter().filter(|r| r.prefix != b'-').count();
        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }
        if old_start < prev_old_end
            || (new_start as i128) != (old_start as i128 + accumulated_delta)
            || (old_start as i128 + accumulated_delta) < 0
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        prev_old_end = old_start + b;
        accumulated_delta += d as i128 - b as i128;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() == 17 {
            return Err(ApplyError::TooLarge);
        }
    }
    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }
    let hunks = parse_patch(patch)?;

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1].split('\n').map(str::to_owned).collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;
    for (idx, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.prefix == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.prefix == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut starts = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    if row.prefix != b'+' {
                        old_payloads.push(row.payload.as_str());
                    }
                }
                let span = hunk.old_count - l - r;
                for s in 0..=working.len().saturating_sub(span) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    if old_payloads.len() != span
                        || old_payloads.iter().enumerate().any(|(j, p)| working[s + l + j] != *p)
                    {
                        continue;
                    }
                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            starts.clear();
                            starts.push((s, l, r));
                        }
                        Some(br) if rank < br => {
                            best_rank = Some(rank);
                            starts.clear();
                            starts.push((s, l, r));
                        }
                        Some(br) if rank == br => starts.push((s, l, r)),
                        _ => {}
                    }
                }
            }
        }

        if starts.is_empty() {
            return Err(ApplyError::NoMatch { hunk: idx + 1 });
        }
        let mut unique_starts: Vec<usize> = starts.iter().map(|x| x.0).collect();
        unique_starts.sort_unstable();
        unique_starts.dedup();
        if unique_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: idx + 1,
                first: unique_starts[0],
                second: unique_starts[1],
            });
        }
        let s = unique_starts[0];
        let (_, l, r) = starts
            .into_iter()
            .filter(|x| x.0 == s)
            .min_by_key(|x| x.1)
            .unwrap();

        let mut replacement = Vec::new();
        for row in &hunk.rows[l..hunk.rows.len() - r] {
            if row.prefix != b'-' {
                replacement.push(row.payload.clone());
            }
        }
        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut out = String::new();
    for line in working {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}
