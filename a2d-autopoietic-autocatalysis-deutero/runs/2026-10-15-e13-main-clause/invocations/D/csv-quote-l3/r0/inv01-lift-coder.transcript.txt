fn parse_csv_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record_index = 0usize;
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: unquoted, 1: quoted, 2: closed quote
    let mut opening_quote = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != 1 && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if state == 1 {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    state = 2;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8");
                field.push(ch);
                i += ch.len_utf8();
            }
            record_started = true;
            continue;
        }

        if state == 2 {
            if b == delimiter {
                fields.push(std::mem::take(&mut field));
                state = 0;
                record_started = true;
                i += 1;
                continue;
            }
            if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record_index += 1;
                state = 0;
                record_started = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            fields.push(std::mem::take(&mut field));
            record_started = true;
            i += 1;
            continue;
        }
        if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            record_index += 1;
            record_started = false;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }
        if b == quote {
            if field.is_empty() && !record_started {
                opening_quote = i;
                state = 1;
                record_started = true;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote {
                record: record_index,
                byte: i,
            });
        }

        let ch = input[i..].chars().next().expect("valid UTF-8");
        field.push(ch);
        record_started = true;
        i += ch.len_utf8();
    }

    if state == 1 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if record_started || !fields.is_empty() || !field.is_empty() || state == 2 {
        fields.push(field);
        records.push(fields);
    }
    Ok(records)
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    let records = parse_csv_with(input, config.delimiter, config.quote)?;
    let mut iter = records.into_iter();
    let (header, rows) = if config.headers {
        match iter.next() {
            Some(header) => (Some(header), iter.collect::<Vec<_>>()),
            None => (None, Vec::new()),
        }
    } else {
        (None, iter.collect::<Vec<_>>())
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in header.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name.as_str(), second);
            }
        }

        for (row_index, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start == b.len() {
        return Err(CellErrorKind::InvalidInteger);
    }
    if b[start] == b'0' {
        if start != 0 || b.len() != 1 {
            return Err(CellErrorKind::InvalidInteger);
        }
    } else if !(b'1'..=b'9').contains(&b[start])
        || !b[start + 1..].iter().all(u8::is_ascii_digit)
    {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_csv_with(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record_index = 0usize;
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start/unquoted, 1: quoted, 2: quote closed
    let mut opening_quote = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != 1 && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if state == 1 {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    state = 2;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8");
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if state == 2 {
            if b == delimiter {
                fields.push(std::mem::take(&mut field));
                state = 0;
                i += 1;
                continue;
            }
            if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record_index += 1;
                state = 0;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            fields.push(std::mem::take(&mut field));
            state = 0;
            i += 1;
            continue;
        }
        if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            record_index += 1;
            state = 0;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }
        if b == quote {
            if field.is_empty() && state == 0 {
                opening_quote = i;
                state = 1;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote {
                record: record_index,
                byte: i,
            });
        }

        let ch = input[i..].chars().next().expect("valid UTF-8");
        field.push(ch);
        state = 0;
        i += ch.len_utf8();
    }

    if state == 1 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if !fields.is_empty() || !field.is_empty() || state == 2 || i > 0 {
        fields.push(field);
        records.push(fields);
    }
    Ok(records)
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    let records = parse_csv_with(input, config.delimiter, config.quote)?;
    let mut iter = records.into_iter();
    let (header, rows) = if config.headers {
        match iter.next() {
            Some(header) => (Some(header), iter.collect::<Vec<_>>()),
            None => (None, Vec::new()),
        }
    } else {
        (None, iter.collect::<Vec<_>>())
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();

                        second,

                    record: row_index + 1,

                    actual: row.len(),
                });
            }

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start == b.len() {
        return Err(CellErrorKind::InvalidInteger);
    }
    if b[start] == b'0' {
        if start != 0 || b.len() != 1 {
            return Err(CellErrorKind::InvalidInteger);
        }
    } else if !(b'1'..=b'9').contains(&b[start])
        || !b[start + 1..].iter().all(u8::is_ascii_digit)
    {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    let Some(dot) = b.iter().position(|&x| x == b'.') else {
        return Err(CellErrorKind::InvalidDecimal);
    };
    if dot < start + 1 || b.len() != dot + 3 {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let integral = &b[start..dot];
    let fraction = &b[dot + 1..];
    let valid_integral = if integral == b"0" {
        true
    } else {
        integral.first().is_some_and(|x| (b'1'..=b'9').contains(x))
            && integral[1..].iter().all(u8::is_ascii_digit)
    };
    if !valid_integral || !fraction.iter().all(u8::is_ascii_digit) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    if start == 1 && integral == b"0" && fraction == b"00" {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let magnitude = s[start..]
        .replace('.', "")
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let signed = if start == 1 { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
