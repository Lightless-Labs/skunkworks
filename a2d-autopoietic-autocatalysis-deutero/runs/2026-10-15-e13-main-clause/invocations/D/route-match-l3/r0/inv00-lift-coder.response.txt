#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded segments are ASCII"))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn split_head<'a>(input: &'a str) -> Result<(&'a str, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok(("/", false));
    }

    let trailing = input.ends_with('/');
    let head = if trailing {
        &input[..input.len() - 1]
    } else {
        input
    };

    let mut start = 1;
    for (i, b) in head.bytes().enumerate().skip(1) {
        if b == b'/' {
            if i == start {
                return Err(start);
            }
            start = i + 1;
        }
    }
    if start == head.len() {
        return Err(start);
    }
    Ok((head, trailing))
}

fn pattern_segments(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), usize> {
    let (head, trailing) = split_head(pattern)?;
    if head == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut rank = Vec::new();
    let mut start = 1;
    let parts: Vec<(usize, &str)> = head[1..]
        .split('/')
        .scan(1usize, |offset, part| {
            let current = *offset;
            *offset += part.len() + 1;
            Some((current, part))
        })
        .collect();

    for (index, (at, part)) in parts.iter().enumerate() {
        if let Some(name) = part.strip_prefix(':') {
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(*at);
            }
            segments.push(Segment::Param(name.to_string()));
            rank.push(1);
        } else if let Some(name) = part.strip_prefix('*') {
            if !valid_capture_name(name)
                || !names.insert(name.to_string())
                || index + 1 != parts.len()
            {
                return Err(*at);
            }
            segments.push(Segment::Wildcard(name.to_string()));
            rank.push(0);
        } else {
            let decoded = decode_segment(part, *at)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }
    let _ = start;
    Ok((segments, trailing, rank))
}

fn parse_request_path(path: &str) -> Result<(Vec<String>, bool), usize> {
    let end = path
        .char_indices()
        .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
        .unwrap_or(path.len());
    let head = &path[..end];
    let (head, trailing) = split_head(head)?;
    if head == "/" {
        return Ok((Vec::new(), false));
    }

    let mut decoded = Vec::new();
    let mut offset = 1;
    for part in head[1..].split('/') {
        decoded.push(decode_segment(part, offset)?);
        offset += part.len() + 1;
    }
    Ok((decoded, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing, rank) =
            pattern_segments(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (request_segments, request_trailing) =
            parse_request_path(path).map_err(|at| Error::InvalidPath { at })?;

        let path_candidates: Vec<&Route> = self
            .routes
            .iter()
            .filter(|route| {
                (self.policy != SlashPolicy::Strict || route.trailing == request_trailing)
                    && route_matches(&route.segments, &request_segments)
            })
            .collect();

        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<&Route> = if method == "HEAD" {
            if path_candidates.iter().any(|r| r.method == "HEAD") {
                path_candidates
                    .iter()
                    .copied()
                    .filter(|r| r.method == "HEAD")
                    .collect()
            } else {
                path_candidates
                    .iter()
                    .copied()
                    .filter(|r| r.method == "GET")
                    .collect()
            }
        } else {
            path_candidates
                .iter()
                .copied()
                .filter(|r| r.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = path_candidates
                .iter()
                .map(|r| r.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut winner: Option<(&Route, Vec<(String, String)>)> = None;
        for route in eligible {
            let Some(params) = capture(&route.segments, &request_segments) else {
                continue;
            };
            let replace = match winner.as_ref() {
                None => true,
                Some((best, _)) => route.rank > best.rank,
            };
            if replace {
                winner = Some((route, params));
            }
        }

        let (route, params) = winner.ok_or(Error::NotFound)?;
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !request_segments.is_empty()
        {
            let head_end = path
                .char_indices()
                .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
                .unwrap_or(path.len());
            let mut original = path[..head_end].to_string();
            if route.trailing {
                original.push('/');
            } else if original.ends_with('/') {
                original.pop();
            }
            Some(original)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}

fn route_matches(pattern: &[Segment], request: &[String]) -> bool {
    let mut i = 0;
    for segment in pattern {
        match segment {
            Segment::Wildcard(_) => return true,
            Segment::Static(value) => {
                if request.get(i) != Some(value) {
                    return false;
                }
            }
            Segment::Param(_) => {
                if request.get(i).is_none() {
                    return false;
                }
            }
        }
        i += 1;
    }
    i == request.len()
}

fn capture(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;
    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if request.get(i)? != value {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(i)?.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), request[i..].join("/")));
                i = request.len();
            }
        }
    }
    if i == request.len() {
        Some(params)
    } else {
        None
    }
}
