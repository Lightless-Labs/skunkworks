use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Byte {
    value: u8,
    origin: usize,
}

#[derive(Clone)]
struct CfwsRun {
    start: usize,
    end: usize,
    origin: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<Byte>, Vec<CfwsRun>), EmailError> {
    let mut retained = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected_quote = false;

    while i < input.len() {
        let b = input[i];

        if protected_quote {
            retained.push(Byte { value: b, origin: i });
            if b == b'\\' {
                if i + 1 < input.len() {
                    retained.push(Byte {
                        value: input[i + 1],
                        origin: i + 1,
                    });
                    i += 2;
                    continue;
                }
                i += 1;
                continue;
            }
            if b == b'"' {
                protected_quote = false;
            }
            i += 1;
            continue;
        }

        if retained.is_empty() && b == b'"' {
            protected_quote = true;
            retained.push(Byte { value: b, origin: i });
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'\r' || b == b'\n' || b == b'(' {
            let start = retained.len();
            let origin = i;

            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];

                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    let outer = i;
                    let mut depth = 1usize;
                    i += 1;
                    while i < input.len() {
                        let x = input[i];
                        if x == b'\\' {
                            if i + 1 >= input.len()
                                || !(0x20..=0x7e).contains(&input[i + 1])
                            {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if x == b'\r' || x == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if x == b'(' {
                            depth += 1;
                            if depth > 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            i += 1;
                        } else if x == b')' {
                            depth -= 1;
                            i += 1;
                            if depth == 0 {
                                break;
                            }
                        } else {
                            let valid = (0x20..=0x7e).contains(&x) || x == b'\t';
                            if !valid {
                                return Err(EmailError::CommentChar { at: i });
                            }
                            i += 1;
                        }
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    continue;
                }
                break;
            }
            runs.push(CfwsRun {
                start,
                end: retained.len(),
                origin,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        retained.push(Byte { value: b, origin: i });
        i += 1;
    }

    Ok((retained, runs))
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_with_policy(input: &str, policy: &Policy, allow_single: bool) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (bytes, runs) = cleanup(raw)?;
    for run in &runs {
        let left_at = run.start > 0 && bytes[run.start - 1].value == b'@';
        let right_at = run.end < bytes.len() && bytes[run.end].value == b'@';
        let boundary = run.start == 0 || run.end == bytes.len();
        if !boundary && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.origin });
        }
    }
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if bytes[0].value == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i].value == b'\\' {
                i += 2;
            } else if bytes[i].value == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: bytes[0].origin,
            });
        }
    }

    let protected_end = quote_close.unwrap_or(0);
    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i].value == b'@' && (quote_close.is_none() || i > protected_end) {
            separators.push(i);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: bytes[separators[1]].origin,
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: bytes[sep].origin,
        });
    }

    let mut decoded = Vec::<(u8, usize)>::new();
    let quoted = bytes[0].value == b'"';
    if !quoted {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: bytes[64].origin,
            });
        }
        let mut previous_dot = false;
        for i in 0..sep {
            let b = bytes[i].value;
            if b == b'.' {
                if i == 0 || i + 1 == sep || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: bytes[i].origin,
                    });
                }
                previous_dot = true;
            } else if is_atom_char(b) {
                previous_dot = false;
            } else {
                return Err(EmailError::LocalChar {
                    at: bytes[i].origin,
                });
            }
            decoded.push((b, bytes[i].origin));
        }
    } else {
        let close = quote_close.unwrap();
        if close + 1 != sep {
            let at = if close + 1 < bytes.len() {
                bytes[close + 1].origin
            } else {
                input.len()
            };
            return Err(EmailError::TrailingLocal { at });
        }
        let mut i = 1;
        while i < close {
            let b = bytes[i].value;
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape {
                        at: bytes[i].origin,
                    });
                }
                let escaped = bytes[i + 1].value;
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: bytes[i].origin,
                    });
                }
                decoded.push((escaped, bytes[i].origin));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: bytes[i].origin,
                    });
                }
                decoded.push((b, bytes[i].origin));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: bytes[close].origin,
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded[64].1,
            });
        }
    }

    let domain_start_index = sep + 1;
    let domain_start = bytes
        .get(domain_start_index)
        .map(|b| b.origin)
        .unwrap_or(input.len());
    let mut domain_end = bytes.len();
    let mut root_dot = None;
    if domain_end > domain_start_index && bytes[domain_end - 1].value == b'.' {
        root_dot = Some(bytes[domain_end - 1].origin);
        domain_end -= 1;
    }
    if domain_start_index == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_end - domain_start_index > 253 {
        return Err(EmailError::DomainTooLong {
            at: bytes[domain_start_index + 253].origin,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start_index;
    for i in domain_start_index..domain_end {
        if bytes[i].value == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }
    labels.push((label_start, domain_end));

    for (start, end) in &labels {
        if start == end {
            let at = if *start == domain_start_index {
                bytes.get(*start).map(|b| b.origin).unwrap_or(domain_start)
            } else if *end == domain_end {
                root_dot.unwrap_or(input.len())
            } else {
                bytes[*start - 1].origin
            };
            return Err(EmailError::EmptyLabel { at });
        }
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: bytes[*start + 63].origin,
            });
        }
        if bytes[*start].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: bytes[*start].origin,
            });
        }
        if bytes[*end - 1].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: bytes[*end - 1].origin,
            });
        }
        for i in *start..*end {
            let b = bytes[i].value;
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: bytes[i].origin,
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end]
        .iter()
        .all(|b| b.value.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: bytes[last_start].origin,
        });
    }

    let mut domain = String::with_capacity(domain_end - domain_start_index);
    for b in &bytes[domain_start_index..domain_end] {
        domain.push((b.value as char).to_ascii_lowercase());
    }

    if policy.local_case == LocalCase::Lower {
        for (b, _) in &mut decoded {
            *b = b.to_ascii_lowercase();
        }
    }

    if policy.strip_subaddress {
        if let Some(plus) = decoded.iter().position(|(b, _)| *b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: decoded[plus].1,
                });
            }
            decoded.truncate(plus);
        }
    }

    let local_bytes: Vec<u8> = decoded.iter().map(|(b, _)| *b).collect();
    let local = String::from_utf8(local_bytes.clone()).unwrap();
    let safe_atom = valid_dot_atom(&local_bytes);
    let out_quoted = if safe_atom {
        if policy.unquote_safe { false } else { quoted }
    } else {
        true
    };

    Ok(Address {
        local,
        domain,
        quoted: out_quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), false)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, policy.allow_single_label)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut first_by_key: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = first_by_key.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    first_by_key.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
