#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit_value(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|n| n as u8)
}

fn encode_group(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        output.extend(encode_group(chunk[0], chunk[1], chunk[2]).map(char::from));
    }

    match chunks.remainder() {
        [] => {}
        [a] => {
            output.push(char::from(ALPHABET[(a >> 2) as usize]));
            output.push(char::from(ALPHABET[((a & 0x03) << 4) as usize]));
            output.push_str("==");
        }
        [a, b] => {
            output.push(char::from(ALPHABET[(a >> 2) as usize]));
            output.push(char::from(
                ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
            ));
            output.push(char::from(ALPHABET[((b & 0x0f) << 2) as usize]));
            output.push('=');
        }
        _ => unreachable!(),
    }

    output
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum GroupState {
    Collecting,
    PendingSecondPadding { offset: usize },
    FinishedPadding,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct DecoderState {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    position_two_padding: Option<usize>,
    group_start: usize,
    state: GroupState,
    next_offset: usize,
}

impl DecoderState {
    fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            position_two_padding: None,
            group_start: 0,
            state: GroupState::Collecting,
            next_offset: 0,
        }
    }

    fn scan_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.state == GroupState::FinishedPadding {
            return Err(Error::BadPadding { offset });
        }

        if let Some(padding_offset) = self.position_two_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.position_two_padding = None;
            self.group[3] = b'=';
            self.offsets[3] = offset;
            self.len = 4;
            let _ = padding_offset;
            self.emit_group(output)?;
            self.state = GroupState::FinishedPadding;
            return Ok(());
        }

        let position = self.len;
        if byte == b'=' {
            if position <= 1 {
                return Err(Error::BadPadding { offset });
            }
            if position == 2 {
                self.position_two_padding = Some(offset);
                self.state = GroupState::PendingSecondPadding { offset };
                return Ok(());
            }
            self.group[3] = b'=';
            self.offsets[3] = offset;
            self.len = 4;
            self.emit_group(output)?;
            self.state = GroupState::FinishedPadding;
            return Ok(());
        }

        let value = digit_value(byte).ok_or(Error::InvalidByte { offset, byte })?;
        if position == 0 {
            self.group_start = offset;
        }
        self.group[position] = value;
        self.offsets[position] = offset;
        self.len += 1;

        if self.len == 4 {
            self.emit_group(output)?;
            self.len = 0;
        }

        Ok(())
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let a = self.group[0];
        let b = self.group[1];

        if self.group[2] == b'=' {
            output.push((a << 2) | (b >> 4));
            self.len = 0;
            self.position_two_padding = None;
            return Ok(());
        }

        let c = digit_value(self.group[2]).ok_or(Error::BadPadding {
            offset: self.offsets[2],
        })?;

        output.push((a << 2) | (b >> 4));

        if self.group[3] == b'=' {
            output.push((b << 4) | (c >> 2));
            self.len = 0;
            self.position_two_padding = None;
            return Ok(());
        }

        let d = digit_value(self.group[3]).ok_or(Error::BadPadding {
            offset: self.offsets[3],
        })?;
        output.push((b << 4) | (c >> 2));
        output.push((c << 6) | d);
        self.len = 0;
        Ok(())
    }

    fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(offset) = self.position_two_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.len != 0 {
            return Err(Error::Truncated {
                offset: self.group_start,
            });
        }

        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let mut output = String::with_capacity(complete_len / 3 * 4);
        for chunk in combined[..complete_len].chunks_exact(3) {
            output.extend(encode_group(chunk[0], chunk[1], chunk[2]).map(char::from));
        }

        self.tail.clear();
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let output = match self.tail.as_slice() {
            [] => String::new(),
            [a] => {
                let mut s = String::new();
                s.push(char::from(ALPHABET[(a >> 2) as usize]));
                s.push(char::from(ALPHABET[((a & 0x03) << 4) as usize]));
                s.push_str("==");
                s
            }
            [a, b] => {
                let mut s = String::new();
                s.push(char::from(ALPHABET[(a >> 2) as usize]));
                s.push(char::from(
                    ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
                ));
                s.push(char::from(ALPHABET[((b & 0x0f) << 2) as usize]));
                s.push('=');
                s
            }
            _ => unreachable!(),
        };
        self.tail.clear();
        Ok(output)
    }
}

pub struct Decoder {
    state: DecoderState,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            state: DecoderState::new(),
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.state.next_offset;
            self.state.next_offset += 1;
            if let Err(error) = self.state.scan_byte(byte, offset, &mut output) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.state.finish() {
            Ok(output) => {
                self.finished = true;
                Ok(output)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }
}
