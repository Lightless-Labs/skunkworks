#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
}

#[derive(Debug, Clone)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { id, segments });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let decoded = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>, Vec<u8>)> = None;

        for route in &self.routes {
            if route.segments.len() != decoded.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut rank = Vec::with_capacity(decoded.len());
            let mut matched = true;

            for (pattern_segment, path_segment) in route.segments.iter().zip(&decoded) {
                match pattern_segment {
                    Segment::Static(value) => {
                        if value != path_segment {
                            matched = false;
                            break;
                        }
                        rank.push(2);
                    }
                    Segment::Param(name) => {
                        params.push((name.clone(), path_segment.clone()));
                        rank.push(1);
                    }
                }
            }

            if matched {
                let replace = match &best {
                    None => true,
                    Some((best_id, _, best_rank)) => {
                        rank > *best_rank || (rank == *best_rank && route.id < *best_id)
                    }
                };
                if replace {
                    best = Some((route.id, params, rank));
                }
            }
        }

        match best {
            Some((route, params, _)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let mut end = pattern.len();
    if pattern.ends_with('/') {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let mut start = 1;

    while start < end {
        let next = pattern[start..end]
            .find('/')
            .map(|i| start + i)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..next];
        if raw.starts_with(':') {
            let name = &raw[1..];
            let valid = !name.is_empty()
                && name.as_bytes()[0].is_ascii_lowercase()
                && name
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_');

            if !valid || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            let name = name.to_string();
            names.push(name.clone());
            segments.push(Segment::Param(name));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }

        start = next + 1;
    }

    if pattern.ends_with('/') {
        return Err(Error::InvalidPattern { at: pattern.len() - 1 });
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let mut end = path.len();
    if path.ends_with('/') {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let next = path[start..end]
            .find('/')
            .map(|i| start + i)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(&path[start..next], start, false)?);
        start = next + 1;
    }

    if path.ends_with('/') {
        return Err(Error::InvalidPath { at: path.len() - 1 });
    }

    Ok(segments)
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(invalid(offset + i, pattern));
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(offset + i, pattern));
            }
            out.push(value);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(invalid(offset + i, pattern));
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn invalid(at: usize, pattern: bool) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => unreachable!(),
    }
}
