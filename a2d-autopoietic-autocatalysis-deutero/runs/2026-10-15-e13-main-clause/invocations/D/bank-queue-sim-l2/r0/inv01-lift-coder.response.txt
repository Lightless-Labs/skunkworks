```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy_until: Option<(u64, u32)>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            idle_since: server.opens,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut arrival_index = 0;
    let mut finished_count = 0;
    let mut time: Option<u64> = None;

    loop {
        if finished_count == customers.len() && states.iter().all(|server| server.opened) {
            break;
        }

        let next_time = if let Some(current) = time {
            current
        } else {
            let mut next = None;
            for customer in &arrivals {
                if next.is_none_or(|t| customer.arrival < t) {
                    next = Some(customer.arrival);
                }
            }
            for server in &states {
                if !server.opened && next.is_none_or(|t| server.opens < t) {
                    next = Some(server.opens);
                }
            }
            let Some(next) = next else {
                break;
            };
            next
        };
        time = Some(next_time);

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, server)| {
                (server.busy_until.is_some_and(|(finish, _)| finish == next_time)).then_some(i)
            })
            .collect();
        finishing.sort_by_key(|&i| states[i].id);
        for i in finishing {
            let server_id = states[i].id;
            let (_, customer_id) = states[i]
                .busy_until
                .take()
                .expect("selected occupied server");
            states[i].idle_since = next_time;
            events.push(Event::Finish {
                time: next_time,
                customer: customer_id,
                server: server_id,
            });
            finished_count += 1;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, server)| {
                (!server.opened && server.opens == next_time).then_some(i)
            })
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: next_time,
                server: states[i].id,
            });
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == next_time {
            let customer = arrivals[arrival_index];
            events.push(Event::Arrive {
                time: next_time,
                customer: customer.id,
            });
            waiting.push(customer);
            arrival_index += 1;
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, server)| server.opened && server.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting[0];
            let server = &states[server_index];

            let after_service =
                next_time
                    .checked_add(customer.service)
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        server: server.id,
                        time: next_time,
                    })?;
            let finish = after_service
                .checked_add(server.overhead)
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: next_time,
                })?;

            let server_id = states[server_index].id;
            events.push(Event::Start {
                time: next_time,
                customer: customer.id,
                server: server_id,
                finish,
            });
            waiting.remove(0);

            if finish == next_time {
                events.push(Event::Finish {
                    time: next_time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = next_time;
                finished_count += 1;
            } else {
                states[server_index].busy_until = Some((finish, customer.id));
            }
        }

        let mut next = None;
        for customer in arrivals.iter().skip(arrival_index) {
            if next.is_none_or(|t| customer.arrival < t) {
                next = Some(customer.arrival);
            }
        }
        for server in &states {
            if !server.opened && next.is_none_or(|t| server.opens < t) {
                next = Some(server.opens);
            }
            if let Some((finish, _)) = server.busy_until {
                if next.is_none_or(|t| finish < t) {
                    next = Some(finish);
                }
            }
        }
        match next {
            Some(t) => time = Some(t),
            None => break,
        }
    }

    Ok(events)
}
```
