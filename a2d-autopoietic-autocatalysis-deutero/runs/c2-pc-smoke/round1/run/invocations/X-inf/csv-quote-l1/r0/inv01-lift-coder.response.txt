#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut quoted = false;
    let mut quote_closed = false;
    let mut quote_open_byte = 0usize;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if quoted {
            if byte == b'"' {
                if pos + 1 < bytes.len() && bytes[pos + 1] == b'"' {
                    field.push('"');
                    pos += 2;
                } else {
                    quoted = false;
                    quote_closed = true;
                    pos += 1;
                }
            } else {
                let start = pos;
                while pos < bytes.len()
                    && bytes[pos] != b'"'
                    && bytes[pos] != 0
                {
                    pos += 1;
                }
                field.push_str(&input[start..pos]);
            }
            continue;
        }

        if byte == b'\r' {
            if pos + 1 >= bytes.len() || bytes[pos + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: pos,
                });
            }
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            quote_closed = false;
            pos += 2;
            continue;
        }

        if quote_closed {
            match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    quote_closed = false;
                    pos += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    quote_closed = false;
                    pos += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
            continue;
        }

        match byte {
            b'"' if field.is_empty() => {
                quoted = true;
                quote_open_byte = pos;
                pos += 1;
            }
            b'"' => {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
            b',' => {
                record.push(std::mem::take(&mut field));
                pos += 1;
            }
            b'\n' => {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                pos += 1;
            }
            _ => {
                let start = pos;
                while pos < bytes.len()
                    && bytes[pos] != b'"'
                    && bytes[pos] != b','
                    && bytes[pos] != b'\n'
                    && bytes[pos] != b'\r'
                    && bytes[pos] != 0
                {
                    pos += 1;
                }
                field.push_str(&input[start..pos]);
            }
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open_byte,
        });
    }

    if !record.is_empty() || !field.is_empty() || quote_closed || bytes.last() == Some(&b',') {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
