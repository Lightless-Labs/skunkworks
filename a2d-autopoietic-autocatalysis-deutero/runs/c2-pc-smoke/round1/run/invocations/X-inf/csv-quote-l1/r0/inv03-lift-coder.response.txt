#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{FEFF}') { 3 } else { 0 };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !matches!(state, State::Quoted { .. }) && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record, byte: i });
            }

            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            state = State::FieldStart;
            record_started = false;
            record += 1;
            i += 2;
            continue;
        }

        match state {
            State::FieldStart => match b {
                b'"' => {
                    state = State::Quoted { opening: i };
                    record_started = true;
                    i += 1;
                }
                b',' => {
                    fields.push(String::new());
                    record_started = true;
                    i += 1;
                }
                b'\n' => {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    record_started = false;
                    record += 1;
                    i += 1;
                }
                _ => {
                    field.push(b as char);
                    state = State::Unquoted;
                    record_started = true;
                    i += 1;
                    while i < bytes.len() && bytes[i].is_ascii() && !matches!(bytes[i], b',' | b'\n' | b'\r' | b'"' | 0) {
                        field.push(bytes[i] as char);
                        i += 1;
                    }
                    if i < bytes.len() && !bytes[i].is_ascii() {
                        let remaining = &input[i..];
                        let ch = remaining.chars().next().unwrap();
                        field.push(ch);
                        i += ch.len_utf8();
                    }
                }
            },
            State::Unquoted => match b {
                b'"' => return Err(CsvError::UnexpectedQuote { record, byte: i }),
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    record_started = true;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    state = State::FieldStart;
                    record_started = false;
                    record += 1;
                    i += 1;
                }
                _ => {
                    if b.is_ascii() {
                        field.push(b as char);
                        i += 1;
                    } else {
                        let ch = input[i..].chars().next().unwrap();
                        field.push(ch);
                        i += ch.len_utf8();
                    }
                }
            },
            State::Quoted { opening } => {
                if b == b'"' {
                    if bytes.get(i + 1) == Some(&b'"') {
                        field.push('"');
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else if b.is_ascii() {
                    field.push(b as char);
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
                if i == bytes.len() {
                    if let State::Quoted { opening } = state {
                        return Err(CsvError::UnclosedQuote { record, byte: opening });
                    }
                }
            }
            State::AfterQuote => match b {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    record_started = true;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    state = State::FieldStart;
                    record_started = false;
                    record += 1;
                    i += 1;
                }
                _ => return Err(CsvError::TrailingByte { record, byte: i }),
            },
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote { record, byte: opening });
    }

    if record_started || !fields.is_empty() || !field.is_empty() || matches!(state, State::AfterQuote) {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}
