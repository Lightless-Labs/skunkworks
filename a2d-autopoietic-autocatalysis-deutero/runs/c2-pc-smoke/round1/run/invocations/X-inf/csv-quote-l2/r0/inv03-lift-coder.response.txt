#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|column| column == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|column| column == name)
            }
        }
    }
}

#[derive(Clone, Copy)]
enum ParseMode {
    Unquoted,
    Quoted { opening: usize },
    AfterQuote,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut index = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if index == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut mode = ParseMode::Unquoted;
    let mut record_index = 0usize;
    let mut record_started = false;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        let inside_quotes = matches!(mode, ParseMode::Quoted { .. });

        if !inside_quotes && byte == b'\r' {
            if bytes.get(index + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: index,
                });
            }

            record.push(String::from_utf8(field).expect("input is valid UTF-8"));
            field = Vec::new();
            records.push(record);
            record = Vec::new();
            record_started = false;
            mode = ParseMode::Unquoted;
            index += 2;
            record_index += 1;
            continue;
        }

        if !inside_quotes && byte == b'\n' {
            record.push(String::from_utf8(field).expect("input is valid UTF-8"));
            field = Vec::new();
            records.push(record);
            record = Vec::new();
            record_started = false;
            mode = ParseMode::Unquoted;
            index += 1;
            record_index += 1;
            continue;
        }

        match mode {
            ParseMode::Unquoted => {
                if byte == delimiter {
                    record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    record_started = true;
                    index += 1;
                } else if byte == quote {
                    if field.is_empty() && !record_started {
                        mode = ParseMode::Quoted { opening: index };
                        record_started = true;
                        index += 1;
                    } else {
                        return Err(CsvError::UnexpectedQuote {
                            record: record_index,
                            byte: index,
                        });
                    }
                } else {
                    field.push(byte);
                    record_started = true;
                    index += 1;
                }
            }
            ParseMode::Quoted { .. } => {
                if byte == quote {
                    if bytes.get(index + 1) == Some(&quote) {
                        field.push(quote);
                        index += 2;
                    } else {
                        mode = ParseMode::AfterQuote;
                        index += 1;
                    }
                } else {
                    field.push(byte);
                    index += 1;
                }
            }
            ParseMode::AfterQuote => {
                if byte == delimiter {
                    record.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    record_started = true;
                    mode = ParseMode::Unquoted;
                    index += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            }
        }
    }

    if let ParseMode::Quoted { opening } = mode {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started || !field.is_empty() || !record.is_empty() {
        record.push(String::from_utf8(field).expect("input is valid UTF-8"));
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut header = None;
    let mut rows = records;

    if config.headers {
        if !rows.is_empty() {
            let parsed_header = rows.remove(0);

            if config.duplicates == DuplicatePolicy::Reject {
                let mut seen: Vec<(String, usize)> = Vec::new();
                for (second, name) in parsed_header.iter().enumerate() {
                    if let Some((_, first)) = seen.iter().find(|(prior, _)| prior == name) {
                        return Err(CsvError::DuplicateHeader {
                            name: name.clone(),
                            first: *first,
                            second,
                        });
                    }
                    seen.push((name.clone(), second));
                }
            }

            for (offset, row) in rows.iter().enumerate() {
                if row.len() != parsed_header.len() {
                    return Err(CsvError::HeaderWidth {
                        record: offset + 1,
                        expected: parsed_header.len(),
                        actual: row.len(),
                    });
                }
            }

            header = Some(parsed_header);
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }

        let result = parse_records(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
