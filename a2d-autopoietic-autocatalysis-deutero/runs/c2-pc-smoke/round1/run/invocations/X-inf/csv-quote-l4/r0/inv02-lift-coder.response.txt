use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

fn valid_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut i = start;
    let mut record_index = 0usize;

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted(usize),
        Closed,
    }
    let mut state = State::Start;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        let outside = !matches!(state, State::Quoted(_));
        if outside && b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        match state {
            State::Quoted(open) => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote);
                        i += 2;
                    } else {
                        state = State::Closed;
                        i += 1;
                    }
                } else {
                    field.push(b);
                    i += 1;
                }
                let _ = open;
            }
            State::Start => {
                if b == quote {
                    state = State::Quoted(i);
                    i += 1;
                } else if b == delimiter {
                    row.push(String::new());
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    i += if b == b'\r' { 2 } else { 1 };
                    state = State::Start;
                } else {
                    field.push(b);
                    state = State::Unquoted;
                    i += 1;
                }
            }
            State::Unquoted => {
                if b == quote {
                    return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
                } else if b == delimiter {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = State::Start;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    state = State::Start;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    field.push(b);
                    i += 1;
                }
            }
            State::Closed => {
                if b == delimiter {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = State::Start;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    state = State::Start;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte { record: record_index, byte: i });
                }
            }
        }
    }

    if let State::Quoted(open) = state {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: open });
    }
    if !matches!(state, State::Start) || !row.is_empty() || !records.is_empty() {
        row.push(String::from_utf8(field).unwrap());
        records.push(row);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name, second);
    }
    None
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(e) = duplicate_header(h) {
                return Err(e);
            }
        }
        for (i, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }
    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        valid_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 || b.last().is_none() { return false; }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(u8::is_ascii_digit) { return false; }
    let int = &b[start..dot];
    if int == b"0" {
        return s != "-0.00";
    }
    !int.is_empty()
        && (b'1'..=b'9').contains(&int[0])
        && int[1..].iter().all(u8::is_ascii_digit)
}

fn eval_cell(
    text: &str,
    spec: &ColumnSpec,
) -> Result<Value, CellErrorKind> {
    if text.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(text.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(text) { return Err(CellErrorKind::InvalidInteger); }
            text.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(text) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = text.starts_with('-');
            let body = if neg { &text[1..] } else { text };
            let mut parts = body.split('.');
            let whole = parts.next().unwrap().parse::<i128>().unwrap();
            let frac = parts.next().unwrap().parse::<i128>().unwrap();
            let value = whole * 100 + frac;
            let signed = if neg { -value } else { value };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match text {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut out_rows = Vec::new();
    let mut errors = Vec::new();
    for (ri, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut failed = false;
        for (si, spec) in columns.iter().enumerate() {
            if failed && policy != ErrorPolicy::Collect {
                out.push(None);
                continue;
            }
            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let result = if index.is_none() {
                Err((None, CellErrorKind::UnknownColumn))
            } else {
                let index = index.unwrap();
                match row.get(index) {
                    None => Err((Some(index), CellErrorKind::MissingCell)),
                    Some(cell) => eval_cell(cell, spec).map_err(|k| (Some(index), k)),
                }
            };
            match result {
                Ok(value) => out.push(Some(value)),
                Err((column, kind)) => {
                    errors.push(CellError { row: ri, spec: si, column, kind });
                    out.push(None);
                    failed = true;
                    if policy == ErrorPolicy::Stop {
                        for _ in si + 1..columns.len() { out.push(None); }
                        out_rows.push(out);
                        return TypedResult { rows: out_rows, errors };
                    }
                    if policy == ErrorPolicy::FirstPerRow {
                        for _ in si + 1..columns.len() { out.push(None); }
                        break;
                    }
                }
            }
        }
        out_rows.push(out);
    }
    TypedResult { rows: out_rows, errors }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    pub line_ending: LineEnding,

}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn rows_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let c = x.as_bytes().cmp(y.as_bytes());
        if c != std::cmp::Ordering::Equal { return c; }
    }
    a.len().cmp(&b.len())
}

fn encode_field(
    s: &str,
    delimiter: u8,
    quote: u8,
    first_output: bool,
    sole: bool,
) -> String {
    let bytes = s.as_bytes();
    let starts_bom = first_output && bytes.starts_with(&[0xef, 0xbb, 0xbf]);
    let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let must_quote = s.as_bytes().contains(&delimiter)
        || s.as_bytes().contains(&quote)
        || s.as_bytes().contains(&b'\r')
        || s.as_bytes().contains(&b'\n')
        || edge_space
        || starts_bom
        || (sole && s.is_empty());
    if !must_quote {
        return s.to_string();
    }
    let mut out = String::new();
    out.push(quote as char);
    for ch in s.chars() {
        if ch as u32 == quote as u32 { out.push(quote as char); }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    valid_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header { original_records.push(header); }
    original_records.extend(table.rows.iter().map(Vec::as_slice));

    for (record, row) in original_records.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            if let Some(byte) = cell.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    if config.mode == WriteMode::Preserve {
        if let Some(h) = &table.header { output_records.push(h.clone()); }
        output_records.extend(table.rows.clone());
    } else if let Some(header) = &table.header {
        let mut latest: HashMap<&str, usize> = HashMap::new();
        for (i, name) in header.iter().enumerate() {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(&first) = latest.get(name.as_str()) {
                    return Err(WriteError::DuplicateHeader {
                        name: name.clone(), first, second: i,
                    });
                }
            }
            latest.insert(name, i);
        }
        let mut retained: Vec<usize> = match table.duplicates {
            DuplicatePolicy::KeepLast => latest.values().copied().collect(),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                let mut seen = HashMap::new();
                let mut v = Vec::new();
                for (i, n) in header.iter().enumerate() {
                    if !seen.contains_key(n.as_str()) {
                        seen.insert(n.as_str(), ());
                        v.push(i);
                    }
                }
                v
            }
        };
        retained.sort_by(|a, b| header[*a].as_bytes().cmp(header[*b].as_bytes()));
        let projected_header = retained.iter().map(|i| header[*i].clone()).collect::<Vec<_>>();
        output_records.push(projected_header);
        let mut projected_rows = table.rows.iter().map(|row| {
            retained.iter().map(|i| row[*i].clone()).collect::<Vec<_>>()
        }).collect::<Vec<_>>();
        projected_rows.sort_by(|a, b| rows_cmp(a, b));
        output_records.extend(projected_rows);
    } else {
        let mut rows = table.rows.clone();
        rows.sort_by(|a, b| rows_cmp(a, b));
        output_records = rows;
    }

    if output_records.is_empty() { return Ok(String::new()); }
    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical || config.line_ending == LineEnding::Lf {
        "\n"
    } else {
        "\r\n"
    };
    let mut out = String::new();
    for (ri, row) in output_records.iter().enumerate() {
        if ri > 0 { out.push_str(ending); }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { out.push(config.delimiter as char); }
            let first = ri == 0 && ci == 0;
            let sole = row.len() == 1;
            out.push_str(&encode_field(field, config.delimiter, config.quote, first, sole));
        }
    }
    if canonical || config.final_terminator { out.push_str(ending); }
    Ok(out)
}
