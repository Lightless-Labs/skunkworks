#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0 = field start, 1 = unquoted, 2 = quoted, 3 = after quote
    let mut quote_open = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record: records.len(), byte: i });
        }

        if state != 2 && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record: records.len(), byte: i });
            }
        }

        if state == 2 {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                state = 3;
                i += 1;
                continue;
            }
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
            continue;
        }

        if state == 3 {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                state = 0;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                state = 0;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record: records.len(), byte: i });
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            state = 0;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = 0;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if state == 0 {
                state = 2;
                quote_open = i;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record: records.len(), byte: i });
            }
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            state = 1;
            i += ch.len_utf8();
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote { record: records.len(), byte: quote_open });
    }
    if !record.is_empty() || !field.is_empty() || state != 0 {
        record.push(field);
        records.push(record);
    } else if !input.is_empty() && !ends_in_terminator(bytes) {
        record.push(String::new());
        records.push(record);
    }

    Ok(records)
}

fn ends_in_terminator(bytes: &[u8]) -> bool {
    bytes.last() == Some(&b'\n')
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: std::collections::HashMap<&str, usize> = std::collections::HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name, second);
    }
    None
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    let header = if config.headers { iter.next() } else { None };
    let rows: Vec<Vec<String>> = iter.collect();

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(e) = duplicate_header(h) {
                return Err(e);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    FirstPerRow,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len() && start == 0;
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 || b[start..].contains(&b'-') {
        return false;
    }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let int = &b[start..dot];
    if int.is_empty() {
        return false;
    }
    let integral_valid = if int == b"0" {
        true
    } else {
        (b'1'..=b'9').contains(&int[0]) && int[1..].iter().all(u8::is_ascii_digit)
    };
    if !integral_valid {
        return false;
    }
    !(start == 1 && int == b"0" && b[dot + 1..] == *b"00")
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let neg = s.starts_with('-');
    let unsigned = if neg { &s[1..] } else { s };
    let dot = unsigned.len() - 3;
    let integral: i128 = unsigned[..dot].parse().unwrap();
    let fractional: i128 = unsigned[dot + 1..].parse().unwrap();
    let magnitude = integral * 100 + fractional;
    let signed = if neg { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn convert_cell(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => parse_integer(s).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(s).map(Value::Decimal),
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut output = Vec::new();
    let mut errors = Vec::new();
    let mut stopped = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stopped {
            break;
        }
        let mut out_row = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                out_row.push(None);
                continue;
            }
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let Some(column) = resolved else {
                out_row.push(None);
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                row_failed = true;
                if policy == ErrorPolicy::Stop {
                    stopped = true;
                }
                continue;
            };
            let Some(cell) = row.get(column) else {
                out_row.push(None);
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column),
                    kind: CellErrorKind::MissingCell,
                });
                row_failed = true;
                if policy == ErrorPolicy::Stop {
                    stopped = true;
                }
                continue;
            };
            match convert_cell(cell, spec) {
                Ok(value) => out_row.push(Some(value)),
                Err(kind) => {
                    out_row.push(None);
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column),
                        kind,
                    });
                    row_failed = true;
                    if policy == ErrorPolicy::Stop {
                        stopped = true;
                    }
                }
            }
            if row_failed && policy != ErrorPolicy::Collect {
                break;
            }
        }

        if row_failed && policy != ErrorPolicy::Collect {
            while out_row.len() < columns.len() {
                out_row.push(None);
            }
        }
        output.push(out_row);
    }

    TypedResult { rows: output, errors }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    original_records.extend(table.rows.iter());

    for (record, row) in original_records.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            if let Some(byte) = cell.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (idx, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: idx + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(CsvError::DuplicateHeader { name, first, second }) =
            duplicate_header(table.header.as_ref().unwrap())
        {
            return Err(WriteError::DuplicateHeader { name, first, second });
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut selected: std::collections::BTreeMap<String, usize> =
                    std::collections::BTreeMap::new();
                for (idx, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => {
                            selected.insert(name.clone(), idx);
                        }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            selected.entry(name.clone()).or_insert(idx);
                        }
                    }
                }
                let indices: Vec<usize> = selected.values().copied().collect();
                let canonical_header: Vec<String> =
                    selected.keys().cloned().collect();
                records.push(canonical_header);
                let mut rows: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| indices.iter().map(|&idx| row[idx].clone()).collect())
                    .collect();
                rows.sort();
                records.extend(rows);
            } else {
                let mut rows = table.rows.clone();
                rows.sort();
                records = rows;
            }
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            if config.mode == WriteMode::Canonical || config.line_ending == LineEnding::Lf {
                out.push('\n');
            } else {
                out.push_str("\r\n");
            }
        }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            let starts_bom = record_index == 0 && column_index == 0 && field.starts_with('\u{feff}');
            let sole_empty = row.len() == 1 && field.is_empty();
            let bytes = field.as_bytes();
            let needs_quote = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || field.starts_with(' ')
                || field.starts_with('\t')
                || field.ends_with(' ')
                || field.ends_with('\t')
                || starts_bom
                || sole_empty;
            if needs_quote {
                out.push(config.quote as char);
                for &b in bytes {
                    if b == config.quote {
                        out.push(config.quote as char);
                    }
                    out.push(b as char);
                }
                out.push(config.quote as char);
            } else {
                out.push_str(field);
            }
        }
    }

    let final_terminator = config.mode == WriteMode::Canonical || config.final_terminator;
    if final_terminator {
        if config.mode == WriteMode::Canonical || config.line_ending == LineEnding::Lf {
            out.push('\n');
        } else {
            out.push_str("\r\n");
        }
    }
    Ok(out)
}
