use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    if pos == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if state != 2 && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        match state {
            0 => {
                if b == delimiter {
                    record.push(String::new());
                    pos += 1;
                } else if b == quote {
                    quote_start = pos;
                    state = 2;
                    pos += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    pos += if b == b'\r' { 2 } else { 1 };
                } else {
                    field.push(b);
                    state = 1;
                    pos += 1;
                }
            }
            1 => {
                if b == delimiter {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = 0;
                    pos += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = 0;
                    pos += if b == b'\r' { 2 } else { 1 };
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                } else {
                    field.push(b);
                    pos += 1;
                }
            }
            2 => {
                if b == quote {
                    if bytes.get(pos + 1) == Some(&quote) {
                        field.push(quote);
                        pos += 2;
                    } else {
                        state = 3;
                        pos += 1;
                    }
                } else {
                    field.push(b);
                    pos += 1;
                }
            }
            _ => {
                if b == delimiter {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = 0;
                    pos += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = 0;
                    pos += if b == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }
    if !record.is_empty() || !field.is_empty() || state != 0 {
        record.push(String::from_utf8(field).unwrap());
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn apply_headers(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    let header = if config.headers { iter.next() } else { None };
    let rows: Vec<Vec<String>> = iter.collect();

    if let Some(ref names) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut first = HashMap::<&str, usize>::new();
            for (second, name) in names.iter().enumerate() {
                if let Some(&first_index) = first.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: first_index,
                        second,
                    });
                }
                first.insert(name, second);
            }
        }
        for (offset, row) in rows.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: offset + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let parsed = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| apply_headers(records, self.config));
        self.result = Some(parsed.clone());
        parsed
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    if b.len() < start + 4 {
        return false;
    }
    let dot = b.len() - 3;
    if b[dot] != b'.'
        || !b[dot + 1..].iter().all(u8::is_ascii_digit)
        || b[dot + 1..].len() != 2
    {
        return false;
    }
    let integral = &b[start..dot];
    if integral.is_empty() {
        return false;
    }
    let valid = if integral == b"0" {
        true
    } else {
        (b'1'..=b'9').contains(&integral[0])
            && integral.iter().all(u8::is_ascii_digit)
    };
    valid && !(start == 1 && integral == b"0")
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output = Vec::new();
    let mut errors = Vec::new();

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out_row = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::Stop && !errors.is_empty() {
                break;
            }
            if row_failed {
                out_row.push(None);
                continue;
            }

            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let Some(index) = index else {
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                out_row.push(None);
                row_failed = true;
                continue;
            };
            let Some(cell) = row.get(index) else {
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(index),
                    kind: CellErrorKind::MissingCell,
                });
                out_row.push(None);
                row_failed = true;
                continue;
            };

            let result = if cell.is_empty() && spec.nullable {
                Ok(Value::Null)
            } else if cell.is_empty() && spec.kind != ColumnType::Text {
                Err(CellErrorKind::Empty)
            } else {
                match spec.kind {
                    ColumnType::Text => Ok(Value::Text(cell.clone())),
                    ColumnType::Integer => {
                        if !integer_syntax(cell) {
                            Err(CellErrorKind::InvalidInteger)
                        } else {
                            cell.parse::<i64>().map(Value::Integer).map_err(|_| {
                                CellErrorKind::IntegerOverflow
                            })
                        }
                    }
                    ColumnType::Decimal => {
                        if !decimal_syntax(cell) {
                            Err(CellErrorKind::InvalidDecimal)
                        } else {
                            let negative = cell.starts_with('-');
                            let body = if negative { &cell[1..] } else { cell.as_str() };
                            let digits: String =
                                body.chars().filter(|&c| c != '.').collect();
                            match digits.parse::<i64>() {
                                Ok(value) => {
                                    if negative {
                                        Ok(Value::Decimal(-value))
                                    } else {
                                        Ok(Value::Decimal(value))
                                    }
                                }
                                Err(_) => {
                                    let fits_min = negative
                                        && digits == "9223372036854775808";
                                    if fits_min {
                                        Ok(Value::Decimal(i64::MIN))
                                    } else {
                                        Err(CellErrorKind::DecimalOverflow)
                                    }
                                }
                            }
                        }
                    }
                    ColumnType::Boolean => match cell.as_str() {
                        "yes" | "1" => Ok(Value::Boolean(true)),
                        "no" | "0" => Ok(Value::Boolean(false)),
                        _ => Err(CellErrorKind::InvalidBoolean),
                    },
                }
            };

            match result {
                Ok(value) => out_row.push(Some(value)),
                Err(kind) => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind,
                    });
                    out_row.push(None);
                    row_failed = true;
                }
            }
        }

        if policy == ErrorPolicy::Stop && !errors.is_empty() {
            output.push(out_row);
            break;
        }
        output.push(out_row);
        if policy == ErrorPolicy::FirstPerRow && row_failed {
            // Remaining spec positions are represented by None.
            let row = output.last_mut().unwrap();
            row.resize(columns.len(), None);
        }
    }

    TypedResult { rows: output, errors }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut all_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        all_records.push(header);
    }
    all_records.extend(table.rows.iter());

    for (record_index, row) in all_records.iter().enumerate() {
        for (column_index, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (row_index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, row) in all_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    let mut records: Vec<Vec<String>> = all_records.into_iter().cloned().collect();

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            let mut retained: HashMap<String, usize> = HashMap::new();
            for (index, name) in header.iter().enumerate() {
                match table.duplicates {
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        retained.entry(name.clone()).or_insert(index);
                    }
                    DuplicatePolicy::KeepLast => {
                        retained.insert(name.clone(), index);
                    }
                }
            }

            if table.duplicates == DuplicatePolicy::Reject {
                let mut seen = HashMap::<&str, usize>::new();
                for (second, name) in header.iter().enumerate() {
                    if let Some(&first) = seen.get(name.as_str()) {
                        return Err(WriteError::DuplicateHeader {
                            name: name.clone(),
                            first,
                            second,
                        });
                    }
                    seen.insert(name, second);
                }
            }

            let mut columns: Vec<(String, usize)> =
                retained.into_iter().map(|(n, i)| (n, i)).collect();
            columns.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
            let indices: Vec<usize> = columns.iter().map(|(_, i)| *i).collect();
            let sorted_header: Vec<String> =
                columns.into_iter().map(|(name, _)| name).collect();

            let mut projected: Vec<Vec<String>> = table
                .rows
                .iter()
                .map(|row| indices.iter().map(|&i| row[i].clone()).collect())
                .collect();
            projected.sort_by(|a, b| compare_rows(a, b));

            records.clear();
            records.push(sorted_header);
            records.extend(projected);
        } else {
            records = table.rows.clone();
            records.sort_by(|a, b| compare_rows(a, b));
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            match config.mode {
                WriteMode::Preserve => match config.line_ending {
                    LineEnding::Lf => output.push('\n'),
                    LineEnding::CrLf => output.push_str("\r\n"),
                },
                WriteMode::Canonical => output.push('\n'),
            }
        }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            let bytes = field.as_bytes();
            let quote_field = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
                || (record_index == 0
                    && column_index == 0
                    && field.starts_with('\u{feff}'))
                || (row.len() == 1 && field.is_empty());

            if quote_field {
                output.push(config.quote as char);
                for ch in field.chars() {
                    if ch as u32 == config.quote as u32 {
                        output.push(config.quote as char);
                        output.push(config.quote as char);
                    } else {
                        output.push(ch);
                    }
                }
                output.push(config.quote as char);
            } else {
                output.push_str(field);
            }
        }
    }

    let final_terminator = config.mode == WriteMode::Canonical || config.final_terminator;
    if final_terminator {
        match config.mode {
            WriteMode::Canonical => output.push('\n'),
            WriteMode::Preserve => match config.line_ending {
                LineEnding::Lf => output.push('\n'),
                LineEnding::CrLf => output.push_str("\r\n"),
            },
        }
    }
    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}
