#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;

    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        let outside_quotes = !matches!(state, State::Quoted { .. });

        if outside_quotes && byte == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        match state {
            State::FieldStart => match byte {
                b'"' => {
                    state = State::Quoted { opening: pos };
                    pos += 1;
                }
                b',' => {
                    record.push(String::new());
                    pos += 1;
                }
                b'\n' => {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::FieldStart;
                    pos += 1;
                }
                b'\r' => {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::FieldStart;
                    pos += 2;
                }
                _ => {
                    field.push(byte as char);
                    state = State::Unquoted;
                    pos += 1;
                }
            },
            State::Unquoted => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    pos += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::FieldStart;
                    pos += 1;
                }
                b'\r' => {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::FieldStart;
                    pos += 2;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                }
                _ => {
                    field.push(byte as char);
                    pos += 1;
                }
            },
            State::Quoted { opening } => {
                if byte == b'"' {
                    if bytes.get(pos + 1) == Some(&b'"') {
                        field.push('"');
                        pos += 2;
                    } else {
                        state = State::AfterQuote;
                        pos += 1;
                    }
                } else {
                    let ch = input[pos..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    pos += ch.len_utf8();
                }

                // Keep the opening offset available for EOF validation.
                if let State::AfterQuote = state {
                    let _ = opening;
                }
            }
            State::AfterQuote => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    pos += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::FieldStart;
                    pos += 1;
                }
                b'\r' => {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::FieldStart;
                    pos += 2;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            },
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::FieldStart => {
            // A nonempty input ending in a delimiter has a final empty field.
            if pos > 0 {
                record.push(String::new());
                records.push(record);
            }
        }
        State::Unquoted | State::AfterQuote => {
            record.push(field);
            records.push(record);
        }
    }

    Ok(records)
}
