#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut field_started_quoted = false;
    let mut quote_open = None;
    let mut after_quote = false;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if quote_open.is_none() && byte == b'\r' {
            if bytes.get(pos + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: pos,
                });
            }

            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            pos += 2;
            record_index += 1;
            field_started_quoted = false;
            after_quote = false;
            continue;
        }

        if let Some(opening) = quote_open {
            if byte == b'"' {
                if bytes.get(pos + 1) == Some(&b'"') {
                    field.push('"');
                    pos += 2;
                } else {
                    quote_open = None;
                    after_quote = true;
                    pos += 1;
                }
            } else {
                let ch = input[pos..].chars().next().unwrap();
                field.push(ch);
                pos += ch.len_utf8();
            }
            let _ = opening;
            continue;
        }

        if after_quote {
            match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    field_started_quoted = false;
                    after_quote = false;
                    pos += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    field_started_quoted = false;
                    after_quote = false;
                    pos += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
            continue;
        }

        match byte {
            b'"' if field.is_empty() && !field_started_quoted => {
                field_started_quoted = true;
                quote_open = Some(pos);
                pos += 1;
            }
            b'"' => {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
            b',' => {
                record.push(std::mem::take(&mut field));
                field_started_quoted = false;
                pos += 1;
            }
            b'\n' => {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                field_started_quoted = false;
                pos += 1;
            }
            _ => {
                let ch = input[pos..].chars().next().unwrap();
                field.push(ch);
                pos += ch.len_utf8();
            }
        }
    }

    if let Some(opening) = quote_open {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if !record.is_empty() || !field.is_empty() || field_started_quoted || after_quote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
