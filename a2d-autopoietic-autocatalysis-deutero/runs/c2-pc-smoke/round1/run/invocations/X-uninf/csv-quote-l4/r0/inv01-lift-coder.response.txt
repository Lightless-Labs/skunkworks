pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|field| field == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|field| field == name)
            }
        }
    }
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if delimiter == quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            row.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            records.push(row);
            row = Vec::new();
            record_index += 1;
            record_started = false;
            after_quote = false;
            i += 2;
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                after_quote = false;
                record_started = true;
                i += 1;
                continue;
            }
            if b == b'\n' {
                row.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                records.push(row);
                row = Vec::new();
                record_index += 1;
                record_started = false;
                after_quote = false;
                i += 1;
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            row.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            record_started = true;
            i += 1;
        } else if b == b'\n' {
            row.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            records.push(row);
            row = Vec::new();
            record_index += 1;
            record_started = false;
            i += 1;
        } else if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_open = i;
                record_started = true;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
        } else {
            field.push(b);
            record_started = true;
            i += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if record_started || after_quote || !row.is_empty() || !field.is_empty() {
        row.push(String::from_utf8(field).unwrap());
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut records = records;
            let header = records.remove(0);
            (Some(header), records)
        }
    } else {
        (None, records)
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(header) {
                return Err(error);
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let negative = b[0] == b'-';
    let body = if negative { &b[1..] } else { b };
    let Some(dot) = body.iter().position(|&c| c == b'.') else {
        return false;
    };
    if body[dot + 1..].len() != 2 || !body[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let integral = &body[..dot];
    if integral.is_empty() {
        return false;
    }
    if integral == b"0" {
        return !negative || body[dot + 1..] != *b"00";
    }
    integral[0].is_ascii_digit()
        && integral[0] != b'0'
        && integral.iter().all(u8::is_ascii_digit)
}

fn parse_decimal_value(s: &str) -> Option<i64> {
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let dot = body.find('.')?;
    let whole: i128 = body[..dot].parse().ok()?;
    let fraction: i128 = body[dot + 1..].parse().ok()?;
    let magnitude = whole.checked_mul(100)?.checked_add(fraction)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).ok()
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());

        for (spec_index, spec) in columns.iter().enumerate() {
            let column_index = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let Some(column_index) = column_index else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                out.push(None);
                if policy != ErrorPolicy::Collect {
                    result.rows.push(out);
                    if policy == ErrorPolicy::Stop {
                        break 'rows;
                    }
                    break;
                }
                continue;
            };

            let Some(cell) = row.get(column_index) else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::MissingCell,
                });
                out.push(None);
                if policy != ErrorPolicy::Collect {
                    result.rows.push(out);
                    if policy == ErrorPolicy::Stop {
                        break 'rows;
                    }
                    break;
                }
                continue;
            };

            if cell.is_empty() {
                if spec.nullable {
                    out.push(Some(Value::Null));
                    continue;
                }
                if spec.kind != ColumnType::Text {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind: CellErrorKind::Empty,
                    });
                    out.push(None);
                    if policy != ErrorPolicy::Collect {
                        result.rows.push(out);
                        if policy == ErrorPolicy::Stop {
                            break 'rows;
                        }
                        break;
                    }
                    continue;
                }
            }

            let parsed = match spec.kind {
                ColumnType::Text => Ok(Value::Text(cell.clone())),
                ColumnType::Integer => {
                    if !integer_syntax(cell) {
                        Err(CellErrorKind::InvalidInteger)
                    } else {
                        match cell.parse::<i64>() {
                            Ok(value) => Ok(Value::Integer(value)),
                            Err(_) => Err(CellErrorKind::IntegerOverflow),
                        }
                    }
                }
                ColumnType::Decimal => {
                    if !decimal_syntax(cell) {
                        Err(CellErrorKind::InvalidDecimal)
                    } else {
                        parse_decimal_value(cell)
                            .map(Value::Decimal)
                            .ok_or(CellErrorKind::DecimalOverflow)
                    }
                }
                ColumnType::Boolean => match cell.as_str() {
                    "yes" | "1" => Ok(Value::Boolean(true)),
                    "no" | "0" => Ok(Value::Boolean(false)),
                    _ => Err(CellErrorKind::InvalidBoolean),
                },
            };

            match parsed {
                Ok(value) => out.push(Some(value)),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    out.push(None);
                    if policy != ErrorPolicy::Collect {
                        result.rows.push(out);
                        if policy == ErrorPolicy::Stop {
                            break 'rows;
                        }
                        break;
                    }
                }
            }
        }

        if out.len() < columns.len() {
            out.resize(columns.len(), None);
        }
        if policy == ErrorPolicy::Collect || out.len() == columns.len() {
            result.rows.push(out);
        }
    }

    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {

        first: usize,
        second: usize,
    },
}

fn validate_write_config(config: WriteConfig) -> Result<(), WriteError> {
    validate_config(config.delimiter, config.quote).map_err(|error| match error {
        CsvError::InvalidConfig { component } => WriteError::InvalidConfig { component },
        _ => unreachable!(),
    })
}

fn bytewise_cmp(a: &String, b: &String) -> std::cmp::Ordering {
    a.as_bytes().cmp(b.as_bytes())
}

fn quote_field(field: &str, delimiter: u8, quote: u8, first: bool, sole_empty: bool) -> String {
    let bytes = field.as_bytes();
    let boundary_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let starts_bom = first && field.starts_with('\u{feff}');
    let needs_quotes = field.as_bytes().contains(&delimiter)
        || field.as_bytes().contains(&quote)
        || field.as_bytes().contains(&b'\r')
        || field.as_bytes().contains(&b'\n')
        || boundary_space
        || starts_bom
        || sole_empty;

    if !needs_quotes {
        return field.to_string();
    }

    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_write_config(config)?;

    let mut original_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    original_records.extend(table.rows.iter());

    for (record_index, record) in original_records.iter().enumerate() {
        for (column_index, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (row_index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in original_records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    let mut records: Vec<Vec<String>> = original_records
        .iter()
        .map(|record| (*record).clone())
        .collect();

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(error) = duplicate_header(header) {
                    if let CsvError::DuplicateHeader { name, first, second } = error {
                        return Err(WriteError::DuplicateHeader { name, first, second });
                    }
                }
            }

            let mut keep = Vec::new();
            for (index, name) in header.iter().enumerate() {
                let chosen = match table.duplicates {
                    DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name) == Some(index),
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        header.iter().position(|n| n == name) == Some(index)
                    }
                };
                if chosen {
                    keep.push(index);
                }
            }
            keep.sort_by(|&a, &b| bytewise_cmp(&header[a], &header[b]));

            let projected_header: Vec<String> = keep.iter().map(|&i| header[i].clone()).collect();
            let projected_rows: Vec<Vec<String>> = table
                .rows
                .iter()
                .map(|row| keep.iter().map(|&i| row[i].clone()).collect())
                .collect();
            records.clear();
            records.push(projected_header);
            records.extend(projected_rows);
            let header_count = 1;
            records[header_count..].sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
        } else {
            records.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let terminator = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        let sole_empty = record.len() == 1 && record[0].is_empty();
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            let first_field_first_record = record_index == 0 && column_index == 0;
            output.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                first_field_first_record,
                sole_empty,
            ));
        }
    }

    if canonical || config.final_terminator {
        output.push_str(terminator);
    }

    Ok(output)
}
