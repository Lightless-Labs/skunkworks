#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut start = 0;
    if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        start = 3;
    }
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut quoted = false;
    let mut closed = false;
    let mut quote_start = 0usize;
    let mut field_started = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if quoted {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                quoted = false;
                closed = true;
                i += 1;
                continue;
            }
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
            continue;
        }

        if b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr { record: record_index, byte: i });
            }
            if closed || !quoted {
                row.push(field);
                records.push(row);
                row = Vec::new();
                field = String::new();
                field_started = false;
                closed = false;
                record_index += 1;
                i += 2;
                continue;
            }
        }

        if closed {
            if b == delimiter {
                row.push(field);
                field = String::new();
                field_started = false;
                closed = false;
                i += 1;
                continue;
            }
            if b == b'\n' {
                row.push(field);
                records.push(row);
                row = Vec::new();
                field = String::new();
                field_started = false;
                closed = false;
                record_index += 1;
                i += 1;
                continue;
            }
            return Err(CsvError::TrailingByte { record: record_index, byte: i });
        }

        if b == delimiter {
            row.push(field);
            field = String::new();
            field_started = false;
            i += 1;
        } else if b == b'\n' {
            row.push(field);
            records.push(row);
            row = Vec::new();
            field = String::new();
            field_started = false;
            record_index += 1;
            i += 1;
        } else if b == quote {
            if field_started || !field.is_empty() {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
            quoted = true;
            quote_start = i;
            i += 1;
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            field_started = true;
            i += ch.len_utf8();
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_start });
    }
    if !row.is_empty() || !field.is_empty() || field_started || closed || !records.is_empty() {
        row.push(field);
        records.push(row);
    } else if bytes[start..].last() == Some(&delimiter) {
        row.push(field);
        records.push(row);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => header.iter().position(|s| s == name),
        }
    }
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|v| v == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;
    let mut records = parse_with(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };
    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(err) = duplicate_header(h) {
                return Err(err);
            }
        }
        for (idx, row) in records.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }
    Ok(Table { header, rows: records, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), result: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = build_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    FirstPerRow,
    Stop,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let mut i = 0;
    if b[0] == b'-' {
        i = 1;
        if i == b.len() { return false; }
    }
    if b[i] == b'0' {
        return i + 1 == b.len() && i == 0;
    }
    if !(b'1'..=b'9').contains(&b[i]) { return false; }
    b[i + 1..].iter().all(|c| c.is_ascii_digit())
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let mut i = 0;
    if b.first() == Some(&b'-') {
        i = 1;
    }
    if i >= b.len() { return false; }
    let int_start = i;
    if b[i] == b'0' {
        i += 1;
        if i < b.len() && b[i].is_ascii_digit() { return false; }
    } else if (b'1'..=b'9').contains(&b[i]) {
        i += 1;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
    } else {
        return false;
    }
    if i >= b.len() || b[i] != b'.' || b.len() != i + 3 { return false; }
    if !b[i + 1..].iter().all(|c| c.is_ascii_digit()) { return false; }
    !(b[0] == b'-' && b[int_start] == b'0' && b[i + 1] == b'0' && b[i + 2] == b'0')
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let neg = s.starts_with('-');
    let body = if neg { &s[1..] } else { s };
    let (whole, frac) = body.split_once('.').unwrap();
    let combined = format!("{}{}", whole, frac);
    let magnitude = combined.parse::<i128>().map_err(|_| CellErrorKind::DecimalOverflow)?;
    let signed = if neg { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'rows: for (row_idx, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut row_failed = false;
        for (spec_idx, spec) in columns.iter().enumerate() {
            if row_failed {
                out.push(None);
                continue;
            }
            let col = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let (value, error) = match col {
                None => (None, Some(CellError { row: row_idx, spec: spec_idx, column: None, kind: CellErrorKind::UnknownColumn })),
                Some(i) if i >= row.len() => (None, Some(CellError { row: row_idx, spec: spec_idx, column: Some(i), kind: CellErrorKind::MissingCell })),
                Some(i) => {
                    let cell = &row[i];
                    if spec.nullable && cell.is_empty() {
                        (Some(Value::Null), None)
                    } else if cell.is_empty() && spec.kind != ColumnType::Text {
                        (None, Some(CellError { row: row_idx, spec: spec_idx, column: Some(i), kind: CellErrorKind::Empty }))
                    } else {
                        let parsed = match spec.kind {
                            ColumnType::Text => Ok(Value::Text(cell.clone())),
                            ColumnType::Integer => parse_integer(cell).map(Value::Integer),
                            ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
                            ColumnType::Boolean => match cell.as_str() {
                                "yes" | "1" => Ok(Value::Boolean(true)),
                                "no" | "0" => Ok(Value::Boolean(false)),
                                _ => Err(CellErrorKind::InvalidBoolean),
                            },
                        };
                        match parsed {
                            Ok(v) => (Some(v), None),
                            Err(kind) => (None, Some(CellError { row: row_idx, spec: spec_idx, column: Some(i), kind })),
                        }
                    }
                }
            };
            if let Some(error) = error {
                result.errors.push(error);
                row_failed = true;
                match policy {
                    ErrorPolicy::Collect => row_failed = false,
                    ErrorPolicy::FirstPerRow => {}
                    ErrorPolicy::Stop => {
                        out.push(None);
                        while out.len() < columns.len() { out.push(None); }
                        result.rows.push(out);
                        break 'rows;
                    }
                }
            }
            out.push(value);
        }
        result.rows.push(out);
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        records.push(header);
    }
    for row in &table.rows {
        records.push(row);
    }

    for (record, row) in records.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            if let Some(byte) = cell.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    let mut output_rows: Vec<Vec<String>> = match config.mode {
        WriteMode::Preserve => records.iter().map(|r| r.to_vec()).collect(),
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                if table.duplicates == DuplicatePolicy::Reject {
                    if let Some(CsvError::DuplicateHeader { name, first, second }) = duplicate_header(header) {
                        return Err(WriteError::DuplicateHeader { name, first, second });
                    }
                }
                let mut chosen = Vec::new();
                for name in header {
                    if chosen.iter().any(|(_, existing): &(usize, &String)| *existing == name) {
                        continue;
                    }
                    let index = match table.duplicates {
                        DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name).unwrap(),
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => header.iter().position(|v| v == name).unwrap(),
                    };
                    chosen.push((index, name));
                }
                chosen.sort_by(|a, b| a.1.as_bytes().cmp(b.1.as_bytes()));
                let indices: Vec<usize> = chosen.iter().map(|(i, _)| *i).collect();
                let mut rows = Vec::new();
                rows.push(indices.iter().map(|i| header[*i].clone()).collect());
                let mut projected: Vec<Vec<String>> = table.rows.iter().map(|row| indices.iter().map(|i| row[*i].clone()).collect()).collect();
                projected.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
                rows.extend(projected);
                rows
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
                rows
            }
        }
    };

    if output_rows.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let term = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };
    let final_term = canonical || config.final_terminator;
    let mut out = String::new();
    for (ri, row) in output_rows.iter_mut().enumerate() {
        if ri > 0 {
            out.push_str(term);
        }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 {
                out.push(config.delimiter as char);
            }
            let bytes = field.as_bytes();
            let quote_needed = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || field.starts_with(' ')
                || field.starts_with('\t')
                || field.ends_with(' ')
                || field.ends_with('\t')
                || (ri == 0 && ci == 0 && field.starts_with('\u{feff}'))
                || (row.len() == 1 && field.is_empty());
            if quote_needed {
                out.push(config.quote as char);
                for ch in field.chars() {
                    if ch as u32 == config.quote as u32 {
                        out.push(config.quote as char);
                    }
                    out.push(ch);
                }
                out.push(config.quote as char);
            } else {
                out.push_str(field);
            }
        }
    }
    if final_term {
        out.push_str(term);
    }
    Ok(out)
}
