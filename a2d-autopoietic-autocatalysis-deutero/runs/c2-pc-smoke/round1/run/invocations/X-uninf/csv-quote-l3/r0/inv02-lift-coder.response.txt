use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;
    let mut i = start;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Quoted { opening } => {
                if byte == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(byte as char);
                    i += 1;
                }
                let _ = opening;
            }
            State::FieldStart | State::Unquoted | State::AfterQuote => {
                if byte == b'\r' {
                    if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                        return Err(CsvError::BareCr {
                            record: record_index,
                            byte: i,
                        });
                    }
                    record.push(field);
                    field = String::new();
                    records.push(record);
                    record = Vec::new();
                    state = State::FieldStart;
                    i += 2;
                    record_index += 1;
                    continue;
                }

                if byte == b'\n' {
                    record.push(field);
                    field = String::new();
                    records.push(record);
                    record = Vec::new();
                    state = State::FieldStart;
                    i += 1;
                    record_index += 1;
                    continue;
                }

                match state {
                    State::FieldStart => {
                        if byte == delimiter {
                            record.push(String::new());
                            i += 1;
                        } else if byte == quote {
                            state = State::Quoted { opening: i };
                            i += 1;
                        } else {
                            field.push(byte as char);
                            state = State::Unquoted;
                            i += 1;
                        }
                    }
                    State::Unquoted => {
                        if byte == delimiter {
                            record.push(field);
                            field = String::new();
                            state = State::FieldStart;
                            i += 1;
                        } else if byte == quote {
                            return Err(CsvError::UnexpectedQuote {
                                record: record_index,
                                byte: i,
                            });
                        } else {
                            field.push(byte as char);
                            i += 1;
                        }
                    }
                    State::AfterQuote => {
                        if byte == delimiter {
                            record.push(field);
                            field = String::new();
                            state = State::FieldStart;
                            i += 1;
                        } else {
                            return Err(CsvError::TrailingByte {
                                record: record_index,
                                byte: i,
                            });
                        }
                    }
                    State::Quoted { .. } => unreachable!(),
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::FieldStart | State::Unquoted | State::AfterQuote => {
            record.push(field);
            records.push(record);
        }
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let parsed = parse_with_dialect(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers && !parsed.is_empty() {
        let mut iter = parsed.into_iter();
        (iter.next(), iter.collect::<Vec<_>>())
    } else {
        (None, parsed)
    };

    if let Some(header_values) = header.as_ref() {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in header_values.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name.as_str(), second);
            }
        }

        let expected = header_values.len();
        for (offset, row) in rows.iter().enumerate() {
            if row.len() != expected {
                return Err(CsvError::HeaderWidth {
                    record: offset + 1,
                    expected,
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = self.result.as_ref() {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }

    let digits = if bytes[0] == b'-' {
        &bytes[1..]
    } else {
        bytes
    };
    if digits.is_empty() || !digits.iter().all(u8::is_ascii_digit) {
        return false;
    }
    digits == b"0" || (digits[0] != b'0')
}

fn parse_integer(value: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(value) {
        return Err(CellErrorKind::InvalidInteger);
    }
    value
        .parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn decimal_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }

    let body = if bytes[0] == b'-' {
        &bytes[1..]
    } else {
        bytes
    };

    if body.len() < 4 || body[body.len() - 3] != b'.' {
        return false;
    }
    let integral = &body[..body.len() - 3];
    let fractional = &body[body.len() - 2..];

    if fractional.len() != 2 || !fractional.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral.is_empty() || !integral.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if !(integral == b"0" || integral[0] != b'0') {
        return false;
    }
    if bytes[0] == b'-' && integral == b"0" && fractional == b"00" {
        return false;
    }
    true
}

fn parse_decimal(value: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(value) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = value.starts_with('-');
    let body = if negative { &value[1..] } else { value };
    let point = body.len() - 3;
    let integral = &body[..point];
    let fractional = &body[point + 1..];

    let whole = integral
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let fraction = ((fractional.as_bytes()[0] - b'0') as i128) * 10
        + (fractional.as_bytes()[1] - b'0') as i128;
    let magnitude = whole
        .checked_mul(100)
        .and_then(|n| n.checked_add(fraction))
        .ok_or(CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    text: Option<&str>,
    column: Option<usize>,
    spec: &ColumnSpec,
) -> Result<Value, (Option<usize>, CellErrorKind)> {
    let text = match text {
        None => {
            return Err((
                column,
                if column.is_none() {
                    CellErrorKind::UnknownColumn
                } else {
                    CellErrorKind::MissingCell
                },
            ));
        }
        Some(text) => text,
    };

    if spec.nullable && text.is_empty() {
        return Ok(Value::Null);
    }

    if text.is_empty() && spec.kind != ColumnType::Text {
        return Err((column, CellErrorKind::Empty));
    }

    let value = match spec.kind {
        ColumnType::Text => Value::Text(text.to_owned()),
        ColumnType::Integer => Value::Integer(parse_integer(text).map_err(|kind| (column, kind))?),
        ColumnType::Decimal => Value::Decimal(parse_decimal(text).map_err(|kind| (column, kind))?),
        ColumnType::Boolean => match text {
            "yes" | "1" => Value::Boolean(true),
            "no" | "0" => Value::Boolean(false),
            _ => return Err((column, CellErrorKind::InvalidBoolean)),
        },
    };
    Ok(value)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                output.push(None);
                continue;
            }

            let column_index = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let cell = column_index.and_then(|index| row.get(index).map(String::as_str));

            match evaluate_cell(cell, column_index, spec) {
                Ok(value) => output.push(Some(value)),
                Err((error_column, kind)) => {
                    output.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: error_column,
                        kind,
                    });
                    if matches!(policy, ErrorPolicy::FirstPerRow | ErrorPolicy::Stop) {
                        failed = true;
                    }
                }
            }

            if policy == ErrorPolicy::Stop && failed {
                break;
            }
        }

        if failed && policy == ErrorPolicy::Stop {
            result.rows.push(output);
            break;
        }

        result.rows.push(output);
    }

    result
}
