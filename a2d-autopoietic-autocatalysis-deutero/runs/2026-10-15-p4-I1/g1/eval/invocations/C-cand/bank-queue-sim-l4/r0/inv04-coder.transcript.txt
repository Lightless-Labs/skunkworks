fn expire_waiters(
    queue: &mut Vec<Waiter>,
    time: u64,
    customers: &[Customer],
    events: &mut Vec<Event>,
    wait_ticks: &mut u128,
    reneged: &mut usize,
) {
    let mut expired = Vec::new();
    let mut retained = Vec::with_capacity(queue.len());

    for waiter in queue.drain(..) {
        if waiter.deadline.is_some_and(|deadline| deadline <= time) {
            expired.push(waiter);
        } else {
            retained.push(waiter);
        }
    }

    expired.sort_by(|a, b| {
        a.deadline
            .cmp(&b.deadline)
            .then(customers[a.index].id.cmp(&customers[b.index].id))
    });

            time,

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let other_half = denominator - remainder;

    let round_up = remainder > other_half
        || (remainder == other_half && quotient % 2 == 1);

    Some(quotient + u128::from(round_up))
}

fn best_waiter(queue: &[Waiter], customers: &[Customer]) -> Option<usize> {
    (0..queue.len()).min_by(|&a, &b| {
        let wa = &queue[a];
        let wb = &queue[b];
        let ca = &customers[wa.index];
        let cb = &customers[wb.index];

        ca.priority
            .cmp(&cb.priority)
            .then_with(|| wb.paused.cmp(&wa.paused))
            .then_with(|| match (wa.paused, wb.paused) {
                (true, true) => wa.pause_time.cmp(&wb.pause_time),
                (false, false) => ca.arrival.cmp(&cb.arrival),
                _ => std::cmp::Ordering::Equal,
            })
            .then_with(|| {
                if wa.paused && wb.paused {
                    ca.id.cmp(&cb.id)
                } else {
                    cb.id.cmp(&ca.id)
                }
            })
    })
}

fn expire_waiters(
    queue: &mut Vec<Waiter>,
    time: u64,
    customers: &[Customer],
    events: &mut Vec<Event>,
    wait_ticks: &mut u128,
    reneged: &mut usize,
) {
    let mut expired = Vec::new();
    let mut retained = Vec::with_capacity(queue.len());

    for waiter in queue.drain(..) {
        if waiter.deadline.is_some_and(|deadline| deadline <= time) {
            expired.push(waiter);
        } else {
            retained.push(waiter);
        }

    expired.sort_by(|a, b| {
        a.deadline
            .cmp(&b.deadline)
