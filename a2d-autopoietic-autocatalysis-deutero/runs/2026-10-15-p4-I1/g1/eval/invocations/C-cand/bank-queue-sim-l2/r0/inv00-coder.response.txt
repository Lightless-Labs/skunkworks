#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy: bool,
    idle_since: u64,
    finish_at: Option<u64>,
    customer_at_finish: Option<u32>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy: false,
            finish_at: None,
            customer_at_finish: None,
        })
        .collect();

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut arrival_index = 0;
    let mut time = 0u64;
    let mut finished_customers = 0usize;

    loop {
        // Complete all previously scheduled work first, in ascending server ID order.
        let mut due_finishes: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (state.busy && state.finish_at == Some(time)).then_some(i)
            })
            .collect();
        due_finishes.sort_by_key(|&i| states[i].server.id);

        for i in due_finishes {
            let state = &mut states[i];
            let customer = state.customer_at_finish.take().unwrap();
            events.push(Event::Finish {
                time,
                customer,
                server: state.server.id,
            });
            state.busy = false;
            state.finish_at = None;
            state.idle_since = time;
            finished_customers += 1;
        }

        // Open servers in descending server ID order.
        let mut due_opens: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.server.opens == time).then_some(i)
            })
            .collect();
        due_opens.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));

        for i in due_opens {
            let state = &mut states[i];
            state.opened = true;
            events.push(Event::Open {
                time,
                server: state.server.id,
            });
        }

        // Emit the complete arrival cohort in the already sorted order.
        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            events.push(Event::Arrive {
                time,
                customer: arrivals[arrival_index].id,
            });
            arrival_index += 1;
        }

        // Dispatch waiting customers in FIFO order. The input order of servers is irrelevant.
        while let Some(customer) = arrivals.get(arrival_index) {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && !state.busy)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer = customer.clone();
            let server = &states[server_index].server;
            let finish = customer
                .service
                .checked_add(time)
                .and_then(|partial| partial.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });
            arrival_index += 1;

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                finished_customers += 1;
                states[server_index].idle_since = time;
            } else {
                states[server_index].busy = true;
                states[server_index].finish_at = Some(finish);
                states[server_index].customer_at_finish = Some(customer.id);
            }
        }

        if finished_customers == customers.len()
            && states.iter().all(|state| state.opened)
        {
            break;
        }

        let next_arrival = arrivals.get(arrival_index).map(|customer| customer.arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_finish = states
            .iter()
            .filter(|state| state.busy)
            .filter_map(|state| state.finish_at)
            .min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&candidate| candidate > time)
            .min();

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(events)
}

fn main() {}
