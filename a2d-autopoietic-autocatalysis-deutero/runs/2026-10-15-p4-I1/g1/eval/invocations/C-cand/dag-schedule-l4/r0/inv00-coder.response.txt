#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
    ResourceShape { task: usize, expected: usize, actual: usize },
    ResourceExceeded { task: usize, resource: usize, need: u32, capacity: u32 },
}

fn resource_snapshot(used: &[u128]) -> String {
    if used.is_empty() {
        return "-".to_owned();
    }
    let mut out = String::new();
    for (i, value) in used.iter().enumerate() {
        if i != 0 {
            out.push(',');
        }
        use std::fmt::Write;
        let _ = write!(out, "{value}");
    }
    out
}

pub fn schedule(tasks: &[Task], cooldowns: &[u64], capacity: &[u32]) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation is performed completely before other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers { count: worker_count });
    }

    // Resource validation is task-major, then resource-major.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &cap)) in task.needs.iter().zip(capacity).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded { task: task_id, resource, need, capacity: cap });
            }
        }
    }

    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Deterministic Kahn order: always remove the smallest available ID.
    let mut ready = std::collections::BTreeSet::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            ready.insert(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(id) = ready.pop_first() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.insert(child);
            }
        }
    }
    if order.len() != n {
        let removed = {
            let mut is_removed = vec![false; n];
            for &id in &order {
                is_removed[id] = true;
            }
            is_removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id].iter().map(|&child| tails[child]).max().unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut trace = String::new();
    use std::fmt::Write;
    let _ = writeln!(trace, "workers={worker_count};resources={}", capacity.len());

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut used = vec![0u128; capacity.len()];

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        finish: u64,
        worker: usize,
    }

    let mut running: Vec<Running> = Vec::new();
    let mut worker_available = vec![0u64; worker_count];
    let mut now = 0u64;
    let mut finished_count = 0usize;
    let mut makespan = 0u64;

    while finished_count < n {
        // Complete all tasks due now in ascending task ID order.
        let mut due: Vec<Running> = running.iter().copied().filter(|r| r.finish == now).collect();
        due.sort_by_key(|r| r.task);
        if !due.is_empty() {
            running.retain(|r| r.finish != now);
            for r in due {
                completed[r.task] = true;
                finished_count += 1;
                makespan = makespan.max(r.finish);
                for (resource, &need) in tasks[r.task].needs.iter().enumerate() {
                    used[resource] -= u128::from(need);
                }
                let _ = writeln!(
                    trace,
                    "{now}|end|{}|{}|{}",
                    r.task,
                    r.worker,
                    resource_snapshot(&used)
                );
                for &child in &children[r.task] {
                    remaining_deps[child] -= 1;
                }
            }
        }

        // Admit one task at a time, reevaluating eligibility and ranking each time.
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_deps[id] == 0
                        && tasks[id].needs.iter().enumerate().all(|(r, &need)| {
                            u128::from(need) <= u128::from(capacity[r]) - used[r]
                        })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now)
                .expect("candidate exists only when a worker can be selected");

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u128::from(need);
            }
            runs[task_id] = Some(Run { start: now, finish, worker });
            let _ = writeln!(
                trace,
                "{now}|start|{task_id}|{worker}|{}",
                resource_snapshot(&used)
            );

            if finish == now {
                // Zero-duration completion is immediate, before the next admission.
                completed[task_id] = true;
                finished_count += 1;
                makespan = makespan.max(finish);
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u128::from(need);
                }
                let _ = writeln!(
                    trace,
                    "{now}|end|{task_id}|{worker}|{}",
                    resource_snapshot(&used)
                );
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push(Running { task: task_id, finish, worker });
            }
        }

        if finished_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change state.
        let mut next = u64::MAX;
        for r in &running {
            if r.finish > now {
                next = next.min(r.finish);
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = next.min(tasks[id].release);
            }
        }
        for &available in &worker_available {
            if available > now {
                next = next.min(available);
            }
        }
        // If no future event exists, the remaining tasks cannot make progress;
        // with a validated acyclic graph this is unreachable.
        if next == u64::MAX {
            break;
        }
        now = next;
    }

    let runs = runs.into_iter().map(|r| r.expect("all tasks completed")).collect();
    let _ = writeln!(trace, "makespan={makespan}");
    Ok(Schedule { runs, makespan, trace })
}

fn main() {}
