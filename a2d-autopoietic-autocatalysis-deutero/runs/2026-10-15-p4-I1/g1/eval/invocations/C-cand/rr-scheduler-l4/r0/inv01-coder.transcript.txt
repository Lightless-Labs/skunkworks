                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let selected_id = processes[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != selected_id) {
            time += u64::from(switch_cost);
        }

        while event_pos < events.len() && events[event_pos].time <= time {
            let event = events[event_pos];
            queues[processes[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        let quantum = quanta[level];
        let consumed = processes[index].remaining.min(quantum);
        let start = time;
        time += u64::from(consumed);
        let end = time;
        processes[index].remaining -= consumed;

        let burst_finished = processes[index].remaining == 0;
        let job_finished =
            burst_finished && processes[index].burst_pos + 1 == processes[index].bursts.len();

        runs.push(Run {
            id: selected_id,
            start,
            end,
            finished: job_finished,
            level,
        });

        if burst_finished && !job_finished {
            let io_position = processes[index].burst_pos + 1;
            let io_delay = u64::from(processes[index].bursts[io_position]);
            let next_cpu_position = io_position + 1;
            processes[index].burst_pos = next_cpu_position;
            processes[index].remaining = processes[index].bursts[next_cpu_position];
            processes[index].level = level.saturating_sub(1);
            events.push(Event {
                time: end + io_delay,
                kind: 1,
                index,
            });
            events.sort_by_key(|event| (event.time, event.kind, event.index));
        }

        while event_pos < events.len() && events[event_pos].time <= end {
            let event = events[event_pos];
            queues[processes[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        if job_finished {
            completion_times[index] = end;
            completed += 1;
        } else if !burst_finished {
            processes[index].level = (level + 1).min(quanta.len() - 1);
            queues[processes[index].level].push_back(index);
        }

        last_dispatch_id = Some(selected_id);
    }

    let mut stats = Vec::with_capacity(n);
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completion_times[index];
        let turnaround = completion - u64::from(job.arrival);
        let total_work: u64 = job.bursts.iter().map(|&burst| u64::from(burst)).sum();
        let waiting = turnaround - total_work;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = n as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;
    let end = completion_times.iter().copied().max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}

fn main() {}
