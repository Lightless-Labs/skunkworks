```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    level: usize,
    burst_pos: usize,
    remaining: u32,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

fn validate(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<(), Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }
    Ok(())
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    validate(jobs, quanta, switch_cost)?;

    let n = jobs.len();
    if n == 0 {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            level: job.level,
            burst_pos: 0,
            remaining: job.bursts[0],
        })
        .collect();

    let mut events = Vec::with_capacity(n * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut event_pos = 0usize;
    let mut time = 0u64;
    let mut completed = 0usize;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completion_times = vec![0u64; n];

    while completed < n {
        while event_pos < events.len() && events[event_pos].time <= time {
            let event = events[event_pos];
            queues[processes[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_pos >= events.len() {
                break;
            }
            let next_time = events[event_pos].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let selected_id = processes[index].id;
        if last_dispatch_id.is_some_and(|last_id| last_id != selected_id) {
            time += u64::from(switch_cost);
        }
        while event_pos < events.len() && events[event_pos].time <= time {
            let event = events[event_pos];
            queues[processes[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        let quantum = quanta[level];
        let consumed = processes[index].remaining.min(quantum);
        let start = time;
        time += u64::from(consumed);
        let end = time;
        processes[index].remaining -= consumed;

        let burst_finished = processes[index].remaining == 0;
        let job_finished = burst_finished && processes[index].burst_pos + 1 == processes[index].bursts.len();
        runs.push(Run {
            id: selected_id,
            start,
            end,
            finished: job_finished,
            level,
        });

        if burst_finished && !job_finished {
            let io_position = processes[index].burst_pos + 1;
            let io_delay = u64::from(processes[index].bursts[io_position]);
            let next_cpu_position = io_position + 1;
            processes[index].burst_pos = next_cpu_position;
            processes[index].remaining = processes[index].bursts[next_cpu_position];
            let wake_level = level.saturating_sub(1);
            processes[index].level = wake_level;
            events.push(Event {
                time: end + io_delay,
                kind: 1,
                index,
            });
        }

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        while event_pos < events.len() && events[event_pos].time <= end {
            let event = events[event_pos];
            queues[processes[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        if job_finished {
            completion_times[index] = end;
            completed += 1;
        } else if burst_finished {
            // The process is blocked until its wake event.
        } else {
            processes[index].level = (level + 1).min(quanta.len() - 1);
            queues[processes[index].level].push_back(index);
        }

        last_dispatch_id = Some(selected_id);
    }

    let mut stats = Vec::with_capacity(n);
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;
    for (index, job) in jobs.iter().enumerate() {
        let completion = completion_times[index];
        let turnaround = completion - u64::from(job.arrival);
        let total_work: u64 = job.bursts.iter().map(|&burst| u64::from(burst)).sum();
        let waiting = turnaround - total_work;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = n as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;
    let end = completion_times.iter().copied().max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}

fn main() {}
```
