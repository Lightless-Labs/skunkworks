#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_char(word: &mut Word, c: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_str(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_char(word, c, active_glob);
    }
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn parse_braced(input: &str, dollar: usize) -> Result<(usize, String, Option<String>), SplitError> {
    let bytes = input.as_bytes();
    let mut close = dollar + 2;

    while close < bytes.len() {
        if bytes[close] == b'\\' {
            close += 1;
            if close < bytes.len() {
                let c = input[close..].chars().next().unwrap();
                close += c.len_utf8();
            }
        } else if bytes[close] == b'}' {
            break;
        } else {
            let c = input[close..].chars().next().unwrap();
            close += c.len_utf8();
        }
    }

    if close >= bytes.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let contents = &input[dollar + 2..close];
    let content_bytes = contents.as_bytes();
    let mut name_end = 0;

    if content_bytes.first().copied().map_or(false, |b| {
        (b as char).is_ascii_alphabetic() || b == b'_'
    }) {
        name_end = 1;
        while name_end < content_bytes.len() {
            let b = content_bytes[name_end];
            if (b as char).is_ascii_alphanumeric() || b == b'_' {
                name_end += 1;
            } else {
                break;
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = contents[..name_end].to_string();
    if name_end == contents.len() {
        return Ok((close + 1, name, None));
    }

    if contents[name_end..].starts_with(":-") {
        let raw_default = &contents[name_end + 2..];
        let mut decoded = String::new();
        let mut i = 0;
        while i < raw_default.len() {
            let c = raw_default[i..].chars().next().unwrap();
            if c == '\\' {
                i += 1;
                if i < raw_default.len() {
                    let escaped = raw_default[i..].chars().next().unwrap();
                    decoded.push(escaped);
                    i += escaped.len_utf8();
                } else {
                    decoded.push('\\');
                }
            } else {
                decoded.push(c);
                i += c.len_utf8();
            }
        }
        return Ok((close + 1, name, Some(decoded)));
    }

    Err(SplitError::InvalidExpansion { pos: dollar })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    let slash = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if matches!(n, '"' | '\\' | '$') {
                        append_char(&mut word, n, false);
                        i = after_n;
                    } else if n == '\n' {
                        i = after_n;
                    } else {
                        append_char(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (after, name, default) = parse_braced(input, i)?;
                        let value = match (env_value(env, &name), default) {
                            (Some(v), Some(d)) if v.is_empty() => d,
                            (Some(v), _) => v.to_string(),
                            (None, Some(d)) => d,
                            (None, None) => String::new(),
                        };
                        append_str(&mut word, &value, false);
                        active = true;
                        i = after;
                    } else if next_i < input.len()
                        && input[next_i..].chars().next().map_or(false, is_name_start)
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if is_name_continue(ch) {
                                end += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &input[next_i..end];
                        if let Some(value) = env_value(env, name) {
                            append_str(&mut word, value, false);
                        }
                        active = true;
                        i = end;
                    } else {
                        append_char(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_char(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '\n' {
                        i = after_n;
                    } else {
                        append_char(&mut word, n, false);
                        active = true;
                        i = after_n;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (after, name, default) = parse_braced(input, i)?;
                        let value = match (env_value(env, &name), default) {
                            (Some(v), Some(d)) if v.is_empty() => d,
                            (Some(v), _) => v.to_string(),
                            (None, Some(d)) => d,
                            (None, None) => String::new(),
                        };
                        append_str(&mut word, &value, true);
                        active = true;
                        i = after;
                    } else if next_i < input.len()
                        && input[next_i..].chars().next().map_or(false, is_name_start)
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if is_name_continue(ch) {
                                end += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &input[next_i..end];
                        if let Some(value) = env_value(env, name) {
                            append_str(&mut word, value, true);
                        }
                        active = true;
                        i = end;
                    } else {
                        append_char(&mut word, '$', false);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn main() {}
