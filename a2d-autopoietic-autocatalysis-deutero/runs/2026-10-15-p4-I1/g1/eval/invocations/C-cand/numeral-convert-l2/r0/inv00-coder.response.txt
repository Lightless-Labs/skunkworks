#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

const LIMB_BASE: u64 = 1_000_000_000;

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u32>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let payload_start = if let Some(hash) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    // Validate all digits before checking separators or leading zeros.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let digit_count = payload.iter().filter(|&&byte| byte != b'_').count();
    let first_digit = payload.iter().copied().find(|&byte| byte != b'_');
    if digit_count >= 2 && first_digit == Some(b'0') {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    let mut value = vec![0u32];
    for &byte in payload {
        if byte == b'_' {
            continue;
        }
        let digit = digit_value(byte).expect("digits were validated") as u64;
        let mut carry = digit;
        for limb in &mut value {
            let current = (*limb as u64) * (radix as u64) + carry;
            *limb = (current % LIMB_BASE) as u32;
            carry = current / LIMB_BASE;
        }
        while carry != 0 {
            value.push((carry % LIMB_BASE) as u32);
            carry /= LIMB_BASE;
        }
    }

    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    Ok(value)
}

fn add_values(mut left: Vec<u32>, right: &[u32]) -> Vec<u32> {
    let max_len = left.len().max(right.len());
    left.resize(max_len, 0);

    let mut carry = 0u64;
    for index in 0..max_len {
        let sum = left[index] as u64
            + right.get(index).copied().unwrap_or(0) as u64
            + carry;
        left[index] = (sum % LIMB_BASE) as u32;
        carry = sum / LIMB_BASE;
    }
    if carry != 0 {
        left.push(carry as u32);
    }
    left
}

fn value_to_digits(mut value: Vec<u32>, radix: u8, upper: bool) -> Vec<u8> {
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }

    let mut digits = Vec::new();
    if value.len() == 1 && value[0] == 0 {
        digits.push(b'0');
        return digits;
    }

    let alphabet = if upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };

    while value.iter().any(|&limb| limb != 0) {
        let mut remainder = 0u64;
        for limb in value.iter_mut().rev() {
            let current = remainder * LIMB_BASE + *limb as u64;
            *limb = (current / radix as u64) as u32;
            remainder = current % radix as u64;
        }
        digits.push(alphabet[remainder as usize]);
        while value.len() > 1 && value.last() == Some(&0) {
            value.pop();
        }
    }
    digits.reverse();
    digits
}

fn format_value(value: Vec<u32>, radix: u8, style: BaseStyle) -> String {
    let mut digits = value_to_digits(value, radix, style.upper);
    if digits.len() < style.min_width {
        let mut padded = vec![b'0'; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    let separators = if style.group == 0 {
        0
    } else {
        (digits.len() - 1) / style.group
    };
    let mut output = String::with_capacity(digits.len() + separators);
    for (index, &digit) in digits.iter().enumerate() {
        if style.group != 0 && index != 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        output.push(digit as char);
    }
    output
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let value = parse_operand(input, from, 0)?;
    Ok(format_value(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_value = parse_operand(left, base, 0)?;
    let right_value = parse_operand(right, base, 1)?;
    let sum = add_values(left_value, &right_value);
    Ok(format_value(sum, base, style))
}

fn main() {}
