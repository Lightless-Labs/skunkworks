use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
    names: BTreeMap<String, usize>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str, offset: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let Some(hi) = hex_value(bytes[i + 1]) else {
                return Err(offset + i);
            };
            let Some(lo) = hex_value(bytes[i + 2]) else {
                return Err(offset + i);
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(offset + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }

    let bytes = pattern.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { pattern.len() - 1 } else { pattern.len() };
    let mut segments = Vec::new();
    let mut names = BTreeSet::new();
    let mut rank = Vec::new();
    let mut start = 1;

    while start < end {
        let next = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..next];
        let first = bytes[start];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' && next != end {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' {
                segments.push(Segment::Wildcard(name.to_string()));
                rank.push(0);
            } else {
                segments.push(Segment::Param(name.to_string()));
                rank.push(1);
            }
        } else {
            match decode_segment(raw, start) {
                Ok(value) => {
                    segments.push(Segment::Static(value));
                    rank.push(2);
                }
                Err(at) => return Err(Error::InvalidPattern { at }),
            }
        }

        if next == end {
            break;
        }
        start = next + 1;
    }

    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }
    Ok((segments, trailing, rank))
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
    head: String,
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    let cut = path
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(path.len());
    let head = &path[..cut];
    if !head.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if head == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
            head: head.to_string(),
        });
    }

    let bytes = head.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { head.len() - 1 } else { head.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let next = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPath { at: start });
        }
        match decode_segment(&head[start..next], start) {
            Ok(value) => segments.push(value),
            Err(at) => return Err(Error::InvalidPath { at }),
        }
        if next == end {
            break;
        }
        start = next + 1;
    }

    Ok(ParsedPath {
        segments,
        trailing,
        head: head.to_string(),
    })
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
            names: BTreeMap::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing, rank) = parse_pattern(pattern)?;
        if self.names.contains_key(name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method != method {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if same_shape(&route.segments, &segments) {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        self.names.insert(name.to_string(), id);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let parsed = parse_path(path)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let selected_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<(&Route, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(route, _)| route.method == selected_method)
            .map(|(route, params)| (*route, params.clone()))
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && !parsed.segments.is_empty()
            && route.trailing != parsed.trailing
        {
            if route.trailing {
                Some(format!("{}/", parsed.head))
            } else {
                Some(parsed.head.clone())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(&id) = self.names.get(name) else {
            return Err(Error::UnknownName);
        };
        let route = &self.routes[id];

        let mut supplied = BTreeMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
            supplied.insert(*key, *value);
        }

        let captures: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|seg| match seg {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &captures {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        if let Some(extra) = supplied
            .keys()
            .filter(|key| !captures.contains(key))
            .min()
        {
            return Err(Error::ExtraParam {
                name: (*extra).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = supplied[name.as_str()];
                    if !valid_value_bytes(value)
                        || value.is_empty()
                        || value.as_bytes().contains(&b'/')
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !valid_value_bytes(value)
                        || (!value.is_empty()
                            && value
                                .split('/')
                                .any(|component| component.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut output = String::new();
        for segment in &route.segments {
            output.push('/');
            match segment {
                Segment::Static(value) => output.push_str(&encode_segment(value)),
                Segment::Param(name) => {
                    output.push_str(&encode_segment(supplied[name.as_str()]));
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        for (i, component) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(component));
                        }
                    }
                }
            }
        }

        if output.is_empty() {
            output.push('/');
        } else if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(left, right)| match (left, right) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let has_wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    let prefix_len = if has_wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };
    if (!has_wildcard && path.len() != prefix_len) || (has_wildcard && path.len() < prefix_len) {
        return None;
    }

    let mut params = Vec::new();
    for (segment, value) in route.segments[..prefix_len].iter().zip(path.iter()) {
        match segment {
            Segment::Static(expected) if expected != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if let Some(Segment::Wildcard(name)) = route.segments.last() {
        let value = path[prefix_len..].join("/");
        params.push((name.clone(), value));
    }

    Some(params)
}

fn valid_value_bytes(value: &str) -> bool {
    value.bytes().all(|b| (0x20..=0x7e).contains(&b))
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn main() {}
