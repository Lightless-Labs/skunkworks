```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    level: usize,
    burst_position: usize,
    remaining: u32,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

fn validate(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<(), Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }
    Ok(())
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    validate(jobs, quanta, switch_cost)?;

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            level: job.level,
            burst_position: 0,
            remaining: job.bursts[0],
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
        })
        .collect();
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut event_cursor = 0usize;
    let mut ready: Vec<std::collections::VecDeque<usize>> = (0..quanta.len())
        .map(|_| std::collections::VecDeque::new())
        .collect();
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;

    let admit_through = |through: u64,
                         events: &mut Vec<Event>,
                         event_cursor: &mut usize,
                         ready: &mut Vec<std::collections::VecDeque<usize>>,
                         processes: &mut Vec<Process>| {
        while *event_cursor < events.len() && events[*event_cursor].time <= through {
            let event = events[*event_cursor];
            *event_cursor += 1;
            if event.kind == 0 {
                ready[processes[event.index].level].push_back(event.index);
            } else {
                let level = processes[event.index].level.saturating_sub(1);
                processes[event.index].level = level;
                ready[level].push_back(event.index);
            }
        }
    };

    loop {
        admit_through(
            time,
            &mut events,
            &mut event_cursor,
            &mut ready,
            &mut processes,
        );

        let Some(level) = ready.iter().position(|queue| !queue.is_empty()) else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = ready[level].pop_front().expect("selected queue is nonempty");
        let job_id = jobs[index].id;

        if last_dispatch_id.is_some_and(|previous| previous != job_id) {
            time += u64::from(switch_cost);
        }
        admit_through(
            time,
            &mut events,
            &mut event_cursor,
            &mut ready,
            &mut processes,
        );

        let consumed = processes[index].remaining.min(quanta[level]);
        let start = time;
        time += u64::from(consumed);
        let end = time;
        processes[index].remaining -= consumed;

        let burst_finished = processes[index].remaining == 0;
        let final_burst = processes[index].burst_position + 1 == jobs[index].bursts.len();
        let finished = burst_finished && final_burst;

        runs.push(Run {
            id: job_id,
            start,
            end,
            finished,
            level,
        });

        admit_through(
            end,
            &mut events,
            &mut event_cursor,
            &mut ready,
            &mut processes,
        );

        if finished {
            processes[index].completion = Some(end);
        } else if burst_finished {
            let next_position = processes[index].burst_position + 2;
            let io_delay = jobs[index].bursts[processes[index].burst_position + 1];
            processes[index].burst_position = next_position;
            processes[index].remaining = jobs[index].bursts[next_position];
            events.push(Event {
                time: end + u64::from(io_delay),
                kind: 1,
                index,
            });
            events[event_cursor..].sort_by_key(|event| (event.time, event.kind, event.index));
            admit_through(
                end,
                &mut events,
                &mut event_cursor,
                &mut ready,
                &mut processes,
            );
        } else {
            processes[index].level = (level + 1).min(quanta.len() - 1);
            ready[processes[index].level].push_back(index);
        }

        last_dispatch_id = Some(job_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = processes[index].completion.expect("all jobs completed");
        let turnaround = completion - u64::from(job.arrival);
        let work: u64 = job.bursts.iter().map(|&burst| u64::from(burst)).sum();
        let waiting = turnaround - work;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 * 2 + count) / (2 * count);
    let mean_turnaround_x100 = (total_turnaround * 100 * 2 + count) / (2 * count);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}

fn main() {}
```
