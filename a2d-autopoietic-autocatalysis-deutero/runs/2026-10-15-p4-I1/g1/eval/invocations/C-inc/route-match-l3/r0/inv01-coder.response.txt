#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(high) = hex_value(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(low) = hex_value(bytes[i + 2]) else {
                return Err(base + i);
            };
            let decoded = (high << 4) | low;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_path(
    input: &str,
    request: bool,
) -> Result<(Vec<(String, usize)>, bool), Error> {
    let head = if request {
        let query = input.find('?');
        let fragment = input.find('#');
        let end = match (query, fragment) {
            (Some(a), Some(b)) => a.min(b),
            (Some(a), None) => a,
            (None, Some(b)) => b,
            (None, None) => input.len(),
        };
        &input[..end]
    } else {
        input
    };

    let invalid = |at| {
        if request {
            Error::InvalidPath { at }
        } else {
            Error::InvalidPattern { at }
        }
    };

    if !head.starts_with('/') {
        return Err(invalid(0));
    }
    if head == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = head.as_bytes();
    let trailing = head.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(invalid(start));
            }
            parts.push((head[start..i].to_string(), start));
            start = i + 1;
        }
        i += 1;
    }

    if parts.is_empty() {
        return Err(invalid(1));
    }
    Ok((parts, trailing))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (parts, trailing) = parse_path(pattern, false)?;
        let mut segments = Vec::with_capacity(parts.len());
        let mut names = std::collections::HashSet::new();
        let mut rank = Vec::with_capacity(parts.len() + 1);

        for (index, (raw, at)) in parts.iter().enumerate() {
            if let Some(name) = raw.strip_prefix(':') {
                if !valid_capture_name(name) || !names.insert(name.to_string()) {
                    return Err(Error::InvalidPattern { at: *at });
                }
                segments.push(Segment::Param(name.to_string()));
                rank.push(1);
            } else if let Some(name) = raw.strip_prefix('*') {
                if index + 1 != parts.len()
                    || !valid_capture_name(name)
                    || !names.insert(name.to_string())
                {
                    return Err(Error::InvalidPattern { at: *at });
                }
                segments.push(Segment::Wildcard(name.to_string()));
                rank.push(0);
            } else {
                let decoded =
                    decode_segment(raw, *at).map_err(|at| Error::InvalidPattern { at })?;
                segments.push(Segment::Static(decoded));
                rank.push(2);
            }
        }

        if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (parts, trailing) = parse_path(path, true)?;
        let mut decoded = Vec::with_capacity(parts.len());
        for (raw, at) in &parts {
            decoded.push(
                decode_segment(raw, *at).map_err(|at| Error::InvalidPath { at })?,
            );
        }

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }

            let mut params = Vec::new();
            let mut path_index = 0usize;
            let mut matched = true;

            for segment in &route.segments {
                match segment {
                    Segment::Static(value) => {
                        if decoded.get(path_index) != Some(value) {
                            matched = false;
                            break;
                        }
                        path_index += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = decoded.get(path_index) {
                            params.push((name.clone(), value.clone()));
                            path_index += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        params.push((name.clone(), decoded[path_index..].join("/")));
                        path_index = decoded.len();
                        break;
                    }
                }
            }

            if matched && path_index == decoded.len() {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let has_head = method == "HEAD"
            && candidates
                .iter()
                .any(|(route, _)| route.method == "HEAD");
        let selected_method = if method == "HEAD" && !has_head {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<(&Route, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(route, _)| route.method == selected_method)
            .map(|(route, params)| (*route, params.clone()))
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != trailing
            && !parts.is_empty()
        {
            let query = path.find('?');
            let fragment = path.find('#');
            let end = match (query, fragment) {
                (Some(a), Some(b)) => a.min(b),
                (Some(a), None) => a,
                (None, Some(b)) => b,
                (None, None) => path.len(),
            };
            let head = &path[..end];

            if route.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head[..head.len() - 1].to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}

fn main() {}
