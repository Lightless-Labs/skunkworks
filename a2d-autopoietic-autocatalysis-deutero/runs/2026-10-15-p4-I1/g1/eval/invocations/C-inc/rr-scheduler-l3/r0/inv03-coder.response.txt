```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrivals: Vec<usize> = (0..jobs.len()).collect();
    arrivals.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut queues: Vec<std::collections::VecDeque<usize>> = (0..quanta.len())
        .map(|_| std::collections::VecDeque::new())
        .collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();

    let mut time = 0_u64;
    let mut next_arrival = 0_usize;
    let mut completed = 0_usize;
    let mut last_dispatched_id: Option<u32> = None;

    while completed < jobs.len() {
        while next_arrival < arrivals.len()
            && u64::from(jobs[arrivals[next_arrival]].arrival) <= time
        {
            let index = arrivals[next_arrival];
            let job = &jobs[index];
            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
                completed += 1;
            } else {
                queues[job.level].push_back(index);
            }
            next_arrival += 1;
        }

        let selected = queues.iter_mut().find_map(|queue| queue.pop_front());

        let Some(index) = selected else {
            if next_arrival >= arrivals.len() {
                break;
            }

            let next_time = u64::from(jobs[arrivals[next_arrival]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let job = &jobs[index];
        if last_dispatched_id.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }

        while next_arrival < arrivals.len()
            && u64::from(jobs[arrivals[next_arrival]].arrival) <= time
        {
            let arrival_index = arrivals[next_arrival];
            let arriving_job = &jobs[arrival_index];
            if arriving_job.ticks == 0 {
                completion[arrival_index] = Some(u64::from(arriving_job.arrival));
                completed += 1;
            } else {
                queues[arriving_job.level].push_back(arrival_index);
            }
            next_arrival += 1;
        }

        let level = job.level;
        let consumed = remaining[index].min(quanta[level]);
        let start = time;
        time += u64::from(consumed);
        let finished = consumed == remaining[index];
        remaining[index] -= consumed;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished,
            level,
        });

        while next_arrival < arrivals.len()
            && u64::from(jobs[arrivals[next_arrival]].arrival) <= time
        {
            let arrival_index = arrivals[next_arrival];
            let arriving_job = &jobs[arrival_index];
            if arriving_job.ticks == 0 {
                completion[arrival_index] = Some(u64::from(arriving_job.arrival));
                completed += 1;
            } else {
                queues[arriving_job.level].push_back(arrival_index);
            }
            next_arrival += 1;
        }

        if finished {
            completion[index] = Some(time);
            completed += 1;
        } else {
            let demoted_level = (level + 1).min(quanta.len() - 1);
            queues[demoted_level].push_back(index);
        }

        last_dispatched_id = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap_or(time)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}

fn main() {}
```
