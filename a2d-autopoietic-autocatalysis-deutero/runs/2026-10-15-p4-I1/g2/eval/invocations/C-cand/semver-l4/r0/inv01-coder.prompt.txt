    let bytes = s.as_bytes();

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}

    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {

            (false, false) => x.as_bytes().cmp(y.as_bytes()),

        }

}

fn full_or_partial(token: &str, at: usize) -> Result<(Version, usize, bool), RangeError> {
    let fields: Vec<&str> = token.split('.').collect();
    let has_suffix = token.contains('-') || token.contains('+');
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        return parse_version(token).map(|v| (v, 3, true))
            .map_err(|error| RangeError::Version { at, error });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for f in &fields {
        if matches!(*f, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || f.is_empty() || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.starts_with('0')) || f.parse::<u64>().is_err()
            {
                return Err(RangeError::Syntax { at });
            }
            nums.push(f.parse::<u64>().unwrap());
        }
    }
    Ok((Version {
        major: *nums.get(0).unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: vec![], build: vec![]
    }, nums.len(), false))
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut n = v.clone();
    let target = match field { 0 => &mut n.major, 1 => &mut n.minor, _ => &mut n.patch };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field {
        0 => { n.minor = 0; n.patch = 0; }
        1 => n.patch = 0,
        _ => {}
    }
    n.pre = vec!["0".into()];
    n.build.clear();
    Ok(n)
}

fn add(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push((v.major, v.minor, v.patch));
    }
    arm.comps.push(Comparator { op, version: v, anchor });
}

fn parse_arm(s: &str, tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comps: vec![], anchors: vec![] };
    if let Some((idx, _)) = tokens.iter().find(|(_, t)| *t == "-") {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: *idx });
        }
        let (lv, lk, lf) = full_or_partial(tokens[0].1, tokens[0].0)?;
        let (uv, uk, uf) = full_or_partial(tokens[2].1, tokens[2].0)?;
        if lk == 0 { return Err(RangeError::Syntax { at: tokens[0].0 }); }
        if uk == 0 { return Err(RangeError::Syntax { at: tokens[2].0 }); }
        add(&mut arm, Op::Ge, lv, true);
        if uf {
            add(&mut arm, Op::Le, uv, true);
        } else {
            let upper = if uk == 1 { bump(&uv, 0, tokens[2].0)? } else { bump(&uv, 1, tokens[2].0)? };
            add(&mut arm, Op::Lt, upper, false);
        }
        let _ = lf;
        return Ok(arm);
    }
    for (at, token) in tokens {
        if *token == "||" { return Err(RangeError::Syntax { at: *at }); }
        if let Some(rest) = token.strip_prefix('^').or_else(|| token.strip_prefix('~')) {
            let caret = token.starts_with('^');
            if rest.is_empty() { return Err(RangeError::Syntax { at: *at }); }
            let (v, k, full) = full_or_partial(rest, at + 1)?;
            if k == 0 { return Err(RangeError::Syntax { at: *at }); }
            add(&mut arm, Op::Ge, v.clone(), true);
            let field = if caret {
                if v.major > 0 || k == 1 { 0 } else if v.minor > 0 || k == 2 { 1 } else { 2 }
            } else if k == 1 { 0 } else { 1 };
            let upper = bump(&v, field, *at)?;
            add(&mut arm, Op::Lt, upper, false);
            let _ = full;
        } else {
            let (op, rest, op_len) = if let Some(x) = token.strip_prefix(">=") { (Op::Ge, x, 2) }
                else if let Some(x) = token.strip_prefix("<=") { (Op::Le, x, 2) }
                else if let Some(x) = token.strip_prefix(">") { (Op::Gt, x, 1) }
                else if let Some(x) = token.strip_prefix("<") { (Op::Lt, x, 1) }
                else if let Some(x) = token.strip_prefix("=") { (Op::Eq, x, 1) }
                else { (Op::Eq, *token, 0) };
            if op_len > 0 {
                if rest.is_empty() || !rest.as_bytes()[0].is_ascii_digit() {
                    return Err(RangeError::Syntax { at: *at });
                }
                let v = parse_version(rest).map_err(|error| RangeError::Version { at: *at + op_len, error })?;
                add(&mut arm, op, v, true);
            } else {
                let first = token.as_bytes()[0];
                if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
                    return Err(RangeError::Syntax { at: *at });
                }
                let (v, k, full) = full_or_partial(token, *at)?;
                if full {
                    add(&mut arm, Op::Eq, v, true);
                } else if k == 0 {
                    // wildcard matches all
                } else {
                    add(&mut arm, Op::Ge, v.clone(), true);
                    if k < 3 {
                        let upper = bump(&v, if k == 1 { 0 } else { 1 }, *at)?;
                        add(&mut arm, Op::Lt, upper, false);
                    }
                }
            }
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut cursor = 0;
    let mut saw_any = false;
    while cursor < bytes.len() && whitespace(bytes[cursor]) { cursor += 1; }
    if cursor == bytes.len() { return Err(RangeError::Empty { at: 0 }); }
    loop {
        let arm_start = cursor;
        let mut sep = None;
        while cursor < bytes.len() {
            if whitespace(bytes[cursor]) {
                let mut j = cursor;
                while j < bytes.len() && whitespace(bytes[j]) { j += 1; }
                if j + 1 < bytes.len() && &s[j..j+2] == "||"
                    && (j == 0 || whitespace(bytes[j-1]))
                    && (j+2 == bytes.len() || whitespace(bytes[j+2]))
                {
                    sep = Some((j, j+2));
                    break;
                }
                cursor = j;
            } else {
                cursor += 1;
            }
        }
        let end = sep.map(|x| x.0).unwrap_or(bytes.len());
        let tokens = split_tokens(s, arm_start, end);
        if tokens.is_empty() {
            return Err(RangeError::Empty { at: if saw_any { arm_start } else { 0 } });
        }
        arms.push(parse_arm(s, &tokens)?);
        saw_any = true;
        if let Some((_, after)) = sep {
            cursor = after;
            while cursor < bytes.len() && whitespace(bytes[cursor]) { cursor += 1; }
            if cursor == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
        } else {
            break;
        }
    }
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !include && !v.pre.is_empty()
        && !arm.anchors.contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }
    arm.comps.iter().all(|c| {
        let o = compare_versions(v, &c.version);
        match c.op { Op::Eq => o == Ordering::Equal, Op::Lt => o == Ordering::Less,
            Op::Le => o != Ordering::Greater, Op::Gt => o == Ordering::Greater,
            Op::Ge => o != Ordering::Less }
    })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index: i, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if matches(v, r) && best.map_or(true, |b| {
            let c = compare_versions(v, &parsed[b]);
            if max { c == Ordering::Greater } else { c == Ordering::Less }
        }) { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    let mut candidates = Vec::new();
    for arm in a.arms.iter().chain(b.arms.iter()) {
        for c in &arm.comps {
            let mut v = c.version.clone();
            candidates.push(v.clone());
            if !v.pre.is_empty() {
                v.pre.push("0".into());
                candidates.push(v);
            }
            let mut stable = c.version.clone();
            stable.pre.clear();
            candidates.push(stable);
        }
    }
    candidates.push(Version { major: 0, minor: 0, patch: 0, pre: vec![], build: vec![] });
    for x in &candidates {
        if matches_with(x, a, include_prerelease) && matches_with(x, b, include_prerelease) {
            return true;
        }
    }
    false
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        for c in &arm.comps {
            match c.op {
                Op::Eq => {
                    lower = Some((c.version.clone(), true));
                    upper = Some((c.version.clone(), true));
                }
                Op::Gt | Op::Ge => {
                    let inc = c.op == Op::Ge;
                    if lower.as_ref().map_or(true, |(v, old)| {
                        compare_versions(&c.version, v) == Ordering::Greater
                            || (compare_versions(&c.version, v) == Ordering::Equal && !inc && *old)
                    }) { lower = Some((c.version.clone(), inc)); }
                }
                Op::Lt | Op::Le => {
                    let inc = c.op == Op::Le;
                    if upper.as_ref().map_or(true, |(v, old)| {
                        compare_versions(&c.version, v) == Ordering::Less
                            || (compare_versions(&c.version, v) == Ordering::Equal && !inc && *old)
                    }) { upper = Some((c.version.clone(), inc)); }
                }
            }
        }
        let anchors = {
            let mut a = arm.anchors.clone();
            a.sort();
            a.dedup();
            if a.is_empty() { "-".to_string() } else {
                a.iter().map(|(x,y,z)| format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")
            }
        };
        let fmt = |x: &Option<(Version, bool)>, lo: bool| -> String {
            if let Some((v, inc)) = x {
                let p = format!("{}.{}.{}{}", v.major, v.minor, v.patch,
                    if v.pre.is_empty() { String::new() } else { format!("-{}", v.pre.join(".")) });
                if lo { format!("{}{}", if *inc { ">=" } else { ">" }, p) }
                else { format!("{}{}", if *inc { "<=" } else { "<" }, p) }
            } else { "*".into() }
        };
        out.push(format!("[{};{};pre={}]", fmt(&lower, true), fmt(&upper, false), anchors));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }

#[derive(Clone, Debug)]
pub struct Constraint { pub label: String, pub expr: SetExpr }

#[derive(Clone, Debug)]
pub struct Package { pub name: String, pub candidates: Vec<String>, pub constraints: Vec<Constraint> }

#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package: String, pub version: String }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn visit_ranges<F: FnMut(&str, Vec<usize>)>(e: &SetExpr, path: &mut Vec<usize>, f: &mut F) {
    match e {
        SetExpr::Range(s) => f(s, path.clone()),
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                visit_ranges(x, path, f);
                path.pop();
            }
        }
    }
}

fn expr_matches(e: &SetExpr, v: &Version, include: bool, ranges: &[Range], cursor: &mut usize) -> bool {
    match e {
        SetExpr::Range(_) => {
            let r = &ranges[*cursor];
            *cursor += 1;
            matches_with(v, r, include)
        }
        SetExpr::Union(xs) => xs.iter().any(|x| expr_matches(x, v, include, ranges, cursor)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| expr_matches(x, v, include, ranges, cursor)),
    }
}

pub fn select_packages(packages: &[Package], preference: Preference, include_prerelease: bool) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) { return Err(SelectError::DuplicatePackage { name: p.name.clone() }); }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut versions: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut parsed = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            parsed.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index: i, error
            })?);
        }
        versions.push(parsed);
    }
    let mut all_ranges = Vec::new();
    for p in packages {
        let mut pranges = Vec::new();
        for c in &p.constraints {
            let mut found = Vec::new();
            visit_ranges(&c.expr, &mut Vec::new(), &mut |s, path| {
                match parse_range(s) {
                    Ok(r) => found.push(r),
                    Err(error) => {
                        // Error is returned after traversal through this single constraint.
                        pranges.push(Err(SelectError::InvalidRange {
                            package: p.name.clone(), label: c.label.clone(), path, error
                        }));
                    }
                }
            });
            pranges.extend(found.into_iter().map(Ok));
        }
        all_ranges.push(pranges);
    }
    for (pi, p) in packages.iter().enumerate() {
        for c in &all_ranges[pi] {
            if let Err(e) = c { return Err(e.clone()); }
        }
        let _ = p;
    }
    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let ranges: Vec<Range> = all_ranges[pi].iter().filter_map(|x| x.as_ref().ok().cloned()).collect();
        let mut qualifies = Vec::new();
        for (ci, v) in versions[pi].iter().enumerate() {
            let mut offset = 0;
            let mut ok = true;
            for c in &p.constraints {
                if !expr_matches(&c.expr, v, include_prerelease, &ranges, &mut offset) { ok = false; }
            }
            if ok { qualifies.push(ci); }
        }
        if qualifies.is_empty() {
            let mut conflict = Vec::new();
            if !p.candidates.is_empty() {
                for c in &p.constraints { conflict.push(c.label.clone()); }
                conflict.sort();
            }
            return Err(SelectError::NoSolution { package: p.name.clone(), labels: conflict });
        }
        let chosen = match preference {
            Preference::Highest => *qualifies.iter().max_by(|&&a, &&b| compare_versions(&versions[pi][a], &versions[pi][b])).unwrap(),
            Preference::Lowest => *qualifies.iter().min_by(|&&a, &&b| compare_versions(&versions[pi][a], &versions[pi][b])).unwrap(),
            Preference::StableFirst => {
                let stable: Vec<usize> = qualifies.iter().copied().filter(|&i| versions[pi][i].pre.is_empty()).collect();
                let pool = if stable.is_empty() { &qualifies } else { &stable };
                *pool.iter().max_by(|&&a, &&b| compare_versions(&versions[pi][a], &versions[pi][b])).unwrap()
            }
        };
        selections.push(Selection { package: p.name.clone(), version: p.candidates[chosen].clone() });
    }
    Ok(selections)
}
