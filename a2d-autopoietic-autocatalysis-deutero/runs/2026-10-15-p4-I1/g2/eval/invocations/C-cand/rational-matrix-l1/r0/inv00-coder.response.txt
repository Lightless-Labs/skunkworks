```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_fraction(mut num: i128, mut den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if den < 0 {
        num = -num;
        den = -den;
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    num /= g;
    den /= g;
    if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: num as i64, den: den as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_i128_fraction(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_fraction(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_fraction(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_i128_fraction(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        from_i128_fraction(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn parse_error(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i += 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let num_digits = num_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(parse_error(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[num_start] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let mut magnitude: u128 = 0;
        for &b in &bytes[num_start..num_end] {
            magnitude = magnitude * 10 + (b - b'0') as u128;
            let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
            if magnitude > limit {
                return Err(parse_error(0, ParseKind::OutOfRange));
            }
        }

        let denominator = if let (Some(start), Some(end)) = (den_start, den_end) {
            let mut value: u128 = 0;
            for &b in &bytes[start..end] {
                value = value * 10 + (b - b'0') as u128;
                if value > i64::MAX as u128 {
                    return Err(parse_error(start, ParseKind::OutOfRange));
                }
            }
            value as u64
        } else {
            1
        };

        if den_start.is_some() && denominator == 0 {
            return Err(parse_error(den_start.unwrap(), ParseKind::ZeroDenominator));
        }

        if let (Some(start), Some(_)) = (den_start, den_end) {
            if denominator == 1 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
            if magnitude == 0 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(magnitude, denominator as u128) > 1 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
        }

        let num = if negative {
            if magnitude == 1u128 << 63 {
                i64::MIN
            } else {
                -(magnitude as i64)
            }
        } else {
            magnitude as i64
        };
        Ok(Rational { num, den: denominator as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn main() {}
```
