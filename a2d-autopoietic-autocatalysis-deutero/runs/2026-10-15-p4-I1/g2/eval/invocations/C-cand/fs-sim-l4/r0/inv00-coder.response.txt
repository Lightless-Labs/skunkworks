```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
struct ParsedPath {
    absolute: bool,
    components: Vec<String>,
}

#[derive(Debug, Clone, Copy)]
enum FinalMode {
    Follow,
    NoFollow,
    CreateFile,
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        let (parent, name) = Self::split_parent(&canonical);

        let physical_parent = self.resolve(&parent, FinalMode::Follow)?;
        self.require_dir(&physical_parent)?;

        let candidate = Self::join(&physical_parent, &name);
        if self.entries.contains_key(&candidate) {
            return Err(Error::Exists(candidate));
        }
        self.entries.insert(candidate, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        let resolved = self.resolve(&canonical, FinalMode::CreateFile)?;

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = Self::split_parent(&resolved);
                self.require_dir(&parent)?;
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        let resolved = self.resolve(&canonical, FinalMode::Follow)?;

        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        let resolved = self.resolve(&canonical, FinalMode::Follow)?;
        self.require_dir(&resolved)?;

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == &resolved || resolved == "/" {
                if resolved == "/" && entry_path == "/" {
                    continue;
                }
            }
            let (parent, name) = Self::split_parent(entry_path);
            if parent == resolved && entry_path != &resolved {
                children.push((name, entry.kind()));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        let resolved = self.resolve(&canonical, FinalMode::Follow)?;
        self.require_dir(&resolved)?;
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from_parsed = Self::validate_path(from)?;
        let to_parsed = Self::validate_path(to)?;
        let from_canonical = Self::normalize(&from_parsed, &self.cwd);
        let to_canonical = Self::normalize(&to_parsed, &self.cwd);

        if from_canonical == "/" || to_canonical == "/" {
            return Err(Error::Root);
        }

        let (from_parent, from_name) = Self::split_parent(&from_canonical);
        let source_parent = self.resolve(&from_parent, FinalMode::Follow)?;
        self.require_dir(&source_parent)?;
        let source = Self::join(&source_parent, &from_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::contains(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (to_parent, to_name) = Self::split_parent(&to_canonical);
        let destination_parent = self.resolve(&to_parent, FinalMode::Follow)?;
        self.require_dir(&destination_parent)?;
        let destination = Self::join(&destination_parent, &to_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::contains(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination) {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        if destination_entry.is_some() {
            self.remove_subtree_entries(&destination);
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| **p == source || Self::contains(&source, p))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        for (old_path, _) in &moving {
            self.entries.remove(old_path);
        }
        for (old_path, entry) in moving {
            let suffix = &old_path[source.len()..];
            let new_path = format!("{destination}{suffix}");
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = Self::split_parent(&canonical);
        let physical_parent = self.resolve(&parent, FinalMode::Follow)?;
        self.require_dir(&physical_parent)?;
        let resolved = Self::join(&physical_parent, &name);

        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir) && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        if recursive {
            self.remove_subtree_entries(&resolved);
        } else {
            self.entries.remove(&resolved);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link_parsed = Self::validate_path(link)?;
        let link_canonical = Self::normalize(&link_parsed, &self.cwd);

        let target_parsed = Self::validate_target(target)?;
        let (parent, name) = Self::split_parent(&link_canonical);
        let physical_parent = self.resolve(&parent, FinalMode::Follow)?;
        self.require_dir(&physical_parent)?;
        let resolved_link = Self::join(&physical_parent, &name);

        if self.entries.contains_key(&resolved_link) {
            return Err(Error::Exists(resolved_link));
        }

        let _ = target_parsed;
        self.entries
            .insert(resolved_link, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        let (parent, name) = Self::split_parent(&canonical);
        let physical_parent = self.resolve(&parent, FinalMode::Follow)?;
        self.require_dir(&physical_parent)?;
        let resolved = Self::join(&physical_parent, &name);

        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(Entry::Dir) | Some(Entry::File(_)) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let parsed = Self::validate_path(path)?;
        let canonical = Self::normalize(&parsed, &self.cwd);
        let resolved = self.resolve(&canonical, FinalMode::Follow)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if !matches!(entry, Entry::Dir) {
            let size = match entry {
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            return Ok(vec![(resolved, size)]);
        }

        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| **p == resolved || Self::contains(&resolved, p))
            .cloned()
            .collect();
        paths.sort();

        let mut sizes = BTreeMap::<String, u64>::new();
        for p in paths.iter().rev() {
            let size = match self.entries.get(p).unwrap() {
                Entry::Dir => {
                    let mut total = 0u64;
                    for child in paths.iter().filter(|c| Self::parent_of(c) == *p) {
                        total = total.saturating_add(*sizes.get(child).unwrap_or(&0));
                    }
                    total
                }
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            sizes.insert(p.clone(), size);
        }

        let mut result: Vec<(String, u64)> = paths
            .into_iter()
            .map(|p| {
                let size = *sizes.get(&p).unwrap_or(&0);
                (p, size)
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    fn validate_path(path: &str) -> Result<ParsedPath, Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        Ok(ParsedPath {
            absolute: path.starts_with('/'),
            components: path
                .split('/')
                .filter(|c| !c.is_empty() && *c != ".")
                .map(str::to_string)
                .collect(),
        })
    }

    fn validate_target(target: &str) -> Result<ParsedPath, Error> {
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        Ok(ParsedPath {
            absolute: target.starts_with('/'),
            components: target
                .split('/')
                .filter(|c| !c.is_empty() && *c != ".")
                .map(str::to_string)
                .collect(),
        })
    }

    fn normalize(parsed: &ParsedPath, cwd: &str) -> String {
        let mut components = if parsed.absolute {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        Self::append_components(&mut components, &parsed.components);
        Self::from_components(&components)
    }

    fn append_components(acc: &mut Vec<String>, components: &[String]) {
        for component in components {
            if component == ".." {
                acc.pop();
            } else if component != "." && !component.is_empty() {
                acc.push(component.clone());
            }
        }
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|c| !c.is_empty())
            .map(str::to_string)
            .collect()
    }

    fn from_components(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn split_parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        match path.rfind('/') {
            Some(0) => ("/".to_string(), path[1..].to_string()),
            Some(i) => (path[..i].to_string(), path[i + 1..].to_string()),
            None => ("/".to_string(), path.to_string()),
        }
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn parent_of(path: &str) -> String {
        Self::split_parent(path).0
    }

    fn contains(parent: &str, path: &str) -> bool {
        if parent == path {
            true
        } else if parent == "/" {
            path.starts_with('/') && path != "/"
        } else {
            path.strip_prefix(parent)
                .is_some_and(|suffix| suffix.starts_with('/'))
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && Self::contains(path, candidate))
    }

    fn remove_subtree_entries(&mut self, path: &str) {
        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| **candidate == path || Self::contains(path, candidate))
            .cloned()
            .collect();
        for candidate in paths {
            self.entries.remove(&candidate);
        }
    }

    fn require_dir(&self, path: &str) -> Result<(), Error> {
        match self.entries.get(path) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_string())),
            None => Err(Error::Missing(path.to_string())),
        }
    }

    fn resolve(&self, path: &str, mode: FinalMode) -> Result<String, Error> {
        let mut pending = Self::components(path);
        let mut physical = Vec::<String>::new();
        let mut visited = BTreeSet::<String>::new();
        let mut expansions = 0usize;

        loop {
            let mut restarted = false;

            while !pending.is_empty() {
                let component = pending.remove(0);
                let candidate_components = {
                    let mut v = physical.clone();
                    v.push(component);
                    v
                };
                let candidate = Self::from_components(&candidate_components);
                let is_final = pending.is_empty();

                if !physical.is_empty() {
                    let parent = Self::from_components(&physical);
                    match self.entries.get(&parent) {
                        Some(Entry::Dir) => {}
                        Some(_) => return Err(Error::NotDir(parent)),
                        None => return Err(Error::Missing(parent)),
                    }
                }

                let entry = self.entries.get(&candidate);
                let should_follow = match entry {
                    Some(Entry::Link(_)) => {
                        !is_final
                            || matches!(mode, FinalMode::Follow | FinalMode::CreateFile)
                    }
                    _ => false,
                };

                if should_follow {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;

                    let target = match entry.unwrap() {
                        Entry::Link(target) => target,
                        _ => unreachable!(),
                    };
                    let parsed_target = Self::validate_target(target)
                        .map_err(|e| match e {
                            Error::InvalidTarget(i) => Error::InvalidTarget(i),
                            other => other,
                        })?;

                    let base = if parsed_target.absolute {
                        Vec::new()
                    } else {
                        physical.clone()
                    };
                    let mut target_components = base;
                    Self::append_components(&mut target_components, &parsed_target.components);
                    target_components.extend(pending);
                    let normalized = Self::from_components(&target_components);
                    pending = Self::components(&normalized);
                    physical.clear();
                    restarted = true;
                    break;
                }

                match entry {
                    Some(Entry::Dir) => {
                        physical.push(component);
                    }
                    Some(_) if !is_final => {
                        return Err(Error::NotDir(candidate));
                    }
                    Some(_) => {
                        physical.push(component);
                    }
                    None if is_final && matches!(mode, FinalMode::CreateFile) => {
                        let parent = Self::from_components(&physical);
                        match self.entries.get(&parent) {
                            Some(Entry::Dir) => {
                                physical.push(component);
                            }
                            Some(_) => return Err(Error::NotDir(parent)),
                            None => return Err(Error::Missing(parent)),
                        }
                    }
                    None => return Err(Error::Missing(candidate)),
                }
            }

            if restarted {
                continue;
            }

            let resolved = Self::from_components(&physical);
            if pending.is_empty() {
                if matches!(mode, FinalMode::Follow | FinalMode::CreateFile)
                    && matches!(self.entries.get(&resolved), Some(Entry::Link(_)))
                {
                    let link_target = match self.entries.get(&resolved).unwrap() {
                        Entry::Link(target) => target.clone(),
                        _ => unreachable!(),
                    };
                    if visited.contains(&resolved) {
                        return Err(Error::Loop(resolved));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(resolved));
                    }
                    visited.insert(resolved.clone());
                    expansions += 1;
                    let parsed_target = Self::validate_target(&link_target)?;
                    let base = if parsed_target.absolute {
                        Vec::new()
                    } else {
                        let (parent, _) = Self::split_parent(&resolved);
                        Self::components(&parent)
                    };
                    let mut target_components = base;
                    Self::append_components(&mut target_components, &parsed_target.components);
                    pending = Self::components(&Self::from_components(&target_components));
                    physical.clear();
                    continue;
                }

                if matches!(mode, FinalMode::CreateFile) && !self.entries.contains_key(&resolved) {
                    let parent = Self::parent_of(&resolved);
                    self.require_dir(&parent)?;
                }
                return Ok(resolved);
            }
        }
    }
}

fn main() {}
```
