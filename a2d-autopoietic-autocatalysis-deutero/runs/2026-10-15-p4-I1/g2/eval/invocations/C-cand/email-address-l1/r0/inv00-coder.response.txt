#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes.iter().enumerate().filter(|(_, &byte)| byte == b'@');
    let separator = match separators.next() {
        Some((at, _)) => at,
        None => return Err(EmailError::MissingAt),
    };

    if let Some((at, _)) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let mut previous_was_dot = false;
    for (at, &byte) in local.iter().enumerate() {
        if byte == b'.' {
            if at == 0 || at + 1 == local.len() || previous_was_dot {
                return Err(EmailError::LocalDot { at });
            }
            previous_was_dot = true;
        } else {
            let permitted = byte.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&byte);
            if !permitted {
                return Err(EmailError::LocalChar { at });
            }
            previous_was_dot = false;
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw_domain[..domain_len];
    let root_dot_at = if has_root_dot {
        Some(domain_start + domain_len)
    } else {
        None
    };

    let mut label_starts = Vec::new();
    let mut label_start = 0;
    for (index, &byte) in domain.iter().enumerate() {
        if byte == b'.' {
            if index == label_start {
                return Err(EmailError::EmptyLabel {
                    at: domain_start + index,
                });
            }
            validate_label(&domain[label_start..index], domain_start + label_start)?;
            label_starts.push(label_start);
            label_start = index + 1;
        }
    }

    if label_start == domain.len() {
        return Err(EmailError::EmptyLabel {
            at: root_dot_at.unwrap_or(domain_start + domain_len),
        });
    }

    validate_label(&domain[label_start..], domain_start + label_start)?;
    label_starts.push(label_start);

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_label_start = *label_starts.last().unwrap();
    let last_label = &domain[last_label_start..];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_label_start,
        });
    }

    let local = String::from_utf8(local.to_vec()).unwrap();
    let normalized_domain = domain
        .iter()
        .map(u8::to_ascii_lowercase)
        .map(char::from)
        .collect();

    Ok(Address {
        local,
        domain: normalized_domain,
        quoted: false,
    })
}

fn validate_label(label: &[u8], start: usize) -> Result<(), EmailError> {
    if label.len() > 63 {
        return Err(EmailError::LabelTooLong { at: start + 63 });
    }

    if label.first() == Some(&b'-') {
        return Err(EmailError::LabelHyphen { at: start });
    }

    if label.last() == Some(&b'-') {
        return Err(EmailError::LabelHyphen {
            at: start + label.len() - 1,
        });
    }

    for (index, &byte) in label.iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || byte == b'-') {
            return Err(EmailError::DomainChar { at: start + index });
        }
    }

    Ok(())
}

fn main() {}
