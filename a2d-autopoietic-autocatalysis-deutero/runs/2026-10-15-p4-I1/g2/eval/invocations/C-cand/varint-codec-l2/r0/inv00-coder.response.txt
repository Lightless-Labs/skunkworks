#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::new();

    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;

        if value == 0 {
            output.push(payload);
            break;
        }

        output.push(payload | 0x80);
    }

    output
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;

    for group in 0..10usize {
        let offset = start + group;
        if offset >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[offset];
        let payload = byte & 0x7f;

        if group == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow { offset, start });
            }
            value |= (payload as u64) << 63;
            return Ok((value, offset + 1));
        }

        value |= (payload as u64) << (group * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count);
    let mut cursor = start;

    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }

    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let encoded = if value >= 0 {
        (value as u64) << 1
    } else {
        ((value.unsigned_abs()) << 1).wrapping_sub(1)
    };
    encode_u64(encoded)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, cursor) = decode_u64(input, start)?;
    let decoded = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((decoded, cursor))
}

#[derive(Debug, Clone)]
pub struct StreamDecoder {
    position: usize,
    pending_start: usize,
    pending_value: u64,
    pending_groups: usize,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            pending_start: 0,
            pending_value: 0,
            pending_groups: 0,
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut next_position = self.position;
        let mut pending_start = self.pending_start;
        let mut pending_value = self.pending_value;
        let mut pending_groups = self.pending_groups;
        let mut output = Vec::new();

        for &byte in chunk {
            let offset = next_position;
            if pending_groups == 0 {
                pending_start = offset;
                pending_value = 0;
            }

            let payload = byte & 0x7f;
            if pending_groups == 9 {
                if payload > 1 || byte & 0x80 != 0 {
                    return Err(Error::Overflow {
                        offset,
                        start: pending_start,
                    });
                }
                pending_value |= (payload as u64) << 63;
                next_position += 1;
                output.push(pending_value);
                pending_value = 0;
                pending_groups = 0;
                pending_start = next_position;
                continue;
            }

            pending_value |= (payload as u64) << (pending_groups * 7);
            next_position += 1;
            pending_groups += 1;

            if byte & 0x80 == 0 {
                output.push(pending_value);
                pending_value = 0;
                pending_groups = 0;
                pending_start = next_position;
            }
        }

        self.position = next_position;
        self.pending_start = pending_start;
        self.pending_value = pending_value;
        self.pending_groups = pending_groups;
        Ok(output)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending_groups == 0 {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.pending_start,
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending_groups
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending_start = 0;
        self.pending_value = 0;
        self.pending_groups = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}

fn main() {}
