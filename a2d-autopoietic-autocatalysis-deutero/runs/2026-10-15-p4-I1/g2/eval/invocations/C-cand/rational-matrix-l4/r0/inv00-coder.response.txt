use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_wide(num as i128, den as i128)
    }

    fn from_wide(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let mut n = num;
        let mut d = den;
        if d < 0 {
            n = -n;
            d = -d;
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128
            || n > i64::MAX as i128
            || d < 1
            || d > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128
            + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128
            - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        Self::from_wide(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_wide(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if b[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < b.len() {
            if b[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < b.len() && b[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < b.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let negative = b[0] == b'-';
        let digits_start = if negative { 1 } else { 0 };
        if num_end - digits_start >= 2 && b[digits_start] == b'0' {
            return Err(parse_err(digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_end == 2 && b[1] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && b[ds] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let mut magnitude: u128 = 0;
        for &digit in &b[digits_start..num_end] {
            magnitude = magnitude * 10 + (digit - b'0') as u128;
            let limit = if negative {
                1u128 << 63
            } else {
                i64::MAX as u128
            };
            if magnitude > limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let num = if negative {
            if magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(magnitude as i64)
            }
        } else {
            magnitude as i64
        };

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            let mut den_mag: u128 = 0;
            for &digit in &b[ds..de] {
                den_mag = den_mag * 10 + (digit - b'0') as u128;
                if den_mag > i64::MAX as u128 {
                    return Err(parse_err(ds, ParseKind::OutOfRange));
                }
            }
            if den_mag == 0 {
                return Err(parse_err(ds, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            if num == 0 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num.unsigned_abs() as u128, den_mag) > 1 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            Ok(Self {
                num,
                den: den_mag as i64,
            })
        } else {
            Ok(Self { num, den: 1 })
        }
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(
        rows: usize,
        cols: usize,
        data: Vec<Rational>,
    ) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = vec![Rational { num: 0, den: 1 }; self.data.len()];
        if self.rows != 0 && self.cols != 0 {
            for r in 0..self.rows {
                for c in 0..self.cols {
                    data[c * self.rows + r] = self.data[r * self.cols + c];
                }
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        mut data: Vec<Rational>,
        rows: usize,
        cols: usize,
        pivot_cols: usize,
    ) -> Result<(Vec<Rational>, Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        let zero = Rational { num: 0, den: 1 };

        for col in 0..pivot_cols {
            if p == rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for r in p..rows {
                let value = data[r * cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), r);
                if chosen.map_or(true, |q| {
                    let other = data[q * cols + col];
                    key < (other.den, other.num.unsigned_abs(), q)
                }) {
                    chosen = Some(r);
                }
            }
            let Some(q) = chosen else {
                continue;
            };

            if q != p {
                for c in 0..cols {
                    data.swap(p * cols + c, q * cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = data[p * cols + col];
            let mut normalized = Vec::with_capacity(cols);
            for c in 0..cols {
                normalized.push(data[p * cols + c].div(&pivot).map_err(|source| {
                    MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    }
                })?);
            }
            for c in 0..cols {
                data[p * cols + c] = normalized[c];
            }

            for r in 0..rows {
                if r == p {
                    continue;
                }
                let factor = data[r * cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..cols {
                    let product = factor.mul(&normalized[c]).map_err(|source| {
                        MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                    data[r * cols + c] = data[r * cols + c]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        })?;
                }
            }

            pivots.push(col);
            p += 1;
        }

        let _ = zero;
        Ok((data, pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (data, pivots, swaps) =
            Self::eliminate(self.data.clone(), self.rows, self.cols, self.cols)?;
        Ok(Reduction {
            matrix: Self {
                rows: self.rows,
                cols: self.cols,
                data,
            },
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut augmented = Vec::with_capacity(len);
        for r in 0..n {
            augmented.extend_from_slice(&self.data[r * n..r * n + n]);
            for c in 0..n {
                augmented.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let (data, pivots, _) = Self::eliminate(augmented, n, width, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&data[r * width + n..r * width + width]);
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: out,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        if m != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: m,
                rhs: rhs.rows,
            });
        }
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut augmented = Vec::with_capacity(len);
        for r in 0..m {
            augmented.extend_from_slice(&self.data[r * n..r * n + n]);
            augmented.extend_from_slice(&rhs.data[r * k..r * k + k]);
        }

        let (reduced, pivots, _) =
            Self::eliminate(augmented, m, width, n).map_err(SolveError::Matrix)?;

        for r in 0..m {
            let coefficient_zero = (0..n).all(|c| reduced[r * width + c].num == 0);
            if coefficient_zero {
                for c in 0..k {
                    if reduced[r * width + n + c].num != 0 {
                        let witness = reduced[r * width..r * width + width].to_vec();
                        return Ok(Solution::None {
                            row: r,
                            rhs_col: c,
                            witness,
                        });
                    }
                }
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(
            MatrixError::ShapeOverflow,
        ))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] = reduced[pivot_row * width + n + c];
            }
        }
        let particular = Self {
            rows: n,
            cols: k,
            data: particular,
        };

        let mut free = Vec::new();
        let mut pivot_mark = vec![false; n];
        for &p in &pivots {
            pivot_mark[p] = true;
        }
        for c in 0..n {
            if !pivot_mark[c] {
                free.push(c);
            }
        }

        if pivots.len() == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let f = free.len();
        let basis_len = n
            .checked_mul(f)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in pivots.iter().enumerate() {
                let coefficient = reduced[i * width + free_col];
                basis[pivot_col * f + j] =
                    coefficient.neg().map_err(|source| SolveError::Matrix(
                        MatrixError::Arithmetic {
                            row: pivot_col,
                            col: j,
                            step: pivots.len(),
                            source,
                        },
                    ))?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self {
                rows: n,
                cols: f,
                data: basis,
            },
            pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn main() {}
