    start: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {

        return Err(syntax_error(field, start));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {

            field,
            offset: start,

                offset: start,

    item_offset: usize,
    field: Field,
    selected: &mut [bool],

            field,
            offset: item_offset,
        });

        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))

        return Err(syntax_error(field, item_offset));

        return Err(syntax_error(field, item_offset));

        (min, max, item_offset, item_offset)

            return Err(syntax_error(field, item_offset));

            return Err(ParseError::ReversedRange {
                field,
                offset: b_start,
            });
        }
        (a, b, a_start, b_start)
    } else {
        let a = parse_endpoint(base, item_offset, field)?;
        let (min, max) = domain(field);

        (a, end, item_offset, item_offset)
    };

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {

        return Err(ParseError::FieldCount { found: parts.len() });
    }

        let (min, max) = domain(FIELDS[i]);
        vec![false; (max - min + 1) as usize]
    });

        let field = FIELDS[i];
        let bytes = part.as_bytes();

            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }

            day_unrestricted[0] = *part == "*";
        } else if i == 4 {
            day_unrestricted[1] = *part == "*";

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn to_ordinal_minute(at: CivilDateTime) -> u64 {
    let mut days = 0u64;
    for y in 2000..at.year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..at.month {
        days += days_in_month(at.year, m) as u64;
    }
    days += at.day as u64 - 1;
    (((days * 24 + at.hour as u64) * 60) + at.minute as u64)
}

fn from_ordinal_minute(mut ordinal: u64) -> CivilDateTime {
    let minute = (ordinal % 60) as u8;
    ordinal /= 60;
    let hour = (ordinal % 24) as u8;
    let mut days = ordinal / 24;
    let mut year = 2000u16;
    loop {
        let yd = if is_leap(year) { 366 } else { 365 };
        if days < yd {
            break;
        }
        days -= yd;
        year += 1;
    }
    let mut month = 1u8;
    loop {
        let md = days_in_month(year, month) as u64;
        if days < md {
            break;
        }
        days -= md;
        month += 1;
    }
    CivilDateTime {
        year,
        month,
        day: days as u8 + 1,
        hour,
        minute,
    }
}
