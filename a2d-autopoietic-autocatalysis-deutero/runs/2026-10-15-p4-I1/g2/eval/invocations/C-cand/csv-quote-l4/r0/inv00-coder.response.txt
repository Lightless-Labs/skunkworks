pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {

    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

    input: String,

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;
    let mut record_index = 0usize;
    let mut touched = false;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if state != 2 && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        match state {
            0 => {
                if b == delimiter {
                    record.push(String::new());
                    touched = true;
                    pos += 1;
                } else if b == quote {
                    quote_start = pos;
                    state = 2;
                    touched = true;
                    pos += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(String::new());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    touched = false;
                    pos += if b == b'\r' { 2 } else { 1 };
                } else {
                    let start = pos;
                    while pos < bytes.len()
                        && bytes[pos] != delimiter
                        && bytes[pos] != quote
                        && bytes[pos] != b'\n'
                        && bytes[pos] != b'\r'
                        && bytes[pos] != 0
                    {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                    state = 1;
                    touched = true;
                }
            }
            1 => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    touched = true;
                    pos += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = 0;
                    touched = false;
                    pos += if b == b'\r' { 2 } else { 1 };
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                } else {
                    let start = pos;
                    while pos < bytes.len()
                        && bytes[pos] != delimiter
                        && bytes[pos] != quote
                        && bytes[pos] != b'\n'
                        && bytes[pos] != b'\r'
                        && bytes[pos] != 0
                    {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                }
            }
            2 => {
                if b == quote {
                    if bytes.get(pos + 1) == Some(&quote) {
                        field.push(quote as char);
                        pos += 2;
                    } else {
                        state = 3;
                        pos += 1;
                    }
                } else {
                    let start = pos;
                    while pos < bytes.len() && bytes[pos] != quote && bytes[pos] != 0 {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                }
            }
            _ => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    touched = true;
                    pos += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = 0;
                    touched = false;
                    pos += if b == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if touched || !record.is_empty() || !field.is_empty() || state != 0 {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn table_from_input(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;

    let records = parse_dialect(input, config.delimiter, config.quote)?;
    let (header, rows) = if config.headers {
        let mut iter = records.into_iter();
        (iter.next(), iter.collect::<Vec<_>>())
    } else {
        (None, records)
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in header.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(seen_name, _)| *seen_name == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }

        for (row_offset, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: row_offset + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = table_from_input(&self.input, self.config);
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    Collect,
    FirstPerRow,

#[derive(Debug, Clone, PartialEq, Eq)]

    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 {
        return false;
    }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if dot <= start {
        return false;
    }
    if b[start] == b'0' {
        dot == start + 1
    } else {
        (b'1'..=b'9').contains(&b[start]) && b[start + 1..dot].iter().all(u8::is_ascii_digit)
    }
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) || s == "-0.00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let dot = body.len() - 3;
    let whole = body[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let fraction = body[dot + 1..]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let magnitude = whole
        .checked_mul(100)
        .and_then(|v| v.checked_add(fraction))
        .ok_or(CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::FirstPerRow && row_failed
                || policy == ErrorPolicy::Stop && !result.errors.is_empty()
            {
                output.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (column, cell) = match resolved {
                None => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    });
                    output.push(None);
                    row_failed = true;
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                    continue;
                }
                Some(index) => match row.get(index) {
                    None => {
                        result.errors.push(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(index),
                            kind: CellErrorKind::MissingCell,
                        });
                        output.push(None);
                        row_failed = true;
                        if policy == ErrorPolicy::Stop {
                            result.rows.push(output);
                            return result;
                        }
                        continue;
                    }
                    Some(cell) => (index, cell),
                },
            };

            let _ = column;
            if cell.is_empty() && spec.nullable {
                output.push(Some(Value::Null));
                continue;
            }
            if cell.is_empty() && spec.kind != ColumnType::Text {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: resolved,
                    kind: CellErrorKind::Empty,
                });
                output.push(None);
                row_failed = true;
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    return result;
                }
                continue;
            }

            let parsed = match spec.kind {
                ColumnType::Text => Ok(Value::Text(cell.clone())),
                ColumnType::Integer => parse_integer(cell).map(Value::Integer),
                ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
                ColumnType::Boolean => match cell.as_str() {
                    "yes" | "1" => Ok(Value::Boolean(true)),
                    "no" | "0" => Ok(Value::Boolean(false)),
                    _ => Err(CellErrorKind::InvalidBoolean),
                },
            };

            match parsed {
                Ok(value) => output.push(Some(value)),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: resolved,
                        kind,
                    });
                    output.push(None);
                    row_failed = true;
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }
        result.rows.push(output);
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn duplicate_header(header: &[String]) -> Option<(String, usize, usize)> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(candidate, _)| *candidate == name) {
            return Some((name.clone(), *first, second));
        }
        seen.push((name, second));
    }
    None
}

fn row_vector_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        match x.as_bytes().cmp(y.as_bytes()) {
            std::cmp::Ordering::Equal => {}
            order => return order,
        }
    }
    a.len().cmp(&b.len())
}

fn encode_field(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) -> String {
    let bytes = field.as_bytes();
    let quote_it = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole_empty && field.is_empty());

    if !quote_it {
        return field.to_string();
    }

    let mut out = String::with_capacity(field.len() + 2);
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let header_record = table.header.as_deref();
    let mut all_records: Vec<&[String]> = Vec::new();
    if let Some(header) = header_record {
        all_records.push(header);
    }
    for row in &table.rows {
        all_records.push(row);
    }

    for (record_index, record) in all_records.iter().enumerate() {
        for (column, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = header_record {
        for (row_index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in all_records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = header_record {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some((name, first, second)) = duplicate_header(header) {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }
        }
    }

    let mut output_header: Option<Vec<String>> = None;
    let mut output_rows: Vec<Vec<String>> = Vec::new();

    match config.mode {
        WriteMode::Preserve => {
            output_header = table.header.clone();
            output_rows = table.rows.clone();
        }
        WriteMode::Canonical => {
            if let Some(header) = header_record {
                let mut chosen: Vec<usize> = Vec::new();
                for (i, name) in header.iter().enumerate() {
                    let seen = chosen.iter().any(|&j| header[j] == *name);
                    if !seen {
                        chosen.push(i);
                    } else if table.duplicates == DuplicatePolicy::KeepLast {
                        chosen.retain(|&j| header[j] != *name);
                        chosen.push(i);
                    }
                }
                chosen.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));

                output_header = Some(chosen.iter().map(|&i| header[i].clone()).collect());
                output_rows = table
                    .rows
                    .iter()
                    .map(|row| chosen.iter().map(|&i| row[i].clone()).collect())
                    .collect();
            } else {
                output_rows = table.rows.clone();
            }
            output_rows.sort_by(|a, b| row_vector_cmp(a, b));
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    if let Some(header) = output_header {
        records.push(header);
    }
    records.extend(output_rows);

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    let terminator = match config.mode {
        WriteMode::Canonical => "\n",
        WriteMode::Preserve => match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        },
    };

    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            // The separator is emitted after the prior record below.
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            let first_first = record_index == 0 && column_index == 0;
            let sole_empty = record.len() == 1 && field.is_empty();
            out.push_str(&encode_field(
                field,
                config.delimiter,
                config.quote,
                first_first,
                sole_empty,
            ));
        }
        let should_terminate = if config.mode == WriteMode::Canonical {
            true
        } else {
            record_index + 1 < records.len() || config.final_terminator
        };
        if should_terminate {
            out.push_str(terminator);
        }
    }

    Ok(out)
}
