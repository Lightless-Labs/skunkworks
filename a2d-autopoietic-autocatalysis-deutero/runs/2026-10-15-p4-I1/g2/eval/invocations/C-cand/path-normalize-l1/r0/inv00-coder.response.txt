```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();

    if let Some(at) = bytes.iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = bytes.iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = bytes
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = bytes[0] == b'/';
    let directory_intent = bytes.last() == Some(&b'/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..");

    let mut components: Vec<(usize, &str)> = Vec::new();
    let mut start = 0;

    while start < path.len() {
        while start < path.len() && bytes[start] == b'/' {
            start += 1;
        }
        if start == path.len() {
            break;
        }

        let component_start = start;
        while start < path.len() && bytes[start] != b'/' {
            start += 1;
        }
        let component = &path[component_start..start];

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|(_, c)| *c != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component_start });
                } else {
                    components.push((component_start, component));
                }
            }
            _ => components.push((component_start, component)),
        }
    }

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    for (index, (_, component)) in components.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

fn main() {}
```
