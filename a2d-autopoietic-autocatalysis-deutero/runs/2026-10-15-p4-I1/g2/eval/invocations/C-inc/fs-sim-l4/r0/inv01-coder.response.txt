```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
struct ParsedPath {
    absolute: bool,
    components: Vec<String>,
}

#[derive(Debug, Clone)]
struct Resolution {
    path: String,
    entry: Entry,
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<ParsedPath, Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(ParsedPath {
            absolute: path.starts_with('/'),
            components: path
                .split('/')
                .filter(|part| !part.is_empty())
                .map(str::to_string)
                .collect(),
        })
    }

    fn cwd_components(&self) -> Vec<String> {
        Self::components_of(&self.cwd)
    }

    fn components_of(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize_parts(base: &[String], components: &[String]) -> Vec<String> {
        let mut parts = base.to_vec();
        for component in components {
            match component.as_str() {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(component.clone()),
            }
        }
        parts
    }

    fn normalize(&self, parsed: &ParsedPath) -> String {
        let base = if parsed.absolute {
            Vec::new()
        } else {
            self.cwd_components()
        };
        Self::canonical(&Self::normalize_parts(&base, &parsed.components))
    }

    fn parent_path(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        let parts = Self::components_of(path);
        Self::canonical(&parts[..parts.len() - 1])
    }

    fn is_within(path: &str, directory: &str) -> bool {
        if directory == "/" {
            return path.starts_with('/');
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn normalize_target(target: &str, link_parent: &str) -> Vec<String> {
        let absolute = target.starts_with('/');
        let base = if absolute {
            Vec::new()
        } else {
            Self::components_of(link_parent)
        };
        let components: Vec<String> = target
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect();
        Self::normalize_parts(&base, &components)
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        allow_missing_final: bool,
    ) -> Result<Option<Resolution>, Error> {
        let mut pending = Self::components_of(path);
        let mut physical: Vec<String> = Vec::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                let canonical = Self::canonical(&physical);
                let entry = self
                    .entries
                    .get(&canonical)
                    .cloned()
                    .ok_or_else(|| Error::Missing(canonical.clone()))?;
                return Ok(Some(Resolution {
                    path: canonical,
                    entry,
                }));
            }

            let component = pending.remove(0);
            let mut candidate_parts = physical.clone();
            candidate_parts.push(component);
            let candidate = Self::canonical(&candidate_parts);
            let is_final = pending.is_empty();

            let entry = match self.entries.get(&candidate) {
                Some(entry) => entry.clone(),
                None if is_final && allow_missing_final => return Ok(None),
                None => return Err(Error::Missing(candidate)),
            };

            if !is_final && entry.kind() != Kind::Dir {
                return Err(Error::NotDir(candidate));
            }

            if let Entry::Link(target) = &entry {
                if !is_final || follow_final {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let mut expanded =
                        Self::normalize_target(target, &Self::canonical(&physical));
                    expanded.extend(pending);
                    physical.clear();
                    pending = expanded;
                    continue;
                }
            }

            physical.push(candidate_parts.last().unwrap().clone());
            if is_final {
                return Ok(Some(Resolution {
                    path: candidate,
                    entry,
                }));
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent_path(path);
        let name = if path == "/" {
            String::new()
        } else {
            Self::components_of(path).last().cloned().unwrap_or_default()
        };
        let resolved = self
            .resolve(&parent, true, false)?
            .expect("resolved parent exists");
        match resolved.entry {
            Entry::Dir => Ok((resolved.path, name)),
            _ => Err(Error::NotDir(resolved.path)),
        }
    }

    fn path_exists(&self, path: &str) -> bool {
        self.entries.contains_key(path)
    }

    fn descendants(&self, root: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| Self::is_within(path, root))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        let (parent, name) = self.resolve_parent(&canonical)?;
        let destination = if name.is_empty() {
            parent
        } else {
            Self::child_path(&parent, &name)
        };
        if self.path_exists(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        let resolved = self.resolve(&canonical, true, true)?;
        let destination = match resolved {
            Some(Resolution {
                path,
                entry: Entry::Dir,
            }) => return Err(Error::IsDir(path)),
            Some(Resolution { path, .. }) => path,
            None => canonical.clone(),
        };

        if !self.path_exists(&destination) {
            let parent = Self::parent_path(&destination);
            let resolved_parent = self
                .resolve(&parent, true, false)?
                .expect("resolved parent exists");
            match resolved_parent.entry {
                Entry::Dir => {}
                _ => return Err(Error::NotDir(resolved_parent.path)),
            }
        }

        self.entries
            .insert(destination, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        let resolved = self
            .resolve(&canonical, true, false)?
            .expect("resolved path exists");
        match resolved.entry {
            Entry::File(bytes) => Ok(bytes),
            Entry::Dir => Err(Error::IsDir(resolved.path)),
            Entry::Link(_) => unreachable!(),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        let resolved = self
            .resolve(&canonical, true, false)?
            .expect("resolved path exists");
        if resolved.entry != Entry::Dir {
            return Err(Error::NotDir(resolved.path));
        }

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == &resolved.path || !Self::is_within(entry_path, &resolved.path) {
                continue;
            }
            let relative = if resolved.path == "/" {
                &entry_path[1..]
            } else {
                &entry_path[resolved.path.len() + 1..]
            };
            if !relative.contains('/') {
                children.push((relative.to_string(), entry.kind()));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        let resolved = self
            .resolve(&canonical, true, false)?
            .expect("resolved path exists");
        match resolved.entry {
            Entry::Dir => {
                self.cwd = resolved.path;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved.path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from_parsed = Self::validate(from)?;
        let to_parsed = Self::validate(to)?;
        let source = self.normalize(&from_parsed);
        let destination = self.normalize(&to_parsed);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::child_path(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if source_entry == Entry::Dir && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = Self::child_path(&destination_parent, &destination_name);

        if self
            .entries
            .get(&destination_path)
            .is_some_and(|entry| *entry == Entry::Dir)
            && Self::is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if source_entry == Entry::Dir && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref target_entry) = destination_entry {
            let source_is_dir = source_entry == Entry::Dir;
            let target_is_dir = *target_entry == Entry::Dir;
            if source_is_dir && !target_is_dir {
                return Err(Error::NotDir(destination_path));
            }
            if !source_is_dir && target_is_dir {
                return Err(Error::IsDir(destination_path));
            }
            if source_is_dir
                && target_is_dir
                && self
                    .entries
                    .keys()
                    .any(|path| path != &destination_path && Self::is_within(path, &destination_path))
            {
                return Err(Error::NotEmpty(destination_path));
            }
        }

        let moving = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &source_path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect::<Vec<_>>();

        if destination_entry.is_some() {
            self.entries.remove(&destination_path);
        }
        for (old, _) in &moving {
            self.entries.remove(old);
        }
        for (old, entry) in moving {
            let suffix = &old[source_path.len()..];
            self.entries.insert(format!("{}{}", destination_path, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = Self::child_path(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if entry == Entry::Dir && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        let descendants = self
            .entries
            .keys()
            .filter(|path| *path != &target && Self::is_within(path, &target))
            .cloned()
            .collect::<Vec<_>>();
        if entry == Entry::Dir && !recursive && !descendants.is_empty() {
            return Err(Error::NotEmpty(target));
        }

        self.entries.remove(&target);
        if recursive {
            for descendant in descendants {
                self.entries.remove(&descendant);
            }
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link_parsed = Self::validate(link)?;
        let link_canonical = self.normalize(&link_parsed);

        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (parent, name) = self.resolve_parent(&link_canonical)?;
        let destination = if name.is_empty() {
            parent
        } else {
            Self::child_path(&parent, &name)
        };
        if self.path_exists(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        if canonical == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target_path = Self::child_path(&parent, &name);
        match self.entries.get(&target_path) {
            None => Err(Error::Missing(target_path)),
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let parsed = Self::validate(path)?;
        let canonical = self.normalize(&parsed);
        let resolved = self
            .resolve(&canonical, true, false)?
            .expect("resolved path exists");
        let root = resolved.path;
        let descendants = self.descendants(&root);
        let mut sizes = BTreeMap::<String, u64>::new();

        for entry_path in &descendants {
            let size = match self.entries.get(entry_path).expect("entry exists") {
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(entry_path.clone(), size);
        }

        let mut directories = descendants
            .iter()
            .filter(|path| self.entries.get(*path) == Some(&Entry::Dir))
            .cloned()
            .collect::<Vec<_>>();
        directories.sort_by_key(|path| std::cmp::Reverse(Self::components_of(path).len()));

        for directory in directories {
            let total = sizes
                .iter()
                .filter(|(path, _)| *path != &directory && Self::is_within(path, &directory))
                .map(|(_, size)| *size)
                .sum::<u64>();
            sizes.insert(directory, total);
        }

        Ok(sizes.into_iter().collect())
    }
}

fn main() {}
```
