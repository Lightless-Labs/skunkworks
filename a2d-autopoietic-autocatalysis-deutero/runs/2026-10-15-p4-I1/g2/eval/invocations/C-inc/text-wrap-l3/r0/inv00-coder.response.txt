#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Piece {
    text: Vec<char>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<Piece>,
    gaps: Vec<usize>,
    content_len: usize,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(segment: &[char]) -> Vec<Vec<char>> {
    let mut words = Vec::new();
    let mut word = Vec::new();

    for &c in segment {
        if is_separator(c) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else if c != '~' {
            word.push(c);
        } else {
            // Markers are removed from display text; handled separately below.
            word.push(c);
        }
    }

    if !word.is_empty() {
        words.push(word);
    }
    words
}

fn line_capacity(options: &WrapOptions, first: bool) -> usize {
    options.width - if first {
        options.first_indent
    } else {
        options.rest_indent
    }
}

fn render_line(line: &Line, width: usize, justify: bool, final_line: bool, line_index: usize) -> String {
    let mut output = " ".repeat(line.indent);
    let gaps_count = line.gaps.len();

    let distribute = justify && !final_line && line.pieces.len() >= 2 && gaps_count > 0;
    let extra = if distribute {
        width.saturating_sub(line.indent + line.content_len)
    } else {
        0
    };
    let each = if distribute { extra / gaps_count } else { 0 };
    let remainder = if distribute { extra % gaps_count } else { 0 };
    let start_gap = if distribute { line_index % gaps_count } else { 0 };

    for (i, piece) in line.pieces.iter().enumerate() {
        output.extend(piece.text.iter());
        if i < gaps_count {
            output.push(' ');
            if distribute {
                output.push_str(&" ".repeat(each));
                let offset = (i + gaps_count - start_gap) % gaps_count;
                if offset < remainder {
                    output.push(' ');
                }
            }
        }
    }
    output
}

fn push_piece(line: &mut Line, chars: Vec<char>, add_gap: bool) {
    if add_gap && !line.pieces.is_empty() {
        line.gaps.push(1);
        line.content_len += 1;
    }
    line.content_len += chars.len();
    line.pieces.push(Piece { text: chars });
}

fn finish_line(lines: &mut Vec<Line>, current: &mut Line) {
    if !current.pieces.is_empty() {
        lines.push(std::mem::replace(
            current,
            Line {
                indent: 0,
                pieces: Vec::new(),
                gaps: Vec::new(),
                content_len: 0,
            },
        ));
    }
}

fn layout_paragraph(words: &[Vec<char>], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut raw_lines: Vec<Line> = Vec::new();
    let mut line_index = 0usize;
    let mut current = Line {
        indent: options.first_indent,
        pieces: Vec::new(),
        gaps: Vec::new(),
        content_len: 0,
    };

    for word in words {
        let mut offset = 0usize;
        let mut add_gap = false;

        while offset < word.len() {
            let capacity = line_capacity(options, line_index == 0);
            let room = capacity.saturating_sub(current.content_len);
            let available = room.saturating_sub(if current.pieces.is_empty() { 0 } else { 1 });

            let suffix_len = word.len() - offset;
            if suffix_len <= available {
                let fragment: Vec<char> = word[offset..].iter().copied().filter(|&c| c != '~').collect();
                push_piece(&mut current, fragment, add_gap);
                offset = word.len();
                add_gap = true;
                continue;
            }

            let mut opportunities = Vec::new();
            for marker in offset + 1..word.len() {
                if word[marker] == '~' {
                    let prefix_len = word[offset..marker].iter().filter(|&&c| c != '~').count();
                    if prefix_len + 1 <= available {
                        opportunities.push((marker, prefix_len));
                    }
                }
            }

            let accepted = if let Some(&(marker, prefix_len)) = opportunities.last() {
                let unused = available - prefix_len - 1;
                if current.pieces.is_empty()
                    || unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                        < room.saturating_mul(room)
                {
                    Some((marker, prefix_len))
                } else {
                    None
                }
            } else {
                None
            };

            if let Some((marker, _)) = accepted {
                let mut fragment: Vec<char> = word[offset..marker]
                    .iter()
                    .copied()
                    .filter(|&c| c != '~')
                    .collect();
                fragment.push('-');
                push_piece(&mut current, fragment, add_gap);
                offset = marker + 1;
                finish_line(&mut raw_lines, &mut current);
                line_index += 1;
                current.indent = options.rest_indent;
                add_gap = false;
                continue;
            }

            if !current.pieces.is_empty() {
                finish_line(&mut raw_lines, &mut current);
                line_index += 1;
                current.indent = options.rest_indent;
                add_gap = false;
                continue;
            }

            let capacity = line_capacity(options, line_index == 0);
            let remaining = word.len() - offset;
            if remaining > capacity {
                let mut consumed = 0usize;
                let mut fragment = Vec::new();
                while offset + consumed < word.len() && fragment.len() < capacity {
                    let c = word[offset + consumed];
                    if c != '~' {
                        fragment.push(c);
                    }
                    consumed += 1;
                }
                push_piece(&mut current, fragment, false);
                offset += consumed;
                finish_line(&mut raw_lines, &mut current);
                line_index += 1;
                current.indent = options.rest_indent;
                add_gap = false;
            } else {
                let fragment = word[offset..].iter().copied().filter(|&c| c != '~').collect();
                push_piece(&mut current, fragment, false);
                offset = word.len();
            }
        }

        add_gap = true;
        let _ = add_gap;
    }

    finish_line(&mut raw_lines, &mut current);
    let last = raw_lines.len().saturating_sub(1);
    raw_lines
        .iter()
        .enumerate()
        .map(|(i, line)| render_line(line, options.width, options.justify, i == last, i))
        .collect()
}

fn process_segment(chars: &[char], options: &WrapOptions) -> Vec<String> {
    let words = parse_words(chars);
    layout_paragraph(&words, options)
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let indexed: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..indexed.len() {
        if indexed[i].1 != '~' {
            continue;
        }
        let valid_left = i > 0 && indexed[i - 1].1 != '~' && !is_separator(indexed[i - 1].1);
        let valid_right = i + 1 < indexed.len()
            && indexed[i + 1].1 != '~'
            && !is_separator(indexed[i + 1].1);
        if !valid_left || !valid_right {
            return Err(WrapError::InvalidMarker {
                index: indexed[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let run_start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                output.extend(process_segment(&normalized[start..run_start], options));
                output.extend(std::iter::repeat_n(String::new(), i - run_start - 1));
                start = i;
            }
        } else {
            i += 1;
        }
    }
    output.extend(process_segment(&normalized[start..], options));
    Ok(output)
}

fn main() {}
