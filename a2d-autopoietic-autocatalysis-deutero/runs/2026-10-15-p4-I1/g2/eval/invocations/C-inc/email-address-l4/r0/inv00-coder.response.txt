use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= input.len() || input[i] != b'\r' || input[i + 1] != b'\n' || !is_hws(input[i + 2]) {
        return None;
    }
    let mut j = i + 2;
    while j < input.len() && is_hws(input[j]) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut first_retained = true;
    let mut quoted_prefix = false;
    let mut quote_closed = false;

    while i < input.len() {
        if quoted_prefix && !quote_closed {
            let b = input[i];
            bytes.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' {
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                quote_closed = true;
            }
            continue;
        }

        let b = input[i];
        if first_retained && b == b'"' {
            quoted_prefix = true;
            quote_closed = false;
            first_retained = false;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        let run_start = i;
        let mut consumed = false;
        loop {
            if i >= input.len() {
                break;
            }
            let b = input[i];
            if is_hws(b) {
                i += 1;
                consumed = true;
            } else if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    consumed = true;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else if b == b'(' {
                consumed = true;
                let outer = i;
                let mut depth = 0usize;
                loop {
                    if i >= input.len() {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    let c = input[i];
                    if c == b'\\' {
                        if i + 1 >= input.len()
                            || !(0x20..=0x7e).contains(&input[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: i });
                        }
                        i += 2;
                    } else if c == b'(' {
                        depth += 1;
                        if depth > 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        i += 1;
                    } else if c == b')' {
                        depth -= 1;
                        i += 1;
                        if depth == 0 {
                            break;
                        }
                    } else if is_hws(c) {
                        i += 1;
                    } else if c == b'\r' || c == b'\n' {
                        if let Some(end) = fold_end(input, i) {
                            i = end;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    } else if (0x20..=0x7e).contains(&c) {
                        i += 1;
                    } else {
                        return Err(EmailError::CommentChar { at: i });
                    }
                }
            } else {
                break;
            }
        }
        if consumed {
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        bytes.push(b);
        positions.push(i);
        first_retained = false;
        i += 1;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn mapped(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_core(input: &str, allow_single_label: bool) -> Result<(Address, usize), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;

    for run in &cleaned.runs {
        let left_boundary = run.start == 0
            || bytes
                .get(run.start.wrapping_sub(1))
                .is_some_and(|&b| b == b'@');
        let right_boundary = run.end == raw.len()
            || bytes.get(run.start).is_some_and(|&b| b == b'@');
        if !left_boundary && !right_boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: mapped(&cleaned, 0, raw.len()),
            });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !(quote_end.is_some() && i <= protected_end) {
            separators.push(i);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: mapped(&cleaned, separators[1], raw.len()),
        });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: mapped(&cleaned, sep, raw.len()),
        });
    }

    let local_bytes = &bytes[..sep];
    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let quoted;

    if local_bytes[0] != b'"' {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: mapped(&cleaned, 64, raw.len()),
            });
        }
        let mut previous_dot = false;
        for (i, &b) in local_bytes.iter().enumerate() {
            let at = mapped(&cleaned, i, raw.len());
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot { at });
                }
                previous_dot = true;
            } else {
                if !is_atom_char(b) {
                    return Err(EmailError::LocalChar { at });
                }
                previous_dot = false;
            }
            local.push(b);
            local_origins.push(at);
        }
        quoted = false;
    } else {
        let closing = quote_end.expect("quoted prefix was checked");
        if closing + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: mapped(&cleaned, closing + 1, raw.len()),
            });
        }
        let mut i = 1;
        while i < closing {
            let b = local_bytes[i];
            let origin = mapped(&cleaned, i, raw.len());
            if b == b'\\' {
                if i + 1 >= closing {
                    return Err(EmailError::BadEscape { at: origin });
                }
                let escaped = local_bytes[i + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: origin });
                }
                local.push(escaped);
                local_origins.push(origin);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: origin });
                }
                local.push(b);
                local_origins.push(origin);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: mapped(&cleaned, closing, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
        quoted = true;
    }

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let mut domain_end = bytes.len();
    if raw_domain.last() == Some(&b'.') {
        domain_end -= 1;
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: mapped(&cleaned, domain_start, raw.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: mapped(&cleaned, domain_start + 253, raw.len()),
        });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    let mut labels = Vec::new();
    let mut label_start = 0;
    for i in 0..=domain_bytes.len() {
        if i == domain_bytes.len() || domain_bytes[i] == b'.' {
            let label = &domain_bytes[label_start..i];
            let absolute = domain_start + label_start;
            if label.is_empty() {
                let error_index = if i == domain_bytes.len() && domain_end < bytes.len() {
                    domain_end
                } else {
                    absolute
                };
                return Err(EmailError::EmptyLabel {
                    at: mapped(&cleaned, error_index, raw.len()),
                });
            }
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: mapped(&cleaned, absolute + 63, raw.len()),
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: mapped(&cleaned, absolute, raw.len()),
                });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: mapped(&cleaned, absolute + label.len() - 1, raw.len()),
                });
            }
            for (j, &b) in label.iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: mapped(&cleaned, absolute + j, raw.len()),
                    });
                }
            }
            labels.push(label);
            label_start = i + 1;
        }
    }
    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(&cleaned, domain_start, raw.len()),
        });
    }
    let last_label = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: mapped(&cleaned, domain_end - last_label.len(), raw.len()),
        });
    }

    let domain: Vec<u8> = domain_bytes.iter().map(u8::to_ascii_lowercase).collect();
    Ok((
        Address {
            local: String::from_utf8(local).expect("ASCII local"),
            domain: String::from_utf8(domain).expect("ASCII domain"),
            quoted,
        },
        local_origins.first().copied().unwrap_or(mapped(&cleaned, 0, raw.len())),
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, first_origin) = parse_core(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();

    if policy.local_case == LocalCase::Lower {
        for b in &mut local {
            *b = b.to_ascii_lowercase();
        }
    }

    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                let origin = if address.quoted {
                    let cleaned = cleanup(input.as_bytes()).expect("parse succeeded");
                    let mut found_at = None;
                    if !cleaned.bytes.is_empty() && cleaned.bytes[0] == b'"' {
                        let mut i = 1;
                        while i < cleaned.bytes.len() {
                            if cleaned.bytes[i] == b'\\' {
                                i += 2;
                            } else if cleaned.bytes[i] == b'"' {
                                found_at = Some(i);
                                break;
                            } else {
                                i += 1;
                            }
                        }
                    }
                    found_at
                        .map(|i| mapped(&cleaned, i + 1, input.len()))
                        .unwrap_or(first_origin)
                } else {
                    first_origin
                };
                return Err(EmailError::EmptyNormalizedLocal { at: origin });
            }
            local.truncate(plus);
        }
    }

    let safe = valid_dot_atom(&local);
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    address.local = String::from_utf8(local).expect("ASCII local");
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = seen.get(&key) {
                    duplicate_of[index] = Some(first);
                } else {
                    seen.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

fn main() {}
