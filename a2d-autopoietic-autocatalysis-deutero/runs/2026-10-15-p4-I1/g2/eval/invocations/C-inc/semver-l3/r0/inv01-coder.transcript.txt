            add_bound(arm, v.clone(), true, true);
            add_bound(arm, v.clone(), false, true);
            if !v.pre.is_empty() { arm.anchors.push(core(&v)); }
            return Ok(());
        }
    };
    if !v.pre.is_empty() { arm.anchors.push(core(&v)); }
    add_bound(arm, v, lower, inclusive);
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { bounds: Vec::new(), anchors: Vec::new() };
    if let Some((_, minus_start)) = tokens.iter().find(|(t, _)| t == "-") {
        if tokens.len() != 3 || tokens[1].0 != "-" {
            return Err(RangeError::Syntax { at: *minus_start });
        }
        let (lo, lo_start) = (&tokens[0].0, tokens[0].1);
        let (hi, hi_start) = (&tokens[2].0, tokens[2].1);
        let lower = parse_partial(lo, lo_start, lo_start)?;
        let upper = parse_partial(hi, hi_start, hi_start)?;
        if lower.all_wild { return Err(RangeError::Syntax { at: lo_start }); }
        if upper.all_wild { return Err(RangeError::Syntax { at: hi_start }); }
        if !lower.version.pre.is_empty() { arm.anchors.push(core(&lower.version)); }
        add_bound(&mut arm, lower.version, true, true);
        if upper.full {
            if !upper.version.pre.is_empty() { arm.anchors.push(core(&upper.version)); }
            add_bound(&mut arm, upper.version, false, true);
        } else {
            let (a,b,c) = if upper.k == 1 {
                bump(upper.version.major, 0, 0, 0)
            } else {
                bump(upper.version.major, upper.version.minor, 0, 1)
            }.ok_or(RangeError::Syntax { at: hi_start })?;
            add_bound(&mut arm, endpoint(a,b,c), false, false);
        }
    } else {
        for (tok, start) in tokens {
            let bytes = tok.as_bytes();
            if tok.starts_with('^') || tok.starts_with('~') {
                let kind = bytes[0] as char;
                let operand = &tok[1..];
                let p = parse_partial(operand, start + 1, start)?;
                expand_partial(&mut arm, p, kind, start, start + 1)?;
            } else if tok.starts_with('<') || tok.starts_with('>') || tok.starts_with('=') {
                parse_comparator(tok, *start, &mut arm)?;
            } else if !bytes.is_empty()
                && (bytes[0].is_ascii_digit() || matches!(bytes[0], b'*' | b'x' | b'X'))
            {
                let p = parse_partial(tok, *start, *start)?;
                expand_partial(&mut arm, p, 'b', *start, *start)?;
            } else {
                return Err(RangeError::Syntax { at: *start });
            }
        }
    }
    Ok(arm)
}

fn is_ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut pos = 0;
    let mut arm_tokens: Vec<(String, usize)> = Vec::new();
    let mut arm_start = None;
    while pos < bytes.len() {
        while pos < bytes.len() && is_ws(bytes[pos]) { pos += 1; }
        if pos >= bytes.len() { break; }
        let start = pos;
        while pos < bytes.len() && !is_ws(bytes[pos]) { pos += 1; }
        let tok = &s[start..pos];
        if tok == "||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(parse_arm(&arm_tokens)?);
            arm_tokens.clear();
            arm_start = Some(pos);
        } else {
            if arm_start.is_none() { arm_start = Some(start); }
            arm_tokens.push((tok.to_string(), start));
        }
    }
    if arm_tokens.is_empty() {
        if arms.is_empty() { return Err(RangeError::Empty { at: 0 }); }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&arm_tokens)?);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_pre: bool) -> bool {
    if !v.pre.is_empty() && !include_pre && !arm.anchors.contains(&core(v)) {
        return false;
    }
    arm.bounds.iter().all(|b| {
        let o = compare_versions(v, &b.version);
        if b.lower {
            o == Ordering::Greater || (b.inclusive && o == Ordering::Equal)
        } else {
            o == Ordering::Less || (b.inclusive && o == Ordering::Equal)
        }
    })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, text) in candidates.iter().enumerate() {
        let v = parse_version(text).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn arm_bounds(arm: &Arm) -> (Option<Bound>, Option<Bound>) {
    let mut lower: Option<Bound> = None;
    let mut upper: Option<Bound> = None;
    for b in &arm.bounds {
        let slot = if b.lower { &mut lower } else { &mut upper };
        let replace = match slot {
            None => true,
            Some(old) => {
                let o = compare_versions(&b.version, &old.version);
                if b.lower {
                    o == Ordering::Greater || (o == Ordering::Equal && !b.inclusive && old.inclusive)
                } else {
                    o == Ordering::Less || (o == Ordering::Equal && !b.inclusive && old.inclusive)
                }
            }
        };
        if replace { *slot = Some(b.clone()); }
    }
    (lower, upper)
}

fn version_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let (lo, hi) = arm_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let atext = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        let lower = lo.map(|b| format!("{}{}", if b.inclusive { ">=" } else { ">" }, version_text(&b.version))).unwrap_or_else(|| "*".to_string());
        let upper = hi.map(|b| format!("{}{}", if b.inclusive { "<=" } else { "<" }, version_text(&b.version))).unwrap_or_else(|| "*".to_string());
        out.push(format!("[{lower};{upper};pre={atext}]"));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".to_string() } else { out.join(" || ") }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut bounds = aa.bounds.clone();
            bounds.extend(bb.bounds.clone());
            let anchors: Vec<_> = if include_prerelease {
                Vec::new()
            } else {
                aa.anchors.iter().copied().filter(|x| bb.anchors.contains(x)).collect()
            };
            let arm = Arm { bounds, anchors };
            // Stable witnesses are checked through interval candidates and all boundary cores.
            if interval_has_stable(&arm) { return true; }
            if !include_prerelease && anchors.iter().any(|t| {
                tuple_has_pre_witness(*t, &arm)
            }) {
                return true;
            }
        }
    }
    false
}

fn interval_has_stable(arm: &Arm) -> bool {
    let (lo, hi) = arm_bounds(arm);
    // Search candidate stable points from lower bound and nearby core tuples.
    let mut candidates = vec![Version { major: 0, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new() }];
    if let Some(l) = lo {
        let mut v = l.version.clone();
        if !l.inclusive && v.pre.is_empty() {
            if let Some(t) = next_core(core(&v)) { v = Version { major:t.0, minor:t.1, patch:t.2, pre:Vec::new(), build:Vec::new() }; }
        } else if !l.inclusive && !v.pre.is_empty() {
            v.pre.clear();
        } else {
            v.pre.clear();
        }
        candidates.push(v);
    }
    if let Some(h) = hi {
        let mut v = h.version.clone();
        v.pre.clear();
        candidates.push(v);
    }
    for v in candidates {
        if arm.bounds.iter().all(|b| {
            let o=compare_versions(&v,&b.version);
            if b.lower { o==Ordering::Greater || (b.inclusive&&o==Ordering::Equal) }
            else { o==Ordering::Less || (b.inclusive&&o==Ordering::Equal) }
        }) { return true; }
    }
    false
}

fn next_core(t: (u64,u64,u64)) -> Option<(u64,u64,u64)> {
    if t.2 < u64::MAX { Some((t.0,t.1,t.2+1)) }
    else if t.1 < u64::MAX { Some((t.0,t.1+1,0)) }
    else if t.0 < u64::MAX { Some((t.0+1,0,0)) }
    else { None }
}

fn tuple_has_pre_witness(t: (u64,u64,u64), arm: &Arm) -> bool {
    let v = endpoint(t.0,t.1,t.2);
    arm.bounds.iter().all(|b| {
        let o=compare_versions(&v,&b.version);
        if b.lower { o==Ordering::Greater || (b.inclusive&&o==Ordering::Equal) }
        else { o==Ordering::Less || (b.inclusive&&o==Ordering::Equal) }
    })
}

fn parse_ids(s: &str, base: usize, numeric_pre: bool) -> Result<Vec<String>, VersionError> {

pub fn parse_version(s: &str) -> Result<Version, VersionError> {

            }

                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

    let mut build = Vec::new();

            .position(|&b| b == b'+')

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }

    a.len().cmp(&b.len())
}

        }

            add_bound(arm, p.version, false, true);
            return Ok(());
        }

        if p.k == 0 {
            return Ok(());
        }

        add_bound(arm, p.version.clone(), true, true);
        let field = if p.k == 1 { 0 } else { 1 };
        let (major, minor, patch) =
            bump(p.version.major, p.version.minor, p.version.patch, field)
                .ok_or(RangeError::Syntax { at: owner })?;
        add_bound(arm, endpoint(major, minor, patch), false, false);
        return Ok(());
    }

    if p.all_wild {
        return Err(RangeError::Syntax { at: owner });
    }

    let lower = p.version.clone();
    if !lower.pre.is_empty() {
        arm.anchors.push(core(&lower));
    }
    add_bound(arm, lower.clone(), true, true);

    let field = if kind == '^' {
        if lower.major > 0 || p.k == 1 {
            0
        } else if lower.minor > 0 || p.k == 2 {
            1
        } else {
            2
        }
    } else if p.k == 1 {
        0
    } else {
        1
    };

    let (major, minor, patch) = bump(lower.major, lower.minor, lower.patch, field)
        .ok_or(RangeError::Syntax { at: owner })?;
    add_bound(arm, endpoint(major, minor, patch), false, false);
    Ok(())
}

fn parse_comparator(tok: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (op_len, op) = if tok.starts_with(">=") {
        (2, ">=")
    } else if tok.starts_with("<=") {
        (2, "<=")
    } else if tok.starts_with('>') {
        (1, ">")
    } else if tok.starts_with('<') {
        (1, "<")
    } else if tok.starts_with('=') {
        (1, "=")
    } else {
        (0, "")
    };

    if op_len == 0 {
        return Err(RangeError::Syntax { at: start });
    }

    let operand = &tok[op_len..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: start });
    }

    let version =
        parse_version(operand).map_err(|error| RangeError::Version {
            at: start + op_len,
            error,
        })?;

    if !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }

    match op {
        ">" => add_bound(arm, version, true, false),
        ">=" => add_bound(arm, version, true, true),
        "<" => add_bound(arm, version, false, false),
        "<=" => add_bound(arm, version, false, true),
        _ => {
            add_bound(arm, version.clone(), true, true);
            add_bound(arm, version, false, true);
        }
    }

    Ok(())
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        bounds: Vec::new(),
        anchors: Vec::new(),
    };

    if let Some((_, minus_start)) = tokens.iter().find(|(token, _)| token == "-") {
        if tokens.len() != 3 || tokens[1].0 != "-" {
            return Err(RangeError::Syntax { at: *minus_start });
        }

        let (lower_text, lower_start) = (&tokens[0].0, tokens[0].1);
        let (upper_text, upper_start) = (&tokens[2].0, tokens[2].1);
        let lower = parse_partial(lower_text, lower_start)?;
        let upper = parse_partial(upper_text, upper_start)?;

        if lower.all_wild {
            return Err(RangeError::Syntax { at: lower_start });
        }
        if upper.all_wild {
            return Err(RangeError::Syntax { at: upper_start });
        }

        if !lower.version.pre.is_empty() {
            arm.anchors.push(core(&lower.version));
        }
        add_bound(&mut arm, lower.version, true, true);

        if upper.full {
            if !upper.version.pre.is_empty() {
                arm.anchors.push(core(&upper.version));
            }
            add_bound(&mut arm, upper.version, false, true);
        } else {
            let (major, minor, patch) = if upper.k == 1 {
                bump(upper.version.major, 0, 0, 0)
            } else {
                bump(upper.version.major, upper.version.minor, 0, 1)
            }
            .ok_or(RangeError::Syntax { at: upper_start })?;
            add_bound(&mut arm, endpoint(major, minor, patch), false, false);
        }

        return Ok(arm);
    }

    for (token, start) in tokens {
        let bytes = token.as_bytes();

        if token.starts_with('^') || token.starts_with('~') {
            let kind = bytes[0] as char;
            let operand = &token[1..];
            let partial = parse_partial(operand, *start + 1)?;
            expand_partial(&mut arm, partial, kind, *start)?;
        } else if token.starts_with('<') || token.starts_with('>') || token.starts_with('=') {
            parse_comparator(token, *start, &mut arm)?;
        } else if !bytes.is_empty()
            && (bytes[0].is_ascii_digit() || matches!(bytes[0], b'*' | b'x' | b'X'))
        {
            let partial = parse_partial(token, *start)?;
            expand_partial(&mut arm, partial, 'b', *start)?;
        } else {
            return Err(RangeError::Syntax { at: *start });
        }
    }

    Ok(arm)
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut tokens: Vec<(String, usize)> = Vec::new();
    let mut pos = 0usize;

    while pos < bytes.len() {
        while pos < bytes.len() && is_ws(bytes[pos]) {
            pos += 1;
        }
        if pos >= bytes.len() {
            break;
        }

        let start = pos;
        while pos < bytes.len() && !is_ws(bytes[pos]) {
            pos += 1;
        }

        let token = &s[start..pos];
        if token == "||" {
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(parse_arm(&tokens)?);
            tokens.clear();
        } else {
            tokens.push((token.to_string(), start));
        }
    }

    if tokens.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }

    arms.push(parse_arm(&tokens)?);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty() && !include_prerelease && !arm.anchors.contains(&core(v)) {
        return false;
    }

    arm.bounds.iter().all(|bound| {
        let order = compare_versions(v, &bound.version);
        if bound.lower {
            order == Ordering::Greater || (bound.inclusive && order == Ordering::Equal)
        } else {
            order == Ordering::Less || (bound.inclusive && order == Ordering::Equal)
        }
    })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());

    for (index, text) in candidates.iter().enumerate() {
        let version = parse_version(text)
            .map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) {
            continue;
        }

        if let Some(j) = best {
            let order = compare_versions(&parsed[i], &parsed[j]);
            if (maximum && order == Ordering::Greater)
                || (!maximum && order == Ordering::Less)
            {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }

    Ok(best.map(|index| candidates[index].clone()))
}

fn arm_bounds(arm: &Arm) -> (Option<Bound>, Option<Bound>) {
    let mut lower: Option<Bound> = None;
    let mut upper: Option<Bound> = None;

    for bound in &arm.bounds {
        let slot = if bound.lower { &mut lower } else { &mut upper };
        let replace = match slot {
            None => true,
            Some(old) => {
                let order = compare_versions(&bound.version, &old.version);
                if bound.lower {
                    order == Ordering::Greater
                        || (order == Ordering::Equal && !bound.inclusive && old.inclusive)
                } else {
                    order == Ordering::Less
                        || (order == Ordering::Equal && !bound.inclusive && old.inclusive)
                }
            }
        };

        if replace {
            *slot = Some(bound.clone());
        }
    }

    (lower, upper)
}

fn version_text(version: &Version) -> String {
    let mut text = format!("{}.{}.{}", version.major, version.minor, version.patch);
    if !version.pre.is_empty() {
        text.push('-');
        text.push_str(&version.pre.join("."));
    }
    text
}

pub fn canonical(r: &Range) -> String {
    let mut output = Vec::new();

    for arm in &r.arms {
        let (lower, upper) = arm_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();

        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(major, minor, patch)| format!("{major}.{minor}.{patch}"))
                .collect::<Vec<_>>()
                .join(",")
        };

        let lower_text = lower
            .map(|bound| {
                format!(
                    "{}{}",
                    if bound.inclusive { ">=" } else { ">" },
                    version_text(&bound.version)
                )
            })
            .unwrap_or_else(|| "*".to_string());

        let upper_text = upper
            .map(|bound| {
                format!(
                    "{}{}",
                    if bound.inclusive { "<=" } else { "<" },
                    version_text(&bound.version)
                )
            })
            .unwrap_or_else(|| "*".to_string());

        output.push(format!(
            "[{lower_text};{upper_text};pre={anchor_text}]"
        ));
    }

    output.sort();
    output.dedup();

    if output.is_empty() {
        "EMPTY".to_string()
    } else {
        output.join(" || ")
    }
}

fn next_core(tuple: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if tuple.2 < u64::MAX {
        Some((tuple.0, tuple.1, tuple.2 + 1))
    } else if tuple.1 < u64::MAX {
        Some((tuple.0, tuple.1 + 1, 0))
    } else if tuple.0 < u64::MAX {
        Some((tuple.0 + 1, 0, 0))
    } else {
        None
    }
}

fn interval_has_stable(arm: &Arm) -> bool {
    let (lower, upper) = arm_bounds(arm);
    let mut candidates = vec![Version {
        major: 0,
        minor: 0,
        patch: 0,
        pre: Vec::new(),
        build: Vec::new(),
    }];

    if let Some(bound) = lower {
        let mut version = bound.version;
        if !bound.inclusive && version.pre.is_empty() {
            if let Some(tuple) = next_core(core(&version)) {
                version = Version {
                    major: tuple.0,
                    minor: tuple.1,
                    patch: tuple.2,
                    pre: Vec::new(),
                    build: Vec::new(),
                };
            }
        } else {
            version.pre.clear();
        }
        candidates.push(version);
    }

    if let Some(bound) = upper {
        let mut version = bound.version;
        version.pre.clear();
        candidates.push(version);
    }

    candidates.into_iter().any(|version| {
        arm.bounds.iter().all(|bound| {
            let order = compare_versions(&version, &bound.version);
            if bound.lower {
                order == Ordering::Greater || (bound.inclusive && order == Ordering::Equal)
            } else {
                order == Ordering::Less || (bound.inclusive && order == Ordering::Equal)
            }
        })
    })
}

fn tuple_has_pre_witness(tuple: (u64, u64, u64), arm: &Arm) -> bool {
    let version = endpoint(tuple.0, tuple.1, tuple.2);
    arm.bounds.iter().all(|bound| {
        let order = compare_versions(&version, &bound.version);
        if bound.lower {
            order == Ordering::Greater || (bound.inclusive && order == Ordering::Equal)
        } else {
            order == Ordering::Less || (bound.inclusive && order == Ordering::Equal)
        }
    })
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for arm_a in &a.arms {
        for arm_b in &b.arms {
            let mut bounds = arm_a.bounds.clone();
            bounds.extend(arm_b.bounds.clone());

            if interval_has_stable(&Arm {
                bounds: bounds.clone(),
                anchors: Vec::new(),
            }) {
                return true;
            }

            if !include_prerelease {
                for tuple in &arm_a.anchors {
                    if arm_b.anchors.contains(tuple)
                        && tuple_has_pre_witness(*tuple, &Arm {
                            bounds: bounds.clone(),
                            anchors: Vec::new(),
                        })
                    {
                        return true;
                    }
                }
            } else if interval_has_stable(&Arm {
                bounds,
                anchors: Vec::new(),
            }) {
                return true;
            }
        }
    }

    false
}
