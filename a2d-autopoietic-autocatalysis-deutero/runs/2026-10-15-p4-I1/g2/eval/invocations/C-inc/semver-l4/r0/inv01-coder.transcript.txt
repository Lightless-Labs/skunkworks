    start: usize,
    end: usize,
    check_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();

    let mut ident_start = start;

    loop {

            .iter()
            .position(|&b| b == b'.')
            .map(|i| ident_start + i)
            .unwrap_or(end);

        if ident_start == ident_end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

            if !is_ident_char(bytes[i]) {

}

fn zero_pre() -> Vec<String> {
    vec!["0".to_string()]
}

fn parse_full_operand(
    text: &str,
    offset: usize,
) -> Result<Version, RangeError> {
    parse_version(text).map_err(|error| RangeError::Version { at: offset, error })
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<u64>,
    full: Option<Version>,
    explicit_numeric: usize,
}

fn partial_operand(text: &str, offset: usize) -> Result<Partial, RangeError> {
    let bytes = text.as_bytes();
    let full_eligible = text.contains('-')
        || text.contains('+')
        || (text.split('.').count() >= 3
            && !text.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if full_eligible {
        let v = parse_full_operand(text, offset)?;
        return Ok(Partial {
            fields: vec![v.major, v.minor, v.patch],
            explicit_numeric: 3,
            full: Some(v),
        });
    }

    let parts: Vec<&str> = text.split('.').collect();
    if parts.is_empty() || parts.len() > 3 {
        return Err(RangeError::Syntax { at: offset });
    }

    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for part in parts {
        if matches!(part, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen || part.is_empty() || !part.bytes().all(is_digit) {
            return Err(RangeError::Syntax { at: offset });
        }
        if part.len() > 1 && part.starts_with('0') {
            return Err(RangeError::Syntax { at: offset });
        }
        let value = part
            .parse::<u64>()
            .map_err(|_| RangeError::Syntax { at: offset })?;
        fields.push(value);
    }

    Ok(Partial {
        explicit_numeric: fields.len(),
        fields,
        full: None,
    })
}

fn partial_core(p: &Partial) -> (u64, u64, u64) {
    (
        p.fields.get(0).copied().unwrap_or(0),
        p.fields.get(1).copied().unwrap_or(0),
        p.fields.get(2).copied().unwrap_or(0),
    )
}

fn increment_upper(
    core: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut c = [core.0, core.1, core.2];
    c[field] = c[field]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at })?;
    for value in c.iter_mut().skip(field + 1) {
        *value = 0;
    }
    Ok(version_with_core((c[0], c[1], c[2]), zero_pre()))
}

fn add_comparator(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(vtuple(&version));
    }
    arm.comparators.push(Comparator { op, version });
}

fn expand_bare(
    arm: &mut Arm,
    token: Token<'_>,
) -> Result<(), RangeError> {
    let p = partial_operand(token.text, token.start)?;
    if let Some(v) = p.full {
        add_comparator(arm, Op::Eq, v, true);
        return Ok(());
    }

    let k = p.explicit_numeric;
    if k == 0 {
        return Ok(());
    }
    let lower = version_with_core(partial_core(&p), Vec::new());
    add_comparator(arm, Op::Ge, lower, false);
    let upper = match k {
        1 => Some(increment_upper(partial_core(&p), 0, token.start)?),
        2 => Some(increment_upper(partial_core(&p), 1, token.start)?),
        _ => None,
    };
    if let Some(upper) = upper {
        add_comparator(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn expand_caret_tilde(
    arm: &mut Arm,
    token: Token<'_>,
    caret: bool,
) -> Result<(), RangeError> {
    let operand_start = token.start + 1;
    let operand = &token.text[1..];
    let p = partial_operand(operand, operand_start)?;
    let k = p.explicit_numeric;
    if k == 0 {
        return Err(RangeError::Syntax { at: token.start });
    }

    let lower = p
        .full
        .clone()
        .unwrap_or_else(|| version_with_core(partial_core(&p), Vec::new()));
    add_comparator(arm, Op::Ge, lower.clone(), true);

    let core = partial_core(&p);
    let field = if caret {
        if core.0 > 0 || k == 1 {
            0
        } else if core.1 > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if k == 1 {
        0
    } else {
        1
    };
    let upper = increment_upper(core, field, operand_start)?;
    add_comparator(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token<'_>], dash_index: usize) -> Result<Arm, RangeError> {
    if dash_index != 1 || tokens.len() != 3 {
        return Err(RangeError::Syntax {
            at: tokens[dash_index].start,
        });
    }

    let lower = partial_operand(tokens[0].text, tokens[0].start)?;
    let upper = partial_operand(tokens[2].text, tokens[2].start)?;
    if lower.explicit_numeric == 0 {
        return Err(RangeError::Syntax {
            at: tokens[0].start,
        });
    }
    if upper.explicit_numeric == 0 {
        return Err(RangeError::Syntax {
            at: tokens[2].start,
        });
    }

    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    let lower_version = lower.full.clone().unwrap_or_else(|| {
        version_with_core(partial_core(&lower), Vec::new())
    });
    add_comparator(&mut arm, Op::Ge, lower_version, true);

    let upper_bound = if let Some(v) = upper.full.clone() {
        v
    } else {
        let core = partial_core(&upper);
        match upper.explicit_numeric {
            1 => increment_upper(core, 0, tokens[2].start)?,
            2 => increment_upper(core, 1, tokens[2].start)?,
            _ => version_with_core(core, Vec::new()),
        }
    };
    let upper_op = if upper.full.is_some() || upper.explicit_numeric == 3 {
        Op::Le
    } else {
        Op::Lt
    };
    add_comparator(&mut arm, upper_op, upper_bound, upper.full.is_some());
    Ok(arm)
}

fn starts_with_ascii_digit(s: &str) -> bool {
    s.as_bytes().first().copied().map(is_digit).unwrap_or(false)
}

fn parse_comparator(token: Token<'_>) -> Result<Comparator, RangeError> {
    let (op, operand_len) = if token.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.text.starts_with("<=") {
        (Op::Le, 2)
    } else if token.text.starts_with('>') {
        (Op::Gt, 1)
    } else if token.text.starts_with('<') {
        (Op::Lt, 1)
    } else if token.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &token.text[operand_len..];
    let operand_start = token.start + operand_len;
    if !starts_with_ascii_digit(operand) {
        return Err(RangeError::Syntax { at: token.start });
    }
    let version = parse_full_operand(operand, operand_start)?;
    Ok(Comparator { op, version })
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let dash = tokens.iter().position(|t| t.text == "-");
    if let Some(index) = dash {
        return parse_hyphen_arm(tokens, index);
    }

    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    for token in tokens {
        let first = token.text.as_bytes().first().copied();
        match first {
            Some(b'^') => expand_caret_tilde(&mut arm, *token, true)?,
            Some(b'~') => expand_caret_tilde(&mut arm, *token, false)?,
            Some(b'<') | Some(b'>') | Some(b'=') => {
                let comparator = parse_comparator(*token)?;
                add_comparator(
                    &mut arm,
                    comparator.op,
                    comparator.version,
                    true,
                );
            }
            Some(b) if is_digit(b) || matches!(b, b'*' | b'x' | b'X') => {
                expand_bare(&mut arm, *token)?;
            }
            _ => return Err(RangeError::Syntax { at: token.start }),
        }
    }
    Ok(arm)
}

fn split_arms<'a>(s: &'a str, tokens: &[Token<'a>]) -> Result<Vec<Vec<Token<'a>>>, RangeError> {
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms: Vec<Vec<Token<'a>>> = vec![Vec::new()];
    let mut previous_separator: Option<usize> = None;
    for token in tokens {
        if token.text == "||" {
            if arms.last().unwrap().is_empty() {
                return Err(RangeError::Empty { at: token.start });
            }
            arms.push(Vec::new());
            previous_separator = Some(token.start);
        } else {
            arms.last_mut().unwrap().push(*token);
            previous_separator = None;
        }
    }
    if arms.last().unwrap().is_empty() {
        return Err(RangeError::Empty {
            at: previous_separator.map(|_| s.len()).unwrap_or(s.len()),
        });
    }
    Ok(arms)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize_range(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let arms_tokens = split_arms(s, &tokens)?;
    let mut arms = Vec::new();
    for tokens in arms_tokens {
        arms.push(parse_arm(&tokens)?);
    }
    Ok(Range { arms })
}

fn comparator_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_admits(v: &Version, arm: &Arm) -> bool {
    arm.anchors.iter().any(|&t| t == vtuple(v))
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if arm.comparators.iter().any(|c| !comparator_matches(v, c)) {
        return false;
    }
    v.pre.is_empty() || include_prerelease || arm_admits(v, arm)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| {
            parse_version(s).map_err(|error| CandidateError { index, error })
        })
        .collect::<Result<_, _>>()?;

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        if let Some(current) = best {
            let ord = compare_versions(version, &parsed[current]);
            if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                best = Some(index);
            }
        } else {
            best = Some(index);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn core_string(core: (u64, u64, u64)) -> String {
    format!("{}.{}.{}", core.0, core.1, core.2)
}

fn version_string(v: &Version) -> String {
    let mut out = core_string(vtuple(v));
    if !v.pre.is_empty() {
        out.push('-');
        out.push_str(&v.pre.join("."));
    }
    out
}

fn compare_bound(a: &Comparator, b: &Comparator) -> Ordering {
    compare_versions(&a.version, &b.version)
}

fn has_stable_witness(arm: &Arm) -> bool {
    let mut lower: Option<&Comparator> = None;
    let mut upper: Option<&Comparator> = None;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                let v = &c.version;
                return arm.comparators.iter().all(|x| comparator_matches(v, x));
            }
            Op::Gt | Op::Ge => {
                if lower.map_or(true, |old| {
                    compare_bound(c, old) == Ordering::Greater
                        || (compare_bound(c, old) == Ordering::Equal
                            && matches!(c.op, Op::Gt))
                }) {
                    lower = Some(c);
                }
            }
            Op::Lt | Op::Le => {
                if upper.map_or(true, |old| {
                    compare_bound(c, old) == Ordering::Less
                        || (compare_bound(c, old) == Ordering::Equal
                            && matches!(c.op, Op::Lt))
                }) {
                    upper = Some(c);
                }
            }
        }
    }

    let lower_core = lower.map(|c| &c.version);
    let upper_core = upper.map(|c| &c.version);
    if let (Some(l), Some(u)) = (lower_core, upper_core) {
        if cmp_core(l, u) == Ordering::Greater {
            return false;
        }
        if cmp_core(l, u) == Ordering::Equal {
            let stable = version_with_core(vtuple(l), Vec::new());
            return arm.comparators.iter().all(|c| comparator_matches(&stable, c));
        }
    }

    if let Some(u) = upper_core {
        if u.pre.is_empty() && matches!(upper.unwrap().op, Op::Lt) {
            let core = vtuple(u);
            return predecessor_core(core).is_some()
                || lower_core.map_or(true, |l| cmp_core(l, u) == Ordering::Less);
        }
    }
    true
}

fn predecessor_core(core: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if core.2 > 0 {
        Some((core.0, core.1, core.2 - 1))
    } else if core.1 > 0 {
        Some((core.0, core.1 - 1, u64::MAX))
    } else if core.0 > 0 {
        Some((core.0 - 1, u64::MAX, u64::MAX))
    } else {
        None
    }
}

fn bounds(arm: &Arm) -> (Option<Comparator>, Option<Comparator>) {
    let mut lower: Option<Comparator> = None;
    let mut upper: Option<Comparator> = None;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                let low = Comparator {
                    op: Op::Ge,
                    version: c.version.clone(),
                };
                let high = Comparator {
                    op: Op::Le,
                    version: c.version.clone(),
                };
                if lower.is_none() {
                    lower = Some(low);
                }
                if upper.is_none() {
                    upper = Some(high);
                }
            }
            Op::Gt | Op::Ge => {
                if lower.as_ref().map_or(true, |old| {
                    let ord = compare_versions(&c.version, &old.version);
                    ord == Ordering::Greater
                        || (ord == Ordering::Equal && matches!(c.op, Op::Gt) && matches!(old.op, Op::Ge))
                }) {
                    lower = Some(c.clone());
                }
            }
            Op::Lt | Op::Le => {
                if upper.as_ref().map_or(true, |old| {
                    let ord = compare_versions(&c.version, &old.version);
                    ord == Ordering::Less
                        || (ord == Ordering::Equal && matches!(c.op, Op::Lt) && matches!(old.op, Op::Le))
                }) {
                    upper = Some(c.clone());
                }
            }
        }
    }
    (lower, upper)
}

fn arm_has_any_version(arm: &Arm) -> bool {
    if !has_stable_witness(arm) {
        return false;
    }
    true
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        if !arm_has_any_version(arm) {
            continue;
        }
        let (lower, upper) = bounds(arm);
        let lower_text = lower
            .as_ref()
            .map(|c| {
                let op = if matches!(c.op, Op::Gt) { ">" } else { ">=" };
                format!("{op}{}", version_string(&c.version))
            })
            .unwrap_or_else(|| "*".to_string());
        let upper_text = upper
            .as_ref()
            .map(|c| {
                let op = if matches!(c.op, Op::Lt) { "<" } else { "<=" };
                format!("{op}{}", version_string(&c.version))
            })
            .unwrap_or_else(|| "*".to_string());
        let mut anchors = arm.anchors.clone();
        anchors.sort_unstable();
        anchors.dedup();
        let anchors_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors.iter().map(|&t| core_string(t)).collect::<Vec<_>>().join(",")
        };
        strings.push(format!(
            "[{};{};pre={}]",
            lower_text, upper_text, anchors_text
        ));
    }
    strings.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".to_string()
    } else {
        strings.join(" || ")
    }
}

fn witness_at_core(core: (u64, u64, u64), arms: &[&Arm], include_pre: bool) -> bool {
    let stable = version_with_core(core, Vec::new());
    if arms.iter().all(|a| arm_matches(&stable, a, include_pre)) {
        return true;
    }
    if include_pre {
        return false;
    }
    if !arms.iter().all(|a| arm_admits(&stable, a)) {
        return false;
    }
    false
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            if aa.anchors.iter().any(|t| bb.anchors.contains(t)) {
                let core = aa.anchors.iter().copied().find(|t| bb.anchors.contains(t)).unwrap();
                let pre = version_with_core(core, vec!["0".to_string()]);
                if arm_matches(&pre, aa, include_prerelease)
                    && arm_matches(&pre, bb, include_prerelease)
                {
                    return true;
                }
            }

            if !include_prerelease && (aa.anchors.is_empty() || bb.anchors.is_empty()) {
                // Anchors are only needed for prerelease witnesses; stable witnesses remain valid.
            }

            let (la, ua) = bounds(aa);
            let (lb, ub) = bounds(bb);
            let mut cores = Vec::new();
            for c in [la.as_ref(), lb.as_ref(), ua.as_ref(), ub.as_ref()].into_iter().flatten() {
                cores.push(vtuple(&c.version));
            }
            cores.push((0, 0, 0));
            cores.push((u64::MAX, u64::MAX, u64::MAX));
            for core in cores {
                if witness_at_core(core, &[aa, bb], include_prerelease) {
                    return true;
                }
                if let Some(next) = successor_core(core) {
                    if witness_at_core(next, &[aa, bb], include_prerelease) {
                        return true;
                    }
                }
            }
        }
    }
    false
}

fn successor_core(core: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if core.2 < u64::MAX {
        Some((core.0, core.1, core.2 + 1))
    } else if core.1 < u64::MAX {
        Some((core.0, core.1 + 1, 0))
    } else if core.0 < u64::MAX {
        Some((core.0 + 1, 0, 0))
    } else {
        None
    }
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(expr: &SetExpr, ranges: &[Option<Range>], leaf_index: &mut usize, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let result = ranges[*leaf_index]
                .as_ref()
                .map(|r| matches_with(v, r, include))
                .unwrap_or(false);
            *leaf_index += 1;
            result
        }
        SetExpr::Union(children) => children
            .iter()
            .any(|child| eval_expr(child, ranges, leaf_index, v, include)),
        SetExpr::Intersection(children) => children
            .iter()
            .all(|child| eval_expr(child, ranges, leaf_index, v, include)),
    }
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    ranges: &mut Vec<Option<Range>>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let parsed = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            ranges.push(Some(parsed));
            Ok(())
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, ranges)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn constraint_accepts(
    expr: &SetExpr,
    ranges: &[Option<Range>],
    v: &Version,
    include: bool,
) -> bool {
    let mut index = 0;
    eval_expr(expr, ranges, &mut index, v, include)
}

fn minimum_conflict(
    package: &Package,
    parsed: &[Version],
    ranges_by_constraint: &[Vec<Option<Range>>],
    include: bool,
) -> Vec<String> {
    if parsed.is_empty() {
        return Vec::new();
    }

    let n = package.constraints.len();
    for size in 1..=n {
        let mut chosen = Vec::new();
        let mut best: Option<Vec<String>> = None;
        fn search(
            start: usize,
            target: usize,
            chosen: &mut Vec<usize>,
            best: &mut Option<Vec<String>>,
            package: &Package,
            parsed: &[Version],
            ranges: &[Vec<Option<Range>>],
            include: bool,
        ) {
            if chosen.len() == target {
                let satisfiable = parsed.iter().any(|v| {
                    chosen.iter().all(|&i| {
                        constraint_accepts(
                            &package.constraints[i].expr,
                            &ranges[i],
                            v,
                            include,
                        )
                    })
                });
                if !satisfiable {
                    let mut labels: Vec<String> = chosen
                        .iter()
                        .map(|&i| package.constraints[i].label.clone())
                        .collect();
                    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                    if best.as_ref().map_or(true, |old| {
                        labels.as_slice().cmp(old.as_slice()) == Ordering::Less
                    }) {
                        *best = Some(labels);
                    }
                }
                return;
            }
            let needed = target - chosen.len();
            for i in start..=package.constraints.len().saturating_sub(needed) {
                chosen.push(i);
                search(i + 1, target, chosen, best, package, parsed, ranges, include);
                chosen.pop();
            }
        }
        search(
            0,
            size,
            &mut chosen,
            &mut best,
            package,
            parsed,
            ranges_by_constraint,
            include,
        );
        if let Some(labels) = best {
            return labels;
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut package_names = std::collections::HashSet::new();
    for package in packages {
        if !package_names.insert(package.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: package.name.clone(),
            });
        }
    }

    for package in packages {
        let mut labels = std::collections::HashSet::new();
        for constraint in &package.constraints {
            if !labels.insert(constraint.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: package.name.clone(),
                    label: constraint.label.clone(),
                });
            }
        }
    }

    let mut parsed_candidates: Vec<Vec<Version>> = Vec::new();
    for package in packages {
        let mut parsed = Vec::new();
        for (index, text) in package.candidates.iter().enumerate() {
            parsed.push(
                parse_version(text).map_err(|error| SelectError::InvalidCandidate {
                    package: package.name.clone(),
                    index,
                    error,
                })?,
            );
        }
        parsed_candidates.push(parsed);
    }

    let mut ranges_per_package: Vec<Vec<Vec<Option<Range>>>> = Vec::new();
    for package in packages {
        let mut constraints_ranges = Vec::new();
        for constraint in &package.constraints {
            let mut ranges = Vec::new();
            validate_expr(
                &constraint.expr,
                &package.name,
                &constraint.label,
                &mut Vec::new(),
                &mut ranges,
            )?;
            constraints_ranges.push(ranges);
        }
        ranges_per_package.push(constraints_ranges);
    }

    let mut selections = Vec::new();
    for (package_index, package) in packages.iter().enumerate() {
        let parsed = &parsed_candidates[package_index];
        let range_sets = &ranges_per_package[package_index];
        let mut qualifying: Vec<usize> = parsed
            .iter()
            .enumerate()
            .filter_map(|(i, version)| {
                let ok = package.constraints.iter().enumerate().all(|(ci, constraint)| {
                    constraint_accepts(
                        &constraint.expr,
                        &range_sets[ci],
                        version,
                        include_prerelease,
                    )
                });
                if ok { Some(i) } else { None }
            })
            .collect();

        if qualifying.is_empty() {
            let labels = minimum_conflict(
                package,
                parsed,
                range_sets,
                include_prerelease,
            );
            return Err(SelectError::NoSolution {
                package: package.name.clone(),
                labels,
            });
        }

        let stable_first = matches!(preference, Preference::StableFirst)
            && qualifying.iter().any(|&i| parsed[i].pre.is_empty());
        qualifying.retain(|&i| {
            !stable_first || parsed[i].pre.is_empty()
        });

        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let ord = compare_versions(&parsed[i], &parsed[best]);
            let improve = match preference {
                Preference::Highest | Preference::StableFirst => ord == Ordering::Greater,
                Preference::Lowest => ord == Ordering::Less,
            };
            if improve {
                best = i;
            }
        }
        selections.push(Selection {
            package: package.name.clone(),
            version: package.candidates[best].clone(),
        });
    }
    Ok(selections)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    check_numeric_leading_zero: bool,

    loop {

            .iter()
            .position(|&b| b == b'.')
            .map(|i| ident_start + i)
            .unwrap_or(end);

            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}

fn partial_operand(text: &str, offset: usize) -> Result<Partial, RangeError> {
    let full_eligible = text.contains('-')
        || text.contains('+')
        || (text.split('.').count() >= 3
            && !text.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if full_eligible {
        let v = parse_full_operand(text, offset)?;
        return Ok(Partial {
            fields: vec![v.major, v.minor, v.patch],
            explicit_numeric: 3,
            full: Some(v),
        });
    }

    let parts: Vec<&str> = text.split('.').collect();
    if parts.is_empty() || parts.len() > 3 {
        return Err(RangeError::Syntax { at: offset });
    }

    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for part in parts {
        if matches!(part, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen || part.is_empty() || !part.bytes().all(is_digit) {
            return Err(RangeError::Syntax { at: offset });
        }

            return Err(RangeError::Syntax { at: offset });
        }
        let value = part
            .parse::<u64>()
            .map_err(|_| RangeError::Syntax { at: offset })?;
        fields.push(value);
    }

    Ok(Partial {
        explicit_numeric: fields.len(),
        fields,
        full: None,
    })
}

fn parse_comparator(token: Token<'_>) -> Result<Comparator, RangeError> {
    let (op, operand_len) = if token.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.text.starts_with("<=") {
        (Op::Le, 2)
    } else if token.text.starts_with('>') {
        (Op::Gt, 1)
    } else if token.text.starts_with('<') {
        (Op::Lt, 1)
    } else if token.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &token.text[operand_len..];
    let operand_start = token.start + operand_len;
    if !starts_with_ascii_digit(operand) {
        return Err(RangeError::Syntax { at: token.start });
    }
    let version = parse_full_operand(operand, operand_start)?;
    Ok(Comparator { op, version })
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let dash = tokens.iter().position(|t| t.text == "-");
    if let Some(index) = dash {
        return parse_hyphen_arm(tokens, index);
    }

    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    for token in tokens {
        let first = token.text.as_bytes().first().copied();
        match first {
            Some(b'^') => expand_caret_tilde(&mut arm, *token, true)?,
            Some(b'~') => expand_caret_tilde(&mut arm, *token, false)?,
            Some(b'<') | Some(b'>') | Some(b'=') => {
                let comparator = parse_comparator(*token)?;
                add_comparator(&mut arm, comparator.op, comparator.version, true);
            }
            Some(b) if is_digit(b) || matches!(b, b'*' | b'x' | b'X') => {
                expand_bare(&mut arm, *token)?;
            }
            _ => return Err(RangeError::Syntax { at: token.start }),
        }
    }
    Ok(arm)
}

fn split_arms<'a>(s: &'a str, tokens: &[Token<'a>]) -> Result<Vec<Vec<Token<'a>>>, RangeError> {
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms: Vec<Vec<Token<'a>>> = vec![Vec::new()];
    let mut previous_separator: Option<usize> = None;
    for token in tokens {
        if token.text == "||" {
            if arms.last().unwrap().is_empty() {
                return Err(RangeError::Empty { at: token.start });
            }
            arms.push(Vec::new());
            previous_separator = Some(token.start);
        } else {
            arms.last_mut().unwrap().push(*token);
            previous_separator = None;
        }
    }
    if arms.last().unwrap().is_empty() {
        return Err(RangeError::Empty {
            at: previous_separator.map(|_| s.len()).unwrap_or(s.len()),
        });
    }
    Ok(arms)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize_range(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let arms_tokens = split_arms(s, &tokens)?;
    let mut arms = Vec::new();
    for tokens in arms_tokens {
        arms.push(parse_arm(&tokens)?);
    }
    Ok(Range { arms })
}

fn comparator_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

    matches_with(v, r, false)

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| {
            parse_version(s).map_err(|error| CandidateError { index, error })
        })
        .collect::<Result<_, _>>()?;

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        if let Some(current) = best {
            let ord = compare_versions(version, &parsed[current]);
            if (want_max && ord == Ordering::Greater)
                || (!want_max && ord == Ordering::Less)
            {
                best = Some(index);
            }
        } else {
            best = Some(index);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,

    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn core_string(core: (u64, u64, u64)) -> String {
    format!("{}.{}.{}", core.0, core.1, core.2)
}

fn version_string(v: &Version) -> String {
    let mut out = core_string(vtuple(v));
    if !v.pre.is_empty() {
        out.push('-');
        out.push_str(&v.pre.join("."));
    }
    out
}

fn compare_bound(a: &Comparator, b: &Comparator) -> Ordering {
    compare_versions(&a.version, &b.version)
}

fn predecessor_core(core: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if core.2 > 0 {
        Some((core.0, core.1, core.2 - 1))
    } else if core.1 > 0 {
        Some((core.0, core.1 - 1, u64::MAX))
    } else if core.0 > 0 {
        Some((core.0 - 1, u64::MAX, u64::MAX))
    } else {
        None
    }
}

fn has_stable_witness(arm: &Arm) -> bool {
    let (lower, upper) = bounds(arm);
    let stable = |core| version_with_core(core, Vec::new());

    if let (Some(l), Some(u)) = (&lower, &upper) {
        let ord = compare_versions(&l.version, &u.version);
        if ord == Ordering::Greater {
            return false;
        }
        if ord == Ordering::Equal {
            return matches!(l.op, Op::Ge | Op::Le)
                && matches!(u.op, Op::Ge | Op::Le)
                && arm
                    .comparators
                    .iter()
                    .all(|c| comparator_matches(&stable(vtuple(&l.version)), c));
        }
    }

    if let Some(u) = &upper {
        if u.version.pre.is_empty() && matches!(u.op, Op::Lt) {
            let core = vtuple(&u.version);
            let Some(prev) = predecessor_core(core) else {
                return false;
            };
            let candidate = stable(prev);
            return arm
                .comparators
                .iter()
                .all(|c| comparator_matches(&candidate, c));
        }
    }

    // If the interval has room at any core, its stable release is a witness.
    if let Some(l) = &lower {
        let core = vtuple(&l.version);
        let candidate = stable(core);
        if arm
            .comparators
            .iter()
            .all(|c| comparator_matches(&candidate, c))
        {
            return true;
        }
        if let Some(next) = successor_core(core) {
            let candidate = stable(next);
            if arm
                .comparators
                .iter()
                .all(|c| comparator_matches(&candidate, c))
            {
                return true;
            }
        }
    } else {
        let candidate = stable((0, 0, 0));
        if arm
            .comparators
            .iter()
            .all(|c| comparator_matches(&candidate, c))
        {
            return true;
        }
    }
    false
}

fn bounds(arm: &Arm) -> (Option<Comparator>, Option<Comparator>) {
    let mut lower: Option<Comparator> = None;
    let mut upper: Option<Comparator> = None;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                let low = Comparator {
                    op: Op::Ge,
                    version: c.version.clone(),
                };
                let high = Comparator {
                    op: Op::Le,
                    version: c.version.clone(),
                };
                if lower.is_none() {
                    lower = Some(low);
                }
                if upper.is_none() {
                    upper = Some(high);
                }
            }
            Op::Gt | Op::Ge => {
                if lower.as_ref().map_or(true, |old| {
                    let ord = compare_versions(&c.version, &old.version);
                    ord == Ordering::Greater
                        || (ord == Ordering::Equal
                            && matches!(c.op, Op::Gt)
                            && matches!(old.op, Op::Ge))
                }) {
                    lower = Some(c.clone());
                }
            }
            Op::Lt | Op::Le => {
                if upper.as_ref().map_or(true, |old| {
                    let ord = compare_versions(&c.version, &old.version);
                    ord == Ordering::Less
                        || (ord == Ordering::Equal
                            && matches!(c.op, Op::Lt)
                            && matches!(old.op, Op::Le))
                }) {
                    upper = Some(c.clone());
                }
            }
        }
    }
    (lower, upper)
}

fn arm_has_any_version(arm: &Arm) -> bool {
    has_stable_witness(arm)
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        if !arm_has_any_version(arm) {
            continue;
        }
        let (lower, upper) = bounds(arm);
        let lower_text = lower
            .as_ref()
            .map(|c| {
                let op = if matches!(c.op, Op::Gt) { ">" } else { ">=" };
                format!("{op}{}", version_string(&c.version))
            })
            .unwrap_or_else(|| "*".to_string());
        let upper_text = upper
            .as_ref()
            .map(|c| {
                let op = if matches!(c.op, Op::Lt) { "<" } else { "<=" };
                format!("{op}{}", version_string(&c.version))
            })
            .unwrap_or_else(|| "*".to_string());
        let mut anchors = arm.anchors.clone();
        anchors.sort_unstable();
        anchors.dedup();
        let anchors_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|&t| core_string(t))
                .collect::<Vec<_>>()
                .join(",")
        };
        strings.push(format!(
            "[{};{};pre={}]",
            lower_text, upper_text, anchors_text
        ));
    }
    strings.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".to_string()
    } else {
        strings.join(" || ")
    }
}

fn successor_core(core: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if core.2 < u64::MAX {
        Some((core.0, core.1, core.2 + 1))
    } else if core.1 < u64::MAX {
        Some((core.0, core.1 + 1, 0))
    } else if core.0 < u64::MAX {
        Some((core.0 + 1, 0, 0))
    } else {
        None
    }
}

fn arm_admits_tuple(arm: &Arm, core: (u64, u64, u64)) -> bool {
    arm.anchors.iter().any(|&t| t == core)
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            // Test stable witnesses at the relevant core boundaries.
            let (la, ua) = bounds(aa);
            let (lb, ub) = bounds(bb);
            let mut cores = vec![(0, 0, 0), (u64::MAX, u64::MAX, u64::MAX)];
            for c in [la.as_ref(), lb.as_ref(), ua.as_ref(), ub.as_ref()]
                .into_iter()
                .flatten()
            {
                cores.push(vtuple(&c.version));
                if let Some(next) = successor_core(vtuple(&c.version)) {
                    cores.push(next);
                }
                if let Some(prev) = predecessor_core(vtuple(&c.version)) {
                    cores.push(prev);
                }
            }
            for core in cores {
                let stable = version_with_core(core, Vec::new());
                if arm_matches(&stable, aa, include_prerelease)
                    && arm_matches(&stable, bb, include_prerelease)
                {
                    return true;
                }
            }

            if include_prerelease {
                for core in aa.anchors.iter().chain(&bb.anchors).copied() {
                    let pre = version_with_core(core, zero_pre());
                    if arm_matches(&pre, aa, true) && arm_matches(&pre, bb, true) {
                        return true;
                    }
                }
            } else {
                for &core in &aa.anchors {
                    if !arm_admits_tuple(bb, core) {
                        continue;
                    }
                    let pre = version_with_core(core, zero_pre());
                    if arm_matches(&pre, aa, false) && arm_matches(&pre, bb, false) {
                        return true;
                    }
                }
            }
        }
    }
    false
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    ranges: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let parsed = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            ranges.push(parsed);
            Ok(())
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, ranges)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn eval_expr(
    expr: &SetExpr,
    ranges: &[Range],
    leaf_index: &mut usize,
    v: &Version,
    include: bool,
) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let result = matches_with(v, &ranges[*leaf_index], include);
            *leaf_index += 1;
            result
        }
        SetExpr::Union(children) => children
            .iter()
            .any(|child| eval_expr(child, ranges, leaf_index, v, include)),
        SetExpr::Intersection(children) => children
            .iter()
            .all(|child| eval_expr(child, ranges, leaf_index, v, include)),
    }
}

fn constraint_accepts(
    expr: &SetExpr,
    ranges: &[Range],
    v: &Version,
    include: bool,
) -> bool {
    let mut index = 0;
    eval_expr(expr, ranges, &mut index, v, include)
}

fn minimum_conflict(
    package: &Package,
    parsed: &[Version],
    ranges_by_constraint: &[Vec<Range>],
    include: bool,
) -> Vec<String> {
    if parsed.is_empty() {
        return Vec::new();
    }

    let n = package.constraints.len();
    for size in 1..=n {
        let mut best: Option<Vec<String>> = None;
        fn search(
            start: usize,
            target: usize,
            chosen: &mut Vec<usize>,
            best: &mut Option<Vec<String>>,
            package: &Package,
            parsed: &[Version],
            ranges: &[Vec<Range>],
            include: bool,
        ) {
            if chosen.len() == target {
                let satisfiable = parsed.iter().any(|v| {
                    chosen.iter().all(|&i| {
                        constraint_accepts(
                            &package.constraints[i].expr,
                            &ranges[i],
                            v,
                            include,
                        )
                    })
                });
                if !satisfiable {
                    let mut labels: Vec<String> = chosen
                        .iter()
                        .map(|&i| package.constraints[i].label.clone())
                        .collect();
                    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                    if best.as_ref().map_or(true, |old| {
                        labels.as_slice().cmp(old.as_slice()) == Ordering::Less
                    }) {
                        *best = Some(labels);
                    }
                }
                return;
            }
            let needed = target - chosen.len();
            for i in start..=package.constraints.len().saturating_sub(needed) {
                chosen.push(i);
                search(
                    i + 1, target, chosen, best, package, parsed, ranges, include,
                );
                chosen.pop();
            }
        }
        let mut chosen = Vec::new();
        search(
            0,
            size,
            &mut chosen,
            &mut best,
            package,
            parsed,
            ranges_by_constraint,
            include,
        );
        if let Some(labels) = best {
            return labels;
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut package_names = std::collections::HashSet::new();
    for package in packages {
        if !package_names.insert(package.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: package.name.clone(),
            });
        }
    }

    for package in packages {
        let mut labels = std::collections::HashSet::new();
        for constraint in &package.constraints {
            if !labels.insert(constraint.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: package.name.clone(),
                    label: constraint.label.clone(),
                });
            }
        }
    }

    let mut parsed_candidates = Vec::new();
    for package in packages {
        let mut parsed = Vec::new();
        for (index, text) in package.candidates.iter().enumerate() {
            parsed.push(
                parse_version(text).map_err(|error| SelectError::InvalidCandidate {
                    package: package.name.clone(),
                    index,
                    error,
                })?,
            );
        }
        parsed_candidates.push(parsed);
    }

    let mut ranges_per_package = Vec::new();
    for package in packages {
        let mut constraints_ranges = Vec::new();
        for constraint in &package.constraints {
            let mut ranges = Vec::new();
            validate_expr(
                &constraint.expr,
                &package.name,
                &constraint.label,
                &mut Vec::new(),
                &mut ranges,
            )?;
            constraints_ranges.push(ranges);
        }
        ranges_per_package.push(constraints_ranges);
    }

    let mut selections = Vec::new();
    for (package_index, package) in packages.iter().enumerate() {
        let parsed = &parsed_candidates[package_index];
        let range_sets = &ranges_per_package[package_index];
        let mut qualifying: Vec<usize> = parsed
            .iter()
            .enumerate()
            .filter_map(|(i, version)| {
                package
                    .constraints
                    .iter()
                    .enumerate()
                    .all(|(ci, constraint)| {
                        constraint_accepts(
                            &constraint.expr,
                            &range_sets[ci],
                            version,
                            include_prerelease,
                        )
                    })
                    .then_some(i)
            })
            .collect();

        if qualifying.is_empty() {
            let labels = minimum_conflict(package, parsed, range_sets, include_prerelease);
            return Err(SelectError::NoSolution {
                package: package.name.clone(),
                labels,
            });
        }

        let stable_first = matches!(preference, Preference::StableFirst)
            && qualifying.iter().any(|&i| parsed[i].pre.is_empty());
        qualifying.retain(|&i| !stable_first || parsed[i].pre.is_empty());

        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let ord = compare_versions(&parsed[i], &parsed[best]);
            let improve = match preference {
                Preference::Highest | Preference::StableFirst => ord == Ordering::Greater,
                Preference::Lowest => ord == Ordering::Less,
            };
            if improve {
                best = i;
            }
        }
        selections.push(Selection {
            package: package.name.clone(),
            version: package.candidates[best].clone(),
        });
    }
    Ok(selections)
}
