#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, String)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + offset,
        });
    }

    let mut content_start = open + 1;
    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    match bytes[content_start] {
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' => {
            return Err(Error::UnsupportedOperator {
                pos: content_start,
            });
        }
        _ => {}
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();

    loop {
        let entry_start = content_start;
        let entry_end = bytes[entry_start..close]
            .iter()
            .position(|&b| b == b',')
            .map_or(close, |offset| entry_start + offset);

        if entry_start == entry_end {
            return Err(Error::EmptyVariable { pos: entry_start });
        }

        let name_bytes = &bytes[entry_start..entry_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: entry_start });
        }

        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: entry_start + offset,
                });
            }
        }

        let name = template[entry_start..entry_end].to_owned();
        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: entry_start });
        }
        seen.push(name.clone());
        names.push((entry_start, name));

        if entry_end == close {
            break;
        }
        content_start = entry_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        names,
    })
}

fn encode_value(value: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        i += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let Some(offset) = bytes[cursor + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: cursor });
                };
                let close = cursor + 1 + offset;
                expressions.push(parse_expression(template, cursor, close)?);
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut first = true;
        for (_, name) in expression.names {
            let value = vars
                .iter()
                .rev()
                .find(|(binding_name, _)| *binding_name == name)
                .map(|(_, value)| *value);

            if let Some(value) = value {
                if !first {
                    output.push(',');
                }
                output.push_str(&encode_value(value));
                first = false;
            }
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn main() {}
