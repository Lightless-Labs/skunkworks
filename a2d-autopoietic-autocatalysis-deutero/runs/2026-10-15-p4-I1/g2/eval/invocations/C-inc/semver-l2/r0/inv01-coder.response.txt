use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }
        if pos == start {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut value = 0u64;
        for &b in &bytes[start..pos] {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        core[i] = value;

        if i < 2 {
            if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let plus = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|n| pos + n);
        let pre_end = plus.unwrap_or(bytes.len());

        parse_identifiers(s, pre_start, pre_end, true, &mut pre)?;
        pos = pre_end;

        if pos < bytes.len() {
            pos += 1;
            parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
            pos = bytes.len();
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    check_pre_zero: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let mut id_start = start;
    loop {
        let id_end = (id_start..end)
            .find(|&i| bytes[i] == b'.')
            .unwrap_or(end);
        if id_end == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        for i in id_start..id_end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }

        if check_pre_zero
            && bytes[id_start..id_end].iter().all(|&b| is_digit(b))
            && id_end - id_start > 1
            && bytes[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        out.push(s[id_start..id_end].to_string());
        if id_end == end {
            break;
        }
        id_start = id_end + 1;
        if id_start == end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(())
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|&b| is_digit(b))
}

fn compare_numeric_ids(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = match (is_numeric_id(x), is_numeric_id(y)) {
                        (true, true) => compare_numeric_ids(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

fn range_tokens(s: &str) -> Vec<(usize, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        out.push((start, i));
    }
    out
}

fn version_at(s: &str, start: usize, end: usize) -> Result<Version, RangeError> {
    parse_version(&s[start..end]).map_err(|error| RangeError::Version { at: start, error })
}

fn numeric_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn release_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn zero_pre_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    }
}

fn comparator(op: Op, version: Version) -> Comparator {
    Comparator { op, version }
}

fn add_expanded(
    out: &mut Vec<Comparator>,
    anchors: &mut Vec<(u64, u64, u64)>,
    lower: Version,
    upper: Option<Version>,
    lower_op: Op,
    lower_is_anchor: bool,
) {
    if lower_is_anchor && !lower.pre.is_empty() {
        anchors.push(numeric_tuple(&lower));
    }
    out.push(comparator(lower_op, lower));
    if let Some(upper) = upper {
        out.push(comparator(Op::Lt, upper));
    }
}

fn increment_tuple(
    major: u64,
    minor: u64,
    patch: u64,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let (m, n, p) = match field {
        0 => (major.checked_add(1), Some(0), Some(0)),
        1 => (Some(major), minor.checked_add(1), Some(0)),
        _ => (Some(major), Some(minor), patch.checked_add(1)),
    };
    match (m, n, p) {
        (Some(m), Some(n), Some(p)) => Ok(zero_pre_version(m, n, p)),
        _ => Err(RangeError::Syntax { at }),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    numbers: Vec<u64>,
    full: Option<Version>,
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn parse_partial(s: &str, start: usize, end: usize) -> Result<Partial, RangeError> {
    let operand = &s[start..end];
    if operand.contains('-') || operand.contains('+') {
        let full = version_at(s, start, end)?;
        return Ok(Partial {
            numbers: vec![full.major, full.minor, full.patch],
            full: Some(full),
        });
    }

    let fields: Vec<&str> = operand.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f)) {
        let full = version_at(s, start, end)?;
        return Ok(Partial {
            numbers: vec![full.major, full.minor, full.patch],
            full: Some(full),
        });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }

    let mut numbers = Vec::new();
    let mut wildcard_seen = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if is_wildcard(field) {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen
            || !field.as_bytes().iter().all(|&b| is_digit(b))
            || (field.len() > 1 && field.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at: start });
        }
        let mut n = 0u64;
        for &b in field.as_bytes() {
            n = n
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u64))
                .ok_or(RangeError::Syntax { at: start })?;
        }
        numbers.push(n);
    }

    Ok(Partial {
        numbers,
        full: None,
    })
}

fn padded(partial: &Partial) -> (u64, u64, u64) {
    let mut vals = [0u64; 3];
    for (i, &n) in partial.numbers.iter().enumerate() {
        vals[i] = n;
    }
    (vals[0], vals[1], vals[2])
}

fn parse_operand(
    s: &str,
    start: usize,
    end: usize,
    kind: u8,
    out: &mut Vec<Comparator>,
    anchors: &mut Vec<(u64, u64, u64)>,
) -> Result<(), RangeError> {
    let partial = parse_partial(s, start, end)?;
    let k = partial.numbers.len();
    let (m, n, p) = padded(&partial);

    match kind {
        b'^' => {
            if k == 0 {
                return Err(RangeError::Syntax { at: start - 1 });
            }
            let lower = partial.full.unwrap_or_else(|| release_version(m, n, p));
            let field = if m > 0 || k == 1 {
                0
            } else if n > 0 || k == 2 {
                1
            } else {
                2
            };
            let upper = increment_tuple(m, n, p, field, start - 1)?;
            add_expanded(out, anchors, lower, Some(upper), Op::Ge, true);
        }
        b'~' => {
            if k == 0 {
                return Err(RangeError::Syntax { at: start - 1 });
            }
            let lower = partial.full.unwrap_or_else(|| release_version(m, n, p));
            let upper = increment_tuple(m, n, p, if k == 1 { 0 } else { 1 }, start - 1)?;
            add_expanded(out, anchors, lower, Some(upper), Op::Ge, true);
        }
        _ => {
            if let Some(full) = partial.full {
                if !full.pre.is_empty() {
                    anchors.push(numeric_tuple(&full));
                }
                out.push(comparator(Op::Eq, full));
            } else {
                match k {
                    0 => {}
                    1 => {
                        out.push(comparator(Op::Ge, release_version(m, n, p)));
                        let upper = increment_tuple(m, n, p, 0, start)?;
                        out.push(comparator(Op::Lt, upper));
                    }
                    2 => {
                        out.push(comparator(Op::Ge, release_version(m, n, p)));
                        let upper = increment_tuple(m, n, p, 1, start)?;
                        out.push(comparator(Op::Lt, upper));
                    }
                    _ => out.push(comparator(Op::Eq, release_version(m, n, p))),
                }
            }
        }
    }
    Ok(())
}

fn parse_comparator_token(
    s: &str,
    start: usize,
    end: usize,
    out: &mut Vec<Comparator>,
    anchors: &mut Vec<(u64, u64, u64)>,
) -> Result<(), RangeError> {
    let bytes = s.as_bytes();
    let mut operand = start;
    let op = if bytes[operand] == b'<' {
        operand += 1;
        if operand < end && bytes[operand] == b'=' {
            operand += 1;
            Op::Le
        } else {
            Op::Lt
        }
    } else if bytes[operand] == b'>' {
        operand += 1;
        if operand < end && bytes[operand] == b'=' {
            operand += 1;
            Op::Ge
        } else {
            Op::Gt
        }
    } else if bytes[operand] == b'=' {
        operand += 1;
        Op::Eq
    } else {
        Op::Eq
    };

    if operand == end || !is_digit(bytes[operand]) {
        return Err(RangeError::Syntax { at: start });
    }
    let v = version_at(s, operand, end)?;
    if !v.pre.is_empty() {
        anchors.push(numeric_tuple(&v));
    }
    out.push(comparator(op, v));
    Ok(())
}

fn parse_arm(s: &str, tokens: &[(usize, usize)]) -> Result<Arm, RangeError> {
    let mut out = Vec::new();
    let mut anchors = Vec::new();

    let hyphen_positions: Vec<usize> = tokens
        .iter()
        .enumerate()
        .filter(|(_, (a, b))| &s[*a..*b] == "-")
        .map(|(i, _)| i)
        .collect();

    if let Some(&minus_index) = hyphen_positions.first() {
        if tokens.len() != 3 || minus_index != 1 {
            return Err(RangeError::Syntax {
                at: tokens[minus_index].0,
            });
        }

        let (l0, l1) = tokens[0];
        let (u0, u1) = tokens[2];
        let lower = parse_partial(s, l0, l1)?;
        let upper = parse_partial(s, u0, u1)?;
        if lower.numbers.is_empty() {
            return Err(RangeError::Syntax { at: l0 });
        }
        if upper.numbers.is_empty() {
            return Err(RangeError::Syntax { at: u0 });
        }

        let (lm, ln, lp) = padded(&lower);
        let lower_v = lower
            .full
            .clone()
            .unwrap_or_else(|| release_version(lm, ln, lp));
        if !lower_v.pre.is_empty() {
            anchors.push(numeric_tuple(&lower_v));
        }
        out.push(comparator(Op::Ge, lower_v));

        if let Some(full) = upper.full {
            if !full.pre.is_empty() {
                anchors.push(numeric_tuple(&full));
            }
            out.push(comparator(Op::Le, full));
        } else {
            let (um, un, up) = padded(&upper);
            let field = if upper.numbers.len() == 1 { 0 } else { 1 };
            let upper_v = increment_tuple(um, un, up, field, u0)?;
            out.push(comparator(Op::Lt, upper_v));
        }

        return Ok(Arm {
            comparators: out,
            anchors,
        });
    }

    for &(start, end) in tokens {
        let bytes = s.as_bytes();
        let first = bytes[start];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_comparator_token(s, start, end, &mut out, &mut anchors)?;
        } else if first == b'^' || first == b'~' {
            if start + 1 == end {
                return Err(RangeError::Syntax { at: start });
            }
            parse_operand(
                s,
                start + 1,
                end,
                first,
                &mut out,
                &mut anchors,
            )?;
        } else if is_digit(first) || is_wildcard(&s[start..end]) {
            parse_operand(s, start, end, 0, &mut out, &mut anchors)?;
        } else {
            return Err(RangeError::Syntax { at: start });
        }
    }

    Ok(Arm {
        comparators: out,
        anchors,
    })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = range_tokens(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        let (start, end) = tokens[i];
        if &s[start..end] == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(parse_arm(s, &current)?);
            current.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push((start, end));
            i += 1;
        }
    }

    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(s, &current)?);

    Ok(Range { arms })
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        (!v.pre.is_empty()
            && arm
                .anchors
                .iter()
                .any(|&tuple| tuple == numeric_tuple(v))
            || v.pre.is_empty()
            || arm
                .anchors
                .iter()
                .any(|&tuple| tuple == numeric_tuple(v)))
            && arm.comparators.iter().all(|c| cmp_matches(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version =
            parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(best_index) => {
                let ord = compare_versions(version, &parsed[best_index]);
                if (want_max && ord == Ordering::Greater)
                    || (!want_max && ord == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }

    Ok(best.map(|index| candidates[index].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
