use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [BTreeSet<u8>; 5],
    day_unrestricted: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn err_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn name_value(field: Field, s: &str) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    let upper = s.to_ascii_uppercase();
    names
        .iter()
        .position(|n| *n == upper)
        .map(|n| n as u32 + if field == Field::Month { 1 } else { 0 })
}

fn parse_atom(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(err_syntax(field, offset));
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut value = 0u32;
        for &b in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let (min, max) = bounds(field);
        if value < min as u32 || value > max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok(value)
    } else if bytes.iter().all(u8::is_ascii_alphabetic) {
        name_value(field, s).ok_or_else(|| err_syntax(field, offset))
    } else {
        Err(err_syntax(field, offset))
    }
}

fn parse_step(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(err_syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (min, max) = bounds(field);
    let width = max as u32 - min as u32 + 1;
    if value == 0 || value > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_item(
    field: Field,
    text: &str,
    start_offset: usize,
    selected: &mut BTreeSet<u8>,
) -> Result<(), ParseError> {
    if text.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: start_offset,
        });
    }

    let slash_count = text.bytes().filter(|&b| b == b'/').count();
    if slash_count > 1 {
        return Err(err_syntax(field, start_offset));
    }
    let (base, step_text) = if let Some(pos) = text.find('/') {
        (&text[..pos], Some((&text[pos + 1..], start_offset + pos + 1)))
    } else {
        (text, None)
    };
    if base.is_empty() || step_text.map_or(false, |(s, _)| s.is_empty()) {
        return Err(err_syntax(field, start_offset));
    }

    let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
    if hyphen_count > 1 {
        return Err(err_syntax(field, start_offset));
    }
    let range_pos = base.find('-');
    if base == "*" {
        // Valid wildcard.
    } else if let Some(pos) = range_pos {
        if pos == 0
            || pos + 1 == base.len()
            || base[..pos].contains('*')
            || base[pos + 1..].contains('*')
        {
            return Err(err_syntax(field, start_offset));
        }
    } else if base.contains('*') {
        return Err(err_syntax(field, start_offset));
    }

    let (min, max) = bounds(field);
    let (from, to) = if base == "*" {
        (min as u32, max as u32)
    } else if let Some(pos) = range_pos {
        let a = parse_atom(field, &base[..pos], start_offset)?;
        let b_offset = start_offset + pos + 1;
        let b = parse_atom(field, &base[pos + 1..], b_offset)?;
        if a > b {
            return Err(ParseError::ReversedRange {
                field,
                offset: b_offset,
            });
        }
        (a, b)
    } else {
        let a = parse_atom(field, base, start_offset)?;
        (a, if step_text.is_some() { max as u32 } else { a })
    };

    let step = if let Some((s, off)) = step_text {
        parse_step(field, s, off)?
    } else {
        1
    };

    let mut value = from;
    while value <= to {
        selected.insert(value as u8);
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    if input.as_bytes().first() == Some(&b'@') {
        let expansion = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        return parse_cron(expansion);
    }

    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    let mut unrestricted = [false; 2];

    for (i, text) in parts.iter().enumerate() {
        let field = FIELDS[i];
        for (offset, &b) in text.as_bytes().iter().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if field == Field::DayOfMonth {
            unrestricted[0] = *text == "*";
        } else if field == Field::DayOfWeek {
            unrestricted[1] = *text == "*";
        }

        let mut offset = 0;
        for item in text.split(',') {
            parse_item(field, item, offset, &mut sets[i])?;
            offset += item.len() + 1;
        }
    }

    Ok(Cron {
        sets,
        day_unrestricted: unrestricted,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].iter().copied().collect()
    }

    fn date_matches(&self, year: u16, month: u8, day: u8, weekday: u8) -> bool {
        let dom = self.sets[2].contains(&day);
        let dow = self.sets[4].contains(&weekday);
        match self.day_unrestricted {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if day == days_in_month(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn validate(at: CivilDateTime) -> Result<(), DateError> {
        if !(2000..=2399).contains(&at.year) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Year,
            });
        }
        if !(1..=12).contains(&at.month) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Month,
            });
        }
        if at.day == 0 || at.day > days_in_month(at.year, at.month) {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Day,
            });
        }
        if at.hour > 23 {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Hour,
            });
        }
        if at.minute > 59 {
            return Err(DateError::InvalidDateTime {
                component: DatePart::Minute,
            });
        }
        Ok(())
    }

    fn time_matches(&self, at: CivilDateTime) -> bool {
        self.sets[0].contains(&at.minute)
            && self.sets[1].contains(&at.hour)
            && self.sets[3].contains(&at.month)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        Self::validate(at)?;
        Ok(self.time_matches(at)
            && self.date_matches(at.year, at.month, at.day, weekday(at.year, at.month, at.day)))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        Self::validate(after)?;
        let mut date = after;
        loop {
            let times = self.times_for_date(date.year, date.month, date.day);
            for t in times {
                if t > after {
                    return Ok(Some(t));
                }
            }
            if !advance_day(&mut date) {
                return Ok(None);
            }
            date.hour = 0;
            date.minute = 0;
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        Self::validate(before)?;
        let mut date = before;
        loop {
            let times = self.times_for_date(date.year, date.month, date.day);
            for t in times.into_iter().rev() {
                if t < before {
                    return Ok(Some(t));
                }
            }
            if !retreat_day(&mut date) {
                return Ok(None);
            }
            date.hour = 23;
            date.minute = 59;
        }
    }

    fn times_for_date(&self, year: u16, month: u8, day: u8) -> Vec<CivilDateTime> {
        let wd = weekday(year, month, day);
        if !self.sets[3].contains(&month) || !self.date_matches(year, month, day, wd) {
            return Vec::new();
        }
        let mut out = Vec::new();
        for &hour in &self.sets[1] {
            for &minute in &self.sets[0] {
                out.push(CivilDateTime {
                    year,
                    month,
                    day,
                    hour,
                    minute,
                });
            }
        }
        out
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        Self::validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        Self::validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        if start == end {
            return Ok(FireWindow {
                fires,
                truncated: false,
            });
        }

        let mut date = start;
        loop {
            for t in self.times_for_date(date.year, date.month, date.day) {
                if t >= start && t < end {
                    if fires.len() == cap {
                        return Ok(FireWindow {
                            fires,
                            truncated: true,
                        });
                    }
                    fires.push(t);
                }
            }
            if date.year == end.year && date.month == end.month && date.day == end.day {
                break;
            }
            if !advance_day(&mut date) {
                break;
            }
        }

        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut result = Vec::new();
        for i in 0..5 {
            let field = FIELDS[i];
            let set = &self.sets[i];
            let (min, max) = bounds(field);
            let full: BTreeSet<u8> = (min..=max).collect();

            let text = if i == 2 && self.day_unrestricted[0] {
                "*".to_string()
            } else if i == 4 && self.day_unrestricted[1] {
                "*".to_string()
            } else if *set == full {
                if i == 2 || i == 4 {
                    "*/1".to_string()
                } else {
                    "*".to_string()
                }
            } else {
                set.iter().map(u8::to_string).collect::<Vec<_>>().join(",")
            };
            result.push(text);
        }
        result.join(" ")
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => {
            if is_leap(year) {
                29
            } else {
                28
            }
        }
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days: u32 = 0;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_day(date: &mut CivilDateTime) -> bool {
    if date.day < days_in_month(date.year, date.month) {
        date.day += 1;
    } else if date.month < 12 {
        date.month += 1;
        date.day = 1;
    } else if date.year < 2399 {
        date.year += 1;
        date.month = 1;
        date.day = 1;
    } else {
        return false;
    }
    true
}

fn retreat_day(date: &mut CivilDateTime) -> bool {
    if date.day > 1 {
        date.day -= 1;
    } else if date.month > 1 {
        date.month -= 1;
        date.day = days_in_month(date.year, date.month);
    } else if date.year > 2000 {
        date.year -= 1;
        date.month = 12;
        date.day = 31;
    } else {
        return false;
    }
    true
}
