fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by(|a, b| constraint_key(*a).cmp(&constraint_key(*b)));
    out.dedup();
    out
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_key(a.constraint)
        .cmp(&constraint_key(b.constraint))
        .then_with(|| a.path.cmp(&b.path))
}

fn evidence_lines(out: &mut String, evidence: &[Evidence]) {
    let mut ev = evidence.to_vec();
    ev.sort_by(evidence_cmp);
    for e in ev {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            evidence_lines(&mut s, evidence);
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let c = canonical_constraints(requirements)
                .iter()
                .map(|x| constraint_text(*x))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!("no-match|package={}|constraints={}", encode(package), c);
            evidence_lines(&mut s, evidence);
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let c = canonical_constraints(requirements)
                .iter()
                .map(|x| constraint_text(*x))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                c
            );
            evidence_lines(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

#[derive(Clone, Debug)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct State {
    constraints: BTreeMap<String, Vec<Constraint>>,
    features: BTreeMap<String, BTreeSet<String>>,
    origins: BTreeMap<String, Vec<Origin>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn sorted_features(features: &[String]) -> String {
    let mut fs = features.to_vec();
    fs.sort();
    fs.dedup();
    fs.join(",")
}

fn root_token(index: usize, root: &Requirement) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        sorted_features(&root.features)
    )
}

fn dependency_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        sorted_features(&dep.features),
        gate,
        platform
    )
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn merge_path(paths: &mut BTreeMap<String, Vec<String>>, package: &str, candidate: Vec<String>) -> bool {
    match paths.get(package) {
        Some(old) if old.len() < candidate.len() || (old.len() == candidate.len() && old <= &candidate) => false,
        _ => {
            paths.insert(package.to_string(), candidate);
            true
        }
    }
}

struct Search<'a> {
    catalog: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    budget: usize,
    attempts: usize,
    first_failure: Option<ResolveError>,
    best: Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>, State)>,
    all_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn make_state(&self, selected: &BTreeMap<String, Version>) -> State {
        let mut state = State::default();
        for (i, root) in self.roots.iter().enumerate() {
            let token = root_token(i, root);
            state
                .constraints
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
            state
                .features
                .entry(root.package.clone())
                .or_default()
                .extend(root.features.iter().cloned());
            state.origins.entry(root.package.clone()).or_default().push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
            merge_path(&mut state.paths, &root.package, vec![token]);
        }

        // Propagate newly activated dependencies/features to a fixed point. Only
        // add each dependency edge once; this avoids repeated origins on cycles.
        let mut emitted = BTreeSet::<(String, usize)>::new();
        loop {
            let mut changed = false;

            for (name, version) in selected {
                let Some(package) = self.catalog.get(name).copied() else {
                    continue;
                };
                let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
                    continue;
                };

                let activated = state.features.get(name).cloned().unwrap_or_default();
                for feature in &activated {
                    if let Some(decl) = release.features.iter().find(|f| f.name == *feature) {
                        let entry = state.features.entry(name.clone()).or_default();
                        for enabled in &decl.enables {
                            if entry.insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }

                let activated = state.features.get(name).cloned().unwrap_or_default();
                for (index, dep) in release.deps.iter().enumerate() {
                    if !target_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                        || emitted.contains(&(name.clone(), index))
                    {
                        continue;
                    }
                    emitted.insert((name.clone(), index));
                    let token = dependency_token(name, *version, index, dep);
                    let mut path = state.paths.get(name).cloned().unwrap_or_default();
                    path.push(token);
                    state
                        .constraints
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                    state
                        .features
                        .entry(dep.package.clone())
                        .or_default()
                        .extend(dep.features.iter().cloned());
                    state.origins.entry(dep.package.clone()).or_default().push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                    if merge_path(&mut state.paths, &dep.package, path) {
                        changed = true;
                    }
                    changed = true;
                }
            }

            if !changed {
                break;
            }
        }
        state
    }

    fn error_evidence(state: &State, package: &str) -> Vec<Evidence> {
        let mut out: Vec<Evidence> = state
            .origins
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect();
        out.sort_by(evidence_cmp);
        out
    }

    fn fail(&mut self, error: ResolveError) {
        if self.first_failure.is_none() {
            self.first_failure = Some(error);
        }
    }

    fn inspect(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let state = self.make_state(selected);
        let active: Vec<String> = state.constraints.keys().cloned().collect();

        for name in &active {
            if !self.catalog.contains_key(name) {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: Self::error_evidence(&state, name),
                });
            }
        }

        for name in selected.keys() {
            let constraints = state.constraints.get(name).cloned().unwrap_or_default();
            if !constraints.iter().all(|c| matches(selected[name], *c)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: selected[name],
                    requirements: canonical_constraints(&constraints),
                    evidence: Self::error_evidence(&state, name),
                });
            }
        }

        for name in selected.keys() {
            let package = self.catalog[name];
            let release = package
                .releases
                .iter()
                .find(|r| r.version == selected[name])
                .expect("selected release must exist");
            let declared: BTreeSet<&str> =
                release.features.iter().map(|f| f.name.as_str()).collect();
            for feature in state.features.get(name).into_iter().flatten() {
                if !declared.contains(feature.as_str()) {
                    return Err(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: selected[name],
                        feature: feature.clone(),
                        path: state.paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        for name in &active {
            if selected.contains_key(name) {
                continue;
            }
            let constraints = state.constraints.get(name).cloned().unwrap_or_default();
            if matching_candidates(self.catalog[name], &constraints).is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(&constraints),
                    evidence: Self::error_evidence(&state, name),
                });
            }
        }

        let next = active.iter().find(|n| !selected.contains_key(*n)).cloned();
        let Some(name) = next else {
            self.record_solution(selected, state);
            return Ok(());
        };

        let constraints = state.constraints.get(&name).cloned().unwrap_or_default();
        let mut candidates = matching_candidates(self.catalog[&name], &constraints);
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        let mut first_local_failure = None;
        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            match self.inspect(selected) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    selected.remove(&name);
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if first_local_failure.is_none() {
                        first_local_failure = Some(error.clone());
                    }
                    self.fail(error);
                }
            }
            selected.remove(&name);
        }

        if self.best.is_none() {
            if self.first_failure.is_none() {
                self.first_failure = first_local_failure;
            }
        }
        Ok(())
    }

    fn record_solution(&mut self, selected: &BTreeMap<String, Version>, state: State) {
        let cost = self
            .locks
            .iter()
            .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
            .count();
        let vector: Vec<Option<Version>> = self
            .all_names
            .iter()
            .map(|name| selected.get(name).copied())
            .collect();
        let better = self.best.as_ref().is_none_or(|(best_cost, best_vector, _, _)| {
            cost < *best_cost || (cost == *best_cost && vector > *best_vector)
        });
        if better {
            self.best = Some((cost, vector, selected.clone(), state));
        }
    }
}

fn matching_candidates(pkg: &Package, constraints: &[Constraint]) -> Vec<Version> {
    let mut versions: Vec<Version> = pkg
        .releases
        .iter()
        .map(|r| r.version)
        .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
        .collect();
    versions.sort_by(|a, b| b.cmp(a));
    versions
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut feature_names = BTreeSet::new();
            for feature in &release.features {
                if !feature_names.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let all_names = catalog_map.keys().cloned().collect();
    let mut search = Search {
        catalog: catalog_map,
        roots,
        locks,
        target,
        budget: max_attempts,
        attempts: 0,
        first_failure: None,
        best: None,
        all_names,
    };
    let mut selected = BTreeMap::new();
    if let Err(ResolveError::Limit { attempts }) = search.inspect(&mut selected) {
        return Err(ResolveError::Limit { attempts });
    }

    let Some((_, _, chosen, final_state)) = search.best else {
        return Err(search.first_failure.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_out = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let changed_names: BTreeSet<String> =
        lock_map.keys().chain(chosen.keys()).cloned().collect();
    let mut changes = Vec::new();

    for package in changed_names {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search.catalog.get(&package).is_some_and(|p| {
                    p.releases.iter().any(|release| release.version == old)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_state
                    .constraints
                    .get(&package)
                    .cloned()
                    .unwrap_or_default()
                    .iter()
                    .all(|constraint| matches(old, *constraint))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
