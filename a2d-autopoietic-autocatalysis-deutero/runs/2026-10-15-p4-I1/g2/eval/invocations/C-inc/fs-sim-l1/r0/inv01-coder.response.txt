```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: std::collections::HashMap<String, Entry>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::HashMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries }
    }

    fn validate(path: &str) -> Result<Vec<String>, Error> {
        if let Some(index) = path.bytes().position(|byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        for (index, byte) in path.bytes().enumerate().skip(1) {
            if byte == b'/' {
                let component = &path[start..index];
                if component.is_empty() || component == "." || component == ".." {
                    return Err(Error::InvalidPath(if component.is_empty() {
                        index
                    } else {
                        start
                    }));
                }
                components.push(component.to_string());
                start = index + 1;
            }
        }

        let component = &path[start..];
        if component.is_empty() || component == "." || component == ".." {
            return Err(Error::InvalidPath(if component.is_empty() {
                path.len() - 1
            } else {
                start
            }));
        }
        components.push(component.to_string());
        Ok(components)
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn lookup_parent(&self, components: &[String]) -> Result<String, Error> {
        let mut prefix = String::from("/");
        for component in components.iter().take(components.len() - 1) {
            prefix = if prefix == "/" {
                format!("/{component}")
            } else {
                format!("{prefix}/{component}")
            };
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
                Some(Entry::Dir) => {}
            }
        }
        Ok(prefix)
    }

    fn lookup_final(&self, components: &[String]) -> Result<Option<&Entry>, Error> {
        if components.is_empty() {
            return Ok(self.entries.get("/"));
        }

        let parent = self.lookup_parent(components)?;
        let path = Self::canonical(components);
        match self.entries.get(&parent) {
            Some(Entry::Dir) => Ok(self.entries.get(&path)),
            Some(Entry::File(_)) => Err(Error::NotDir(parent)),
            None => Err(Error::Missing(parent)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        let canonical = Self::canonical(&components);
        if components.is_empty() {
            return Err(Error::Exists(canonical));
        }

        let parent = self.lookup_parent(&components)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }

        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        let canonical = Self::canonical(&components);
        if components.is_empty() {
            return Err(Error::IsDir(canonical));
        }

        let parent = self.lookup_parent(&components)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }

        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        let canonical = Self::canonical(&components);
        match self.lookup_final(&components)? {
            None => Err(Error::Missing(canonical)),
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let canonical = Self::canonical(&components);
        match self.lookup_final(&components)? {
            None => Err(Error::Missing(canonical)),
            Some(Entry::File(_)) => Err(Error::NotDir(canonical)),
            Some(Entry::Dir) => {
                let mut children = Vec::new();
                for (entry_path, entry) in &self.entries {
                    let basename = if canonical == "/" {
                        match entry_path.strip_prefix('/') {
                            Some(rest) => rest,
                            None => continue,
                        }
                    } else if let Some(rest) = entry_path.strip_prefix(&canonical) {
                        match rest.strip_prefix('/') {
                            Some(child) => child,
                            None => continue,
                        }
                    } else {
                        continue;
                    };

                    if !basename.is_empty() && !basename.contains('/') {
                        let kind = match entry {
                            Entry::Dir => Kind::Dir,
                            Entry::File(_) => Kind::File,
                        };
                        children.push((basename.to_string(), kind));
                    }
                }
                children.sort_by(|left, right| left.0.cmp(&right.0));
                Ok(children)
            }
        }
    }
}

fn main() {}
```
