#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
enum RowKind {
    Context,
    Remove,
    Add,
}

#[derive(Clone)]
struct Row {
    kind: RowKind,
    payload: String,
    line: usize,
}

struct Hunk {
    header_line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.starts_with('0') {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, tail) = rest.split_once(" @@")?;
    if tail != "" {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    let a = parse_number(a)?;
    let b = parse_number(b)?;
    let c = parse_number(c)?;
    let d = parse_number(d)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn nominal_start(start: usize, count: usize) -> i128 {
    if count > 0 {
        start as i128 - 1
    } else {
        start as i128
    }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut index = 2;
    let mut previous_old_end: Option<i128> = None;
    let mut net: i128 = 0;

    while index < lines.len() {
        let header_line = index + 1;
        let (a, b, c, d) = parse_hunk_header(lines[index])
            .ok_or(ApplyError::BadHunkHeader { line: header_line })?;
        let old_start = nominal_start(a, b);
        let new_start = nominal_start(c, d);
        index += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while index < lines.len() && !lines[index].starts_with("@@") {
            let line = lines[index];
            let physical_line = index + 1;
            let bytes = line.as_bytes();
            if bytes.is_empty() || bytes[0] == b'\r' || line.contains('\r') {
                return Err(ApplyError::BadBody { line: physical_line });
            }
            let kind = match bytes[0] {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                    RowKind::Context
                }
                b'-' => {
                    old_seen += 1;
                    RowKind::Remove
                }
                b'+' => {
                    new_seen += 1;
                    RowKind::Add
                }
                _ => return Err(ApplyError::BadBody { line: physical_line }),
            };
            rows.push(Row {
                kind,
                payload: line[1..].to_string(),
                line: physical_line,
            });
            index += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows
            .iter()
            .any(|row| matches!(row.kind, RowKind::Add | RowKind::Remove))
        {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if previous_old_end.is_some_and(|end| old_start < end)
            || new_start != old_start + net
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = Some(old_start + b as i128);
        net += d as i128 - b as i128;

        hunks.push(Hunk {
            header_line,
            old_start: old_start as usize,
            old_count: b,
            new_start: new_start as usize,
            new_count: d,
            rows,
        });

        if hunks.len() == 17 {
            return Err(ApplyError::TooLarge);
        }
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.bytes().position(|b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_string)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_distance(a: i128, b: i128) -> i128 {
    (a - b).abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk
            .rows
            .iter()
            .take_while(|r| matches!(r.kind, RowKind::Context))
            .count();
        let trail = hunk
            .rows
            .iter()
            .rev()
            .take_while(|r| matches!(r.kind, RowKind::Context))
            .count();

        let expected = hunk.old_start as i128 + net + drift;
        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_starts = Vec::<(usize, usize, usize)>::new();

        for left in 0..=options.fuzz.min(lead) {
            for right in 0..=options.fuzz.min(trail) {
                if left + right > hunk.rows.len() {
                    continue;
                }
                let trimmed: Vec<&Row> = hunk
                    .rows
                    .iter()
                    .enumerate()
                    .filter_map(|(i, row)| {
                        if i < left || i >= hunk.rows.len().saturating_sub(right) {
                            None
                        } else {
                            Some(row)
                        }
                    })
                    .collect();

                let old_payloads: Vec<&str> = trimmed
                    .iter()
                    .filter(|r| matches!(r.kind, RowKind::Context | RowKind::Remove))
                    .map(|r| r.payload.as_str())
                    .collect();

                if hunk.old_count > working.len() {
                    continue;
                }
                for start in 0..=working.len() - hunk.old_count {
                    let distance = abs_distance(start as i128, expected);
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let match_start = start + left;
                    let match_end = start + hunk.old_count - right;
                    if match_end > working.len()
                        || old_payloads.len() != match_end.saturating_sub(match_start)
                        || !old_payloads
                            .iter()
                            .zip(&working[match_start..match_end])
                            .all(|(a, b)| *a == b)
                    {
                        continue;
                    }

                    let rank = (left + right, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((start, left, right));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((start, left, right));
                        }
                        Some(current) if rank == current => {
                            best_starts.push((start, left, right));
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }
        best_starts.sort_by_key(|item| (item.0, item.1));
        let distinct: Vec<usize> = best_starts
            .iter()
            .map(|item| item.0)
            .fold(Vec::new(), |mut acc, start| {
                if acc.last().copied() != Some(start) {
                    acc.push(start);
                }
                acc
            });
        if distinct.len() >= 2 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }

        let (start, left, right) = best_starts[0];
        let trimmed_rows: Vec<&Row> = hunk
            .rows
            .iter()
            .enumerate()
            .filter_map(|(i, row)| {
                if i < left || i >= hunk.rows.len().saturating_sub(right) {
                    None
                } else {
                    Some(row)
                }
            })
            .collect();

        let mut replacement = Vec::new();
        for row in &trimmed_rows {
            if matches!(row.kind, RowKind::Context | RowKind::Add) {
                replacement.push(row.payload.clone());
            }
        }
        let replace_start = start + left;
        let replace_end = start + hunk.old_count - right;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = start as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

fn main() {}
