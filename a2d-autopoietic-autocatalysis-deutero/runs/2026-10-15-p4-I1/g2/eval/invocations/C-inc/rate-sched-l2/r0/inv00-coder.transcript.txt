#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        for (i, b) in text.bytes().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = text
            .bytes()
            .enumerate()
            .filter_map(|(i, b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 {
            return Err(CronError::Layout(0));
        }
        if split + 1 == text.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        Self::parse_field(&text[..split], 0, 59, &mut minutes)?;
        Self::parse_field(&text[split + 1..], split + 1, 23, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }

    fn parse_field(
        field: &str,
        offset: usize,
        max: usize,
        selected: &mut [bool],
    ) -> Result<(), CronError> {
        let mut start = 0;
        loop {
            let end = field[start..]
                .find(',')
                .map(|n| start + n)
                .unwrap_or(field.len());
            if start == end {
                return Err(CronError::Syntax(offset + start));
            }
            Self::parse_item(&field[start..end], offset + start, max, selected)?;
            if end == field.len() {
                break;
            }
            start = end + 1;
        }
        Ok(())
    }

    fn parse_item(
        item: &str,
        offset: usize,
        max: usize,
        selected: &mut [bool],
    ) -> Result<(), CronError> {
        let slash = item.find('/');
        if let Some(s) = slash {
            if let Some(second) = item[s + 1..].find('/') {
                return Err(CronError::Syntax(offset + s + 1 + second));
            }
        }

        let base_end = slash.unwrap_or(item.len());
        let base = &item[..base_end];
        let step_start = slash.map(|s| s + 1);

        let is_star = base == "*";
        let dash = base.find('-');
        let is_range = dash.is_some();

        if slash.is_some() && !is_star && !is_range {
            return Err(CronError::Syntax(offset + slash.unwrap()));
        }

        let (lo, hi, dash_pos) = if is_star {
            (0, max, None)
        } else if let Some(d) = dash {
            let lo = Self::parse_number(&base[..d], offset, max)?;
            let hi = Self::parse_number(&base[d + 1..], offset + d + 1, max)?;
            if lo > hi {
                return Err(CronError::Descending(offset + d));
            }
            (lo, hi, Some(d))
        } else {
            let n = Self::parse_number(base, offset, max)?;
            (n, n, None)
        };

        let _ = dash_pos;
        let step = if let Some(s) = step_start {
            let frag = &item[s..];
            let value = Self::parse_number(frag, offset + s, usize::MAX)?;
            let step_max = if max == 59 { 60 } else { 24 };
            if value == 0 || value > step_max {
                return Err(CronError::Step(offset + s));
            }
            value
        } else {
            1
        };

        let mut n = lo;
        while n <= hi {
            selected[n] = true;
            match n.checked_add(step) {
                Some(next) => n = next,
                None => break,
            }
        }
        Ok(())
    }

    fn parse_number(fragment: &str, offset: usize, max: usize) -> Result<usize, CronError> {
        if fragment.is_empty() {
            return Err(CronError::Syntax(offset));
        }
        for (i, b) in fragment.bytes().enumerate() {
            if !b.is_ascii_digit() {
                return Err(CronError::Syntax(offset + i));
            }
        }
        if fragment.len() > 1 && fragment.as_bytes()[0] == b'0' {
            return Err(CronError::Number(offset));
        }

        let mut value = 0usize;
        for b in fragment.bytes() {
            let digit = (b - b'0') as usize;
            value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
                Some(v) => v,
                None => return Err(CronError::Bounds(offset)),
            };
        }
        if value > max {
            return Err(CronError::Bounds(offset));
        }
        Ok(value)
    }
}

    cron: Cron,

    policy: Policy,

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|known| known == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let active_from = self.last_poll.unwrap_or(0);
        let order = self.jobs.len();
        let owned_id = id.to_owned();
        self.ids.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order,
            active_from,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let lower = match lower_exclusive {
                Some(v) => v.max(job.active_from),
                None => job.active_from,
            };
            let first_minute = if lower_exclusive.is_none() && job.active_from == 0 {
                0
            } else {
                lower / 60_000 + 1
            };
            let last_minute = now / 60_000;
            if first_minute > last_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    let mut m = first_minute;
                    while m <= last_minute {
                        if job.cron.matches(m) {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: m * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                        m += 1;
                    }
                }
                Policy::Once => {
                    let mut found = None;
                    let mut m = first_minute;
                    while m <= last_minute {
                        if job.cron.matches(m) {
                            found = Some(m);
                        }
                        m += 1;
                    }
                    if let Some(m) = found {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: m * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let m = now / 60_000;
                        if m >= first_minute && m <= last_minute && job.cron.matches(m) {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
