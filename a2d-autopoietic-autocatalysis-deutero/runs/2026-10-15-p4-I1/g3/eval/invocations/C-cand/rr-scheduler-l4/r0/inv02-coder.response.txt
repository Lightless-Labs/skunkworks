```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct RuntimeJob {
    burst_pos: usize,
    remaining: u32,
    level: usize,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut runtime: Vec<RuntimeJob> = jobs
        .iter()
        .map(|job| RuntimeJob {
            burst_pos: 0,
            remaining: job.bursts[0],
            level: job.level,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completions = vec![0u64; jobs.len()];
    let mut time = 0u64;
    let mut done = 0usize;
    let mut last_dispatch_id: Option<u32> = None;

    while done < jobs.len() {
        admit_events(&mut events, time, &mut queues);

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            let next_time = events.iter().map(|event| event.time).min().unwrap();
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let job_index = queues[level].pop_front().unwrap();
        let job_id = jobs[job_index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != job_id) {
            time += u64::from(switch_cost);
        }
        admit_events(&mut events, time, &mut queues);

        let consumed = runtime[job_index].remaining.min(quanta[level]);
        let start = time;
        time += u64::from(consumed);
        runtime[job_index].remaining -= consumed;

        let burst_finished = runtime[job_index].remaining == 0;
        let final_burst = runtime[job_index].burst_pos + 1 == jobs[job_index].bursts.len();
        let job_finished = burst_finished && final_burst;

        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished: job_finished,
            level,
        });

        admit_events(&mut events, time, &mut queues);

        if job_finished {
            completions[job_index] = time;
            done += 1;
        } else if burst_finished {
            let io_duration = jobs[job_index].bursts[runtime[job_index].burst_pos + 1];
            runtime[job_index].burst_pos += 2;
            runtime[job_index].remaining = jobs[job_index].bursts[runtime[job_index].burst_pos];
            runtime[job_index].level = level.saturating_sub(1);
            events.push(Event {
                time: time + u64::from(io_duration),
                kind: 1,
                index: job_index,
                level: runtime[job_index].level,
            });
            admit_events(&mut events, time, &mut queues);
        } else {
            runtime[job_index].level = (level + 1).min(quanta.len() - 1);
            queues[runtime[job_index].level].push_back(job_index);
        }

        last_dispatch_id = Some(job_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completions[index];
        let turnaround = completion - u64::from(job.arrival);
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&duration| u64::from(duration))
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&duration| u64::from(duration))
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}

fn admit_events(events: &mut Vec<Event>, through: u64, queues: &mut [VecDeque<usize>]) {
    events.sort_by_key(|event| (event.time, event.kind, event.index));
    let count = events.partition_point(|event| event.time <= through);
    for event in events.drain(..count) {
        queues[event.level].push_back(event.index);
    }
}

fn main() {}
```
