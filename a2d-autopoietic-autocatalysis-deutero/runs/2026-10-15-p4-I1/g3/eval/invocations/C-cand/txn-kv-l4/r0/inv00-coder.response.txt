```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    value: Option<String>,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Write {
    value: Option<String>,
    deadline: Option<u64>,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

#[derive(Clone)]
struct SnapshotEntry {
    value: String,
    deadline: Option<u64>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    versions: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, SnapshotEntry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            versions: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn validate_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn validate_key(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn visible_write(&self, tx: Option<u64>, key: &str) -> Option<Write> {
        tx.and_then(|id| self.transactions.get(&id))
            .and_then(|t| t.writes.get(key))
            .cloned()
    }

    fn selected_version(&self, tx: Option<u64>, key: &str) -> Option<&Version> {
        let versions = self.versions.get(key)?;
        match tx {
            None => versions.last(),
            Some(id) => {
                let revision = self.transactions.get(&id)?.begin_revision;
                versions.iter().rev().find(|v| v.revision <= revision)
            }
        }
    }

    fn visible_value(&self, tx: Option<u64>, key: &str, now: u64) -> Option<String> {
        if let Some(write) = self.visible_write(tx, key) {
            return match write.value {
                Some(value) if write.deadline.map_or(true, |d| d > now) => Some(value),
                _ => None,
            };
        }
        let version = self.selected_version(tx, key)?;
        match &version.value {
            Some(value) if version.deadline.map_or(true, |d| d > now) => Some(value.clone()),
            _ => None,
        }
    }

    fn visible_entries(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.versions.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(t) = self.transactions.get(&id) {
                keys.extend(t.writes.keys().cloned());
            }
        }
        let mut result = BTreeMap::new();
        for key in keys {
            if let Some(value) = self.visible_value(tx, &key, now) {
                result.insert(key, value);
            }
        }
        result
    }

    fn append_version(&mut self, key: String, write: Write, revision: u64) {
        self.versions.entry(key).or_default().push(Version {
            revision,
            value: write.value,
            deadline: write.deadline,
        });
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let tx = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;

        if !tx.writes.is_empty() {
            for key in tx.writes.keys() {
                if self
                    .versions
                    .get(key)
                    .and_then(|vs| vs.last())
                    .is_some_and(|v| v.revision > tx.begin_revision)
                {
                    return Err(Error::Conflict {
                        tx: id,
                        key: key.clone(),
                    });
                }
            }
            let next_revision = self.revision + 1;
            let writes = tx.writes.clone();
            for (key, write) in writes {
                self.append_version(key, write, next_revision);
            }
            self.revision = next_revision;
            self.transactions.remove(&id);
            Ok(next_revision)
        } else {
            self.transactions.remove(&id);
            Ok(self.revision)
        }
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let deadline = match ttl {
            Some(d) => Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };
        let write = if ttl == Some(0) {
            Write {
                value: None,
                deadline: None,
            }
        } else {
            Write {
                value: Some(value.to_string()),
                deadline,
            }
        };
        match tx {
            Some(id) => {
                self.transactions
                    .get_mut(&id)
                    .unwrap()
                    .writes
                    .insert(key.to_string(), write);
            }
            None => {
                self.revision += 1;
                self.append_version(key.to_string(), write, self.revision);
            }
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let was_live = self.visible_value(tx, key, now).is_some();
        let write = Write {
            value: None,
            deadline: None,
        };
        match tx {
            Some(id) => {
                self.transactions
                    .get_mut(&id)
                    .unwrap()
                    .writes
                    .insert(key.to_string(), write);
            }
            None => {
                self.revision += 1;
                self.append_version(key.to_string(), write, self.revision);
            }
        }
        Ok(was_live)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_key(key)?;
        Ok(self.visible_value(tx, key, now))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Ok(self
            .visible_entries(tx, now)
            .values()
            .filter(|v| v.as_str() == value)
            .count())
    }

    fn range_matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_range(start, end)?;
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::range_matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut matching: Vec<(String, String)> = self
            .visible_entries(tx, now)
            .into_iter()
            .filter(|(k, _)| Self::range_matches(k, start, end, prefix))
            .filter(|(k, _)| cursor.map_or(true, |c| k > &c.last))
            .collect();

        let skip = if cursor.is_some() { 0 } else { offset };
        let take_from = skip.min(matching.len());
        matching.drain(..take_from);
        let has_more = matching.len() > limit;
        matching.truncate(limit);

        let next = if has_more {
            let last = matching.last().unwrap().0.clone();
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last,
            })
        } else {
            None
        };
        Ok(Page {
            items: matching,
            next,
        })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Self::validate_range(start, end)?;
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.visible_entries(tx, now) {
            if !Self::range_matches(&key, start, end, prefix) {
                continue;
            }
            let parsed = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (i, b) in name.bytes().enumerate() {
            let valid = if i == 0 {
                b.is_ascii_lowercase()
            } else {
                b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'
            };
            if !valid {
                return Err(Error::NameInvalid { at: i });
            }
        }
        Ok(())
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        let exists = self.snapshots.contains_key(name);
        if !exists && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let entries = self
            .visible_entries(None, now)
            .into_iter()
            .filter_map(|key_value| {
                let key = key_value.0;
                let version = self.versions.get(&key)?.last()?;
                Some((
                    key,
                    SnapshotEntry {
                        value: key_value.1,
                        deadline: version.deadline,
                    },
                ))
            })
            .collect();
        self.snapshots.insert(name.to_string(), entries);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let saved_live: BTreeMap<String, Write> = saved
            .into_iter()
            .filter(|(_, entry)| entry.deadline.map_or(true, |d| d > now))
            .map(|(key, entry)| {
                (
                    key,
                    Write {
                        value: Some(entry.value),
                        deadline: entry.deadline,
                    },
                )
            })
            .collect();

        let current = self.visible_entries(None, now);
        let mut union: BTreeSet<String> = current.keys().cloned().collect();
        union.extend(saved_live.keys().cloned());
        if union.is_empty() {
            return Ok(self.revision);
        }

        let next_revision = self.revision + 1;
        for key in union {
            let write = saved_live.get(&key).cloned().unwrap_or(Write {
                value: None,
                deadline: None,
            });
            self.append_version(key, write, next_revision);
        }
        self.revision = next_revision;
        Ok(next_revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0usize;
        for versions in self.versions.values_mut() {
            if versions.len() <= 1 {
                continue;
            }
            let mut keep = BTreeSet::new();
            keep.insert(versions.len() - 1);
            for tx in self.transactions.values() {
                if let Some(index) = versions
                    .iter()
                    .rposition(|v| v.revision <= tx.begin_revision)
                {
                    keep.insert(index);
                }
            }
            let old_len = versions.len();
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                retain
            });
            removed += old_len - versions.len();
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let live_keys = self.visible_entries(None, now).len();
        let versions = self.versions.values().map(Vec::len).sum();
        let tombstones = self
            .versions
            .values()
            .flat_map(|vs| vs.iter())
            .filter(|v| v.value.is_none())
            .count();
        let read_only_transactions = self
            .transactions
            .values()
            .filter(|t| t.read_only)
            .count();
        let staged_writes = self
            .transactions
            .values()
            .map(|t| t.writes.len())
            .sum();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.versions.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}

fn parse_canonical_i64(s: &str) -> Option<i64> {
    if s == "0" {
        return Some(0);
    }
    let digits = if let Some(rest) = s.strip_prefix('-') {
        if rest.is_empty() || rest == "0" {
            return None;
        }
        rest
    } else {
        s
    };
    if digits.is_empty()
        || !digits.bytes().all(|b| b.is_ascii_digit())
        || digits.starts_with('0')
    {
        return None;
    }
    s.parse::<i64>().ok()
}

fn main() {}
```
