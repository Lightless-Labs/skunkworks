#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

#[derive(Clone)]
struct Runtime {
    remaining: u64,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
    started: bool,
}

#[derive(Clone)]
struct Active {
    idx: usize,
    dispatch: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: usize, b: usize, customers: &[Customer], rt: &[Runtime]) -> std::cmp::Ordering {
    let ca = &customers[a];
    let cb = &customers[b];
    ca.priority
        .cmp(&cb.priority)
        .then_with(|| match (rt[a].paused, rt[b].paused) {
            (true, false) => std::cmp::Ordering::Less,
            (false, true) => std::cmp::Ordering::Greater,
            (true, true) => rt[a]
                .pause_time
                .cmp(&rt[b].pause_time)
                .then_with(|| ca.id.cmp(&cb.id)),
            (false, false) => ca
                .arrival
                .cmp(&cb.arrival)
                .then_with(|| cb.id.cmp(&ca.id)),
        })
}

fn best_waiter(waiting: &[usize], customers: &[Customer], rt: &[Runtime]) -> Option<usize> {
    waiting
        .iter()
        .copied()
        .min_by(|a, b| waiter_cmp(*a, *b, customers, rt))
}

fn add_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId {
                index: i,
                id: c.id,
            });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index: i,
                value: c.priority,
            });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow {
                    index: i,
                    id: c.id,
                });
            }
        }
    }

    let n = customers.len();
    let mut rt: Vec<Runtime> = customers
        .iter()
        .map(|c| Runtime {
            remaining: c.service,
            entry: c.arrival,
            deadline: c.patience.map(|p| c.arrival + p),
            waited: 0,
            paused: false,
            pause_time: 0,
            started: false,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..n).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatch: Option<u32> = None;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();
    let mut now = customers
        .iter()
        .map(|c| c.arrival)
        .min()
        .unwrap_or(0);

    loop {
        // Advance to the next event time when no immediate work remains.
        if active.is_none() && waiting.is_empty() && next_arrival < n {
            now = now.max(customers[arrivals[next_arrival]].arrival);
        }

        // Phase A: finish the active job at its scheduled end.
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let idx = a.idx;
                active = None;
                add_event(&mut events, now, customers[idx].id, Kind::Finish);
                completed += 1;
                turnaround_ticks += (now - customers[idx].arrival) as u128;
                completion_order.push(customers[idx].id);
            }
        }

        // Phase B: expire waiters present at the start of the tick.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| rt[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            rt[a].deadline
                .cmp(&rt[b].deadline)
                .then_with(|| customers[a].id.cmp(&customers[b].id))
        });
        for i in expired {
            waiting.retain(|&x| x != i);
            let segment = now - rt[i].entry;
            rt[i].waited = rt[i].waited.saturating_add(segment);
            wait_ticks += segment as u128;
            add_event(
                &mut events,
                now,
                customers[i].id,
                Kind::Renege { waited: rt[i].waited },
            );
            reneged += 1;
        }

        // Phase C: process the full arrival cohort.
        while next_arrival < n && customers[arrivals[next_arrival]].arrival == now {
            let i = arrivals[next_arrival];
            next_arrival += 1;
            add_event(&mut events, now, customers[i].id, Kind::Arrive);
            let idle_extra = usize::from(active.is_none());
            let gate = config.waiting_capacity.saturating_add(idle_extra);
            if waiting.len() >= gate {
                add_event(&mut events, now, customers[i].id, Kind::Balk);
                balked += 1;
            } else {
                waiting.push(i);
                accepted += 1;
            }
        }

        // Phase D: preempt only after switching protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, customers, &rt) {
                    if customers[w].priority < customers[a.idx].priority {
                        let old = a.idx;
                        let remaining = a.finish - now;
                        let by = customers[w].id;
                        let old_active = active.take().unwrap();
                        busy_ticks += (now - old_active.dispatch) as u128;
                        add_event(
                            &mut events,
                            now,
                            customers[old].id,
                            Kind::Preempt { remaining, by },
                        );
                        preemptions += 1;
                        rt[old].remaining = remaining;
                        rt[old].entry = now;
                        rt[old].paused = true;
                        rt[old].pause_time = now;
                        if let Some(budget) = rt[old]
                            .deadline
                            .map(|d| d.saturating_sub(old_active.dispatch))
                        {
                            // Recover the unconsumed budget from the prior deadline and elapsed
                            // active time; the prior waiting segment was consumed at dispatch.
                            let elapsed_active = now - old_active.dispatch;
                            let budget_left = budget.saturating_add(elapsed_active);
                            rt[old].deadline = Some(now.checked_add(budget_left).ok_or(
                                Error::TimeOverflow {
                                    customer: customers[old].id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?);
                        }
                        waiting.push(old);
                    }
                }
            }
        }

        // Phase E: repeatedly dispatch and inline-finish zero-length work.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, customers, &rt) else {
                break;
            };
            waiting.retain(|&x| x != i);

            let segment = now - rt[i].entry;
            rt[i].waited = rt[i].waited.saturating_add(segment);
            wait_ticks += segment as u128;
            if let Some(d) = rt[i].deadline {
                let budget = d.saturating_sub(rt[i].entry);
                let remaining_budget = budget.saturating_sub(segment);
                rt[i].deadline = Some(now.saturating_add(remaining_budget));
            }

            let switching = match last_dispatch {
                None => 0,
                Some(id) if id == customers[i].id => 0,
                Some(_) => config.switch_cost,
            };
            let work_at = now.checked_add(switching).ok_or(Error::TimeOverflow {
                customer: customers[i].id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(rt[i].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: customers[i].id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let was_started = rt[i].started;
            if !was_started {
                rt[i].started = true;
                started += 1;
                add_event(
                    &mut events,
                    now,
                    customers[i].id,
                    Kind::Start {
                        remaining: rt[i].remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                add_event(
                    &mut events,
                    now,
                    customers[i].id,
                    Kind::Resume {
                        remaining: rt[i].remaining,
                        work_at,
                        finish,
                    },
                );
            }
            last_dispatch = Some(customers[i].id);
            active = Some(Active {
                idx: i,
                dispatch: now,
                work_at,
                finish,
            });

            if finish == now {
                active = None;
                add_event(&mut events, now, customers[i].id, Kind::Finish);
                completed += 1;
                turnaround_ticks += (now - customers[i].arrival) as u128;
                completion_order.push(customers[i].id);
                continue;
            }
            break;
        }

        // Phase F: expire anything newly queued at this tick.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| rt[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            rt[a].deadline
                .cmp(&rt[b].deadline)
                .then_with(|| customers[a].id.cmp(&customers[b].id))
        });
        for i in expired {
            waiting.retain(|&x| x != i);
            let segment = now - rt[i].entry;
            rt[i].waited = rt[i].waited.saturating_add(segment);
            wait_ticks += segment as u128;
            add_event(
                &mut events,
                now,
                customers[i].id,
                Kind::Renege { waited: rt[i].waited },
            );
            reneged += 1;
        }

        if completed + reneged + balked == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < n {
            next_time = Some(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            next_time = Some(next_time.map_or(a.finish, |x| x.min(a.finish)));
            if a.work_at > now {
                next_time = Some(next_time.map_or(a.work_at, |x| x.min(a.work_at)));
            }
        }
        for &i in &waiting {
            if let Some(d) = rt[i].deadline {
                if d > now {
                    next_time = Some(next_time.map_or(d, |x| x.min(d)));
                }
            }
        }
        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap_or(0)
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap_or(0)
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10_000, elapsed as u128)
            .unwrap_or(0)
            .min(u32::MAX as u128) as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

fn main() {}
