        for (position, &value) in task.deps.iter().enumerate() {

                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

            count: worker_count,
        });

            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });

            if need > cap {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: cap,
                });
            }
        }
    }

    let mut indegree = vec![0usize; n];

        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    let mut available = Vec::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut kahn_order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        kahn_order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }

            removed[id] = true;

                blocked.push(id);

}

fn append_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task_id: usize,
    worker_id: usize,
    used: &[u64],
) {
    trace.push_str(&time.to_string());
    trace.push('|');
    trace.push_str(kind);
    trace.push('|');
    trace.push_str(&task_id.to_string());
    trace.push('|');
    trace.push_str(&worker_id.to_string());
    trace.push('|');
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index > 0 {
                trace.push(',');
            }
            trace.push_str(&value.to_string());
        }
    }
    trace.push('\n');
}
