use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method == method
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
                && same_shape(&route.segments, &segments)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        let rank = make_rank(&segments);
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_request_path(head)?;

        let mut candidates = Vec::new();
        for route in &self.routes {
            if let Some(params) = match_segments(&route.segments, &parsed.segments) {
                let slash_ok = self.policy != SlashPolicy::Strict || route.trailing == parsed.trailing;
                if slash_ok {
                    candidates.push((route, params));
                }
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<(&Route, Vec<(String, String)>)> = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                candidates
                    .into_iter()
                    .filter(|(r, _)| r.method == "HEAD")
                    .collect()
            } else {
                candidates
                    .into_iter()
                    .filter(|(r, _)| r.method == "GET")
                    .collect()
            }
        } else {
            candidates
                .into_iter()
                .filter(|(r, _)| r.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = self
                .routes
                .iter()
                .filter(|route| {
                    match_segments(&route.segments, &parsed.segments).is_some()
                        && (self.policy != SlashPolicy::Strict
                            || route.trailing == parsed.trailing)
                })
                .map(|route| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                a.rank
                    .cmp(&b.rank)
                    .then_with(|| b.id.cmp(&a.id))
            })
            .unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect
            && winner.trailing != parsed.trailing
            && !parsed.segments.is_empty()
        {
            let mut target = head.to_owned();
            if winner.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut values: HashMap<&str, &str> = HashMap::new();
        for (key, value) in params {
            if values.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let capture_names: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &capture_names {
            if !values.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extras: Vec<&str> = values
            .keys()
            .copied()
            .filter(|key| !capture_names.contains(key))
            .collect();
        extras.sort();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = values[name.as_str()];
                    if !valid_param_value(value) {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = values[name.as_str()];
                    if !valid_wildcard_value(value) {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if !first {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(value));
                    first = false;
                }
                Segment::Param(name) => {
                    if !first {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(values[name.as_str()]));
                    first = false;
                }
                Segment::Wildcard(name) => {
                    let value = values[name.as_str()];
                    if !value.is_empty() {
                        if !first {
                            output.push('/');
                        }
                        for (i, component) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(component));
                        }
                        first = false;
                    }
                }
            }
        }

        if route.trailing && !route.segments.is_empty() && !output.ends_with('/') {
            output.push('/');
        }
        Ok(output)
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = pattern.as_bytes();
    let mut end = bytes.len();
    let trailing = bytes.last() == Some(&b'/');
    if trailing {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut seen = HashSet::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|b| *b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);

        if segment_end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..segment_end];
        if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.starts_with('*');
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && segment_end != end {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start, true)?;
            segments.push(Segment::Static(decoded));
        }

        if segment_end == end {
            break;
        }
        start = segment_end + 1;
        if start == end {
            return Err(Error::InvalidPattern { at: start });
        }
    }

    Ok((segments, trailing))
}

fn parse_request_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|b| *b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);

        if segment_end == start {
            return Err(Error::InvalidPath { at: start });
        }
        let decoded = decode_segment(&path[start..segment_end], start, false)?;
        segments.push(decoded);

        if segment_end == end {
            break;
        }
        start = segment_end + 1;
        if start == end {
            return Err(Error::InvalidPath { at: start });
        }
    }

    Ok(ParsedPath { segments, trailing })
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            let decoded = match (hi, lo) {
                (Some(h), Some(l)) => h * 16 + l,
                _ => {
                    return Err(if pattern {
                        Error::InvalidPattern { at: offset + i }
                    } else {
                        Error::InvalidPath { at: offset + i }
                    });
                }
            };
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: offset + i }
            } else {
                Error::InvalidPath { at: offset + i }
            });
        }
    }

    Ok(String::from_utf8(out).unwrap())
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    if a.len() != b.len() {
        return false;
    }
    a.iter().zip(b).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn match_segments(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = pattern
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));

    let fixed_count = wildcard_index.unwrap_or(pattern.len());
    if wildcard_index.is_none() && request.len() != pattern.len() {
        return None;
    }
    if wildcard_index.is_some() && request.len() < fixed_count {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_count {
        match &pattern[i] {
            Segment::Static(value) if value != &request[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), request[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_index {
        if let Segment::Wildcard(name) = &pattern[index] {
            params.push((name.clone(), request[index..].join("/")));
        }
    }
    Some(params)
}

fn valid_param_value(value: &str) -> bool {
    !value.is_empty()
        && value
            .bytes()
            .all(|b| (0x20..=0x7e).contains(&b) && b != b'/')
}

fn valid_wildcard_value(value: &str) -> bool {
    if value.is_empty() {
        return true;
    }
    value
        .split('/')
        .all(|part| !part.is_empty() && part.bytes().all(|b| (0x20..=0x7e).contains(&b)))
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}
