#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                append_full_group(
                    &mut output,
                    self.tail[0],
                    self.tail[1],
                    input[index - 1],
                );
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while input.len() - index >= 3 {
            append_full_group(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        let remaining = input.len() - index;
        self.tail_len = remaining;
        if remaining > 0 {
            self.tail[0] = input[index];
        }
        if remaining > 1 {
            self.tail[1] = input[index + 1];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn append_full_group(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub struct Decoder {
    values: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    next_offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    sticky_error: Option<Error>,
    finished: bool,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            values: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            next_offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            sticky_error: None,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset = self.next_offset.saturating_add(1);

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset });
            }

            if self.pending_second_padding.is_some() {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset });
                }
                self.pending_second_padding = None;
                self.group_len = 0;
                self.terminal_padding = true;
                output.push((self.values[0] << 2) | (self.values[1] >> 4));
                continue;
            }

            if byte == b'=' {
                match self.group_len {
                    0 | 1 => return self.fail(Error::BadPadding { offset }),
                    2 => {
                        self.pending_second_padding = Some(offset);
                    }
                    3 => {
                        self.group_len = 0;
                        self.terminal_padding = true;
                        output.push((self.values[0] << 2) | (self.values[1] >> 4));
                        output.push((self.values[1] << 4) | (self.values[2] >> 2));
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let Some(value) = digit_value(byte) else {
                return self.fail(Error::InvalidByte { offset, byte });
            };

            self.values[self.group_len] = value;
            self.offsets[self.group_len] = offset;
            self.group_len += 1;

            if self.group_len == 4 {
                output.push((self.values[0] << 2) | (self.values[1] >> 4));
                output.push((self.values[1] << 4) | (self.values[2] >> 2));
                output.push((self.values[2] << 6) | self.values[3]);
                self.group_len = 0;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if self.group_len != 0 {
            return self.fail(Error::Truncated {
                offset: self.offsets[0],
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}
