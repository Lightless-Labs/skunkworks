#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| b.arrival.cmp(&a.arrival).then_with(|| a.id.cmp(&b.id)));

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut now = 0u64;
    let mut current: Option<(Customer, u64)> = None;
    let mut previous_finish: Option<u64> = None;

    loop {
        let next_time = match (current.as_ref().map(|(_, finish)| *finish), arrivals.get(next_arrival)) {
            (Some(finish), Some(customer)) => Some(finish.min(customer.arrival)),
            (Some(finish), None) => Some(finish),
            (None, Some(customer)) => Some(customer.arrival),
            (None, None) => None,
        };
        let Some(time) = next_time else {
            break;
        };
        now = time;

        if current.as_ref().is_some_and(|(_, finish)| *finish == now) {
            let (customer, _) = current.take().unwrap();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Finish,
            });
            previous_finish = Some(now);
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if current.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = previous_finish.map_or(true, |finish| now > finish);
                let finish = now
                    .checked_add(customer.service)
                    .and_then(|t| t.checked_add(if wake { wake_cost } else { 0 }))
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    })?;

                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == now {
                    events.push(Event {
                        time: now,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(now);
                } else {
                    current = Some((customer, finish));
                }
            }
        }

        // A zero-duration completion can leave more customers ready at this same tick.
        while current.is_none() && !waiting.is_empty() {
            let customer = waiting.pop_front().unwrap();
            let wake = previous_finish.map_or(true, |finish| now > finish);
            let finish = now
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(if wake { wake_cost } else { 0 }))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time: now,
                })?;

            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Start { finish, wake },
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                previous_finish = Some(now);
            } else {
                current = Some((customer, finish));
            }
        }
    }

    Ok(events)
}

fn main() {}
