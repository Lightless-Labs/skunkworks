#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { id, segments });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let segments = parse_path(head)?;

        let mut best: Option<(&Route, Vec<u8>, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if route.segments.len() != segments.len() {
                continue;
            }

            let mut rank = Vec::with_capacity(segments.len());
            let mut params = Vec::new();
            let mut matched = true;

            for (pattern_segment, request_segment) in route.segments.iter().zip(&segments) {
                match pattern_segment {
                    Segment::Static(value) => {
                        rank.push(2);
                        if value != request_segment {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        rank.push(1);
                        params.push((name.clone(), request_segment.clone()));
                    }
                }
            }

            if !matched {
                continue;
            }

            let replace = best.as_ref().map_or(true, |(_, best_rank, _)| rank > *best_rank);
            if replace {
                best = Some((route, rank, params));
            }
        }

        match best {
            Some((route, _, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if bytes.len() == 1 {
        return Ok(Vec::new());
    }

    let end = if bytes.last() == Some(&b'/') {
        bytes.len() - 1
    } else {
        bytes.len()
    };

    let mut result = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);

        if stop == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &bytes[start..stop];
        if raw[0] == b':' {
            let name_bytes = &raw[1..];
            let valid = !name_bytes.is_empty()
                && name_bytes[0].is_ascii_lowercase()
                && name_bytes
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');

            if !valid {
                return Err(Error::InvalidPattern { at: start });
            }

            let name = std::str::from_utf8(name_bytes).unwrap().to_owned();
            if !seen.insert(name.clone()) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(Segment::Param(name));
        } else if raw[0] == b'*' {
            return Err(Error::InvalidPattern { at: start });
        } else {
            let decoded = decode_static_segment(raw, start, true)?;
            result.push(Segment::Static(decoded));
        }

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }

    Ok(result)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if bytes.len() == 1 {
        return Ok(Vec::new());
    }

    let end = if bytes.last() == Some(&b'/') {
        bytes.len() - 1
    } else {
        bytes.len()
    };

    let mut result = Vec::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);

        if stop == start {
            return Err(Error::InvalidPath { at: start });
        }

        result.push(decode_static_segment(&bytes[start..stop], start, false)?);

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPath { at: bytes.len() - 1 });
    }

    Ok(result)
}

fn decode_static_segment(
    raw: &[u8],
    base: usize,
    pattern: bool,
) -> Result<String, Error> {
    let mut decoded = Vec::with_capacity(raw.len());
    let mut i = 0;

    while i < raw.len() {
        let byte = raw[i];
        if byte == b'%' {
            if i + 2 >= raw.len()
                || !is_upper_hex(raw[i + 1])
                || !is_upper_hex(raw[i + 2])
            {
                return invalid_segment(pattern, base + i);
            }

            let value = (hex_value(raw[i + 1]) << 4) | hex_value(raw[i + 2]);
            let unreserved = is_unreserved(value);
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved {
                return invalid_segment(pattern, base + i);
            }

            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return invalid_segment(pattern, base + i);
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn invalid_segment<T>(pattern: bool, at: usize) -> Result<T, Error> {
    if pattern {
        Err(Error::InvalidPattern { at })
    } else {
        Err(Error::InvalidPath { at })
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

fn main() {}
