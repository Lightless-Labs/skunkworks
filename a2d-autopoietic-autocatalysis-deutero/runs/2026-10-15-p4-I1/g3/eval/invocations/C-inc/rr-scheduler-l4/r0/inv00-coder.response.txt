#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

use std::collections::VecDeque;

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

#[derive(Clone, Copy)]
struct Ready {
    job: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut queues: Vec<VecDeque<Ready>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut burst_index = vec![0usize; jobs.len()];
    let mut remaining = jobs
        .iter()
        .map(|job| u64::from(job.bursts[0]))
        .collect::<Vec<_>>();
    let mut levels = jobs.iter().map(|job| job.level).collect::<Vec<_>>();
    let mut completion = vec![0u64; jobs.len()];
    let mut done = 0usize;
    let mut event_pos = 0usize;
    let mut now = 0u64;
    let mut last_id: Option<u32> = None;

    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let admit_through = |limit: u64,
                         event_pos: &mut usize,
                         events: &[Event],
                         queues: &mut Vec<VecDeque<Ready>>,
                         levels: &[usize]| {
        while *event_pos < events.len() && events[*event_pos].time <= limit {
            let event = events[*event_pos];
            queues[levels[event.index]].push_back(Ready { job: event.index });
            *event_pos += 1;
        }
    };

    while done < jobs.len() {
        admit_through(now, &mut event_pos, &events, &mut queues, &levels);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_pos >= events.len() {
                break;
            }
            let next_time = events[event_pos].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_id = None;
            }
            continue;
        };

        let selected = queues[level].pop_front().unwrap();
        let job_index = selected.job;
        let job = &jobs[job_index];

        if last_id.is_some_and(|id| id != job.id) {
            now += u64::from(switch_cost);
        }
        admit_through(now, &mut event_pos, &events, &mut queues, &levels);

        let start = now;
        let consumed = remaining[job_index].min(u64::from(quanta[level]));
        now += consumed;
        let end = now;
        remaining[job_index] -= consumed;

        let burst_finished = remaining[job_index] == 0;
        let final_burst = burst_finished && burst_index[job_index] + 1 == job.bursts.len();
        runs.push(Run {
            id: job.id,
            start,
            end,
            finished: final_burst,
            level,
        });

        if final_burst {
            completion[job_index] = end;
            done += 1;
        } else if burst_finished {
            burst_index[job_index] += 2;
            remaining[job_index] = u64::from(job.bursts[burst_index[job_index]]);
            let io_delay = u64::from(job.bursts[burst_index[job_index] - 1]);
            let wake_time = end + io_delay;
            levels[job_index] = level.saturating_sub(1);
            events.push(Event {
                time: wake_time,
                kind: 1,
                index: job_index,
            });
            events[event_pos..].sort_by_key(|event| (event.time, event.kind, event.index));
        }

        admit_through(end, &mut event_pos, &events, &mut queues, &levels);

        if !burst_finished {
            levels[job_index] = (level + 1).min(quanta.len() - 1);
            queues[levels[job_index]].push_back(Ready { job: job_index });
        }

        last_id = Some(job.id);
    }

    let stats = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let turnaround = completion[index] - u64::from(job.arrival);
            let total_work: u64 = job.bursts.iter().map(|&value| u64::from(value)).sum();
            Stat {
                id: job.id,
                completion: completion[index],
                turnaround,
                waiting: turnaround - total_work,
            }
        })
        .collect::<Vec<_>>();

    let count = jobs.len() as u64;
    let total_waiting: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        end: completion.iter().copied().max().unwrap_or(0),
        runs,
        idle,
        mean_waiting_x100: (total_waiting * 100 + count / 2) / count,
        mean_turnaround_x100: (total_turnaround * 100 + count / 2) / count,
        stats,
    })
}

fn main() {}
