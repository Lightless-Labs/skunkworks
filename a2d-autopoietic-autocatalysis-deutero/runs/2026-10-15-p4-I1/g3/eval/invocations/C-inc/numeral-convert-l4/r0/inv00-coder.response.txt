#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(&str, u128); 6] = [
    ("thousand", 1_000),
    ("million", 1_000_000),
    ("billion", 1_000_000_000),
    ("trillion", 1_000_000_000_000),
    ("quadrillion", 1_000_000_000_000_000),
    ("quintillion", 1_000_000_000_000_000_000),
];

fn encode_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }
    if n < 100 {
        let tens = TENS[(n / 10 - 2) as usize];
        let units = n % 10;
        if units == 0 {
            tens.to_string()
        } else {
            format!("{tens}-{}", SMALL[units as usize])
        }
    } else {
        let hundreds = format!("{} hundred", SMALL[(n / 100) as usize]);
        let remainder = n % 100;
        if remainder == 0 {
            hundreds
        } else {
            format!("{hundreds} and {}", encode_under_thousand(remainder))
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    while remaining != 0 {
        groups.push((remaining % 1_000) as u16);
        remaining /= 1_000;
    }

    let mut parts = Vec::new();
    for (index, &group) in groups.iter().enumerate().rev() {
        if group == 0 {
            continue;
        }
        let mut part = encode_under_thousand(group);
        if index > 0 {
            part.push(' ');
            part.push_str(SCALES[index - 1].0);
        }
        parts.push((index, group, part));
    }

    let units_and = parts.len() > 1
        && parts.last().is_some_and(|(index, group, _)| *index == 0 && *group < 100);

    let mut result = String::new();
    for (i, (_, _, part)) in parts.iter().enumerate() {
        if i > 0 {
            if units_and && i + 1 == parts.len() {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        result.push_str(part);
    }

    if value < 0 {
        Ok(format!("minus {result}"))
    } else {
        Ok(result)
    }
}

#[derive(Clone, Copy)]
enum TokenKind {
    Zero,
    Add(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

fn dictionary_word(word: &str) -> Option<TokenKind> {
    if word == "zero" {
        return Some(TokenKind::Zero);
    }
    for (value, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(TokenKind::Add(value as u128));
        }
    }
    for (index, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(TokenKind::Add(((index + 2) * 10) as u128));
        }
    }
    if word == "hundred" {
        return Some(TokenKind::Hundred);
    }
    for (spelling, value) in SCALES {
        if word == spelling {
            return Some(TokenKind::Scale(value));
        }
    }
    if word == "and" {
        return Some(TokenKind::And);
    }
    if word == "minus" {
        return Some(TokenKind::Minus);
    }
    None
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    if let Some(index) = bytes
        .iter()
        .position(|&b| !(b.is_ascii_lowercase() || b == b' ' || b == b'-'))
    {
        return Err(WordsError::InvalidByte { index });
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || bytes[index - 1] == b' '
                || bytes[index - 1] == b'-'
                || bytes[index + 1] == b' '
                || bytes[index + 1] == b'-'
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize, TokenKind)> = Vec::new();
    let mut start = 0;
    for end in 0..=bytes.len() {
        if end == bytes.len() || bytes[end] == b' ' || bytes[end] == b'-' {
            let word = &input[start..end];
            let Some(kind) = dictionary_word(word) else {
                return Err(WordsError::InvalidWord { index: start });
            };
            tokens.push((word, start, kind));
            start = end + 1;
        }
    }

    let negative = matches!(tokens.first(), Some((_, _, TokenKind::Minus)));
    for (i, (_, index, kind)) in tokens.iter().enumerate() {
        if matches!(kind, TokenKind::Minus) && i != 0 {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let arithmetic_start = usize::from(negative);
    let mut total = 0u128;
    let mut group = 0u128;

    for (_, _, kind) in tokens.iter().skip(arithmetic_start) {
        match kind {
            TokenKind::Zero | TokenKind::Minus => {}
            TokenKind::Add(value) => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
            TokenKind::And => {}
        }

        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(bytes)
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }

    Ok(value)
}

fn main() {}
