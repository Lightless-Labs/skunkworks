#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8]) -> String {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    let mut out = String::with_capacity(4);
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        output.push_str(&encode_group(chunk));
    }

    if !chunks.remainder().is_empty() {
        output.push_str(&encode_group(chunks.remainder()));
    }

    output
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.tail.len() + input.len());
        data.extend_from_slice(&self.tail);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut output = String::with_capacity((complete_len / 3) * 4);

        for chunk in data[..complete_len].chunks_exact(3) {
            output.push_str(&encode_group(chunk));
        }

        self.tail.clear();
        self.tail.extend_from_slice(&data[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        self.finished = true;
        let output = if self.tail.is_empty() {
            String::new()
        } else {
            encode_group(&self.tail)
        };
        self.tail.clear();
        Ok(output)
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    value: u8,
    offset: usize,
}

pub struct Decoder {
    group: Vec<Symbol>,
    absolute_offset: usize,
    terminal_padding: bool,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            absolute_offset: 0,
            terminal_padding: false,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.absolute_offset;
            self.absolute_offset += 1;

            if self.terminal_padding {
                return Err(self.fail(Error::BadPadding { offset }));
            }

            if self.group.len() == 3 && self.group[2].byte == b'=' {
                if byte != b'=' {
                    return Err(self.fail(Error::BadPadding { offset }));
                }
                self.group.push(Symbol {
                    byte,
                    value: 0,
                    offset,
                });
                output.push((self.group[0].value << 2) | (self.group[1].value >> 4));
                self.group.clear();
                self.terminal_padding = true;
                continue;
            }

            let position = self.group.len();
            if byte == b'=' {
                match position {
                    0 | 1 => return Err(self.fail(Error::BadPadding { offset })),
                    2 => {
                        self.group.push(Symbol {
                            byte,
                            value: 0,
                            offset,
                        });
                    }
                    3 => {
                        let a = self.group[0].value;
                        let b = self.group[1].value;
                        let c = self.group[2].value;
                        output.push((a << 2) | (b >> 4));
                        output.push((b << 4) | (c >> 2));
                        self.group.clear();
                        self.terminal_padding = true;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let value = match digit(byte) {
                Some(value) => value,
                None => {
                    return Err(self.fail(Error::InvalidByte { offset, byte }));
                }
            };

            self.group.push(Symbol {
                byte,
                value,
                offset,
            });

            if self.group.len() == 4 {
                let a = self.group[0].value;
                let b = self.group[1].value;
                let c = self.group[2].value;
                let d = self.group[3].value;
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
                self.group.clear();
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if self.group.len() == 3 && self.group[2].byte == b'=' {
            let offset = self.group[2].offset;
            return Err(self.fail(Error::BadPadding { offset }));
        }

        if !self.group.is_empty() {
            let offset = self.group[0].offset;
            return Err(self.fail(Error::Truncated { offset }));
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

fn main() {}
