#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes
        .iter()
        .any(|b| !b.is_ascii_lowercase() && !b.is_ascii_digit() && *b != b'-')
    {
        return false;
    }
    bytes.last() != Some(&b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(version) => format!("={}", version_text(version)),
        Constraint::AtLeast(version) => format!(">={}", version_text(version)),
        Constraint::Before(version) => format!("<{}", version_text(version)),
        Constraint::Caret(version) => format!("^{}", version_text(version)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut output = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            output.push(*byte as char);
        } else {
            output.push_str(&format!("%{:02X}", byte));
        }
    }
    output
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(required) => version == required,
        Constraint::AtLeast(required) => version >= required,
        Constraint::Before(required) => version < required,
        Constraint::Caret(required) => {
            version >= required
                && if required.major > 0 {
                    version.major == required.major
                } else if required.minor > 0 {
                    version.major == 0 && version.minor == required.minor
                } else {
                    version == required
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    package_indices: Vec<usize>,
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, Vec<Selection>, Vec<(String, Vec<Constraint>)>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn package_index(&self, name: &str) -> Option<usize> {
        self.package_indices
            .binary_search_by(|index| self.catalog[*index].name.as_str().cmp(name))
            .ok()
    }

    fn inspect(&mut self, selected: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

        fn add_constraint(
            active: &mut Vec<(String, Vec<Constraint>)>,
            package: &str,
            constraint: Constraint,
        ) {
            match active.binary_search_by(|(name, _)| name.as_str().cmp(package)) {
                Ok(index) => active[index].1.push(constraint),
                Err(index) => active.insert(index, (package.to_string(), vec![constraint])),
            }
        }

        for root in self.roots {
            add_constraint(&mut active, &root.package, root.constraint);
        }

        for (name, version) in selected.iter() {
            if let Some(index) = self.package_index(name) {
                if let Some(release) = self.catalog[index]
                    .releases
                    .iter()
                    .find(|release| release.version == *version)
                {
                    for dependency in &release.deps {
                        add_constraint(
                            &mut active,
                            &dependency.package,
                            dependency.constraint,
                        );
                    }
                }
            }
        }

        for (_, constraints) in &mut active {
            *constraints = canonical(std::mem::take(constraints));
        }

        if let Some((name, _)) = active
            .iter()
            .find(|(name, _)| self.package_index(name).is_none())
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        selected.sort_by(|left, right| left.0.cmp(&right.0));
        for (name, version) in selected.iter() {
            if let Ok(index) = active.binary_search_by(|(active_name, _)| active_name.cmp(name)) {
                let constraints = &active[index].1;
                if !matches_all(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.binary_search_by(|(selected_name, _)| selected_name.cmp(name)).is_err() {
                let index = self.package_index(name).unwrap();
                if !self.catalog[index]
                    .releases
                    .iter()
                    .any(|release| matches_all(release.version, constraints))
                {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let next = active
            .iter()
            .find(|(name, _)| {
                selected
                    .binary_search_by(|(selected_name, _)| selected_name.cmp(name))
                    .is_err()
            })
            .cloned();

        if let Some((name, constraints)) = next {
            let index = self.package_index(&name).unwrap();
            let mut candidates: Vec<Version> = self.catalog[index]
                .releases
                .iter()
                .map(|release| release.version)
                .filter(|version| matches_all(*version, &constraints))
                .collect();
            candidates.sort_by(|left, right| right.cmp(left));

            if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
                if let Some(position) = candidates.iter().position(|version| *version == lock.version)
                {
                    let locked = candidates.remove(position);
                    candidates.insert(0, locked);
                }
            }

            for candidate in candidates {
                if self.attempts == self.max_attempts {
                    return Err(ResolveError::Limit {
                        attempts: self.max_attempts,
                    });
                }
                self.attempts += 1;
                selected.push((name.clone(), candidate));
                match self.inspect(selected) {
                    Ok(()) => {}
                    Err(ResolveError::Limit { attempts }) => {
                        selected.pop();
                        return Err(ResolveError::Limit { attempts });
                    }
                    Err(error) => {
                        if self.first_error.is_none() {
                            self.first_error = Some(error);
                        }
                        selected.pop();
                    }
                }
            }
            Ok(())
        } else {
            let selections: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            self.record_solution(selections, active);
            Ok(())
        }
    }

    fn record_solution(
        &mut self,
        selections: Vec<Selection>,
        active: Vec<(String, Vec<Constraint>)>,
    ) {
        let cost = self
            .locks
            .iter()
            .filter(|lock| {
                !selections
                    .iter()
                    .any(|selection| {
                        selection.package == lock.package && selection.version == lock.version
                    })
            })
            .count();

        let vector: Vec<Option<Version>> = self
            .package_indices
            .iter()
            .map(|index| {
                let name = &self.catalog[*index].name;
                selections
                    .iter()
                    .find(|selection| selection.package == *name)
                    .map(|selection| selection.version)
            })
            .collect();

        let better = match &self.best {
            None => true,
            Some((best_cost, best_vector, _, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };

        if better {
            self.best = Some((cost, vector, selections, active));
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", index),
                name: root.package.clone(),
            });
        }
    }

    for (package_index, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", package_index),
                name: package.name.clone(),
            });
        }
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!(
                            "catalog[{}].release[{}].dep[{}]",
                            package_index, release_index, dependency_index
                        ),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (index, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", index),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut package_indices: Vec<usize> = (0..catalog.len()).collect();
    package_indices.sort_by(|left, right| catalog[*left].name.cmp(&catalog[*right].name));

    let mut search = Search {
        catalog,
        roots,
        package_indices,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = Vec::new();
    let result = search.inspect(&mut selected);

    if let Err(ResolveError::Limit { attempts }) = result {
        return Err(ResolveError::Limit { attempts });
    }

    let Some((_, _, selected, active)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let mut changes = Vec::new();
    for lock in locks {
        let after = selected
            .iter()
            .find(|selection| selection.package == lock.package)
            .map(|selection| selection.version);
        if after != Some(lock.version) {
            let reason = match after {
                None => ChangeReason::Removed,
                Some(_) => {
                    let exists = search
                        .package_index(&lock.package)
                        .map(|index| {
                            catalog[index]
                                .releases
                                .iter()
                                .any(|release| release.version == lock.version)
                        })
                        .unwrap_or(false);
                    if !exists {
                        ChangeReason::Unavailable
                    } else {
                        let constraints = active
                            .iter()
                            .find(|(name, _)| name == &lock.package)
                            .map(|(_, constraints)| constraints.as_slice())
                            .unwrap_or(&[]);
                        if !matches_all(lock.version, constraints) {
                            ChangeReason::Rejected
                        } else {
                            ChangeReason::Rebalanced
                        }
                    }
                }
            };
            changes.push(Change {
                package: lock.package.clone(),
                before: Some(lock.version),
                after,
                reason,
            });
        }
    }

    for selection in &selected {
        if !locks.iter().any(|lock| lock.package == selection.package) {
            changes.push(Change {
                package: selection.package.clone(),
                before: None,
                after: Some(selection.version),
                reason: ChangeReason::Added,
            });
        }
    }
    changes.sort_by(|left, right| left.package.cmp(&right.package));

    Ok(Resolution { selected, changes })
}
