use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    base: usize,
    check_pre_numeric: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut start = 0;
    loop {
        let mut end = start;
        while end < bytes.len() && bytes[end] != b'.' {
            if !is_ident_char(bytes[end]) {
                return Err(VersionError::InvalidChar { at: base + end });
            }
            end += 1;
        }
        if end == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
        let part = &s[start..end];
        if check_pre_numeric
            && part.bytes().all(is_digit)
            && part.len() > 1
            && part.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        ids.push(part.to_string());
        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
    }
    Ok(ids)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut nums = [0u64; 3];
    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }
        if pos == start {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let n = s[start..pos]
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;
        nums[field] = n;
        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let pstart = pos + 1;
        let plus = bytes[pstart..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pstart + i)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(&s[pstart..plus], pstart, true)?;
        pos = plus;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        let bstart = pos + 1;
        build = parse_identifiers(&s[bstart..], bstart, false)?;
        pos = bytes.len();
    }
    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn vcore(t: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: t.0,
        minor: t.1,
        patch: t.2,
        pre,
        build: Vec::new(),
    }
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        let v = parse_full_operand(s, at)?;
        return Ok(Partial {
            nums: vec![v.major, v.minor, v.patch],
            full: Some(v),
        });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")) {
        let v = parse_full_operand(s, at)?;
        return Ok(Partial {
            nums: vec![v.major, v.minor, v.patch],
            full: Some(v),
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial { nums, full: None })
}

fn bump_tuple(mut t: (u64, u64, u64), field: usize, at: usize) -> Result<(u64, u64, u64), RangeError> {
    match field {
        0 => t.0 = t.0.checked_add(1).ok_or(RangeError::Syntax { at })?,
        1 => t.1 = t.1.checked_add(1).ok_or(RangeError::Syntax { at })?,
        _ => t.2 = t.2.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    if field <= 1 { t.2 = 0; }
    if field == 0 { t.1 = 0; }
    Ok(t)
}

fn padded(nums: &[u64]) -> (u64, u64, u64) {
    (nums.first().copied().unwrap_or(0), nums.get(1).copied().unwrap_or(0), nums.get(2).copied().unwrap_or(0))
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.cmps.push(Cmp { op, version, anchor });
}

fn parse_bare(arm: &mut Arm, token: &str, at: usize) -> Result<(), RangeError> {
    let p = parse_partial(token, at)?;
    if let Some(v) = p.full {
        add_cmp(arm, Op::Eq, v, true);
        return Ok(());
    }
    let k = p.nums.len();
    if k == 0 {
        return Ok(());
    }
    let t = padded(&p.nums);
    add_cmp(arm, Op::Ge, vcore(t, Vec::new()), false);
    if k < 3 {
        let upper = bump_tuple(t, k - 1, at)?;
        add_cmp(arm, Op::Lt, vcore(upper, vec!["0".into()]), false);
    }
    Ok(())
}

fn parse_caret_tilde(arm: &mut Arm, token: &str, at: usize, caret: bool) -> Result<(), RangeError> {
    let p = parse_partial(&token[1..], at + 1)?;
    let k = p.nums.len();
    if k == 0 {
        return Err(RangeError::Syntax { at });
    }
    let lower = p.full.unwrap_or_else(|| vcore(padded(&p.nums), Vec::new()));
    let t = core(&lower);
    add_cmp(arm, Op::Ge, lower, true);
    let field = if caret {
        if t.0 > 0 || k == 1 { 0 } else if t.1 > 0 || k == 2 { 1 } else { 2 }
    } else if k == 1 { 0 } else { 1 };
    let upper = bump_tuple(t, field, at)?;
    add_cmp(arm, Op::Lt, vcore(upper, vec!["0".into()]), false);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[(usize, &str)], arm: &mut Arm) -> Result<(), RangeError> {
    let dash = tokens.iter().position(|(_, t)| *t == "-").unwrap();
    if tokens.len() != 3 || dash != 1 {
        return Err(RangeError::Syntax { at: tokens[dash].0 });
    }
    let lo = parse_partial(tokens[0].1, tokens[0].0)?;
    let hi = parse_partial(tokens[2].1, tokens[2].0)?;
    if lo.nums.is_empty() || hi.nums.is_empty() {
        return Err(RangeError::Syntax {
            at: if lo.nums.is_empty() { tokens[0].0 } else { tokens[2].0 },
        });
    }
    let lv = lo.full.unwrap_or_else(|| vcore(padded(&lo.nums), Vec::new()));
    add_cmp(arm, Op::Ge, lv, true);
    if let Some(v) = hi.full {
        add_cmp(arm, Op::Le, v, true);
    } else {
        let t = padded(&hi.nums);
        let k = hi.nums.len();
        if k >= 3 {
            add_cmp(arm, Op::Le, vcore(t, Vec::new()), false);
        } else {
            let upper = bump_tuple(t, k - 1, tokens[2].0)?;
            add_cmp(arm, Op::Lt, vcore(upper, vec!["0".into()]), false);
        }
    }
    Ok(())
}

fn parse_arm(s: &str, offset: usize) -> Result<Arm, RangeError> {
    let b = s.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && ws(b[i]) { i += 1; }
        if i == b.len() { break; }
        let start = i;
        while i < b.len() && !ws(b[i]) { i += 1; }
        tokens.push((offset + start, &s[start..i]));
    }
    let mut arm = Arm { cmps: Vec::new(), anchors: Vec::new() };
    if tokens.iter().any(|(_, t)| *t == "-") {
        return parse_hyphen_arm(&tokens, &mut arm).map(|_| arm);
    }
    for (at, token) in tokens {
        let first = token.as_bytes()[0];
        match first {
            b'^' | b'~' => parse_caret_tilde(&mut arm, token, at, first == b'^')?,
            b'<' | b'>' | b'=' => {
                let (op, n) = if token.starts_with("<=") { (Op::Le, 2) }
                    else if token.starts_with(">=") { (Op::Ge, 2) }
                    else if token.starts_with('<') { (Op::Lt, 1) }
                    else if token.starts_with('>') { (Op::Gt, 1) }
                    else { (Op::Eq, 1) };
                if n == token.len() || !token.as_bytes()[n].is_ascii_digit() {
                    return Err(RangeError::Syntax { at });
                }
                let v = parse_full_operand(&token[n..], at + n)?;
                add_cmp(&mut arm, op, v, true);
            }
            b'0'..=b'9' | b'*' | b'x' | b'X' => parse_bare(&mut arm, token, at)?,
            _ => return Err(RangeError::Syntax { at }),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;
    let mut saw_any = false;
    while i < b.len() {
        while i < b.len() && ws(b[i]) { i += 1; }
        if i == b.len() { break; }
        if i + 1 < b.len() && b[i] == b'|' && b[i + 1] == b'|' {
            return Err(RangeError::Empty { at: i });
        }
        let arm_start = i;
        while i < b.len() {
            if i + 1 < b.len() && b[i] == b'|' && b[i + 1] == b'|' {
                break;
            }
            i += 1;
        }
        let arm_end = i;
        let piece = &s[arm_start..arm_end];
        arms.push(parse_arm(piece, arm_start)?);
        saw_any = true;
        if i < b.len() {
            let sep = i;
            if i > 0 && !ws(b[i - 1]) || i + 2 < b.len() && !ws(b[i + 2]) {
                return Err(RangeError::Syntax { at: sep });
            }
            i += 2;
            let mut j = i;
            while j < b.len() && ws(b[j]) { j += 1; }
            if j == b.len() {
                return Err(RangeError::Empty { at: b.len() });
            }
            if j + 1 < b.len() && b[j] == b'|' && b[j + 1] == b'|' {
                return Err(RangeError::Empty { at: j });
            }
            i = j;
        }
        start = i;
    }
    let _ = start;
    if !saw_any {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

fn cmp_matches(v: &Version, c: &Cmp) -> bool {
    let o = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => o == Ordering::Equal,
        Op::Lt => o == Ordering::Less,
        Op::Le => o != Ordering::Greater,
        Op::Gt => o == Ordering::Greater,
        Op::Ge => o != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.cmps.iter().all(|c| cmp_matches(v, c)) {
        return false;
    }
    include || v.pre.is_empty() || arm.anchors.contains(&core(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let cmp = compare_versions(v, &parsed[j]);
            if (max && cmp == Ordering::Greater) || (!max && cmp == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn checked_next_core(t: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if let Some(p) = t.2.checked_add(1) { return Some((t.0, t.1, p)); }
    if let Some(m) = t.1.checked_add(1) { return Some((t.0, m, 0)); }
    t.0.checked_add(1).map(|a| (a, 0, 0))
}

fn arm_has_stable_witness(arm: &Arm) -> bool {
    let mut lo: Option<(Version, bool)> = None;
    let mut hi: Option<(Version, bool)> = None;
    for c in &arm.cmps {
        match c.op {
            Op::Eq => {
                lo = Some((c.version.clone(), true));
                hi = Some((c.version.clone(), true));
            }
            Op::Ge | Op::Gt => {
                let inc = matches!(c.op, Op::Ge);
                if lo.as_ref().map_or(true, |(v, old)| {
                    let o = compare_versions(&c.version, v);
                    o == Ordering::Greater || (o == Ordering::Equal && !inc && *old)
                }) { lo = Some((c.version.clone(), inc)); }
            }
            Op::Le | Op::Lt => {
                let inc = matches!(c.op, Op::Le);
                if hi.as_ref().map_or(true, |(v, old)| {
                    let o = compare_versions(&c.version, v);
                    o == Ordering::Less || (o == Ordering::Equal && !inc && *old)
                }) { hi = Some((c.version.clone(), inc)); }
            }
        }
    }
    let lower = lo.map(|(v, inclusive)| (v, inclusive));
    let upper = hi.map(|(v, inclusive)| (v, inclusive));
    match (lower, upper) {
        (Some((l, li)), Some((u, ui))) => {
            let c = compare_versions(&l, &u);
            if c == Ordering::Less { true }
            else if c == Ordering::Greater { false }
            else if li && ui { l.pre.is_empty() }
            else if !l.pre.is_empty() {
                let successor = {
                    let mut p = l.pre.clone();
                    p.push("0".into());
                    vcore(core(&l), p)
                };
                compare_versions(&successor, &u) != Ordering::Greater
                    && (compare_versions(&successor, &u) != Ordering::Equal || ui)
            } else if let Some(next) = checked_next_core(core(&l)) {
                let n = vcore(next, Vec::new());
                compare_versions(&n, &u) != Ordering::Greater
                    && (compare_versions(&n, &u) != Ordering::Equal || ui)
            } else { false }
        }
        _ => true,
    }
}

fn admitted_prerelease_witness(arm: &Arm) -> bool {
    for anchor in &arm.anchors {
        let least = vcore(*anchor, vec!["0".into()]);
        let mut candidate = least.clone();
        let mut ok = true;
        for c in &arm.cmps {
            if !cmp_matches(&candidate, c) { ok = false; break; }
        }
        if ok { return true; }
        // A finite conjunction changes truth only at a comparator boundary.
        let mut probes = Vec::new();
        for c in &arm.cmps {
            if core(&c.version) == *anchor && !c.version.pre.is_empty() {
                probes.push(c.version.clone());
                let mut p = c.version.pre.clone();
                p.push("0".into());
                probes.push(vcore(*anchor, p));
            }
        }
        probes.push(least);
        for p in probes {
            if !p.pre.is_empty() && arm.cmps.iter().all(|c| cmp_matches(&p, c)) {
                return true;
            }
        }
        let _ = &mut candidate;
        let _ = &mut ok;
    }
    false
}

fn arm_possible_default(arm: &Arm) -> bool {
    arm_has_stable_witness(arm) || admitted_prerelease_witness(arm)
}

fn strongest_bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lo: Option<(Version, bool)> = None;
    let mut hi: Option<(Version, bool)> = None;
    for c in &arm.cmps {
        match c.op {
            Op::Eq => {
                lo = Some((c.version.clone(), true));
                hi = Some((c.version.clone(), true));
            }
            Op::Ge | Op::Gt => {
                let inc = matches!(c.op, Op::Ge);
                if lo.as_ref().map_or(true, |(v, old)| {
                    let o = compare_versions(&c.version, v);
                    o == Ordering::Greater || (o == Ordering::Equal && !inc && *old)
                }) { lo = Some((c.version.clone(), inc)); }
            }
            Op::Le | Op::Lt => {
                let inc = matches!(c.op, Op::Le);
                if hi.as_ref().map_or(true, |(v, old)| {
                    let o = compare_versions(&c.version, v);
                    o == Ordering::Less || (o == Ordering::Equal && !inc && *old)
                }) { hi = Some((c.version.clone(), inc)); }
            }
        }
    }
    (lo, hi)
}

fn version_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        if !arm_possible_default(arm) { continue; }
        let (lo, hi) = strongest_bounds(arm);
        let l = lo.map_or("*".to_string(), |(v, inc)| format!("{}{}", if inc { ">=" } else { ">" }, version_text(&v)));
        let u = hi.map_or("*".to_string(), |(v, inc)| format!("{}{}", if inc { "<=" } else { "<" }, version_text(&v)));
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let a = if anchors.is_empty() { "-".to_string() } else {
            anchors.iter().map(|t| format!("{}.{}.{}", t.0, t.1, t.2)).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{l};{u};pre={a}]"));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

fn arms_intersect(a: &Arm, b: &Arm, include: bool) -> bool {
    let mut combined = Arm {
        cmps: a.cmps.iter().chain(&b.cmps).cloned().collect(),
        anchors: Vec::new(),
    };
    if include {
        return arm_has_stable_witness(&combined) || admitted_prerelease_witness(&{
            combined.anchors = a.anchors.iter().chain(&b.anchors).copied().collect();
            combined
        });
    }
    combined.anchors = a.anchors.iter().chain(&b.anchors).copied().collect();
    arm_has_stable_witness(&combined) || admitted_prerelease_witness(&combined)
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    a.arms.iter().any(|x| b.arms.iter().any(|y| arms_intersect(x, y, include_prerelease)))
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(expr: &SetExpr, parsed: &[(String, Range)], v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parsed.iter().find(|(x, _)| x == s).map_or(false, |(_, r)| matches_with(v, r, include)),
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, parsed, v, include)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, parsed, v, include)),
    }
}

fn validate_ranges(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    out: &mut Vec<(String, Range)>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            out.push((s.clone(), r));
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, child) in xs.iter().enumerate() {
                path.push(i);
                validate_ranges(child, package, label, path, out)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn minimum_conflict(
    constraints: &[Constraint],
    parsed_ranges: &[Vec<(String, Range)>],
    candidates: &[Version],
    include: bool,
) -> Vec<String> {
    let n = constraints.len();
    if candidates.is_empty() { return Vec::new(); }
    for size in 1..=n {
        let mut chosen = Vec::new();
        if let Some(indices) = find_conflict(size, 0, n, &mut chosen, constraints, parsed_ranges, candidates, include) {
            let mut labels = indices.iter().map(|&i| constraints[i].label.clone()).collect::<Vec<_>>();
            labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            return labels;
        }
    }
    Vec::new()
}

fn find_conflict(
    need: usize,
    start: usize,
    n: usize,
    chosen: &mut Vec<usize>,
    constraints: &[Constraint],
    parsed_ranges: &[Vec<(String, Range)>],
    candidates: &[Version],
    include: bool,
) -> Option<Vec<usize>> {
    if need == 0 {
        let any = candidates.iter().any(|v| chosen.iter().all(|&i| eval_expr(&constraints[i].expr, &parsed_ranges[i], v, include)));
        return if any { None } else { Some(chosen.clone()) };
    }
    if n - start < need { return None; }
    for i in start..=n - need {
        chosen.push(i);
        if let Some(c) = find_conflict(need - 1, i + 1, n, chosen, constraints, parsed_ranges, candidates, include) {
            return Some(c);
        }
        chosen.pop();
    }
    None
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut parsed_candidates = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index, error,
            })?);
        }
        parsed_candidates.push(vs);
    }
    let mut all_ranges = Vec::new();
    for p in packages {
        let mut ranges_for_package = Vec::new();
        for c in &p.constraints {
            let mut found = Vec::new();
            validate_ranges(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut found)?;
            ranges_for_package.push(found);
        }
        all_ranges.push(ranges_for_package);
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let candidates = &parsed_candidates[pi];
        let qualifies = |i: usize| -> bool {
            p.constraints.iter().enumerate().all(|(ci, c)| {
                eval_expr(&c.expr, &all_ranges[pi][ci], &candidates[i], include_prerelease)
            })
        };
        let mut best: Option<usize> = None;
        for i in 0..candidates.len() {
            if !qualifies(i) { continue; }
            if let Some(j) = best {
                let cmp = compare_versions(&candidates[i], &candidates[j]);
                let take = match preference {
                    Preference::Highest => cmp == Ordering::Greater,
                    Preference::Lowest => cmp == Ordering::Less,
                    Preference::StableFirst => {
                        let si = candidates[i].pre.is_empty();
                        let sj = candidates[j].pre.is_empty();
                        if si != sj { si } else { cmp == Ordering::Greater }
                    }
                };
                if take { best = Some(i); }
            } else {
                best = Some(i);
            }
        }
        if let Some(i) = best {
            selections.push(Selection { package: p.name.clone(), version: p.candidates[i].clone() });
        } else {
            let labels = minimum_conflict(&p.constraints, &all_ranges[pi], candidates, include_prerelease);
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
    }
    Ok(selections)
}
