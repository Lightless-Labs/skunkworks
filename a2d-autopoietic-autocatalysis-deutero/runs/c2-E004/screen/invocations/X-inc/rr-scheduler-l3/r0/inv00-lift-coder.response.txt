```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    use std::collections::VecDeque;

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut event_cursor = 0;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completed = 0usize;

    while completed < jobs.len() {
        while event_cursor < events.len()
            && u64::from(jobs[events[event_cursor]].arrival) <= time
        {
            let index = events[event_cursor];
            if remaining[index] == 0 {
                completion[index] = Some(u64::from(jobs[index].arrival));
                completed += 1;
            } else {
                queues[levels[index]].push_back(index);
            }
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = u64::from(jobs[events[event_cursor]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let job_id = jobs[index].id;

        if last_dispatched.is_some_and(|previous_id| previous_id != job_id) {
            time += u64::from(switch_cost);
        }

        while event_cursor < events.len()
            && u64::from(jobs[events[event_cursor]].arrival) <= time
        {
            let arriving = events[event_cursor];
            if remaining[arriving] == 0 {
                completion[arriving] = Some(u64::from(jobs[arriving].arrival));
                completed += 1;
            } else {
                queues[levels[arriving]].push_back(arriving);
            }
            event_cursor += 1;
        }

        let start = time;
        let consumed = remaining[index].min(quanta[level]);
        remaining[index] -= consumed;
        time += u64::from(consumed);
        let finished = remaining[index] == 0;

        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished,
            level,
        });

        while event_cursor < events.len()
            && u64::from(jobs[events[event_cursor]].arrival) <= time
        {
            let arriving = events[event_cursor];
            if remaining[arriving] == 0 {
                completion[arriving] = Some(u64::from(jobs[arriving].arrival));
                completed += 1;
            } else {
                queues[levels[arriving]].push_back(arriving);
            }
            event_cursor += 1;
        }

        if finished {
            completion[index] = Some(time);
            completed += 1;
        } else {
            levels[index] = (level + 1).min(quanta.len() - 1);
            queues[levels[index]].push_back(index);
        }

        last_dispatched = Some(job_id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
```
