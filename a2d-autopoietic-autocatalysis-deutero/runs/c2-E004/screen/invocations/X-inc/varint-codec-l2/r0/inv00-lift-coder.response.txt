#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::with_capacity(10);
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            output.push(payload);
            break;
        }
        output.push(payload | 0x80);
    }
    output
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    for index in 0..10 {
        let offset = start + index;
        if offset >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[offset];
        let payload = byte & 0x7f;
        if index == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow { offset, start });
            }
            value |= (payload as u64) << 63;
            return Ok((value, offset + 1));
        }

        value |= (payload as u64) << (index * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count);
    let mut cursor = start;
    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }
    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let zigzag = if value >= 0 {
        (value as u64) << 1
    } else {
        ((value as u64) << 1) ^ u64::MAX
    };
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, cursor) = decode_u64(input, start)?;
    let decoded = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((decoded, cursor))
}

#[derive(Debug, Clone)]
pub struct StreamDecoder {
    position: usize,
    pending: Vec<u8>,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            pending: Vec::new(),
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut next_pending = self.pending.clone();
        let mut values = Vec::new();
        let mut integer_start = self.position - self.pending.len();
        let mut group_index = next_pending.len();

        for (chunk_index, &byte) in chunk.iter().enumerate() {
            let absolute_offset = self.position + chunk_index;

            if group_index == 9 {
                let payload = byte & 0x7f;
                if payload > 1 || byte & 0x80 != 0 {
                    return Err(Error::Overflow {
                        offset: absolute_offset,
                        start: integer_start,
                    });
                }

                let mut value = 0u64;
                for (i, &pending_byte) in next_pending.iter().enumerate() {
                    value |= ((pending_byte & 0x7f) as u64) << (i * 7);
                }
                value |= (payload as u64) << 63;
                values.push(value);
                next_pending.clear();
                group_index = 0;
                integer_start = absolute_offset + 1;
            } else if byte & 0x80 != 0 {
                next_pending.push(byte);
                group_index += 1;
            } else {
                let mut value = 0u64;
                for (i, &pending_byte) in next_pending.iter().enumerate() {
                    value |= ((pending_byte & 0x7f) as u64) << (i * 7);
                }
                value |= (byte as u64) << (group_index * 7);
                values.push(value);
                next_pending.clear();
                group_index = 0;
                integer_start = absolute_offset + 1;
            }
        }

        self.position += chunk.len();
        self.pending = next_pending;
        Ok(values)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending.is_empty() {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.position - self.pending.len(),
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending.len()
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending.clear();
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}
