use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(mut num: i128, mut den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    if den < 0 {
        num = num.checked_neg().ok_or(RationalError::Overflow)?;
        den = den.checked_neg().ok_or(RationalError::Overflow)?;
    }
    let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    num /= g;
    den /= g;
    if num < i64::MIN as i128
        || num > i64::MAX as i128
        || den < 1
        || den > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: num as i64,
        den: den as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let num = (self.num as i128) * a + (rhs.num as i128) * b;
        let den = (self.den as i128) * a;
        rational_from_i128(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let num = (self.num as i128) * a - (rhs.num as i128) * b;
        let den = (self.den as i128) * a;
        rational_from_i128(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128) as i128;
        rational_from_i128(
            (self.num as i128 / g1) * (rhs.num as i128 / g2),
            (self.den as i128 / g2) * (rhs.den as i128 / g1),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(
            self.num.unsigned_abs() as u128,
            rhs.num.unsigned_abs() as u128,
        ) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let mut n = (self.num as i128 / g1) * (rhs.den as i128 / g2);
        let mut d = (self.den as i128 / g2) * (rhs.num as i128 / g1);
        if d < 0 {
            n = -n;
            d = -d;
        }
        rational_from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Rational {
            num: n,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn err(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(err(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
                continue;
            }
            let valid_digit = b.is_ascii_digit();
            let valid_minus = b == b'-' && i == 0;
            if !valid_digit && !valid_minus {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let n_end = slash.unwrap_or(bytes.len());
        let n_start = usize::from(bytes[0] == b'-');
        if n_start == n_end {
            return Err(err(n_end, ParseKind::InvalidByte));
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(err(bytes.len(), ParseKind::InvalidByte));
            }
            if s == 0 {
                return Err(err(0, ParseKind::InvalidByte));
            }
        }
        for i in n_start..n_end {
            if !bytes[i].is_ascii_digit() {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }
        if let Some(s) = slash {
            for i in s + 1..bytes.len() {
                if !bytes[i].is_ascii_digit() {
                    return Err(err(i, ParseKind::InvalidByte));
                }
            }
        }

        let n_digits = n_end - n_start;
        if n_digits >= 2 && bytes[n_start] == b'0' {
            return Err(err(n_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && n_digits == 1 && bytes[1] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some(s) = slash {
            if bytes.len() - s - 1 >= 2 && bytes[s + 1] == b'0' {
                return Err(err(s + 2, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let mut nmag: u128 = 0;
        let nlimit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        for &b in &bytes[n_start..n_end] {
            nmag = nmag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
            if nmag > nlimit {
                return Err(err(0, ParseKind::OutOfRange));
            }
        }
        let num = if negative {
            if nmag == 1u128 << 63 {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        let Some(s) = slash else {
            return Ok(Rational { num, den: 1 });
        };

        let mut dmag: u128 = 0;
        for &b in &bytes[s + 1..] {
            dmag = dmag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
            if dmag > i64::MAX as u128 {
                return Err(err(s + 1, ParseKind::OutOfRange));
            }
        }
        if dmag == 0 {
            return Err(err(s + 1, ParseKind::ZeroDenominator));
        }
        if dmag == 1 {
            return Err(err(s, ParseKind::NonCanonical));
        }
        if num == 0 {
            return Err(err(s, ParseKind::NonCanonical));
        }
        if gcd_u128(num.unsigned_abs() as u128, dmag) > 1 {
            return Err(err(s, ParseKind::NonCanonical));
        }
        Ok(Rational {
            num,
            den: dmag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(&self, pivot_cols: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_cols {
            if p == m.rows {
                break;
            }
            let mut best: Option<(i64, u64, usize)> = None;
            for row in p..m.rows {
                let v = m.data[row * m.cols + col];
                if v.num == 0 {
                    continue;
                }
                let key = (v.den, v.num.unsigned_abs(), row);
                if best.map_or(true, |old| key < old) {
                    best = Some(key);
                }
            }
            let Some((_, _, q)) = best else {
                continue;
            };
            if q != p {
                for c in 0..m.cols {
                    m.data.swap(p * m.cols + c, q * m.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = m.data[p * m.cols + col];
            let mut normalized = Vec::with_capacity(m.cols);
            for c in 0..m.cols {
                let v = m.data[p * m.cols + c];
                normalized.push(v.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                    row: p,
                    col: c,
                    step: p,
                    source,
                })?);
            }
            for c in 0..m.cols {
                m.data[p * m.cols + c] = normalized[c];
            }

            for row in 0..m.rows {
                if row == p {
                    continue;
                }
                let factor = m.data[row * m.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..m.cols {
                    let product = factor
                        .mul(&m.data[p * m.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    let value = m.data[row * m.cols + c]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    m.data[row * m.cols + c] = value;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction {
            matrix: m,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..width {
                if c < n {
                    data.push(self.data[r * n + c]);
                } else {
                    data.push(Rational {
                        num: if c - n == r { 1 } else { 0 },
                        den: 1,
                    });
                }
            }
        }
        let aug = Self {
            rows: n,
            cols: width,
            data,
        };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: red.pivots.len(),
            });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n
            .checked_add(k)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                data.push(rhs.data[r * k + c]);
            }
        }
        let augmented = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let red = augmented.eliminate(n).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            if red.matrix.data[r * width..r * width + n]
                .iter()
                .all(|v| v.num == 0)
            {
                if let Some(c) =
                    (0..k).find(|&c| red.matrix.data[r * width + n + c].num != 0)
                {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col: c,
                        witness: red.matrix.data[r * width..(r + 1) * width].to_vec(),
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let mut free = Vec::new();
        let mut is_pivot = vec![false; n];
        for &c in &red.pivots {
            is_pivot[c] = true;
        }
        for c in 0..n {
            if !is_pivot[c] {
                free.push(c);
            }
        }

        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (i, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] =
                    red.matrix.data[i * width + n + c];
            }
        }
        let particular = Self {
            rows: n,
            cols: k,
            data: particular_data,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: red.pivots,
            });
        }

        let basis_len = n
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let free_count = free.len();
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free_count + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let value = red.matrix.data[i * width + free_col].neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: rank,
                        source,
                    })
                })?;
                basis_data[pivot_col * free_count + j] = value;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self {
                rows: n,
                cols: free_count,
                data: basis_data,
            },
            pivots: red.pivots,
        })
    }
}
