use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_number(s: &str, bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    if *pos == start {
        return Err(match bytes.get(start) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }

    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    bytes: &[u8],
    pos: &mut usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut ids = Vec::new();

    loop {
        let start = *pos;
        while *pos < end {
            let b = bytes[*pos];
            if b.is_ascii_alphanumeric() || b == b'-' {
                *pos += 1;
            } else if b == b'.' {
                break;
            } else {
                return Err(VersionError::InvalidChar { at: *pos });
            }
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let id = &s[start..*pos];
        if prerelease
            && id.len() > 1
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }

        ids.push(id.to_string());
        if *pos == end {
            break;
        }

        *pos += 1;
        if *pos == end {
            return Err(VersionError::EmptyIdentifier { at: *pos });
        }
    }

    Ok(ids)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;
    let major = parse_core_number(s, bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(match bytes.get(pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let minor = parse_core_number(s, bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(match bytes.get(pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;

    let patch = parse_core_number(s, bytes, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(s, bytes, &mut pos, end, true)?;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, bytes, &mut pos, bytes.len(), false)?;
    }

    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b) {
        let xn = is_numeric_id(x);
        let yn = is_numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn tokenize(s: &str, base: usize) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !ws(bytes[i]) {
            i += 1;
        }
        out.push(Token {
            text: &s[start..i],
            start: base + start,
        });
    }
    out
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: *fields.first().unwrap_or(&0),
        minor: *fields.get(1).unwrap_or(&0),
        patch: *fields.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<u64>,
    full: Option<Version>,
}

fn parse_partial(operand: &str, at: usize) -> Result<Partial, RangeError> {
    let full_candidate = operand.contains('-')
        || operand.contains('+')
        || (operand.split('.').count() >= 3
            && !operand
                .split('.')
                .any(|f| matches!(f, "*" | "x" | "X")));

    if full_candidate {
        let v = parse_version(operand).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            fields: vec![v.major, v.minor, v.patch],
            full: Some(v),
        });
    }

    let parts: Vec<&str> = operand.split('.').collect();
    if parts.is_empty() || parts.len() > 3 || parts.iter().any(|p| p.is_empty()) {
        return Err(RangeError::Syntax { at });
    }

    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for p in parts {
        if matches!(p, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !p.bytes().all(|b| b.is_ascii_digit())
                || (p.len() > 1 && p.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            fields.push(n);
        }
    }

    Ok(Partial { fields, full: None })
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = v.clone();
    out.pre = vec!["0".to_string()];
    out.build.clear();
    let n = match field {
        0 => &mut out.major,
        1 => &mut out.minor,
        _ => &mut out.patch,
    };
    *n = n.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field {
        0 => {
            out.minor = 0;
            out.patch = 0;
        }
        1 => out.patch = 0,
        _ => {}
    }
    Ok(out)
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        let c = core(&version);
        if !arm.anchors.contains(&c) {
            arm.anchors.push(c);
        }
    }
    arm.comparators.push(Comparator { op, version });
}

fn comparator_ok(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn parse_comparator_token(tok: &Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, skip) = if tok.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if tok.text.starts_with("<=") {
        (Op::Le, 2)
    } else if tok.text.starts_with('>') {
        (Op::Gt, 1)
    } else if tok.text.starts_with('<') {
        (Op::Lt, 1)
    } else if tok.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };

    let operand = &tok.text[skip..];
    let operand_at = tok.start + skip;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let version = parse_version(operand)
        .map_err(|error| RangeError::Version { at: operand_at, error })?;
    add_cmp(arm, op, version, true);
    Ok(())
}

fn parse_partial_operand(
    tok: &Token<'_>,
    operand: &str,
    offset: usize,
) -> Result<Partial, RangeError> {
    parse_partial(operand, tok.start + offset)
}

fn parse_expanded_token(tok: &Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let b = tok.text.as_bytes();
    if b[0] == b'^' || b[0] == b'~' {
        let p = parse_partial_operand(tok, &tok.text[1..], 1)?;
        let k = p.fields.len();
        if k == 0 {
            return Err(RangeError::Syntax { at: tok.start });
        }
        let lower = p.full.clone().unwrap_or_else(|| padded(&p.fields));
        add_cmp(arm, Op::Ge, lower.clone(), true);

        let upper = if b[0] == b'^' {
            if p.fields[0] > 0 || k == 1 {
                bump(&lower, 0, tok.start)?
            } else if p.fields.get(1).copied().unwrap_or(0) > 0 || k == 2 {
                bump(&lower, 1, tok.start)?
            } else {
                bump(&lower, 2, tok.start)?
            }
        } else if k == 1 {
            bump(&lower, 0, tok.start)?
        } else {
            bump(&lower, 1, tok.start)?
        };
        add_cmp(arm, Op::Lt, upper, false);
        return Ok(());
    }

    let p = parse_partial_operand(tok, tok.text, 0)?;
    if let Some(v) = p.full {
        add_cmp(arm, Op::Eq, v, true);
        return Ok(());
    }

    let k = p.fields.len();
    if k == 0 {
        return Ok(());
    }
    let lower = padded(&p.fields);
    add_cmp(arm, Op::Ge, lower.clone(), false);
    if k < 3 {
        let upper = bump(&lower, k - 1, tok.start)?;
        add_cmp(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_hyphen_arm(
    tokens: &[Token<'_>],
    minus: usize,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    if tokens.len() != 3 || tokens[1].text != "-" {
        return Err(RangeError::Syntax { at: minus });
    }

    let lower = parse_partial_operand(&tokens[0], tokens[0].text, 0)?;
    let upper = parse_partial_operand(&tokens[2], tokens[2].text, 0)?;
    if lower.fields.is_empty() {
        return Err(RangeError::Syntax {
            at: tokens[0].start,
        });
    }
    if upper.fields.is_empty() {
        return Err(RangeError::Syntax {
            at: tokens[2].start,
        });
    }

    let lower_v = lower.full.clone().unwrap_or_else(|| padded(&lower.fields));
    add_cmp(arm, Op::Ge, lower_v, true);

    if let Some(v) = upper.full {
        add_cmp(arm, Op::Le, v, true);
    } else {
        let upper_v = padded(&upper.fields);
        if upper.fields.len() < 3 {
            let exclusive = bump(
                &upper_v,
                upper.fields.len() - 1,
                tokens[2].start,
            )?;
            add_cmp(arm, Op::Lt, exclusive, false);
        } else {
            add_cmp(arm, Op::Le, upper_v, false);
        }
    }
    Ok(())
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    if let Some(minus) = tokens.iter().find(|t| t.text == "-") {
        parse_hyphen_arm(tokens, minus.start, &mut arm)?;
        return Ok(arm);
    }

    for tok in tokens {
        let first = tok.text.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_comparator_token(tok, &mut arm)?;
        } else if matches!(first, b'^' | b'~' | b'*' | b'x' | b'X')
            || first.is_ascii_digit()
        {
            parse_expanded_token(tok, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: tok.start });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;

    while i <= bytes.len() {
        let separator = i < bytes.len()
            && bytes[i..].starts_with(b"||")
            && (i == 0 || ws(bytes[i - 1]))
            && (i + 2 == bytes.len() || ws(bytes[i + 2]));

        if i == bytes.len() || separator {
            let chunk = &s[arm_start..i];
            let tokens = tokenize(chunk, arm_start);
            if tokens.is_empty() {
                let at = if arms.is_empty() { 0 } else { i };
                return Err(RangeError::Empty { at });
            }
            arms.push(parse_arm(&tokens)?);

            if i == bytes.len() {
                break;
            }
            i += 2;
            while i < bytes.len() && ws(bytes[i]) {
                i += 1;
            }
            if i == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
            arm_start = i;
        } else {
            i += 1;
        }
    }

    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparators.iter().all(|c| comparator_ok(v, c)) {
        return false;
    }
    if include_prerelease || v.pre.is_empty() {
        return true;
    }
    arm.anchors.contains(&core(v))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(bi) = best {
            let ord = compare_versions(v, &parsed[bi]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }

    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug)]
struct Bounds {
    lower: Option<(Version, bool)>,
    upper: Option<(Version, bool)>,
}

fn arm_bounds(arm: &Arm) -> Bounds {
    let mut bounds = Bounds {
        lower: None,
        upper: None,
    };
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                choose_lower(&mut bounds.lower, c.version.clone(), true);
                choose_upper(&mut bounds.upper, c.version.clone(), true);
            }
            Op::Gt | Op::Ge => {
                choose_lower(&mut bounds.lower, c.version.clone(), c.op == Op::Ge)
            }
            Op::Lt | Op::Le => {
                choose_upper(&mut bounds.upper, c.version.clone(), c.op == Op::Le)
            }
        }
    }
    bounds
}

fn choose_lower(slot: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match slot {
        None => *slot = Some((v, inclusive)),
        Some((old, old_inc)) => match compare_versions(&v, old) {
            Ordering::Greater => *slot = Some((v, inclusive)),
            Ordering::Equal if !inclusive && *old_inc => *slot = Some((v, false)),
            _ => {}
        },
    }
}

fn choose_upper(slot: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match slot {
        None => *slot = Some((v, inclusive)),
        Some((old, old_inc)) => match compare_versions(&v, old) {
            Ordering::Less => *slot = Some((v, inclusive)),
            Ordering::Equal if !inclusive && *old_inc => *slot = Some((v, false)),
            _ => {}
        },
    }
}

fn tuple_version(c: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: c.0,
        minor: c.1,
        patch: c.2,
        pre,
        build: Vec::new(),
    }
}

fn next_core(c: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if c.2 < u64::MAX {
        Some((c.0, c.1, c.2 + 1))
    } else if c.1 < u64::MAX {
        Some((c.0, c.1 + 1, 0))
    } else if c.0 < u64::MAX {
        Some((c.0 + 1, 0, 0))
    } else {
        None
    }
}

fn candidate_at_least(v: &Version, bound: &(Version, bool)) -> bool {
    let o = compare_versions(v, &bound.0);
    o == Ordering::Greater || (o == Ordering::Equal && bound.1)
}

fn candidate_at_most(v: &Version, bound: &(Version, bool)) -> bool {
    let o = compare_versions(v, &bound.0);
    o == Ordering::Less || (o == Ordering::Equal && bound.1)
}

fn stable_witness(bounds: &Bounds) -> bool {
    let (lo, hi) = match (&bounds.lower, &bounds.upper) {
        (Some(l), Some(h)) => (l, h),
        _ => return true,
    };
    let ord = compare_versions(&lo.0, &hi.0);
    if ord == Ordering::Greater || (ord == Ordering::Equal && !(lo.1 && hi.1)) {
        return false;
    }
    if ord == Ordering::Equal {
        return lo.0.pre.is_empty();
    }

    let mut candidate = lo.0.clone();
    if !lo.1 {
        if !candidate.pre.is_empty() {
            candidate.pre.push("0".to_string());
        } else if let Some(c) = next_core(core(&candidate)) {
            candidate = tuple_version(c, Vec::new());
        } else {
            return false;
        }
    }
    if !candidate.pre.is_empty() {
        if let Some(c) = next_core(core(&candidate)) {
            candidate = tuple_version(c, Vec::new());
        } else {
            return false;
        }
    }
    candidate_at_most(&candidate, hi)
}

fn prerelease_witness(bounds: &Bounds, anchors: &[(u64, u64, u64)]) -> bool {
    for &c in anchors {
        let v = tuple_version(c, vec!["0".to_string()]);
        if bounds
            .lower
            .as_ref()
            .map_or(true, |b| candidate_at_least(&v, b))
            && bounds
                .upper
                .as_ref()
                .map_or(true, |b| candidate_at_most(&v, b))
        {
            return true;
        }
    }
    false
}

fn arm_possible(arm: &Arm) -> bool {
    let bounds = arm_bounds(arm);
    stable_witness(&bounds) || prerelease_witness(&bounds, &arm.anchors)
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut bounds = arm_bounds(aa);
            let other = arm_bounds(bb);
            if let Some((v, i)) = other.lower {
                choose_lower(&mut bounds.lower, v, i);
            }
            if let Some((v, i)) = other.upper {
                choose_upper(&mut bounds.upper, v, i);
            }

            if include_prerelease {
                if stable_witness(&bounds) || prerelease_witness(&bounds, &[]) {
                    return true;
                }
            } else {
                let mut shared = aa.anchors.clone();
                shared.retain(|x| bb.anchors.contains(x));
                if stable_witness(&bounds) || prerelease_witness(&bounds, &shared) {
                    return true;
                }
            }
        }
    }
    false
}

fn version_display(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn bound_text(bound: &Option<(Version, bool)>, lower: bool) -> String {
    match bound {
        None => "*".to_string(),
        Some((v, inclusive)) => {
            let op = match (lower, inclusive) {
                (true, true) => ">=",
                (true, false) => ">",
                (false, true) => "<=",
                (false, false) => "<",
            };
            format!("{op}{}", version_display(v))
        }
    }
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        if !arm_possible(arm) {
            continue;
        }
        let bounds = arm_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(a, b, c)| format!("{a}.{b}.{c}"))
                .collect::<Vec<_>>()
                .join(",")
        };
        strings.push(format!(
            "[{};{};pre={}]",
            bound_text(&bounds.lower, true),
            bound_text(&bounds.upper, false),
            anchor_text
        ));
    }
    strings.sort();
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".to_string()
    } else {
        strings.join(" || ")
    }
}
