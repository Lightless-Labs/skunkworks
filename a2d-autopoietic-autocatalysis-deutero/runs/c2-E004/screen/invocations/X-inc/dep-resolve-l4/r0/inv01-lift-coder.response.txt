fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(items: &[Evidence]) -> Vec<Evidence> {
    let mut out = items.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(first: String, items: &[Evidence]) -> String {
    let mut out = first;
    for e in evidence_sorted(items) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct State {
    selected: BTreeMap<String, Version>,
}

#[derive(Clone)]
struct Active {
    constraints: BTreeMap<String, Vec<Constraint>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    evidence: BTreeMap<String, Vec<Evidence>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn platform_matches(p: Platform, t: Target) -> bool {
    match p {
        Platform::Any => true,
        Platform::Linux => t == Target::Linux,
        Platform::Mac => t == Target::Mac,
        Platform::Windows => t == Target::Windows,
    }
}

fn root_token(r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root:{}@{}+{{{}}}",
        r.package,
        constraint_text(r.constraint),
        fs.join(",")
    )
}

fn dep_token(source: &str, version: Version, index: usize, d: &Dependency) -> String {
    let mut fs = d.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        d.package,
        constraint_text(d.constraint),
        fs.join(","),
        d.gate.as_deref().unwrap_or("*"),
        match d.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn better_path(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    state: &State,
    target: Target,
) -> Active {
    let mut active = Active {
        constraints: BTreeMap::new(),
        requests: BTreeMap::new(),
        evidence: BTreeMap::new(),
        paths: BTreeMap::new(),
    };

    for r in roots {
        active
            .constraints
            .entry(r.package.clone())
            .or_default()
            .push(r.constraint);
        active
            .requests
            .entry(r.package.clone())
            .or_default()
            .extend(r.features.iter().cloned());
        let token = root_token(r);
        active
            .evidence
            .entry(r.package.clone())
            .or_default()
            .push(Evidence {
                constraint: r.constraint,
                path: vec![token.clone()],
            });
        let path = vec![token];
        match active.paths.get(&r.package) {
            None => {
                active.paths.insert(r.package.clone(), path);
            }
            Some(old) if better_path(&path, old) => {
                active.paths.insert(r.package.clone(), path);
            }
            _ => {}
        }
    }

    // Recompute the joint feature/dependency closure until no new requests or edges appear.
    let mut processed = BTreeSet::<(String, usize, Version)>::new();
    loop {
        let mut changed = false;
        let names: Vec<String> = state.selected.keys().cloned().collect();

        for source in names {
            let version = state.selected[&source];
            let Some(package) = catalog.get(&source) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = active
                .requests
                .get(&source)
                .cloned()
                .unwrap_or_default();
            let mut feature_queue: Vec<String> = requested.into_iter().collect();
            let mut activated = BTreeSet::<String>::new();

            while let Some(feature) = feature_queue.pop() {
                if !activated.insert(feature.clone()) {
                    continue;
                }
                if let Some(decl) = release.features.iter().find(|f| f.name == feature) {
                    feature_queue.extend(decl.enables.iter().cloned());
                }
            }

            for feature in &activated {
                if let Some(decl) = release.features.iter().find(|f| f.name == *feature) {
                    let entry = active.requests.entry(source.clone()).or_default();
                    for enabled in &decl.enables {
                        if entry.insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            for (idx, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }

                if !processed.insert((source.clone(), idx, version)) {
                    continue;
                }

                active
                    .constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                active
                    .requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());

                let token = dep_token(&source, version, idx, dep);
                if let Some(source_path) = active.paths.get(&source).cloned() {
                    let mut path = source_path;
                    path.push(token.clone());
                    match active.paths.get(&dep.package) {
                        None => {
                            active.paths.insert(dep.package.clone(), path);
                        }
                        Some(old) if better_path(&path, old) => {
                            active.paths.insert(dep.package.clone(), path);
                        }
                        _ => {}
                    }

                    let mut evidence_path = active.paths[&source].clone();
                    evidence_path.push(token);
                    active
                        .evidence
                        .entry(dep.package.clone())
                        .or_default()
                        .push(Evidence {
                            constraint: dep.constraint,
                            path: evidence_path,
                        });
                }
                changed = true;
            }
        }

        if !changed {
            break;
        }

        // A request added during this pass may have enabled a gate on an earlier release.
        let mut has_unprocessed_active_dep = false;
        for (source, version) in &state.selected {
            if let Some(package) = catalog.get(source) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    let requested = active
                        .requests
                        .get(source)
                        .cloned()
                        .unwrap_or_default();
                    let mut queue: Vec<String> = requested.into_iter().collect();
                    let mut activated = BTreeSet::new();
                    while let Some(feature) = queue.pop() {
                        if !activated.insert(feature.clone()) {
                            continue;
                        }
                        if let Some(decl) = release.features.iter().find(|f| f.name == feature) {
                            queue.extend(decl.enables.iter().cloned());
                        }
                    }
                    for (idx, dep) in release.deps.iter().enumerate() {
                        if platform_matches(dep.platform, target)
                            && dep.gate.as_ref().is_none_or(|g| activated.contains(g))
                            && !processed.contains(&(source.clone(), idx, *version))
                        {
                            has_unprocessed_active_dep = true;
                        }
                    }
                }
            }
        }
        if !has_unprocessed_active_dep {
            let mut added = false;
            for (source, version) in &state.selected {
                if let Some(package) = catalog.get(source) {
                    if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                        let requested = active
                            .requests
                            .get(source)
                            .cloned()
                            .unwrap_or_default();
                        let mut queue: Vec<String> = requested.into_iter().collect();
                        let mut activated = BTreeSet::new();
                        while let Some(feature) = queue.pop() {
                            if !activated.insert(feature.clone()) {
                                continue;
                            }
                            if let Some(decl) =
                                release.features.iter().find(|f| f.name == feature)
                            {
                                for enabled in &decl.enables {
                                    if active
                                        .requests
                                        .entry(source.clone())
                                        .or_default()
                                        .insert(enabled.clone())
                                    {
                                        added = true;
                                    }
                                }
                            }
                        }
                    }
                }
            }
            if !added {
                break;
            }
        }
    }

    // Recompute canonical shortest paths on the final active dependency graph.
    let mut best_paths = BTreeMap::<String, Vec<String>>::new();
    for r in roots {
        let path = vec![root_token(r)];
        match best_paths.get(&r.package) {
            None => {
                best_paths.insert(r.package.clone(), path);
            }
            Some(old) if better_path(&path, old) => {
                best_paths.insert(r.package.clone(), path);
            }
            _ => {}
        }
    }

    let mut edges = Vec::<(String, String, String)>::new();
    for (source, version) in &state.selected {
        let Some(package) = catalog.get(source) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        let requested = active
            .requests
            .get(source)
            .cloned()
            .unwrap_or_default();
        let mut queue: Vec<String> = requested.into_iter().collect();
        let mut activated = BTreeSet::new();
        while let Some(feature) = queue.pop() {
            if !activated.insert(feature.clone()) {
                continue;
            }
            if let Some(decl) = release.features.iter().find(|f| f.name == feature) {
                queue.extend(decl.enables.iter().cloned());
            }
        }
        for (idx, dep) in release.deps.iter().enumerate() {
            if platform_matches(dep.platform, target)
                && dep.gate.as_ref().is_none_or(|g| activated.contains(g))
            {
                edges.push((
                    source.clone(),
                    dep.package.clone(),
                    dep_token(source, *version, idx, dep),
                ));
            }
        }
    }

    loop {
        let mut changed = false;
        for (source, destination, token) in &edges {
            if let Some(source_path) = best_paths.get(source).cloned() {
                let mut candidate = source_path;
                candidate.push(token.clone());
                match best_paths.get(destination) {
                    None => {
                        best_paths.insert(destination.clone(), candidate);
                        changed = true;
                    }
                    Some(old) if better_path(&candidate, old) => {
                        best_paths.insert(destination.clone(), candidate);
                        changed = true;
                    }
                    _ => {}
                }
            }
        }
        if !changed {
            break;
        }
    }

    active.paths = best_paths;
    active.evidence.clear();
    for r in roots {
        active
            .evidence
            .entry(r.package.clone())
            .or_default()
            .push(Evidence {
                constraint: r.constraint,
                path: vec![root_token(r)],
            });
    }
    for (source, destination, token) in edges {
        if let Some(path) = active.paths.get(&source) {
            let mut incoming = path.clone();
            incoming.push(token);
            if let Some(package) = catalog.get(&source) {
                if let Some(release) = package.releases.iter().find(|r| {
                    state.selected.get(source) == Some(&r.version)
                }) {
                    if let Some(dep) = release
                        .deps
                        .iter()
                        .enumerate()
                        .find(|(idx, d)| dep_token(source, release.version, *idx, d) == incoming.last().unwrap())
                        .map(|(_, d)| d)
                    {
                        active
                            .evidence
                            .entry(destination.clone())
                            .or_default()
                            .push(Evidence {
                                constraint: dep.constraint,
                                path: incoming,
                            });
                    }
                }
            }
        }
    }

    for cs in active.constraints.values_mut() {
        *cs = canonical_constraints(cs);
    }
    for evidence in active.evidence.values_mut() {
        evidence.sort_by(|a, b| {
            constraint_key(a.constraint)
                .cmp(&constraint_key(b.constraint))
                .then_with(|| a.path.cmp(&b.path))
        });
    }
    active
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, Vec<Option<Version>>, State, Active)>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, state: &State) -> Result<(), ResolveError> {
        let active = compute_active(self.catalog, self.roots, state, self.target);

        if let Some(name) = active
            .constraints
            .keys()
            .filter(|n| !self.catalog.contains_key(*n))
            .min()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
                evidence: active.evidence.get(name).cloned().unwrap_or_default(),
            });
        }

        for (name, version) in &state.selected {
            let cs = active.constraints.get(name).cloned().unwrap_or_default();
            if cs.iter().any(|c| !matches(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: cs,
                    evidence: active.evidence.get(name).cloned().unwrap_or_default(),
                });
            }
        }

        for (name, version) in &state.selected {
            if let Some(package) = self.catalog.get(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    let requested = active.requests.get(name).cloned().unwrap_or_default();
                    let mut closure = BTreeSet::new();
                    let mut stack: Vec<String> = requested.into_iter().collect();
                    while let Some(feature) = stack.pop() {
                        if !closure.insert(feature.clone()) {
                            continue;
                        }
                        if let Some(decl) = release.features.iter().find(|x| x.name == feature) {
                            stack.extend(decl.enables.iter().cloned());
                        }
                    }
                    if let Some(feature) = closure
                        .iter()
                        .find(|f| !release.features.iter().any(|decl| &decl.name == *f))
                    {
                        return Err(ResolveError::MissingFeature {
                            package: name.clone(),
                            selected: *version,
                            feature: feature.clone(),
                            path: active.paths.get(name).cloned().unwrap_or_default(),
                        });
                    }
                }
            }
        }

        for name in active.constraints.keys() {
            if state.selected.contains_key(name) {
                continue;
            }
            let cs = active.constraints.get(name).cloned().unwrap_or_default();
            let package = self.catalog[name];
            let candidates: Vec<Version> = package
                .releases
                .iter()
                .map(|r| r.version)
                .filter(|v| cs.iter().all(|c| matches(*c, *v)))
                .collect();
            if candidates.is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs,
                    evidence: active.evidence.get(name).cloned().unwrap_or_default(),
                });
            }
        }

        let next = active
            .constraints
            .keys()
            .find(|name| !state.selected.contains_key(*name))
            .cloned();

        let Some(package_name) = next else {
            let mut cost = 0usize;
            for lock in self.locks {
                if state.selected.get(&lock.package) != Some(&lock.version) {
                    cost += 1;
                }
            }
            let vector = self
                .catalog_names
                .iter()
                .map(|n| state.selected.get(n).copied())
                .collect::<Vec<_>>();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_vector, _, _)) => {
                    cost < *best_cost || (cost == *best_cost && vector > *best_vector)
                }
            };
            if replace {
                self.best = Some((cost, vector, state.clone(), active));
            }
            return Ok(());
        };

        let package = self.catalog[&package_name];
        let cs = active
            .constraints
            .get(&package_name)
            .cloned()
            .unwrap_or_default();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| cs.iter().all(|c| matches(*c, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self
            .locks
            .iter()
            .find(|l| l.package == package_name)
            .map(|l| l.version)
        {
            if let Some(pos) = candidates.iter().position(|v| *v == lock) {
                candidates.remove(pos);
                candidates.insert(0, lock);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let mut child = state.clone();
            child.selected.insert(package_name.clone(), version);
            if let Err(err) = self.inspect(&child) {
                if matches!(err, ResolveError::Limit { .. }) {
                    return Err(err);
                }
                if self.first_error.is_none() {
                    self.first_error = Some(err);
                }
            }
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let dep_site = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: dep_site.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{dep_site}.gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{dep_site}.feature[{f}]"),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let feature_site = format!("catalog[{i}].release[{j}].feature[{f}]");
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: feature_site.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{feature_site}.enable[{e}]"),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let catalog_names: Vec<String> = map.keys().cloned().collect();

    let mut search = Search {
        catalog: &map,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        catalog_names,
    };

    let initial = State {
        selected: BTreeMap::new(),
    };
    search.inspect(&initial)?;

    let Some((_, _, state, active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: vec![],
            evidence: vec![],
        }));
    };

    let selected: Vec<Selection> = state
        .selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut old = BTreeMap::new();
    for lock in locks {
        old.insert(lock.package.clone(), lock.version);
    }

    let mut all_names: BTreeSet<String> = old.keys().cloned().collect();
    all_names.extend(state.selected.keys().cloned());

    let mut changes = Vec::new();
    for package in all_names {
        let before = old.get(&package).copied();
        let after = state.selected.get(&package).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old_version));
                if !exists {
                    ChangeReason::Unavailable
                } else if active
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| cs.iter().any(|c| !matches(*c, old_version)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };

        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
