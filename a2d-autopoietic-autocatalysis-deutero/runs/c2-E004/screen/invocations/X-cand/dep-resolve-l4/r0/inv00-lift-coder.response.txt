use std::collections::{BTreeMap, BTreeSet};

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Target {
    Linux,
    Mac,
    Windows,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Platform {
    Any,
    Linux,
    Mac,
    Windows,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {

}

#[derive(Clone, Debug, PartialEq, Eq)]

    pub version: Version,

    Rebalanced,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateFeature { package: String, version: Version, feature: String },
    DuplicateLock { package: String },
    Missing { package: String, evidence: Vec<Evidence> },
    NoMatch { package: String, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    Conflict { package: String, selected: Version, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    MissingFeature { package: String, selected: Version, feature: String, path: Vec<String> },
    Limit { attempts: usize },
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(ev: &[Evidence]) -> Vec<Evidence> {
    let mut e = ev.to_vec();
    e.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    e
}

fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } =>
            format!("invalid-name|site={}|name={}", encode(site), encode(name)),
        ResolveError::DuplicatePackage { package } =>
            format!("duplicate-package|package={}", encode(package)),
        ResolveError::DuplicateVersion { package, version } =>
            format!("duplicate-version|package={}|version={}", encode(package), version_text(*version)),
        ResolveError::DuplicateFeature { package, version, feature } =>
            format!("duplicate-feature|package={}|version={}|feature={}", encode(package), version_text(*version), encode(feature)),
        ResolveError::DuplicateLock { package } =>
            format!("duplicate-lock|package={}", encode(package)),
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch { package, requirements, evidence } => {
            let mut s = format!("no-match|package={}|constraints={}", encode(package), constraint_list(requirements));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict { package, selected, requirements, evidence } => {
            let mut s = format!("conflict|package={}|selected={}|constraints={}", encode(package), version_text(*selected), constraint_list(requirements));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature { package, selected, feature, path } =>
            format!("missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
                encode(package), version_text(*selected), encode(feature), encode(feature), path.join(" -> ")),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn constraint_list(cs: &[Constraint]) -> String {
    canonical(cs.to_vec()).iter().map(|c| constraint_text(*c)).collect::<Vec<_>>().join(",")
}

fn append_evidence(s: &mut String, evidence: &[Evidence]) {
    for e in evidence_sorted(evidence) {
        s.push_str("\n  C:");
        s.push_str(&constraint_text(e.constraint));
        s.push_str(" via ");
        s.push_str(&e.path.join(" -> "));
    }
}

fn target_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct State {
    selected: BTreeMap<String, Version>,
}

struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    features: BTreeMap<String, BTreeSet<String>>,
    origins: BTreeMap<String, Vec<Origin>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!("root[{}]:{}@{}+{{{}}}", i, r.package, constraint_text(r.constraint), fs.join(","))
}

fn dep_token(source: &str, version: Version, index: usize, d: &Dependency) -> String {
    let mut fs = d.features.clone();
    fs.sort();
    fs.dedup();
    format!("{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source, version_text(version), index, d.package, constraint_text(d.constraint),
        fs.join(","), d.gate.as_deref().unwrap_or("*"), platform_text(d.platform))
}

fn better_path(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn closure(
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    state: &State,
    target: Target,
) -> Closure {
    let mut c = Closure {
        constraints: BTreeMap::new(),
        features: BTreeMap::new(),
        origins: BTreeMap::new(),
        paths: BTreeMap::new(),
    };

    for (i, r) in roots.iter().enumerate() {
        let path = vec![root_token(i, r)];
        c.constraints.entry(r.package.clone()).or_default().push(r.constraint);
        c.features.entry(r.package.clone()).or_default().extend(r.features.iter().cloned());
        c.origins.entry(r.package.clone()).or_default().push(Origin { constraint: r.constraint, path: path.clone() });
        let replace = c.paths.get(&r.package).map_or(true, |old| better_path(&path, old));
        if replace {
            c.paths.insert(r.package.clone(), path);
        }
    }

    loop {
        let mut changed = false;
        let packages: Vec<(String, Version)> = state.selected.iter().map(|(p, v)| (p.clone(), *v)).collect();
        for (source, version) in packages {
            let Some(pkg) = catalog.get(&source) else { continue };
            let Some(release) = pkg.releases.iter().find(|r| r.version == version) else { continue };

            let requested: Vec<String> = c.features.get(&source).map(|f| f.iter().cloned().collect()).unwrap_or_default();
            for feature_name in requested {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if c.features.entry(source.clone()).or_default().insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            let active_features = c.features.get(&source).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !active_features.contains(g))
                {
                    continue;
                }
                let token = dep_token(&source, version, j, dep);
                let source_path = c.paths.get(&source).cloned().unwrap_or_default();
                let mut path = source_path;
                path.push(token);
                c.constraints.entry(dep.package.clone()).or_default().push(dep.constraint);
                c.features.entry(dep.package.clone()).or_default().extend(dep.features.iter().cloned());
                c.origins.entry(dep.package.clone()).or_default().push(Origin { constraint: dep.constraint, path: path.clone() });
                let replace = c.paths.get(&dep.package).map_or(true, |old| better_path(&path, old));
                if replace {
                    c.paths.insert(dep.package.clone(), path);
                    changed = true;
                }
            }
        }
        if !changed {
            break;
        }
    }

    for cs in c.constraints.values_mut() {
        *cs = canonical(std::mem::take(cs));
    }
    c
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: BTreeMap<String, &'a Package>,
    locks: BTreeMap<String, Version>,
    target: Target,
    budget: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, state: State) -> Result<(), ResolveError> {
        let c = closure(self.roots, &self.catalog, &state, self.target);

        for p in c.constraints.keys() {
            if !self.catalog.contains_key(p) {
                let mut ev: Vec<Evidence> = c.origins.get(p).into_iter().flatten()
                    .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() }).collect();
                ev.sort_by(|a,b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
                return Err(ResolveError::Missing { package: p.clone(), evidence: ev });
            }
        }

        for (p, v) in &state.selected {
            if let Some(cs) = c.constraints.get(p) {
                if !cs.iter().all(|x| matches(*v, *x)) {
                    let ev = c.origins.get(p).into_iter().flatten()
                        .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() }).collect();
                    return Err(ResolveError::Conflict { package: p.clone(), selected: *v, requirements: cs.clone(), evidence: ev });
                }
            }
        }

        for (p, v) in &state.selected {
            let Some(pkg) = self.catalog.get(p) else { continue };
            let release = pkg.releases.iter().find(|r| r.version == *v).unwrap();
            let names = c.features.get(p).cloned().unwrap_or_default();
            for f in names {
                if !release.features.iter().any(|decl| decl.name == f) {
                    return Err(ResolveError::MissingFeature {
                        package: p.clone(),
                        selected: *v,
                        feature: f,
                        path: c.paths.get(p).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        let mut active: Vec<String> = c.constraints.keys().cloned().collect();
        active.sort();
        for p in &active {
            if state.selected.contains_key(p) { continue; }
            let cs = c.constraints.get(p).unwrap();
            let pkg = self.catalog.get(p).unwrap();
            let candidates: Vec<Version> = pkg.releases.iter().map(|r| r.version)
                .filter(|v| cs.iter().all(|x| matches(*v, *x))).collect();
            if candidates.is_empty() {
                let ev = c.origins.get(p).into_iter().flatten()
                    .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() }).collect();
                return Err(ResolveError::NoMatch { package: p.clone(), requirements: cs.clone(), evidence: ev });
            }
        }

        let next = active.into_iter().find(|p| !state.selected.contains_key(p));
        let Some(package) = next else {
            let selected: Vec<Selection> = state.selected.iter()
                .map(|(p, v)| Selection { package: p.clone(), version: *v }).collect();
            let cost = self.locks.iter().filter(|(p, v)| state.selected.get(*p) != Some(*v)).count();
            let replace = self.best.as_ref().map_or(true, |(best_cost, best)| {
                if cost != *best_cost { cost < *best_cost } else {
                    let names: BTreeSet<String> = self.catalog.keys().cloned().collect();
                    let a: BTreeMap<&str, Version> = selected.iter().map(|s| (s.package.as_str(), s.version)).collect();
                    let b: BTreeMap<&str, Version> = best.iter().map(|s| (s.package.as_str(), s.version)).collect();
                    for name in names {
                        match (a.get(name.as_str()), b.get(name.as_str())) {
                            (None, None) => {}
                            (None, Some(_)) => return false,
                            (Some(_), None) => return true,
                            (Some(av), Some(bv)) if av != bv => return av > bv,
                            _ => {}
                        }
                    }
                    false
                }
            });
            if replace {
                self.best = Some((cost, selected));
            }
            return Ok(());
        };

        let cs = c.constraints.get(&package).unwrap();
        let pkg = self.catalog.get(&package).unwrap();
        let mut candidates: Vec<Version> = pkg.releases.iter().map(|r| r.version)
            .filter(|v| cs.iter().all(|x| matches(*v, *x))).collect();
        candidates.sort_by(|a,b| b.cmp(a));
        if let Some(lock) = self.locks.get(&package) {
            if let Some(i) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        let mut branch_error = None;
        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit { attempts: self.budget });
            }
            self.attempts += 1;
            let mut child = state.clone();
            child.selected.insert(package.clone(), version);
            match self.inspect(child) {
                Ok(()) => {}
                Err(e @ ResolveError::Limit { .. }) => return Err(e),
                Err(e) => {
                    if branch_error.is_none() {
                        branch_error = Some(e);
                    }
                }
            }
        }
        if let Some(e) = branch_error { Err(e) } else {
            Err(ResolveError::NoMatch {
                package,
                requirements: cs.clone(),
                evidence: c.origins.get(cs.first().map(|_| "").unwrap_or("")).map(|_| vec![]).unwrap_or_default(),
            })
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]", i), name: r.package.clone() });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName { site: format!("root[{}].feature[{}]", i, f), name: name.clone() });
            }
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]", i), name: p.name.clone() });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].dep[{}]", i, j, k), name: d.package.clone() });
                }
                if let Some(g) = &d.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k), name: g.clone() });
                    }
                }
                for (f, name) in d.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f), name: name.clone() });
                    }
                }
            }
            for (f, feature) in r.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].feature[{}]", i, j, f), name: feature.name.clone() });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].feature[{}].enable[{}]", i, j, f, e), name: name.clone() });
                    }
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for p in catalog {
        if !package_names.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }
    for p in catalog {
        let mut seen = BTreeSet::new();
        for r in &p.releases {
            if !seen.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut seen = BTreeSet::new();
            for f in &r.features {
                if !seen.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature { package: p.name.clone(), version: r.version, feature: f.name.clone() });
                }
            }
        }
    }
    for (i, l) in locks.iter().enumerate() {
        if !valid_name(&l.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{}]", i), name: l.package.clone() });
        }
    }
    let mut lock_names = BTreeSet::new();
    for l in locks {
        if !lock_names.insert(l.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: l.package.clone() });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks.iter().map(|l| (l.package.clone(), l.version)).collect();
    let mut search = Search {
        roots,
        catalog: catalog_map,
        locks: lock_map,
        target,
        budget: max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let result = search.inspect(State::default());
    if let Err(e @ ResolveError::Limit { .. }) = result {
        return Err(e);
    }
    let selected = if let Some((_, selected)) = search.best {
        selected
    } else {
        return Err(result.err().or(search.first_error).unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: vec![],
            evidence: vec![],
        }));
    };

    let final_state = State { selected: selected.iter().map(|s| (s.package.clone(), s.version)).collect() };
    let final_closure = closure(roots, &search.catalog, &final_state, target);
    let selected_map: BTreeMap<String, Version> = selected.iter().map(|s| (s.package.clone(), s.version)).collect();
    let mut names: BTreeSet<String> = search.locks.keys().cloned().collect();
    names.extend(selected_map.keys().cloned());
    let mut changes = Vec::new();
    for name in names {
        let before = search.locks.get(&name).copied();
        let after = selected_map.get(&name).copied();
        if before == after { continue; }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search.catalog.get(&name).is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure.constraints.get(&name).is_some_and(|cs| !cs.iter().all(|c| matches(old, *c))) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change { package: name, before, after, reason });
    }
    Ok(Resolution { selected, changes })
}
