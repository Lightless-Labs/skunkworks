fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_key(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b as char, '.' | '_' | '-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_key(a.constraint)
        .cmp(&constraint_key(b.constraint))
        .then_with(|| a.path.cmp(&b.path))
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            let mut ev = evidence.clone();
            ev.sort_by(evidence_cmp);
            for e in ev {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            let mut ev = evidence.clone();
            ev.sort_by(evidence_cmp);
            for e in ev {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            let mut ev = evidence.clone();
            ev.sort_by(evidence_cmp);
            for e in ev {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone)]
struct Active {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    features: BTreeSet<String>,
    path: Vec<String>,
}

use std::collections::{BTreeMap, BTreeSet};

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>, BTreeMap<String, Active>)>,
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn release(&self, name: &str, version: Version) -> Option<&'a Release> {
        self.package(name)?
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn closure(&self, selected: &BTreeMap<String, Version>) -> BTreeMap<String, Active> {
        let mut active: BTreeMap<String, Active> = BTreeMap::new();
        for (i, root) in self.roots.iter().enumerate() {
            let fs: BTreeSet<String> = root.features.iter().cloned().collect();
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                root.package,
                constraint_text(root.constraint),
                fs.iter().cloned().collect::<Vec<_>>().join(",")
            );
            let a = active.entry(root.package.clone()).or_insert_with(|| Active {
                constraints: Vec::new(),
                origins: Vec::new(),
                features: BTreeSet::new(),
                path: vec![token.clone()],
            });
            a.constraints.push(root.constraint);
            a.origins.push(Origin {
                constraint: root.constraint,
                path: vec![token],
            });
            a.features.extend(fs);
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = active.keys().cloned().collect();
            for source in names {
                let Some(&version) = selected.get(&source) else {
                    continue;
                };
                let Some(release) = self.release(&source, version) else {
                    continue;
                };

                // Propagate locally enabled features to their least fixed point.
                let mut queue: Vec<String> = active[&source].features.iter().cloned().collect();
                let mut seen = BTreeSet::new();
                while let Some(feature_name) = queue.pop() {
                    if !seen.insert(feature_name.clone()) {
                        continue;
                    }
                    if let Some(decl) = release.features.iter().find(|f| f.name == feature_name) {
                        for en in &decl.enables {
                            if active.get_mut(&source).unwrap().features.insert(en.clone()) {
                                changed = true;
                            }
                            queue.push(en.clone());
                        }
                    }
                }

                let activated = active[&source].features.clone();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !target_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }
                    let dep_features: BTreeSet<String> = dep.features.iter().cloned().collect();
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        source,
                        version_text(version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        dep_features.iter().cloned().collect::<Vec<_>>().join(","),
                        dep.gate.as_deref().unwrap_or("*"),
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );
                    let source_path = active[&source].path.clone();
                    let mut path = source_path;
                    path.push(token);
                    let dest = active.entry(dep.package.clone()).or_insert_with(|| Active {
                        constraints: Vec::new(),
                        origins: Vec::new(),
                        features: BTreeSet::new(),
                        path: path.clone(),
                    });
                    let origin = Origin {
                        constraint: dep.constraint,
                        path,
                    };
                    if !dest.origins.iter().any(|o| {
                        o.constraint == origin.constraint && o.path == origin.path
                    }) {
                        dest.origins.push(origin);
                        changed = true;
                    }
                    if !dest.constraints.contains(&dep.constraint) {
                        dest.constraints.push(dep.constraint);
                        changed = true;
                    }
                    for f in dep_features {
                        if dest.features.insert(f) {
                            changed = true;
                        }
                    }
                }
            }

            // Canonical shortest paths can improve after a cycle or alternate path is found.
            let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
            for (i, root) in self.roots.iter().enumerate() {
                let fs: BTreeSet<String> = root.features.iter().cloned().collect();
                let p = vec![format!(
                    "root[{}]:{}@{}+{{{}}}",
                    i,
                    root.package,
                    constraint_text(root.constraint),
                    fs.into_iter().collect::<Vec<_>>().join(",")
                )];
                let ent = paths.entry(root.package.clone()).or_insert_with(|| p.clone());
                if path_better(&p, ent) {
                    *ent = p;
                    changed = true;
                }
            }
            let mut progress = true;
            while progress {
                progress = false;
                let current: Vec<(String, Vec<String>)> =
                    paths.iter().map(|(n, p)| (n.clone(), p.clone())).collect();
                for (source, source_path) in current {
                    let Some(&version) = selected.get(&source) else {
                        continue;
                    };
                    let Some(release) = self.release(&source, version) else {
                        continue;
                    };
                    let features = active.get(&source).map(|a| &a.features);
                    for (j, dep) in release.deps.iter().enumerate() {
                        if !target_matches(dep.platform, self.target)
                            || dep
                                .gate
                                .as_ref()
                                .is_some_and(|g| !features.is_some_and(|fs| fs.contains(g)))
                        {
                            continue;
                        }
                        let fs: BTreeSet<String> = dep.features.iter().cloned().collect();
                        let token = format!(
                            "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                            source,
                            version_text(version),
                            j,
                            dep.package,
                            constraint_text(dep.constraint),
                            fs.into_iter().collect::<Vec<_>>().join(","),
                            dep.gate.as_deref().unwrap_or("*"),
                            match dep.platform {
                                Platform::Any => "any",
                                Platform::Linux => "linux",
                                Platform::Mac => "mac",
                                Platform::Windows => "windows",
                            }
                        );
                        let mut candidate = source_path.clone();
                        candidate.push(token);
                        match paths.get_mut(&dep.package) {
                            Some(old) if path_better(&candidate, old) => {
                                *old = candidate;
                                progress = true;
                            }
                            None => {
                                paths.insert(dep.package.clone(), candidate);
                                progress = true;
                            }
                            _ => {}
                        }
                    }
                }
            }
            for (name, path) in paths {
                if let Some(a) = active.get_mut(&name) {
                    if a.path != path {
                        a.path = path;
                        changed = true;
                    }
                }
            }
            if !changed {
                break;
            }
        }

        for a in active.values_mut() {
            a.constraints = canonical_constraints(a.constraints.clone());
            a.origins.sort_by(|x, y| {
                constraint_key(x.constraint)
                    .cmp(&constraint_key(y.constraint))
                    .then_with(|| x.path.cmp(&y.path))
            });
        }
        active
    }

    fn dynamic_error(
        &self,
        selected: &BTreeMap<String, Version>,
        active: &BTreeMap<String, Active>,
    ) -> Option<ResolveError> {
        if let Some(name) = active.keys().find(|n| self.package(n).is_none()) {
            let ev = active[*name]
                .origins
                .iter()
                .map(|o| Evidence {
                    constraint: o.constraint,
                    path: o.path.clone(),
                })
                .collect();
            return Some(ResolveError::Missing {
                package: (*name).clone(),
                evidence: ev,
            });
        }

        for (name, version) in selected {
            let Some(a) = active.get(name) else {
                continue;
            };
            let missing = a.features.iter().find(|f| {
                self.release(name, *version)
                    .is_some_and(|r| !r.features.iter().any(|d| &d.name == *f))
            });
            if let Some(feature) = missing {
                return Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: a.path.clone(),
                });
            }
        }

        for (name, version) in selected {
            let Some(a) = active.get(name) else {
                continue;
            };
            if !a.constraints.iter().all(|c| matches(*version, *c)) {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: a.constraints.clone(),
                    evidence: a
                        .origins
                        .iter()
                        .map(|o| Evidence {
                            constraint: o.constraint,
                            path: o.path.clone(),
                        })
                        .collect(),
                });
            }
        }

        for (name, a) in active {
            if selected.contains_key(name) {
                continue;
            }
            let has_candidate = self.package(name).is_some_and(|p| {
                p.releases.iter().any(|r| {
                    a.constraints.iter().all(|c| matches(r.version, *c))
                })
            });
            if !has_candidate {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: a.constraints.clone(),
                    evidence: a
                        .origins
                        .iter()
                        .map(|o| Evidence {
                            constraint: o.constraint,
                            path: o.path.clone(),
                        })
                        .collect(),
                });
            }
        }
        None
    }

    fn solution_better(
        &self,
        selected: &BTreeMap<String, Version>,
        best: &(
            usize,
            Vec<Option<Version>>,
            BTreeMap<String, Version>,
            BTreeMap<String, Active>,
        ),
    ) -> bool {
        let cost = self
            .locks
            .iter()
            .filter(|l| selected.get(&l.package) != Some(&l.version))
            .count();
        let names: BTreeSet<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
        let vec: Vec<Option<Version>> = names
            .iter()
            .map(|n| selected.get(n).copied())
            .collect();
        cost < best.0 || (cost == best.0 && vec > best.1)
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = self.closure(selected);
        if let Some(err) = self.dynamic_error(selected, &active) {
            if self.first_error.is_none() {
                self.first_error = Some(err);
            }
            return Ok(());
        }

        let next = active.keys().find(|n| !selected.contains_key(*n)).cloned();
        let Some(name) = next else {
            if let Some(old) = &self.best {
                if self.solution_better(selected, old) {
                    let cost = self
                        .locks
                        .iter()
                        .filter(|l| selected.get(&l.package) != Some(&l.version))
                        .count();
                    let names: BTreeSet<String> =
                        self.catalog.iter().map(|p| p.name.clone()).collect();
                    self.best = Some((
                        cost,
                        names.iter().map(|n| selected.get(n).copied()).collect(),
                        selected.clone(),
                        active,
                    ));
                }
            } else {
                let cost = self
                    .locks
                    .iter()
                    .filter(|l| selected.get(&l.package) != Some(&l.version))
                    .count();
                let names: BTreeSet<String> =
                    self.catalog.iter().map(|p| p.name.clone()).collect();
                self.best = Some((
                    cost,
                    names.iter().map(|n| selected.get(n).copied()).collect(),
                    selected.clone(),
                    active,
                ));
            }
            return Ok(());
        };

        let a = &active[&name];
        let mut candidates: Vec<Version> = self
            .package(&name)
            .unwrap()
            .releases
            .iter()
            .filter(|r| a.constraints.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            let result = self.visit(selected);
            selected.remove(&name);
            result?;
        }
        Ok(())
    }
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn validate_names(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!(
                            "catalog[{}].release[{}].feature[{}]",
                            i, j, f
                        ),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;

    let mut packages = BTreeSet::new();
    for p in catalog {
        if !packages.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.as_str()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }
    let mut lock_names = BTreeSet::new();
    for l in locks {
        if !lock_names.insert(l.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: l.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
    };
    let mut selected = BTreeMap::new();
    search.visit(&mut selected)?;
    let Some((_, _, selected, active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec: Vec<Selection> = selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut before: BTreeMap<String, Version> = BTreeMap::new();
    for l in locks {
        before.insert(l.package.clone(), l.version);
    }
    let names: BTreeSet<String> = before
        .keys()
        .chain(selected.keys())
        .cloned()
        .collect();
    let mut changes = Vec::new();
    for name in names {
        let b = before.get(&name).copied();
        let a = selected.get(&name).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                if search
                    .package(&name)
                    .is_none_or(|p| !p.releases.iter().any(|r| r.version == old))
                {
                    ChangeReason::Unavailable
                } else if active.get(&name).is_some_and(|act| {
                    !act.constraints.iter().all(|c| matches(old, *c))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
