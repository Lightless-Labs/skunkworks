#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000_000, "quintillion"),
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
];

fn append_under_thousand(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_under_hundred(remainder, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

fn append_under_hundred(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut remainder = magnitude;
    let mut has_higher = false;

    for &(scale, name) in &SCALES {
        let group = remainder / scale;
        if group != 0 {
            if has_higher {
                out.push(' ');
            }
            append_under_thousand(group as u16, &mut out);
            out.push(' ');
            out.push_str(name);
            remainder %= scale;
            has_higher = true;
        }
    }

    if remainder != 0 {
        if has_higher {
            out.push(' ');
            if remainder < 100 {
                out.push_str("and ");
            }
        }
        append_under_thousand(remainder as u16, &mut out);
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<(u128, WordKind)> {
    if word == "zero" {
        return Some((0, WordKind::Number));
    }
    for (i, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some((i as u128, WordKind::Number));
        }
    }
    for (i, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(((20 + i * 10) as u128, WordKind::Number));
        }
    }
    if word == "hundred" {
        return Some((100, WordKind::Hundred));
    }
    if word == "and" {
        return Some((0, WordKind::And));
    }
    for &(scale, spelling) in &SCALES {
        if word == spelling {
            return Some((scale, WordKind::Scale));
        }
    }
    None
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(usize, &str)> = Vec::new();
    let mut start = 0;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    tokens.push((start, &input[start..]));

    for (i, &(position, token)) in tokens.iter().enumerate() {
        if token == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: position });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: position });
        }
    }

    let negative = tokens.first().is_some_and(|(_, token)| *token == "minus");
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(_, token) in arithmetic_tokens {
        if token == "minus" {
            unreachable!();
        }
        let (value, kind) = word_value(token).expect("dictionary scan validated tokens");
        match kind {
            WordKind::Number => group += value,
            WordKind::Hundred => group *= 100,
            WordKind::Scale => {
                total += group * value;
                group = 0;
            }
            WordKind::And => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != bytes {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(signed)
}
