#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty",
    "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000_000, "quintillion"),
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
];

fn spell_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_owned();
    }
    if n < 100 {
        let tens = TENS[(n / 10) as usize];
        let units = n % 10;
        return if units == 0 {
            tens.to_owned()
        } else {
            format!("{tens}-{}", SMALL[units as usize])
        };
    }

    let hundreds = n / 100;
    let remainder = n % 100;
    if remainder == 0 {
        format!("{} hundred", SMALL[hundreds as usize])
    } else {
        format!(
            "{} hundred and {}",
            SMALL[hundreds as usize],
            spell_under_thousand(remainder)
        )
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_owned());
    }

    let mut remainder = magnitude;
    let mut parts = Vec::new();
    let mut has_higher_group = false;

    for &(scale, name) in &SCALES {
        let group = remainder / scale;
        remainder %= scale;
        if group != 0 {
            parts.push(format!("{} {name}", spell_under_thousand(group as u16)));
            has_higher_group = true;
        }
    }

    if remainder != 0 {
        let units = spell_under_thousand(remainder as u16);
        if has_higher_group && remainder < 100 {
            parts.push(format!("and {units}"));
        } else {
            parts.push(units);
        }
    }

    let spelling = parts.join(" ");
    if value < 0 {
        Ok(format!("minus {spelling}"))
    } else {
        Ok(spelling)
    }
}

fn word_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }
    if let Some(index) = SMALL.iter().position(|&candidate| candidate == word) {
        if index > 0 {
            return Some(index as u128);
        }
    }
    if let Some(index) = TENS.iter().position(|&candidate| candidate == word) {
        if index >= 2 {
            return Some((index * 10) as u128);
        }
    }

    match word {
        "hundred" => Some(100),
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        "and" | "minus" => Some(0),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (index, &(word, position)) in tokens.iter().enumerate() {
        if word == "minus" {
            if index != 0 {
                return Err(WordsError::InvalidWord { index: position });
            }
        } else if word_value(word).is_none() {
            return Err(WordsError::InvalidWord { index: position });
        }
    }

    let negative = tokens.first().is_some_and(|(word, _)| *word == "minus");
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(word, _) in arithmetic_tokens {
        match word {
            "and" => {}
            "hundred" => group = group.saturating_mul(100),
            "thousand" => {
                total = total.saturating_add(group.saturating_mul(1_000));
                group = 0;
            }
            "million" => {
                total = total.saturating_add(group.saturating_mul(1_000_000));
                group = 0;
            }
            "billion" => {
                total = total.saturating_add(group.saturating_mul(1_000_000_000));
                group = 0;
            }
            "trillion" => {
                total = total.saturating_add(group.saturating_mul(1_000_000_000_000));
                group = 0;
            }
            "quadrillion" => {
                total = total.saturating_add(group.saturating_mul(1_000_000_000_000_000));
                group = 0;
            }
            "quintillion" => {
                total = total.saturating_add(group.saturating_mul(LIMIT));
                group = 0;
            }
            _ => group = group.saturating_add(word_value(word).unwrap_or(0)),
        }

        if total.saturating_add(group) > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.saturating_add(group);
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let mismatch = canonical
            .bytes()
            .zip(bytes.iter().copied())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(value)
}
