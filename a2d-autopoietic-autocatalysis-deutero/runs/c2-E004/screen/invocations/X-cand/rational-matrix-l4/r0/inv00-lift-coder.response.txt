use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if b[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(err(i, if i == b.len() {
                ParseKind::InvalidByte
            } else {
                ParseKind::InvalidByte
            }));
        }

        let slash = if i < b.len() && b[i] == b'/' {
            let s = i;
            i += 1;
            let den_start = i;
            while i < b.len() && b[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start {
                return Err(err(i, ParseKind::InvalidByte));
            }
            Some((s, den_start, i))
        } else {
            None
        };

        if i != b.len() {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let num_end = slash.map_or(b.len(), |x| x.0);
        let negative = b[0] == b'-';
        let digits_start = usize::from(negative);
        if num_end - digits_start >= 2 && b[digits_start] == b'0' {
            return Err(err(digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && &b[..num_end] == b"-0" {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some((_, ds, de)) = slash {
            if de - ds >= 2 && b[ds] == b'0' {
                return Err(err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let mut nmag: u128 = 0;
        for &c in &b[digits_start..num_end] {
            nmag = nmag.saturating_mul(10).saturating_add((c - b'0') as u128);
            let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
            if nmag > limit {
                return Err(err(0, ParseKind::OutOfRange));
            }
        }

        let (den, den_at) = if let Some((_, ds, de)) = slash {
            let mut value: u128 = 0;
            for &c in &b[ds..de] {
                value = value.saturating_mul(10).saturating_add((c - b'0') as u128);
                if value > i64::MAX as u128 {
                    return Err(err(ds, ParseKind::OutOfRange));
                }
            }
            (value as i64, ds)
        } else {
            (1, 0)
        };

        if slash.is_some() && den == 0 {
            return Err(err(den_at, ParseKind::ZeroDenominator));
        }
        if let Some((s, _, _)) = slash {
            if den == 1 {
                return Err(err(s, ParseKind::NonCanonical));
            }
            if nmag == 0 {
                return Err(err(s, ParseKind::NonCanonical));
            }
            if gcd_u128(nmag, den as u128) > 1 {
                return Err(err(s, ParseKind::NonCanonical));
            }
        }

        let signed = if negative {
            if nmag == (1u128 << 63) {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };
        Ok(Rational {
            num: signed,
            den,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let size = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(size);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let product = self.data[r * self.cols + k]
                        .mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: k,
                            source,
                        })?;
                    acc = acc.add(&product).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        data: &mut [Rational],
        rows: usize,
        width: usize,
        candidates: usize,
    ) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;
        for col in 0..candidates {
            if p == rows {
                break;
            }
            let mut best: Option<usize> = None;
            for r in p..rows {
                let v = data[r * width + col];
                if v.num == 0 {
                    continue;
                }
                let key = (v.den, v.num.unsigned_abs(), r);
                if best.is_none_or(|br| {
                    let b = data[br * width + col];
                    key < (b.den, b.num.unsigned_abs(), br)
                }) {
                    best = Some(r);
                }
            }
            let Some(q) = best else {
                continue;
            };
            if q != p {
                for c in 0..width {
                    data.swap(p * width + c, q * width + c);
                }
                swaps.push((p, q));
            }

            let pivot = data[p * width + col];
            for c in 0..width {
                let original = data[p * width + c];
                data[p * width + c] =
                    original
                        .div(&pivot)
                        .map_err(|source| MatrixError::Arithmetic {
                            row: p,
                            col: c,
                            step: p,
                            source,
                        })?;
            }

            for r in 0..rows {
                if r == p {
                    continue;
                }
                let factor = data[r * width + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..width {
                    let product = factor.mul(&data[p * width + c]).map_err(|source| {
                        MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                    data[r * width + c] = data[r * width + c].sub(&product).map_err(|source| {
                        MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                }
            }

            pivots.push(col);
            p += 1;
        }
        Ok((pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut data = self.data.clone();
        let (pivots, swaps) = Self::eliminate(&mut data, self.rows, self.cols, self.cols)?;
        Ok(Reduction {
            matrix: Self {
                rows: self.rows,
                cols: self.cols,
                data,
            },
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let size = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(size);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { one } else { zero });
            }
        }
        let (pivots, _) = Self::eliminate(&mut data, n, width, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut right = Vec::with_capacity(size / 2);
        for r in 0..n {
            right.extend_from_slice(&data[r * width + n..r * width + width]);
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: right,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let width = self
            .cols
            .checked_add(rhs.cols)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let size = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(size);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * self.cols..(r + 1) * self.cols]);
            data.extend_from_slice(&rhs.data[r * rhs.cols..(r + 1) * rhs.cols]);
        }
        let (pivots, _) =
            Self::eliminate(&mut data, self.rows, width, self.cols).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..self.cols).all(|c| data[r * width + c].num == 0);
            if coeff_zero {
                if let Some(c) = (0..rhs.cols).find(|&c| data[r * width + self.cols + c].num != 0) {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col: c,
                        witness: data[r * width..r * width + width].to_vec(),
                    });
                }
            }
        }

        let free: Vec<usize> = (0..self.cols)
            .filter(|c| !pivots.contains(c))
            .collect();
        let particular_size = self
            .cols
            .checked_mul(rhs.cols)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_size];
        for (i, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..rhs.cols {
                particular[pivot_col * rhs.cols + c] =
                    data[i * width + self.cols + c];
            }
        }
        let particular = Self {
            rows: self.cols,
            cols: rhs.cols,
            data: particular,
        };

        if pivots.len() == self.cols {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let basis_size = self
            .cols
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_size];
        let one = Rational { num: 1, den: 1 };
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * free.len() + j] = one;
            for (i, &pivot_col) in pivots.iter().enumerate() {
                let value = data[i * width + free_col];
                basis[pivot_col * free.len() + j] =
                    value.neg().map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: j,
                            step: pivots.len(),
                            source,
                        })
                    })?;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self {
                rows: self.cols,
                cols: basis_size / self.cols.max(1),
                data: basis,
            },
            pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
