use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_wide(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let neg = (num < 0) ^ (den < 0);
    let n = num.unsigned_abs();
    let d = den.unsigned_abs();
    let g = gcd_u128(n, d);
    let rn = n / g;
    let rd = d / g;

    if rd == 0 || rd > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    let max_num = if neg { 1u128 << 63 } else { i64::MAX as u128 };
    if rn > max_num {
        return Err(RationalError::Overflow);
    }

    let signed_num = if neg {
        if rn == 1u128 << 63 {
            i64::MIN
        } else {
            -(rn as i64)
        }
    } else {
        rn as i64
    };

    Ok(Rational {
        num: signed_num,
        den: rd as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_wide(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_wide(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_wide(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_wide(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_wide(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_wide(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let slash = b.iter().position(|&x| x == b'/');
        let num_end = slash.unwrap_or(b.len());
        let mut i = 0;
        if b[0] == b'-' {
            i = 1;
        }
        let num_digits_start = i;
        while i < num_end && b[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_digits_start {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }
        if i < num_end {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let mut den_start = None;
        if let Some(s) = slash {
            let start = s + 1;
            den_start = Some(start);
            if start == b.len() {
                return Err(RationalError::Parse {
                    at: b.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
            let mut j = start;
            while j < b.len() && b[j].is_ascii_digit() {
                j += 1;
            }
            if j < b.len() {
                return Err(RationalError::Parse {
                    at: j,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        if num_end - num_digits_start >= 2 && b[num_digits_start] == b'0' {
            return Err(RationalError::Parse {
                at: num_digits_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if b[0] == b'-' && num_end == 2 && b[1] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }
        if let Some(ds) = den_start {
            if b.len() - ds >= 2 && b[ds] == b'0' {
                return Err(RationalError::Parse {
                    at: ds + 1,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let negative = b[0] == b'-';
        let mut nmag: u128 = 0;
        for &ch in &b[num_digits_start..num_end] {
            nmag = nmag.saturating_mul(10).saturating_add((ch - b'0') as u128);
            let lim = if negative { 1u128 << 63 } else { i64::MAX as u128 };
            if nmag > lim {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: ParseKind::OutOfRange,
                });
            }
        }
        let num = if negative {
            if nmag == 1u128 << 63 {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        let mut den: i64 = 1;
        if let Some(ds) = den_start {
            let mut dmag: u128 = 0;
            for &ch in &b[ds..] {
                dmag = dmag.saturating_mul(10).saturating_add((ch - b'0') as u128);
                if dmag > i64::MAX as u128 {
                    return Err(RationalError::Parse {
                        at: ds,
                        kind: ParseKind::OutOfRange,
                    });
                }
            }
            if dmag == 0 {
                return Err(RationalError::Parse {
                    at: ds,
                    kind: ParseKind::ZeroDenominator,
                });
            }
            den = dmag as i64;
            if den == 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if num == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(num.unsigned_abs() as u128, den as u128) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        Ok(Rational { num, den })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(&self, pivot_cols: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_cols {
            if p == m.rows {
                break;
            }
            let mut best: Option<(i64, u128, usize)> = None;
            for row in p..m.rows {
                let v = m.data[row * m.cols + col];
                if v.num == 0 {
                    continue;
                }
                let key = (v.den, v.num.unsigned_abs() as u128, row);
                if best.map_or(true, |x| key < x) {
                    best = Some(key);
                }
            }
            let Some((_, _, q)) = best else {
                continue;
            };
            if q != p {
                for c in 0..m.cols {
                    m.data.swap(p * m.cols + c, q * m.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = m.data[p * m.cols + col];
            for c in 0..m.cols {
                let v = m.data[p * m.cols + c]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
                m.data[p * m.cols + c] = v;
            }

            for row in 0..m.rows {
                if row == p {
                    continue;
                }
                let factor = m.data[row * m.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..m.cols {
                    let product = factor
                        .mul(&m.data[p * m.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    let value = m.data[row * m.cols + c]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    m.data[row * m.cols + c] = value;
                }
            }
            pivots.push(col);
            p += 1;
        }

        Ok(Reduction {
            matrix: m,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let aug = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: red.pivots.len(),
            });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: out,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(len);
        for r in 0..m {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            data.extend_from_slice(&rhs.data[r * k..(r + 1) * k]);
        }

        let aug = Matrix {
            rows: m,
            cols: width,
            data,
        };
        let red = aug.eliminate(n).map_err(SolveError::Matrix)?;

        for row in 0..m {
            let coeff_zero = (0..n).all(|c| red.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                if let Some(rhs_col) = (0..k).find(|&c| red.matrix.data[row * width + n + c].num != 0)
                {
                    let witness = red.matrix.data[row * width..(row + 1) * width].to_vec();
                    return Ok(Solution::None {
                        row,
                        rhs_col,
                        witness,
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular = vec![zero; particular_len];
        for (j, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] = red.matrix.data[j * width + n + c];
            }
        }

        let free: Vec<usize> = (0..n)
            .filter(|c| !red.pivots.contains(c))
            .collect();

        if rank == n {
            return Ok(Solution::Unique {
                values: Matrix {
                    rows: n,
                    cols: k,
                    data: particular,
                },
                pivots: red.pivots,
            });
        }

        let basis_len = n
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero; basis_len];
        let one = Rational { num: 1, den: 1 };

        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * free.len() + j] = one;
            for (pivot_row, &pivot_col) in red.pivots.iter().enumerate() {
                let value = red.matrix.data[pivot_row * width + free_col]
                    .neg()
                    .map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: j,
                            step: rank,
                            source,
                        })
                    })?;
                basis[pivot_col * free.len() + j] = value;
            }
        }

        Ok(Solution::Infinite {
            particular: Matrix {
                rows: n,
                cols: k,
                data: particular,
            },
            free,
            basis: Matrix {
                rows: n,
                cols: free.len(),
                data: basis,
            },
            pivots: red.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
