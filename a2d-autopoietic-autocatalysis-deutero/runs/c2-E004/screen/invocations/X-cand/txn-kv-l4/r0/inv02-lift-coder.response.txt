```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    value: Option<String>,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Write {
    value: Option<String>,
    deadline: Option<u64>,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, (String, Option<u64>)>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn visible_write(&self, tx: Option<u64>, key: &str) -> Option<Write> {
        tx.and_then(|id| self.transactions.get(&id))
            .and_then(|t| t.writes.get(key))
            .cloned()
    }

    fn selected_version(&self, tx: Option<u64>, key: &str) -> Option<&Version> {
        let versions = self.data.get(key)?;
        if let Some(id) = tx {
            let begin_revision = self.transactions.get(&id)?.begin_revision;
            versions
                .iter()
                .rev()
                .find(|version| version.revision <= begin_revision)
        } else {
            versions.last()
        }
    }

    fn visible_entry(
        &self,
        tx: Option<u64>,
        key: &str,
        now: u64,
    ) -> Option<(String, Option<u64>)> {
        if let Some(write) = self.visible_write(tx, key) {
            return match write.value {
                Some(value) if write.deadline.map_or(true, |d| d > now) => {
                    Some((value, write.deadline))
                }
                _ => None,
            };
        }
        let version = self.selected_version(tx, key)?;
        match &version.value {
            Some(value) if version.deadline.map_or(true, |d| d > now) => {
                Some((value.clone(), version.deadline))
            }
            _ => None,
        }
    }

    fn all_keys(&self) -> BTreeSet<String> {
        self.data.keys().cloned().collect()
    }

    fn visible_keys(&self, tx: Option<u64>) -> BTreeSet<String> {
        let mut keys = self.all_keys();
        if let Some(id) = tx {
            if let Some(t) = self.transactions.get(&id) {
                keys.extend(t.writes.keys().cloned());
            }
        }
        keys
    }

    fn append(&mut self, key: String, write: Write, revision: u64) {
        self.data.entry(key).or_default().push(Version {
            revision,
            value: write.value,
            deadline: write.deadline,
        });
    }

    fn key_check(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn name_check(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let tx = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;

        for key in tx.writes.keys() {
            if self
                .data
                .get(key)
                .and_then(|versions| versions.last())
                .is_some_and(|v| v.revision > tx.begin_revision)
            {
                return Err(Error::Conflict {
                    tx: id,
                    key: key.clone(),
                });
            }
        }

        let tx = self.transactions.remove(&id).expect("transaction checked");
        if tx.writes.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        for (key, write) in tx.writes {
            self.append(key, write, self.revision);
        }
        Ok(self.revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        self.transactions
            .remove(&id)
            .map(|_| ())
            .ok_or(Error::UnknownTransaction(id))
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::key_check(key)?;

        if let Some(id) = tx {
            if self.transactions.get(&id).is_some_and(|t| t.read_only) {
                return Err(Error::ReadOnly(id));
            }
        }

        let write = match ttl {
            Some(0) => Write {
                value: None,
                deadline: None,
            },
            Some(duration) => Write {
                value: Some(value.to_string()),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            },
            None => Write {
                value: Some(value.to_string()),
                deadline: None,
            },
        };

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .expect("transaction checked")
                .writes
                .insert(key.to_string(), write);
        } else {
            self.revision += 1;
            self.append(key.to_string(), write, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::key_check(key)?;

        if let Some(id) = tx {
            if self.transactions.get(&id).is_some_and(|t| t.read_only) {
                return Err(Error::ReadOnly(id));
            }
        }

        let existed = self.visible_entry(tx, key, now).is_some();
        let write = Write {
            value: None,
            deadline: None,
        };
        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .expect("transaction checked")
                .writes
                .insert(key.to_string(), write);
        } else {
            self.revision += 1;
            self.append(key.to_string(), write, self.revision);
        }
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::key_check(key)?;
        Ok(self.visible_entry(tx, key, now).map(|(value, _)| value))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self
            .visible_keys(tx)
            .iter()
            .filter(|key| {
                self.visible_entry(tx, key, now)
                    .is_some_and(|(current, _)| current == value)
            })
            .count())
    }

    fn range_check(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::range_check(start, end)?;

        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut entries = Vec::new();
        for key in self.visible_keys(tx) {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            if cursor.is_some_and(|c| key <= c.last) {
                continue;
            }
            if let Some((value, _)) = self.visible_entry(tx, &key, now) {
                entries.push((key, value));
            }
        }

        let skip = if cursor.is_some() { 0 } else { offset };
        let items: Vec<_> = entries.into_iter().skip(skip).take(limit).collect();
        let next = items.last().and_then(|(last, _)| {
            let has_more = self
                .visible_keys(tx)
                .iter()
                .any(|key| {
                    Self::matches(key, start, end, prefix)
                        && key > last
                        && cursor.map_or(true, |c| key > c.last)
                        && self.visible_entry(tx, key, now).is_some()
                });
            has_more.then(|| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        });

        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::range_check(start, end)?;

        let mut result = Aggregate { count: 0, sum: 0 };
        for key in self.visible_keys(tx) {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            let Some((value, _)) = self.visible_entry(tx, &key, now) else {
                continue;
            };
            let parsed = Self::parse_number(&key, &value)?;
            result.sum = result.sum.checked_add(parsed).ok_or_else(|| Error::SumOverflow {
                key: key.clone(),
            })?;
            result.count += 1;
        }
        Ok(result)
    }

    fn parse_number(key: &str, value: &str) -> Result<i64, Error> {
        let bytes = value.as_bytes();
        let valid = if bytes == b"0" {
            true
        } else if bytes.first() == Some(&b'-') {
            bytes.len() >= 2
                && bytes[1] >= b'1'
                && bytes[1] <= b'9'
                && bytes[2..].iter().all(u8::is_ascii_digit)
        } else {
            !bytes.is_empty()
                && bytes[0] >= b'1'
                && bytes[0] <= b'9'
                && bytes[1..].iter().all(u8::is_ascii_digit)
        };
        if !valid {
            return Err(Error::InvalidNumber {
                key: key.to_string(),
                value: value.to_string(),
            });
        }
        value.parse::<i64>().map_err(|_| Error::InvalidNumber {
            key: key.to_string(),
            value: value.to_string(),
        })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::name_check(name)?;
        if self.snapshots.len() >= 3 && !self.snapshots.contains_key(name) {
            return Err(Error::SnapshotLimit);
        }
        let mut copy = BTreeMap::new();
        for key in self.all_keys() {
            if let Some((value, deadline)) = self.visible_entry(None, &key, now) {
                copy.insert(key, (value, deadline));
            }
        }
        self.snapshots.insert(name.to_string(), copy);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        Self::name_check(name)?;
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let saved_live: BTreeMap<String, (String, Option<u64>)> = saved
            .into_iter()
            .filter(|(_, (_, deadline))| deadline.map_or(true, |d| d > now))
            .collect();

        let mut current_live = BTreeMap::new();
        for key in self.all_keys() {
            if let Some((value, deadline)) = self.visible_entry(None, &key, now) {
                current_live.insert(key, (value, deadline));
            }
        }

        let union: BTreeSet<String> = current_live
            .keys()
            .chain(saved_live.keys())
            .cloned()
            .collect();
        if union.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        for key in union {
            let write = match saved_live.get(&key) {
                Some((value, deadline)) => Write {
                    value: Some(value.clone()),
                    deadline: *deadline,
                },
                None => Write {
                    value: None,
                    deadline: None,
                },
            };
            self.append(key, write, self.revision);
        }
        Ok(self.revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::name_check(name)?;
        self.snapshots
            .remove(name)
            .map(|_| ())
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;

        let begin_revisions: Vec<u64> = self
            .transactions
            .values()
            .map(|t| t.begin_revision)
            .collect();

        let mut removed = 0;
        for versions in self.data.values_mut() {
            let mut retain = BTreeSet::new();
            if !versions.is_empty() {
                retain.insert(versions.len() - 1);
            }
            for begin_revision in &begin_revisions {
                if let Some(index) = versions
                    .iter()
                    .rposition(|v| v.revision <= *begin_revision)
                {
                    retain.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let keep = retain.contains(&index);
                index += 1;
                if !keep {
                    removed += 1;
                }
                keep
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let versions: usize = self.data.values().map(Vec::len).sum();
        let tombstones: usize = self
            .data
            .values()
            .flat_map(|versions| versions.iter())
            .filter(|version| version.value.is_none())
            .count();

        let live_keys = self
            .all_keys()
            .iter()
            .filter(|key| self.visible_entry(None, key, now).is_some())
            .count();

        Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self
                .transactions
                .values()
                .filter(|t| t.read_only)
                .count(),
            staged_writes: self
                .transactions
                .values()
                .map(|t| t.writes.len())
                .sum(),
            snapshots: self.snapshots.len(),
        }
        .pipe(Ok)
    }
}

trait Pipe: Sized {
    fn pipe<T>(self, f: impl FnOnce(Self) -> T) -> T {
        f(self)
    }
}
impl<T> Pipe for T {}
```
