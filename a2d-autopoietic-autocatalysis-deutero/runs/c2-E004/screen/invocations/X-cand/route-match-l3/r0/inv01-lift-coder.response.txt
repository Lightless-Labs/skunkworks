#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(input: &str, base: usize) -> Result<String, usize> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_path_head<'a>(input: &'a str, pattern: bool) -> Result<(Vec<(usize, &'a str)>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut parts = Vec::new();
    let mut start = 1usize;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(start);
                }
                parts.push((start, &input[start..i]));
                return Ok((parts, true));
            }
            if start == i {
                return Err(start);
            }
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(start);
    }
    parts.push((start, &input[start..]));
    let _ = pattern;
    Ok((parts, false))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (parts, trailing) =
            parse_path_head(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        let mut segments = Vec::with_capacity(parts.len());
        let mut names = Vec::<String>::new();
        let mut rank = Vec::new();

        for (index, (at, part)) in parts.iter().enumerate() {
            if let Some(name) = part.strip_prefix(':') {
                if !valid_capture_name(name) || names.iter().any(|n| n == name) {
                    return Err(Error::InvalidPattern { at: *at });
                }
                names.push(name.to_string());
                segments.push(Segment::Param(name.to_string()));
                rank.push(1);
            } else if let Some(name) = part.strip_prefix('*') {
                if index + 1 != parts.len() {
                    return Err(Error::InvalidPattern { at: *at });
                }
                if !valid_capture_name(name) || names.iter().any(|n| n == name) {
                    return Err(Error::InvalidPattern { at: *at });
                }
                names.push(name.to_string());
                segments.push(Segment::Wildcard(name.to_string()));
                rank.push(0);
            } else {
                let decoded = decode_segment(part, *at)
                    .map_err(|at| Error::InvalidPattern { at })?;
                segments.push(Segment::Static(decoded));
                rank.push(2);
            }
        }

        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];

        let (parts, trailing) =
            parse_path_head(head, false).map_err(|at| Error::InvalidPath { at })?;
        let mut decoded = Vec::with_capacity(parts.len());
        for (at, part) in parts {
            decoded.push(decode_segment(part, at).map_err(|at| Error::InvalidPath { at })?);
        }

        let mut candidates = Vec::<(&Route, Vec<(String, String)>)>::new();
        for route in &self.routes {
            let mut params = Vec::new();
            let mut path_index = 0usize;
            let mut matched = true;

            for segment in &route.segments {
                match segment {
                    Segment::Static(value) => {
                        if decoded.get(path_index) != Some(value) {
                            matched = false;
                            break;
                        }
                        path_index += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = decoded.get(path_index) {
                            params.push((name.clone(), value.clone()));
                            path_index += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = decoded[path_index..].join("/");
                        params.push((name.clone(), value));
                        path_index = decoded.len();
                        break;
                    }
                }
            }

            if !matched || path_index != decoded.len() {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            candidates.push((route, params));
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<_> = if method == "HEAD" {
            let has_head = candidates.iter().any(|(r, _)| r.method == "HEAD");
            candidates
                .into_iter()
                .filter(|(r, _)| r.method == if has_head { "HEAD" } else { "GET" })
                .collect()
        } else {
            candidates
                .into_iter()
                .filter(|(r, _)| r.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = self
                .routes
                .iter()
                .filter(|r| {
                    // Candidate methods are collected below from path matches; this branch
                    // is retained for the set computed before method filtering.
                    let _ = r;
                    false
                })
                .map(|r| r.method.clone())
                .collect();
            let _ = &mut allowed;
            return Err(Error::MethodNotAllowed {
                allowed: Vec::new(),
            });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id)))
            .expect("eligible candidates are nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && !winner.segments.is_empty()
            && winner.trailing != trailing
        {
            if winner.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params,
            redirect,
        })
    }
}
