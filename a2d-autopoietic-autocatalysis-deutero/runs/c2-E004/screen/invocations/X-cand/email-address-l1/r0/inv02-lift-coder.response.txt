#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index));

    let separator = match (separators.next(), separators.next()) {
        (None, _) => return Err(EmailError::MissingAt),
        (Some(_), Some(second)) => return Err(EmailError::MultipleAt { at: second }),
        (Some(first), None) => first,
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let allowed_special = b"!#$%&'*+-/=?^_`{|}~";
    for (index, &byte) in local_bytes.iter().enumerate() {
        if byte == b'.' {
            if index == 0
                || index + 1 == local_bytes.len()
                || local_bytes[index - 1] == b'.'
            {
                return Err(EmailError::LocalDot { at: index });
            }
        } else if !byte.is_ascii_alphanumeric() && !allowed_special.contains(&byte) {
            return Err(EmailError::LocalChar { at: index });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let removed_dot = raw_domain.last() == Some(&b'.');
    let domain_bytes = if removed_dot {
        &raw_domain[..raw_domain.len() - 1]
    } else {
        raw_domain
    };
    let domain_len = domain_bytes.len();
    let root_dot_at = input.len() - 1;

    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ranges = Vec::new();
    let mut start = 0;

    for index in 0..=domain_len {
        if index == domain_len || domain_bytes[index] == b'.' {
            let label = &domain_bytes[start..index];
            let label_start = domain_start + start;

            if label.is_empty() {
                let at = if index == domain_len && removed_dot {
                    root_dot_at
                } else {
                    label_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: label_start + 63,
                });
            }

            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_start });
            }

            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: label_start + label.len() - 1,
                });
            }

            for (offset, &byte) in label.iter().enumerate() {
                if !byte.is_ascii_alphanumeric() && byte != b'-' {
                    return Err(EmailError::DomainChar {
                        at: label_start + offset,
                    });
                }
            }

            label_starts.push(label_start);
            label_ranges.push(start..index);
            start = index + 1;
        }
    }

    if label_ranges.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_range = label_ranges.last().expect("validated nonempty labels");
    let last_label = &domain_bytes[last_range.clone()];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_range.start,
        });
    }

    let local = std::str::from_utf8(local_bytes)
        .expect("ASCII input")
        .to_owned();
    let domain = domain_bytes
        .iter()
        .map(|byte| byte.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}
