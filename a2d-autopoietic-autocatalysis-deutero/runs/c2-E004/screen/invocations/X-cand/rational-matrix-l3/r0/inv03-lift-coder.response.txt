```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }

    let gcd = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= gcd;
    d /= gcd;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_parts(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_parts(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_parts(
            (self.num as i128) * (rhs.num as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_parts(
            (self.num as i128) * (rhs.den as i128),
            (self.den as i128) * (rhs.num as i128),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        // Validate the full grammar before checking canonicality or magnitude.
        let slash = bytes.iter().position(|&b| b == b'/');
        let numerator_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');

        if num_start == numerator_end {
            return Err(parse_error(numerator_end, ParseKind::InvalidByte));
        }

        for i in 0..numerator_end {
            if i == 0 && bytes[i] == b'-' {
                continue;
            }
            if !bytes[i].is_ascii_digit() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(parse_error(bytes.len(), ParseKind::InvalidByte));
            }
            for i in (s + 1)..bytes.len() {
                if !bytes[i].is_ascii_digit() {
                    return Err(parse_error(i, ParseKind::InvalidByte));
                }
            }
        }

        let num_digits = numerator_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(parse_error(num_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == 1 && bytes[1] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        let den_start = slash.map(|s| s + 1);
        if let Some(ds) = den_start {
            if bytes.len() - ds >= 2 && bytes[ds] == b'0' {
                return Err(parse_error(ds + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let num_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };

        let mut num_mag = 0u128;
        for &b in &bytes[num_start..numerator_end] {
            let digit = (b - b'0') as u128;
            if num_mag > (num_limit - digit) / 10 {
                return Err(parse_error(0, ParseKind::OutOfRange));
            }
            num_mag = num_mag * 10 + digit;
        }

        let mut den_mag = 1u128;
        if let Some(ds) = den_start {
            for &b in &bytes[ds..] {
                let digit = (b - b'0') as u128;
                if den_mag > (i64::MAX as u128 - digit) / 10 {
                    return Err(parse_error(ds, ParseKind::OutOfRange));
                }
                den_mag = den_mag * 10 + digit;
            }

            if den_mag == 0 {
                return Err(parse_error(ds, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(parse_error(slash.unwrap(), ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_error(slash.unwrap(), ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(parse_error(slash.unwrap(), ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            -(num_mag as i128)
        } else {
            num_mag as i128
        };

        rational_from_parts(signed_num, den_mag as i128)
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };

        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let lhs = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = lhs.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        reduce(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };

        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { one } else { zero });
            }
        }

        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduction = reduce(augmented, n)?;
        let rank = reduction.pivots.len();
        if rank < n {
            return Err(MatrixError::Singular { rank });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                inverse_data.push(reduction.matrix.data[r * width + n + c]);
            }
        }

        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

fn reduce(mut matrix: Matrix, pivot_cols: usize) -> Result<Reduction, MatrixError> {
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;
    let rows = matrix.rows;
    let cols = matrix.cols;

    for c in 0..pivot_cols {
        if p == rows {
            break;
        }

        let mut chosen: Option<usize> = None;
        for r in p..rows {
            let value = matrix.data[r * cols + c];
            if value.num == 0 {
                continue;
            }
            let key = (value.den, value.num.unsigned_abs(), r);
            if chosen
                .map(|q| {
                    let old = matrix.data[q * cols + c];
                    key < (old.den, old.num.unsigned_abs(), q)
                })
                .unwrap_or(true)
            {
                chosen = Some(r);
            }
        }

        let Some(q) = chosen else {
            continue;
        };

        if q != p {
            for j in 0..cols {
                matrix.data.swap(p * cols + j, q * cols + j);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + c];
        for j in 0..cols {
            let original = matrix.data[p * cols + j];
            matrix.data[p * cols + j] =
                original
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: j,
                        step: p,
                        source,
                    })?;
        }

        for r in 0..rows {
            if r == p {
                continue;
            }
            let factor = matrix.data[r * cols + c];
            if factor.num == 0 {
                continue;
            }
            for j in 0..cols {
                let product = factor
                    .mul(&matrix.data[p * cols + j])
                    .map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: j,
                        step: p,
                        source,
                    })?;
                matrix.data[r * cols + j] = matrix.data[r * cols + j]
                    .sub(&product)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: j,
                        step: p,
                        source,
                    })?;
            }
        }

        pivots.push(c);
        p += 1;
    }

    Ok(Reduction {
        matrix,
        pivots,
        swaps,
    })
}
```
