use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut fields = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }

        if pos == start {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut value = 0u64;
        for &b in &bytes[start..pos] {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((b - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        fields[i] = value;

        if i < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let start = pos;
        while pos < bytes.len() && bytes[pos] != b'+' {
            pos += 1;
        }
        parse_identifiers(s, start, pos, true, &mut pre)?;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: fields[0],
        minor: fields[1],
        patch: fields[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
    output: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        if prerelease
            && bytes[id_start..i].iter().all(|b| is_digit(*b))
            && i - id_start > 1
            && bytes[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        output.push(s[id_start..i].to_owned());
        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(())
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core_order = core(a).cmp(&core(b));
    if core_order != Ordering::Equal {
        return core_order;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let xn = numeric_identifier(x);
                let yn = numeric_identifier(y);
                let order = match (xn, yn) {
                    (true, true) => compare_numeric_identifiers(x, y),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    (false, false) => x.as_bytes().cmp(y.as_bytes()),
                };
                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    admission: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

impl Comparator {
    fn accepts(&self, candidate: &Version) -> bool {
        match (self.op, compare_versions(candidate, &self.version)) {
            (Op::Eq, Ordering::Equal) => true,
            (Op::Lt, Ordering::Less) => true,
            (Op::Le, Ordering::Less | Ordering::Equal) => true,
            (Op::Gt, Ordering::Greater) => true,
            (Op::Ge, Ordering::Greater | Ordering::Equal) => true,
            _ => false,
        }
    }
}

#[derive(Clone, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_range_space(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_range_space(bytes[i]) {
            i += 1;
        }
        tokens.push(Token {
            text: &s[start..i],
            start,
        });
    }
    tokens
}

fn parse_full_operand(
    operand: &str,
    operand_start: usize,
) -> Result<Version, RangeError> {
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: operand_start });
    }
    parse_version(operand).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })
}

#[derive(Clone, Debug)]
enum ParsedOperand {
    Full(Version),
    Partial {
        fields: Vec<u64>,
        count: usize,
    },
}

fn parse_flexible_operand(
    operand: &str,
    start: usize,
) -> Result<ParsedOperand, RangeError> {
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: start });
    }

    let fields: Vec<&str> = operand.split('.').collect();
    let has_suffix = operand.contains('-') || operand.contains('+');
    let no_wildcards = !fields
        .iter()
        .any(|f| matches!(*f, "*" | "x" | "X"));

    if has_suffix || (fields.len() >= 3 && no_wildcards) {
        return parse_full_operand(operand, start).map(ParsedOperand::Full);
    }

    if fields.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }

    let mut numeric = Vec::new();
    let mut wildcard_seen = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen
            || !field.bytes().all(is_digit)
            || (field.len() > 1 && field.starts_with('0'))
        {
            return Err(RangeError::Syntax { at: start });
        }
        let mut n = 0u64;
        for b in field.bytes() {
            n = n
                .checked_mul(10)
                .and_then(|x| x.checked_add((b - b'0') as u64))
                .ok_or(RangeError::Syntax { at: start })?;
        }
        numeric.push(n);
    }

    Ok(ParsedOperand::Partial {
        count: numeric.len(),
        fields: numeric,
    })
}

fn version_from_fields(fields: &[u64], pre: Vec<String>, build: Vec<String>) -> Version {
    Version {
        major: *fields.first().unwrap_or(&0),
        minor: *fields.get(1).unwrap_or(&0),
        patch: *fields.get(2).unwrap_or(&0),
        pre,
        build,
    }
}

fn bound_version(fields: [u64; 3]) -> Version {
    Version {
        major: fields[0],
        minor: fields[1],
        patch: fields[2],
        pre: vec!["0".to_owned()],
        build: Vec::new(),
    }
}

fn inc_field(fields: &mut [u64; 3], index: usize, start: usize) -> Result<(), RangeError> {
    fields[index] = fields[index]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at: start })?;
    for field in fields.iter_mut().skip(index + 1) {
        *field = 0;
    }
    Ok(())
}

fn cmp(op: Op, version: Version, admission: bool) -> Comparator {
    Comparator {
        op,
        version,
        admission,
    }
}

fn add_expansion(
    target: &mut Vec<Comparator>,
    parsed: ParsedOperand,
    kind: u8,
    start: usize,
) -> Result<(), RangeError> {
    match (kind, parsed) {
        (0, ParsedOperand::Full(v)) => {
            let admission = !v.pre.is_empty();
            target.push(cmp(Op::Eq, v, admission));
        }
        (0, ParsedOperand::Partial { fields, count }) => {
            if count == 0 {
                return Ok(());
            }
            let lower = version_from_fields(&fields, Vec::new(), Vec::new());
            target.push(cmp(Op::Ge, lower, false));
            if count < 3 {
                let mut upper_fields = [0, 0, 0];
                for (i, value) in fields.iter().enumerate() {
                    upper_fields[i] = *value;
                }
                inc_field(&mut upper_fields, count - 1, start)?;
                target.push(cmp(Op::Lt, bound_version(upper_fields), false));
            }
        }
        (1, ParsedOperand::Full(v)) => {
            if v.pre.is_empty() && v.build.is_empty() {
                // Full versions retain their parsed tuple as the inclusive lower endpoint.
            }
            let admission = !v.pre.is_empty();
            let lower = v.clone();
            target.push(cmp(Op::Ge, lower, admission));
            let mut f = [v.major, v.minor, v.patch];
            let index = if v.major > 0 { 0 } else if v.minor > 0 { 1 } else { 2 };
            inc_field(&mut f, index, start)?;
            target.push(cmp(Op::Lt, bound_version(f), false));
        }
        (1, ParsedOperand::Partial { fields, count }) => {
            if count == 0 {
                return Err(RangeError::Syntax { at: start });
            }
            let lower = version_from_fields(&fields, Vec::new(), Vec::new());
            target.push(cmp(Op::Ge, lower, false));
            let mut f = [0, 0, 0];
            for (i, value) in fields.iter().enumerate() {
                f[i] = *value;
            }
            let index = if fields[0] > 0 || count == 1 {
                0
            } else if fields.get(1).copied().unwrap_or(0) > 0 || count == 2 {
                1
            } else {
                2
            };
            inc_field(&mut f, index, start)?;
            target.push(cmp(Op::Lt, bound_version(f), false));
        }
        (2, ParsedOperand::Full(v)) => {
            let admission = !v.pre.is_empty();
            target.push(cmp(Op::Ge, v.clone(), admission));
            let mut f = [v.major, v.minor, v.patch];
            let index = if v.major > 0 { 0 } else { 1 };
            inc_field(&mut f, index, start)?;
            target.push(cmp(Op::Lt, bound_version(f), false));
        }
        (2, ParsedOperand::Partial { fields, count }) => {
            if count == 0 {
                return Err(RangeError::Syntax { at: start });
            }
            let lower = version_from_fields(&fields, Vec::new(), Vec::new());
            target.push(cmp(Op::Ge, lower, false));
            let mut f = [0, 0, 0];
            for (i, value) in fields.iter().enumerate() {
                f[i] = *value;
            }
            let index = if count == 1 { 0 } else { 1 };
            inc_field(&mut f, index, start)?;
            target.push(cmp(Op::Lt, bound_version(f), false));
        }
        _ => unreachable!(),
    }
    Ok(())
}

fn parse_comparator(token: &Token<'_>) -> Result<Comparator, RangeError> {
    let bytes = token.text.as_bytes();
    let (op, offset) = if token.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.text.starts_with("<=") {
        (Op::Le, 2)
    } else if token.text.starts_with('>') {
        (Op::Gt, 1)
    } else if token.text.starts_with('<') {
        (Op::Lt, 1)
    } else if token.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    if offset == bytes.len() || !bytes[offset].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = parse_full_operand(
        &token.text[offset..],
        token.start + offset,
    )?;
    let admission = !v.pre.is_empty();
    Ok(cmp(op, v, admission))
}

fn is_standalone_minus(token: &Token<'_>) -> bool {
    token.text == "-"
}

fn parse_hyphen_arm(tokens: &[Token<'_>]) -> Result<Vec<Comparator>, RangeError> {
    let minus = tokens.iter().position(is_standalone_minus).unwrap();
    if tokens.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax {
            at: tokens[minus].start,
        });
    }

    let lower_parsed = parse_flexible_operand(tokens[0].text, tokens[0].start)?;
    let upper_parsed = parse_flexible_operand(tokens[2].text, tokens[2].start)?;

    let lower = match lower_parsed {
        ParsedOperand::Full(v) => v,
        ParsedOperand::Partial { fields, count } if count > 0 => {
            version_from_fields(&fields, Vec::new(), Vec::new())
        }
        ParsedOperand::Partial { .. } => {
            return Err(RangeError::Syntax { at: tokens[0].start });
        }
    };

    let mut result = vec![cmp(Op::Ge, lower.clone(), !lower.pre.is_empty())];

    match upper_parsed {
        ParsedOperand::Full(v) => {
            let admission = !v.pre.is_empty();
            result.push(cmp(Op::Le, v, admission));
        }
        ParsedOperand::Partial { fields, count } => {
            if count == 0 {
                return Err(RangeError::Syntax { at: tokens[2].start });
            }
            let mut f = [0, 0, 0];
            for (i, value) in fields.iter().enumerate() {
                f[i] = *value;
            }
            let index = count - 1;
            inc_field(&mut f, index, tokens[2].start)?;
            result.push(cmp(Op::Lt, bound_version(f), false));
        }
    }
    Ok(result)
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Vec<Comparator>, RangeError> {
    if let Some(index) = tokens.iter().position(is_standalone_minus) {
        if tokens.len() != 3 || index != 1 {
            return Err(RangeError::Syntax {
                at: tokens[index].start,
            });
        }
        return parse_hyphen_arm(tokens);
    }

    let mut result = Vec::new();
    for token in tokens {
        let first = token.text.as_bytes()[0];
        match first {
            b'<' | b'>' | b'=' => result.push(parse_comparator(token)?),
            b'^' | b'~' => {
                let kind = if first == b'^' { 1 } else { 2 };
                let operand = &token.text[1..];
                let operand_start = token.start + 1;
                let parsed = parse_flexible_operand(operand, operand_start)?;
                add_expansion(&mut result, parsed, kind, token.start)?;
            }
            b'0'..=b'9' | b'*' | b'x' | b'X' => {
                let parsed = parse_flexible_operand(token.text, token.start)?;
                add_expansion(&mut result, parsed, 0, token.start)?;
            }
            _ => return Err(RangeError::Syntax { at: token.start }),
        }
    }
    Ok(result)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;

    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty {
                    at: tokens[i].start,
                });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(tokens[i].clone());
            i += 1;
        }
    }

    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !arm.iter().all(|c| c.accepts(v)) {
            return false;
        }
        if v.pre.is_empty() {
            return true;
        }
        arm.iter().any(|c| c.admission && core(&c.version) == core(v))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(previous) => {
                let order = compare_versions(version, &parsed[previous]);
                if (want_max && order == Ordering::Greater)
                    || (!want_max && order == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }
    Ok(best.map(|index| candidates[index].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
