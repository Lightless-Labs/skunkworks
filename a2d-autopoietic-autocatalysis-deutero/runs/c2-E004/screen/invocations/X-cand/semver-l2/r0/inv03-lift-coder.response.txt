use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_field(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len()
            || matches!(bytes[start], b'.' | b'-' | b'+')
        {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let field = &bytes[start..*pos];
    if field.len() > 1 && field[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &b in field {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    base: usize,
    numeric_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut start = 0;
    loop {
        let mut i = start;
        while i < bytes.len()
            && (bytes[i].is_ascii_alphanumeric() || bytes[i] == b'-')
        {
            i += 1;
        }
        if i != bytes.len() && bytes[i] != b'.' {
            return Err(VersionError::InvalidChar { at: base + i });
        }
        if i == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
        let part = &s[start..i];
        if numeric_pre
            && part.len() > 1
            && part.as_bytes()[0] == b'0'
            && part.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        out.push(part.to_owned());
        if i == bytes.len() {
            break;
        }
        start = i + 1;
        if start == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0;
    let major = parse_core_field(bytes, &mut pos)?;
    for _ in 0..2 {
        if pos >= bytes.len() || bytes[pos] != b'.' {
            return Err(if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        if _ == 0 {}
    }
    // Reparse the remaining core fields with their separator checks.
    let mut pos = 0;
    let major = parse_core_field(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else { VersionError::InvalidChar { at: pos } });
    }
    pos += 1;
    let minor = parse_core_field(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else { VersionError::InvalidChar { at: pos } });
    }
    pos += 1;
    let patch = parse_core_field(bytes, &mut pos)?;
    if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = s[pre_start..].find('+').map(|i| pre_start + i);
        let end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(&s[pre_start..end], pre_start, true)?;
        pos = end;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(&s[build_start..], build_start, false)?;
        pos = bytes.len();
    }
    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn make_version(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version { major, minor, patch, pre, build: Vec::new() }
}

fn upper_bump(
    mut v: Version,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    match field {
        0 => v.major = v.major.checked_add(1).ok_or(RangeError::Syntax { at })?,
        1 => v.minor = v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?,
        _ => v.patch = v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    if field <= 1 { v.minor = if field == 0 { 0 } else { v.minor }; }
    if field == 0 { v.minor = 0; v.patch = 0; }
    if field == 1 { v.patch = 0; }
    v.pre = vec!["0".to_owned()];
    v.build.clear();
    Ok(v)
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
    numeric: usize,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_sign = s.contains('-') || s.contains('+');
    let parts: Vec<&str> = s.split('.').collect();
    let any_wild = parts.iter().any(|p| matches!(*p, "*" | "x" | "X"));
    if has_sign || (parts.len() >= 3 && !any_wild) {
        return parse_version(s)
            .map(|v| Partial { fields: vec![Some(v.major), Some(v.minor), Some(v.patch)], full: Some(v), numeric: 3 })
            .map_err(|error| RangeError::Version { at, error });
    }
    if parts.is_empty() || parts.len() > 3 || parts.iter().any(|p| p.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut numeric = 0;
    let mut wild = false;
    for part in parts {
        if matches!(part, "*" | "x" | "X") {
            wild = true;
            fields.push(None);
        } else {
            if wild || !part.bytes().all(|b| b.is_ascii_digit())
                || (part.len() > 1 && part.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = part.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            numeric += 1;
            fields.push(Some(n));
        }
    }
    Ok(Partial { fields, full: None, numeric })
}

fn padded(p: &Partial) -> Version {
    let get = |i: usize| p.fields.get(i).and_then(|x| *x).unwrap_or(0);
    make_version(get(0), get(1), get(2), Vec::new())
}

fn add_check(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
    arm.checks.push(Check { op, version });
}

fn parse_operand_version(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn parse_hyphen_arm(tokens: &[(usize, &str)], arm: &mut Arm) -> Result<(), RangeError> {
    let minus: Vec<_> = tokens.iter().filter(|(_, t)| *t == "-").collect();
    if minus.is_empty() {
        return Err(RangeError::Syntax { at: tokens[0].0 });
    }
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: minus[0].0 });
    }
    let (la, ls) = tokens[0];
    let (ua, us) = tokens[2];
    let lower = parse_partial(ls, la)?;
    let upper = parse_partial(us, ua)?;
    if lower.numeric == 0 { return Err(RangeError::Syntax { at: la }); }
    if upper.numeric == 0 { return Err(RangeError::Syntax { at: ua }); }
    let lowv = lower.full.clone().unwrap_or_else(|| padded(&lower));
    add_check(arm, Op::Ge, lowv, true);
    if let Some(full) = upper.full {
        add_check(arm, Op::Le, full, true);
    } else {
        let upper_v = padded(&upper);
        let k = upper.numeric;
        let field = if k == 1 { 0 } else { 1 };
        let bound = upper_bump(upper_v, field, ua)?;
        add_check(arm, Op::Lt, bound, false);
    }
    Ok(())
}

fn parse_token(token: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let bytes = token.as_bytes();
    if token.starts_with('^') || token.starts_with('~') {
        let caret = bytes[0] == b'^';
        let op_at = start + 1;
        let partial = parse_partial(&token[1..], op_at)?;
        if partial.numeric == 0 { return Err(RangeError::Syntax { at: start }); }
        let lower = partial.full.clone().unwrap_or_else(|| padded(&partial));
        add_check(arm, Op::Ge, lower, true);
        let k = partial.numeric;
        let field = if caret {
            if lower.major > 0 || k == 1 { 0 }
            else if lower.minor > 0 || k == 2 { 1 }
            else { 2 }
        } else if k == 1 { 0 } else { 1 };
        let bound = upper_bump(padded(&partial), field, op_at)?;
        add_check(arm, Op::Lt, bound, false);
        return Ok(());
    }

    let mut op = Op::Eq;
    let mut i = 0;
    if bytes.starts_with(b">=") { op = Op::Ge; i = 2; }
    else if bytes.starts_with(b"<=") { op = Op::Le; i = 2; }
    else if bytes.starts_with(b">") { op = Op::Gt; i = 1; }
    else if bytes.starts_with(b"<") { op = Op::Lt; i = 1; }
    else if bytes.starts_with(b"=") { op = Op::Eq; i = 1; }

    if i > 0 {
        if i == bytes.len() || !bytes[i].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }
        let version = parse_operand_version(&token[i..], start + i)?;
        add_check(arm, op, version, true);
        return Ok(());
    }

    if !bytes[0].is_ascii_digit() && !matches!(bytes[0], b'*' | b'x' | b'X') {
        return Err(RangeError::Syntax { at: start });
    }
    let partial = parse_partial(token, start)?;
    if let Some(full) = partial.full {
        add_check(arm, Op::Eq, full, true);
    } else if partial.numeric == 0 {
        // Wildcard-only operand imposes no checks.
    } else {
        let lower = padded(&partial);
        let k = partial.numeric;
        add_check(arm, Op::Ge, lower.clone(), true);
        if k < 3 {
            let bound = upper_bump(lower, if k == 1 { 0 } else { 1 }, start)?;
            add_check(arm, Op::Lt, bound, false);
        }
    }
    Ok(())
}

fn tokenize(s: &str) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let bytes = s.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) { i += 1; }
        if i == bytes.len() { break; }
        let start = i;
        while i < bytes.len() && !ws(bytes[i]) { i += 1; }
        out.push((start, &s[start..i]));
    }
    out
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut current = Arm { checks: Vec::new(), anchors: Vec::new() };
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].1 == "||" {
            if current.checks.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].0 });
            }
            arms.push(current);
            current = Arm { checks: Vec::new(), anchors: Vec::new() };
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
            continue;
        }
        let begin = i;
        while i < tokens.len() && tokens[i].1 != "||" { i += 1; }
        let slice = &tokens[begin..i];
        if slice.iter().any(|(_, t)| *t == "-") {
            parse_hyphen_arm(slice, &mut current)?;
        } else {
            for &(start, token) in slice {
                parse_token(token, start, &mut current)?;
            }
        }
        if i < tokens.len() {
            if current.checks.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].0 });
            }
            arms.push(current);
            current = Arm { checks: Vec::new(), anchors: Vec::new() };
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        }
    }
    if current.checks.is_empty() && !arms.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    if !current.checks.is_empty() { arms.push(current); }
    Ok(Range { arms })
}

fn check_one(v: &Version, c: &Check) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.contains(&(v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.checks.iter().all(|c| check_one(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, text) in candidates.iter().enumerate() {
        let version = parse_version(text).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }
    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(version, &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
