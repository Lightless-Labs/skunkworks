use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;

        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            if pos == bytes.len()
                || matches!(bytes[pos], b'.' | b'-' | b'+')
            {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in &bytes[start..pos] {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        core[field] = value;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len()
                || matches!(bytes[pos], b'-' | b'+')
            {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        } else if pos < bytes.len()
            && !matches!(bytes[pos], b'-' | b'+')
        {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let build_marker = bytes[pos..].iter().position(|&b| b == b'+')
            .map(|offset| pos + offset);
        let pre_end = build_marker.unwrap_or(bytes.len());

        parse_identifiers(bytes, pre_start, pre_end, true, &mut pre)?;

        if let Some(marker) = build_marker {
            pos = marker + 1;
            parse_identifiers(bytes, pos, bytes.len(), false, &mut build)?;
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(bytes, pos, bytes.len(), false, &mut build)?;
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let mut ident_start = start;
    let mut pos = start;

    loop {
        while pos < end && bytes[pos] != b'.' {
            let b = bytes[pos];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &bytes[ident_start..pos];
        if is_pre
            && ident.len() > 1
            && ident[0] == b'0'
            && ident.iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        out.push(String::from_utf8(ident.to_vec()).unwrap());

        if pos == end {
            break;
        }

        pos += 1;
        ident_start = pos;
    }

    Ok(())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.iter().zip(b.iter()) {
        let left_numeric = left.bytes().all(|byte| byte.is_ascii_digit());
        let right_numeric = right.bytes().all(|byte| byte.is_ascii_digit());

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => left
                .len()
                .cmp(&right.len())
                .then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    admits_pre: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut comparators = Vec::new();
    let mut admits_pre = Vec::new();
    let mut pos = 0usize;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let mut operand_start = token_start;
        let op = match bytes[token_start] {
            b'=' => {
                operand_start += 1;
                Op::Eq
            }
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Op::Le
            }
            b'<' => {
                operand_start += 1;
                Op::Lt
            }
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Op::Ge
            }
            b'>' => {
                operand_start += 1;
                Op::Gt
            }
            _ => Op::Eq,
        };

        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            admits_pre.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        comparators,
        admits_pre,
    })
}

fn is_range_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .admits_pre
            .iter()
            .any(|&(major, minor, patch)| {
                (v.major, v.minor, v.patch) == (major, minor, patch)
            })
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.op {
            Op::Eq => ordering == Ordering::Equal,
            Op::Lt => ordering == Ordering::Less,
            Op::Le => ordering != Ordering::Greater,
            Op::Gt => ordering == Ordering::Greater,
            Op::Ge => ordering != Ordering::Less,
        }
    })
}
