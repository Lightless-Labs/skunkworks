#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: u8) -> bool {
    matches!(b, b'.' | b'-' | b'+')
}

fn parse_core_field(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;

    if start >= bytes.len() || is_core_terminator(bytes[start]) {
        return Err(VersionError::MissingCore { at: start });
    }
    if !bytes[start].is_ascii_digit() {
        return Err(VersionError::InvalidChar { at: start });
    }

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    let field = &bytes[start..*pos];
    if field.len() > 1 && field[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in field {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn is_identifier_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut cursor = ident_start;
        while cursor < end && bytes[cursor] != b'.' {
            if !is_identifier_byte(bytes[cursor]) {
                return Err(VersionError::InvalidChar { at: cursor });
            }
            cursor += 1;
        }

        if cursor == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let identifier = &bytes[ident_start..cursor];
        if is_pre
            && identifier.len() > 1

            && identifier.iter().all(u8::is_ascii_digit)

            return Err(VersionError::LeadingZero { at: ident_start });

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {

    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut comparators = Vec::new();
    let mut prerelease_cores = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
        pos += 1;
    }
    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (ordering, inclusive, op_len) = if token.starts_with(">=") {
            (Ordering::Greater, true, 2)
        } else if token.starts_with("<=") {
            (Ordering::Less, true, 2)
        } else if token.starts_with('>') {
            (Ordering::Greater, false, 1)
        } else if token.starts_with('<') {
            (Ordering::Less, false, 1)
        } else if token.starts_with('=') {
            (Ordering::Equal, true, 1)
        } else {
            (Ordering::Equal, true, 0)
        };

        let operand_start = token_start + op_len;
        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            prerelease_cores.push((version.major, version.minor, version.patch));
        }
        comparators.push((ordering, inclusive, version));

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        comparators,
        prerelease_cores,
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .prerelease_cores
            .iter()
            .any(|&(major, minor, patch)| {
                (v.major, v.minor, v.patch) == (major, minor, patch)
            })
    {
        return false;
    }

    r.comparators.iter().all(|(operator, inclusive, version)| {
        let ordering = compare_versions(v, version);
        match operator {
            Ordering::Equal => ordering == Ordering::Equal,
            Ordering::Less => {
                ordering == Ordering::Less || (*inclusive && ordering == Ordering::Equal)
            }
            Ordering::Greater => {
                ordering == Ordering::Greater || (*inclusive && ordering == Ordering::Equal)
            }
        }
    })
}

            return Err(RangeError::Syntax { at: token_start });

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;

        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            if pos == bytes.len()
                || matches!(bytes[pos], b'.' | b'-' | b'+')
            {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in &bytes[start..pos] {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        core[field] = value;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len()
                || matches!(bytes[pos], b'-' | b'+')
            {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        } else if pos < bytes.len()
            && !matches!(bytes[pos], b'-' | b'+')
        {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let build_marker = bytes[pos..].iter().position(|&b| b == b'+')
            .map(|offset| pos + offset);
        let pre_end = build_marker.unwrap_or(bytes.len());

        parse_identifiers(bytes, pre_start, pre_end, true, &mut pre)?;

        if let Some(marker) = build_marker {
            pos = marker + 1;
            parse_identifiers(bytes, pos, bytes.len(), false, &mut build)?;
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(bytes, pos, bytes.len(), false, &mut build)?;
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let mut ident_start = start;
    let mut pos = start;

    loop {
        while pos < end && bytes[pos] != b'.' {
            let b = bytes[pos];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &bytes[ident_start..pos];
        if is_pre
            && ident.len() > 1
            && ident[0] == b'0'
            && ident.iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        out.push(String::from_utf8(ident.to_vec()).unwrap());

        if pos == end {
            break;
        }

        pos += 1;
        ident_start = pos;
    }

    Ok(())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.iter().zip(b.iter()) {
        let left_numeric = left.bytes().all(|byte| byte.is_ascii_digit());
        let right_numeric = right.bytes().all(|byte| byte.is_ascii_digit());

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => left
                .len()
                .cmp(&right.len())
                .then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

    let mut comparators = Vec::new();
    let mut admits_pre = Vec::new();
    let mut pos = 0usize;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let mut operand_start = token_start;
        let op = match bytes[token_start] {
            b'=' => {
                operand_start += 1;
                Op::Eq
            }
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Op::Le
            }
            b'<' => {
                operand_start += 1;
                Op::Lt
            }
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2;
                Op::Ge
            }
            b'>' => {
                operand_start += 1;
                Op::Gt
            }
            _ => Op::Eq,
        };

        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            admits_pre.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        comparators,
        admits_pre,
    })
}

fn is_core_terminator(byte: Option<u8>) -> bool {
    matches!(byte, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();

    if start >= bytes.len() || !bytes[start].is_ascii_digit() {
        return if is_core_terminator(bytes.get(start).copied()) {
            Err(VersionError::MissingCore { at: start })
        } else {
            Err(VersionError::InvalidChar { at: start })
        };
    }

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }

    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut identifiers = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let identifier = &s[ident_start..i];
        if is_pre
            && identifier.len() > 1
            && identifier.starts_with('0')
            && identifier.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        identifiers.push(identifier.to_owned());

        if i == end {
            break;
        }

        ident_start = i + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(identifiers)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;

    if bytes.get(pos) != Some(&b'.') {
        return if is_core_terminator(bytes.get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let minor = parse_core_field(s, &mut pos)?;

    if bytes.get(pos) != Some(&b'.') {
        return if is_core_terminator(bytes.get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let patch = parse_core_field(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();

    match bytes.get(pos).copied() {
        None => {}
        Some(b'-') => {
            let pre_start = pos + 1;
            let build_marker = bytes[pre_start..]
                .iter()
                .position(|&b| b == b'+')
                .map(|offset| pre_start + offset);
            let pre_end = build_marker.unwrap_or(bytes.len());

            pre = parse_identifiers(s, pre_start, pre_end, true)?;

            if let Some(marker) = build_marker {
                let build_start = marker + 1;
                build = parse_identifiers(s, build_start, bytes.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, pos + 1, bytes.len(), false)?;
        }
        Some(_) => return Err(VersionError::InvalidChar { at: pos }),
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core_order = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core_order != Ordering::Equal {
        return core_order;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.pre.iter().zip(&b.pre) {
        let left_numeric = left.bytes().all(|c| c.is_ascii_digit());
        let right_numeric = right.bytes().all(|c| c.is_ascii_digit());

        let order = match (left_numeric, right_numeric) {
            (true, true) => compare_numeric_identifiers(left, right),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if order != Ordering::Equal {
            return order;
        }
    }

    a.pre.len().cmp(&b.pre.len())
}
