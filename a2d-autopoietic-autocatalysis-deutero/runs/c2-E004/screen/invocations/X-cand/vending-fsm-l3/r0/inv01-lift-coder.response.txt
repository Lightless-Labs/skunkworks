```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn product_index(products: &[Product], id: &str) -> Option<usize> {
    products.iter().position(|product| product.id == id)
}

fn event(events: &mut Vec<Event>, kind: EventKind) {
    let seq = events.len();
    events.push(Event { seq, kind });
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount;
    let mut result = [0; 4];
    for i in (0..4).rev() {
        let denomination = DENOMINATIONS[i];
        let take = tubes[i].min(remaining / denomination);
        result[i] = take;
        remaining -= take * denomination;
    }
    if remaining == 0 {
        Some(result)
    } else {
        None
    }
}

fn better_change(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
    let candidate_count: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
    let current_count: u32 = current.iter().map(|&n| u32::from(n)).sum();
    if candidate_count != current_count {
        return candidate_count < current_count;
    }
    for i in (0..4).rev() {
        if candidate[i] != current[i] {
            return candidate[i] > current[i];
        }
    }
    false
}

fn exact_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<[u16; 4]> = None;
    let max_tens = tubes[3].min(amount / 10);
    for tens in 0..=max_tens {
        let left_after_tens = amount - tens * 10;
        let max_fours = tubes[2].min(left_after_tens / 4);
        for fours in 0..=max_fours {
            let left_after_fours = left_after_tens - fours * 4;
            let max_threes = tubes[1].min(left_after_fours / 3);
            for threes in 0..=max_threes {
                let left_after_threes = left_after_fours - threes * 3;
                if left_after_threes > tubes[0] {
                    continue;
                }
                let candidate = [left_after_threes, threes, fours, tens];
                if best
                    .as_ref()
                    .is_none_or(|current| better_change(&candidate, current))
                {
                    best = Some(candidate);
                }
            }
        }
    }
    best
}

fn change_vector(counts: &[u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            coins.push(DENOMINATIONS[i]);
        }
    }
    coins
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let new_credit = u32::from(self.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit = new_credit as u16;
        self.escrow.push(coin);
        event(
            &mut self.events,
            EventKind::Inserted {
                coin,
                credit: self.credit,
            },
        );
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(index) = product_index(&self.products, id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(self.products[index].id.as_str()) {
            return Err(Error::SameSelection);
        }

        let selected_id = self.products[index].id.clone();
        self.selected = Some(selected_id.clone());
        event(
            &mut self.events,
            EventKind::Selected { id: selected_id },
        );
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let Some(index) = product_index(&self.products, &selected_id) else {
            return Err(Error::NoSelection);
        };

        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut {
                id: product.id.clone(),
            });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let counts = if let Some(greedy) = greedy_change(amount, &self.tubes) {
            greedy
        } else {
            exact_change(amount, &self.tubes).ok_or(Error::NoChange { amount })?
        };
        let change = change_vector(&counts);

        let mut proposed_tubes = self.tubes;
        for i in 0..4 {
            proposed_tubes[i] -= counts[i];
        }
        for &coin in &self.escrow {
            let i = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow only contains accepted denominations");
            let proposed = u32::from(proposed_tubes[i]) + 1;
            if proposed > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
            proposed_tubes[i] = proposed as u16;
        }

        let id = self.products[index].id.clone();
        self.products[index].stock -= 1;
        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        event(
            &mut self.events,
            EventKind::Vended {
                id: id.clone(),
                price,
                change: change.clone(),
            },
        );
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        event(
            &mut self.events,
            EventKind::Refunded {
                coins: coins.clone(),
            },
        );
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = product_index(&self.products, id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        event(
            &mut self.events,
            EventKind::PriceChanged {
                id: self.products[index].id.clone(),
                old,
                new: price,
            },
        );
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = product_index(&self.products, id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = u32::from(self.products[index].stock) + u32::from(count);
        if new_stock > 100 {
            return Err(Error::StockLimit {
                id: self.products[index].id.clone(),
            });
        }

        self.products[index].stock = new_stock as u16;
        event(
            &mut self.events,
            EventKind::Restocked {
                id: self.products[index].id.clone(),
                added: count,
                stock: self.products[index].stock,
            },
        );
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}
```
