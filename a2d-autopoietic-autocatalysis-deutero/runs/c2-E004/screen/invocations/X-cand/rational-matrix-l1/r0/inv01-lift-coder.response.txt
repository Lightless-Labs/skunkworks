```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_wide(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd(n.unsigned_abs(), d as u128);
    let rn = n / g as i128;
    let rd = d / g as i128;

    if rn < i64::MIN as i128 || rn > i64::MAX as i128 || rd < 1 || rd > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: rn as i64,
        den: rd as i64,
    })
}

fn rational_from_signed_magnitude(negative: bool, magnitude: u128, den: u128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if magnitude == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd(magnitude, den);
    let n = magnitude / g;
    let d = den / g;
    let max_n = if negative { 1u128 << 63 } else { i64::MAX as u128 };

    if n > max_n || d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }

    let num = if negative {
        if n == (1u128 << 63) {
            i64::MIN
        } else {
            -(n as i64)
        }
    } else {
        n as i64
    };

    Ok(Rational {
        num,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let negative = (num < 0) ^ (den < 0);
        rational_from_signed_magnitude(negative, num.unsigned_abs() as u128, den.unsigned_abs() as u128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_wide(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_wide(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_wide(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        rational_from_wide(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_error(i, if i == bytes.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }

        let numerator_end = i;
        let mut denominator_start = None;
        let mut denominator_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            if i == bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            if !bytes[i].is_ascii_digit() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            denominator_start = Some(start);
            denominator_end = Some(i);

            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let num_token = &bytes[numerator_start..numerator_end];
        if num_token.len() >= 2 && num_token[0] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_token == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            let den_token = &bytes[start..end];
            if den_token.len() >= 2 && den_token[0] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let num_mag = parse_magnitude(num_token);
        let max_num = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag.map_or(true, |n| n > max_num) {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let den_mag = if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            let token = &bytes[start..end];
            let mag = parse_magnitude(token);
            if mag.map_or(true, |d| d > i64::MAX as u128) {
                return Err(parse_error(start, ParseKind::OutOfRange));
            }
            Some(mag.unwrap())
        } else {
            None
        };

        if let (Some(start), Some(end), Some(d)) = (denominator_start, denominator_end, den_mag) {
            if d == 0 {
                return Err(parse_error(start, ParseKind::ZeroDenominator));
            }

            let slash = start - 1;
            if d == 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            if num_mag == Some(0) {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }

            let n = num_mag.unwrap();
            if gcd(n, d) > 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }

            let _ = end;
            rational_from_signed_magnitude(negative, n, d)
                .map_err(|_| parse_error(0, ParseKind::OutOfRange))
        } else {
            let n = num_mag.unwrap();
            let signed = if negative {
                if n == (1u128 << 63) {
                    i64::MIN
                } else {
                    -(n as i64)
                }
            } else {
                n as i64
            };
            Ok(Rational { num: signed, den: 1 })
        }
    }
}

fn parse_magnitude(bytes: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        let digit = (b - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
