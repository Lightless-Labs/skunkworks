```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u128,
}

#[derive(Debug, Clone)]
pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u128,
    capacity: Option<usize>,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::with_capacity_inner(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_capacity_inner(Some(capacity))
    }

    fn with_capacity_inner(capacity: Option<usize>) -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
            capacity,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn normalize_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }

        if start == end {
            return Err(Error::EmptyKey);
        }

        let trimmed = &bytes[start..end];

        if !trimmed[0].is_ascii_alphabetic() {
            return Err(Error::InvalidKey { at: start });
        }

        for (i, &byte) in trimmed.iter().enumerate().skip(1) {
            if !(byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-') {
                return Err(Error::InvalidKey { at: start + i });
            }
        }

        let key = std::str::from_utf8(trimmed).expect("validated ASCII key");
        Ok(key.to_ascii_lowercase())
    }

    fn compare(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn sorted_indices(&self) -> Vec<usize> {
        let mut order: Vec<usize> = (0..self.entries.len()).collect();
        order.sort_by(|&a, &b| Self::compare(&self.entries[a], &self.entries[b]));
        order
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (index, entry) in self.entries.iter().enumerate() {
            self.indices.insert(entry.item.key.clone(), index);
        }
    }

    fn remove_at(&mut self, index: usize) -> Item {
        let removed = self.entries.swap_remove(index);
        self.indices.remove(&removed.item.key);
        if index < self.entries.len() {
            self.indices
                .insert(self.entries[index].item.key.clone(), index);
        }
        removed.item
    }

    fn fresh_age(&mut self) -> u128 {
        // A queue cannot contain more than usize::MAX items, so renumbering
        // preserves all observable age ordering if the counter ever wraps.
        if self.next_age == u128::MAX {
            let order = self.sorted_indices();
            let mut ages = vec![0u128; self.entries.len()];
            for (age, index) in order.into_iter().enumerate() {
                ages[index] = age as u128;
            }
            for (entry, age) in self.entries.iter_mut().zip(ages) {
                entry.age = age;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = Self::normalize_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed.item));
        }

        if let Some(capacity) = self.capacity {
            if self.entries.len() >= capacity {
                let worst_index = self
                    .entries
                    .iter()
                    .enumerate()
                    .max_by(|(_, a), (_, b)| Self::compare(a, b))
                    .map(|(index, _)| index)
                    .expect("full positive-capacity queue is nonempty");

                if Self::compare(&proposed, &self.entries[worst_index])
                    != std::cmp::Ordering::Less
                {
                    return Ok(Admission::Discarded(proposed.item));
                }

                let evicted = self.remove_at(worst_index);
                let age = self.fresh_age();
                self.indices.insert(key, self.entries.len());
                self.entries.push(Entry {
                    item: proposed.item,
                    age,
                });
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.fresh_age();
        self.indices.insert(key, self.entries.len());
        self.entries.push(Entry {
            item: proposed.item,
            age,
        });
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.sorted_indices()
            .first()
            .map(|&index| self.entries[index].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let index = self
            .sorted_indices()
            .first()
            .copied()?;
        Some(self.remove_at(index))
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::normalize_key(key)?;
        Ok(self
            .indices
            .get(&key)
            .map(|&index| self.entries[index].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::normalize_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::normalize_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::normalize_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let old = std::mem::replace(&mut self.entries[index].item.value, value.to_owned());
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::normalize_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.remove_at(index))
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let Some(limit) = capacity else {
            return Vec::new();
        };

        let mut evicted = Vec::new();
        while self.entries.len() > limit {
            let worst_index = self
                .entries
                .iter()
                .enumerate()
                .max_by(|(_, a), (_, b)| Self::compare(a, b))
                .map(|(index, _)| index)
                .expect("excess items imply a nonempty queue");
            evicted.push(self.remove_at(worst_index));
        }
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.sorted_indices()
            .into_iter()
            .map(|index| self.entries[index].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order = other.sorted_indices();
        let source_items: Vec<Item> = source_order
            .iter()
            .map(|&index| other.entries[index].item.clone())
            .collect();

        if policy == CollisionPolicy::Reject {
            for item in &source_items {
                if self.indices.contains_key(&item.key) {
                    return Err(Error::Collision {
                        key: item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for item in source_items {
            if let Some(&index) = self.indices.get(&item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = item.priority;
                        self.entries[index].item.value = item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                self.indices.insert(item.key.clone(), self.entries.len());
                self.entries.push(Entry { item, age });
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.indices.clear();

        if let Some(limit) = self.capacity {
            while self.entries.len() > limit {
                let worst_index = self
                    .entries
                    .iter()
                    .enumerate()
                    .max_by(|(_, a), (_, b)| Self::compare(a, b))
                    .map(|(index, _)| index)
                    .expect("excess items imply a nonempty queue");
                report.evicted.push(self.remove_at(worst_index));
            }
        }

        Ok(report)
    }
}
```
