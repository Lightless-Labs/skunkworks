#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1_i128 << 120) - 1;

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    if !(-LIMIT..=LIMIT).contains(&value) {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut reversed = Vec::new();

    if n == 0 {
        reversed.push(b'0');
    } else {
        while n != 0 {
            match system {
                SignedSystem::BalancedTernary => {
                    let mut remainder = n % 3;
                    if remainder < 0 {
                        remainder += 3;
                    }
                    let digit: i128 = match remainder {
                        0 => 0,
                        1 => 1,
                        _ => -1,
                    };
                    reversed.push(match digit {
                        -1 => b'T',
                        0 => b'0',
                        _ => b'1',
                    });
                    n = (n - digit) / 3;
                }
                SignedSystem::Negabinary => {
                    let remainder = n.rem_euclid(2);
                    reversed.push(if remainder == 0 { b'0' } else { b'1' });
                    n = (n - remainder) / -2;
                }
            }
        }
    }

    reversed.reverse();
    let mut output = String::from_utf8(reversed).expect("digits are ASCII");

    if checksum {
        let sum = output
            .bytes()
            .enumerate()
            .map(|(i, byte)| {
                let code = match byte {
                    b'T' => 2_u32,
                    b'1' => 1,
                    _ => 0,
                };
                (((i as u32 + 1) % 7) * code) % 7
            })
            .fold(0_u32, |acc, term| (acc + term) % 7);
        output.push(':');
        output.push(char::from(b'0' + sum as u8));
    }

    Ok(output)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();
    let colon = bytes.iter().position(|&b| b == b':');
    let payload_end = colon.unwrap_or(bytes.len());

    if payload_end == 0 {
        return Err(SignedError::Empty);
    }

    for (index, &byte) in bytes[..payload_end].iter().enumerate() {
        let valid = match system {
            SignedSystem::BalancedTernary => matches!(byte, b'T' | b'0' | b'1'),
            SignedSystem::Negabinary => matches!(byte, b'0' | b'1'),
        };
        if !valid {
            return Err(SignedError::InvalidByte { index });
        }
    }

    if let Some(colon_index) = colon {
        let suffix = &bytes[colon_index + 1..];
        if suffix.len() != 1 || !matches!(suffix[0], b'0'..=b'6') {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
    }

    if payload_end > 1 && bytes[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let base: i128 = match system {
        SignedSystem::BalancedTernary => 3,
        SignedSystem::Negabinary => -2,
    };

    let mut value = 0_i128;
    for (index, &byte) in bytes[..payload_end].iter().enumerate() {
        let digit: i128 = match byte {
            b'T' => -1,
            b'0' => 0,
            b'1' => 1,
            _ => unreachable!("payload was validated"),
        };

        value = value * base + digit;
        if !(-LIMIT..=LIMIT).contains(&value) {
            return Err(SignedError::Range { index });
        }
    }

    if let Some(colon_index) = colon {
        let mut sum = 0_u32;
        for (i, &byte) in bytes[..payload_end].iter().enumerate() {
            let code = match byte {
                b'T' => 2_u32,
                b'1' => 1,
                _ => 0,
            };
            sum = (sum + (((i as u32 + 1) % 7) * code) % 7) % 7;
        }
        let expected = sum as u8;
        let found = bytes[colon_index + 1] - b'0';
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(value)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, checksum)
}
