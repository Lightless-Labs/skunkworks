fn is_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'a'..=b'f').contains(&b) || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'a'..=b'f' => b - b'a' + 10,
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }

fn utf8_error_offset(error: std::str::Utf8Error) -> usize {
    error.valid_up_to()
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let n = utf8_error_offset(e);
            Err(CodecError::InvalidUtf8 {
                at: offsets.get(n).copied().unwrap_or(input.len()),
            })
        }
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
        return true;
    }
    match component {
        Component::UserInfo => b == b':',
        Component::Path => b == b'/',
        Component::Query => matches!(b, b'/' | b'?'),
        Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if component_safe(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let after_marker = colon + 3;
    let auth_end = bytes[after_marker..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| after_marker + n)
        .unwrap_or(bytes.len());
    let authority = &input[after_marker..auth_end];
    let auth_bytes = authority.as_bytes();

    let mut at_sign = None;
    for (i, &b) in auth_bytes.iter().enumerate() {
        if b == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: after_marker + i,
                });
            }
            at_sign = Some(i);
        }
    }

    let (userinfo, host_start) = if let Some(i) = at_sign {
        if i == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: after_marker + i,
            });
        }
        (Some(authority[..i].to_owned()), i + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_start..];
    let host_abs = after_marker + host_start;
    let mut ipv6 = false;
    let (host_raw, port_raw, port_abs) = if host_port.starts_with('[') {
        ipv6 = true;
        let close = match host_port.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(n) => 1 + n,
            None => return Err(ParseError::InvalidIpv6 { at: host_abs }),
        };
        let interior = &host_port[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_abs });
        }
        let suffix = &host_port[close + 1..];
        let p = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some((&suffix[1..], host_abs + close + 2))
        } else {
            return Err(ParseError::InvalidHost {
                at: host_abs + close + 1,
            });
        };
        (interior, p.map(|v| v.0), p.map(|v| v.1).unwrap_or(0))
    } else {
        if let Some(i) = host_port.as_bytes().iter().position(|&b| b == b':') {
            (&host_port[..i], Some(&host_port[i + 1..]), host_abs + i + 1)
        } else {
            (host_port, None, 0)
        }
    };

    if host_raw.is_empty() {
        return Err(ParseError::InvalidHost { at: host_abs });
    }

    if !ipv6 {
        for (i, &b) in host_raw.as_bytes().iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_abs + i,
                });
            }
        }

        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_abs });
        }
        for (i, &b) in port_text.as_bytes().iter().enumerate() {
            if !b.is_ascii_digit() {
                return Err(ParseError::InvalidPort { at: port_abs + i });
            }
        }
        if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_abs });
        }
        let mut value: u32 = 0;
        for &b in port_text.as_bytes() {
            value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: port_abs }),
            };
        }
        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_abs });
        }
        Some(value as u16)
    } else {
        None
    };

    let fragment_mark = bytes[auth_end..].iter().position(|&b| b == b'#').map(|n| auth_end + n);
    let pre_fragment_end = fragment_mark.unwrap_or(bytes.len());
    let query_mark = bytes[auth_end..pre_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|n| auth_end + n);

    let path_end = query_mark.unwrap_or(pre_fragment_end);
    let raw_path = &input[auth_end..path_end];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    let query = query_mark.map(|q| input[q + 1..pre_fragment_end].to_owned());
    let fragment = fragment_mark.map(|f| input[f + 1..].to_owned());

    if let Some(ref u) = userinfo {
        let start = after_marker;
        decode_component(Component::UserInfo, u).map_err(|e| map_codec_parse(e, start))?;
    }
    if !raw_path.is_empty() {
        decode_component(Component::Path, raw_path)
            .map_err(|e| map_codec_parse(e, auth_end))?;
    }
    if let (Some(q), Some(qmark)) = (&query, query_mark) {
        decode_component(Component::Query, q)
            .map_err(|e| map_codec_parse(e, qmark + 1))?;
    }
    if let (Some(f), Some(fmark)) = (&fragment, fragment_mark) {
        decode_component(Component::Fragment, f)
            .map_err(|e| map_codec_parse(e, fmark + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host_raw.to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn map_codec_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut terminal_navigation = false;

    for (i, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        let dot = decoded == ".";
        let dotdot = decoded == "..";
        if i + 1 == segments.len() {
            terminal_navigation = dot || dotdot;
        }
        if dot {
            continue;
        } else if dotdot {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn map_path_resolve(error: PathError) -> ResolveError {
    ResolveError::Path(error)
}

fn scan_reference(input: &str) -> Result<(), ParseError> {
    for (i, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn split_reference_parts(input: &str) -> (String, Option<String>, Option<String>) {
    let bytes = input.as_bytes();
    let fragment_mark = bytes.iter().position(|&b| b == b'#');
    let before_fragment_end = fragment_mark.unwrap_or(bytes.len());
    let query_mark = bytes[..before_fragment_end].iter().position(|&b| b == b'?');
    let path_end = query_mark.unwrap_or(before_fragment_end);
    let path = input[..path_end].to_owned();
    let query = query_mark.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_mark.map(|f| input[f + 1..].to_owned());
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(map_path_resolve)?;

    scan_reference(reference).map_err(ResolveError::Reference)?;

    let first_delim = reference
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference.as_bytes()[..first_delim].contains(&b':') {
        let mut result = parse_url(reference).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(map_path_resolve)?;
        return Ok(result);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut result = parse_url(&absolute).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(map_path_resolve)?;
        return Ok(result);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference_parts(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.clone(),
            if ref_query.is_none() {
                base_url.query.clone()
            } else {
                ref_query
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..slash], ref_path)
        };
        (merged, ref_query)
    };

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(u) = &base_url.userinfo {
        raw.push_str(u);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        raw.push(':');
        raw.push_str(&port.to_string());
    }
    raw.push_str(&selected_path);
    if let Some(q) = &selected_query {
        raw.push('?');
        raw.push_str(q);
    }
    if let Some(f) = &ref_fragment {
        raw.push('#');
        raw.push_str(f);
    }

    let mut result = parse_url(&raw).map_err(ResolveError::Reference)?;
    result.path = normalise_path(&result.path).map_err(map_path_resolve)?;
    Ok(result)
}

fn map_codec_query(error: CodecError, base: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

fn form_decode(input: &str) -> String {
    input.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut offset = 0usize;

    for field in input.split('&') {
        let field_start = offset;
        offset += field.len() + 1;
        if field.is_empty() {
            continue;
        }

        let (raw_key, raw_value, value_start) = if let Some(eq) = field.find('=') {
            (&field[..eq], &field[eq + 1..], field_start + eq + 1)
        } else {
            (field, "", field_start + field.len())
        };

        let key_source = form_decode(raw_key);
        let key = decode_component(Component::Query, &key_source)
            .map_err(|e| map_codec_query(e, field_start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }

        let value_source = form_decode(raw_value);
        let value = decode_component(Component::Query, &value_source)
            .map_err(|e| map_codec_query(e, value_start))?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| k == &key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(&key), form_encode(&value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, input.find(userinfo.as_str()).unwrap_or(0))))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let normalized = url.host.parse::<std::net::Ipv6Addr>().map_err(|_| {
            CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 })
        })?;
        out.push('[');
        out.push_str(&normalized.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let default = (url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443);
        if !default {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(map_codec_path(e)))?;
        canonical_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(q) = query {
        out.push('?');
        out.push_str(&serialise_query(&q));
    }

    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| {
                CanonicalError::Parse(map_codec_parse(
                    e,
                    input.rfind('#').map(|i| i + 1).unwrap_or(input.len()),
                ))
            })?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(out)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(error) => {
            let at = origins
                .get(error.valid_up_to())
                .copied()
                .unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let safe = unreserved
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (at, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| auth_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];
    let auth_bytes = authority.as_bytes();

    let ats: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + ats[1],
        });
    }
    let (userinfo, host_start_in_auth) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: auth_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_start = auth_start + host_start_in_auth;
    let hostport = &authority[host_start_in_auth..];
    let hostport_bytes = hostport.as_bytes();

    let (host, port_text, port_offset, ipv6) = if hostport.starts_with('[') {
        let close = match hostport_bytes.iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix_start = close + 1;
        let suffix = &hostport[suffix_start..];
        if !suffix.is_empty() && !suffix.starts_with(':') {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        }
        let port = if suffix.is_empty() {
            None
        } else {
            Some(&suffix[1..])
        };
        (
            interior.to_ascii_lowercase(),
            port,
            port.map(|_| host_start + suffix_start + 1),
            true,
        )
    } else {
        let split = hostport_bytes.iter().position(|&b| b == b':');
        let (host, port) = match split {
            Some(i) => (&hostport[..i], Some(&hostport[i + 1..])),
            None => (hostport, None),
        };
        if host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, &b) in host.as_bytes().iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start + i });
            }
        }
        let port_at = split.map(|i| host_start + i + 1);
        (host.to_ascii_lowercase(), port, port_at, false)
    };

    let port = if let Some(text) = port_text {
        let at = port_offset.unwrap_or(auth_end);
        if text.is_empty() {
            return Err(ParseError::InvalidPort { at });
        }
        for (i, &b) in text.as_bytes().iter().enumerate() {
            if !b.is_ascii_digit() {
                return Err(ParseError::InvalidPort { at: at + i });
            }
        }
        if text.len() >= 2 && text.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at });
        }
        let mut value: u32 = 0;
        for &b in text.as_bytes() {
            value = match value
                .checked_mul(10)
                .and_then(|n| n.checked_add((b - b'0') as u32))
            {
                Some(n) => n,
                None => return Err(ParseError::InvalidPort { at }),
            };
            if value > 65535 {
                return Err(ParseError::InvalidPort { at });
            }
        }
        if value == 0 {
            return Err(ParseError::InvalidPort { at });
        }
        Some(value as u16)
    } else {
        None
    };

    let tail = &input[auth_end..];
    let fragment_pos = tail.find('#');
    let before_fragment = fragment_pos.map(|p| &tail[..p]).unwrap_or(tail);
    let fragment = fragment_pos.map(|p| tail[p + 1..].to_owned());
    let query_pos = before_fragment.find('?');
    let before_query = query_pos
        .map(|p| &before_fragment[..p])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|p| before_fragment[p + 1..].to_owned());

    let path = if before_query.is_empty() {
        "/".to_owned()
    } else {
        before_query.to_owned()
    };

    let userinfo_offset = auth_start;
    if let Some(ref raw) = userinfo {
        decode_component(Component::UserInfo, raw)
            .map_err(|e| map_codec_parse(e, userinfo_offset))?;
    }
    if before_query.len() > 0 {
        decode_component(Component::Path, before_query)
            .map_err(|e| map_codec_parse(e, auth_end))?;
    }
    if let Some(ref raw) = query {
        let query_offset = auth_end + query_pos.unwrap_or(0) + 1;
        decode_component(Component::Query, raw)
            .map_err(|e| map_codec_parse(e, query_offset))?;
    }
    if let Some(ref raw) = fragment {
        let fragment_offset = auth_end + fragment_pos.unwrap_or(0) + 1;
        decode_component(Component::Fragment, raw)
            .map_err(|e| map_codec_parse(e, fragment_offset))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let pieces: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (i, raw) in pieces.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(map_codec_path)?;
        let dot = decoded == ".";
        let dotdot = decoded == "..";
        if i + 1 == pieces.len() {
            terminal_navigation = dot || dotdot;
        }
        if dot {
            continue;
        } else if dotdot {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }

    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),

fn global_scan(input: &str) -> Result<(), ParseError> {
    for (at, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())

fn normalise_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn reference_components(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_at = reference.find('#');
    let before_fragment = fragment_at.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = fragment_at.map(|i| &reference[i + 1..]);
    let query_at = before_fragment.find('?');
    let path = query_at
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_at.map(|i| &before_fragment[i + 1..]);
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalise_url_path(base_url)?;

    global_scan(reference).map_err(ResolveError::Reference)?;
    let first_delimiter = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalise_url_path(url);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        return normalise_url_path(url);
    }

    let (reference_path, reference_query, reference_fragment) =
        reference_components(reference);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if reference_query.is_some() {
                reference_query.map(str::to_owned)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let base_dir_end = base_url.path.rfind('/').unwrap_or(0) + 1;
            format!("{}{}", &base_url.path[..base_dir_end], reference_path)
        };
        (merged, reference_query.map(str::to_owned))
    };

    let userinfo = base_url
        .userinfo
        .as_ref()
        .map(|u| format!("{u}@"))
        .unwrap_or_default();
    let host = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let port = base_url
        .port
        .map(|p| format!(":{p}"))
        .unwrap_or_default();
    let query = selected_query
        .as_ref()
        .map(|q| format!("?{q}"))
        .unwrap_or_default();
    let fragment = reference_fragment
        .map(|f| format!("#{f}"))
        .unwrap_or_default();
    let constructed = format!(
        "{}://{}{}{}{}{}{}",
        base_url.scheme, userinfo, host, port, selected_path, query, fragment
    );
    let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalise_url_path(url)
}

pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

fn form_decode(raw: &str, offset: usize) -> Result<String, QueryError> {
    let replaced = raw.replace('+', " ");
    decode_component(Component::Query, &replaced).map_err(|e| match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    })
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;

    for field in input.split('&') {
        if field.is_empty() {
            field_start += 1;
            continue;
        }

        let eq = field.find('=');
        let (raw_key, raw_value, value_offset) = match eq {
            Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
            None => (field, "", field_start + field.len()),
        };

        let key = form_decode(raw_key, field_start)?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value = form_decode(raw_value, value_offset)?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| k == &key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }

        field_start += field.len() + 1;
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn canonical_component(text: &str, component: Component) -> Result<String, CodecError> {
    let decoded = decode_component(component, text)?;

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let canonical =
            canonical_component(userinfo, Component::UserInfo).map_err(|e| {
                CanonicalError::Parse(map_codec_parse(e, input.find(userinfo).unwrap_or(0)))
            })?;
        output.push_str(&canonical);
        output.push('@');
    }

    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host).expect("validated IPv6 host");
        output.push('[');
        output.push_str(&address.to_string());
        output.push(']');
    } else {
        output.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let default = (url.scheme == "http" && port == 80)
            || (url.scheme == "https" && port == 443);
        if !default {
            output.push(':');
            output.push_str(&port.to_string());
        }
    }

    let segments: Result<Vec<String>, CanonicalError> = path
        .split('/')
        .map(|segment| {
            let decoded = decode_component(Component::Path, segment)
                .map_err(|_| CanonicalError::Path(PathError::NotAbsolute))?;
            Ok(encode_component(Component::UserInfo, &decoded))
        })
        .collect();
    output.push_str(&segments?.join("/"));

    if let Some(map) = query_map {
        output.push('?');
        output.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| {
                CanonicalError::Parse(map_codec_parse(
                    e,
                    input.rfind('#').map(|i| i + 1).unwrap_or(input.len()),
                ))
            })?;
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}

fn parse_port(text: &str, colon_at: usize) -> Result<Option<u16>, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: colon_at + 1 });
    }

    if let Some((index, _)) = text.bytes().enumerate().find(|(_, byte)| !byte.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: colon_at + 1 + index });
    }

    if text.len() >= 2 && text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: colon_at + 1 });
    }

    let mut value: u32 = 0;
    for byte in text.bytes() {
        value = value
            .saturating_mul(10)
            .saturating_add((byte - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: colon_at + 1 });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: colon_at + 1 });
    }
    Ok(Some(value as u16))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_byte_scan(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(pos) => pos,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .map(|p| authority_start + p)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| if b == b'@' { Some(i) } else { None })
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: authority_start + ats[1] });
    }

    let (userinfo, hostport, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: authority_start });
        }
        (
            Some(authority[..at].to_string()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            parse_port(&suffix[1..], hostport_start + close + 1)?
        } else {
            return Err(ParseError::InvalidHost { at: hostport_start + close + 1 });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.find(':');
        let (host_text, port) = if let Some(index) = split {
            (&hostport[..index], parse_port(&hostport[index + 1..], hostport_start + index)?)
        } else {
            (hostport, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        if let Some((index, _)) = host_text
            .bytes()
            .enumerate()
            .find(|(_, byte)| !(byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'-' | b'_')))
        {
            return Err(ParseError::InvalidHost { at: hostport_start + index });
        }
        (host_text.to_ascii_lowercase(), port, false)
    };

    let mut fragment = None;
    let before_fragment_end = if let Some(hash) = input[authority_end..].find('#') {
        let hash_at = authority_end + hash;
        fragment = Some(input[hash_at + 1..].to_string());
        hash_at
    } else {
        input.len()
    };

    let mut query = None;
    let before_query_end = if let Some(question) = input[authority_end..before_fragment_end].find('?') {
        let question_at = authority_end + question;
        query = Some(input[question_at + 1..before_fragment_end].to_string());
        question_at
    } else {
        before_fragment_end
    };

    let path = if before_query_end == authority_end {
        "/".to_string()
    } else {
        input[authority_end..before_query_end].to_string()
    };

    if let Some(ref info) = userinfo {
        validate_component(Component::UserInfo, info, authority_start)?;
    }
    if before_query_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let Some(ref q) = query {
        let start = input[authority_end..before_fragment_end]
            .find('?')
            .map(|p| authority_end + p + 1)
            .unwrap_or(before_fragment_end);
        validate_component(Component::Query, q, start)?;
    }
    if let Some(ref f) = fragment {
        let start = input.find('#').map(|p| p + 1).unwrap_or(input.len());
        validate_component(Component::Fragment, f, start)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = match hex_value(bytes[i + 1]) {
                Some(value) => value,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            let low = match hex_value(bytes[i + 2]) {
                Some(value) => value,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            decoded.push((high << 4) | low);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(value) => Ok(value.to_string()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            let original_at = origins.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

fn component_safe(component: Component, byte: u8) -> bool {
    if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
        return true;
    }
    match component {
        Component::UserInfo => byte == b':',
        Component::Path => byte == b'/',
        Component::Query => matches!(byte, b'/' | b'?'),
        Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in text.as_bytes() {
        if component_safe(component, byte) {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn codec_error_to_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_error_to_path)?;

    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut last_navigation = false;

    for segment in &segments {
        let decoded = decode_component(Component::Path, segment).map_err(codec_error_to_path)?;
        if decoded == "." {
            last_navigation = true;
        } else if decoded == ".." {
            stack.pop();
            last_navigation = true;
        } else {
            stack.push(segment);
            last_navigation = false;
        }
    }

    if last_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_string())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    attach_path(&mut base_url).map_err(ResolveError::Path)?;
    global_byte_scan(reference).map_err(|e| match e {
        ParseError::InvalidByte { at } => ResolveError::Reference(ParseError::InvalidByte { at }),
        other => ResolveError::Reference(other),
    })?;

    if let Some(delimiter) = reference.find(|c| matches!(c, '/' | '?' | '#')) {
        if reference[..delimiter].contains(':') {
            let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
            attach_path(&mut url).map_err(ResolveError::Path)?;
            return Ok(url);
        }
    } else if reference.contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        attach_path(&mut url).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        attach_path(&mut url).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let fragment_at = reference.find('#');
    let before_fragment = fragment_at.map(|at| &reference[..at]).unwrap_or(reference);
    let fragment = fragment_at.map(|at| &reference[at + 1..]);

    let query_at = before_fragment.find('?');
    let ref_path = query_at.map(|at| &before_fragment[..at]).unwrap_or(before_fragment);
    let ref_query = query_at.map(|at| &before_fragment[at + 1..]);

    let selected_path = if ref_path.is_empty() {
        base_url.path.clone()
    } else if ref_path.starts_with('/') {
        ref_path.to_string()
    } else {
        let slash = base_url.path.rfind('/').unwrap_or(0);
        format!("{}{}", &base_url.path[..=slash], ref_path)
    };

    let selected_query = if ref_path.is_empty() && ref_query.is_none() {
        base_url.query.clone()
    } else {
        ref_query.map(str::to_string)
    };

    let userinfo = base_url
        .userinfo
        .as_ref()
        .map(|value| format!("{}@", value))
        .unwrap_or_default();
    let host = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let port = base_url.port.map(|value| format!(":{}", value)).unwrap_or_default();
    let query = selected_query.map(|value| format!("?{}", value)).unwrap_or_default();
    let fragment = fragment.map(|value| format!("#{}", value)).unwrap_or_default();

    let constructed = format!(
        "{}://{}{}{}{}{}{}",
        base_url.scheme, userinfo, host, port, selected_path, query, fragment
    );
    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    attach_path(&mut url).map_err(ResolveError::Path)?;
    Ok(url)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

fn codec_error_to_query(error: CodecError, offset: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

fn replace_plus(text: &str) -> String {
    text.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();

    for (field_start, field) in {
        let mut items = Vec::new();
        let mut start = 0;
        for (at, byte) in input.bytes().enumerate() {
            if byte == b'&' {
                items.push((start, &input[start..at]));
                start = at + 1;
            }
        }
        items.push((start, &input[start..]));
        items
    } {
        if field.is_empty() {
            continue;
        }
        let equals = field.find('=');
        let (raw_key, raw_value, value_offset) = if let Some(at) = equals {
            (&field[..at], &field[at + 1..], field_start + at + 1)
        } else {
            (field, "", field_start + field.len())
        };

        let key_input = replace_plus(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|e| codec_error_to_query(e, field_start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }

        let value_input = replace_plus(raw_value);
        let value = decode_component(Component::Query, &value_input)
            .map_err(|e| codec_error_to_query(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| *existing == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in text.as_bytes() {
        if byte == b' ' {
            out.push('+');
        } else if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();

    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing_values)) = merged.iter_mut().find(|(existing, _)| existing == key) {
            existing_values.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(&key), form_encode(&value)));
        }
    }
    fields.join("&")
}

fn canonical_segment(segment: &str) -> Result<String, CodecError> {
    let decoded = decode_component(Component::Path, segment)?;
    let mut result = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in decoded.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    Ok(result)
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let parsed = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&parsed.path).map_err(CanonicalError::Path)?;
    let query_map = match parsed.query.as_deref() {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = format!("{}://", parsed.scheme);

    if let Some(userinfo) = parsed.userinfo.as_deref() {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .map_err(|error| CanonicalError::Parse(codec_error_to_parse(error, 0)))?;
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }

    if parsed.ipv6 {
        let address = Ipv6Addr::from_str(&parsed.host).expect("parser validated IPv6 address");
        output.push('[');
        output.push_str(&address.to_string());
        output.push(']');
    } else {
        output.push_str(&parsed.host);
    }

    let keep_port = parsed.port.is_some_and(|port| {
        !((parsed.scheme == "http" && port == 80) || (parsed.scheme == "https" && port == 443))
    });
    if keep_port {
        output.push(':');
        output.push_str(&parsed.port.unwrap().to_string());
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        canonical_segments.push(
            canonical_segment(segment)
                .map_err(|error| CanonicalError::Path(codec_error_to_path(error)))?,
        );
    }
    output.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        output.push('?');
        output.push_str(&serialise_query(&map));
    }

    if let Some(fragment) = parsed.fragment.as_deref() {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|error| CanonicalError::Parse(codec_error_to_parse(error, 0)))?;
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(err) => {
            let decoded_at = err.valid_up_to();
            let original_at = offsets.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(char::from(b));
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(char::from(HEX[(b >> 4) as usize]));
            out.push(char::from(HEX[(b & 0x0f) as usize]));
        }
    }
    out
}

fn invalid_byte(input: &str) -> Option<usize> {
    input
        .as_bytes()

    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },

}

fn parse_port(port: &str, at: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }

    let bytes = port.as_bytes();
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: at + i });
    }

    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }

    let mut value: u32 = 0;
    for &b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add(u32::from(b - b'0')))
        {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }

    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    if let Some(at) = invalid_byte(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let after_marker = colon + 3;
    let authority_end = bytes[after_marker..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|offset| after_marker + offset)
        .unwrap_or(bytes.len());
    let authority = &input[after_marker..authority_end];

    let mut at_positions = authority.match_indices('@').map(|(i, _)| i);
    let first_at = at_positions.next();
    if let Some(second_at) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo {
            at: after_marker + second_at,
        });
    }

    let (userinfo, hostport, hostport_start) = match first_at {
        Some(i) => {
            if i == 0 {
                return Err(ParseError::InvalidUserInfo { at: after_marker });
            }
            (
                Some(authority[..i].to_owned()),
                &authority[i + 1..],
                after_marker + i + 1,
            )
        }
        None => (None, authority, after_marker),
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let opening_at = hostport_start;
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: opening_at }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: opening_at });
        }

        let suffix = &hostport[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let host_colon = hostport.find(':');
        let (host_part, port_part, port_start) = match host_colon {
            Some(i) => (
                &hostport[..i],
                Some(&hostport[i + 1..]),
                hostport_start + i + 1,
            ),
            None => (hostport, None, 0),
        };

        if host_part.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host_part.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }

        let parsed_port = match port_part {
            Some(p) => Some(parse_port(p, port_start)?),
            None => None,
        };
        (host_part.to_ascii_lowercase(), parsed_port)
    };

    let fragment_pos = input[authority_end..]
        .find('#')
        .map(|i| authority_end + i);
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());
    let query_pos = input[authority_end..before_fragment_end]
        .find('?')
        .map(|i| authority_end + i);

    let query = query_pos.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|f| input[f + 1..].to_owned());

    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path = if authority_end == path_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    if let Some(ref user) = userinfo {
        let user_start = after_marker;
        validate_component(user, user_start, Component::UserInfo)?;
    }
    if path != "/" || authority_end != path_end {
        validate_component(&path, authority_end, Component::Path)?;
    }
    if let (Some(q), Some(qpos)) = (&query, query_pos) {
        validate_component(q, qpos + 1, Component::Query)?;
    }
    if let (Some(f), Some(fpos)) = (&fragment, fragment_pos) {
        validate_component(f, fpos + 1, Component::Fragment)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (i, raw) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 + segment_offset(&raw_segments, i) },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 + segment_offset(&raw_segments, i) },
        })?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if is_dot {
            // no-op
        } else if is_dotdot {
            stack.pop();
        } else {
            stack.push(raw);
        }
        if i + 1 == raw_segments.len() {
            terminal_navigation = is_dot || is_dotdot;
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn segment_offset(segments: &[&str], index: usize) -> usize {
    let mut offset = 0;
    for segment in segments.iter().take(index) {
        offset += segment.len() + 1;
    }
    offset
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn normalize_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn split_reference_components(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = fragment_pos.map(|i| &reference[i + 1..]);
    let query_pos = before_fragment.find('?');
    let path = query_pos.map(|i| &before_fragment[..i]).unwrap_or(before_fragment);
    let query = query_pos.map(|i| &before_fragment[i + 1..]);
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base = normalize_url_path(base_url).map_err(|e| e)?;

    if let Some(at) = invalid_byte(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let first_delim = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delim].contains(':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", normalized_base.scheme, reference);
        let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference_components(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            normalized_base.path.clone(),
            if ref_query.is_none() {
                normalized_base.query.clone()
            } else {
                ref_query.map(str::to_owned)
            },
        )
    } else {
        let path = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = normalized_base.path.rfind('/').unwrap_or(0);
            format!("{}{}", &normalized_base.path[..=slash], ref_path)
        };
        (path, ref_query.map(str::to_owned))
    };

    let mut constructed = format!("{}://", normalized_base.scheme);
    if let Some(userinfo) = &normalized_base.userinfo {
        constructed.push_str(userinfo);
        constructed.push('@');
    }
    if normalized_base.ipv6 {
        constructed.push('[');
        constructed.push_str(&normalized_base.host);
        constructed.push(']');
    } else {
        constructed.push_str(&normalized_base.host);
    }
    if let Some(port) = normalized_base.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&selected_path);
    if let Some(query) = &selected_query {
        constructed.push('?');
        constructed.push_str(query);
    }
    if let Some(fragment) = ref_fragment {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalize_url_path(url)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

fn form_decode_piece(input: &str, offset: usize) -> Result<String, QueryError> {
    let replaced = input.replace('+', " ");
    decode_component(Component::Query, &replaced).map_err(|error| match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    })
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;

    for field in input.split('&') {
        if !field.is_empty() {
            let (raw_key, raw_value, value_offset) = match field.find('=') {
                Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
                None => (field, "", field_start + field.len()),
            };
            let key = form_decode_piece(raw_key, field_start)?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: field_start });
            }
            let value = form_decode_piece(raw_value, value_offset)?;

            if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| existing == &key) {
                values.push(value);
            } else {
                entries.push((key, vec![value]));
            }
        }
        field_start += field.len() + 1;
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(char::from(b));
        } else {
            out.push('%');
            out.push(char::from(HEX[(b >> 4) as usize]));
            out.push(char::from(HEX[(b & 0x0f) as usize]));
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, collected)) = merged.iter_mut().find(|(existing, _)| *existing == key) {
            collected.extend(values.iter().map(String::as_str));
        } else {
            merged.push((key, values.iter().map(String::as_str).collect()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(value)));
        }
    }
    fields.join("&")
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn map_parse_codec(error: CodecError) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at },
    }
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded =
            decode_component(Component::UserInfo, userinfo).map_err(|e| CanonicalError::Parse(map_parse_codec(e)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).expect("parsed IPv6 host remains valid");
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    let keep_port = match (url.scheme.as_str(), url.port) {
        ("http", Some(80)) | ("https", Some(443)) => false,
        (_, Some(_)) => true,
        (_, None) => false,
    };
    if keep_port {
        out.push(':');
        out.push_str(&url.port.unwrap().to_string());
    }

    let segments: Result<Vec<String>, CanonicalError> = path
        .split('/')
        .map(|segment| {
            let decoded =
                decode_component(Component::Path, segment).map_err(|e| CanonicalError::Parse(map_parse_codec(e)))?;
            let mut encoded = String::with_capacity(decoded.len());
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            for &b in decoded.as_bytes() {
                if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
                    encoded.push(char::from(b));
                } else {
                    encoded.push('%');
                    encoded.push(char::from(HEX[(b >> 4) as usize]));
                    encoded.push(char::from(HEX[(b & 0x0f) as usize]));
                }
            }
            Ok(encoded)
        })
        .collect();
    out.push_str(&segments?.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(map_parse_codec(e)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(out)
}
