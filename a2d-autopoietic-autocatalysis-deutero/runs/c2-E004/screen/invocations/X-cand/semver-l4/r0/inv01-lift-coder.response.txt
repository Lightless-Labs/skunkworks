use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn missing_core_byte(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }
        if pos == start {
            if missing_core_byte(bytes.get(pos).copied()) {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &b in &bytes[start..pos] {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        core[field] = value;

        if field < 2 {
            if bytes.get(pos) != Some(&b'.') {
                if missing_core_byte(bytes.get(pos).copied()) {
                    return Err(VersionError::MissingCore { at: pos });
                }
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if bytes.get(pos) == Some(&b'-') {
        pos += 1;
        let pre_start = pos;
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());

        if pre_start == end {
            return Err(VersionError::EmptyIdentifier { at: pre_start });
        }

        let mut start = pre_start;
        while start < end {
            let mut stop = start;
            while stop < end && bytes[stop] != b'.' {
                if !bytes[stop].is_ascii_alphanumeric() && bytes[stop] != b'-' {
                    return Err(VersionError::InvalidChar { at: stop });
                }
                stop += 1;
            }
            if stop == start {
                return Err(VersionError::EmptyIdentifier { at: start });
            }
            let id = &s[start..stop];
            if id.bytes().all(is_digit) && id.len() > 1 && id.as_bytes()[0] == b'0' {
                return Err(VersionError::LeadingZero { at: start });
            }
            pre.push(id.to_string());
            if stop == end {
                break;
            }
            start = stop + 1;
            if start == end {
                return Err(VersionError::EmptyIdentifier { at: start });
            }
        }
        pos = end;
    }

    if bytes.get(pos) == Some(&b'+') {
        pos += 1;
        let start0 = pos;
        if start0 == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: start0 });
        }
        let mut start = start0;
        while start < bytes.len() {
            let mut stop = start;
            while stop < bytes.len() && bytes[stop] != b'.' {
                if !bytes[stop].is_ascii_alphanumeric() && bytes[stop] != b'-' {
                    return Err(VersionError::InvalidChar { at: stop });
                }
                stop += 1;
            }
            if stop == start {
                return Err(VersionError::EmptyIdentifier { at: start });
            }
            build.push(s[start..stop].to_string());
            if stop == bytes.len() {
                break;
            }
            start = stop + 1;
            if start == bytes.len() {
                return Err(VersionError::EmptyIdentifier { at: start });
            }
        }
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn compare_numeric_ids(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn id_is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = id_is_numeric(x);
        let yn = id_is_numeric(y);
        let cmp = match (xn, yn) {
            (true, true) => compare_numeric_ids(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if cmp != Ordering::Equal {
            return cmp;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum CmpOp {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CmpOp,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Copy)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens<'a>(s: &'a str, base: usize) -> Vec<Token<'a>> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) {
            i += 1;
        }
        result.push(Token {
            text: &s[start..i],
            start: base + start,
        });
    }
    result
}

fn cmp(op: CmpOp, version: Version) -> Comparator {
    Comparator { op, version }
}

fn add_anchor(anchors: &mut Vec<(u64, u64, u64)>, v: &Version) {
    if !v.pre.is_empty() {
        anchors.push((v.major, v.minor, v.patch));
    }
}

fn upper_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
enum Operand {
    Full(Version),
    Partial {
        fields: Vec<Option<u64>>,
        numeric: usize,
    },
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn parse_operand(raw: &str, start: usize) -> Result<Operand, RangeError> {
    let has_suffix = raw.contains('-') || raw.contains('+');
    let fields_raw: Vec<&str> = raw.split('.').collect();
    let has_wild = fields_raw.iter().any(|x| is_wildcard(x));
    if has_suffix || (fields_raw.len() >= 3 && !has_wild) {
        return parse_version(raw)
            .map(Operand::Full)
            .map_err(|error| RangeError::Version { at: start, error });
    }

    if raw.is_empty() || fields_raw.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }
    let mut fields = Vec::new();
    let mut numeric = 0;
    let mut wildcard_seen = false;
    for field in fields_raw {
        if field.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if is_wildcard(field) {
            wildcard_seen = true;
            fields.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(is_digit)
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: start });
            }
            let value = field.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            fields.push(Some(value));
            numeric += 1;
        }
    }
    Ok(Operand::Partial { fields, numeric })
}

fn padded(fields: &[Option<u64>]) -> (u64, u64, u64) {
    let mut out = [0u64; 3];
    for (i, field) in fields.iter().enumerate().take(3) {
        out[i] = field.unwrap_or(0);
    }
    (out[0], out[1], out[2])
}

fn partial_end(fields: &[Option<u64>], numeric: usize, start: usize) -> Result<Version, RangeError> {
    let (mut major, mut minor, mut patch) = padded(fields);
    match numeric {
        1 => major = major.checked_add(1).ok_or(RangeError::Syntax { at: start })?,
        2 => minor = minor.checked_add(1).ok_or(RangeError::Syntax { at: start })?,
        _ => patch = patch.checked_add(1).ok_or(RangeError::Syntax { at: start })?,
    }
    if numeric == 1 {
        minor = 0;
        patch = 0;
    } else if numeric == 2 {
        patch = 0;
    }
    Ok(upper_version(major, minor, patch))
}

fn add_operand(
    arm: &mut Arm,
    token: Token<'_>,
    mode: Option<u8>,
) -> Result<(), RangeError> {
    let raw = if mode.is_some() { &token.text[1..] } else { token.text };
    let start = token.start + usize::from(mode.is_some());
    if raw.is_empty() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let opnd = parse_operand(raw, start)?;
    match mode {
        None => match opnd {
            Operand::Full(v) => {
                add_anchor(&mut arm.anchors, &v);
                arm.comparators.push(cmp(CmpOp::Eq, v));
            }
            Operand::Partial { fields, numeric } => {
                if numeric == 0 {
                    return Ok(());
                }
                let (major, minor, patch) = padded(&fields);
                arm.comparators.push(cmp(
                    CmpOp::Ge,
                    Version { major, minor, patch, pre: Vec::new(), build: Vec::new() },
                ));
                if numeric < 3 {
                    arm.comparators.push(cmp(
                        CmpOp::Lt,
                        partial_end(&fields, numeric, token.start)?,
                    ));
                }
            }
        },
        Some(b'^') | Some(b'~') => {
            let (fields, numeric, mut lower) = match opnd {
                Operand::Full(v) => {
                    let fs = vec![Some(v.major), Some(v.minor), Some(v.patch)];
                    (fs, 3, v)
                }
                Operand::Partial { fields, numeric } => {
                    if numeric == 0 {
                        return Err(RangeError::Syntax { at: token.start });
                    }
                    let (major, minor, patch) = padded(&fields);
                    (
                        fields,
                        numeric,
                        Version { major, minor, patch, pre: Vec::new(), build: Vec::new() },
                    )
                }
            };
            add_anchor(&mut arm.anchors, &lower);
            arm.comparators.push(cmp(CmpOp::Ge, lower.clone()));
            let (mut major, mut minor, mut patch) = (lower.major, lower.minor, lower.patch);
            let mode_byte = mode.unwrap();
            if mode_byte == b'~' {
                if numeric == 1 {
                    major = major.checked_add(1).ok_or(RangeError::Syntax { at: token.start })?;
                    minor = 0;
                    patch = 0;
                } else {
                    minor = minor.checked_add(1).ok_or(RangeError::Syntax { at: token.start })?;
                    patch = 0;
                }
            } else if major > 0 || numeric == 1 {
                major = major.checked_add(1).ok_or(RangeError::Syntax { at: token.start })?;
                minor = 0;
                patch = 0;
            } else if minor > 0 || numeric == 2 {
                minor = minor.checked_add(1).ok_or(RangeError::Syntax { at: token.start })?;
                patch = 0;
            } else {
                patch = patch.checked_add(1).ok_or(RangeError::Syntax { at: token.start })?;
            }
            lower.build.clear();
            arm.comparators.push(cmp(CmpOp::Lt, upper_version(major, minor, patch)));
            let _ = fields;
        }
        _ => unreachable!(),
    }
    Ok(())
}

fn parse_explicit_comparator(arm: &mut Arm, tok: Token<'_>) -> Result<(), RangeError> {
    let bytes = tok.text.as_bytes();
    let (op, n) = if bytes.starts_with(b">=") {
        (CmpOp::Ge, 2)
    } else if bytes.starts_with(b"<=") {
        (CmpOp::Le, 2)
    } else if bytes.starts_with(b">") {
        (CmpOp::Gt, 1)
    } else if bytes.starts_with(b"<") {
        (CmpOp::Lt, 1)
    } else if bytes.starts_with(b"=") {
        (CmpOp::Eq, 1)
    } else {
        unreachable!()
    };
    if n >= bytes.len() || !is_digit(bytes[n]) {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let raw = &tok.text[n..];
    let v = parse_version(raw).map_err(|error| RangeError::Version {
        at: tok.start + n,
        error,
    })?;
    add_anchor(&mut arm.anchors, &v);
    arm.comparators.push(cmp(op, v));
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token<'_>], minus_index: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if minus_index != 1 || tokens.len() != 3 {
        return Err(RangeError::Syntax { at: tokens[minus_index].start });
    }
    let lower = parse_operand(tokens[0].text, tokens[0].start)?;
    let upper = parse_operand(tokens[2].text, tokens[2].start)?;
    let (lf, ln, lv) = match lower {
        Operand::Full(v) => (vec![Some(v.major), Some(v.minor), Some(v.patch)], 3, v),
        Operand::Partial { fields, numeric } => {
            if numeric == 0 {
                return Err(RangeError::Syntax { at: tokens[0].start });
            }
            let (major, minor, patch) = padded(&fields);
            (fields, numeric, Version { major, minor, patch, pre: Vec::new(), build: Vec::new() })
        }
    };
    let upper_is_full = matches!(&upper, Operand::Full(_));
    let (uf, un, uv) = match upper {
        Operand::Full(v) => (vec![Some(v.major), Some(v.minor), Some(v.patch)], 3, v),
        Operand::Partial { fields, numeric } => {
            if numeric == 0 {
                return Err(RangeError::Syntax { at: tokens[2].start });
            }
            let (major, minor, patch) = padded(&fields);
            (fields, numeric, Version { major, minor, patch, pre: Vec::new(), build: Vec::new() })
        }
    };
    let _ = (lf, ln);
    add_anchor(&mut arm.anchors, &lv);
    add_anchor(&mut arm.anchors, &uv);
    arm.comparators.push(cmp(CmpOp::Ge, lv));
    if upper_is_full {
        arm.comparators.push(cmp(CmpOp::Le, uv));
    } else {
        arm.comparators.push(cmp(CmpOp::Lt, partial_end(&uf, un, tokens[2].start)?));
    }
    Ok(())
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    if let Some(i) = tokens.iter().position(|t| t.text == "-") {
        parse_hyphen_arm(tokens, i, &mut arm)?;
        return Ok(arm);
    }

    for tok in tokens {
        let first = tok.text.as_bytes().first().copied();
        match first {
            Some(b'>') | Some(b'<') | Some(b'=') => parse_explicit_comparator(&mut arm, *tok)?,
            Some(b'^') | Some(b'~') => add_operand(&mut arm, *tok, first)?,
            Some(b) if is_digit(b) || matches!(b, b'*' | b'x' | b'X') => {
                add_operand(&mut arm, *tok, None)?
            }
            _ => return Err(RangeError::Syntax { at: tok.start }),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    if bytes.iter().all(|&b| is_ws(b)) {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;
    let mut previous_sep = None;
    while i < bytes.len() {
        if !is_ws(bytes[i]) {
            i += 1;
            continue;
        }
        let ws_start = i;
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        let left = &s[start..ws_start];
        let right_start = i;
        if left == "||" {
            let _ = right_start;
        }
        let mut token_start = ws_start;
        while token_start > start && is_ws(bytes[token_start - 1]) {
            token_start -= 1;
        }
        let mut token_end = right_start;
        while token_end < bytes.len() && !is_ws(bytes[token_end]) {
            token_end += 1;
        }
        if token_end > right_start && &s[right_start..token_end] == "||" {
            let before = split_tokens(&s[start..token_start], start);
            if before.is_empty() {
                return Err(RangeError::Empty { at: token_start });
            }
            arms.push(parse_arm(&before)?);
            previous_sep = Some(right_start);
            start = token_end;
            i = token_end;
        }
    }

    let tail = split_tokens(&s[start..], start);
    if tail.is_empty() {
        if previous_sep.is_some() {
            return Err(RangeError::Empty { at: s.len() });
        }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(&tail)?);
    Ok(Range { arms })
}

fn comparator_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        CmpOp::Eq => ord == Ordering::Equal,
        CmpOp::Lt => ord == Ordering::Less,
        CmpOp::Le => ord != Ordering::Greater,
        CmpOp::Gt => ord == Ordering::Greater,
        CmpOp::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparators.iter().all(|c| comparator_matches(v, c)) {
        return false;
    }
    include_prerelease
        || v.pre.is_empty()
        || arm.anchors.contains(&(v.major, v.minor, v.patch))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(b) = best {
            let ord = compare_versions(v, &parsed[b]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn core_next(v: &Version) -> Option<(u64, u64, u64)> {
    if v.patch < u64::MAX {
        Some((v.major, v.minor, v.patch + 1))
    } else if v.minor < u64::MAX {
        Some((v.major, v.minor + 1, 0))
    } else if v.major < u64::MAX {
        Some((v.major + 1, 0, 0))
    } else {
        None
    }
}

fn pre_zero() -> Vec<String> {
    vec!["0".to_string()]
}

fn has_stable_witness(_arm: &Arm) -> bool {
    true
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            if arms_intersect(aa, bb, include_prerelease) {
                return true;
            }
        }
    }
    false
}

fn arms_intersect(a: &Arm, b: &Arm, include_prerelease: bool) -> bool {
    let combined = Arm {
        comparators: a.comparators.iter().chain(b.comparators.iter()).cloned().collect(),
        anchors: a.anchors.iter().chain(b.anchors.iter()).copied().collect(),
    };
    if include_prerelease {
        return find_witness(&combined, true, &[]).is_some();
    }

    if let Some(v) = find_witness(&combined, false, &[]) {
        if v.pre.is_empty() {
            return true;
        }
    }
    let common: Vec<_> = a
        .anchors
        .iter()
        .copied()
        .filter(|x| b.anchors.contains(x))
        .collect();
    find_witness(&combined, true, &common).is_some()
}

fn find_witness(arm: &Arm, allow_pre: bool, allowed: &[(u64, u64, u64)]) -> Option<Version> {
    let mut candidates = Vec::new();
    for c in &arm.comparators {
        let v = &c.version;
        candidates.push(Version {
            major: v.major,
            minor: v.minor,
            patch: v.patch,
            pre: Vec::new(),
            build: Vec::new(),
        });
        candidates.push(Version {
            major: v.major,
            minor: v.minor,
            patch: v.patch,
            pre: pre_zero(),
            build: Vec::new(),
        });
        if let Some((major, minor, patch)) = core_next(v) {
            candidates.push(Version {
                major,
                minor,
                patch,
                pre: Vec::new(),
                build: Vec::new(),
            });
            candidates.push(Version {
                major,
                minor,
                patch,
                pre: pre_zero(),
                build: Vec::new(),
            });
        }
    }
    for &(major, minor, patch) in allowed {
        candidates.push(Version {
            major,
            minor,
            patch,
            pre: pre_zero(),
            build: Vec::new(),
        });
    }
    candidates.push(Version { major: 0, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new() });
    for v in candidates {
        if (allow_pre || v.pre.is_empty())
            && (allow_pre || allowed.contains(&(v.major, v.minor, v.patch)) || v.pre.is_empty())
            && arm.comparators.iter().all(|c| comparator_matches(&v, c))
        {
            return Some(v);
        }
    }
    None
}

fn strip_build(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn canonical_arm(arm: &Arm) -> Option<String> {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.comparators {
        match c.op {
            CmpOp::Eq => {
                let lo = (c.version.clone(), true);
                let hi = (c.version.clone(), true);
                lower = Some(match lower {
                    Some(old) if compare_versions(&old.0, &lo.0) == Ordering::Greater => old,
                    Some(old) if compare_versions(&old.0, &lo.0) == Ordering::Equal => (old.0, old.1 && lo.1),
                    _ => lo,
                });
                upper = Some(match upper {
                    Some(old) if compare_versions(&old.0, &hi.0) == Ordering::Less => old,
                    Some(old) if compare_versions(&old.0, &hi.0) == Ordering::Equal => (old.0, old.1 && hi.1),
                    _ => hi,
                });
            }
            CmpOp::Gt | CmpOp::Ge => {
                let inclusive = matches!(c.op, CmpOp::Ge);
                lower = Some(match lower {
                    Some(old) if compare_versions(&old.0, &c.version) == Ordering::Greater => old,
                    Some(old) if compare_versions(&old.0, &c.version) == Ordering::Equal => (old.0, old.1 && inclusive),
                    _ => (c.version.clone(), inclusive),
                });
            }
            CmpOp::Lt | CmpOp::Le => {
                let inclusive = matches!(c.op, CmpOp::Le);
                upper = Some(match upper {
                    Some(old) if compare_versions(&old.0, &c.version) == Ordering::Less => old,
                    Some(old) if compare_versions(&old.0, &c.version) == Ordering::Equal => (old.0, old.1 && inclusive),
                    _ => (c.version.clone(), inclusive),
                });
            }
        }
    }

    let low_ok = lower.as_ref().map(|x| if x.1 {
        compare_versions(&x.0, &x.0) == Ordering::Equal
    } else {
        true
    }).unwrap_or(true);
    let _ = low_ok;

    if !has_stable_witness(arm) && arm.anchors.is_empty() {
        return None;
    }
    let mut anchors = arm.anchors.clone();
    anchors.sort_unstable();
    anchors.dedup();
    let anchors_s = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
    };
    let lo = lower.map(|(v, inclusive)| format!("{}{}", if inclusive { ">=" } else { ">" }, strip_build(&v))).unwrap_or_else(|| "*".to_string());
    let hi = upper.map(|(v, inclusive)| format!("{}{}", if inclusive { "<=" } else { "<" }, strip_build(&v))).unwrap_or_else(|| "*".to_string());
    Some(format!("[{lo};{hi};pre={anchors_s}]"))
}

pub fn canonical(r: &Range) -> String {
    let mut arms: Vec<String> = r.arms.iter().filter_map(canonical_arm).collect();
    arms.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    arms.dedup();
    if arms.is_empty() {
        "EMPTY".to_string()
    } else {
        arms.join(" || ")
    }
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    parsed: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            parsed.push(r);
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, parsed)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn eval_expr(expr: &SetExpr, candidate: &Version, include: bool, ranges: &mut std::slice::Iter<'_, Range>) -> bool {
    match expr {
        SetExpr::Range(_) => ranges.next().map(|r| matches_with(candidate, r, include)).unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|c| eval_expr(c, candidate, include, ranges)),
        SetExpr::Intersection(children) => children.iter().all(|c| eval_expr(c, candidate, include, ranges)),
    }
}

fn eval_expr_indexed(expr: &SetExpr, candidate: &Version, include: bool, map: &mut Vec<Range>) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let r = map.remove(0);
            matches_with(candidate, &r, include)
        }
        SetExpr::Union(children) => {
            let mut result = false;
            for child in children {
                result |= eval_expr_indexed(child, candidate, include, map);
            }
            result
        }
        SetExpr::Intersection(children) => {
            let mut result = true;
            for child in children {
                result &= eval_expr_indexed(child, candidate, include, map);
            }
            result
        }
    }
}

fn expr_match(expr: &SetExpr, candidate: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(candidate, &r, include)).unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|c| expr_match(c, candidate, include)),
        SetExpr::Intersection(children) => children.iter().all(|c| expr_match(c, candidate, include)),
    }
}

fn conflict_labels(
    constraints: &[Constraint],
    candidates: &[Version],
    include: bool,
) -> Vec<String> {
    if candidates.is_empty() {
        return Vec::new();
    }
    let n = constraints.len();
    for size in 1..=n {
        let mut indices = Vec::new();
        if let Some(found) = find_conflict_subset(0, size, n, &mut indices, constraints, candidates, include) {
            let mut labels: Vec<String> = found.into_iter().map(|i| constraints[i].label.clone()).collect();
            labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            return labels;
        }
    }
    Vec::new()
}

fn find_conflict_subset(
    next: usize,
    remaining: usize,
    n: usize,
    selected: &mut Vec<usize>,
    constraints: &[Constraint],
    candidates: &[Version],
    include: bool,
) -> Option<Vec<usize>> {
    if remaining == 0 {
        let satisfiable = candidates.iter().any(|v| {
            selected.iter().all(|&i| expr_match(&constraints[i].expr, v, include))
        });
        return if satisfiable { None } else { Some(selected.clone()) };
    }
    if n - next < remaining {
        return None;
    }
    for i in next..=n - remaining {
        selected.push(i);
        if let Some(found) = find_conflict_subset(
            i + 1, remaining - 1, n, selected, constraints, candidates, include,
        ) {
            return Some(found);
        }
        selected.pop();
    }
    None
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = Vec::<&str>::new();
    for p in packages {
        if names.contains(&p.name.as_str()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
        names.push(&p.name);
    }

    for p in packages {
        let mut labels = Vec::<&str>::new();
        for c in &p.constraints {
            if labels.contains(&c.label.as_str()) {
                return Err(SelectError::DuplicateLabel {
                    package: p.name.clone(),
                    label: c.label.clone(),
                });
            }
            labels.push(&c.label);
        }
    }

    let mut parsed_candidates = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            let v = parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(),
                index,
                error,
            })?;
            vs.push(v);
        }
        parsed_candidates.push(vs);
    }

    for p in packages {
        for c in &p.constraints {
            let mut ranges = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut ranges)?;
        }
    }

    let mut result = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let candidates = &parsed_candidates[pi];
        let qualifying: Vec<usize> = candidates
            .iter()
            .enumerate()
            .filter(|(_, v)| p.constraints.iter().all(|c| expr_match(&c.expr, v, include_prerelease)))
            .map(|(i, _)| i)
            .collect();

        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflict_labels(&p.constraints, candidates, include_prerelease),
            });
        }

        let chosen = match preference {
            Preference::Highest => *qualifying.iter().max_by(|&&i, &&j| compare_versions(&candidates[i], &candidates[j])).unwrap(),
            Preference::Lowest => *qualifying.iter().min_by(|&&i, &&j| compare_versions(&candidates[i], &candidates[j])).unwrap(),
            Preference::StableFirst => {
                let stable: Vec<usize> = qualifying.iter().copied().filter(|&i| candidates[i].pre.is_empty()).collect();
                let pool = if stable.is_empty() { &qualifying } else { &stable };
                *pool.iter().max_by(|&&i, &&j| compare_versions(&candidates[i], &candidates[j])).unwrap()
            }
        };
        result.push(Selection {
            package: p.name.clone(),
            version: p.candidates[chosen].clone(),
        });
    }
    Ok(result)
}
