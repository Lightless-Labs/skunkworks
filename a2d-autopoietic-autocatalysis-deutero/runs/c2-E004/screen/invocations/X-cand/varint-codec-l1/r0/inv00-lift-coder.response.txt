#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut encoded = Vec::new();

    loop {
        let mut byte = (value & 0x7f) as u8;
        value >>= 7;

        if value != 0 {
            byte |= 0x80;
        }

        encoded.push(byte);

        if value == 0 {
            break;
        }
    }

    encoded
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;

    for index in 0..10 {
        let Some(&byte) = input.get(start + index) else {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        };

        let payload = (byte & 0x7f) as u64;

        if index == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow {
                    offset: start + 9,
                    start,
                });
            }
            value |= payload << 63;
            return Ok((value, start + 10));
        }

        value |= payload << (index * 7);

        if byte & 0x80 == 0 {
            return Ok((value, start + index + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count);
    let mut cursor = start;

    for _ in 0..count {
        let (value, next_cursor) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next_cursor;
    }

    Ok((values, cursor))
}
