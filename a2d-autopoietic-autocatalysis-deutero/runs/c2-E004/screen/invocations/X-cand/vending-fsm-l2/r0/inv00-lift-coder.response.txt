```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::<String>::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|byte| byte.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match &self.selected {
            Some(id) => id.clone(),
            None => return Err(Error::NoSelection),
        };

        let price = self
            .products
            .iter()
            .find(|product| product.id == selected_id)
            .map(|product| product.price)
            .expect("selected products are always present in the catalog");

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = if amount == 0 {
            Vec::new()
        } else {
            let mut greedy_counts = [0u16; 4];
            let mut remaining = amount;
            for index in (0..4).rev() {
                let denomination = DENOMINATIONS[index];
                let count = self.tubes[index].min(remaining / denomination);
                greedy_counts[index] = count;
                remaining -= count * denomination;
            }

            if remaining == 0 {
                counts_to_descending_coins(greedy_counts)
            } else {
                match best_exact_change(amount, self.tubes) {
                    Some(counts) => counts_to_descending_coins(counts),
                    None => return Err(Error::NoChange { amount }),
                }
            }
        };

        let mut proposed_tubes = self.tubes;
        for &coin in &change {
            let index = denomination_index(coin).expect("change uses valid denominations");
            proposed_tubes[index] -= 1;
        }
        for &coin in &self.escrow {
            let index = denomination_index(coin).expect("escrow uses valid denominations");
            proposed_tubes[index] += 1;
            if proposed_tubes[index] > 100 {
                return Err(Error::TubeLimit { coin });
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt {
            id: selected_id,
            price,
            change,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.selected = None;
        self.credit = 0;
        Ok(refunded)
    }
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&denomination| denomination == coin)
}

fn counts_to_descending_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

fn best_exact_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<([u16; 4], u16)> = None;

    for tens in 0..=tubes[3].min(amount / 10) {
        for fours in 0..=tubes[2].min(amount / 4) {
            for threes in 0..=tubes[1].min(amount / 3) {
                let subtotal = tens * 10 + fours * 4 + threes * 3;
                if subtotal > amount {
                    continue;
                }
                let ones = amount - subtotal;
                if ones > tubes[0] {
                    continue;
                }

                let counts = [ones, threes, fours, tens];
                let total = ones + threes + fours + tens;
                let is_better = match best {
                    None => true,
                    Some((best_counts, best_total)) => {
                        total < best_total
                            || (total == best_total
                                && (counts[3], counts[2], counts[1], counts[0])
                                    > (
                                        best_counts[3],
                                        best_counts[2],
                                        best_counts[1],
                                        best_counts[0],
                                    ))
                    }
                };
                if is_better {
                    best = Some((counts, total));
                }
            }
        }
    }

    best.map(|(counts, _)| counts)
}
```
