```rust
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashSet::new();
        for product in &products {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        const DENOMS: [u16; 4] = [1, 3, 4, 10];
        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = match coin {
            1 => 0,
            3 => 1,
            4 => 2,
            10 => 3,
            _ => return Err(Error::BadCoin { coin }),
        };

        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        let _ = index;
        self.escrow.push(coin);
        self.credit = new_credit as u16;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|p| p.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_ref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };
        let product = self
            .products
            .iter()
            .find(|p| p.id == *selected_id)
            .expect("selected product is always in the catalog");
        let price = product.price;
        let id = product.id.clone();

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let mut change_counts = [0u16; 4];
        let mut remainder = amount;

        for i in [3usize, 2, 1, 0] {
            let coin = [1u16, 3, 4, 10][i];
            let take = self.tubes[i].min(remainder / coin);
            change_counts[i] = take;
            remainder -= take * coin;
        }

        if remainder != 0 {
            let mut best: Option<[u16; 4]> = None;
            let mut best_total = u32::MAX;

            fn search(
                index: usize,
                remaining: u16,
                tubes: &[u16; 4],
                current: &mut [u16; 4],
                best: &mut Option<[u16; 4]>,
                best_total: &mut u32,
            ) {
                const DENOMS: [u16; 4] = [1, 3, 4, 10];
                if index == 4 {
                    if remaining == 0 {
                        let total: u32 = current.iter().map(|&n| n as u32).sum();
                        let better = total < *best_total
                            || (total == *best_total
                                && best.map_or(true, |old| {
                                    for i in (0..4).rev() {
                                        if current[i] != old[i] {
                                            return current[i] > old[i];
                                        }
                                    }
                                    false
                                }));
                        if better {
                            *best_total = total;
                            *best = Some(*current);
                        }
                    }
                    return;
                }

                let coin = DENOMS[index];
                let max_take = tubes[index].min(remaining / coin);
                for take in 0..=max_take {
                    current[index] = take;
                    search(
                        index + 1,
                        remaining - take * coin,
                        tubes,
                        current,
                        best,
                        best_total,
                    );
                }
                current[index] = 0;
            }

            search(
                0,
                amount,
                &self.tubes,
                &mut [0; 4],
                &mut best,
                &mut best_total,
            );

            match best {
                Some(counts) => change_counts = counts,
                None => return Err(Error::NoChange { amount }),
            }
        }

        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= change_counts[i];
        }

        for coin in &self.escrow {
            let index = match *coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed[index] += 1;
        }

        const DENOMS: [u16; 4] = [1, 3, 4, 10];
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change_counts[i] {
                change.push([1u16, 3, 4, 10][i]);
            }
        }

        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.selected = None;
        self.credit = 0;
        Ok(returned)
    }
}
```
