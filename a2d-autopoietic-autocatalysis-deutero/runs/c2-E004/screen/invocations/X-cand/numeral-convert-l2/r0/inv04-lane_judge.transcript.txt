#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;
    if let Some(hash_index) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash_index] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash_index + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + i,
                });
            }
        }
    }

    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (i == 0
                || i + 1 == payload.len()
                || digit_value(payload[i - 1]).is_none()
                || digit_value(payload[i + 1]).is_none())
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + i,
            });
        }
    }

    let digit_count = payload.iter().filter(|&&b| b != b'_').count();
    if digit_count == 0 {
        return Err(BaseError::Separator {
            operand,
            index: payload_start,
        });
    }

    let first_digit = payload.iter().copied().find(|&b| b != b'_').unwrap();
    if digit_count >= 2 && digit_value(first_digit) == Some(0) {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(payload
        .iter()
        .filter(|&&b| b != b'_')
        .map(|&b| digit_value(b).unwrap())
        .collect())
}

fn add_small(number: &mut Vec<u8>, value: u8, radix: u8) {
    let mut carry = value as u16;
    let mut i = number.len();
    while carry > 0 && i > 0 {
        i -= 1;
        let sum = number[i] as u16 + carry;
        number[i] = (sum % radix as u16) as u8;
        carry = sum / radix as u16;
    }
    while carry > 0 {
        number.insert(0, (carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

fn multiply_small(number: &mut Vec<u8>, factor: u8, radix: u8) {
    let mut carry = 0u16;
    for digit in number.iter_mut().rev() {
        let product = *digit as u16 * factor as u16 + carry;
        *digit = (product % radix as u16) as u8;
        carry = product / radix as u16;
    }
    while carry > 0 {
        number.insert(0, (carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

fn to_decimal(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut result = vec![0];
    for &digit in digits {
        multiply_small(&mut result, radix, 10);
        add_small(&mut result, digit, 10);
    }
    result
}

fn from_decimal(decimal: &[u8], radix: u8) -> Vec<u8> {
    let mut remaining = decimal.to_vec();
    let mut output = Vec::new();

    while remaining.iter().any(|&d| d != 0) {
        let mut quotient = Vec::with_capacity(remaining.len());
        let mut remainder = 0u16;
        for &digit in &remaining {
            let value = remainder * 10 + digit as u16;
            let q = value / radix as u16;
            remainder = value % radix as u16;
            if !quotient.is_empty() || q != 0 {
                quotient.push(q as u8);
            }
        }
        output.push(remainder as u8);
        remaining = if quotient.is_empty() { vec![0] } else { quotient };
    }

    if output.is_empty() {
        output.push(0);
    }
    output.reverse();
    output
}

fn add_decimal(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u8;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry > 0 {
        let a = if i > 0 {
            i -= 1;
            left[i]
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            right[j]
        } else {
            0
        };
        let sum = a + b + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }

    if result.is_empty() {
        result.push(0);
    }
    result.reverse();
    result
}

fn render(digits: Vec<u8>, style: BaseStyle) -> String {
    let zero = if style.upper { b'0' } else { b'0' };
    let mut chars: Vec<u8> = digits
        .into_iter()
        .map(|digit| {
            if digit < 10 {
                zero + digit
            } else if style.upper {
                b'A' + digit - 10
            } else {
                b'a' + digit - 10
            }
        })
        .collect();

    while chars.len() < style.min_width {
        chars.insert(0, b'0');
    }

    if style.group == 0 || chars.len() <= style.group {
        return String::from_utf8(chars).unwrap();
    }

    let mut output = Vec::with_capacity(chars.len() + chars.len() / style.group);
    for (i, &byte) in chars.iter().enumerate() {
        if i > 0 && (chars.len() - i) % style.group == 0 {
            output.push(b'_');
        }
        output.push(byte);
    }
    String::from_utf8(output).unwrap()
}

pub fn convert_base(

    from: u8,
    to: u8,
    style: BaseStyle,

    style: BaseStyle,

    let right_decimal = to_decimal(&right_digits, base);
    let sum = add_decimal(&left_decimal, &right_decimal);
    Ok(render(from_decimal(&sum, base), style))
}

        }

        if byte == b'_' {
            continue;

            Some(value) if value < base => digits.push(value),
            _ => {

                    operand,

        }

            operand,
            index: payload_start,

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for digit in value.iter_mut() {
        let product = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (product % 256) as u8;
        carry = product / 256;
    }
    while carry != 0 {
        value.push((carry % 256) as u8);
        carry /= 256;
    }
}

fn convert_digits(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut number = vec![0u8]; // Little-endian bytes in base 256.
    for &digit in digits {
        multiply_add(&mut number, from, digit);
    }

    while number.len() > 1 && number.last() == Some(&0) {
        number.pop();
    }

    let mut output = Vec::new();
    if number.len() == 1 && number[0] == 0 {
        output.push(0);
        return output;
    }

    while number.iter().any(|&byte| byte != 0) {
        let mut remainder = 0u16;
        for byte in number.iter_mut().rev() {
            let current = remainder * 256 + *byte as u16;
            *byte = (current / to as u16) as u8;
            remainder = current % to as u16;
        }
        output.push(remainder as u8);
        while number.len() > 1 && number.last() == Some(&0) {
            number.pop();
        }
    }
    output.reverse();
    output
}

fn format_digits(digits: &[u8], style: BaseStyle) -> String {
    let padding = style.min_width.saturating_sub(digits.len());
    let total = padding + digits.len();
    let mut result = String::with_capacity(total + total / style.group.max(1));

    for index in 0..total {
        if style.group != 0 && index != 0 && (total - index) % style.group == 0 {
            result.push('_');
        }
        let digit = if index < padding {
            0
        } else {
            digits[index - padding]
        };
        let byte = match digit {
            0..=9 => b'0' + digit,
            10..=35 if style.upper => b'A' + (digit - 10),
            10..=35 => b'a' + (digit - 10),
            _ => unreachable!(),
        };
        result.push(byte as char);
    }

    result
}

    right: &str,
