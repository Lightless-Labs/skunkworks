#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if (2..=36).contains(&value) {
        Ok(())
    } else {
        Err(BaseError::BadBase { which, value })
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0usize;

    if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if bytes[..hash] != *expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let left_is_digit = offset > 0 && payload[offset - 1] != b'_';
            let right_is_digit = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !left_is_digit || !right_is_digit {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u32;
    for limb in value.iter_mut() {
        let current = (*limb as u32) * (radix as u32) + carry;
        *limb = (current % 1_000_000_000) as u32;
        carry = current / 1_000_000_000;
    }
    while carry != 0 {
        value.push((carry % 1_000_000_000) as u32);
        carry /= 1_000_000_000;
    }
}

fn parse_number(digits: &[u8], radix: u8) -> Vec<u32> {
    let mut value = vec![0u32];
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    value
}

fn add_numbers(left: &[u32], right: &[u32]) -> Vec<u32> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for index in 0..left.len().max(right.len()) {
        let sum = left.get(index).copied().unwrap_or(0) as u64
            + right.get(index).copied().unwrap_or(0) as u64
            + carry;
        result.push((sum % 1_000_000_000) as u32);
        carry = sum / 1_000_000_000;
    }
    if carry != 0 {
        result.push(carry as u32);
    }
    result
}

fn divide_number(value: &mut Vec<u32>, divisor: u32) -> u32 {
    let mut remainder = 0u64;
    for limb in value.iter_mut().rev() {
        let current = remainder * 1_000_000_000 + *limb as u64;
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    remainder as u32
}

fn render_number(mut value: Vec<u32>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if value.iter().all(|&limb| limb == 0) {
        digits.push(0u8);
    } else {
        while value.iter().any(|&limb| limb != 0) {
            digits.push(divide_number(&mut value, radix as u32) as u8);
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let mut padded = vec![0u8; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::new();
    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };
    for (index, digit) in digits.iter().enumerate() {
        if style.group != 0
            && index > 0
            && (digits.len() - index) % style.group == 0
        {
            output.push('_');
        }
        output.push(alphabet[*digit as usize] as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let value = parse_number(&digits, from);
    Ok(render_number(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = parse_number(&left_digits, base);
    let right_value = parse_number(&right_digits, base);
    Ok(render_number(add_numbers(&left_value, &right_value), base, style))
}
