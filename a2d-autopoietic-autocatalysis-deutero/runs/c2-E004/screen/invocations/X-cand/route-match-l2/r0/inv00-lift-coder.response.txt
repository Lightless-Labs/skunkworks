#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let path = path.split(['?', '#']).next().unwrap_or(path);
        let request_segments = parse_path(path)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &request_segments) {
                let replace = match &best {
                    None => true,
                    Some((best_index, _)) => {
                        let best_route = &self.routes[*best_index];
                        route.rank > best_route.rank
                            || (route.rank == best_route.rank && route.id < best_route.id)
                    }
                };
                if replace {
                    best = Some((route.id, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut ranges = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(Error::InvalidPattern { at: start });
                }
                ranges.push((start, i));
                let segments = parse_pattern_segments(pattern, &ranges)?;
                return Err(Error::InvalidPattern { at: i }).or(Ok(segments));
            }
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(Error::InvalidPattern { at: start });
    }
    ranges.push((start, bytes.len()));
    parse_pattern_segments(pattern, &ranges)
}

fn parse_pattern_segments(
    pattern: &str,
    ranges: &[(usize, usize)],
) -> Result<Vec<Segment>, Error> {
    let bytes = pattern.as_bytes();
    let mut segments = Vec::with_capacity(ranges.len());
    let mut names = std::collections::HashSet::new();

    for (index, &(start, end)) in ranges.iter().enumerate() {
        let first = bytes[start];
        if first == b':' || first == b'*' {
            let name_bytes = &bytes[start + 1..end];
            if !valid_name(name_bytes) {
                return Err(Error::InvalidPattern { at: start });
            }
            let name = std::str::from_utf8(name_bytes)
                .expect("capture names are ASCII")
                .to_owned();
            if !names.insert(name.clone()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' && index + 1 != ranges.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if first == b':' {
                Segment::Param(name)
            } else {
                Segment::Wildcard(name)
            });
        } else {
            segments.push(Segment::Static(decode_segment(bytes, start, end, true)?));
        }
    }
    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            segments.push(decode_segment(bytes, start, i, false)?);
            if i == bytes.len() - 1 {
                return Err(Error::InvalidPath { at: i });
            }
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPath { at: start });
    }
    segments.push(decode_segment(bytes, start, bytes.len(), false)?);
    Ok(segments)
}

fn decode_segment(
    bytes: &[u8],
    start: usize,
    end: usize,
    is_pattern: bool,
) -> Result<String, Error> {
    let mut decoded = Vec::new();
    let mut i = start;
    while i < end {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= end {
                return Err(invalid_error(is_pattern, i));
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return Err(invalid_error(is_pattern, i));
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return Err(invalid_error(is_pattern, i));
            };
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid_error(is_pattern, i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid_error(is_pattern, i));
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn invalid_error(is_pattern: bool, at: usize) -> Error {
    if is_pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn valid_name(name: &[u8]) -> bool {
    if name.is_empty() || !name[0].is_ascii_lowercase() {
        return false;
    }
    name[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn match_route(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut request_index = 0;

    for segment in pattern {
        match segment {
            Segment::Wildcard(name) => {
                let value = request[request_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
            Segment::Static(expected) => {
                if request.get(request_index)? != expected {
                    return None;
                }
                request_index += 1;
            }
            Segment::Param(name) => {
                let value = request.get(request_index)?;
                params.push((name.clone(), value.clone()));
                request_index += 1;
            }
        }
    }

    if request_index == request.len() {
        Some(params)
    } else {
        None
    }
}
