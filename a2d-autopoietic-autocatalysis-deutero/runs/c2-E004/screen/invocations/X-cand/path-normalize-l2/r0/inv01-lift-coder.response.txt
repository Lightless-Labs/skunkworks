#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.match_indices('/')
        .fold(Vec::new(), |mut parts, (slash_at, _)| {
            let start = slash_at + 1;
            let rest = &path[start..];
            let end = rest.find('/').map_or(path.len(), |i| start + i);
            if start < path.len() && start <= end && !path[start..end].is_empty() {
                parts.push((start, &path[start..end]));
            }
            parts
        })
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/').find(|part| !part.is_empty()).is_some_and(|part| part == "." || part == "..")
}

fn reduce(
    mut stack: Vec<String>,
    path: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<Vec<String>, PathError> {
    for (at, part) in components(path) {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: offset_base + at,
                    });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(part.to_string()),
        }
    }
    Ok(stack)
}

fn render(stack: Vec<String>, absolute: bool, dir_intent: bool) -> String {
    let joined = stack.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };

    if dir_intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let stack = reduce(Vec::new(), path, 0, absolute)?;
    Ok(render(stack, absolute, directory_intent(path)))
}

fn normalized_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    validate_raw(path).map_err(|_| ())?;
    let stack = reduce(Vec::new(), path, 0, true).map_err(|_| ())?;
    Ok(stack)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_stack = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(home) => Some(normalized_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let tilde = path.as_bytes()[0] == b'~';
    let (mut stack, remainder, offset_base, absolute) = if tilde {
        if path.len() > 1 && !path[1..].starts_with('/') {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_stack = home_stack.ok_or(PathError::MissingHome)?;
        let consumed = if path.len() > 1 { 2 } else { 1 };
        (home_stack, &path[consumed..], consumed, true)
    } else if path.as_bytes()[0] == b'/' {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_stack, path, 0, true)
    };

    stack = reduce(stack, remainder, offset_base, absolute)?;
    Ok(render(stack, true, directory_intent(path)))
}
