#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn reduce<'a>(
    parts: impl IntoIterator<Item = (&'a str, usize)>,
    stack: &mut Vec<(String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (part, offset) in parts {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|(name, _)| name != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset });
                } else {
                    stack.push(("..".to_string(), offset));
                }
            }
            _ => stack.push((part.to_string(), offset)),
        }
    }
    Ok(())
}

fn render(stack: &[(String, usize)], absolute: bool, intent: bool) -> String {
    let joined = stack
        .iter()
        .map(|(part, _)| part.as_str())
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let mut stack = Vec::new();
    let mut search_from = 0usize;

    for part in path.split('/') {
        let start = search_from;
        search_from = search_from.saturating_add(part.len()).saturating_add(1);
        if part.is_empty() {
            continue;
        }
        reduce(
            std::iter::once((part, start)),
            &mut stack,
            absolute,
        )?;
    }

    Ok(render(&stack, absolute, directory_intent(path)))
}

fn validate_context(path: &str) -> Result<Vec<(String, usize)>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    validate_raw(path).map_err(|_| PathError::InvalidCwd)?;

    let mut stack = Vec::new();
    let mut search_from = 0usize;
    for part in path.split('/') {
        let start = search_from;
        search_from = search_from.saturating_add(part.len()).saturating_add(1);
        if part.is_empty() {
            continue;
        }
        reduce(std::iter::once((part, start)), &mut stack, true)
            .map_err(|_| PathError::InvalidCwd)?;
    }
    Ok(stack)
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_stack = validate_context(cwd)?;

    let home_stack = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    if tilde && bytes.len() > 1 && bytes[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    let intent = directory_intent(path);
    let (absolute, mut stack, remainder_start) = if tilde {
        let stack = home_stack.ok_or(PathError::MissingHome)?;
        (true, stack, if bytes.len() > 1 { 2 } else { 1 })
    } else if bytes.first() == Some(&b'/') {
        (true, Vec::new(), 0)
    } else {
        (true, cwd_stack, 0)
    };

    let remainder = &path[remainder_start..];
    let mut search_from = remainder_start;
    for part in remainder.split('/') {
        let start = search_from;
        search_from = search_from.saturating_add(part.len()).saturating_add(1);
        if part.is_empty() {
            continue;
        }
        reduce(
            std::iter::once((part, start)),
            &mut stack,
            absolute,
        )?;
    }

    Ok(render(&stack, true, intent))
}
