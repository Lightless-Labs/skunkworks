pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components(
    components: &mut Vec<String>,
    path: &str,
    offset: usize,
    reject_escape: bool,
) -> Result<(), PathError> {
    let bytes = path.as_bytes();
    let mut i = offset;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        let component = &path[start..i];

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|last| last != "..") {
                    components.pop();
                } else if reject_escape {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push("..".to_owned());
                }
            }
            _ => components.push(component.to_owned()),
        }
    }

    Ok(())
}

fn render(components: &[String], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", components.join("/"))
    } else if components.is_empty() {
        ".".to_owned()
    } else {
        components.join("/")
    };

    if absolute && components.is_empty() {
        return "/".to_owned();
    }
    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn components(path: &str, offset: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_owned(), offset + start));
    }

    result
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    components(path, 0)
        .last()
        .is_some_and(|(component, _)| component == "." || component == "..")
}

    stack: &mut Vec<String>,
    parts: impl IntoIterator<Item = (String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {

            "." => {}

                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    reduce(&mut stack, components(path, 0), absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn validated_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {

        .map(|(component, _)| component)
        .collect())
}

    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn reduce<'a>(
    parts: impl IntoIterator<Item = (&'a str, usize)>,
    stack: &mut Vec<(String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (part, offset) in parts {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|(name, _)| name != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset });
                } else {
                    stack.push(("..".to_string(), offset));
                }
            }
            _ => stack.push((part.to_string(), offset)),
        }
    }
    Ok(())
}

fn render(stack: &[(String, usize)], absolute: bool, intent: bool) -> String {
    let joined = stack
        .iter()
        .map(|(part, _)| part.as_str())
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let mut stack = Vec::new();
    let mut search_from = 0usize;

    for part in path.split('/') {
        let start = search_from;
        search_from = search_from.saturating_add(part.len()).saturating_add(1);
        if part.is_empty() {
            continue;
        }
        reduce(
            std::iter::once((part, start)),
            &mut stack,
            absolute,
        )?;
    }

    Ok(render(&stack, absolute, directory_intent(path)))
}

fn validate_context(path: &str) -> Result<Vec<(String, usize)>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    validate_raw(path).map_err(|_| PathError::InvalidCwd)?;

    let mut stack = Vec::new();
    let mut search_from = 0usize;
    for part in path.split('/') {
        let start = search_from;
        search_from = search_from.saturating_add(part.len()).saturating_add(1);
        if part.is_empty() {
            continue;
        }
        reduce(std::iter::once((part, start)), &mut stack, true)
            .map_err(|_| PathError::InvalidCwd)?;
    }
    Ok(stack)
}

    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {

        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),

            continue;
        }
        reduce(
            std::iter::once((part, start)),
            &mut stack,
            absolute,
        )?;
    }

    Ok(render(&stack, true, intent))
}
