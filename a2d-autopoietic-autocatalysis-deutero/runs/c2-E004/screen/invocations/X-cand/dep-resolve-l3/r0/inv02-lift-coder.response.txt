#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn matches_constraint(version: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn satisfies(version: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches_constraint(version, *c))
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut names = std::collections::HashSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::HashSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::HashSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Active = std::collections::BTreeMap<String, Vec<Constraint>>;
type Assigned = std::collections::BTreeMap<String, Version>;

fn active_constraints(
    roots: &[Requirement],
    catalog_by_name: &std::collections::BTreeMap<&str, &Package>,
    assigned: &Assigned,
) -> Active {
    let mut active = Active::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assigned {
        if let Some(package) = catalog_by_name.get(name.as_str()) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for cs in active.values_mut() {
        *cs = canonical(std::mem::take(cs));
    }
    active
}

fn search(
    catalog_by_name: &std::collections::BTreeMap<&str, &Package>,
    catalog_names: &[String],
    roots: &[Requirement],
    locks: &std::collections::BTreeMap<String, Version>,
    assigned: &mut Assigned,
    attempts: &mut usize,
    max_attempts: usize,
    best: &mut Option<(usize, Vec<Option<Version>>, Assigned)>,
    first_error: &mut Option<ResolveError>,
) -> Result<(), ResolveError> {
    let active = active_constraints(roots, catalog_by_name, assigned);

    if let Some(name) = active
        .keys()
        .filter(|name| !catalog_by_name.contains_key(name.as_str()))
        .min()
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in assigned.iter() {
        if let Some(cs) = active.get(name) {
            if !satisfies(*version, cs) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: cs.clone(),
                });
            }
        }
    }

    for (name, cs) in &active {
        if assigned.contains_key(name) {
            continue;
        }
        let package = catalog_by_name[name.as_str()];
        if !package
            .releases
            .iter()
            .any(|release| satisfies(release.version, cs))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: cs.clone(),
            });
        }
    }

    let next = active.keys().find(|name| !assigned.contains_key(*name));
    let Some(name) = next else {
        let cost = locks
            .iter()
            .filter(|(name, version)| assigned.get(*name) != Some(*version))
            .count();
        let vector = catalog_names
            .iter()
            .map(|name| assigned.get(name).copied())
            .collect::<Vec<_>>();
        let better = match best {
            None => true,
            Some((best_cost, best_vector, _)) => {
                cost < *best_cost
                    || (cost == *best_cost
                        && vector
                            .iter()
                            .zip(best_vector)
                            .find_map(|(a, b)| (a != b).then_some(a > b))
                            .unwrap_or(false))
            }
        };
        if better {
            *best = Some((cost, vector, assigned.clone()));
        }
        return Ok(());
    };

    let package = catalog_by_name[name.as_str()];
    let cs = &active[name];
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .map(|r| r.version)
        .filter(|v| satisfies(*v, cs))
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(locked) = locks.get(name) {
        if let Some(pos) = candidates.iter().position(|v| v == locked) {
            candidates.remove(pos);
            candidates.insert(0, *locked);
        }
    }

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        assigned.insert(name.clone(), version);
        match search(
            catalog_by_name,
            catalog_names,
            roots,
            locks,
            assigned,
            attempts,
            max_attempts,
            best,
            first_error,
        ) {
            Ok(()) => {}
            Err(ResolveError::Limit { attempts }) => {
                assigned.remove(name);
                return Err(ResolveError::Limit { attempts });
            }
            Err(error) => {
                assigned.remove(name);
                if first_error.is_none() {
                    *first_error = Some(error);
                }
            }
        }
    }
    Ok(())
}

fn final_active(roots: &[Requirement], selected: &Assigned, catalog: &[Package]) -> Active {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    active_constraints(roots, &by_name, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let catalog_by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let lock_map: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();

    let mut attempts = 0;
    let mut assigned = Assigned::new();
    let mut best = None;
    let mut first_error = None;

    match search(
        &catalog_by_name,
        &catalog_names,
        roots,
        &lock_map,
        &mut assigned,
        &mut attempts,
        max_attempts,
        &mut best,
        &mut first_error,
    ) {
        Err(ResolveError::Limit { attempts }) => return Err(ResolveError::Limit { attempts }),
        Err(error) => {
            if first_error.is_none() {
                first_error = Some(error);
            }
        }
        Ok(()) => {}
    }

    let Some((_, _, selected_map)) = best else {
        return Err(first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_constraints = final_active(roots, &selected_map, catalog);
    let lock_map_by_name: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect();
    let mut all_names = std::collections::BTreeSet::new();
    all_names.extend(lock_map_by_name.keys().map(|s| (*s).to_string()));
    all_names.extend(selected_map.keys().cloned());

    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map_by_name.get(package.as_str()).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = catalog_by_name
                    .get(package.as_str())
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == locked));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(&package)
                    .is_some_and(|cs| !satisfies(locked, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
