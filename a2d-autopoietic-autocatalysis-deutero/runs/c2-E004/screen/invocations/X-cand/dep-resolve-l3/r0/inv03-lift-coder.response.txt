#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateLock { package: String },
    Missing { package: String },
    NoMatch { package: String, requirements: Vec<Constraint> },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit { attempts: usize },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes.iter().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn check_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_owned(),
        })
    }
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(char::from_digit((b >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((b & 15) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|c| matches_constraint(version, *c))
}

struct Search<'a> {
    catalog: &'a [Package],
    catalog_by_name: Vec<usize>,
    locks: &'a [Lock],
    attempts: usize,
    max_attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<Vec<Selection>>,
    best_cost: usize,
}

impl<'a> Search<'a> {
    fn package_index(&self, name: &str) -> Option<usize> {
        self.catalog_by_name
            .binary_search_by(|i| self.catalog[*i].name.as_str().cmp(name))
            .ok()
            .map(|pos| self.catalog_by_name[pos])
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn solution_cost(&self, selected: &[Selection]) -> usize {
        self.locks
            .iter()
            .filter(|lock| {
                !selected
                    .iter()
                    .any(|s| s.package == lock.package && s.version == lock.version)
            })
            .count()
    }

    fn is_better(&self, candidate: &[Selection], cost: usize) -> bool {
        let Some(best) = &self.best else {
            return true;
        };
        if cost != self.best_cost {
            return cost < self.best_cost;
        }
        for &idx in &self.catalog_by_name {
            let name = &self.catalog[idx].name;
            let a = candidate.iter().find(|s| &s.package == name).map(|s| s.version);
            let b = best.iter().find(|s| &s.package == name).map(|s| s.version);
            match (a, b) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(av), Some(bv)) if av != bv => return av > bv,
                _ => {}
            }
        }
        false
    }

    fn inspect(
        &mut self,
        roots: &[Requirement],
        assigned: &mut Vec<Selection>,
    ) -> Result<(), ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();

        for root in roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for selection in assigned.iter() {
            if let Some(pkg_idx) = self.package_index(&selection.package) {
                if let Some(release) = self.catalog[pkg_idx]
                    .releases
                    .iter()
                    .find(|r| r.version == selection.version)
                {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for requirements in active.values_mut() {
            *requirements = canonical(std::mem::take(requirements));
        }

        if let Some(name) = active
            .keys()
            .filter(|name| self.package_index(name).is_none())
            .next()
            .cloned()
        {
            let error = ResolveError::Missing { package: name };
            self.record_error(error.clone());
            return Ok(());
        }

        let mut assigned_sorted = assigned.clone();
        assigned_sorted.sort_by(|a, b| a.package.cmp(&b.package));
        for selection in &assigned_sorted {
            if let Some(requirements) = active.get(&selection.package) {
                if !matches_all(selection.version, requirements) {
                    let error = ResolveError::Conflict {
                        package: selection.package.clone(),
                        selected: selection.version,
                        requirements: requirements.clone(),
                    };
                    self.record_error(error);
                    return Ok(());
                }
            }
        }

        let selected_names: std::collections::BTreeSet<String> =
            assigned.iter().map(|s| s.package.clone()).collect();

        for (name, requirements) in &active {
            if selected_names.contains(name) {
                continue;
            }
            let idx = self.package_index(name).unwrap();
            let has_match = self.catalog[idx]
                .releases
                .iter()
                .any(|r| matches_all(r.version, requirements));
            if !has_match {
                let error = ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: requirements.clone(),
                };
                self.record_error(error);
                return Ok(());
            }
        }

        let next = active
            .keys()
            .find(|name| !selected_names.contains(*name))
            .cloned();

        let Some(name) = next else {
            let mut solution = assigned.clone();
            solution.sort_by(|a, b| a.package.cmp(&b.package));
            let cost = self.solution_cost(&solution);
            if self.is_better(&solution, cost) {
                self.best_cost = cost;
                self.best = Some(solution);
            }
            return Ok(());
        };

        let idx = self.package_index(&name).unwrap();
        let requirements = active.get(&name).unwrap();
        let mut candidates: Vec<Version> = self.catalog[idx]
            .releases
            .iter()
            .filter(|r| matches_all(r.version, requirements))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let locked_version = candidates.remove(pos);
                candidates.insert(0, locked_version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.push(Selection {
                package: name.clone(),
                version,
            });
            let result = self.inspect(roots, assigned);
            assigned.pop();
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        check_name(format!("root[{i}]"), &root.package)?;
    }
    for (i, package) in catalog.iter().enumerate() {
        check_name(format!("catalog[{i}]"), &package.name)?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                check_name(
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                    &dep.package,
                )?;
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        check_name(format!("lock[{i}]"), &lock.package)?;
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut catalog_by_name: Vec<usize> = (0..catalog.len()).collect();
    catalog_by_name.sort_by(|a, b| catalog[*a].name.cmp(&catalog[*b].name));

    let mut search = Search {
        catalog,
        catalog_by_name,
        locks,
        attempts: 0,
        max_attempts,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
    };
    search.inspect(roots, &mut Vec::new())?;
    let selected = match search.best {
        Some(selected) => selected,
        None => return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    };

    let mut names = std::collections::BTreeSet::new();
    for lock in locks {
        names.insert(lock.package.clone());
    }
    for selection in &selected {
        names.insert(selection.package.clone());
    }

    let final_constraints = {
        let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
        for root in roots {
            active.entry(root.package.clone()).or_default().push(root.constraint);
        }
        for selection in &selected {
            if let Some(idx) = search.package_index(&selection.package) {
                if let Some(release) = catalog[idx]
                    .releases
                    .iter()
                    .find(|r| r.version == selection.version)
                {
                    for dep in &release.deps {
                        active.entry(dep.package.clone()).or_default().push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }
        active
    };

    let mut changes = Vec::new();
    for name in names {
        let before = locks.iter().find(|l| l.package == name).map(|l| l.version);
        let after = selected.iter().find(|s| s.package == name).map(|s| s.version);
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .package_index(&name)
                    .map(|idx| catalog[idx].releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(&name)
                    .is_some_and(|cs| !matches_all(old, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));

    Ok(Resolution { selected, changes })
}
