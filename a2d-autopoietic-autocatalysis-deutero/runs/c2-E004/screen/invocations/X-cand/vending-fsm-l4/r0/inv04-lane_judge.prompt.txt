pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,

    refunded: u64,

const DENOMS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let index = match DENOMS.iter().position(|d| *d == coin) {
            Some(index) => index,
            None => return Err(Error::BadCoin { coin }),
        };
        let new_credit = u32::from(self.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        let new_credit = new_credit as u16;
        self.escrow.push(coin);
        self.credit = new_credit;
        self.inserted = self.inserted.saturating_add(u64::from(coin));
        let _ = index;
        self.push_event(EventKind::Inserted {
            coin,
            credit: new_credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        self.push_event(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = match self.selected.as_deref() {
            Some(id) => id.to_owned(),
            None => return Err(Error::NoSelection),
        };
        let product_index = self
            .products
            .iter()
            .position(|product| product.id == id)
            .expect("selected product is always in the catalog");

        if self.products[product_index].stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = self.products[product_index].price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let chosen = Self::choose_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for index in 0..4 {
            proposed[index] -= chosen[index];
        }
        for coin in &self.escrow {
            let index = DENOMS
                .iter()
                .position(|denom| denom == coin)
                .expect("escrow contains only accepted denominations");
            proposed[index] += 1;
        }
        for index in 0..4 {
            if proposed[index] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMS[index],
                });
            }
        }

        let mut change_coins = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..chosen[index] {
                change_coins.push(DENOMS[index]);
            }
        }

        self.tubes = proposed;
        self.products[product_index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        self.change = self.change.saturating_add(u64::from(amount));
        self.push_event(EventKind::Vended {
            id: id.clone(),
            price,
            change: change_coins.clone(),
        });

        Ok(Receipt {
            id,
            price,
            change: change_coins,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let refunded = u64::from(self.credit);
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        self.refunded = self.refunded.saturating_add(refunded);
        self.push_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {

        result.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = match self.products.iter().position(|product| product.id == id) {
            Some(index) => index,
            None => return Err(Error::UnknownProduct { id: id.to_owned() }),
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.push_event(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = match self.products.iter().position(|product| product.id == id) {
            Some(index) => index,
            None => return Err(Error::UnknownProduct { id: id.to_owned() }),
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let total = u32::from(self.products[index].stock) + u32::from(count);
        if total > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = total as u16;
        self.push_event(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: total as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }

        self.maintenance = enabled;
        self.selected = None;
        self.push_event(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let tubes_value = self.tube_value(self.tubes);
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tubes_value.saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(
        &mut self,
        expected: [u16; 4],
        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for index in 0..4 {
            if expected[index] != self.tubes[index] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[index],
                    expected: expected[index],
                    found: self.tubes[index],
                });
            }
        }
        for index in 0..4 {
            if actual[index] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMS[index],
                });
            }
        }

        let before_value = self.tube_value(self.tubes);
        let after_value = self.tube_value(actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift {
                before: before_value,
                after: after_value,
            });
        }

        let delta = i128::from(after_value) - i128::from(before_value);
        self.adjustment = (i128::from(self.adjustment) + delta)
            .clamp(i128::from(i64::MIN), i128::from(i64::MAX))
            as i64;
        let before = self.tubes;
        self.tubes = actual;
        self.push_event(EventKind::Reconciled {
            before,
            after: actual,
        });
        Ok(())
    }
}

impl Machine {
    fn push_event(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }

    fn tube_value(&self, tubes: [u16; 4]) -> u64 {
        tubes
            .iter()
            .zip(DENOMS)
            .map(|(count, denom)| u64::from(*count) * u64::from(denom))
            .sum()
    }

    fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
        if amount == 0 {
            return Some([0; 4]);
        }

        let mut greedy = [0u16; 4];
        let mut remainder = amount;
        for index in (0..4).rev() {
            let denomination = DENOMS[index];
            let take = tubes[index].min(remainder / denomination);
            greedy[index] = take;
            remainder -= take * denomination;
        }
        if remainder == 0 {
            return Some(greedy);
        }

        let target = usize::from(amount);
        let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
        best[0] = Some([0; 4]);

        for index in 0..4 {
            let denomination = usize::from(DENOMS[index]);
            for _ in 0..tubes[index] {
                for value in (denomination..=target).rev() {
                    if let Some(previous) = best[value - denomination] {
                        let mut candidate = previous;
                        candidate[index] += 1;
                        let replace = match best[value] {
                            None => true,
                            Some(current) => {
                                let candidate_count: u32 =
                                    candidate.iter().map(|x| u32::from(*x)).sum();
                                let current_count: u32 =
                                    current.iter().map(|x| u32::from(*x)).sum();
                                if candidate_count < current_count {
                                    true
                                } else if candidate_count > current_count {
                                    false
                                } else {
                                    Self::descending_lex_greater(candidate, current)
                                }
                            }
                        };
                        if replace {
                            best[value] = Some(candidate);
                        }
                    }
                }
            }
        }
        best[target]
    }

    fn descending_lex_greater(left: [u16; 4], right: [u16; 4]) -> bool {
        for index in (0..4).rev() {
            if left[index] != right[index] {
                return left[index] > right[index];
            }
        }
        false
    }
}

            coin,
            credit: self.credit,

        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }

            return Err(Error::NoSelection);

            new: price,

            return Err(Error::MaintenanceRequired);

        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });

        if count == 0 {
            return Err(Error::ZeroRestock);

        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let tubes_value = self.tubes.iter().zip(DENOMS).fold(0u64, |sum, (&count, coin)| {
            sum.saturating_add(u64::from(count) * u64::from(coin))
        });
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tubes_value.saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(
        &mut self,
        expected: [u16; 4],
        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        let before_value = tube_value(self.tubes);
        let after_value = tube_value(actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift {
                before: before_value,
                after: after_value,
            });
        }

        let difference = i128::from(after_value) - i128::from(before_value);
        let delta = difference as i64;
        self.adjustment = self.adjustment.saturating_add(delta);
        let before = self.tubes;
        self.tubes = actual;
        self.log(EventKind::Reconciled {
            before,
            after: actual,
        });
        Ok(())
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len())
        && id
            .as_bytes()
            .iter()
            .all(|byte| byte.is_ascii_lowercase())
}

fn denom_index(coin: u16) -> Option<usize> {
    DENOMS.iter().position(|&denom| denom == coin)
}

fn tube_value(tubes: [u16; 4]) -> u64 {
    tubes.iter().zip(DENOMS).fold(0u64, |sum, (&count, coin)| {
        sum + u64::from(count) * u64::from(coin)
    })
}

fn make_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / DENOMS[i]);
        greedy_counts[i] = take;
        remaining -= take * DENOMS[i];
    }
    if remaining == 0 {
        return Some(counts_to_descending(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;

    fn search(
        i: isize,
        remaining: u16,
        tubes: &[u16; 4],
        counts: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_count: &mut u32,
    ) {
        if i < 0 {
            if remaining == 0 {
                let total: u32 = counts.iter().map(|&n| u32::from(n)).sum();
                if total < *best_count
                    || (total == *best_count
                        && best
                            .as_ref()
                            .map_or(true, |current| descending_lex_greater(counts, current)))
                {
                    *best_count = total;
                    *best = Some(*counts);
                }
            }
            return;
        }

        let index = i as usize;
        let coin = DENOMS[index];
        let max_take = tubes[index].min(remaining / coin);
        for take in 0..=max_take {
            let next_count: u32 = counts
                .iter()
                .map(|&n| u32::from(n))
                .sum::<u32>()
                .saturating_add(u32::from(take));
            if next_count > *best_count {
                continue;
            }
            counts[index] = take;
            search(
                i - 1,
                remaining - take * coin,
                tubes,
                counts,
                best,
                best_count,
            );
        }
        counts[index] = 0;
    }

    search(
        3,
        amount,
        tubes,
        &mut [0; 4],
        &mut best,
        &mut best_count,
    );
    best.map(counts_to_descending)
}

fn descending_lex_greater(a: &[u16; 4], b: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if a[i] != b[i] {
            return a[i] > b[i];
        }
    }
    false
}

fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    result
}

        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        let change_value = change
            .iter()
            .fold(0u64, |sum, &coin| sum.saturating_add(u64::from(coin)));
        self.change = self.change.saturating_add(change_value);
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.refunded = self.refunded.saturating_add(u64::from(self.credit));
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&index) = self.product_indices.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&index) = self.product_indices.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = u32::from(self.products[index].stock) + u32::from(count);
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = new_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let tube_value = self.tubes.iter().enumerate().fold(0u64, |sum, (i, count)| {
            sum.saturating_add(u64::from(*count) * u64::from(DENOMINATIONS[i]))
        });
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value.saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMINATIONS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let before_value = tube_value(self.tubes);
        let after_value = tube_value(actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift {
                before: before_value,
                after: after_value,
            });
        }

        let delta = i128::from(after_value) - i128::from(before_value);
        self.adjustment = saturating_add_i64_i128(self.adjustment, delta);
        let before = self.tubes;
        self.tubes = actual;
        self.log(EventKind::Reconciled {
            before,
            after: actual,
        });
        Ok(())
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&d| d == coin)
}

fn tube_value(tubes: [u16; 4]) -> u64 {
    tubes.iter().enumerate().fold(0u64, |sum, (i, count)| {
        sum + u64::from(*count) * u64::from(DENOMINATIONS[i])
    })
}

fn saturating_add_i64_i128(value: i64, delta: i128) -> i64 {
    let sum = i128::from(value) + delta;
    if sum > i128::from(i64::MAX) {
        i64::MAX
    } else if sum < i128::from(i64::MIN) {
        i64::MIN
    } else {
        sum as i64
    }
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];
    for i in (0..4).rev() {
        let coin = DENOMINATIONS[i];
        let take = tubes[i].min(remaining / coin);
        greedy_counts[i] = take;
        remaining -= take * coin;
    }
    if remaining == 0 {
        return Some(counts_to_descending(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut counts = [0u16; 4];
    search_combinations(3, amount, &tubes, &mut counts, &mut best);
    best.map(counts_to_descending)
}

fn search_combinations(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if remaining == 0 {
        let replace = match best {
            None => true,
            Some(current) => {
                let new_total: u32 = counts.iter().map(|&x| u32::from(x)).sum();
                let old_total: u32 = current.iter().map(|&x| u32::from(x)).sum();
                if new_total < old_total {
                    true
                } else if new_total > old_total {
                    false
                } else {
                    descending_lex_greater(counts, current)
                }
            }
        };
        if replace {
            *best = Some(*counts);
        }
        return;
    }
    if index < 0 {
        return;
    }
    let i = index as usize;
    let coin = DENOMINATIONS[i];
    let max_take = tubes[i].min(remaining / coin);

    for take in (0..=max_take).rev() {
        counts[i] = take;
        search_combinations(
            index - 1,
            remaining - take * coin,
            tubes,
            counts,
            best,
        );
    }
    counts[i] = 0;
}

fn descending_lex_greater(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if left[i] != right[i] {
            return left[i] > right[i];
        }
    }
    false
}

fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for i in (0..4).rev() {
        coins.extend(std::iter::repeat_n(DENOMINATIONS[i], usize::from(counts[i])));
    }
    coins
}

    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = std::collections::HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),

                    id: product.id.clone(),

            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,

            change: 0,
            adjustment: 0,
        })

            credit: self.credit,

            .product_index(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }

        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self
            .product_index(&id)
            .expect("selected product must exist");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change_counts = choose_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

            proposed[i] -= change_counts[i];

                    coin: DENOMS[index],
                });

        for coin in &coins {

            id: id.clone(),
            price,
            change: coins.clone(),

        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

            id,
            price,
            change: coins,

        .zip(DENOMS.iter())

fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        coins.extend(std::iter::repeat_n(DENOMS[index], counts[index] as usize));
    }
    coins

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for index in (0..4).rev() {
        let coin = DENOMS[index];
        let taken = tubes[index].min(remaining / coin);
        greedy[index] = taken;
        remaining -= taken * coin;
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_total = u32::MAX;

    fn search(
        index: isize,
        remaining: u16,
        tubes: [u16; 4],
        current: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_total: &mut u32,
    ) {
        if index < 0 {
            if remaining == 0 {
                let total: u32 = current.iter().map(|count| u32::from(*count)).sum();
                if total < *best_total
                    || (total == *best_total
                        && best.map_or(true, |previous| {
                            for i in (0..4).rev() {
                                if current[i] != previous[i] {
                                    return current[i] > previous[i];
                                }
                            }
                            false
                        }))
                {
                    *best = Some(*current);
                    *best_total = total;
                }
            }
            return;
        }

        let i = index as usize;
        let coin = DENOMS[i];
        let maximum = tubes[i].min(remaining / coin);
        for count in 0..=maximum {
            current[i] = count;
            search(
                index - 1,
                remaining - count * coin,
                tubes,
                current,
                best,
                best_total,
            );
        }
        current[i] = 0;
    }

    search(
        3,
        amount,
        tubes,
        &mut [0; 4],
        &mut best,
        &mut best_total,
    );
    best
}
