#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut found = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    found = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match found {
            Some(end) => Some(end),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, is_quoted) = if let Some(end) = quoted_end.filter(|_| bytes[0] == b'"') {
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            if bytes[j] == b'\\' {
                let next = bytes[j + 1];
                if !matches!(next, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: j });
                }
                decoded.push(next);
                origins.push(j);
                j += 2;
            } else {
                let b = bytes[j];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: j });
                }
                decoded.push(b);
                origins.push(j);
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for j in 0..sep {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: j });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: j });
            }
        }
        (input[..sep].to_owned(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let mut removed_dot = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_dot = Some(domain_end);
    }

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_count = 0usize;
    let mut last_label_start = domain_start;
    let mut last_label_end = domain_start;
    let mut label_start = domain_start;

    loop {
        let label_end = bytes[label_start..domain_end]
            .iter()
            .position(|&b| b == b'.')
            .map(|offset| label_start + offset)
            .unwrap_or(domain_end);

        if label_start == label_end {
            let at = if label_start == domain_end {
                removed_dot.unwrap_or(domain_end)
            } else {
                label_start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        let label_len = label_end - label_start;
        if label_len > 63 {
            return Err(EmailError::LabelTooLong {
                at: label_start + 63,
            });
        }
        if bytes[label_start] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_start });
        }
        if bytes[label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_end - 1 });
        }
        for j in label_start..label_end {
            let b = bytes[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: j });
            }
        }

        label_count += 1;
        last_label_start = label_start;
        last_label_end = label_end;

        if label_end == domain_end {
            break;
        }
        label_start = label_end + 1;
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    if bytes[last_label_start..last_label_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: last_label_start,
        });
    }

    let domain = input[domain_start..domain_end].to_ascii_lowercase();
    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}
