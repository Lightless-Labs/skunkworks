#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/');
    let mut components: Vec<(String, usize)> = Vec::new();
    let mut last_nonempty: Option<(&str, usize)> = None;

    let mut start = 0;
    for (index, byte) in path.bytes().enumerate() {
        if byte == b'/' {
            if start < index {
                let component = &path[start..index];
                last_nonempty = Some((component, start));
                process_component(component, start, absolute, &mut components)?;
            }
            start = index + 1;
        }
    }
    if start < path.len() {
        let component = &path[start..];
        last_nonempty = Some((component, start));
        process_component(component, start, absolute, &mut components)?;
    }

    let directory_intent = directory_intent
        || matches!(last_nonempty, Some((".", _) | ("..", _)));

    let joined = components
        .into_iter()
        .map(|(component, _)| component)
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

fn process_component(
    component: &str,
    at: usize,
    absolute: bool,
    components: &mut Vec<(String, usize)>,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if components
                .last()
                .is_some_and(|(previous, _)| previous != "..")
            {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                components.push(("..".to_string(), at));
            }
        }
        _ => components.push((component.to_string(), at)),
    }
    Ok(())
}
