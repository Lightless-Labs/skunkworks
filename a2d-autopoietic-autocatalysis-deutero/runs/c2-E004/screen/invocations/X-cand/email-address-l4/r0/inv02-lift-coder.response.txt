#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn clean_cfws(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_prefix = false;
    let mut quote_open = 0;

    while i < input.len() {
        let b = input[i];

        if quote_prefix {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    bytes.push(input[i + 1]);
                    positions.push(i + 1);
                    i += 2;
                } else {
                    i += 1;
                }
            } else {
                if b == b'"' {
                    quote_prefix = false;
                }
                i += 1;
            }
            continue;
        }

        if bytes.is_empty() && b == b'"' {
            quote_prefix = true;
            quote_open = i;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'(' || b == b'\r' || b == b'\n' {
            let run_start = i;
            let mut depth = 0usize;
            let mut outer_open = None;

            loop {
                if i >= input.len() {
                    break;
                }

                let c = input[i];

                if depth > 0 {
                    if c == b'\\' {
                        if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                            return Err(EmailError::CommentEscape { at: i });
                        }
                        i += 2;
                        continue;
                    }

                    if c == b'\r' || c == b'\n' {
                        if c == b'\r'
                            && i + 2 < input.len()
                            && input[i + 1] == b'\n'
                            && is_hws(input[i + 2])
                        {
                            i += 2;
                            while i < input.len() && is_hws(input[i]) {
                                i += 1;
                            }
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }

                    if c == b'(' {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        depth += 1;
                        i += 1;
                        continue;
                    }

                    if c == b')' {
                        depth -= 1;
                        i += 1;
                        if depth == 0 {
                            continue;
                        }
                        continue;
                    }

                    if !is_printable(c) && c != b'\t' {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                    continue;
                }

                if is_hws(c) {
                    i += 1;
                    continue;
                }

                if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && i + 2 < input.len()
                        && input[i + 1] == b'\n'
                        && is_hws(input[i + 2])
                    {
                        i += 2;
                        while i < input.len() && is_hws(input[i]) {
                            i += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if c == b'(' {
                    outer_open = Some(i);
                    depth = 1;
                    i += 1;
                    continue;
                }

                break;
            }

            if depth > 0 {
                return Err(EmailError::UnclosedComment {
                    at: outer_open.unwrap_or(run_start),
                });
            }

            runs.push((run_start, i));
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if quote_prefix {
        let _ = quote_open;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn mapped(pos: usize, positions: &[usize], input_len: usize) -> usize {
    positions.get(pos).copied().unwrap_or(input_len)
}

fn is_atom_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else if is_atom_byte(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    !previous_dot
}

fn parse_cleaned(
    input: &[u8],
    cleaned: &Cleaned,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let bytes = &cleaned.bytes;
    let positions = &cleaned.positions;

    for &(start, end) in &cleaned.runs {
        let left_at = start > 0 && bytes[start - 1] == b'@';
        let right_at = end < bytes.len() && bytes[end] == b'@';
        if start != 0 && end != bytes.len() && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary {
                at: start.min(input.len()),
            });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
                continue;
            }
            if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            }
            i += 1;
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: mapped(0, positions, input.len()),
            });
        }
    }

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' {
            if quote_end.map_or(true, |q| i > q) {
                separators.push(i);
            }
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: mapped(separators[1], positions, input.len()),
        });
    }

    let at = separators[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: mapped(at, positions, input.len()),
        });
    }

    let local_bytes = &bytes[..at];
    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let quoted;

    if local_bytes[0] == b'"' {
        quoted = true;
        let qend = quote_end.unwrap();
        if qend + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: mapped(qend + 1, positions, input.len()),
            });
        }

        let mut i = 1;
        while i < qend {
            let b = local_bytes[i];
            if b == b'\\' {
                let next = local_bytes.get(i + 1).copied();
                match next {
                    Some(b'"' | b'\\' | b'@') => {
                        local.push(next.unwrap());
                        local_origins.push(mapped(i, positions, input.len()));
                        i += 2;
                    }
                    _ => {
                        return Err(EmailError::BadEscape {
                            at: mapped(i, positions, input.len()),
                        });
                    }
                }
            } else if is_printable(b) {
                local.push(b);
                local_origins.push(mapped(i, positions, input.len()));
                i += 1;
            } else {
                return Err(EmailError::QuotedChar {
                    at: mapped(i, positions, input.len()),
                });
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: mapped(qend, positions, input.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: mapped(64, positions, input.len()),
            });
        }

        let mut previous_dot = false;
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: mapped(i, positions, input.len()),
                    });
                }
                previous_dot = true;
            } else {
                if !is_atom_byte(b) {
                    return Err(EmailError::LocalChar {
                        at: mapped(i, positions, input.len()),
                    });
                }
                previous_dot = false;
            }
            local.push(b);
            local_origins.push(mapped(i, positions, input.len()));
        }
    }

    let domain_start = at + 1;
    let raw_domain = &bytes[domain_start..];
    let mut domain_end = bytes.len();
    let removed_root = if raw_domain.last() == Some(&b'.') {
        domain_end -= 1;
        Some(mapped(domain_end, positions, input.len()))
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: mapped(domain_start, positions, input.len()),
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: mapped(domain_start + 253, positions, input.len()),
        });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut label_start = 0;

    for (i, &b) in domain_bytes.iter().enumerate() {
        if b == b'.' {
            if i == label_start {
                return Err(EmailError::EmptyLabel {
                    at: mapped(domain_start + i, positions, input.len()),
                });
            }
            label_starts.push(label_start);
            label_ends.push(i);
            label_start = i + 1;
        }
    }

    if label_start == domain_bytes.len() {
        return Err(EmailError::EmptyLabel {
            at: removed_root.unwrap_or_else(|| mapped(domain_end, positions, input.len())),
        });
    }
    label_starts.push(label_start);
    label_ends.push(domain_bytes.len());

    for (&start, &end) in label_starts.iter().zip(label_ends.iter()) {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: mapped(domain_start + start + 63, positions, input.len()),
            });
        }

        if domain_bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(domain_start + start, positions, input.len()),
            });
        }
        if domain_bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(domain_start + end - 1, positions, input.len()),
            });
        }

        for i in start..end {
            let b = domain_bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: mapped(domain_start + i, positions, input.len()),
                });
            }
        }
    }

    if label_starts.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(domain_start, positions, input.len()),
        });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if domain_bytes[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: mapped(domain_start + last_start, positions, input.len()),
        });
    }

    let mut lowered_domain = domain_bytes.to_vec();
    lowered_domain.make_ascii_lowercase();

    Ok((
        Address {
            local: String::from_utf8(local).unwrap(),
            domain: String::from_utf8(lowered_domain).unwrap(),
            quoted,
        },
        local_origins,
    ))
}

fn parse_internal(
    input: &str,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = clean_cfws(raw)?;
    parse_cleaned(raw, &cleaned, allow_single_label)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }

    let atom = valid_dot_atom(address.local.as_bytes());
    if !atom {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    result.duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }

    result
}
