#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct CfwsRun {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<CfwsRun>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn scan_fold(input: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= input.len() || input[i] != b'\r' || input[i + 1] != b'\n' || !is_hws(input[i + 2]) {
        return None;
    }
    let mut j = i + 3;
    while j < input.len() && is_hws(input[j]) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::new();
    let mut positions = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut protected_quote = false;
    let mut escaped_quote = false;

    while i < input.len() {
        let b = input[i];

        if protected_quote {
            bytes.push(b);
            positions.push(i);
            if escaped_quote {
                escaped_quote = false;
            } else if b == b'\\' {
                escaped_quote = true;
            } else if b == b'"' {
                protected_quote = false;
            }
            i += 1;
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = scan_fold(input, i) {
                let start = i;
                i = end;
                if let Some(last) = runs.last_mut() {
                    if last.end == start {
                        last.end = end;
                    } else {
                        runs.push(CfwsRun { start, end });
                    }
                } else {
                    runs.push(CfwsRun { start, end });
                }
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if is_hws(b) {
            let start = i;
            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }
            if let Some(last) = runs.last_mut() {
                if last.end == start {
                    last.end = i;
                    continue;
                }
            }
            runs.push(CfwsRun { start, end: i });
            continue;
        }

        if b == b'(' {
            let outer = i;
            let mut depth = 1usize;
            i += 1;
            while i < input.len() {
                let c = input[i];
                if c == b'\\' {
                    let at = i;
                    if i + 1 >= input.len() || !(0x20..=0x7e).contains(&input[i + 1]) {
                        return Err(EmailError::CommentEscape { at });
                    }
                    i += 2;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = scan_fold(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    depth += 1;
                    if depth > 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    i += 1;
                    continue;
                }
                if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 {
                        break;
                    }
                    continue;
                }
                if c == b'\t' || (0x20..=0x7e).contains(&c) {
                    i += 1;
                    continue;
                }
                return Err(EmailError::CommentChar { at: i });
            }
            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            if let Some(last) = runs.last_mut() {
                if last.end == outer {
                    last.end = i;
                    continue;
                }
            }
            runs.push(CfwsRun { start: outer, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if bytes.is_empty() && b == b'"' {
            protected_quote = true;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    for run in &runs {
        let left_at = bytes
            .iter()
            .position(|&b| b == b'@')
            .map(|p| positions[p]);
        let right_at = bytes
            .iter()
            .enumerate()
            .find(|(p, b)| **b == b'@' && positions[*p] >= run.end)
            .map(|(_, &p)| p);

        let legal = run.start == 0
            || run.end == input.len()
            || left_at.map_or(false, |p| p < run.start)
            || right_at.map_or(false, |p| p >= run.end);
        if !legal {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn mapped(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if b.is_ascii_alphanumeric()
            || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_cleaned(
    input_len: usize,
    clean: &Cleaned,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let b = &clean.bytes;
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let quote_end = if b[0] == b'"' {
        let mut i = 1usize;
        let mut found = None;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                found = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match found {
            Some(end) => Some(end),
            None => return Err(EmailError::UnclosedQuote { at: mapped(clean, 0, input_len) }),
        }
    } else {
        None
    };

    let mut ats = Vec::new();
    for i in 0..b.len() {
        if b[i] == b'@' && !quote_end.map_or(false, |end| i <= end) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: mapped(clean, ats[1], input_len) });
    }
    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal { at: mapped(clean, at, input_len) });
    }

    let local_bytes = &b[..at];
    let (local, quoted, origins) = if local_bytes[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != at {
            return Err(EmailError::TrailingLocal { at: mapped(clean, end + 1, input_len) });
        }
        let mut decoded = Vec::new();
        let mut origin = Vec::new();
        let mut i = 1usize;
        while i < end {
            let c = local_bytes[i];
            if c == b'\\' {
                if i + 1 >= end {
                    return Err(EmailError::BadEscape { at: mapped(clean, i, input_len) });
                }
                let n = local_bytes[i + 1];
                if !matches!(n, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: mapped(clean, i, input_len) });
                }
                decoded.push(n);
                origin.push(mapped(clean, i, input_len));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&c) {
                    return Err(EmailError::QuotedChar { at: mapped(clean, i, input_len) });
                }
                decoded.push(c);
                origin.push(mapped(clean, i, input_len));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: mapped(clean, end, input_len) });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origin[64] });
        }
        (
            String::from_utf8(decoded).unwrap(),
            true,
            origin,
        )
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: mapped(clean, 64, input_len) });
        }
        let mut prev_dot = false;
        for (i, &c) in local_bytes.iter().enumerate() {
            if c == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || prev_dot {
                    return Err(EmailError::LocalDot { at: mapped(clean, i, input_len) });
                }
                prev_dot = true;
            } else {
                if !(c.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&c)) {
                    return Err(EmailError::LocalChar { at: mapped(clean, i, input_len) });
                }
                prev_dot = false;
            }
        }
        (
            String::from_utf8(local_bytes.to_vec()).unwrap(),
            false,
            (0..local_bytes.len()).map(|i| mapped(clean, i, input_len)).collect(),
        )
    };

    let domain_start = at + 1;
    let mut domain_end = b.len();
    let root_position = if domain_end > domain_start && b[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: mapped(clean, domain_start, input_len),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: mapped(clean, domain_start + 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if b[i] == b'.' {
            if i == start {
                return Err(EmailError::EmptyLabel { at: mapped(clean, i, input_len) });
            }
            labels.push((start, i));
            start = i + 1;
        }
    }
    if start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: root_position.map(|p| mapped(clean, p, input_len))
                .unwrap_or_else(|| mapped(clean, start, input_len)),
        });
    }
    labels.push((start, domain_end));

    for &(ls, le) in &labels {
        let len = le - ls;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: mapped(clean, ls + 63, input_len),
            });
        }
        if b[ls] == b'-' {
            return Err(EmailError::LabelHyphen { at: mapped(clean, ls, input_len) });
        }
        if b[le - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: mapped(clean, le - 1, input_len) });
        }
        for i in ls..le {
            let c = b[i];
            if !(c.is_ascii_alphanumeric() || c == b'-') {
                return Err(EmailError::DomainChar { at: mapped(clean, i, input_len) });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(clean, domain_start, input_len),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if b[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: mapped(clean, last_start, input_len),
        });
    }

    let local = local;
    let domain = String::from_utf8(b[domain_start..domain_end].iter().map(u8::to_ascii_lowercase).collect()).unwrap();
    Ok((Address { local, domain, quoted }, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let clean = cleanup(raw)?;
    let (address, _) = parse_cleaned(input.len(), &clean, false)?;
    Ok(address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let clean = cleanup(raw)?;
    let (mut address, origins) = parse_cleaned(input.len(), &clean, policy.allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.get(plus).copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }

    if address.local.is_empty() {
        return Err(EmailError::EmptyNormalizedLocal { at: input.len() });
    }

    let safe = valid_dot_atom(address.local.as_bytes());
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                result.push('\\');
            }
            result.push(c);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: std::collections::HashMap<(String, String), usize> =
        std::collections::HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = keys.get(&key) {
                    duplicate_of[index] = Some(first);
                } else {
                    keys.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
