#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn is_ascii_letter(b: u8) -> bool {
    b.is_ascii_alphabetic()
}

fn is_valid_input_byte(b: u8) -> bool {
    (0x21..=0x7e).contains(&b)
}

fn validate_global_bytes(input: &str) -> Result<(), ParseError> {
    for (at, &b) in input.as_bytes().iter().enumerate() {
        if !is_valid_input_byte(b) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(err) => {
            let decoded_at = err.valid_up_to();
            let original_at = origins.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };

        if safe {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn map_codec_parse(err: CodecError, offset: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn validate_component(input: &str, offset: usize, component: Component) -> Result<(), ParseError> {
    decode_component(component, input)
        .map(|_| ())
        .map_err(|e| map_codec_parse(e, offset))
}

fn parse_port(text: &str, absolute_start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }
    if let Some((i, _)) = text
        .as_bytes()
        .iter()
        .enumerate()
        .find(|(_, b)| !b.is_ascii_digit())
    {
        return Err(ParseError::InvalidPort {
            at: absolute_start + i,
        });
    }
    let bytes = text.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }

    let mut value: u32 = 0;
    for &b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
        {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: absolute_start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    validate_global_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !is_ascii_letter(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = (authority_start..bytes.len())
        .find(|&i| matches!(bytes[i], b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_rel) = if let Some(&at_rel) = at_positions.first() {
        if at_rel == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at_rel,
            });
        }
        (Some(authority[..at_rel].to_owned()), at_rel + 1)
    } else {
        (None, 0)
    };
    let host_start = authority_start + host_start_rel;
    let host_port = &authority[host_start_rel..];

    let mut ipv6 = false;
    let (host, port, port_start_rel) = if host_port.starts_with('[') {
        let close = match host_port.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => {
                return Err(ParseError::InvalidIpv6 { at: host_start });
            }
        };
        let inside = &host_port[1..close];
        if inside.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix_start = close + 1;
        let suffix = &host_port[suffix_start..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(
                &suffix[1..],
                host_start + suffix_start + 1,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        };
        ipv6 = true;
        (inside.to_ascii_lowercase(), parsed_port, suffix_start)
    } else {
        let split = host_port.as_bytes().iter().position(|&b| b == b':');
        let (host_text, parsed_port, port_rel) = if let Some(split) = split {
            let host_text = &host_port[..split];
            let parsed_port = Some(parse_port(&host_port[split + 1..], host_start + split + 1)?);
            (host_text, parsed_port, split)
        } else {
            (host_port, None, host_port.len())
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, &b) in host_text.as_bytes().iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + i,
                });
            }
        }
        (host_text.to_ascii_lowercase(), parsed_port, port_rel)
    };

    let _ = port_start_rel;
    let component_start = authority_end;
    let before_fragment_end = (component_start..bytes.len())
        .find(|&i| bytes[i] == b'#')
        .unwrap_or(bytes.len());
    let fragment = if before_fragment_end < bytes.len() {
        Some(input[before_fragment_end + 1..].to_owned())
    } else {
        None
    };

    let before_query_end = (component_start..before_fragment_end)
        .find(|&i| bytes[i] == b'?')
        .unwrap_or(before_fragment_end);
    let query = if before_query_end < before_fragment_end {
        Some(input[before_query_end + 1..before_fragment_end].to_owned())
    } else {
        None
    };

    let path = if component_start == before_query_end {
        "/".to_owned()
    } else {
        input[component_start..before_query_end].to_owned()
    };

    if let Some(info) = &userinfo {
        validate_component(info, authority_start, Component::UserInfo)?;
    }
    if path != "/" || component_start != before_query_end {
        validate_component(&path, component_start, Component::Path)?;
    }
    if let Some(q) = &query {
        validate_component(q, before_query_end + 1, Component::Query)?;
    }
    if let Some(f) = &fragment {
        validate_component(f, before_fragment_end + 1, Component::Fragment)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut terminal_navigation = false;

    for (i, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 + segment_offset(path, i) },
            CodecError::InvalidUtf8 { at } => {
                PathError::InvalidUtf8 { at: at + 1 + segment_offset(path, i) }
            }
        })?;

        if decoded == "." {
            terminal_navigation = i + 1 == raw_segments.len();
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            terminal_navigation = i + 1 == raw_segments.len();
        } else {
            stack.push((*segment).to_owned());
            terminal_navigation = false;
        }
    }

    if terminal_navigation {
        stack.push(String::new());
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn segment_offset(path: &str, index: usize) -> usize {
    let bytes = path.as_bytes();
    let mut seen = 0;
    for i in 1..bytes.len() {
        if bytes[i - 1] == b'/' {
            seen += 1;
            if seen == index {
                return i - 1;
            }
        }
    }
    0
}

fn split_reference_components(input: &str) -> (&str, Option<&str>, Option<&str>) {
    let bytes = input.as_bytes();
    let fragment_at = bytes.iter().position(|&b| b == b'#').unwrap_or(bytes.len());
    let query_at = bytes[..fragment_at]
        .iter()
        .position(|&b| b == b'?')
        .unwrap_or(fragment_at);

    let path = &input[..query_at];
    let query = if query_at < fragment_at {
        Some(&input[query_at + 1..fragment_at])
    } else {
        None
    };
    let fragment = if fragment_at < input.len() {
        Some(&input[fragment_at + 1..])
    } else {
        None
    };
    (path, query, fragment)
}

fn construct_url(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut result = String::new();
    result.push_str(&url.scheme);
    result.push_str("://");
    if let Some(info) = &url.userinfo {
        result.push_str(info);
        result.push('@');
    }
    if url.ipv6 {
        result.push('[');
        result.push_str(&url.host);
        result.push(']');
    } else {
        result.push_str(&url.host);
    }
    if let Some(port) = url.port {
        result.push(':');
        result.push_str(&port.to_string());
    }
    result.push_str(path);
    if let Some(q) = query {
        result.push('?');
        result.push_str(q);
    }
    if let Some(f) = fragment {
        result.push('#');
        result.push_str(f);
    }
    result
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base_path =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    validate_global_bytes(reference).map_err(ResolveError::Reference)?;

    let reference_bytes = reference.as_bytes();
    let first_delimiter = reference_bytes
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference_bytes[..first_delimiter].contains(&b':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (reference_path, reference_query, reference_fragment) =
        split_reference_components(reference);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        let query = reference_query.or(base_url.query.as_deref());
        (normalized_base_path.as_str(), query)
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = normalized_base_path
                .rfind('/')
                .map(|i| &normalized_base_path[..=i])
                .unwrap_or("/");
            format!("{slash}{reference_path}")
        };
        // Keep the constructed URL validation offsets and ordering authoritative.
        let raw = construct_url(
            &base_url,
            &merged,
            reference_query,
            reference_fragment,
        );
        let mut url = parse_url(&raw).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    };

    let raw = construct_url(
        &base_url,
        selected_path,
        selected_query,
        reference_fragment,
    );
    let mut url = parse_url(&raw).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}
