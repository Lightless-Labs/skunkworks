#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

struct Running {
    task: usize,
    worker: usize,
    finish: u64,
}

fn append_resources(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index > 0 {
                trace.push(',');
            }
            trace.push_str(&value.to_string());
        }
    }
}

fn append_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    trace.push_str(&time.to_string());
    trace.push('|');
    trace.push_str(kind);
    trace.push('|');
    trace.push_str(&task.to_string());
    trace.push('|');
    trace.push_str(&worker.to_string());
    trace.push('|');
    append_resources(trace, used);
    trace.push('\n');
}

fn fits(task: &Task, used: &[u64], capacity: &[u32]) -> bool {
    task.needs
        .iter()
        .enumerate()
        .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r])
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shape and demands in task-major order.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Kahn traversal using a min-heap so newly available smaller IDs
    // take precedence over IDs that were already available.
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    let mut available = std::collections::BinaryHeap::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &task_id in &order {
            removed[task_id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = String::new();
    trace.push_str("workers=");
    trace.push_str(&worker_count.to_string());
    trace.push_str(";resources=");
    trace.push_str(&capacity.len().to_string());
    trace.push('\n');

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];
    let mut running: Vec<Running> = Vec::new();
    let mut current_time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every task finishing now, in task ID order.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, item)| {
                if item.finish == current_time {
                    Some(index)
                } else {
                    None
                }
            })
            .collect();
        finishing.sort_by_key(|&index| running[index].task);

        for index in finishing.into_iter().rev() {
            let item = running.remove(index);
            for (resource, &need) in tasks[item.task].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            completed[item.task] = true;
            completed_count += 1;
            makespan = makespan.max(item.finish);
            append_event(
                &mut trace,
                current_time,
                "end",
                item.task,
                item.worker,
                &used,
            );
        }

        // Repeatedly admit one task, then reconsider eligibility and rank.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > current_time
                    || !tasks[task_id]
                        .deps
                        .iter()
                        .all(|&predecessor| completed[predecessor])
                    || !fits(&tasks[task_id], &used, capacity)
                {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(previous) => {
                        tails[task_id] > tails[previous]
                            || (tails[task_id] == tails[previous]
                                && (tasks[task_id].priority > tasks[previous].priority
                                    || (tasks[task_id].priority
                                        == tasks[previous].priority
                                        && task_id < previous)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let task_id = match best {
                Some(task_id) => task_id,
                None => break,
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= current_time);

            let Some(worker) = worker else {
                break;
            };

            let finish = current_time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: current_time,
                finish,
                worker,
            });
            append_event(
                &mut trace,
                current_time,
                "start",
                task_id,
                worker,
                &used,
            );

            if finish == current_time {
                // Zero-duration tasks complete immediately before another start.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                append_event(
                    &mut trace,
                    current_time,
                    "end",
                    task_id,
                    worker,
                    &used,
                );
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest future event that can change scheduling.
        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > current_time {
                next_time = Some(next_time.map_or(candidate, |old| old.min(candidate)));
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for task_id in 0..n {
            if !started[task_id] {
                consider(tasks[task_id].release);
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > current_time {
                consider(worker_available[worker]);
            }
        }

        current_time = next_time.expect("unfinished tasks must have a future event");
    }

    trace.push_str("makespan=");
    trace.push_str(&makespan.to_string());
    trace.push('\n');

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}
