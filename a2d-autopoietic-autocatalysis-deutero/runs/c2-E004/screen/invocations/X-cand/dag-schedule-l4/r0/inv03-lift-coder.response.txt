#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

#[derive(Clone)]
struct Running {
    task: usize,
    worker: usize,
    finish: u64,
    available: u64,
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependencies in the specified order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for resource in 0..capacity.len() {
            if task.needs[resource] > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need: task.needs[resource],
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build successor lists and indegrees. Dependencies are already validated.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            children[dependency].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(task_id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, task_id)| **task_id)
            .map(|(position, _)| position)
            .unwrap();
        let task_id = available.swap_remove(smallest_pos);
        order.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }
    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &task_id in &order {
                flags[task_id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let mut max_child_tail = 0u64;
        for &child in &children[task_id] {
            max_child_tail = max_child_tail.max(tails[child]);
        }
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut remaining_deps: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut running = Vec::<Running>::new();
    let mut worker_available = vec![0u64; worker_count];
    let mut used = vec![0u64; capacity.len()];
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));

    while completed_count < n {
        // Complete every task finishing now, in ascending task ID.
        let mut ending: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, item)| (item.finish == now).then_some(index))
            .collect();
        ending.sort_by_key(|&index| running[index].task);
        for index in ending.into_iter().rev() {
            let item = running.remove(index);
            let task_id = item.task;
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(item.finish);

            for resource in 0..capacity.len() {
                used[resource] -= tasks[task_id].needs[resource] as u64;
            }
            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }

            trace.push_str(&format!("{}|end|{}|{}|", now, task_id, item.worker));
            append_resources(&mut trace, &used);
            trace.push('\n');
        }

        // Admit one task at a time, reevaluating after each start.
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    if started[task_id]
                        || tasks[task_id].release > now
                        || remaining_deps[task_id] != 0
                    {
                        return false;
                    }
                    (0..capacity.len()).all(|resource| {
                        tasks[task_id].needs[resource] as u64
                            <= capacity[resource] as u64 - used[resource]
                    })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            // Choose a worker only after choosing the task.
            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now);
            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let turnaround = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = turnaround;
            for resource in 0..capacity.len() {
                used[resource] += tasks[task_id].needs[resource] as u64;
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            trace.push_str(&format!("{}|start|{}|{}|", now, task_id, worker));
            append_resources(&mut trace, &used);
            trace.push('\n');

            if tasks[task_id].duration == 0 {
                // Immediate completion occurs before selecting another task.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for resource in 0..capacity.len() {
                    used[resource] -= tasks[task_id].needs[resource] as u64;
                }
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }

                trace.push_str(&format!("{}|end|{}|{}|", now, task_id, worker));
                append_resources(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                    available: turnaround,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest event that can change eligibility or worker
        // availability. The current time has already had all starts attempted.
        let mut next_time: Option<u64> = None;
        for item in &running {
            if item.finish > now {
                next_time = Some(next_time.map_or(item.finish, |t| t.min(item.finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                next_time = Some(next_time.map_or(tasks[task_id].release, |t| {
                    t.min(tasks[task_id].release)
                }));
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > now {
                next_time = Some(next_time.map_or(worker_available[worker], |t| {
                    t.min(worker_available[worker])
                }));
            }
        }

        // A task can be blocked only by a running predecessor, resource holder,
        // release, or worker cooldown, so a future event must exist.
        now = next_time.expect("unfinished tasks must have a future event");
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task starts exactly once"))
        .collect();

    trace.push_str(&format!("makespan={}\n", makespan));

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn append_resources(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index > 0 {
                trace.push(',');
            }
            trace.push_str(&value.to_string());
        }
    }
}
