#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let route = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .bytes()
            .position(|byte| byte == b'?' || byte == b'#')
            .unwrap_or(path.len());
        let path_head = &path[..head_end];
        let request_segments = parse_path(path_head)?;

        for (route_id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != request_segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_segment, request_segment) in
                route.segments.iter().zip(request_segments.iter())
            {
                match pattern_segment {
                    Segment::Static(expected) => {
                        if expected != request_segment {
                            matches = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        params.push((name.clone(), request_segment.clone()));
                    }
                }
            }

            if matches {
                return Ok(Match {
                    route: route_id,
                    params,
                });
            }
        }

        Err(Error::NotFound)
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(pattern: &str) -> Result<Route, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    if pattern == "/" {
        return Ok(Route {
            segments: Vec::new(),
            rank: Vec::new(),
        });
    }

    let bytes = pattern.as_bytes();
    let mut segments = Vec::new();
    let mut rank = Vec::new();
    let mut names = Vec::<String>::new();

    let mut start = 1;
    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&byte| byte == b'/')
            .map(|relative| start + relative)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw_segment = &pattern[start..end];
        let first = bytes[start];

        if first == b'*' {
            return Err(Error::InvalidPattern { at: start });
        }

        if first == b':' {
            let name = &raw_segment[1..];
            if !valid_capture_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }

            let name = name.to_owned();
            names.push(name.clone());
            segments.push(Segment::Param(name));
            rank.push(1);
        } else {
            let decoded = decode_segment(raw_segment, start, ErrorKind::Pattern)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }

        if end == bytes.len() {
            break;
        }

        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPattern { at: bytes.len() - 1 });
        }
    }

    Ok(Route { segments, rank })
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&byte| byte == b'/')
            .map(|relative| start + relative)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }

        let raw_segment = &path[start..end];
        segments.push(decode_segment(raw_segment, start, ErrorKind::Path)?);

        if end == bytes.len() {
            break;
        }

        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPath { at: bytes.len() - 1 });
        }
    }

    Ok(segments)
}

#[derive(Clone, Copy)]
enum ErrorKind {
    Pattern,
    Path,
}

impl ErrorKind {
    fn make(self, at: usize) -> Error {
        match self {
            Self::Pattern => Error::InvalidPattern { at },
            Self::Path => Error::InvalidPath { at },
        }
    }
}

fn decode_segment(segment: &str, base: usize, kind: ErrorKind) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' {
            if index + 2 >= bytes.len() {
                return Err(kind.make(base + index));
            }

            let high = uppercase_hex(bytes[index + 1]);
            let low = uppercase_hex(bytes[index + 2]);
            let Some(high) = high else {
                return Err(kind.make(base + index));
            };
            let Some(low) = low else {
                return Err(kind.make(base + index));
            };

            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(kind.make(base + index));
            }

            decoded.push(value);
            index += 3;
            continue;
        }

        if !byte.is_ascii() || !is_unreserved(byte) {
            return Err(kind.make(base + index));
        }

        decoded.push(byte);
        index += 1;
    }

    // All accepted decoded bytes are ASCII.
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    let Some((&first, rest)) = bytes.split_first() else {
        return false;
    };

    first.is_ascii_lowercase()
        && rest
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}
