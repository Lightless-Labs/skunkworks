use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if start == *pos {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let field = &s[start..*pos];
    if field.len() > 1 && field.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    field.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut part_start = start;
    let mut i = start;
    while i <= end {
        if i == end || bytes[i] == b'.' {
            if i == part_start {
                return Err(VersionError::EmptyIdentifier { at: part_start });
            }
            let slice = &s[part_start..i];
            if pre && slice.len() > 1 && slice.bytes().all(is_digit) && slice.starts_with('0') {
                return Err(VersionError::LeadingZero { at: part_start });
            }
            result.push(slice.to_string());
            part_start = i + 1;
        } else if !(bytes[i].is_ascii_alphanumeric() || bytes[i] == b'-') {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let n = parse_core_number(s, &mut pos)?;
        if _ == 0 {
            let minor = n;
            // Keep the parsed minor while processing patch.
            let patch = {
                if pos >= s.len() || s.as_bytes()[pos] != b'.' {
                    if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                        return Err(VersionError::MissingCore { at: pos });
                    }
                    return Err(VersionError::InvalidChar { at: pos });
                }
                pos += 1;
                parse_core_number(s, &mut pos)?
            };
            let (pre, build) = parse_tail(s, pos)?;
            return Ok(Version { major, minor, patch, pre, build });
        }
        let _ = n;
    }
    unreachable!()
}

fn parse_tail(s: &str, mut pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    let bytes = s.as_bytes();
    if pos == bytes.len() {
        return Ok((Vec::new(), Vec::new()));
    }
    if bytes[pos] == b'-' {
        pos += 1;
        let plus = s[pos..].find('+').map(|n| pos + n).unwrap_or(s.len());
        let pre = parse_identifiers(s, pos, plus, true)?;
        if plus == s.len() {
            return Ok((pre, Vec::new()));
        }
        let build_start = plus + 1;
        let build = parse_identifiers(s, build_start, s.len(), false)?;
        return Ok((pre, build));
    }
    if bytes[pos] == b'+' {
        let build_start = pos + 1;
        let build = parse_identifiers(s, build_start, s.len(), false)?;
        return Ok((Vec::new(), build));
    }
    Err(VersionError::InvalidChar { at: pos })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Cmp {
    Lt(Version),
    Le(Version),
    Eq(Version),
    Ge(Version),
    Gt(Version),
}

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

fn cmp_ok(v: &Version, c: &Cmp) -> bool {
    let o = compare_versions(v, match c {
        Cmp::Lt(x) | Cmp::Le(x) | Cmp::Eq(x) | Cmp::Ge(x) | Cmp::Gt(x) => x,
    });
    match c {
        Cmp::Lt(_) => o == Ordering::Less,
        Cmp::Le(_) => o != Ordering::Greater,
        Cmp::Eq(_) => o == Ordering::Equal,
        Cmp::Ge(_) => o != Ordering::Less,
        Cmp::Gt(_) => o == Ordering::Greater,
    }
}

fn split_ws(s: &str) -> Vec<(usize, &str)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') { i += 1; }
        if i == b.len() { break; }
        let st = i;
        while i < b.len() && !matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') { i += 1; }
        out.push((st, &s[st..i]));
    }
    out
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone)]
enum Partial {
    Full(Version),
    Fields(Vec<u64>),
}
fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let contains_mark = s.contains('-') || s.contains('+');
    let has_three_nonwild = fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X"));
    if contains_mark || has_three_nonwild {
        return parse_full_operand(s, at).map(Partial::Full);
    }
    if fields.len() > 3 || fields.is_empty() {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for f in fields {
        if f.is_empty() { return Err(RangeError::Syntax { at }); }
        if matches!(f, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial::Fields(nums))
}

fn vtuple(t: (u64,u64,u64), pre: Vec<String>) -> Version {
    Version { major:t.0, minor:t.1, patch:t.2, pre, build:Vec::new() }
}
fn padded(nums: &[u64]) -> (u64,u64,u64) {
    (nums.get(0).copied().unwrap_or(0), nums.get(1).copied().unwrap_or(0), nums.get(2).copied().unwrap_or(0))
}
fn bump(t:(u64,u64,u64), which:usize, at:usize) -> Result<Version,RangeError> {
    let (mut a,mut b,mut c)=t;
    match which {
        0 => { a=a.checked_add(1).ok_or(RangeError::Syntax{at})?; b=0;c=0; }
        1 => { b=b.checked_add(1).ok_or(RangeError::Syntax{at})?; c=0; }
        _ => { c=c.checked_add(1).ok_or(RangeError::Syntax{at})?; }
    }
    Ok(vtuple((a,b,c),vec!["0".into()]))
}

fn add_partial(cmps:&mut Vec<Cmp>, anchors:&mut Vec<(u64,u64,u64)>, p:Partial, kind:u8, at:usize)->Result<(),RangeError>{
    match p {
        Partial::Full(v) => {
            if !v.pre.is_empty(){anchors.push((v.major,v.minor,v.patch));}
            match kind {
                0 => cmps.push(Cmp::Eq(v)),
                1 => {
                    if v.pre.is_empty() { return Err(RangeError::Syntax{at}); }
                    cmps.push(Cmp::Ge(v));
                }
                2 => cmps.push(Cmp::Ge(v)),
                _ => {}
            }
        }
        Partial::Fields(nums) => {
            let k=nums.len();
            let t=padded(&nums);
            if kind==0 {
                if k==0 { return Ok(()); }
                cmps.push(Cmp::Ge(vtuple(t,Vec::new())));
                if k<3 {
                    let upper=bump(t, if k==1 {0}else{1},at)?;
                    cmps.push(Cmp::Lt(upper));
                }
            } else if kind==1 || kind==2 {
                if k==0 { return Err(RangeError::Syntax{at}); }
                cmps.push(Cmp::Ge(vtuple(t,Vec::new())));
                let up = if kind==1 {
                    if t.0>0 || k==1 {0} else if t.1>0 || k==2 {1} else {2}
                } else if k==1 {0} else {1};
                cmps.push(Cmp::Lt(bump(t,up,at)?));
            }
        }
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens=split_ws(s);
    if tokens.is_empty(){return Err(RangeError::Empty{at:0});}
    let mut arms=Vec::new();
    let mut start=0usize;
    let mut current=Vec::new();
    for (i,t) in tokens.iter().enumerate() {
        if t.1=="||" {
            if current.is_empty(){return Err(RangeError::Empty{at:t.0});}
            arms.push(parse_arm(&current)?);
            current.clear();
            start=i+1;
        } else { current.push(*t); }
    }
    if current.is_empty() {
        let at=if let Some((pos,_))=tokens.last(){*pos+tokens.last().unwrap().1.len()}else{0};
        return Err(RangeError::Empty{at});
    }
    let _=start;
    arms.push(parse_arm(&current)?);
    Ok(Range{arms})
}

fn parse_arm(tokens:&[(usize,&str)]) -> Result<Arm,RangeError>{
    let mut cmps=Vec::new(); let mut anchors=Vec::new();
    if let Some((idx,_))=tokens.iter().enumerate().find(|(_,t)|t.1=="-"){
        if tokens.len()!=3 || idx!=1 { return Err(RangeError::Syntax{at:tokens[idx].0}); }
        for endpoint in [tokens[0],tokens[2]] {
            let p=parse_partial(endpoint.1,endpoint.0)?;
            if matches!(&p,Partial::Fields(v) if v.is_empty()){return Err(RangeError::Syntax{at:endpoint.0});}
            if let Partial::Full(v)=&p { if !v.pre.is_empty(){anchors.push((v.major,v.minor,v.patch));} }
            if endpoint.0==tokens[0].0 {
                match p { Partial::Full(v)=>cmps.push(Cmp::Ge(v)), Partial::Fields(n)=>cmps.push(Cmp::Ge(vtuple(padded(&n),Vec::new()))) }
            } else {
                match p {
                    Partial::Full(v)=>cmps.push(Cmp::Le(v)),
                    Partial::Fields(n)=> { let k=n.len(); let t=padded(&n); if k<3 { cmps.push(Cmp::Lt(bump(t,if k==1{0}else{1},endpoint.0)?)); } else { cmps.push(Cmp::Le(vtuple(t,Vec::new()))); } }
                }
            }
        }
        return Ok(Arm{cmps,anchors});
    }
    for (at,t) in tokens {
        let bytes=t.as_bytes();
        if bytes.is_empty(){return Err(RangeError::Syntax{at:*at});}
        if matches!(bytes[0],b'<'|b'>'|b'=') {
            let (op,n)=if t.starts_with("<="){("<=",2)}else if t.starts_with(">="){(">=",2)}else if t.starts_with('<'){("<",1)}else if t.starts_with('>'){(">",1)}else if t.starts_with('='){("=",1)}else{unreachable!()};
            let operand=&t[n..]; let oa=*at+n;
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit(){return Err(RangeError::Syntax{at:*at});}
            let v=parse_full_operand(operand,oa)?;
            if !v.pre.is_empty(){anchors.push((v.major,v.minor,v.patch));}
            cmps.push(match op {"<"=>Cmp::Lt(v),"<="=>Cmp::Le(v),">"=>Cmp::Gt(v),">="=>Cmp::Ge(v),_=>Cmp::Eq(v)});
        } else if bytes[0]==b'^' || bytes[0]==b'~' {
            let kind=if bytes[0]==b'^'{1}else{2};
            let operand=&t[1..]; let oa=*at+1;
            let p=parse_partial(operand,oa)?;
            add_partial(&mut cmps,&mut anchors,p,kind,oa)?;
        } else if bytes[0].is_ascii_digit() || matches!(bytes[0],b'*'|b'x'|b'X') {
            let p=parse_partial(t,*at)?;
            add_partial(&mut cmps,&mut anchors,p,0,*at)?;
        } else {return Err(RangeError::Syntax{at:*at});}
    }
    Ok(Arm{cmps,anchors})
}

fn arm_matches(v:&Version,a:&Arm,include:bool)->bool{
    a.cmps.iter().all(|c|cmp_ok(v,c)) &&
    (include || v.pre.is_empty() || a.anchors.contains(&(v.major,v.minor,v.patch)))
}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool{
    r.arms.iter().any(|a|arm_matches(v,a,include_prerelease))
}
pub fn matches(v:&Version,r:&Range)->bool{matches_with(v,r,false)}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{
    extrema(candidates,r,true)
}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{
    extrema(candidates,r,false)
}
fn extrema(candidates:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in candidates.iter().enumerate(){ parsed.push(parse_version(s).map_err(|error|CandidateError{index:i,error})?); }
    let mut best:Option<(usize,usize)>=None;
    for (i,v) in parsed.iter().enumerate(){
        if !matches(v,r){continue;}
        if let Some((bi,_))=best{
            let ord=compare_versions(v,&parsed[bi]);
            if (max && ord==Ordering::Greater)||(!max&&ord==Ordering::Less){best=Some((i,i));}
        }else{best=Some((i,i));}
    }
    Ok(best.map(|(i,_)|candidates[i].clone()))
}

pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool{
    for aa in &a.arms {
        for bb in &b.arms {
            if arm_intersects(aa,bb,include_prerelease){return true;}
        }
    }
    false
}
fn arm_intersects(a:&Arm,b:&Arm,include:bool)->bool{
    let mut anchors=a.anchors.clone();
    anchors.extend(b.anchors.iter().copied());
    for t in anchors {
        let v=vtuple(t,vec!["0".into()]);
        if arm_matches(&v,a,include) && arm_matches(&v,b,include){return true;}
    }
    if include {
        return stable_intersection(a,b);
    }
    stable_intersection(a,b)
}
fn stable_intersection(a:&Arm,b:&Arm)->bool{
    let mut candidates=vec![vtuple((0,0,0),Vec::new()),vtuple((u64::MAX,u64::MAX,u64::MAX),Vec::new())];
    for c in a.cmps.iter().chain(&b.cmps){
        let v=match c {Cmp::Lt(v)|Cmp::Le(v)|Cmp::Eq(v)|Cmp::Ge(v)|Cmp::Gt(v)=>v};
        candidates.push(vtuple((v.major,v.minor,v.patch),Vec::new()));
        if v.patch<u64::MAX { candidates.push(vtuple((v.major,v.minor,v.patch+1),Vec::new())); }
    }
    candidates.into_iter().any(|v|v.pre.is_empty() && a.cmps.iter().all(|c|cmp_ok(&v,c)) && b.cmps.iter().all(|c|cmp_ok(&v,c)))
}

pub fn canonical(r:&Range)->String{
    let mut out=Vec::new();
    for a in &r.arms {
        let mut lo:Option<(Version,bool)>=None; let mut hi:Option<(Version,bool)>=None;
        for c in &a.cmps {
            match c {
                Cmp::Eq(v)=>{set_lower(&mut lo,v,true);set_upper(&mut hi,v,true);}
                Cmp::Ge(v)=>set_lower(&mut lo,v,true),
                Cmp::Gt(v)=>set_lower(&mut lo,v,false),
                Cmp::Le(v)=>set_upper(&mut hi,v,true),
                Cmp::Lt(v)=>set_upper(&mut hi,v,false),
            }
        }
        if !stable_intersection(a,a) && !a.anchors.iter().any(|t| {
            let v=vtuple(*t,vec!["0".into()]);
            a.cmps.iter().all(|c|cmp_ok(&v,c))
        }) {continue;}
        let l=lo.map(|(v,inc)|format!("{}{}",if inc{">="}else{">"},fmt_version(&v))).unwrap_or("*".into());
        let h=hi.map(|(v,inc)|format!("{}{}",if inc{"<="}else{"<"},fmt_version(&v))).unwrap_or("*".into());
        let mut anc=a.anchors.clone();anc.sort();anc.dedup();
        let anchor=if anc.is_empty(){"-".into()}else{anc.iter().map(|t|format!("{}.{}.{}",t.0,t.1,t.2)).collect::<Vec<_>>().join(",")};
        out.push(format!("[{};{};pre={}]",l,h,anchor));
    }
    out.sort();out.dedup();
    if out.is_empty(){"EMPTY".into()}else{out.join(" || ")}
}
fn set_lower(dst:&mut Option<(Version,bool)>,v:&Version,inc:bool){
    match dst {None=>*dst=Some((v.clone(),inc)),Some((old,oi))=>{
        let o=compare_versions(v,old); if o==Ordering::Greater || (o==Ordering::Equal && !inc && *oi){*dst=Some((v.clone(),inc));}
    }}
}
fn set_upper(dst:&mut Option<(Version,bool)>,v:&Version,inc:bool){
    match dst {None=>*dst=Some((v.clone(),inc)),Some((old,oi))=>{
        let o=compare_versions(v,old); if o==Ordering::Less || (o==Ordering::Equal && !inc && *oi){*dst=Some((v.clone(),inc));}
    }}
}
fn fmt_version(v:&Version)->String{
    let mut s=format!("{}.{}.{}",v.major,v.minor,v.patch);
    if !v.pre.is_empty(){s.push('-');s.push_str(&v.pre.join("."));}
    s
}
