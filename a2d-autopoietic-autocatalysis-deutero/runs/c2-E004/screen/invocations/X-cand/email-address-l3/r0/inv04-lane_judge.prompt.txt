pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(bytes)?;
    check_boundaries(&cleaned)?;

    if cleaned.retained.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let data = &cleaned.retained;
    let mut quote_close = None;
    if data[0].byte == b'"' {
        let mut i = 1;
        while i < data.len() {
            match data[i].byte {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    quote_close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote { at: data[0].pos });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    if let Some(close) = quote_close {
        i = close + 1;
    }
    while i < data.len() {
        if data[i].byte == b'@' {
            ats.push(i);
        }
        i += 1;
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: data[ats[1]].pos,
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: data[sep].pos });
    }

    let local = if data[0].byte == b'"' {
        let close = quote_close.expect("opening quote was scanned");
        if close + 1 != sep {
            let at = data.get(close + 1).map_or(input.len(), |r| r.pos);
            return Err(EmailError::TrailingLocal { at });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut p = 1;
        while p < close {
            let b = data[p].byte;
            if b == b'\\' {
                let next = data.get(p + 1).map(|r| r.byte);
                match next {
                    Some(b'"' | b'\\' | b'@') => {
                        decoded.push(next.unwrap());
                        origins.push(data[p].pos);
                        p += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: data[p].pos }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: data[p].pos });
                }
                decoded.push(b);
                origins.push(data[p].pos);
                p += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: data[close].pos });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).expect("validated ASCII"), true)
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: data[64].pos });
        }

        for p in 0..sep {
            let b = data[p].byte;
            if b == b'.' {
                if p == 0 || p + 1 == sep || data[p - 1].byte == b'.' {
                    return Err(EmailError::LocalDot { at: data[p].pos });
                }
            } else if !is_atext(b) {
                return Err(EmailError::LocalChar { at: data[p].pos });
            }
        }
        (
            String::from_utf8(data[..sep].iter().map(|r| r.byte).collect())
                .expect("validated ASCII"),
            false,
        )
    };

    let domain_start = sep + 1;
    if domain_start >= data.len() {
        return Err(EmailError::EmptyDomain { at: input.len() });
    }

    let mut domain_end = data.len();
    let root_pos;
    if data[domain_end - 1].byte == b'.' {
        root_pos = Some(data[domain_end - 1].pos);
        domain_end -= 1;
    } else {
        root_pos = None;
    }

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: data.get(domain_start).map_or(input.len(), |r| r.pos),
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: data[domain_start + 253].pos,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    let mut p = domain_start;
    while p <= domain_end {
        if p == domain_end || data[p].byte == b'.' {
            if label_start == p {
                let at = if p == domain_end {
                    root_pos.unwrap_or(input.len())
                } else {
                    data[p].pos
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, p));
            label_start = p + 1;
        }
        p += 1;
    }

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: data[start + 63].pos,
            });
        }
        if data[start].byte == b'-' {
            return Err(EmailError::LabelHyphen { at: data[start].pos });
        }
        if data[end - 1].byte == b'-' {
            return Err(EmailError::LabelHyphen { at: data[end - 1].pos });
        }
        for r in &data[start..end] {
            if !r.byte.is_ascii_alphanumeric() && r.byte != b'-' {
                return Err(EmailError::DomainChar { at: r.pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: data[domain_start].pos,
        });
    }

    let (last_start, last_end) = *labels.last().expect("at least one label");
    if data[last_start..last_end]
        .iter()
        .all(|r| r.byte.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: data[last_start].pos,
        });
    }

    let domain = data[domain_start..domain_end]
        .iter()
        .map(|r| (r.byte as char).to_ascii_lowercase())
        .collect();

    Ok(Address {
        local: local.0,
        domain,
        quoted: local.1,
    })
}

fn is_atext(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn cleanup(bytes: &[u8]) -> Result<Cleanup, EmailError> {
    let mut retained = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut in_run = false;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut protected_quote = false;

    while i < bytes.len() {
        let b = bytes[i];

        if protected_quote {
            retained.push(Retained { byte: b, pos: i });
            if b == b'\\' {
                if i + 1 < bytes.len() {
                    i += 1;
                    retained.push(Retained {
                        byte: bytes[i],
                        pos: i,
                    });
                }
            } else if b == b'"' {
                protected_quote = false;
            }
            i += 1;
            continue;
        }

        if comment_depth > 0 {
            if b == b'\\' {
                if i + 1 >= bytes.len() || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if !valid_fold(bytes, i) {
                    return Err(EmailError::BadFold { at: i });
                }
                i = fold_end(bytes, i);
                continue;
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if b == b'\t' || (0x20..=0x7e).contains(&b) {
                i += 1;
                continue;
            }
            return Err(EmailError::CommentChar { at: i });
        }

        if b == b' ' || b == b'\t' || b == b'(' || b == b'\r' || b == b'\n' {
            if !in_run {
                runs.push(i);
                in_run = true;
            }
            if b == b'(' {
                comment_depth = 1;
                outer_comment = i;
                i += 1;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if !valid_fold(bytes, i) {
                    return Err(EmailError::BadFold { at: i });
                }
                i = fold_end(bytes, i);
                continue;
            }
            i += 1;
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if !protected_quote && retained.is_empty() && b == b'"' {
            in_run = false;
            protected_quote = true;
            retained.push(Retained { byte: b, pos: i });
            i += 1;
            continue;
        }

        in_run = false;
        retained.push(Retained { byte: b, pos: i });
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    Ok(Cleanup { retained, runs })
}

fn valid_fold(bytes: &[u8], at: usize) -> bool {
    at + 2 < bytes.len()
        && bytes[at] == b'\r'
        && bytes[at + 1] == b'\n'
        && (bytes[at + 2] == b' ' || bytes[at + 2] == b'\t')
}

fn fold_end(bytes: &[u8], at: usize) -> usize {
    let mut i = at + 2;
    while i < bytes.len() && (bytes[i] == b' ' || bytes[i] == b'\t') {
        i += 1;
    }
    i
}

fn check_boundaries(cleaned: &Cleanup) -> Result<(), EmailError> {
    for &run_start in &cleaned.runs {
        let left_is_at = cleaned
            .retained
            .iter()
            .rev()
            .find(|r| r.pos < run_start)
            .is_some_and(|r| r.byte == b'@');
        let right_is_at = cleaned
            .retained
            .iter()
            .find(|r| r.pos > run_start)
            .is_some_and(|r| r.byte == b'@');

        let has_retained_before = cleaned.retained.iter().any(|r| r.pos < run_start);
        let has_retained_after = cleaned.retained.iter().any(|r| r.pos > run_start);

        if has_retained_before && has_retained_after && !left_is_at && !right_is_at {
            return Err(EmailError::CfwsBoundary { at: run_start });
        }
    }
    Ok(())
}

fn consume_fold(raw: &[u8], i: usize) -> Option<usize> {
    if raw.get(i) != Some(&b'\r')
        || raw.get(i + 1) != Some(&b'\n')
        || !matches!(raw.get(i + 2), Some(b' ') | Some(b'\t'))
    {
        return None;
    }

    let mut end = i + 2;
    while matches!(raw.get(end), Some(b' ') | Some(b'\t')) {
        end += 1;
    }
    Some(end)
}

fn cleanup(input: &str) -> Result<Cleaned, EmailError> {

    let mut positions = Vec::with_capacity(raw.len());

    let mut quote_protected = false;
    let mut in_quote = false;

            let b = raw[i];
            bytes.push(b);
            positions.push(i);
            i += 1;

                    i += 1;

        }

        if matches!(raw[i], b' ' | b'\t' | b'\r' | b'\n' | b'(') {
            let run_start = i;

            loop {
                if i >= raw.len() {
                    break;
                }

                match raw[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        if let Some(end) = consume_fold(raw, i) {
                            i = end;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut stack = Vec::with_capacity(3);

                        while i < raw.len() {
                            match raw[i] {
                                b'(' => {
                                    if stack.len() == 3 {
                                        return Err(EmailError::CommentDepth { at: i });
                                    }
                                    stack.push(i);
                                    i += 1;
                                }
                                b')' => {
                                    stack.pop();
                                    i += 1;
                                    if stack.is_empty() {
                                        break;
                                    }
                                }
                                b'\\' => {
                                    let slash = i;
                                    if i + 1 >= raw.len() || !printable(raw[i + 1]) {
                                        return Err(EmailError::CommentEscape { at: slash });
                                    }
                                    i += 2;
                                }
                                b'\r' | b'\n' => {
                                    if let Some(end) = consume_fold(raw, i) {
                                        i = end;
                                    } else {
                                        return Err(EmailError::BadFold { at: i });
                                    }
                                }
                                b if printable(b) || b == b'\t' => i += 1,
                                _ => return Err(EmailError::CommentChar { at: i }),
                            }
                        }

                        if !stack.is_empty() {
                            return Err(EmailError::UnclosedComment { at: outer });
                        }
                    }
                    _ => break,
                }
            }

            if i > run_start {
                runs.push((run_start, i));
                continue;
            }
        }

        if raw[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if !quote_protected && bytes.is_empty() && raw[i] == b'"' {
            quote_protected = true;
            in_quote = true;
        }

        bytes.push(raw[i]);
        positions.push(i);
        i += 1;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn parse_email_impl(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();

    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(input)?;
    let bytes = &cleaned.bytes;
    let positions = &cleaned.positions;

    // A run is at an edge if it has no retained bytes on that side. Otherwise
    // it is legal only when its immediately adjacent retained byte is '@'.
    for &(start, end) in &cleaned.runs {
        let left = positions.iter().rposition(|&p| p < start);
        let right = positions.iter().position(|&p| p >= end);

        let at_edge = left.is_none() || right.is_none();
        let adjacent_to_at = left.is_some_and(|idx| bytes[idx] == b'@')
            || right.is_some_and(|idx| bytes[idx] == b'@');

        if !(at_edge || adjacent_to_at) {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let pos = |idx: usize| positions.get(idx).copied().unwrap_or(input.len());

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut j = 1;
        while j < bytes.len() {
            if bytes[j] == b'\\' {
                j += 2;
            } else if bytes[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: pos(0) });
        }
    }

    let protected_end = quote_end.map_or(0, |end| end + 1);
    let mut separators = Vec::new();
    for (idx, &b) in bytes.iter().enumerate() {
        if b == b'@' && (quote_end.is_none() || idx >= protected_end) {
            separators.push(idx);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos(separators[1]),
        });
    }

    let at_idx = separators[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: pos(at_idx) });
    }

    let mut local = Vec::new();
    let quoted;

    if let Some(qend) = quote_end {
        if qend + 1 != at_idx {
            return Err(EmailError::TrailingLocal { at: pos(qend + 1) });
        }

        quoted = true;
        let mut j = 1;
        while j < qend {
            if bytes[j] == b'\\' {
                let slash = j;
                if j + 1 >= qend || !matches!(bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: pos(slash) });
                }
                local.push((bytes[j + 1], pos(slash)));
                j += 2;
            } else {
                if !printable(bytes[j]) {
                    return Err(EmailError::QuotedChar { at: pos(j) });
                }
                local.push((bytes[j], pos(j)));
                j += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(qend) });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: local[64].1 });
        }
    } else {
        quoted = false;

        if at_idx > 64 {
            return Err(EmailError::LocalTooLong { at: pos(64) });
        }

        for j in 0..at_idx {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == at_idx || bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: pos(j) });
                }
            } else if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                return Err(EmailError::LocalChar { at: pos(j) });
            }
        }

        local.extend((0..at_idx).map(|j| (bytes[j], pos(j))));
    }

    let domain_start = at_idx + 1;
    let domain_start_pos = pos(domain_start);
    if domain_start >= bytes.len() {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    let mut domain_end = bytes.len();
    let root_pos = if bytes.last() == Some(&b'.') {
        domain_end -= 1;
        Some(pos(bytes.len() - 1))
    } else {
        None
    };

    if domain_end <= domain_start {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(domain_start + 253),
        });
    }

    let mut labels = Vec::new();
    let mut label_start = domain_start;
    for cursor in domain_start..=domain_end {
        if cursor == domain_end || bytes[cursor] == b'.' {
            labels.push((label_start, cursor));
            label_start = cursor + 1;
        }
    }

    for (index, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let at = if index + 1 == labels.len() {
                root_pos.unwrap_or_else(|| pos(start))
            } else {
                pos(start)
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong { at: pos(start + 63) });
        }
        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(start) });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(end - 1) });
        }
        for j in start..end {
            let b = bytes[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: pos(j) });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(last_start) });
    }

    let local_string: String = local.into_iter().map(|(b, _)| b as char).collect();
    let domain_string: String = bytes[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_email_impl(input)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    for &run_start in &cleaned.runs {
        let retained_index = cleaned
            .positions
            .partition_point(|&p| p < run_start);
        let at_boundary = retained_index == 0
            || retained_index == cleaned.bytes.len()
            || cleaned.bytes.get(retained_index.wrapping_sub(1)) == Some(&b'@')
            || cleaned.bytes.get(retained_index) == Some(&b'@');
        if !at_boundary {
            return Err(EmailError::CfwsBoundary { at: run_start });
        }
    }

    if cleaned.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let bytes = &cleaned.bytes;
    let positions = &cleaned.positions;
    let n = bytes.len();

    let quoted_prefix_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut close = None;
        while i < n {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match close {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: positions[0] }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < n {
        if let Some(end) = quoted_prefix_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: positions[sep] });
    }

    let (local, quoted) = if bytes[0] == b'"' {
        let close = quoted_prefix_end.expect("opening quote has a closing quote");
        if close + 1 != sep {
            let at = if close + 1 < n {
                positions[close + 1]
            } else {
                input.len()
            };
            return Err(EmailError::TrailingLocal { at });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < close {
            let b = bytes[j];
            if b == b'\\' {
                if j + 1 >= close {
                    return Err(EmailError::BadEscape { at: positions[j] });
                }
                let escaped = bytes[j + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: positions[j] });
                }
                decoded.push(escaped);
                origins.push(positions[j]);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: positions[j] });
                }
                decoded.push(b);
                origins.push(positions[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[close],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (
            String::from_utf8(decoded).expect("decoded bytes are ASCII"),
            true,
        )
    } else {
        let len = sep;
        if len > 64 {
            return Err(EmailError::LocalTooLong { at: positions[64] });
        }
        let mut previous_dot = false;
        for j in 0..len {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == len || previous_dot {
                    return Err(EmailError::LocalDot { at: positions[j] });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !b.is_ascii_alphanumeric()
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
                {
                    return Err(EmailError::LocalChar { at: positions[j] });
                }
            }
        }
        (
            String::from_utf8(bytes[..len].to_vec()).expect("cleaned bytes are ASCII"),
            false,
        )
    };

    let domain_start = sep + 1;
    let domain_end = n;
    let has_root_dot = domain_start < domain_end && bytes[domain_end - 1] == b'.';
    let effective_end = if has_root_dot { domain_end - 1 } else { domain_end };
    if effective_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: if domain_start < n {
                positions[domain_start]
            } else {
                input.len()
            },
        });
    }

    if effective_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..=effective_end {
        if j == effective_end || bytes[j] == b'.' {
            if label_start == j {
                let at = if has_root_dot && j == effective_end {
                    positions[domain_end - 1]
                } else if j < n {
                    positions[j]
                } else {
                    input.len()
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: positions[start] });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for j in start..end {
            let b = bytes[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: positions[j] });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: positions[domain_start],
        });
    }

    let (last_start, last_end) = *labels.last().expect("at least one label");
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = bytes[domain_start..effective_end]
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    Ok(Address {
        local,
        domain: String::from_utf8(domain).expect("domain is ASCII"),
        quoted,
    })
}

fn cleanup(raw: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(raw.len());
    let mut positions = Vec::with_capacity(raw.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut in_run = false;
    let mut first_retained_is_quote = false;
    let mut quote_open = false;

    while i < raw.len() {
        let b = raw[i];

        if quote_open {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < raw.len() {
                    bytes.push(raw[i]);
                    positions.push(i);
                }
            } else if b == b'"' {
                quote_open = false;
            }
            i += 1;
            continue;
        }

        let is_cfws_start = b == b' ' || b == b'\t' || b == b'(' || b == b'\r' || b == b'\n';
        if is_cfws_start {
            if !in_run {
                runs.push(i);
                in_run = true;
            }
            if b == b' ' || b == b'\t' {
                i += 1;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if b != b'\r'
                    || i + 2 >= raw.len()
                    || raw[i + 1] != b'\n'
                    || !(raw[i + 2] == b' ' || raw[i + 2] == b'\t')
                {
                    return Err(EmailError::BadFold { at: i });
                }
                i += 2;
                while i < raw.len() && (raw[i] == b' ' || raw[i] == b'\t') {
                    i += 1;
                }
                continue;
            }

            let outer_open = i;
            let mut stack = vec![i];
            i += 1;
            while let Some(&outermost) = stack.first() {
                if i >= raw.len() {
                    return Err(EmailError::UnclosedComment { at: outermost });
                }
                let c = raw[i];
                if c == b'\\' {
                    if i + 1 >= raw.len()
                        || !(0x20..=0x7e).contains(&raw[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if c != b'\r'
                        || i + 2 >= raw.len()
                        || raw[i + 1] != b'\n'
                        || !(raw[i + 2] == b' ' || raw[i + 2] == b'\t')
                    {
                        return Err(EmailError::BadFold { at: i });
                    }
                    i += 2;
                    while i < raw.len() && (raw[i] == b' ' || raw[i] == b'\t') {
                        i += 1;
                    }
                    continue;
                }
                if c == b'(' {
                    if stack.len() == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    stack.push(i);
                    i += 1;
                    continue;
                }
                if c == b')' {
                    stack.pop();
                    i += 1;
                    continue;
                }
                if c == b'\t' || (0x20..=0x7e).contains(&c) {
                    i += 1;
                    continue;
                }
                return Err(EmailError::CommentChar { at: i });
            }
            let _ = outer_open;
            continue;
        }

        in_run = false;
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if bytes.is_empty() && b == b'"' {
            first_retained_is_quote = true;
            quote_open = true;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    let _ = first_retained_is_quote;
    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}
