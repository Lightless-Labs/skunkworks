#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn parse_email_impl(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut bytes = Vec::with_capacity(raw.len());
    let mut positions = Vec::with_capacity(raw.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_protected = false;
    let mut in_quote = false;

    while i < raw.len() {
        if in_quote {
            let b = raw[i];
            bytes.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' && i < raw.len() {
                bytes.push(raw[i]);
                positions.push(i);
                i += 1;
            } else if b == b'"' {
                in_quote = false;
            }
            continue;
        }

        let start = i;
        let b = raw[i];

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let run_start = i;
            loop {
                if i >= raw.len() {
                    break;
                }
                match raw[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        let p = i;
                        if raw[i] != b'\r'
                            || i + 2 >= raw.len()
                            || raw[i + 1] != b'\n'
                            || !(raw[i + 2] == b' ' || raw[i + 2] == b'\t')
                        {
                            return Err(EmailError::BadFold { at: p });
                        }
                        i += 2;
                        while i < raw.len() && (raw[i] == b' ' || raw[i] == b'\t') {
                            i += 1;
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut depth = 0usize;
                        while i < raw.len() {
                            match raw[i] {
                                b'(' => {
                                    depth += 1;
                                    if depth > 3 {
                                        return Err(EmailError::CommentDepth { at: i });
                                    }
                                    i += 1;
                                }
                                b')' => {
                                    depth -= 1;
                                    i += 1;
                                    if depth == 0 {
                                        break;
                                    }
                                }
                                b'\\' => {
                                    let slash = i;
                                    if i + 1 >= raw.len() || !printable(raw[i + 1]) {
                                        return Err(EmailError::CommentEscape { at: slash });
                                    }
                                    i += 2;
                                }
                                b'\r' | b'\n' => {
                                    let p = i;
                                    if raw[i] != b'\r'
                                        || i + 2 >= raw.len()
                                        || raw[i + 1] != b'\n'
                                        || !(raw[i + 2] == b' ' || raw[i + 2] == b'\t')
                                    {
                                        return Err(EmailError::BadFold { at: p });
                                    }
                                    i += 2;
                                    while i < raw.len()
                                        && (raw[i] == b' ' || raw[i] == b'\t')
                                    {
                                        i += 1;
                                    }
                                }
                                c if printable(c) || c == b'\t' => i += 1,
                                _ => return Err(EmailError::CommentChar { at: i }),
                            }
                        }
                        if depth != 0 {
                            return Err(EmailError::UnclosedComment { at: outer });
                        }
                    }
                    _ => break,
                }
            }
            if i > run_start {
                runs.push((run_start, i));
                continue;
            }
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if !quote_protected && bytes.is_empty() && b == b'"' {
            quote_protected = true;
            in_quote = true;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
        let _ = start;
    }

    for &(start, end) in &runs {
        let left_is_at = positions.first().is_some_and(|&p| p >= end)
            && bytes.first() == Some(&b'@');
        let right_is_at = positions.last().is_some_and(|&p| p < start)
            && bytes.last() == Some(&b'@');
        let at_edge = !positions.iter().any(|&p| p < start)
            || !positions.iter().any(|&p| p >= end);
        if !(at_edge || left_is_at || right_is_at) {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let pos = |idx: usize| positions.get(idx).copied().unwrap_or(input.len());

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut j = 1;
        while j < bytes.len() {
            if bytes[j] == b'\\' {
                j += 2;
            } else if bytes[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: pos(0) });
        }
    }
    let protected_end = quote_end.map_or(0, |x| x + 1);
    let ats: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(idx, &b)| {
            if b == b'@' && (quote_end.is_none() || idx >= protected_end) {
                Some(idx)
            } else {
                None
            }
        })
        .collect();

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt { at: pos(ats[1]) });
    }
    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: pos(at_idx) });
    }

    let mut local = Vec::new();
    let quoted;
    if let Some(qend) = quote_end {
        if qend + 1 != at_idx {
            return Err(EmailError::TrailingLocal { at: pos(qend + 1) });
        }
        quoted = true;
        let mut j = 1;
        while j < qend {
            if bytes[j] == b'\\' {
                let slash = j;
                if j + 1 >= qend || !matches!(bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: pos(slash) });
                }
                local.push((bytes[j + 1], pos(slash)));
                j += 2;
            } else {
                if !printable(bytes[j]) {
                    return Err(EmailError::QuotedChar { at: pos(j) });
                }
                local.push((bytes[j], pos(j)));
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(qend) });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: local[64].1 });
        }
    } else {
        quoted = false;
        if at_idx > 64 {
            return Err(EmailError::LocalTooLong { at: pos(64) });
        }
        for j in 0..at_idx {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == at_idx || bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: pos(j) });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: pos(j) });
            }
        }
        local.extend((0..at_idx).map(|j| (bytes[j], pos(j))));
    }

    let domain_start_idx = at_idx + 1;
    let domain_start_pos = pos(domain_start_idx);
    if domain_start_idx >= bytes.len() {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    let mut domain_end = bytes.len();
    let root_pos;
    if bytes.last() == Some(&b'.') {
        domain_end -= 1;
        root_pos = Some(pos(bytes.len() - 1));
    } else {
        root_pos = None;
    }
    if domain_end <= domain_start_idx {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }
    let domain_len = domain_end - domain_start_idx;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(domain_start_idx + 253),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start_idx;
    let mut cursor = domain_start_idx;
    while cursor <= domain_end {
        if cursor == domain_end || bytes[cursor] == b'.' {
            labels.push((label_start, cursor));
            label_start = cursor + 1;
        }
        cursor += 1;
    }

    for (n, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let where_at = if n + 1 == labels.len() {
                root_pos.unwrap_or_else(|| pos(start))
            } else {
                pos(start)
            };
            return Err(EmailError::EmptyLabel { at: where_at });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong { at: pos(start + 63) });
        }
        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(start) });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(end - 1) });
        }
        for j in start..end {
            let b = bytes[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: pos(j) });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(last_start) });
    }

    let local_string: String = local.into_iter().map(|(b, _)| b as char).collect();
    let domain_string: String = bytes[domain_start_idx..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_email_impl(input)
}
