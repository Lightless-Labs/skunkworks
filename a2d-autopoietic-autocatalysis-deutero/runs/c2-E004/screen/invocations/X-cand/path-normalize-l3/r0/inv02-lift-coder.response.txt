use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> impl Iterator<Item = (usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, component| {
            let current = *offset;
            *offset += component.len() + 1;
            Some((current, component))
        })
        .filter(|(_, component)| !component.is_empty())
}

fn format_result(stack: &[String], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        if stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn apply_component(
    stack: &mut Vec<String>,
    absolute: bool,
    component: &str,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_owned());
            }
            Ok(())
        }
        _ => {
            stack.push(component.to_owned());
            Ok(())
        }
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory = directory_intent(path);
    let mut stack = Vec::new();

    for (at, component) in components(path) {
        apply_component(&mut stack, absolute, component, at)?;
    }

    Ok(format_result(&stack, absolute, directory))
}

fn normalized_absolute_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

fn validate_contexts(cwd: &str, home: Option<&str>) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_components = normalized_absolute_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(path) => Some(
            normalized_absolute_context(path).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };
    Ok((cwd_components, home_components))
}

fn request_start<'a>(
    path: &'a str,
    home: Option<&'a Vec<String>>,
) -> Result<(bool, Vec<String>, &'a str, usize), PathError> {
    if path.starts_with('~') {
        if path == "~" {
            return Ok((
                true,
                home.ok_or(PathError::MissingHome)?.clone(),
                "",
                path.len(),
            ));
        }
        if path.starts_with("~/") {
            return Ok((
                true,
                home.ok_or(PathError::MissingHome)?.clone(),
                &path[2..],
                2,
            ));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    if path.starts_with('/') {
        Ok((true, Vec::new(), path, 0))
    } else {
        Ok((true, Vec::new(), path, 0))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (cwd_components, home_components) = validate_contexts(cwd, home)?;
    validate_raw(path)?;

    let (absolute, mut stack, remainder, base_offset) =
        if path.starts_with('~') {
            let (absolute, stack, remainder, offset) =
                request_start(path, home_components.as_ref())?;
            (absolute, stack, remainder, offset)
        } else if path.starts_with('/') {
            (true, Vec::new(), path, 0)
        } else {
            (true, cwd_components, path, 0)
        };

    let directory = directory_intent(path);
    for (at, component) in components(remainder) {
        apply_component(&mut stack, absolute, component, at + base_offset)?;
    }

    Ok(format_result(&stack, true, directory))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut map = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();

        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(&first) = map.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        map.insert(link.path.clone(), index);
    }

    Ok(map)
}

#[derive(Clone)]
struct WorkItem {
    raw: String,
    offset_base: usize,
    pos: usize,
    link_index: Option<usize>,
}

impl WorkItem {
    fn new(raw: &str, offset_base: usize, link_index: Option<usize>) -> Self {
        Self {
            raw: raw.to_owned(),
            offset_base,
            pos: 0,
            link_index,
        }
    }

    fn next_component(&mut self) -> Option<(usize, String)> {
        let bytes = self.raw.as_bytes();
        while self.pos < bytes.len() && bytes[self.pos] == b'/' {
            self.pos += 1;
        }
        if self.pos >= bytes.len() {
            return None;
        }

        let start = self.pos;
        while self.pos < bytes.len() && bytes[self.pos] != b'/' {
            self.pos += 1;
        }
        Some((
            self.offset_base + start,
            self.raw[start..self.pos].to_owned(),
        ))
    }
}

fn current_absolute_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_components, home_components) = validate_contexts(cwd, home)?;
    let link_map = validate_links(links)?;
    validate_raw(path)?;

    let directory = directory_intent(path);
    let (mut stack, remainder, base_offset) = if path.starts_with('~') {
        if path == "~" {
            (
                home_components.clone().ok_or(PathError::MissingHome)?,
                "",
                path.len(),
            )
        } else if path.starts_with("~/") {
            (
                home_components.clone().ok_or(PathError::MissingHome)?,
                &path[2..],
                2,
            )
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_components, path, 0)
    };

    let mut queue = vec![WorkItem::new(remainder, base_offset, None)];
    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    while let Some(item) = queue.last_mut() {
        if let Some((at, component)) = item.next_component() {
            if component == "." {
                continue;
            }
            if component == ".." {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
                continue;
            }

            stack.push(component);
            let key = current_absolute_path(&stack);
            if let Some(&index) = link_map.get(&key) {
                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if expansions >= max_expansions {
                    return Err(PathError::Limit { index });
                }
                expansions += 1;

                let link = &links[index];
                stack.pop();
                if link.target.starts_with('/') {
                    stack.clear();
                }

                active.push(index);
                queue.push(WorkItem::new(&link.target, 0, Some(index)));
            }
        } else {
            let finished = queue.pop().expect("work item exists");
            if let Some(index) = finished.link_index {
                let popped = active.pop();
                debug_assert_eq!(popped, Some(index));
            }
        }
    }

    Ok(format_result(&stack, true, directory))
}
