#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }

    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }

        result.push((path[start..i].to_owned(), start));
    }

    result
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }

    result
}

fn reduce_component(
    stack: &mut Vec<String>,
    component: &str,
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    match component {
        "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
                Ok(())
            } else if absolute {
                Err(PathError::Escape { at: offset })
            } else {
                stack.push("..".to_owned());
                Ok(())
            }
        }
        _ => {
            stack.push(component.to_owned());
            Ok(())
        }
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    for (component, offset) in components(path) {
        reduce_component(&mut stack, &component, offset, absolute)?;
    }

    Ok(render(&stack, absolute, intent))
}

fn validate_context(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }

    let normalized = normalize(path).map_err(|_| invalid)?;
    Ok(components(&normalized)
        .into_iter()
        .map(|(component, _)| component)
        .collect())
}

fn validate_cwd(cwd: &str) -> Result<Vec<String>, PathError> {
    validate_context(cwd, PathError::InvalidCwd)
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    home.map(|path| validate_context(path, PathError::InvalidHome))
        .transpose()
}

enum RequestStart {
    Absolute,
    Relative,
    Home(usize),
}

fn parse_request_start(
    path: &str,
    home: Option<&[String]>,
) -> Result<RequestStart, PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            if home.is_none() {
                return Err(PathError::MissingHome);
            }
            return Ok(RequestStart::Home(if bytes.len() > 1 { 2 } else { 1 }));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    if path.starts_with('/') {
        Ok(RequestStart::Absolute)
    } else {
        Ok(RequestStart::Relative)
    }
}

fn resolve_core(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: Option<(&[Link], usize)>,
) -> Result<String, PathError> {
    let cwd_components = validate_cwd(cwd)?;
    let home_components = validate_home(home)?;

    let validated_links = if let Some((links, _)) = links {
        let mut seen: Vec<(&str, usize)> = Vec::new();

        for (index, link) in links.iter().enumerate() {
            let normalized_path = normalize(&link.path);
            let path_valid = normalized_path
                .as_ref()
                .is_ok_and(|normalized| {
                    link.path.starts_with('/')
                        && !link.path.ends_with('/')
                        && link.path != "/"
                        && normalized == &link.path
                });
            let target_valid = validate_raw(&link.target).is_ok();

            if !path_valid || !target_valid {
                return Err(PathError::InvalidLink { index });
            }

            if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
                return Err(PathError::DuplicateLink {
                    first: *first,
                    second: index,
                });
            }

            seen.push((&link.path, index));
        }

        Some(links)
    } else {
        None
    };

    validate_raw(path)?;

    let request_start = parse_request_start(path, home_components.as_deref())?;
    let intent = directory_intent(path);
    let mut stack = match request_start {
        RequestStart::Absolute => Vec::new(),
        RequestStart::Relative => cwd_components,
        RequestStart::Home(_) => home_components.unwrap_or_default(),
    };

    let request_offset = match request_start {
        RequestStart::Home(offset) => offset,
        _ => 0,
    };

    let mut queue: Vec<(String, usize, bool)> = components(path)
        .into_iter()
        .filter(|(_, offset)| *offset >= request_offset)
        .map(|(component, offset)| (component, offset, false))
        .collect();

    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;
    let max_expansions = links.map_or(0, |(_, max)| max);

    let mut cursor = 0usize;
    while cursor < queue.len() {
        let (component, offset, is_target) = queue[cursor].clone();
        cursor += 1;

        match component.as_str() {
            "." => continue,
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else {
                    let absolute = true;
                    if absolute {
                        return Err(PathError::Escape { at: offset });
                    }
                    stack.push("..".to_owned());
                }
            }
            _ => {
                stack.push(component);

                if let Some(links) = validated_links {
                    let matched = {
                        let candidate = format!("/{}", stack.join("/"));
                        links.iter().position(|link| link.path == candidate)
                    };

                    if let Some(index) = matched {
                        if active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if expansions >= max_expansions {
                            return Err(PathError::Limit { index });
                        }
                        expansions += 1;
                        active.push(index);

                        stack.pop();
                        let target = &links[index].target;
                        let target_absolute = target.starts_with('/');
                        if target_absolute {
                            stack.clear();
                        }

                        let target_components: Vec<(String, usize, bool)> = components(target)
                            .into_iter()
                            .map(|(component, offset)| (component, offset, true))
                            .collect();

                        let remaining: Vec<(String, usize, bool)> =
                            queue[cursor..].to_vec();
                        queue.truncate(cursor);
                        queue.extend(target_components);
                        queue.extend(remaining);

                        // The link remains active until its target (including nested
                        // expansions) has finished. A marker records that boundary.
                        queue.insert(cursor, (format!("\0LINK_END:{index}"), 0, true));
                    }
                }
            }
        }

        if let Some(index) = component_link_end(&queue, cursor) {
            if active.last() == Some(&index) {
                active.pop();
            }
        }

        let _ = is_target;
    }

    Ok(render(&stack, true, intent))
}

fn component_link_end(queue: &[(String, usize, bool)], cursor: usize) -> Option<usize> {
    if cursor == 0 {
        return None;
    }

    let (component, _, is_target) = &queue[cursor - 1];
    if !is_target || !component.starts_with("\0LINK_END:") {
        return None;
    }

    component[10..].parse().ok()
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home, None)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home, Some((links, max_expansions)))
}
