#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let bytes = input.as_bytes();
    let mut clean = Vec::with_capacity(bytes.len());
    let mut positions = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut first_retained = true;

    while i < bytes.len() {
        if first_retained && bytes[i] == b'"' {
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
            while i < bytes.len() {
                let b = bytes[i];
                clean.push(b);
                positions.push(i);
                i += 1;
                if b == b'\\' {
                    if i < bytes.len() {
                        clean.push(bytes[i]);
                        positions.push(i);
                        i += 1;
                    }
                } else if b == b'"' {
                    break;
                }
            }
            first_retained = false;
            continue;
        }

        let start = i;
        let mut j = i;
        let mut depth = 0usize;
        let mut outer = None;
        let mut consumed_cfws = false;

        loop {
            if j >= bytes.len() {
                break;
            }

            let b = bytes[j];
            if depth == 0 {
                if is_hws(b) {
                    consumed_cfws = true;
                    j += 1;
                    continue;
                }
                if b == b'\r' || b == b'\n' {
                    consumed_cfws = true;
                    if let Some(end) = is_fold(bytes, j) {
                        j = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: j });
                }
                if b == b'(' {
                    consumed_cfws = true;
                    outer = Some(j);
                    depth = 1;
                    j += 1;
                    continue;
                }
                if b == b')' {
                    return Err(EmailError::UnexpectedClose { at: j });
                }
                break;
            }

            match b {
                b'\\' => {
                    let slash = j;
                    if j + 1 >= bytes.len()
                        || !(0x20..=0x7e).contains(&bytes[j + 1])
                    {
                        return Err(EmailError::CommentEscape { at: slash });
                    }
                    j += 2;
                }
                b'\r' | b'\n' => {
                    if let Some(end) = is_fold(bytes, j) {
                        j = end;
                    } else {
                        return Err(EmailError::BadFold { at: j });
                    }
                }
                b'(' => {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: j });
                    }
                    depth += 1;
                    j += 1;
                }
                b')' => {
                    depth -= 1;
                    j += 1;
                }
                b'\t' => j += 1,
                0x20..=0x7e => j += 1,
                _ => return Err(EmailError::CommentChar { at: j }),
            }
        }

        if depth != 0 {
            return Err(EmailError::UnclosedComment {
                at: outer.unwrap_or(start),
            });
        }

        if consumed_cfws {
            runs.push(Run { start, end: j });
            i = j;
        } else {
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
            first_retained = false;
        }
    }

    Ok((clean, positions, runs))
}

fn mapped(pos: &[usize], input_len: usize, idx: usize) -> usize {
    pos.get(idx).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, pos, runs) = cleanup(input)?;
    for run in runs {
        let left_boundary = run.start == 0;
        let right_boundary = run.end == bytes.len();
        let left_at = run.start > 0 && clean.get(
            clean.iter().take_while(|_| false).count()
        ).is_some();
        let _ = left_at;

        let left_retained = clean.iter().enumerate().find_map(|(idx, _)| {
            (pos[idx] < run.start).then_some(idx)
        });
        let right_retained = clean.iter().enumerate().find_map(|(idx, _)| {
            (pos[idx] >= run.end).then_some(idx)
        });

        let adjacent_at = left_retained
            .and_then(|idx| clean.get(idx).copied())
            .is_some_and(|b| b == b'@')
            || right_retained
                .and_then(|idx| clean.get(idx).copied())
                .is_some_and(|b| b == b'@');

        if !(left_boundary || right_boundary || adjacent_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if clean.first() == Some(&b'"') {
        let mut i = 1usize;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: mapped(&pos, bytes.len(), 0),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0usize;
    while i < clean.len() {
        if let Some(close) = quote_close {
            if i <= close {
                i = close + 1;
                continue;
            }
        }
        if clean[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: mapped(&pos, bytes.len(), separators[1]),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: mapped(&pos, bytes.len(), sep),
        });
    }

    let local_bytes = &clean[..sep];
    let (local, quoted) = if local_bytes.first() == Some(&b'"') {
        let close = quote_close.unwrap_or(0);
        if close + 1 < sep {
            return Err(EmailError::TrailingLocal {
                at: mapped(&pos, bytes.len(), close + 1),
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < close {
            let b = clean[j];
            if b == b'\\' {
                let next = clean.get(j + 1).copied().unwrap_or(0);
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: mapped(&pos, bytes.len(), j),
                    });
                }
                decoded.push(next);
                origins.push(mapped(&pos, bytes.len(), j));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: mapped(&pos, bytes.len(), j),
                    });
                }
                decoded.push(b);
                origins.push(mapped(&pos, bytes.len(), j));
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: mapped(&pos, bytes.len(), close),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: mapped(&pos, bytes.len(), 64),
            });
        }
        for (idx, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if idx == 0 || idx + 1 == local_bytes.len() || local_bytes[idx - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: mapped(&pos, bytes.len(), idx),
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: mapped(&pos, bytes.len(), idx),
                });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: mapped(&pos, bytes.len(), domain_start),
        });
    }
    if clean[domain_end - 1] == b'.' {
        domain_end -= 1;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: mapped(&pos, bytes.len(), domain_start),
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: mapped(&pos, bytes.len(), domain_start + 253),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    let mut k = domain_start;
    while k <= domain_end {
        if k == domain_end || clean[k] == b'.' {
            let label_end = k;
            if label_start == label_end {
                let at = if k == domain_end && domain_end < clean.len() {
                    domain_end
                } else if label_start > domain_start {
                    label_start
                } else {
                    label_start
                };
                return Err(EmailError::EmptyLabel {
                    at: mapped(&pos, bytes.len(), at),
                });
            }

            let len = label_end - label_start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: mapped(&pos, bytes.len(), label_start + 63),
                });
            }
            if clean[label_start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: mapped(&pos, bytes.len(), label_start),
                });
            }
            if clean[label_end - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: mapped(&pos, bytes.len(), label_end - 1),
                });
            }
            for p in label_start..label_end {
                let b = clean[p];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: mapped(&pos, bytes.len(), p),
                    });
                }
            }
            labels.push((label_start, label_end));
            label_start = k + 1;
        }
        k += 1;
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(&pos, bytes.len(), domain_start),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: mapped(&pos, bytes.len(), last_start),
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
