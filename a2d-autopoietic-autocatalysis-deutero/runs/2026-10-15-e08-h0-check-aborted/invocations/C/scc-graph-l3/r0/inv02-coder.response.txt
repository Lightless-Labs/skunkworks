use std::cmp::Reverse;
use std::collections::{BinaryHeap, HashMap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn shortest_route(adjacency: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let mut reverse = vec![Vec::new(); adjacency.len()];
    for (from, tos) in adjacency.iter().enumerate() {
        for &to in tos {
            reverse[to].push(from);
        }
    }

    let mut distance = vec![usize::MAX; adjacency.len()];
    let mut queue = VecDeque::new();
    distance[goal] = 0;
    queue.push_back(goal);

    while let Some(node) = queue.pop_front() {
        for &previous in &reverse[node] {
            if distance[previous] == usize::MAX {
                distance[previous] = distance[node] + 1;
                queue.push_back(previous);
            }
        }
    }

    if distance[start] == usize::MAX {
        return None;
    }

    let mut route = vec![start];
    let mut current = start;
    while current != goal {
        let next = adjacency[current]
            .iter()
            .copied()
            .filter(|&v| distance[v] != usize::MAX && distance[v] + 1 == distance[current])
            .min()?;
        route.push(next);
        current = next;
    }
    Some(route)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];

        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut adjacency = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adjacency[arc.from].push(arc.to);
        }
        for neighbors in &mut adjacency {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        if let Some(route) = shortest_route(&adjacency, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_route(&adjacency, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }

        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (index, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative Kosaraju traversal avoids recursion on long paths.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((node, next_index)) = stack.last_mut() {
            if *next_index < adjacency[*node].len() {
                let next = adjacency[*node][*next_index];
                *next_index += 1;
                if !seen[next] {
                    seen[next] = true;
                    stack.push((next, 0));
                }
            } else {
                finish.push(*node);
                stack.pop();
            }
        }
    }

    seen.fill(false);
    let mut components = Vec::new();
    for &root in finish.iter().rev() {
        if seen[root] {
            continue;
        }
        let mut members = Vec::new();
        let mut stack = vec![root];
        seen[root] = true;
        while let Some(node) = stack.pop() {
            members.push(node);
            for &previous in &reverse[node] {
                if !seen[previous] {
                    seen[previous] = true;
                    stack.push(previous);
                }
            }
        }
        members.sort_unstable();
        components.push(members);
    }

    components.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));
    let mut component_of = vec![0; n];
    for (id, members) in components.iter().enumerate() {
        for &vertex in members {
            component_of[vertex] = id;
        }
    }

    let count = components.len();
    let mut cyclic: Vec<bool> = components.iter().map(|c| c.len() >= 2).collect();
    let mut internal_edges = vec![0usize; count];
    let mut arc_data: HashMap<(usize, usize), (usize, usize)> = HashMap::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, edge_index));
            entry.0 += 1;
            entry.1 = entry.1.min(edge_index);
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = BinaryHeap::new();
    for id in 0..count {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], Reverse(id)));
        }
    }

    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while let Some((_, _, Reverse(node))) = ready.pop() {
        topo.push(node);
        for &next in &outgoing[node] {
            depth[next] = depth[next].max(depth[node] + 1);
            indegree[next] -= 1;
            if indegree[next] == 0 {
                ready.push((internal_edges[next], components[next][0], Reverse(next)));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn validates_precedence_and_endpoint_order() {
        assert_eq!(
            analyze(MAX_VERTICES + 1, &[(usize::MAX, usize::MAX)]),
            Err(GraphError::TooManyVertices {
                count: MAX_VERTICES + 1,
                limit: MAX_VERTICES
            })
        );
        assert_eq!(
            analyze(1, &[(usize::MAX, usize::MAX)]),
            Err(GraphError::BadEndpoint {
                edge: 0,
                side: 0,
                vertex: usize::MAX
            })
        );
        assert_eq!(
            analyze(1, &[(0, usize::MAX)]),
            Err(GraphError::BadEndpoint {
                edge: 0,
                side: 1,
                vertex: usize::MAX
            })
        );
    }

    #[test]
    fn duplicate_edges_count_and_retain_original_witness() {
        let g = analyze(2, &[(0, 0), (0, 1), (0, 1), (1, 1)]).unwrap();
        assert_eq!(g.internal_edges, vec![1, 1]);
        assert_eq!(
            g.arcs,
            vec![Arc {
                from: 0,
                to: 1,
                multiplicity: 2,
                first_edge: 1
            }]
        );
        assert_eq!(g.cyclic, vec![true, true]);
    }

    #[test]
    fn priority_topology_and_longest_depth() {
        let g = analyze(4, &[(0, 2), (1, 2), (2, 3), (1, 1), (0, 0)]).unwrap();
        assert_eq!(g.topo, vec![1, 0, 2, 3]);
        assert_eq!(g.depth, vec![0, 0, 1, 2]);
    }

    #[test]
    fn shortest_route_uses_lexicographic_component_ids() {
        let g = analyze(4, &[(0, 1), (1, 3), (0, 2), (2, 3)]).unwrap();
        assert_eq!(
            g.query(0, 3).unwrap(),
            Query {
                kind: Relation::Forward,
                route: vec![0, 1, 3]
            }
        );
    }

    #[test]
    fn empty_graph_rejects_queries() {
        let g = analyze(0, &[]).unwrap();
        assert_eq!(
            g.query(0, 0),
            Err(GraphError::BadQuery {
                side: 0,
                vertex: 0
            })
        );
        assert!(g.components.is_empty());
        assert!(g.arcs.is_empty());
    }
}

fn main() {}
