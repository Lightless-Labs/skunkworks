#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = None;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if let Some(end) = previous_end {
            if span.start < end {
                return Err(SpanError::Overlap { span: i });
            }
        }
        previous_end = Some(span.end);
    }
    Ok(())
}

fn crc32_update(mut state: u32, data: &[u8]) -> u32 {
    for &byte in data {
        state ^= byte as u32;
        for _ in 0..8 {
            state = if state & 1 != 0 {
                (state >> 1) ^ 0xedb88320
            } else {
                state >> 1
            };
        }
    }
    state
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_update(0xffff_ffff, data) ^ 0xffff_ffff
}

fn adler_update(mut a: u32, mut b: u32, data: &[u8]) -> (u32, u32) {
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (a, b)
}

pub fn adler32(data: &[u8]) -> u32 {
    let (a, b) = adler_update(1, 0, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update(crc, selected);
        (a, b) = adler_update(a, b, selected);
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc_state: u32,
    adler_a: u32,
    adler_b: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_state: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_state ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_state: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc_state = crc32_update(self.crc_state, data);
        (self.adler_a, self.adler_b) =
            adler_update(self.adler_a, self.adler_b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let crc = parse_hex_u32(bytes, 1, 9)?;
        let adler = parse_hex_u32(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for i in start..end {
        let digit = hex_value(bytes[i]).ok_or(SnapshotError::Hex { offset: i })?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc: u16 = if variant == Crc16::False { 0xffff } else { 0 };
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    crc = if crc & 0x8000 != 0 {
                        (crc << 1) ^ 0x1021
                    } else {
                        crc << 1
                    };
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc: u16 = 0xffff;
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    crc = if crc & 1 != 0 {
                        (crc >> 1) ^ 0x8408
                    } else {
                        crc >> 1
                    };
                }
            }
            crc ^ 0xffff
        }
    }
}

fn fletcher32_update(mut a: u32, mut b: u32, data: &[u8], pending: &mut Option<u8>) -> (u32, u32) {
    for &byte in data {
        if let Some(high) = pending.take() {
            let word = ((high as u32) << 8) | byte as u32;
            a = (a + word) % 65535;
            b = (b + a) % 65535;
        } else {
            *pending = Some(byte);
        }
    }
    (a, b)
}

fn fletcher32_finalize(a: u32, b: u32, pending: Option<u8>) -> u32 {
    let (mut a, mut b) = (a, b);
    if let Some(high) = pending {
        let word = (high as u32) << 8;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }
    (b << 16) | a
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut pending = None;
    let (a, b) = fletcher32_update(0, 0, data, &mut pending);
    fletcher32_finalize(a, b, pending)
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut pending = None;
    let (mut a, mut b) = (0, 0);
    for span in spans {
        (a, b) = fletcher32_update(a, b, &data[span.start..span.end], &mut pending);
    }
    Ok(fletcher32_finalize(a, b, pending))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

struct Accumulator {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    f16_a: u32,
    f16_b: u32,
    f32_a: u32,
    f32_b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
    variant: Crc16,
}

impl Accumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            f16_a: 0,
            f16_b: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        self.crc = crc32_update(self.crc, data);
        (self.adler_a, self.adler_b) =
            adler_update(self.adler_a, self.adler_b, data);

        for &byte in data {
            self.f16_a = (self.f16_a + byte as u32) % 255;
            self.f16_b = (self.f16_b + self.f16_a) % 255;

            (self.f32_a, self.f32_b) =
                fletcher32_update(self.f32_a, self.f32_b, &[byte], &mut self.f32_pending);

            match self.variant {
                Crc16::Xmodem | Crc16::False => {
                    self.crc16 ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        self.crc16 = if self.crc16 & 0x8000 != 0 {
                            (self.crc16 << 1) ^ 0x1021
                        } else {
                            self.crc16 << 1
                        };
                    }
                }
                Crc16::X25 => {
                    self.crc16 ^= byte as u16;
                    for _ in 0..8 {
                        self.crc16 = if self.crc16 & 1 != 0 {
                            (self.crc16 >> 1) ^ 0x8408
                        } else {
                            self.crc16 >> 1
                        };
                    }
                }
            }
        }
    }

    fn finish(self) -> Multi {
        Multi {
            base: Hashes {
                crc32: self.crc ^ 0xffff_ffff,
                adler32: (self.adler_b << 16) | self.adler_a,
            },
            fletcher16: ((self.f16_b as u16) << 8) | self.f16_a as u16,
            fletcher32: fletcher32_finalize(self.f32_a, self.f32_b, self.f32_pending),
            crc16: if self.variant == Crc16::X25 {
                self.crc16 ^ 0xffff
            } else {
                self.crc16
            },
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(variant);
    for span in spans {
        acc.update(&data[span.start..span.end]);
    }
    Ok(acc.finish())
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for n in 0..32 {
        square[n] = gf2_matrix_times(matrix, matrix[n]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb88320;
    let mut row = 1u32;
    for n in 1..32 {
        odd[n] = row;
        row <<= 1;
    }

    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }

    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let sums = checksums_spans(data, spans)?;
    let len: u64 = spans.iter().map(|s| (s.end - s.start) as u64).sum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, sums.crc32, sums.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch { field: usize, expected: u32, actual: u32 },
}

fn parse_hex_range(bytes: &[u8], start: usize, end: usize) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for i in start..end {
        let digit = hex_value(bytes[i]).ok_or(VerifyError::Hex { offset: i })?;
        value = (value << 4) | digit as u64;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let len = parse_hex_range(bytes, 4, 20)?;
    let crc = parse_hex_range(bytes, 25, 33)? as u32;
    let adler = parse_hex_range(bytes, 40, 48)? as u32;

    let mut selected_len = 0u64;
    for span in spans {
        selected_len += (span.end - span.start) as u64;
    }
    if len != selected_len {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
