#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<u8>; 5],
    dom_is_exact_star: bool,
    dow_is_exact_star: bool,
}

fn field_info(field: Field) -> (usize, u8, u8, u32) {
    match field {
        Field::Minute => (0, 0, 59, 60),
        Field::Hour => (1, 0, 23, 24),
        Field::DayOfMonth => (2, 1, 31, 31),
        Field::Month => (3, 1, 12, 12),
        Field::DayOfWeek => (4, 0, 6, 7),
    }
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        let digit = (b - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (_, min, max, width) = field_info(field);
    let mut selected = vec![false; 256];
    let mut item_start = 0;

    for item in text.split(',') {
        let start_offset = item_start;
        item_start += item.len() + 1;

        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: start_offset,
            });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        let (base, step_text) = if slash_count == 0 {
            (item, None)
        } else if slash_count == 1 {
            let (a, b) = item.split_once('/').unwrap();
            (a, Some(b))
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: start_offset,
            });
        };

        let base_bytes = base.as_bytes();
        let (base_kind, first_text, second_text) =
            if base == "*" {
                (0, None, None)
            } else if base_bytes.iter().all(u8::is_ascii_digit) && !base.is_empty() {
                (1, Some(base), None)
            } else if base_bytes.iter().filter(|&&b| b == b'-').count() == 1 {
                let (a, b) = base.split_once('-').unwrap();
                if a.is_empty()
                    || b.is_empty()
                    || !a.bytes().all(|x| x.is_ascii_digit())
                    || !b.bytes().all(|x| x.is_ascii_digit())
                {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: start_offset,
                    });
                }
                (2, Some(a), Some(b))
            } else {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: start_offset,
                });
            };

        if let Some(st) = step_text {
            if st.is_empty() || !st.bytes().all(|b| b.is_ascii_digit()) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: start_offset,
                });
            }
        }

        let mut first = min as u32;
        let mut end = max as u32;
        let mut end_number_offset = start_offset;

        if let Some(a) = first_text {
            let a_offset = start_offset + (a.as_ptr() as usize - item.as_ptr() as usize);
            let value = parse_number(a, field, a_offset)?;
            if value < min as u32 || value > max as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: a_offset,
                });
            }
            first = value;
        }

        if let Some(b) = second_text {
            let b_offset = start_offset + (b.as_ptr() as usize - item.as_ptr() as usize);
            let value = parse_number(b, field, b_offset)?;
            if value < min as u32 || value > max as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: b_offset,
                });
            }
            if first > value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: b_offset,
                });
            }
            end = value;
            end_number_offset = b_offset;
        } else if base_kind == 1 && step_text.is_none() {
            end = first;
        }

        let step = if let Some(st) = step_text {
            let step_offset = start_offset + (st.as_ptr() as usize - item.as_ptr() as usize);
            let value = parse_number(st, field, step_offset)?;
            if value == 0 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            value
        } else {
            1
        };

        let _ = end_number_offset;
        let mut value = first;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
    }

    Ok((0..=255)
        .filter(|&n| selected[n])
        .map(|n| n as u8)
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut values: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());

    for (i, field) in fields.into_iter().enumerate() {
        values[i] = parse_field(parts[i], field)?;
    }

    Ok(Cron {
        values,
        dom_is_exact_star: parts[2] == "*",
        dow_is_exact_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.values[field_info(field).0].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.values[0].contains(&at.minute)
            && self.values[1].contains(&at.hour)
            && self.values[2].contains(&at.day)
            && self.values[3].contains(&at.month)
            && self.values[4].contains(&weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

impl PartialEq for Cron {
    fn eq(&self, other: &Self) -> bool {
        self.values == other.values
            && self.dom_is_exact_star == other.dom_is_exact_star
            && self.dow_is_exact_star == other.dow_is_exact_star
    }
}
impl Eq for Cron {}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        1 | 3 | 5 | 7 | 8 | 10 | 12 => 31,
        4 | 6 | 9 | 11 => 30,
        2 if is_leap_year(year) => 29,
        2 => 28,
        _ => 0,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(value: &mut CivilDateTime) -> bool {
    if value.minute < 59 {
        value.minute += 1;
        return true;
    }
    value.minute = 0;
    if value.hour < 23 {
        value.hour += 1;
        return true;
    }
    value.hour = 0;
    if value.day < days_in_month(value.year, value.month) {
        value.day += 1;
        return true;
    }
    value.day = 1;
    if value.month < 12 {
        value.month += 1;
        return true;
    }
    value.month = 1;
    if value.year < 2399 {
        value.year += 1;
        return true;
    }
    false
}
