fn parse_number(text: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        1 | 3 | 5 | 7 | 8 | 10 | 12 => 31,
        4 | 6 | 9 | 11 => 30,
        2 if is_leap_year(year) => 29,
        2 => 28,
        _ => 0,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut elapsed = 0u32;
    for y in 2000..year {
        elapsed += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        elapsed += days_in_month(year, m) as u32;
    }
    elapsed += (day - 1) as u32;
    ((6 + elapsed) % 7) as u8
}

        let stepped = cron("*/20 0 * */3 *");
        assert_eq!(stepped.allowed(Field::Minute), vec![0, 20, 40]);
        assert_eq!(stepped.allowed(Field::Month), vec![1, 4, 7, 10]);
    }

    fn error_precedence_and_offsets() {
        assert_eq!(
            parse_cron("1,,x 0 * * *"),
            Err(ParseError::InvalidCharacter { field: Field::Minute, offset: 3 })
        );
        assert_eq!(
            parse_cron("1,,2 0 * * *"),
            Err(ParseError::EmptyItem { field: Field::Minute, offset: 2 })
        );
        assert_eq!(
            parse_cron("01 0 * * *"),
            Err(ParseError::LeadingZero { field: Field::Minute, offset: 0 })
        );
        assert_eq!(
            parse_cron("0 0 * * 7"),
            Err(ParseError::OutOfRange { field: Field::DayOfWeek, offset: 0 })
        );
    }

    fn matching_and_next_fire() {

        assert!(!q.matches(dt(2026, 5, 4, 9, 31)).unwrap());
        assert_eq!(q.next_fire(dt(2026, 5, 4, 9, 0)).unwrap(), Some(dt(2026, 5, 4, 9, 30)));
