#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::new();
    loop {
        let mut byte = (value & 0x7f) as u8;
        value >>= 7;
        if value != 0 {
            byte |= 0x80;
        }
        output.push(byte);
        if value == 0 {
            return output;
        }
    }
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    for group in 0..10usize {
        let offset = start + group;
        if offset >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[offset];
        let payload = (byte & 0x7f) as u64;

        if group == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow { offset, start });
            }
            value |= payload << 63;
            return Ok((value, offset + 1));
        }

        value |= payload << (group * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::new();
    let mut cursor = start;
    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }
    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let encoded = if value >= 0 {
        (value as u64) << 1
    } else {
        (value.unsigned_abs() << 1).wrapping_sub(1)
    };
    encode_u64(encoded)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, cursor) = decode_u64(input, start)?;
    let decoded = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((decoded, cursor))
}

#[derive(Debug, Clone)]
pub struct StreamDecoder {
    position: usize,
    pending_start: usize,
    pending_len: usize,
    pending_value: u64,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            pending_start: 0,
            pending_len: 0,
            pending_value: 0,
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut position = self.position;
        let mut pending_start = self.pending_start;
        let mut pending_len = self.pending_len;
        let mut pending_value = self.pending_value;
        let mut output = Vec::new();

        for &byte in chunk {
            let offset = position;
            let payload = (byte & 0x7f) as u64;

            if pending_len == 9 {
                if payload > 1 || byte & 0x80 != 0 {
                    return Err(Error::Overflow {
                        offset,
                        start: pending_start,
                    });
                }
                pending_value |= payload << 63;
                position += 1;
                output.push(pending_value);
                pending_len = 0;
                pending_value = 0;
                pending_start = position;
                continue;
            }

            pending_value |= payload << (pending_len * 7);
            position += 1;
            pending_len += 1;

            if byte & 0x80 == 0 {
                output.push(pending_value);
                pending_len = 0;
                pending_value = 0;
                pending_start = position;
            }
        }

        self.position = position;
        self.pending_start = pending_start;
        self.pending_len = pending_len;
        self.pending_value = pending_value;
        Ok(output)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending_len == 0 {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.position - self.pending_len,
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending_len
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending_start = 0;
        self.pending_len = 0;
        self.pending_value = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn unsigned_roundtrip_and_maximum() {
        for value in [0, 2, 17, 126, 129, 12345, u64::MAX] {
            let bytes = encode_u64(value);
            assert_eq!(decode_u64(&bytes, 0), Ok((value, bytes.len())));
        }
        assert_eq!(encode_u64(12345), vec![0xb9, 0x60]);
    }

    #[test]
    fn cursor_errors_and_batches() {
        assert_eq!(
            decode_u64(&[1], 2),
            Err(Error::StartOutOfBounds { offset: 2 })
        );
        assert_eq!(
            decode_u64(&[], 0),
            Err(Error::Truncated { offset: 0, start: 0 })
        );
        assert_eq!(decode_many(&[1, 128, 1, 8], 0, 2), Ok((vec![1, 128], 3)));
        assert_eq!(decode_many(&[], 0, 0), Ok((vec![], 0)));
    }

    #[test]
    fn ten_byte_overflow_and_overlong_values() {
        assert_eq!(decode_u64(&[0x80, 0], 0), Ok((0, 2)));
        assert_eq!(decode_u64(&[0x80; 10], 0), Err(Error::Overflow {
            offset: 9,
            start: 0
        }));
        let mut bytes = vec![0x80; 9];
        bytes.push(1);
        assert_eq!(decode_u64(&bytes, 0), Ok((1u64 << 63, 10)));
    }

    #[test]
    fn signed_extremes() {
        for value in [i64::MIN, -300, -1, 0, 1, 300, i64::MAX] {
            let bytes = encode_i64(value);
            assert_eq!(decode_i64(&bytes, 0), Ok((value, bytes.len())));
        }
        assert_eq!(encode_i64(-1), vec![1]);
    }

    #[test]
    fn stream_atomicity_and_resume() {
        let mut decoder = StreamDecoder::new();
        assert_eq!(decoder.push(&[0x80]), Ok(vec![]));
        assert_eq!(decoder.finish(), Err(Error::Truncated { offset: 1, start: 0 }));
        assert_eq!(decoder.push(&[1, 4]), Ok(vec![128, 4]));
        assert_eq!(decoder.finish(), Ok(()));

        let mut decoder = StreamDecoder::new();
        let mut invalid = vec![7];
        invalid.extend([0x80; 10]);
        assert_eq!(
            decoder.push(&invalid),
            Err(Error::Overflow { offset: 10, start: 1 })
        );
        assert_eq!(decoder.position(), 0);
        assert_eq!(decoder.pending_len(), 0);
        assert_eq!(decoder.push(&[7]), Ok(vec![7]));
    }
}
