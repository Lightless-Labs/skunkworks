pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }
    if n < 100 {
        let tens = TENS[(n / 10 - 2) as usize];
        let unit = n % 10;
        return if unit == 0 {
            tens.to_string()
        } else {
            format!("{}-{}", tens, SMALL[unit as usize])
        };
    }

    let hundreds = format!("{} hundred", SMALL[(n / 100) as usize]);
    let remainder = n % 100;
    if remainder == 0 {
        hundreds
    } else {
        format!("{} and {}", hundreds, under_thousand(remainder))
    }
}

fn encode_magnitude(mut n: u128) -> String {
    if n == 0 {
        return "zero".to_string();
    }

    let mut parts: Vec<String> = Vec::new();
    let mut groups: Vec<(usize, u16)> = Vec::new();

    for i in (0..=6).rev() {
        let divisor = 1000u128.pow(i as u32);
        let group = (n / divisor) as u16;
        n %= divisor;
        if group != 0 {
            groups.push((i, group));
        }
    }

    for (i, group) in &groups {
        let mut part = under_thousand(*group);
        if *i > 0 {
            part.push(' ');
            part.push_str(SCALES[*i - 1]);
        }
        parts.push(part);
    }

    if let Some((last_i, last_group)) = groups.last() {
        if *last_i == 0 && *last_group < 100 && groups.len() > 1 {
            let last = parts.pop().unwrap();
            parts.push(format!("and {}", last));
            return parts.join(" ");
        }
    }

    parts.join(" ")
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if value < 0 {
        Ok(format!("minus {}", encode_magnitude(magnitude)))
    } else {
        Ok(encode_magnitude(magnitude))
    }
}

fn word_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }
    for (i, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(i as u128);
        }
    }
    for (i, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some((20 + i * 10) as u128);
        }
    }
    None
}

fn scale_value(word: &str) -> Option<u128> {
    match word {
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(LIMIT),
        _ => None,
    }
}

fn is_dictionary_word(word: &str, first: bool) -> bool {
    word == "zero"
        || word == "hundred"
        || word == "and"
        || scale_value(word).is_some()
        || word_value(word).is_some()
        || (first && word == "minus")
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, (word, index)) in tokens.iter().enumerate() {
        if !is_dictionary_word(word, i == 0) {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (word, _) in arithmetic_tokens {
        if let Some(value) = word_value(word) {
            group = group.checked_add(value).ok_or(WordsError::Range)?;
        } else if *word == "hundred" {
            group = group.checked_mul(100).ok_or(WordsError::Range)?;
        } else if let Some(scale) = scale_value(word) {
            let scaled = group.checked_mul(scale).ok_or(WordsError::Range)?;
            total = total.checked_add(scaled).ok_or(WordsError::Range)?;
            group = 0;
        }

        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let differing = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(value)
}
