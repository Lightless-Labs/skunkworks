#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
    order: usize,
}

struct Component {
    start: usize,
    end: usize,
    changes: Vec<Change>,
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect()
    }
}

fn edit_changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0usize, 0usize, 0usize);
    let mut changes = Vec::new();
    let mut in_change = false;
    let (mut change_start, mut change_end) = (0usize, 0usize);
    let mut replacement = Vec::new();

    let finish_change = |changes: &mut Vec<Change>,
                         in_change: &mut bool,
                         start: &mut usize,
                         end: &mut usize,
                         replacement: &mut Vec<String>| {
        if *in_change {
            let order = changes.len();
            changes.push(Change {
                start: *start,
                end: *end,
                replacement: std::mem::take(replacement),
                side,
                order,
            });
            *in_change = false;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            finish_change(
                &mut changes,
                &mut in_change,
                &mut change_start,
                &mut change_end,
                &mut replacement,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else {
            let take_delete = if i == n {
                false
            } else if j == m {
                true
            } else {
                let delete_lcs = lcs[i + 1][j];
                let insert_lcs = lcs[i][j + 1];
                if delete_lcs > insert_lcs {
                    true
                } else if insert_lcs > delete_lcs {
                    false
                } else {
                    (i + j) % 2 == 0
                }
            };

            if !in_change {
                in_change = true;
                change_start = base_pos;
                change_end = base_pos;
                replacement.clear();
            }

            if take_delete {
                i += 1;
                base_pos += 1;
                change_end += 1;
            } else {
                replacement.push(new[j].clone());
                j += 1;
            }
        }
    }

    finish_change(
        &mut changes,
        &mut in_change,
        &mut change_start,
        &mut change_end,
        &mut replacement,
    );
    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn make_components(changes: Vec<Change>) -> Vec<Component> {
    let count = changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            let p = parent[x];
            parent[x] = parent[p];
            x = parent[x];
        }
        x
    }

    for i in 0..count {
        for j in i + 1..count {
            if interact(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<Component> = Vec::new();
    let mut roots: Vec<usize> = Vec::new();
    for i in 0..count {
        let r = root(&mut parent, i);
        let index = if let Some(pos) = roots.iter().position(|&x| x == r) {
            pos
        } else {
            roots.push(r);
            groups.push(Component {
                start: changes[i].start,
                end: changes[i].end,
                changes: Vec::new(),
            });
            groups.len() - 1
        };

        let group = &mut groups[index];
        group.start = group.start.min(changes[i].start);
        group.end = group.end.max(changes[i].end);
        group.changes.push(changes[i].clone());
    }

    for group in &mut groups {
        group.changes.sort_by_key(|c| (c.start, c.end, c.order));
    }
    groups.sort_by_key(|c| (c.start, usize::from(c.end > c.start), c.end));
    groups
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut out = Vec::new();
    let mut cursor = start;

    for change in changes {
        if change.start > cursor {
            out.extend(base[cursor..change.start].iter().cloned());
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < end {
        out.extend(base[cursor..end].iter().cloned());
    }
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = edit_changes(&base_lines, &left_lines, 0);
    changes.extend(edit_changes(&base_lines, &right_lines, 1));
    let components = make_components(changes);

    let mut output_lines = Vec::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0usize;

    for component in components {
        if cursor < component.start {
            output_lines.extend(base_lines[cursor..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let mut left_changes: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == 0)
            .cloned()
            .collect();
        let mut right_changes: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == 1)
            .cloned()
            .collect();

        left_changes.sort_by_key(|c| (c.start, c.end, c.order));
        right_changes.sort_by_key(|c| (c.start, c.end, c.order));

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
        );

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines.len();
                    let base_len = component.end - component.start;
                    let marker_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output_lines.push(format!("<<<<<<< LEFT {}", ordinal));
                    output_lines.extend(left_projection.iter().cloned());
                    output_lines.push(format!(
                        "||||||| BASE {},{}",
                        marker_start, base_len
                    ));
                    output_lines.extend(base_projection.iter().cloned());
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection.iter().cloned());
                    output_lines.push(format!(">>>>>>> RIGHT {}", ordinal));

                    let output_len =
                        4 + left_projection.len() + base_projection.len() + right_projection.len();
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        cursor = component.end;
    }

    if cursor < base_lines.len() {
        output_lines.extend(base_lines[cursor..].iter().cloned());
    }

    let mut text = String::new();
    for line in output_lines {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        conflicts: conflicts.len(),
        text,
        regions: conflicts,
    })
}

fn main() {
    let _ = merge_three_way("", "", "", Resolution::Mark);
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn unchanged_input() {
        let result = merge_three_way("a\n", "a\n", "a\n", Resolution::Mark).unwrap();
        assert_eq!(
            result,
            MergeResult {
                text: "a\n".into(),
                conflicts: 0,
                regions: vec![]
            }
        );
    }
}
