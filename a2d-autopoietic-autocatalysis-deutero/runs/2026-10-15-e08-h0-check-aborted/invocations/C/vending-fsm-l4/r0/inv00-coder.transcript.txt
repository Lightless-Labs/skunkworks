        self.log(EventKind::Vended { id: id.clone(), price, change: change_coins.clone() });
        Ok(Receipt { id, price, change: change_coins })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let value = u64::from(self.credit);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.refunded = self.refunded.saturating_add(value);
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.products.clone();
        result.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged { id: id.to_owned(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = stock as u16;
        self.log(EventKind::Restocked { id: id.to_owned(), added: count, stock: stock as u16 });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value(&self.tubes).saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before_value = tube_value(&self.tubes);
        let after_value = tube_value(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let delta = after_value as i128 - before_value as i128;
        self.adjustment = (i128::from(self.adjustment) + delta)
            .clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        let before = self.tubes;
        self.tubes = actual;
        self.log(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let descending = [10usize, 2, 1, 0];
    let mut remaining = u32::from(amount);
    let mut greedy_counts = [0u16; 4];
    for &i in &descending {
        let denom = u32::from(DENOMS[i]);
        let take = (remaining / denom).min(u32::from(tubes[i])) as u16;
        greedy_counts[i] = take;
        remaining -= u32::from(take) * denom;
    }
    if remaining == 0 {
        return Some(expand_counts(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;
    let mut counts = [0u16; 4];
    search_change(0, u32::from(amount), tubes, &mut counts, &mut best, &mut best_count);
    best.map(expand_counts)
}

fn search_change(
    pos: usize,
    remaining: u32,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
    best_count: &mut u32,
) {
    if pos == 4 {
        if remaining == 0 {
            let total: u32 = counts.iter().map(|&n| u32::from(n)).sum();
            if total < *best_count || (total == *best_count && lex_greater(counts, best.as_ref())) {
                *best = Some(*counts);
                *best_count = total;
            }
        }
        return;
    }

    let i = [3usize, 2, 1, 0][pos];
    let denom = u32::from(DENOMS[i]);
    let max_take = (remaining / denom).min(u32::from(tubes[i])) as u16;
    for take in (0..=max_take).rev() {
        counts[i] = take;
        search_change(
            pos + 1,
            remaining - u32::from(take) * denom,
            tubes,
            counts,
            best,
            best_count,
        );
    }
    counts[i] = 0;
}

fn lex_greater(candidate: &[u16; 4], current: Option<&[u16; 4]>) -> bool {
    let Some(current) = current else {
        return true;
    };
    for &i in &[3usize, 2, 1, 0] {
        if candidate[i] != current[i] {
            return candidate[i] > current[i];
        }
    }
    false
}

fn expand_counts(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for &i in &[3usize, 2, 1, 0] {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    result
}
