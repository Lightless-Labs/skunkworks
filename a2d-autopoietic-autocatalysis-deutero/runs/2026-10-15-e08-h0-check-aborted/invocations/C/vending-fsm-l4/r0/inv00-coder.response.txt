#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
    pub maintenance: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
    MaintenanceActive,
    MaintenanceRequired,
    AlreadyMode,
    ReconcileMismatch { coin: u16, expected: u16, found: u16 },
    ValueDrift { before: u64, after: u64 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
    Maintenance { enabled: bool },
    Reconciled { before: [u16; 4], after: [u16; 4] },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Audit {
    pub inserted: u64,
    pub refunded: u64,
    pub sales: u64,
    pub change: u64,
    pub adjustment: i64,
    pub cash: u64,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    !id.is_empty() && id.len() <= 8 && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn tube_value(tubes: &[u16; 4]) -> u64 {
    tubes.iter().zip(DENOMS).map(|(&n, d)| u64::from(n) * u64::from(d)).sum()
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }
        let mut seen = std::collections::HashSet::new();
        for p in &products {
            if !valid_id(&p.id) {
                return Err(Error::BadId { id: p.id.clone() });
            }
            if !seen.insert(p.id.clone()) {
                return Err(Error::DuplicateId { id: p.id.clone() });
            }
            if !(1..=100).contains(&p.price) {
                return Err(Error::BadPrice { id: p.id.clone() });
            }
            if p.stock > 100 {
                return Err(Error::StockLimit { id: p.id.clone() });
            }
        }
        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    fn log(&mut self, kind: EventKind) {
        self.events.push(Event { seq: self.events.len(), kind });
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|p| p.id == id)
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let new_credit = u32::from(self.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(u64::from(coin));
        self.log(EventKind::Inserted { coin, credit: self.credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        let selected_id = self.products[index].id.clone();
        self.selected = Some(selected_id.clone());
        self.log(EventKind::Selected { id: selected_id });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let index = self.product_index(&id).expect("selected product exists");
        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }
        let amount = self.credit - price;
        let change_coins = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for &coin in &change_coins {
            let i = DENOMS.iter().position(|&d| d == coin).unwrap();
            proposed[i] -= 1;
        }
        for &coin in &self.escrow {
            let i = DENOMS.iter().position(|&d| d == coin).unwrap();
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        let id = self.products[index].id.clone();
        self.products[index].stock -= 1;
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        let change_value: u64 = change_coins.iter().map(|&c| u64::from(c)).sum();
        self.change = self.change.saturating_add(change_value);
        self.log(EventKind::Vended { id: id.clone(), price, change: change_coins.clone() });
        Ok(Receipt { id, price, change: change_coins })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let value = u64::from(self.credit);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.refunded = self.refunded.saturating_add(value);
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.products.clone();
        result.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged { id: id.to_owned(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = stock as u16;
        self.log(EventKind::Restocked { id: id.to_owned(), added: count, stock: stock as u16 });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value(&self.tubes).saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before_value = tube_value(&self.tubes);
        let after_value = tube_value(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let delta = after_value as i128 - before_value as i128;
        self.adjustment = (i128::from(self.adjustment) + delta)
            .clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        let before = self.tubes;
        self.tubes = actual;
        self.log(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let descending = [10usize, 2, 1, 0];
    let mut remaining = u32::from(amount);
    let mut greedy_counts = [0u16; 4];
    for &i in &descending {
        let denom = u32::from(DENOMS[i]);
        let take = (remaining / denom).min(u32::from(tubes[i])) as u16;
        greedy_counts[i] = take;
        remaining -= u32::from(take) * denom;
    }
    if remaining == 0 {
        return Some(expand_counts(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;
    let mut counts = [0u16; 4];
    search_change(0, u32::from(amount), tubes, &mut counts, &mut best, &mut best_count);
    best.map(expand_counts)
}

fn search_change(
    pos: usize,
    remaining: u32,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
    best_count: &mut u32,
) {
    if pos == 4 {
        if remaining == 0 {
            let total: u32 = counts.iter().map(|&n| u32::from(n)).sum();
            if total < *best_count || (total == *best_count && lex_greater(counts, best.as_ref())) {
                *best = Some(*counts);
                *best_count = total;
            }
        }
        return;
    }

    let i = [3usize, 2, 1, 0][pos];
    let denom = u32::from(DENOMS[i]);
    let max_take = (remaining / denom).min(u32::from(tubes[i])) as u16;
    for take in (0..=max_take).rev() {
        counts[i] = take;
        search_change(
            pos + 1,
            remaining - u32::from(take) * denom,
            tubes,
            counts,
            best,
            best_count,
        );
    }
    counts[i] = 0;
}

fn lex_greater(candidate: &[u16; 4], current: Option<&[u16; 4]>) -> bool {
    let Some(current) = current else {
        return true;
    };
    for &i in &[3usize, 2, 1, 0] {
        if candidate[i] != current[i] {
            return candidate[i] > current[i];
        }
    }
    false
}

fn expand_counts(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for &i in &[3usize, 2, 1, 0] {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    result
}
