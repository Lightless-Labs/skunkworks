```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: HashMap<String, u16>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut catalog = HashMap::new();
        for product in products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            catalog.insert(product.id, product.price);
        }

        let denominations = [1, 3, 4, 10];
        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: denominations[i],
                });
            }
        }

        Ok(Self {
            products: catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = match coin {
            1 => 0,
            3 => 1,
            4 => 2,
            10 => 3,
            _ => return Err(Error::BadCoin { coin }),
        };

        let _ = index;
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(_) = self.products.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let id = id.clone();
        let price = self.products[&id];
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let coins = [1u16, 3, 4, 10];
        let mut remaining = amount;
        let mut change_counts = [0u16; 4];

        for i in (0..4).rev() {
            let take = self.tubes[i].min(remaining / coins[i]);
            change_counts[i] = take;
            remaining -= take * coins[i];
        }

        if remaining != 0 {
            let mut best: Option<[u16; 4]> = None;
            let mut best_total = u32::MAX;

            fn search(
                idx: i32,
                remaining: u16,
                coins: &[u16; 4],
                tubes: &[u16; 4],
                counts: &mut [u16; 4],
                best: &mut Option<[u16; 4]>,
                best_total: &mut u32,
            ) {
                if idx < 0 {
                    if remaining == 0 {
                        let total: u32 = counts.iter().map(|&x| u32::from(x)).sum();
                        let better = total < *best_total
                            || (total == *best_total
                                && best.as_ref().map_or(true, |old| {
                                    for i in (0..4).rev() {
                                        if counts[i] != old[i] {
                                            return counts[i] > old[i];
                                        }
                                    }
                                    false
                                }));
                        if better {
                            *best_total = total;
                            *best = Some(*counts);
                        }
                    }
                    return;
                }

                let i = idx as usize;
                let max_take = tubes[i].min(remaining / coins[i]);
                for take in 0..=max_take {
                    counts[i] = take;
                    search(
                        idx - 1,
                        remaining - take * coins[i],
                        coins,
                        tubes,
                        counts,
                        best,
                        best_total,
                    );
                }
                counts[i] = 0;
            }

            search(
                3,
                amount,
                &coins,
                &self.tubes,
                &mut [0; 4],
                &mut best,
                &mut best_total,
            );

            let Some(found) = best else {
                return Err(Error::NoChange { amount });
            };
            change_counts = found;
        }

        let mut proposed_tubes = self.tubes;
        for i in 0..4 {
            proposed_tubes[i] -= change_counts[i];
        }
        for &coin in &self.escrow {
            let index = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed_tubes[index] += 1;
        }

        for i in 0..4 {
            if proposed_tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: coins[i] });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change_counts[i] {
                change.push(coins[i]);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(returned)
    }
}
```
