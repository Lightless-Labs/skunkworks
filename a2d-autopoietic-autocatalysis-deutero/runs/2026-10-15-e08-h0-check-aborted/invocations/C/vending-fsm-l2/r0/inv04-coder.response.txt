use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

#[derive(Debug)]
pub struct Machine {
    products: HashMap<String, u16>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut catalog = HashMap::new();
        for product in products {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|byte| byte.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            catalog.insert(product.id, product.price);
        }

        let denominations = [1, 3, 4, 10];
        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: denominations[index],
                });
            }
        }

        Ok(Self {
            products: catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        match coin {
            1 | 3 | 4 | 10 => {}
            _ => return Err(Error::BadCoin { coin }),
        }

        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let id = selected_id.clone();
        let price = self.products[&id];

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let denominations = [1u16, 3, 4, 10];
        let mut remaining = amount;
        let mut change_counts = [0u16; 4];

        for index in (0..4).rev() {
            let take = self.tubes[index].min(remaining / denominations[index]);
            change_counts[index] = take;
            remaining -= take * denominations[index];
        }

        if remaining != 0 {
            fn search(
                index: i32,
                remaining: u16,
                denominations: &[u16; 4],
                tubes: &[u16; 4],
                counts: &mut [u16; 4],
                best: &mut Option<[u16; 4]>,
                best_total: &mut u32,
            ) {
                if index < 0 {
                    if remaining == 0 {
                        let total: u32 = counts.iter().map(|&count| u32::from(count)).sum();
                        let better = total < *best_total
                            || (total == *best_total
                                && best.as_ref().is_none_or(|old| {
                                    for i in (0..4).rev() {
                                        if counts[i] != old[i] {
                                            return counts[i] > old[i];
                                        }
                                    }
                                    false
                                }));
                        if better {
                            *best_total = total;
                            *best = Some(*counts);
                        }
                    }
                    return;
                }

                let i = index as usize;
                let max_take = tubes[i].min(remaining / denominations[i]);
                for take in 0..=max_take {
                    counts[i] = take;
                    search(
                        index - 1,
                        remaining - take * denominations[i],
                        denominations,
                        tubes,
                        counts,
                        best,
                        best_total,
                    );
                }
                counts[i] = 0;
            }

            let mut best = None;
            let mut best_total = u32::MAX;
            search(
                3,
                amount,
                &denominations,
                &self.tubes,
                &mut [0; 4],
                &mut best,
                &mut best_total,
            );

            let Some(found) = best else {
                return Err(Error::NoChange { amount });
            };
            change_counts = found;
        }

        let mut proposed_tubes = self.tubes;
        for index in 0..4 {
            proposed_tubes[index] -= change_counts[index];
        }
        for &coin in &self.escrow {
            let index = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed_tubes[index] += 1;
        }

        for index in 0..4 {
            if proposed_tubes[index] > 100 {
                return Err(Error::TubeLimit {
                    coin: denominations[index],
                });
            }
        }

        let mut change = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..change_counts[index] {
                change.push(denominations[index]);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(returned)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn product(id: &str, price: u16) -> Product {
        Product {
            id: id.to_owned(),
            price,
        }
    }

    fn machine(price: u16, tubes: [u16; 4]) -> Machine {
        Machine::new(vec![product("tea", price), product("juice", 7)], tubes).unwrap()
    }

    #[test]
    fn exact_sale_and_snapshot_are_owned() {
        let mut machine = machine(6, [20, 10, 10, 10]);
        let mut snapshot = machine.state();
        snapshot.credit = 99;
        machine.select("tea").unwrap();
        machine.insert(3).unwrap();
        machine.insert(3).unwrap();
        assert_eq!(
            machine.vend().unwrap(),
            Receipt {
                id: "tea".into(),
                price: 6,
                change: vec![]
            }
        );
        assert_eq!(machine.state().credit, 0);
        assert_eq!(machine.state().selected, None);
    }

    #[test]
    fn refund_returns_coins_in_reverse_order() {
        let mut machine = machine(6, [20, 10, 10, 10]);
        machine.select("juice").unwrap();
        machine.insert(1).unwrap();
        machine.insert(4).unwrap();
        assert_eq!(machine.refund().unwrap(), vec![4, 1]);
        assert_eq!(machine.state().escrow, Vec::<u16>::new());
    }

    #[test]
    fn greedy_change_is_used() {
        let mut machine = machine(6, [20, 10, 10, 10]);
        machine.insert(10).unwrap();
        machine.select("tea").unwrap();
        assert_eq!(machine.vend().unwrap().change, vec![4]);
        assert_eq!(machine.state().tubes, [20, 10, 9, 11]);
    }

    #[test]
    fn validation_precedence_and_atomicity() {
        let err = Machine::new(
            vec![product("tea", 0), product("tea", 0)],
            [101, 0, 0, 0],
        )
        .unwrap_err();
        assert_eq!(err, Error::BadPrice { id: "tea".into() });

        let mut machine = machine(6, [0, 0, 0, 0]);
        machine.select("tea").unwrap();
        assert_eq!(
            machine.vend(),
            Err(Error::Underpaid {
                price: 6,
                credit: 0
            })
        );
        assert_eq!(machine.state().selected.as_deref(), Some("tea"));
    }
}

fn main() {}
