#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut index = start;
    let mut record_index = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut field_started = false;
    let mut opening_quote = 0usize;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        if !in_quotes && byte == b'\r' && bytes.get(index + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: index,
            });
        }

        if in_quotes {
            if byte == quote {
                if bytes.get(index + 1) == Some(&quote) {
                    field.push(quote as char);
                    index += 2;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                index += 1;
                continue;
            }

            let ch = input[index..].chars().next().expect("valid UTF-8 boundary");
            field.push(ch);
            index += ch.len_utf8();
            continue;
        }

        if after_quote {
            if byte == delimiter {
                record.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                index += 1;
                continue;
            }
            if byte == b'\n' || byte == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                field_started = false;
                after_quote = false;
                index += if byte == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: index,
            });
        }

        if byte == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            index += 1;
            continue;
        }

        if byte == b'\n' || byte == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            field_started = false;
            index += if byte == b'\r' { 2 } else { 1 };
            continue;
        }

        if byte == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: index,
                });
            }
            in_quotes = true;
            opening_quote = index;
            field_started = true;
            index += 1;
            continue;
        }

        let ch = input[index..].chars().next().expect("valid UTF-8 boundary");
        field.push(ch);
        field_started = true;
        index += ch.len_utf8();
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if after_quote || field_started || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();

    if !config.headers {
        return Ok(Table {
            header: None,
            rows: iter.collect(),
            duplicates: config.duplicates,
        });
    }

    let Some(header) = iter.next() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen = std::collections::HashMap::<String, usize>::new();
        for (column, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second: column,
                });
            }
            seen.insert(name.clone(), column);
        }
    }

    let expected = header.len();
    let mut rows = Vec::new();
    for (offset, row) in iter.enumerate() {
        if row.len() != expected {
            return Err(CsvError::HeaderWidth {
                record: offset + 1,
                expected,
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }

        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn main() {
    let _ = parse_csv("");
}

#[cfg(test)]
mod tests {
    use super::*;

    fn strings(values: &[&str]) -> Vec<String> {
        values.iter().map(|s| s.to_string()).collect()
    }

    fn cfg(headers: bool, duplicates: DuplicatePolicy) -> Config {
        Config {
            delimiter: b',',
            quote: b'"',
            headers,
            duplicates,
        }
    }

    #[test]
    fn parses_quotes_and_streams() {
        assert_eq!(
            parse_csv("\"red,blue\",\"say \"\"hi\"\"\""),
            Ok(vec![strings(&["red,blue", "say \"hi\""])])
        );

        let mut reader = StreamReader::new(cfg(false, DuplicatePolicy::KeepFirst)).unwrap();
        reader.feed("left,").unwrap();
        reader.feed("\"right\"\r").unwrap();
        reader.feed("\n").unwrap();
        assert_eq!(reader.finish().unwrap().rows, vec![strings(&["left", "right"])]);
    }
}
