```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(index) = path.bytes().position(|byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            components.extend(self.cwd.split('/').filter(|part| !part.is_empty()));
        }

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                _ => components.push(part),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn parent_path(path: &str) -> &str {
        match path.rfind('/') {
            Some(0) | None => "/",
            Some(index) => &path[..index],
        }
    }

    fn lookup_parent(&self, path: &str) -> Result<(), Error> {
        let mut prefix = String::new();
        let components: Vec<&str> = path.split('/').filter(|part| !part.is_empty()).collect();

        for component in components.iter().take(components.len().saturating_sub(1)) {
            prefix.push('/');
            prefix.push_str(component);

            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn lookup_final(&self, path: &str) -> Result<Option<&Entry>, Error> {
        if path == "/" {
            return Ok(self.entries.get("/"));
        }

        self.lookup_parent(path)?;
        match self.entries.get(path) {
            Some(entry) => Ok(Some(entry)),
            None => Ok(None),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Exists(canonical));
        }

        self.lookup_parent(&canonical)?;
        let parent = Self::parent_path(&canonical);
        match self.entries.get(parent) {
            None => return Err(Error::Missing(parent.to_string())),
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent.to_string())),
        }

        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::IsDir(canonical));
        }

        self.lookup_parent(&canonical)?;
        let parent = Self::parent_path(&canonical);
        match self.entries.get(parent) {
            None => return Err(Error::Missing(parent.to_string())),
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent.to_string())),
        }

        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_final(&canonical)? {
            None => Err(Error::Missing(canonical)),
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_final(&canonical)? {
            None => Err(Error::Missing(canonical)),
            Some(Entry::File(_)) => Err(Error::NotDir(canonical)),
            Some(Entry::Dir) => {
                let prefix = if canonical == "/" {
                    "/".to_string()
                } else {
                    format!("{}/", canonical)
                };
                let mut children = Vec::new();

                for (entry_path, entry) in &self.entries {
                    if entry_path == &canonical || !entry_path.starts_with(&prefix) {
                        continue;
                    }
                    let remainder = &entry_path[prefix.len()..];
                    if remainder.is_empty() || remainder.contains('/') {
                        continue;
                    }

                    let kind = match entry {
                        Entry::Dir => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    children.push((remainder.to_string(), kind));
                }

                children.sort_by(|left, right| left.0.cmp(&right.0));
                Ok(children)
            }
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_final(&canonical)? {
            None => Err(Error::Missing(canonical)),
            Some(Entry::File(_)) => Err(Error::NotDir(canonical)),
            Some(Entry::Dir) => {
                self.cwd = canonical;
                Ok(())
            }
        }
    }
}
```
