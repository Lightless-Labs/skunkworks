#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized = input.replace("\r\n", "\n").replace('\r', "\n");
    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let mut end = i + 1;
            while end < chars.len() && chars[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                wrap_segment(&chars[segment_start..i], options, &mut output);
                for _ in 1..(end - i) {
                    output.push(String::new());
                }
                segment_start = end;
            }
            i = end;
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00A0}'
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }

    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut line_content = String::new();
    let mut content_length = 0usize;

    for word in words {
        let mut suffix: Vec<char> = word;
        while !suffix.is_empty() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;

            if suffix.len() <= capacity {
                let needed = suffix.len() + usize::from(content_length > 0);
                if content_length + needed <= capacity {
                    if content_length > 0 {
                        line_content.push(' ');
                        content_length += 1;
                    }
                    for ch in suffix {
                        line_content.push(ch);
                        content_length += 1;
                    }
                    suffix = Vec::new();
                } else {
                    emit_line(
                        &mut line_content,
                        indent,
                        output,
                    );
                    content_length = 0;
                    first_line = false;
                }
            } else if content_length == 0 {
                for &ch in suffix.iter().take(capacity) {
                    line_content.push(ch);
                }
                suffix.drain(..capacity);
                emit_line(&mut line_content, indent, output);
                content_length = 0;
                first_line = false;
            } else {
                let available = capacity.saturating_sub(content_length + 1);
                if available >= 2 {
                    line_content.push(' ');
                    content_length += 1;
                    for &ch in suffix.iter().take(available) {
                        line_content.push(ch);
                    }
                    suffix.drain(..available);
                    emit_line(&mut line_content, indent, output);
                    content_length = 0;
                    first_line = false;
                } else {
                    emit_line(&mut line_content, indent, output);
                    content_length = 0;
                    first_line = false;
                }
            }
        }
    }

    if content_length > 0 {
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        emit_line(&mut line_content, indent, output);
    }
}

fn emit_line(content: &mut String, indent: usize, output: &mut Vec<String>) {
    let mut line = String::with_capacity(indent + content.len());
    line.push_str(&" ".repeat(indent));
    line.push_str(content);
    output.push(line);
    content.clear();
}
