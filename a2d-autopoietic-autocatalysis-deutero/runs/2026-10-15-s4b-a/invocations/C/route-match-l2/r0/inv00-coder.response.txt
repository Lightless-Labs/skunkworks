#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = rank_for(&segments);
        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;
        for (route_id, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(&route.segments, &request) {
                let replace = match &best {
                    None => true,
                    Some((best_id, _)) => {
                        route.rank > self.routes[*best_id].rank
                            || (route.rank == self.routes[*best_id].rank && route_id < *best_id)
                    }
                };
                if replace {
                    best = Some((route_id, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn rank_for(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let (parts, offsets) = split_segments(input);
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (i, (part, start)) in parts.iter().zip(offsets.iter()).enumerate() {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        if let Some(name) = part.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = part.strip_prefix('*') {
            if i + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            if !valid_name(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            segments.push(Segment::Static(decode_segment(part, *start, true)?));
        }
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }
    Ok(segments)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let (parts, offsets) = split_segments(input);
    let mut decoded = Vec::with_capacity(parts.len());
    for (part, start) in parts.iter().zip(offsets.iter()) {
        if part.is_empty() {
            return Err(Error::InvalidPath { at: *start });
        }
        decoded.push(decode_segment(part, *start, false)?);
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }
    Ok(decoded)
}

fn split_segments(input: &str) -> (Vec<&str>, Vec<usize>) {
    let mut parts = Vec::new();
    let mut offsets = Vec::new();
    let mut start = 1;
    for (i, byte) in input.bytes().enumerate().skip(1) {
        if byte == b'/' {
            parts.push(&input[start..i]);
            offsets.push(start);
            start = i + 1;
        }
    }
    parts.push(&input[start..]);
    offsets.push(start);
    (parts, offsets)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(
    segment: &str,
    offset: usize,
    pattern: bool,
) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return invalid(pattern, offset + i);
            }
            let Some(hi) = upper_hex(bytes[i + 1]) else {
                return invalid(pattern, offset + i);
            };
            let Some(lo) = upper_hex(bytes[i + 2]) else {
                return invalid(pattern, offset + i);
            };
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return invalid(pattern, offset + i);
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return invalid(pattern, offset + i);
        }
    }

    // Decoding only permits ASCII bytes.
    Ok(String::from_utf8(out).expect("decoded segment is ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn invalid<T>(pattern: bool, at: usize) -> Result<T, Error> {
    if pattern {
        Err(Error::InvalidPattern { at })
    } else {
        Err(Error::InvalidPath { at })
    }
}

fn match_route(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut ri = 0;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if request.get(ri)? != value {
                    return None;
                }
                ri += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(ri)?.clone()));
                ri += 1;
            }
            Segment::Wildcard(name) => {
                let value = request[ri..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    if ri == request.len() {
        Some(params)
    } else {
        None
    }
}

fn main() {}
