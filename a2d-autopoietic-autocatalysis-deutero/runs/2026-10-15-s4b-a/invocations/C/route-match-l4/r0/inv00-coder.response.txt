use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str, base: usize, pattern: bool) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(h) = hex_upper(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(l) = hex_upper(bytes[i + 2]) else {
                return Err(base + i);
            };
            let value = (h << 4) | l;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(base + i);
            }
            out.push(value);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            let _ = pattern;
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, is_pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let body_end = if trailing { input.len() - 1 } else { input.len() };
    let body = &input[1..body_end];
    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let parts: Vec<(usize, &str)> = body
        .split('/')
        .scan(1usize, |pos, part| {
            let start = *pos;
            *pos += part.len() + 1;
            Some((start, part))
        })
        .collect();

    for (idx, (start, raw)) in parts.iter().enumerate() {
        if raw.is_empty() {
            return Err(*start);
        }
        if is_pattern && raw.starts_with('*') {
            let name = &raw[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(*start);
            }
            if idx + 1 != parts.len() {
                return Err(*start);
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else if is_pattern && raw.starts_with(':') {
            let name = &raw[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(*start);
            }
            segments.push(Segment::Param(name.to_owned()));
        } else {
            let decoded = decode_segment(raw, *start, is_pattern)
                .map_err(|at| at)?;
            segments.push(Segment::Static(decoded));
        }
    }

    let _ = bytes;
    Ok((segments, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;

        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method != method || route.segments.len() != segments.len() {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            let same_shape = route
                .segments
                .iter()
                .zip(&segments)
                .all(|(a, b)| match (a, b) {
                    (Segment::Static(x), Segment::Static(y)) => x == y,
                    (Segment::Param(_), Segment::Param(_))
                    | (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                    _ => false,
                });
            if same_shape {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) =
            parse_path(head, false).map_err(|at| Error::InvalidPath { at })?;

        let mut path_candidates: Vec<(&Route, Vec<(String, String)>, Vec<u8>)> = Vec::new();
        for route in &self.routes {
            let wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            let fixed_count = if wildcard {
                route.segments.len() - 1
            } else {
                route.segments.len()
            };
            if (!wildcard && request_segments.len() != fixed_count)
                || (wildcard && request_segments.len() < fixed_count)
            {
                continue;
            }

            let mut captures = Vec::new();
            let mut rank = Vec::new();
            let mut matched = true;
            for i in 0..fixed_count {
                match (&route.segments[i], &request_segments[i]) {
                    (Segment::Static(a), Segment::Static(b)) if a == b => rank.push(2),
                    (Segment::Param(name), Segment::Static(value)) => {
                        rank.push(1);
                        captures.push((name.clone(), value.clone()));
                    }
                    _ => {
                        matched = false;
                        break;
                    }
                }
            }
            if !matched {
                continue;
            }
            if wildcard {
                if let Segment::Wildcard(name) = &route.segments[fixed_count] {
                    let tail = request_segments[fixed_count..]
                        .iter()
                        .map(|s| match s {
                            Segment::Static(v) => v.as_str(),
                            _ => "",
                        })
                        .collect::<Vec<_>>()
                        .join("/");
                    rank.push(0);
                    captures.push((name.clone(), tail));
                }
            } else {
                rank.push(3);
            }
            path_candidates.push((route, captures, rank));
        }

        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && path_candidates.iter().any(|(r, _, _)| r.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = path_candidates
            .iter()
            .filter(|(r, _, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = path_candidates
                .iter()
                .map(|(r, _, _)| r.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let winner = eligible
            .into_iter()
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.0.id.cmp(&a.0.id)))
            .expect("eligible candidates are nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && winner.0.trailing != request_trailing
        {
            if winner.0.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.0.id,
            params: winner.1.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|r| r.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = HashSet::new();
        for (key, _) in params {
            if !supplied.insert(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let captures: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|s| match s {
                Segment::Param(n) | Segment::Wildcard(n) => Some(n.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &captures {
            if !params.iter().any(|(k, _)| k == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extras: Vec<&str> = params
            .iter()
            .map(|(k, _)| *k)
            .filter(|k| !captures.contains(k))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for key in &captures {
            let value = params.iter().find(|(k, _)| k == key).unwrap().1;
            let valid_ascii = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
            let is_wild = route
                .segments
                .iter()
                .any(|s| matches!(s, Segment::Wildcard(n) if n == key));
            let valid_shape = if is_wild {
                value.is_empty()
                    || value
                        .split('/')
                        .all(|part| !part.is_empty())
            } else {
                !value.is_empty() && !value.contains('/')
            };
            if !valid_ascii || !valid_shape {
                return Err(Error::InvalidValue {
                    name: (*key).to_owned(),
                });
            }
        }

        fn encode_component(s: &str) -> String {
            let mut out = String::new();
            for b in s.bytes() {
                if unreserved(b) {
                    out.push(b as char);
                } else {
                    out.push('%');
                    out.push(b"0123456789ABCDEF"[(b >> 4) as usize] as char);
                    out.push(b"0123456789ABCDEF"[(b & 15) as usize] as char);
                }
            }
            out
        }

        if route.segments.is_empty() {
            return Ok("/".to_owned());
        }

        let mut out = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if !first {
                        out.push('/');
                    }
                    out.push_str(&encode_component(value));
                    first = false;
                }
                Segment::Param(key) => {
                    if !first {
                        out.push('/');
                    }
                    let value = params.iter().find(|(k, _)| *k == key).unwrap().1;
                    out.push_str(&encode_component(value));
                    first = false;
                }
                Segment::Wildcard(key) => {
                    let value = params.iter().find(|(k, _)| *k == key).unwrap().1;
                    if !value.is_empty() {
                        if !first {
                            out.push('/');
                        }
                        let mut component_first = true;
                        for component in value.split('/') {
                            if !component_first {
                                out.push('/');
                            }
                            out.push_str(&encode_component(component));
                            component_first = false;
                        }
                    }
                    first = false;
                }
            }
        }

        if route.trailing && !out.ends_with('/') {
            out.push('/');
        }
        Ok(out)
    }
}

fn main() {}
