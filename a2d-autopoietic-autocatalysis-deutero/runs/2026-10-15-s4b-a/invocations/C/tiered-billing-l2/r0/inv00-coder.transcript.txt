fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index != tiers.len() - 1 {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().unwrap().up_to.is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn band_charge(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    let product = units.checked_mul(unit).ok_or(Error::Overflow { index })?;
    product.checked_add(start).ok_or(Error::Overflow { index })
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut amount = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                previous = bound;
                remaining.min(width)
            }
            None => remaining,
        };

        if units > 0 {
            let charge = band_charge(units, tier.unit, tier.start, index)?;
            amount = amount.checked_add(charge).ok_or(Error::Overflow { index })?;
            bands.push(Band {
                index,
                units,
                amount: charge,
            });
            remaining -= units;
        }
    }

    Ok(Charge { amount, bands })
}

fn volume(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let index = tiers
        .iter()
        .position(|tier| tier.up_to.map_or(true, |bound| bound >= quantity))
        .unwrap();
    let tier = &tiers[index];
    let amount = band_charge(quantity, tier.unit, tier.start, index)?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index,
            units: quantity,
            amount,
        }],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                })
            } else {
                let amount = band_charge(quantity, *unit, *start, 0)?;
                Ok(Charge {
                    amount,
                    bands: vec![Band {
                        index: 0,
                        units: quantity,
                        amount,
                    }],
                })
            }
        }
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match price {
        Price::Flat { .. } => quote(quantity, price)?,
        Price::Tiers(tiers) => match mode {
            Mode::Graduated => graduated(quantity, tiers)?,
            Mode::Volume => volume(quantity, tiers)?,
        },
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::Overflow {
            index: charge.bands.last().map_or(0, |band| band.index),
        })?;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
