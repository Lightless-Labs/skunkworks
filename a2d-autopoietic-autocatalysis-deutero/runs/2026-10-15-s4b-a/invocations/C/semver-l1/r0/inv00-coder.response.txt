use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_field(bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }

    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &b in &bytes[start..*pos] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    base: usize,
    allow_numeric_leading_zero: bool,
    stop_at_plus: bool,
) -> Result<(Vec<String>, usize), VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;

    loop {
        while i < bytes.len()
            && bytes[i] != b'.'
            && !(stop_at_plus && bytes[i] == b'+')
        {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: base + i });
            }
            i += 1;
        }

        if i == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }

        let part = &s[start..i];
        if !allow_numeric_leading_zero
            && part.bytes().all(is_digit)
            && part.len() > 1
            && part.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        ids.push(part.to_owned());

        if i == bytes.len() || (stop_at_plus && bytes[i] == b'+') {
            return Ok((ids, i));
        }

        // The only remaining delimiter is a dot.
        i += 1;
        start = i;
    }
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;

    let major = parse_core_field(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core_field(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core_field(bytes, &mut pos)?;
    if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let start = pos + 1;
        let tail = &s[start..];
        let (ids, consumed) = parse_identifiers(tail, start, false, true)?;
        pre = ids;
        pos = start + consumed;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let start = pos + 1;
        let tail = &s[start..];
        let (ids, _) = parse_identifiers(tail, start, true, false)?;
        build = ids;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core_cmp(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch))
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_numeric_strings(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = core_cmp(a, b);
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (ai, bi) in a.pre.iter().zip(&b.pre) {
        let an = is_numeric_id(ai);
        let bn = is_numeric_id(bi);
        let cmp = match (an, bn) {
            (true, true) => compare_numeric_strings(ai, bi),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => ai.as_bytes().cmp(bi.as_bytes()),
        };
        if cmp != Ordering::Equal {
            return cmp;
        }
    }

    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut i = 0usize;
    let mut comparators = Vec::new();

    while i < bytes.len() {
        while i < bytes.len() && is_range_space(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let token_start = i;
        while i < bytes.len() && !is_range_space(bytes[i]) {
            i += 1;
        }
        let token_end = i;
        let token = &s[token_start..token_end];

        let (op, op_len) = if token.starts_with(">=") {
            (Op::Ge, 2)
        } else if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with('>') {
            (Op::Gt, 1)
        } else if token.starts_with('<') {
            (Op::Lt, 1)
        } else if token.starts_with('=') {
            (Op::Eq, 1)
        } else {
            (Op::Eq, 0)
        };

        let operand_start = token_start + op_len;
        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !arm.iter().all(|c| {
            let cmp = compare_versions(v, &c.version);
            match c.op {
                Op::Eq => cmp == Ordering::Equal,
                Op::Lt => cmp == Ordering::Less,
                Op::Le => cmp != Ordering::Greater,
                Op::Gt => cmp == Ordering::Greater,
                Op::Ge => cmp != Ordering::Less,
            }
        }) {
            return false;
        }

        if v.pre.is_empty() {
            return true;
        }

        arm.iter().any(|c| {
            !c.version.pre.is_empty() && core_cmp(v, &c.version) == Ordering::Equal
        })
    })
}

fn main() {}
