#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn find_braced_end(input: &str, start: usize) -> Option<usize> {
    let mut iter = input[start..].char_indices().peekable();

    while let Some((offset, c)) = iter.next() {
        if c == '\\' {
            iter.next();
        } else if c == '}' {
            return Some(start + offset);
        }
    }

    None
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let Some(close) = find_braced_end(input, content_start) else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let content = &input[content_start..close];
                        let (name, default) = if let Some(index) = content.find(":-") {
                            (&content[..index], Some(&content[index + 2..]))
                        } else {
                            (content, None)
                        };

                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if default.is_none() && content.contains(':') {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }

                        match (env_value(env, name), default) {
                            (Some(value), None) => word.push_str(value),
                            (Some(value), Some(_)) if !value.is_empty() => word.push_str(value),
                            (_, Some(text)) => word.push_str(&decode_default(text)),
                            (None, None) => {}
                        }
                        i = close + 1;
                    } else if next_i < input.len()
                        && input[next_i..].chars().next().is_some_and(is_name_start)
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_i..end];
                        if let Some(value) = env_value(env, name) {
                            word.push_str(value);
                        }
                        i = end;
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' || c == '"' {
                    active = true;
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_start = i;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        word.push(next);
                        i = after_next;
                    }
                } else if c == '$' {
                    active = true;
                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let Some(close) = find_braced_end(input, content_start) else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let content = &input[content_start..close];
                        let (name, default) = if let Some(index) = content.find(":-") {
                            (&content[..index], Some(&content[index + 2..]))
                        } else {
                            (content, None)
                        };

                        let valid_name = !name.is_empty()
                            && name.chars().next().is_some_and(is_name_start)
                            && name.chars().all(is_name_continue);
                        if !valid_name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if default.is_none() && content.contains(':') {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }

                        match (env_value(env, name), default) {
                            (Some(value), None) => word.push_str(value),
                            (Some(value), Some(_)) if !value.is_empty() => word.push_str(value),
                            (_, Some(text)) => word.push_str(&decode_default(text)),
                            (None, None) => {}
                        }
                        i = close + 1;
                    } else if next_i < input.len()
                        && input[next_i..].chars().next().is_some_and(is_name_start)
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_i..end];
                        if let Some(value) = env_value(env, name) {
                            word.push_str(value);
                        }
                        i = end;
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    word.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn main() {}
