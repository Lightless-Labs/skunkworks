#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

fn rounded_sqrt(n: u128) -> u64 {
    let mut lo = 0_u128;
    let mut hi = u64::MAX as u128 + 1;
    while lo + 1 < hi {
        let mid = (lo + hi) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let lower_sq = lo * lo;
    let upper = lo + 1;
    let upper_sq = upper * upper;
    let result = if n - lower_sq >= upper_sq - n { upper } else { lo };
    result as u64
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -1_000_000_000
            || point.x > 1_000_000_000
            || point.y < -1_000_000_000
            || point.y > 1_000_000_000
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = points.len()
        - usize::from(points.len() >= 2 && points.last() == points.first());
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let n = polygon.len();
    let mut signed_area2 = 0_i128;
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];

        signed_area2 += (a.x as i128) * (b.y as i128) - (a.y as i128) * (b.x as i128);

        let dx = (b.x as i128) - (a.x as i128);
        let dy = (b.y as i128) - (a.y as i128);
        let d2 = (dx * dx + dy * dy) as u128;
        let rounded = rounded_sqrt(d2 * 1_000_000);
        edge_milli.push(rounded);
        perimeter_milli += rounded;

        match longest_edge {
            None => longest_edge = Some(i),
            Some(current) => {
                let current_len = edge_milli[current];
                let pair = if a <= b { (a, b) } else { (b, a) };
                let current_a = polygon[current];
                let current_b = polygon[(current + 1) % n];
                let current_pair = if current_a <= current_b {
                    (current_a, current_b)
                } else {
                    (current_b, current_a)
                };
                if rounded > current_len
                    || (rounded == current_len
                        && (pair < current_pair || (pair == current_pair && i < current)))
                {
                    longest_edge = Some(i);
                }
            }
        }
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn main() {}
