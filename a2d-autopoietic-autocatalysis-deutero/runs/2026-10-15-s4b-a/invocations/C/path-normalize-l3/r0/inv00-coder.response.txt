#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str, offset: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        out.push((path[start..i].to_string(), offset + start));
    }
    out
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    components(path, 0)
        .last()
        .map(|(c, _)| c == "." || c == "..")
        .unwrap_or(false)
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };
    if absolute && stack.is_empty() {
        result = "/".to_string();
    }
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce_components(
    stack: &mut Vec<String>,
    items: &[(String, usize)],
    absolute: bool,
) -> Result<(), PathError> {
    for (component, at) in items {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|s| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: *at });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(component.clone()),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let items = components(path, 0);
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce_components(&mut stack, &items, absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn normalized_context(path: &str, is_home: bool) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(if is_home {
            PathError::InvalidHome
        } else {
            PathError::InvalidCwd
        });
    }
    let normalized = normalize(path).map_err(|_| {
        if is_home {
            PathError::InvalidHome
        } else {
            PathError::InvalidCwd
        }
    })?;
    Ok(components(&normalized, 0)
        .into_iter()
        .map(|(c, _)| c)
        .collect())
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(&'a str, Vec<String>), PathError> {
    validate_raw(path)?;
    if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let home = home.ok_or(PathError::MissingHome)?;
            let remainder = if path == "~" { "" } else { &path[2..] };
            let base = components(home, 0)
                .into_iter()
                .map(|(c, _)| c)
                .collect();
            Ok((remainder, base))
        } else {
            Err(PathError::UnsupportedTilde { at: 1 })
        }
    } else {
        Ok((path, Vec::new()))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd, false)?;
    let home_components = match home {
        Some(h) => Some(normalized_context(h, true)?),
        None => None,
    };
    validate_raw(path)?;

    let expanded = if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let base = home_components.ok_or(PathError::MissingHome)?;
            let remainder = if path == "~" { "" } else { &path[2..] };
            (remainder, base)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (path, Vec::new())
    } else {
        (path, cwd_components)
    };

    let (remainder, mut stack) = expanded;
    let items = components(remainder, if path.starts_with("~/") { 2 } else { 0 });
    reduce_components(&mut stack, &items, true)?;
    Ok(render(&stack, true, directory_intent(path)))
}

fn canonical_link_path(path: &str) -> bool {
    if !path.starts_with('/') || path == "/" || path.ends_with('/') {
        return false;
    }
    normalize(path).is_ok_and(|normalized| normalized == path)
}

#[derive(Clone)]
struct WorkItem {
    text: String,
    offset: usize,
    origin_request: bool,
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd, false)?;
    let home_components = match home {
        Some(h) => Some(normalized_context(h, true)?),
        None => None,
    };

    let mut first_indices = std::collections::HashMap::<String, usize>::new();
    for (index, link) in links.iter().enumerate() {
        if !canonical_link_path(&link.path) || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = first_indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        first_indices.insert(link.path.clone(), index);
    }

    validate_raw(path)?;

    let intent = directory_intent(path);
    let (request_remainder, mut stack, request_offset) = if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let base = home_components.ok_or(PathError::MissingHome)?;
            let remainder = if path == "~" { "" } else { &path[2..] };
            (remainder, base, if path == "~" { 1 } else { 2 })
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (path, Vec::new(), 0)
    } else {
        (path, cwd_components, 0)
    };

    let mut queue: std::collections::VecDeque<WorkItem> = components(request_remainder, request_offset)
        .into_iter()
        .map(|(text, offset)| WorkItem {
            text,
            offset,
            origin_request: true,
        })
        .collect();

    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(item) = queue.pop_front() {
        match item.text.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: item.offset });
                }
                continue;
            }
            _ => {}
        }

        stack.push(item.text.clone());
        let absolute = render(&stack, true, false);
        if let Some((index, link)) = links
            .iter()
            .enumerate()
            .find(|(_, link)| link.path == absolute)
        {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);
            stack.pop();

            let target_absolute = link.target.starts_with('/');
            if target_absolute {
                stack.clear();
            }
            let target_items = components(&link.target, 0);
            let mut target_queue: std::collections::VecDeque<WorkItem> = target_items
                .into_iter()
                .map(|(text, offset)| WorkItem {
                    text,
                    offset,
                    origin_request: false,
                })
                .collect();

            let mut pending: Vec<WorkItem> = target_queue.drain(..).collect();
            pending.extend(queue.drain(..));
            queue = pending.into();

            // Mark target completion using an internal sentinel.
            queue.push_front(WorkItem {
                text: format!("\0LINK_END:{index}"),
                offset: 0,
                origin_request: false,
            });
        }
    }

    Ok(render(&stack, true, intent))
}

fn main() {}
