use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut burst_positions: Vec<usize> = vec![0; jobs.len()];
    let mut completions: Vec<u64> = vec![0; jobs.len()];
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut time: u64 = 0;
    let mut completed: usize = 0;
    let mut last_id: Option<u32> = None;

    while completed < jobs.len() {
        admit_events(&mut events, &mut queues, &mut levels, time);

        if let Some(level) = queues.iter().position(|queue| !queue.is_empty()) {
            let index = queues[level].pop_front().unwrap();
            let job = &jobs[index];

            if last_id.is_some_and(|id| id != job.id) {
                time += switch_cost as u64;
            }

            admit_events(&mut events, &mut queues, &mut levels, time);

            let position = burst_positions[index];
            let remaining = job.bursts[position] as u64;
            let consumed = remaining.min(quanta[level] as u64);
            let start = time;
            time += consumed;
            let end = time;
            let burst_done = consumed == remaining;
            let finished = burst_done && position + 1 == job.bursts.len();

            runs.push(Run {
                id: job.id,
                start,
                end,
                finished,
                level,
            });

            if finished {
                completions[index] = end;
                completed += 1;
            } else if burst_done {
                let io_delay = job.bursts[position + 1] as u64;
                let next_level = level.saturating_sub(1);
                burst_positions[index] += 2;
                events.push(Event {
                    time: end + io_delay,
                    kind: EventKind::Wake { level: next_level },
                    index,
                });
            } else {
                levels[index] = (level + 1).min(quanta.len() - 1);
                queues[levels[index]].push_back(index);
            }

            admit_events(&mut events, &mut queues, &mut levels, end);
            last_id = Some(job.id);
        } else if let Some(next_time) = events.iter().map(|event| event.time).min() {
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_id = None;
            } else {
                admit_events(&mut events, &mut queues, &mut levels, time);
            }
        }
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let completion = completions[index];
            let turnaround = completion - job.arrival as u64;
            let cpu: u64 = job
                .bursts
                .iter()
                .step_by(2)
                .map(|&burst| burst as u64)
                .sum();
            let io: u64 = job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&burst| burst as u64)
                .sum();

            Stat {
                id: job.id,
                completion,
                turnaround,
                waiting: turnaround - cpu - io,
            }
        })
        .collect();

    let count = jobs.len() as u64;
    let waiting_total: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let turnaround_total: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: (waiting_total * 100 + count / 2) / count,
        mean_turnaround_x100: (turnaround_total * 100 + count / 2) / count,
        end: time,
    })
}

fn admit_events(
    events: &mut Vec<Event>,
    queues: &mut [VecDeque<usize>],
    levels: &mut [usize],
    through: u64,
) {
    events.sort_by_key(|event| {
        let kind_order = match event.kind {
            EventKind::Arrival => 0usize,
            EventKind::Wake { .. } => 1usize,
        };
        (event.time, kind_order, event.index)
    });

    let admitted = events.partition_point(|event| event.time <= through);
    let ready: Vec<Event> = events.drain(..admitted).collect();

    for event in ready {
        match event.kind {
            EventKind::Arrival => {
                let level = levels[event.index];
                queues[level].push_back(event.index);
            }
            EventKind::Wake { level } => {
                levels[event.index] = level;
                queues[level].push_back(event.index);
            }
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn cpu_only_demotion() {
        assert_eq!(
            schedule(
                &[
                    Job {
                        id: 1,
                        arrival: 0,
                        bursts: vec![6],
                        level: 0
                    },
                    Job {
                        id: 2,
                        arrival: 0,
                        bursts: vec![2],
                        level: 0
                    }
                ],
                &[2, 4],
                0
            ),
            Ok(Schedule {
                runs: vec![
                    Run {
                        id: 1,
                        start: 0,
                        end: 2,
                        finished: false,
                        level: 0
                    },
                    Run {
                        id: 2,
                        start: 2,
                        end: 4,
                        finished: true,
                        level: 0
                    },
                    Run {
                        id: 1,
                        start: 4,
                        end: 8,
                        finished: true,
                        level: 1
                    }
                ],
                idle: vec![],
                stats: vec![
                    Stat {
                        id: 1,
                        completion: 8,
                        turnaround: 8,
                        waiting: 2
                    },
                    Stat {
                        id: 2,
                        completion: 4,
                        turnaround: 4,
                        waiting: 2
                    }
                ],
                mean_waiting_x100: 200,
                mean_turnaround_x100: 600,
                end: 8
            })
        );
    }

    #[test]
    fn io_idle_and_promotion() {
        assert_eq!(
            schedule(
                &[Job {
                    id: 5,
                    arrival: 2,
                    bursts: vec![1, 3, 2],
                    level: 1
                }],
                &[3, 4],
                0
            ),
            Ok(Schedule {
                runs: vec![
                    Run {
                        id: 5,
                        start: 2,
                        end: 3,
                        finished: false,
                        level: 1
                    },
                    Run {
                        id: 5,
                        start: 6,
                        end: 8,
                        finished: true,
                        level: 0
                    }
                ],
                idle: vec![(0, 2), (3, 6)],
                stats: vec![Stat {
                    id: 5,
                    completion: 8,
                    turnaround: 6,
                    waiting: 0
                }],
                mean_waiting_x100: 0,
                mean_turnaround_x100: 600,
                end: 8
            })
        );
    }

    #[test]
    fn empty_statistics() {
        assert_eq!(
            schedule(&[], &[1], 0),
            Ok(Schedule {
                runs: vec![],
                idle: vec![],
                stats: vec![],
                mean_waiting_x100: 0,
                mean_turnaround_x100: 0,
                end: 0
            })
        );
    }

    #[test]
    fn invalid_cpu_burst() {
        assert_eq!(
            schedule(
                &[Job {
                    id: 1,
                    arrival: 0,
                    bursts: vec![0],
                    level: 0
                }],
                &[2],
                0
            ),
            Err(Error::Burst {
                index: 0,
                position: 0
            })
        );
    }
}

fn main() {}
