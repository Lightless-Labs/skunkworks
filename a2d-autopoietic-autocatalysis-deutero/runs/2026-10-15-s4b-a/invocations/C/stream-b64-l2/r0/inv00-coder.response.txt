#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|i| i as u8)
}

fn encode_group(input: &[u8]) -> String {
    let a = input[0];
    let b = if input.len() > 1 { input[1] } else { 0 };
    let c = if input.len() > 2 { input[2] } else { 0 };

    let mut out = String::with_capacity(4);
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if input.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if input.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }
    out
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut out = encoder.push(input).unwrap();
    out.push_str(&encoder.finish().unwrap());
    out
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.tail.len() + input.len());
        data.extend_from_slice(&self.tail);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut out = String::with_capacity(complete_len / 3 * 4);
        for group in data[..complete_len].chunks_exact(3) {
            out.push_str(&encode_group(group));
        }
        self.tail.clear();
        self.tail.extend_from_slice(&data[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        if self.tail.is_empty() {
            Ok(String::new())
        } else {
            let out = encode_group(&self.tail);
            self.tail.clear();
            Ok(out)
        }
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Decoder {
    group: Vec<(u8, usize)>,
    offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(err) = &self.sticky_error {
            return Err(err.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let at = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset: at });
            }

            if let Some(pending_at) = self.pending_second_padding {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset: at });
                }
                self.pending_second_padding = None;
                self.group.push((byte, at));
                self.complete_group(&mut out);
                self.terminal_padding = true;
                continue;
            }

            let pos = self.group.len();
            if byte == b'=' {
                match pos {
                    0 | 1 => return self.fail(Error::BadPadding { offset: at }),
                    2 => self.pending_second_padding = Some(at),
                    3 => {
                        self.group.push((byte, at));
                        self.complete_group(&mut out);
                        self.terminal_padding = true;
                    }
                    _ => unreachable!(),
                }
                if pos == 2 {
                    self.group.push((byte, at));
                }
                continue;
            }

            let Some(value) = digit(byte) else {
                return self.fail(Error::InvalidByte { offset: at, byte });
            };
            self.group.push((value, at));

            if self.group.len() == 4 {
                self.complete_group(&mut out);
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(err) = &self.sticky_error {
            return Err(err.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            return self.fail(Error::Truncated {
                offset: self.group[0].1,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn complete_group(&mut self, out: &mut Vec<u8>) {
        let g = &self.group;
        let a = g[0].0;
        let b = g[1].0;

        out.push((a << 2) | (b >> 4));

        if g[2].0 != b'=' {
            let c = g[2].0;
            out.push((b << 4) | (c >> 2));
            if g[3].0 != b'=' {
                let d = g[3].0;
                out.push((c << 6) | d);
            }
        }
        self.group.clear();
    }

    fn fail<T>(&mut self, err: Error) -> Result<T, Error> {
        self.sticky_error = Some(err.clone());
        Err(err)
    }
}

fn main() {}
