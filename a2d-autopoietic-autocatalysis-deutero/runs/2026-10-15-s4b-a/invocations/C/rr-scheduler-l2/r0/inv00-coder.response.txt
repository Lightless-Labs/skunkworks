#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    use std::collections::VecDeque;

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by_key(|&i| (jobs[i].arrival, i));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion_times: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let admit_through = |time: u64,
                         next_arrival: &mut usize,
                         ready: &mut VecDeque<usize>,
                         completion_times: &mut Vec<Option<u64>>| {
        while *next_arrival < arrival_order.len() {
            let index = arrival_order[*next_arrival];
            let job = &jobs[index];
            if u64::from(job.arrival) > time {
                break;
            }
            *next_arrival += 1;
            if job.ticks == 0 {
                completion_times[index] = Some(u64::from(job.arrival));
            } else {
                ready.push_back(index);
            }
        }
    };

    loop {
        admit_through(now, &mut next_arrival, &mut ready, &mut completion_times);

        if let Some(index) = ready.pop_front() {
            let job = &jobs[index];

            if last_id.is_some_and(|id| id != job.id) {
                now += u64::from(switch_cost);
            }

            admit_through(now, &mut next_arrival, &mut ready, &mut completion_times);

            let consumed = remaining[index].min(quantum);
            let start = now;
            now += u64::from(consumed);
            remaining[index] -= consumed;
            let finished = remaining[index] == 0;

            runs.push(Run {
                id: job.id,
                start,
                end: now,
                finished,
            });

            admit_through(now, &mut next_arrival, &mut ready, &mut completion_times);

            if finished {
                completion_times[index] = Some(now);
            } else {
                ready.push_back(index);
            }

            last_id = Some(job.id);
        } else if next_arrival < arrival_order.len() {
            let next_time = u64::from(jobs[arrival_order[next_arrival]].arrival);
            if next_time > now {
                if let Some((_, end)) = idle.last_mut() {
                    if *end == now {
                        *end = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_id = None;
            }
        } else {
            break;
        }
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion_times[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}

fn main() {}
