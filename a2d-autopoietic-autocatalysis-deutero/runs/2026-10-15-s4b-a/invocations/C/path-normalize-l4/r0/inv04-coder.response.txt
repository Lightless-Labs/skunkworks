#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn raw_validate(s: &str) -> Result<(), PathError> {
    if s.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = s.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = s.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = s.as_bytes().iter().position(|&b| b <= 31 || b == 127) {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(s: &str) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        out.push((s[start..i].to_string(), start));
    }
    out
}

fn directory_intent(s: &str) -> bool {
    s.ends_with('/')
        || components(s)
            .last()
            .map(|(c, _)| c == "." || c == "..")
            .unwrap_or(false)
}

fn format_absolute(stack: &[String], intent: bool) -> String {
    let mut result = if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    };
    if intent && result != "/" {
        result.push('/');
    }
    result
}

fn format_normalized(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if stack.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce_normalized(s: &str) -> Result<String, PathError> {
    raw_validate(s)?;
    let absolute = s.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(s);
    let mut stack = Vec::<String>::new();

    for (part, at) in components(s) {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().map(|x| x != "..").unwrap_or(false) {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }

    Ok(format_normalized(&stack, absolute, intent))
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    reduce_normalized(path)
}

fn normalized_context(s: &str, home: bool) -> Result<Vec<String>, PathError> {
    if !s.starts_with('/') {
        return Err(if home {
            PathError::InvalidHome
        } else {
            PathError::InvalidCwd
        });
    }
    match reduce_normalized(s) {
        Ok(value) => Ok(components(&value).into_iter().map(|(c, _)| c).collect()),
        Err(_) => Err(if home {
            PathError::InvalidHome
        } else {
            PathError::InvalidCwd
        }),
    }
}

fn request_parts(
    path: &str,
    home: Option<&str>,
) -> Result<(bool, Vec<(String, usize)>, bool), PathError> {
    raw_validate(path)?;
    let intent = directory_intent(path);

    if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let _ = home.ok_or(PathError::MissingHome)?;
            let rest = if path == "~" { "" } else { &path[2..] };
            let parts = components(rest)
                .into_iter()
                .map(|(part, at)| (part, at + 2))
                .collect();
            return Ok((false, parts, intent));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    Ok((path.starts_with('/'), components(path), intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd, false)?;
    let home_parts = match home {
        Some(value) => Some(normalized_context(value, true)?),
        None => None,
    };
    let (absolute, parts, intent) = request_parts(path, home)?;

    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        home_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    for (part, at) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(format_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && link.path != "/"
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .map(|value| value == link.path)
                .unwrap_or(false);
        if !path_valid || raw_validate(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

#[derive(Clone)]
struct Work {
    parts: Vec<(String, usize)>,
    pos: usize,
    link_index: Option<usize>,
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd, false)?;
    let home_parts = match home {
        Some(value) => Some(normalized_context(value, true)?),
        None => None,
    };
    validate_links(links)?;
    let (absolute, request, intent) = request_parts(path, home)?;

    let start = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        home_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    let mut stack = Vec::<String>::new();
    let mut work = vec![Work {
        parts: start.into_iter().map(|part| (part, 0)).collect(),
        pos: 0,
        link_index: None,
    }];
    work.push(Work {
        parts: request,
        pos: 0,
        link_index: None,
    });

    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while !work.is_empty() {
        if work.last().unwrap().pos >= work.last().unwrap().parts.len() {
            let completed = work.pop().unwrap();
            if let Some(index) = completed.link_index {
                if let Some(pos) = active.iter().rposition(|&value| value == index) {
                    active.remove(pos);
                }
            }
            continue;
        }

        let (part, at) = {
            let top = work.last_mut().unwrap();
            let item = top.parts[top.pos].clone();
            top.pos += 1;
            item
        };

        match part.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(part);
                let current = format!("/{}", stack.join("/"));
                if let Some((index, link)) =
                    links.iter().enumerate().find(|(_, link)| link.path == current)
                {
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    expansions += 1;
                    active.push(index);
                    stack.pop();

                    if link.target.starts_with('/') {
                        stack.clear();
                    }
                    work.push(Work {
                        parts: components(&link.target),
                        pos: 0,
                        link_index: Some(index),
                    });
                }
            }
        }
    }

    Ok(format_absolute(&stack, intent))
}

fn canonical_parts(path: &str) -> Vec<String> {
    components(path).into_iter().map(|(part, _)| part).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_string(); base.len() - common];
    parts.extend_from_slice(&target[common..]);
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = canonical_parts(&resolved_base);
    let target_parts = canonical_parts(&resolved_target);
    let target_is_dir = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        if let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let probe_parts = canonical_parts(&probe);
            if probe_parts.len() <= target_parts.len()
                && probe_parts
                    .iter()
                    .zip(target_parts.iter())
                    .all(|(a, b)| a == b)
            {
                let mut candidate = route(&base_parts, &canonical_parts(&link.path));
                let suffix = &target_parts[probe_parts.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let mut valid = Vec::<String>::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_is_dir && candidate != "." {
            candidate.push('/');
        }
        if let Ok(roundtrip) =
            resolve_links(&candidate, &resolved_base, home, links, max_expansions)
        {
            if canonical_parts(&roundtrip) == target_parts {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Empty)
}

fn main() {}
