use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if start == *pos {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: *pos });
        }
        return Err(VersionError::InvalidChar { at: *pos });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_ids(s: &str, start: usize, end: usize, pre: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            if !is_ident_char(bytes[p]) {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if pre
            && bytes[id_start..p].iter().all(|b| is_digit(*b))
            && p - id_start > 1
            && bytes[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(s[id_start..p].to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p == s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p == s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    if p < s.len() && !matches!(s.as_bytes()[p], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: p });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let start = p + 1;
        let end = s[start..]
            .find('+')
            .map(|i| start + i)
            .unwrap_or(s.len());
        pre = parse_ids(s, start, end, true)?;
        p = end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        build = parse_ids(s, p + 1, s.len(), false)?;
        p = s.len();
    }
    if p != s.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => cmp_numeric_str(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => cmp_pre(&a.pre, &b.pre),
        })
}

pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn version_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn full_or_partial(s: &str, at: usize) -> Result<(Version, usize, bool), RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let full_rule = s.contains('-')
        || s.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X")));
    if full_rule {
        return Ok((version_at(s, at)?, 3, true));
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }

    let mut vals = Vec::new();
    let mut wildcard = false;
    for field in &fields {
        if matches!(*field, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard
                || !field.bytes().all(is_digit)
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let n = field
                .parse::<u64>()
                .map_err(|_| RangeError::Syntax { at })?;
            vals.push(n);
        }
    }
    let k = vals.len();
    while vals.len() < 3 {
        vals.push(0);
    }
    Ok((
        Version {
            major: vals[0],
            minor: vals[1],
            patch: vals[2],
            pre: Vec::new(),
            build: Vec::new(),
        },
        k,
        false,
    ))
}

fn endpoint(version: Version) -> Cmp {
    Cmp {
        op: Op::Ge,
        version,
    }
}

fn upper(version: Version) -> Cmp {
    Cmp {
        op: Op::Lt,
        version,
    }
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (major, minor, patch) = match field {
        0 => (v.major.checked_add(1), Some(0), Some(0)),
        1 => (Some(v.major), v.minor.checked_add(1), Some(0)),
        _ => (Some(v.major), Some(v.minor), v.patch.checked_add(1)),
    };
    let (Some(major), Some(minor), Some(patch)) = (major, minor, patch) else {
        return Err(RangeError::Syntax { at });
    };
    Ok(Version {
        major,
        minor,
        patch,
        pre: vec!["0".into()],
        build: Vec::new(),
    })
}

fn parse_partial_token(
    s: &str,
    at: usize,
    mode: u8,
) -> Result<(Vec<Cmp>, Option<(u64, u64, u64)>), RangeError> {
    let (v, k, full) = full_or_partial(s, at)?;
    if mode != 0 && k == 0 {
        return Err(RangeError::Syntax { at });
    }
    let anchor = if full && !v.pre.is_empty() {
        Some((v.major, v.minor, v.patch))
    } else {
        None
    };
    let mut comps = Vec::new();
    match mode {
        0 => {
            if full {
                comps.push(Cmp {
                    op: Op::Eq,
                    version: v,
                });
            } else if k > 0 {
                comps.push(endpoint(v.clone()));
                comps.push(upper(bump(&v, if k == 1 { 0 } else { 1 }, at)?));
            }
        }
        1 => {
            comps.push(endpoint(v.clone()));
            let field = if v.major > 0 || k == 1 {
                0
            } else if v.minor > 0 || k == 2 {
                1
            } else {
                2
            };
            comps.push(upper(bump(&v, field, at)?));
        }
        2 => {
            comps.push(endpoint(v.clone()));
            comps.push(upper(bump(&v, if k == 1 { 0 } else { 1 }, at)?));
        }
        _ => {}
    }
    Ok((comps, anchor))
}

fn parse_token(t: &str, at: usize) -> Result<(Vec<Cmp>, Option<(u64, u64, u64)>), RangeError> {
    if t.starts_with('^') || t.starts_with('~') {
        return parse_partial_token(
            &t[1..],
            at + 1,
            if t.starts_with('^') { 1 } else { 2 },
        );
    }

    if let Some(operand) = t.strip_prefix(">=") {
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at });
        }
        return Ok((
            vec![Cmp {
                op: Op::Ge,
                version: version_at(operand, at + 2)?,
            }],
            None,
        ));
    }
    if let Some(operand) = t.strip_prefix("<=") {
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at });
        }
        return Ok((
            vec![Cmp {
                op: Op::Le,
                version: version_at(operand, at + 2)?,
            }],
            None,
        ));
    }
    let (op, offset) = match t.as_bytes().first() {
        Some(b'>') => (Some(Op::Gt), 1),
        Some(b'<') => (Some(Op::Lt), 1),
        Some(b'=') => (Some(Op::Eq), 1),
        _ => (None, 0),
    };
    if let Some(op) = op {
        let operand = &t[offset..];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at });
        }
        return Ok((
            vec![Cmp {
                op,
                version: version_at(operand, at + offset)?,
            }],
            None,
        ));
    }
    if !t.as_bytes().first().is_some_and(|b| {
        b.is_ascii_digit() || matches!(*b, b'*' | b'x' | b'X')
    }) {
        return Err(RangeError::Syntax { at });
    }
    parse_partial_token(t, at, 0)
}

fn split_tokens(s: &str, base: usize) -> Vec<(&str, usize)> {
    let mut out = Vec::new();
    let bytes = s.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !ws(bytes[i]) {
            i += 1;
        }
        out.push((&s[start..i], base + start));
    }
    out
}

fn parse_hyphen_endpoint(token: &str, at: usize) -> Result<(Version, usize, bool), RangeError> {
    let (v, k, full) = full_or_partial(token, at)?;
    if k == 0 {
        return Err(RangeError::Syntax { at });
    }
    Ok((v, k, full))
}

fn parse_arm(tokens: &[(&str, usize)]) -> Result<Arm, RangeError> {
    if let Some((idx, _)) = tokens.iter().enumerate().find(|(_, token)| token.0 == "-") {
        if tokens.len() != 3 || idx != 1 {
            return Err(RangeError::Syntax { at: tokens[idx].1 });
        }

        let (lower, _, lower_full) = parse_hyphen_endpoint(tokens[0].0, tokens[0].1)?;
        let (upper_version, upper_k, upper_full) =
            parse_hyphen_endpoint(tokens[2].0, tokens[2].1)?;

        let mut comps = vec![Cmp {
            op: Op::Ge,
            version: lower.clone(),
        }];
        if upper_full {
            comps.push(Cmp {
                op: Op::Le,
                version: upper_version.clone(),
            });
        } else {
            comps.push(upper(bump(
                &upper_version,
                if upper_k == 1 { 0 } else { 1 },
                tokens[2].1,
            )?));
        }

        let mut anchors = Vec::new();
        if lower_full && !lower.pre.is_empty() {
            anchors.push((lower.major, lower.minor, lower.patch));
        }
        if upper_full && !upper_version.pre.is_empty() {
            anchors.push((
                upper_version.major,
                upper_version.minor,
                upper_version.patch,
            ));
        }
        return Ok(Arm { comps, anchors });
    }

    let mut arm = Arm {
        comps: Vec::new(),
        anchors: Vec::new(),
    };
    for (token, at) in tokens {
        let (mut comps, anchor) = parse_token(token, *at)?;
        arm.comps.append(&mut comps);
        if let Some(anchor) = anchor {
            arm.anchors.push(anchor);
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;

    while i + 1 < bytes.len() {
        if bytes[i] == b'|'
            && bytes[i + 1] == b'|'
            && i > 0
            && ws(bytes[i - 1])
            && i + 2 < bytes.len()
            && ws(bytes[i + 2])
        {
            let tokens = split_tokens(&s[start..i], start);
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(&tokens)?);
            i += 2;
            while i < bytes.len() && ws(bytes[i]) {
                i += 1;
            }
            start = i;
            continue;
        }
        i += 1;
    }

    let tokens = split_tokens(&s[start..], start);
    if tokens.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&tokens)?);
    Ok(Range { arms })
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn satisfies_cmp(v: &Version, c: &Cmp) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    arm.comps.iter().all(|c| satisfies_cmp(v, c))
        && (include_prerelease || v.pre.is_empty() || arm.anchors.contains(&tuple(v)))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| parse_version(s).map_err(|error| CandidateError { index, error }))
        .collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        if best.is_none_or(|j| {
            if max {
                compare_versions(&parsed[i], &parsed[j]) == Ordering::Greater
            } else {
                compare_versions(&parsed[i], &parsed[j]) == Ordering::Less
            }
        }) {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn ver_string(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    // A prerelease endpoint is an admission anchor. Check the endpoint and its
    // immediate successor, which are the only possible witnesses at a strict
    // boundary before the next distinct identifier prefix.
    for arm_a in &a.arms {
        for arm_b in &b.arms {
            let mut comps = arm_a.comps.clone();
            comps.extend(arm_b.comps.clone());
            let mut probes = Vec::new();
            for c in &comps {
                probes.push(c.version.clone());
                if !c.version.pre.is_empty() {
                    let mut successor = c.version.clone();
                    successor.pre.push("0".into());
                    probes.push(successor);
                    let mut stable = c.version.clone();
                    stable.pre.clear();
                    probes.push(stable);
                } else if let Some(patch) = c.version.patch.checked_add(1) {
                    let mut next = c.version.clone();
                    next.patch = patch;
                    next.pre = vec!["0".into()];
                    next.build.clear();
                    probes.push(next);
                }
            }
            probes.push(Version {
                major: 0,
                minor: 0,
                patch: 0,
                pre: Vec::new(),
                build: Vec::new(),
            });
            if probes.iter().any(|v| {
                arm_matches(v, arm_a, include_prerelease)
                    && arm_matches(v, arm_b, include_prerelease)
            }) {
                return true;
            }
        }
    }
    false
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper_bound: Option<(Version, bool)> = None;
        for c in &arm.comps {
            match c.op {
                Op::Eq => {
                    lower = Some((c.version.clone(), true));
                    upper_bound = Some((c.version.clone(), true));
                }
                Op::Gt | Op::Ge => {
                    let inclusive = matches!(c.op, Op::Ge);
                    if lower.as_ref().is_none_or(|(v, i)| {
                        compare_versions(&c.version, v) == Ordering::Greater
                            || (compare_versions(&c.version, v) == Ordering::Equal
                                && !inclusive
                                && *i)
                    }) {
                        lower = Some((c.version.clone(), inclusive));
                    }
                }
                Op::Lt | Op::Le => {
                    let inclusive = matches!(c.op, Op::Le);
                    if upper_bound.as_ref().is_none_or(|(v, i)| {
                        compare_versions(&c.version, v) == Ordering::Less
                            || (compare_versions(&c.version, v) == Ordering::Equal
                                && !inclusive
                                && *i)
                    }) {
                        upper_bound = Some((c.version.clone(), inclusive));
                    }
                }
            }
        }
        let lo = lower
            .map(|(v, inclusive)| {
                format!("{}{}", if inclusive { ">=" } else { ">" }, ver_string(&v))
            })
            .unwrap_or_else(|| "*".into());
        let hi = upper_bound
            .map(|(v, inclusive)| {
                format!("{}{}", if inclusive { "<=" } else { "<" }, ver_string(&v))
            })
            .unwrap_or_else(|| "*".into());
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors
                .iter()
                .map(|(major, minor, patch)| format!("{major}.{minor}.{patch}"))
                .collect::<Vec<_>>()
                .join(",")
        };
        strings.push(format!("[{lo};{hi};pre={anchor_text}]"));
    }
    strings.sort();
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".into()
    } else {
        strings.join(" || ")
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage {
        name: String,
    },
    DuplicateLabel {
        package: String,
        label: String,
    },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution {
        package: String,
        labels: Vec<String>,
    },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    out: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s)
            .map(|range| out.push(range))
            .map_err(|error| SelectError::InvalidRange {
                package: package.into(),
                label: label.into(),
                path: path.clone(),
                error,
            }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, out)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn eval_expr<'a>(
    expr: &SetExpr,
    v: &Version,
    include: bool,
    iter: &mut impl Iterator<Item = &'a Range>,
) -> bool {
    match expr {
        SetExpr::Range(_) => iter.next().is_some_and(|r| matches_with(v, r, include)),
        SetExpr::Union(children) => children.iter().any(|x| eval_expr(x, v, include, iter)),
        SetExpr::Intersection(children) => {
            children.iter().all(|x| eval_expr(x, v, include, iter))
        }
    }
}

fn conflict_labels(
    package: &Package,
    versions: &[Version],
    include: bool,
    parsed: &[Vec<Range>],
) -> Vec<String> {
    let n = package.constraints.len();
    if versions.is_empty() {
        return Vec::new();
    }

    for size in 1..=n {
        let mut found: Vec<Vec<String>> = Vec::new();

        fn rec(
            start: usize,
            size: usize,
            chosen: &mut Vec<usize>,
            n: usize,
            found: &mut Vec<Vec<String>>,
            package: &Package,
            versions: &[Version],
            include: bool,
            parsed: &[Vec<Range>],
        ) {
            if chosen.len() == size {
                let satisfiable = versions.iter().any(|v| {
                    chosen.iter().all(|&ci| {
                        let mut iter = parsed[ci].iter();
                        eval_expr(&package.constraints[ci].expr, v, include, &mut iter)
                    })
                });
                if !satisfiable {
                    let mut labels: Vec<String> = chosen
                        .iter()
                        .map(|&i| package.constraints[i].label.clone())
                        .collect();
                    labels.sort();
                    found.push(labels);
                }
                return;
            }
            for i in start..n {
                chosen.push(i);
                rec(
                    i + 1,
                    size,
                    chosen,
                    n,
                    found,
                    package,
                    versions,
                    include,
                    parsed,
                );
                chosen.pop();
            }
        }

        rec(
            0,
            size,
            &mut Vec::new(),
            n,
            &mut found,
            package,
            versions,
            include,
            parsed,
        );
        if !found.is_empty() {
            found.sort();
            return found.remove(0);
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for package in packages {
        if !names.insert(package.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: package.name.clone(),
            });
        }
    }

    for package in packages {
        let mut labels = std::collections::HashSet::new();
        for constraint in &package.constraints {
            if !labels.insert(constraint.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: package.name.clone(),
                    label: constraint.label.clone(),
                });
            }
        }
    }

    let mut versions_all = Vec::new();
    for package in packages {
        let mut versions = Vec::new();
        for (index, candidate) in package.candidates.iter().enumerate() {
            versions.push(
                parse_version(candidate).map_err(|error| SelectError::InvalidCandidate {
                    package: package.name.clone(),
                    index,
                    error,
                })?,
            );
        }
        versions_all.push(versions);
    }

    let mut ranges_all = Vec::new();
    for package in packages {
        let mut package_ranges = Vec::new();
        for constraint in &package.constraints {
            let mut ranges = Vec::new();
            validate_expr(
                &constraint.expr,
                &package.name,
                &constraint.label,
                &mut Vec::new(),
                &mut ranges,
            )?;
            package_ranges.push(ranges);
        }
        ranges_all.push(package_ranges);
    }

    let mut result = Vec::new();
    for (pi, package) in packages.iter().enumerate() {
        let versions = &versions_all[pi];
        let parsed_ranges = &ranges_all[pi];
        let mut eligible = Vec::new();

        for i in 0..versions.len() {
            let accepted = package.constraints.iter().enumerate().all(|(ci, constraint)| {
                let mut iter = parsed_ranges[ci].iter();
                eval_expr(&constraint.expr, &versions[i], include_prerelease, &mut iter)
            });
            if accepted {
                eligible.push(i);
            }
        }

        if eligible.is_empty() {
            return Err(SelectError::NoSolution {
                package: package.name.clone(),
                labels: conflict_labels(package, versions, include_prerelease, parsed_ranges),
            });
        }

        let stable: Vec<usize> = eligible
            .iter()
            .copied()
            .filter(|&i| versions[i].pre.is_empty())
            .collect();
        let pool = if matches!(preference, Preference::StableFirst) && !stable.is_empty() {
            stable.as_slice()
        } else {
            eligible.as_slice()
        };

        let mut best = pool[0];
        for &i in &pool[1..] {
            let order = compare_versions(&versions[i], &versions[best]);
            let choose = match preference {
                Preference::Lowest => order == Ordering::Less,
                Preference::Highest | Preference::StableFirst => order == Ordering::Greater,
            };
            if choose {
                best = i;
            }
        }
        result.push(Selection {
            package: package.name.clone(),
            version: package.candidates[best].clone(),
        });
    }
    Ok(result)
}
