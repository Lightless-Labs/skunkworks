use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}
fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}
fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if start == *pos {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: *pos });
        }
        return Err(VersionError::InvalidChar { at: *pos });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}
fn parse_ids(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            if !is_ident_char(bytes[p]) {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if pre && bytes[id_start..p].iter().all(|b| is_digit(*b))
            && p - id_start > 1
            && bytes[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(s[id_start..p].to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    for _ in 0..1 {
        if p >= s.len() || s.as_bytes()[p] != b'.' {
            if p == s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: p });
            }
            return Err(VersionError::InvalidChar { at: p });
        }
        p += 1;
    }
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p == s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    if p < s.len() && !matches!(s.as_bytes()[p], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: p });
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let st = p + 1;
        let end = s[st..]
            .find('+')
            .map(|i| st + i)
            .unwrap_or(s.len());
        pre = parse_ids(s, st, end, true)?;
        p = end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        build = parse_ids(s, p + 1, s.len(), false)?;
        p = s.len();
    }
    if p != s.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}
fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}
fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => cmp_numeric_str(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => cmp_pre(&a.pre, &b.pre),
        })
}

fn ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }
fn version_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })

fn full_or_partial(s: &str, at: usize) -> Result<(Version, usize, bool), RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let full_rule = s.contains('-') || s.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X")));
    if full_rule {
        let v = version_at(s, at)?;
        return Ok((v, 3, true));
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut vals = Vec::new();
    let mut wildcard = false;
    for f in &fields {
        if matches!(*f, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            vals.push(n);
        }
    }
    let k = vals.len();
    while vals.len() < 3 { vals.push(0); }
    Ok((Version { major: vals[0], minor: vals[1], patch: vals[2], pre: vec![], build: vec![] }, k, false))
}
fn endpoint(v: Version) -> Cmp { Cmp { op: Op::Ge, version: v } }
fn upper(v: Version) -> Cmp { Cmp { op: Op::Lt, version: v } }
fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (major,minor,patch) = match field {
        0 => (v.major.checked_add(1), Some(0), Some(0)),
        1 => (Some(v.major), v.minor.checked_add(1), Some(0)),
        _ => (Some(v.major), Some(v.minor), v.patch.checked_add(1)),
    };
    let (Some(major),Some(minor),Some(patch))=(major,minor,patch) else { return Err(RangeError::Syntax{at}); };
    Ok(Version{major,minor,patch,pre:vec!["0".into()],build:vec![]})
}
fn parse_partial_token(s: &str, at: usize, mode: u8) -> Result<(Vec<Cmp>, Option<(u64,u64,u64)>), RangeError> {
    let (v,k,full)=full_or_partial(s,at)?;
    if mode != 0 && k == 0 { return Err(RangeError::Syntax{at}); }
    let mut cs=Vec::new();
    let anchor=if full && !v.pre.is_empty(){Some((v.major,v.minor,v.patch))}else{None};
    match mode {
        0 => {
            if full { cs.push(Cmp{op:Op::Eq,version:v}); }
            else if k>0 {
                cs.push(endpoint(v.clone()));
                let upper_field=if k==1 {0}else{1};
                cs.push(upper(bump(&v,upper_field,at)?));
            }
        }
        1 => {
            cs.push(endpoint(v.clone()));
            let field=if v.major>0 || k==1 {0} else if v.minor>0 || k==2 {1} else {2};
            cs.push(upper(bump(&v,field,at)?));
        }
        2 => {
            cs.push(endpoint(v.clone()));
            cs.push(upper(bump(&v,if k==1 {0}else{1},at)?));
        }
        _ => {}
    }
    Ok((cs,anchor))
}
fn parse_token(t: &str, at: usize) -> Result<(Vec<Cmp>, Option<(u64,u64,u64)>), RangeError> {
    if t.starts_with('^') || t.starts_with('~') {
        return parse_partial_token(&t[1..],at+1,if t.starts_with('^'){1}else{2});
    }
    if let Some(op)=t.strip_prefix(">=") {
        if op.is_empty() || !op.as_bytes()[0].is_ascii_digit(){return Err(RangeError::Syntax{at});}
        return Ok((vec![Cmp{op:Op::Ge,version:version_at(op,at+2)?}],None));
    }
    if let Some(op)=t.strip_prefix("<=") {
        if op.is_empty() || !op.as_bytes()[0].is_ascii_digit(){return Err(RangeError::Syntax{at});}
        return Ok((vec![Cmp{op:Op::Le,version:version_at(op,at+2)?}],None));
    }
    let (op,off)=match t.as_bytes().first(){Some(b'>')=>(Some(Op::Gt),1),Some(b'<')=>(Some(Op::Lt),1),Some(b'=')=>(Some(Op::Eq),1),_=>(None,0)};
    if let Some(op)=op {
        let operand=&t[off..];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit(){return Err(RangeError::Syntax{at});}
        return Ok((vec![Cmp{op,version:version_at(operand,at+off)?}],None));
    }
    if !t.as_bytes().first().is_some_and(|b| b.is_ascii_digit() || matches!(*b,b'*'|b'x'|b'X')) {
        return Err(RangeError::Syntax{at});
    }
    parse_partial_token(t,at,0)
}
fn split_tokens(s: &str, base: usize) -> Vec<(&str,usize)> {
    let mut out=Vec::new(); let b=s.as_bytes(); let mut i=0;
    while i<b.len(){while i<b.len()&&ws(b[i]){i+=1;} if i==b.len(){break;} let st=i; while i<b.len()&&!ws(b[i]){i+=1;} out.push((&s[st..i],base+st));}
    out
}
fn parse_arm(tokens: &[(&str,usize)]) -> Result<Arm,RangeError> {
    if let Some((idx,_))=tokens.iter().enumerate().find(|(_,x)| x.0=="-") {
        if tokens.len()!=3 || idx!=1 { return Err(RangeError::Syntax{at:tokens[idx].1}); }
        let (mut lc,la)=parse_partial_token(tokens[0].0,tokens[0].1,0)?;
        let (mut uc,ua)=parse_partial_token(tokens[2].0,tokens[2].1,0)?;
        if la.is_none() && tokens[0].0.split('.').all(|x| matches!(x,"*"|"x"|"X")) {return Err(RangeError::Syntax{at:tokens[0].1});}
        if ua.is_none() && tokens[2].0.split('.').all(|x| matches!(x,"*"|"x"|"X")) {return Err(RangeError::Syntax{at:tokens[2].1});}
        let (lv,_,lf)=full_or_partial(tokens[0].0,tokens[0].1)?;
        let (uv,uk,uf)=full_or_partial(tokens[2].0,tokens[2].1)?;
        lc.clear(); uc.clear();
        lc.push(Cmp{op:Op::Ge,version:lv.clone()});
        if uf {uc.push(Cmp{op:Op::Le,version:uv.clone()});}
        else if uk==1 {uc.push(upper(bump(&uv,0,tokens[2].1)?));}
        else {uc.push(upper(bump(&uv,1,tokens[2].1)?));}
        let mut anchors=Vec::new();
        if lf && !lv.pre.is_empty(){anchors.push((lv.major,lv.minor,lv.patch));}
        if uf && !uv.pre.is_empty(){anchors.push((uv.major,uv.minor,uv.patch));}
        return Ok(Arm{comps:lc.into_iter().chain(uc).collect(),anchors});
    }
    let mut arm=Arm{comps:vec![],anchors:vec![]};
    for (t,at) in tokens {
        let (mut cs,a)=parse_token(t,*at)?;
        arm.comps.append(&mut cs);
        if let Some(a)=a {arm.anchors.push(a);}
    }
    Ok(arm)
}
pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b=s.as_bytes();
    let mut arms=Vec::new();
    let mut start=0;
    let mut i=0;
    while i+1<b.len() {
        if b[i]==b'|'&&b[i+1]==b'|' {
            if i>0 && ws(b[i-1]) && i+2<b.len() && ws(b[i+2]) {
                let tok=split_tokens(&s[start..i],start);
                if tok.is_empty(){return Err(RangeError::Empty{at:i});}
                arms.push(parse_arm(&tok)?);
                i+=2;
                while i<b.len()&&ws(b[i]){i+=1;}
                start=i;
                continue;
            }
        }
        i+=1;
    }
    let tok=split_tokens(&s[start..],start);
    if tok.is_empty() {
        if arms.is_empty(){return Err(RangeError::Empty{at:0});}
        return Err(RangeError::Empty{at:s.len()});
    }
    arms.push(parse_arm(&tok)?);
    Ok(Range{arms})
}
fn tuple(v:&Version)->(u64,u64,u64){(v.major,v.minor,v.patch)}
fn satisfies_cmp(v:&Version,c:&Cmp)->bool{
    match c.op {
        Op::Eq=>compare_versions(v,&c.version)==Ordering::Equal,
        Op::Lt=>compare_versions(v,&c.version)==Ordering::Less,
        Op::Le=>compare_versions(v,&c.version)!=Ordering::Greater,
        Op::Gt=>compare_versions(v,&c.version)==Ordering::Greater,
        Op::Ge=>compare_versions(v,&c.version)!=Ordering::Less,
    }
}
fn arm_matches(v:&Version,a:&Arm,include:bool)->bool{
    if !a.comps.iter().all(|c|satisfies_cmp(v,c)){return false;}
    include || v.pre.is_empty() || a.anchors.contains(&tuple(v))
}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool{
    r.arms.iter().any(|a|arm_matches(v,a,include_prerelease))
}
pub fn matches(v:&Version,r:&Range)->bool{matches_with(v,r,false)}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }
fn extrema(candidates:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let parsed:Vec<Version>=candidates.iter().enumerate().map(|(i,s)|parse_version(s).map_err(|error|CandidateError{index:i,error})).collect::<Result<_,_>>()?;
    let mut best:Option<usize>=None;
    for i in 0..parsed.len(){
        if !matches(&parsed[i],r){continue;}
        if best.is_none_or(|j| if max {compare_versions(&parsed[i],&parsed[j])==Ordering::Greater} else {compare_versions(&parsed[i],&parsed[j])==Ordering::Less}) {best=Some(i);}
    }
    Ok(best.map(|i|candidates[i].clone()))
}
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extrema(candidates,r,true)}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extrema(candidates,r,false)}

pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool{
    // Boundaries are represented by the exact comparator set. Candidate generation
    // covers all stable tuple boundaries and explicit admission anchors.
    for aa in &a.arms { for bb in &b.arms {
        let mut cs=aa.comps.clone(); cs.extend(bb.comps.clone());
        let mut probes=Vec::new();
        for c in &cs {
            probes.push(c.version.clone());
            if !c.version.pre.is_empty() {
                let mut x=c.version.clone(); x.pre.push("0".into()); probes.push(x);
                let mut x=c.version.clone(); x.pre.clear(); probes.push(x);
            }
            if c.version.pre.is_empty() {
                if let Ok(x)=bump(&c.version,2,0){probes.push(x);}
            }
        }
        probes.push(Version{major:0,minor:0,patch:0,pre:vec![],build:vec![]});
        for v in probes {
            let in_a=arm_matches(&v,aa,include_prerelease);
            let in_b=arm_matches(&v,bb,include_prerelease);
            if in_a&&in_b{return true;}
        }
    }}
    false
}
fn ver_string(v:&Version)->String{
    let mut s=format!("{}.{}.{}",v.major,v.minor,v.patch);
    if !v.pre.is_empty(){s.push('-');s.push_str(&v.pre.join("."));}
    s
}
pub fn canonical(r:&Range)->String{
    let mut strings=Vec::new();
    for a in &r.arms {
        let mut low:Option<(Version,bool)>=None; let mut high:Option<(Version,bool)>=None;
        let mut eq=false;
        for c in &a.comps {
            match c.op {
                Op::Eq=>{eq=true;low=Some((c.version.clone(),true));high=Some((c.version.clone(),true));}
                Op::Gt|Op::Ge=>{
                    let inc=matches!(c.op,Op::Ge);
                    if low.as_ref().is_none_or(|(v,i)| compare_versions(&c.version,v)==Ordering::Greater || (compare_versions(&c.version,v)==Ordering::Equal && !inc && *i)){low=Some((c.version.clone(),inc));}
                }
                Op::Lt|Op::Le=>{
                    let inc=matches!(c.op,Op::Le);
                    if high.as_ref().is_none_or(|(v,i)| compare_versions(&c.version,v)==Ordering::Less || (compare_versions(&c.version,v)==Ordering::Equal && !inc && *i)){high=Some((c.version.clone(),inc));}
                }
            }
        }
        let lo=low.map(|(v,i)|format!("{}{}",if i{">="}else{">"},ver_string(&v))).unwrap_or("*".into());
        let hi=high.map(|(v,i)|format!("{}{}",if i{"<="}else{"<"},ver_string(&v))).unwrap_or("*".into());
        let mut anchors=a.anchors.clone(); anchors.sort(); anchors.dedup();
        let an=if anchors.is_empty(){"-".to_string()}else{anchors.iter().map(|(x,y,z)|format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")};
        let _=eq;
        strings.push(format!("[{lo};{hi};pre={an}]"));
    }
    strings.sort(); strings.dedup();
    if strings.is_empty(){"EMPTY".into()}else{strings.join(" || ")}
}

#[derive(Clone, Debug, PartialEq, Eq)]

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name:String },
    DuplicateLabel { package:String, label:String },
    InvalidCandidate { package:String, index:usize, error:VersionError },
    InvalidRange { package:String, label:String, path:Vec<usize>, error:RangeError },
    NoSolution { package:String, labels:Vec<String> },
}
fn validate_expr(expr:&SetExpr,package:&str,label:&str,path:&mut Vec<usize>,out:&mut Vec<Range>)->Result<(),SelectError>{
    match expr {
        SetExpr::Range(s)=>parse_range(s).map(|r|out.push(r)).map_err(|error|SelectError::InvalidRange{package:package.into(),label:label.into(),path:path.clone(),error}),
        SetExpr::Union(xs)|SetExpr::Intersection(xs)=>{for (i,x) in xs.iter().enumerate(){path.push(i);validate_expr(x,package,label,path,out)?;path.pop();}Ok(())}
    }
}
fn eval_expr(expr:&SetExpr,v:&Version,include:bool,iter:&mut impl Iterator<Item=&Range>)->bool{
    match expr {
        SetExpr::Range(_)=>iter.next().is_some_and(|r|matches_with(v,r,include)),
        SetExpr::Union(xs)=>xs.iter().any(|x|eval_expr(x,v,include,iter)),
        SetExpr::Intersection(xs)=>xs.iter().all(|x|eval_expr(x,v,include,iter)),
    }
}
fn conflict_labels(pkg:&Package,versions:&[Version],include:bool,parsed:&[Vec<Range>])->Vec<String>{
    let n=pkg.constraints.len();
    if versions.is_empty(){return vec![];}
    for size in 1..=n {
        let mut found:Vec<Vec<String>>=Vec::new();
        fn rec(start:usize,size:usize,chosen:&mut Vec<usize>,n:usize,found:&mut Vec<Vec<String>>,pkg:&Package,versions:&[Version],include:bool,parsed:&[Vec<Range>]){
            if chosen.len()==size{
                let sat=versions.iter().any(|v|chosen.iter().all(|&ci|{
                    let mut it=parsed[ci].iter();
                    eval_expr(&pkg.constraints[ci].expr,v,include,&mut it)
                }));
                if !sat {let mut labels:Vec<String>=chosen.iter().map(|&i|pkg.constraints[i].label.clone()).collect();labels.sort();found.push(labels);}
                return;
            }
            for i in start..n{chosen.push(i);rec(i+1,size,chosen,n,found,pkg,versions,include,parsed);chosen.pop();}
        }
        rec(0,size,&mut Vec::new(),n,&mut found,pkg,versions,include,parsed);
        if !found.is_empty(){found.sort();return found.remove(0);}
    }
    vec![]
}
pub fn select_packages(packages:&[Package],preference:Preference,include_prerelease:bool)->Result<Vec<Selection>,SelectError>{
    let mut names=std::collections::HashSet::new();
    for p in packages {if !names.insert(p.name.clone()){return Err(SelectError::DuplicatePackage{name:p.name.clone()});}}
    for p in packages {
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints {if !labels.insert(c.label.clone()){return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()});}}
    }
    let mut versions_all=Vec::new();
    for p in packages {
        let mut vs=Vec::new();
        for (i,s) in p.candidates.iter().enumerate(){vs.push(parse_version(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);}
        versions_all.push(vs);
    }
    let mut ranges_all=Vec::new();
    for p in packages {
        let mut pr=Vec::new();
        for c in &p.constraints {let mut rs=Vec::new();validate_expr(&c.expr,&p.name,&c.label,&mut Vec::new(),&mut rs)?;pr.push(rs);}
        ranges_all.push(pr);
    }
    let mut result=Vec::new();
    for (pi,p) in packages.iter().enumerate(){
        let vs=&versions_all[pi]; let pr=&ranges_all[pi];
        let mut eligible=Vec::new();
        for i in 0..vs.len(){
            let ok=p.constraints.iter().enumerate().all(|(ci,c)|{let mut it=pr[ci].iter();eval_expr(&c.expr,&vs[i],include_prerelease,&mut it)});
            if ok {eligible.push(i);}
        }
        if eligible.is_empty(){
            return Err(SelectError::NoSolution{package:p.name.clone(),labels:conflict_labels(p,vs,include_prerelease,pr)});
        }
        let stable:Vec<usize>=eligible.iter().copied().filter(|&i|vs[i].pre.is_empty()).collect();
        let pool=if matches!(preference,Preference::StableFirst)&&!stable.is_empty(){stable.as_slice()}else{eligible.as_slice()};
        let mut best=pool[0];
        for &i in &pool[1..]{
            let ord=compare_versions(&vs[i],&vs[best]);
            let choose=match preference {Preference::Lowest=>ord==Ordering::Less,_=>ord==Ordering::Greater};
            if choose {best=i;}
        }
        result.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }
    Ok(result)
}
