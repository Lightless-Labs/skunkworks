#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_group(input: &[u8], out: &mut String) {
    let a = input[0];
    let b = if input.len() > 1 { input[1] } else { 0 };
    let c = if input.len() > 2 { input[2] } else { 0 };

    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if input.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if input.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity((input.len() + 2) / 3 * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        encode_group(chunk, &mut out);
    }

    let tail = chunks.remainder();
    if !tail.is_empty() {
        encode_group(tail, &mut out);
    }

    out
}

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

#[derive(Default)]
struct DecodeState {
    group: Vec<Symbol>,
    pending_padding: Option<usize>,
    terminal_padding: bool,
    next_offset: usize,
}

impl DecodeState {
    fn consume(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if let Some(padding_offset) = self.pending_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_padding = None;
            self.group.push(Symbol { byte, offset });
            self.emit_group(out);
            self.terminal_padding = true;
            let _ = padding_offset;
            return Ok(());
        }

        let position = self.group.len();
        if byte == b'=' {
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_padding = Some(offset);
                    self.group.push(Symbol { byte, offset });
                    return Ok(());
                }
                3 => {
                    self.group.push(Symbol { byte, offset });
                    self.emit_group(out);
                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
        self.group.push(Symbol { byte, offset });
        if self.group.len() == 4 {
            let _ = value;
            self.emit_group(out);
        }
        Ok(())
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) {
        let a = digit(self.group[0].byte).unwrap();
        let b = digit(self.group[1].byte).unwrap();

        out.push((a << 2) | (b >> 4));

        if self.group[2].byte != b'=' {
            let c = digit(self.group[2].byte).unwrap();
            out.push((b << 4) | (c >> 2));

            if self.group[3].byte != b'=' {
                let d = digit(self.group[3].byte).unwrap();
                out.push((c << 6) | d);
            }
        }

        self.group.clear();
    }

    fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(offset) = self.pending_padding {
            return Err(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            return Err(Error::Truncated {
                offset: self.group[0].offset,
            });
        }
        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut state = DecodeState::default();
    let mut out = Vec::new();

    for (offset, &byte) in input.iter().enumerate() {
        state.consume(byte, offset, &mut out)?;
    }

    state.finish()?;
    Ok(out)
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let mut out = String::with_capacity(complete_len / 3 * 4);
        for chunk in combined[..complete_len].chunks_exact(3) {
            encode_group(chunk, &mut out);
        }

        self.tail.clear();
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::new();
        if !self.tail.is_empty() {
            encode_group(&self.tail, &mut out);
        }
        self.tail.clear();
        self.finished = true;
        Ok(out)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    state: DecodeState,
    sticky_error: Option<Error>,
    finished: bool,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            state: DecodeState::default(),
            sticky_error: None,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.state.next_offset;
            self.state.next_offset += 1;
            if let Err(error) = self.state.consume(byte, offset, &mut out) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.state.finish() {
            Ok(out) => {
                self.finished = true;
                Ok(out)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
