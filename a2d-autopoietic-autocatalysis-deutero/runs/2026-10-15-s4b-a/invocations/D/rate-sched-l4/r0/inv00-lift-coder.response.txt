```rust
use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Runtime {
    state: State,
    age_from: u64,
    remaining: Option<u64>,
    segment_start: u64,
    segment_order: u64,
}

pub struct Pool {
    jobs: Vec<Job>,
    states: Vec<Runtime>,
    ids: HashMap<String, usize>,
    dependencies: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    started: bool,
    next_segment_order: u64,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job
                .tag
                .as_ref()
                .is_some_and(|tag| !valid_token(tag, 1, 8))
            {
                return Err(Error::BadTag(i));
            }
            if ids.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            ids.insert(job.id.clone(), i);
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dep) {
                    Some(&index) => dependencies[i].push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        let mut colors = vec![0u8; jobs.len()];
        fn visit(i: usize, deps: &[Vec<usize>], colors: &mut [u8]) -> bool {
            if colors[i] == 1 {
                return true;
            }
            if colors[i] == 2 {
                return false;
            }
            colors[i] = 1;
            for &d in &deps[i] {
                if visit(d, deps, colors) {
                    return true;
                }
            }
            colors[i] = 2;
            false
        }
        for i in 0..jobs.len() {
            if visit(i, &dependencies, &mut colors) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|job| Runtime {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                age_from: job.release_at,
                remaining: None,
                segment_start: 0,
                segment_order: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            ids,
            dependencies,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            started: false,
            next_segment_order: 0,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.ids.get(id).ok_or(Error::Unknown)?;
        Ok(self.states[i].state.clone())
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.started && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut t = if self.started { self.horizon } else { 0 };
        let mut first_phase = !self.started;

        loop {
            if first_phase || t > self.horizon {
                self.process_timestamp(t, &mut lines);
            } else if self.started {
                // The prior horizon was already processed.
            }

            if t == until {
                break;
            }

            let mut next: Option<u64> = None;
            for (i, rt) in self.states.iter().enumerate() {
                match rt.state {
                    State::Waiting { ready_at, .. } if ready_at > t => {
                        next = Some(next.map_or(ready_at, |n| n.min(ready_at)));
                    }
                    State::Running { finish_at, .. } if finish_at > t => {
                        next = Some(next.map_or(finish_at, |n| n.min(finish_at)));
                    }
                    _ => {}
                }
                let _ = i;
            }
            match next {
                Some(nt) if nt <= until => {
                    t = nt;
                    first_phase = true;
                }
                _ => break,
            }
        }

        self.horizon = until;
        self.started = true;
        Ok(lines)
    }

    fn process_timestamp(&mut self, t: u64, lines: &mut Vec<String>) {
        let mut completed: Vec<usize> = self
            .states
            .iter()
            .enumerate()
            .filter_map(|(i, rt)| match rt.state {
                State::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        completed.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));

        for i in completed {
            let attempt = match self.states[i].state {
                State::Running { attempt, .. } => attempt,
                _ => unreachable!(),
            };
            let ok = attempt > self.jobs[i].failures;
            lines.push(format!(
                "{} finish {} {} {}",
                t,
                self.jobs[i].id,
                attempt,
                if ok { "ok" } else { "fail" }
            ));
            if ok {
                self.states[i].state = State::Succeeded;
                self.states[i].remaining = None;
            } else if attempt >= self.jobs[i].max_attempts {
                self.states[i].state = State::Failed;
                self.states[i].remaining = None;
            } else {
                let shift = u32::from(attempt.saturating_sub(1));
                let delay = self
                    .retry_base
                    .checked_shl(shift)
                    .unwrap_or(u64::MAX)
                    .min(self.retry_cap);
                let ready_at = t.saturating_add(delay);
                self.states[i].state = State::Waiting { ready_at, attempt };
                self.states[i].age_from = ready_at;
                self.states[i].remaining = None;
                lines.push(format!(
                    "{} retry {} {} {}",
                    t, self.jobs[i].id, attempt, ready_at
                ));
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if !matches!(self.states[i].state, State::Waiting { .. }) {
                    continue;
                }
                if let Some(&dep) = self.dependencies[i].iter().find(|&&d| {
                    matches!(
                        self.states[d].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    self.states[i].state = State::Blocked;
                    self.states[i].remaining = None;
                    newly_blocked.push((i, dep));
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|(a, _), (b, _)| {
            self.jobs[*a].id.as_bytes().cmp(self.jobs[*b].id.as_bytes())
        });
        for (i, dep) in newly_blocked {
            lines.push(format!(
                "{} skip {} {}",
                t, self.jobs[i].id, self.jobs[dep].id
            ));
        }

        loop {
            let mut candidates: Vec<usize> = (0..self.jobs.len())
                .filter(|&i| {
                    matches!(
                        self.states[i].state,
                        State::Waiting { ready_at, .. }
                            if ready_at <= t
                                && self.dependencies[i].iter().all(|&d| {
                                    matches!(self.states[d].state, State::Succeeded)
                                })
                    )
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut selected = None;
            for candidate in candidates {
                if let Some(victim) = self.choose_victim(candidate, t) {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
                if self.can_start_without_preemption(candidate) {
                    selected = Some((candidate, None));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(victim) = victim {
                let finish_at = match self.states[victim].state {
                    State::Running { finish_at, .. } => finish_at,
                    _ => unreachable!(),
                };
                let remaining = finish_at.saturating_sub(t);
                let attempt = match self.states[victim].state {
                    State::Running { attempt, .. } => attempt,
                    _ => unreachable!(),
                };
                self.states[victim].state = State::Waiting { ready_at: t, attempt };
                self.states[victim].age_from = t;
                self.states[victim].remaining = Some(remaining);
                lines.push(format!(
                    "{} preempt {} by {}",
                    t, self.jobs[victim].id, self.jobs[candidate].id
                ));
            }

            let (attempt, is_resume) = match self.states[candidate].state {
                State::Waiting { attempt, .. } => {
                    (attempt, self.states[candidate].remaining.is_some())
                }
                _ => unreachable!(),
            };
            let (attempt, duration) = if is_resume {
                (
                    attempt,
                    self.states[candidate].remaining.take().unwrap_or(0),
                )
            } else {
                (attempt.saturating_add(1), self.jobs[candidate].duration)
            };
            let finish_at = t.saturating_add(duration);
            self.next_segment_order = self.next_segment_order.saturating_add(1);
            self.states[candidate].state = State::Running { attempt, finish_at };
            self.states[candidate].segment_start = t;
            self.states[candidate].segment_order = self.next_segment_order;
            lines.push(format!(
                "{} start {} {} {}",
                t, self.jobs[candidate].id, attempt, duration
            ));
        }
    }

    fn effective_priority(&self, i: usize, t: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[i].priority;
        }
        let elapsed = t.saturating_sub(self.states[i].age_from);
        let gain = elapsed / self.aging_ms;
        (u64::from(self.jobs[i].priority) + gain).min(9) as u8
    }

    fn can_start_without_preemption(&self, candidate: usize) -> bool {
        let running = self.running_jobs();
        running.len() < self.capacity
            && self.tag_is_free(candidate, &running)
    }

    fn choose_victim(&self, candidate: usize, t: u64) -> Option<usize> {
        let running = self.running_jobs();
        let candidate_tag = self.jobs[candidate].tag.as_deref();

        let mut victims: Vec<usize> = running
            .iter()
            .copied()
            .filter(|&v| {
                self.jobs[candidate].priority
                    >= self.jobs[v].priority.saturating_add(2)
                    && (running.len() == self.capacity || self.tag_is_occupied_by(candidate_tag, v, &running))
                    && self.can_resolve_by_removing(candidate_tag, v, &running)
                    && matches!(
                        self.states[v].state,
                        State::Running { finish_at, .. } if finish_at > t
                    )
            })
            .collect();

        victims.sort_by(|&a, &b| {
            self.jobs[a]
                .priority
                .cmp(&self.jobs[b].priority)
                .then_with(|| {
                    self.states[b]
                        .segment_order
                        .cmp(&self.states[a].segment_order)
                })
                .then_with(|| {
                    self.jobs[b]
                        .id
                        .as_bytes()
                        .cmp(self.jobs[a].id.as_bytes())
                })
        });
        victims.into_iter().next()
    }

    fn can_resolve_by_removing(
        &self,
        candidate_tag: Option<&str>,
        victim: usize,
        running: &[usize],
    ) -> bool {
        let slots_after = running.len().saturating_sub(1) < self.capacity;
        let tag_free_after = match candidate_tag {
            None => true,
            Some(tag) => !running.iter().any(|&i| {
                i != victim && self.jobs[i].tag.as_deref() == Some(tag)
            }),
        };
        slots_after && tag_free_after
    }

    fn tag_is_occupied_by(
        &self,
        candidate_tag: Option<&str>,
        victim: usize,
        running: &[usize],
    ) -> bool {
        match candidate_tag {
            None => false,
            Some(tag) => {
                self.jobs[victim].tag.as_deref() == Some(tag)
                    && running.iter().any(|&i| self.jobs[i].tag.as_deref() == Some(tag))
            }
        }
    }

    fn tag_is_free(&self, candidate: usize, running: &[usize]) -> bool {
        match self.jobs[candidate].tag.as_deref() {
            None => true,
            Some(tag) => !running.iter().any(|&i| self.jobs[i].tag.as_deref() == Some(tag)),
        }
    }

    fn running_jobs(&self) -> Vec<usize> {
        self.states
            .iter()
            .enumerate()
            .filter_map(|(i, rt)| matches!(rt.state, State::Running { .. }).then_some(i))
            .collect()
    }
}

fn valid_token(s: &str, min: usize, max: usize) -> bool {
    let bytes = s.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}
```
