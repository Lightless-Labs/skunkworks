#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    modifier: Modifier,
    pos: usize,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone)]
struct Expression {
    op: Option<char>,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'_'
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let op = if content_start < end && b"+#./;?&".contains(&bytes[content_start]) {
        let c = bytes[content_start] as char;
        if c == '+' || c == '#' || c == '.' || c == '/' || c == ';' || c == '?' || c == '&' {
            content_start += 1;
            Some(c)
        } else {
            None
        }
    } else if content_start < end && b"=!@$|,".contains(&bytes[content_start]) {
        return Err(Error::UnsupportedOperator { pos: content_start });
    } else {
        None
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut piece_start = content_start;

    loop {
        let piece_end = bytes[piece_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| piece_start + offset)
            .unwrap_or(end);

        if piece_start == piece_end {
            return Err(Error::EmptyVariable { pos: piece_start });
        }

        let piece = &bytes[piece_start..piece_end];
        let modifier_offset = piece.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_offset.map(|offset| piece_start + offset).unwrap_or(piece_end);

        if name_end == piece_start {
            return Err(Error::InvalidName { pos: piece_start });
        }

        let name_bytes = &bytes[piece_start..name_end];
        let mut invalid = None;
        if !is_name_start(name_bytes[0]) {
            invalid = Some(piece_start);
        } else {
            for (i, &b) in name_bytes.iter().enumerate().skip(1) {
                if !is_name_continue(b) {
                    invalid = Some(piece_start + i);
                    break;
                }
            }
        }
        if let Some(pos) = invalid {
            return Err(Error::InvalidName { pos });
        }

        let name = template[piece_start..name_end].to_owned();
        let modifier = if let Some(offset) = modifier_offset {
            let mod_pos = piece_start + offset;
            let suffix = &bytes[mod_pos..piece_end];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.len() >= 2
                && suffix[0] == b':'
                && suffix.len() <= 5
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut n = 0usize;
                for &digit in &suffix[1..] {
                    n = n * 10 + usize::from(digit - b'0');
                }
                if (1..=9999).contains(&n) {
                    Modifier::Prefix(n)
                } else {
                    return Err(Error::InvalidModifier { pos: mod_pos });
                }
            } else {
                return Err(Error::InvalidModifier { pos: mod_pos });
            }
        } else {
            Modifier::None
        };

        if seen.iter().any(|prior| prior == &name) {
            return Err(Error::DuplicateVariable { pos: piece_start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            modifier,
            pos: piece_start,
        });

        if piece_end == end {
            break;
        }
        piece_start = piece_end + 1;
    }

    Ok(Expression { op, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit() {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let is_reserved = reserved
            && matches!(
                b,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
                    | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );
        if unreserved || is_reserved {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn binding<'a>(vars: &'a [(&str, Value)], name: &str) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn encode_scalar(value: &str, modifier: Modifier, reserved: bool) -> String {
    let selected = match modifier {
        Modifier::Prefix(n) => value.chars().take(n).collect::<String>(),
        _ => value.to_owned(),
    };
    encode(&selected, reserved)
}

fn named_field(name: &str, value: String, op: char) -> String {
    if op == ';' && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.op.unwrap_or('\0');
    let reserved = matches!(op, '+' | '#');
    let separator = match op {
        '.' => ".",
        '/' => "/",
        ';' | '?' | '&' => "&".replace('&', if op == ';' { ';' } else { '&' }).as_str().to_owned().leak(),
        _ => ",",
    };

    let mut fields = Vec::<String>::new();
    for spec in &expr.specs {
        let Some(value) = binding(vars, &spec.name) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(s) => {
                fields.push(encode_scalar(s, spec.modifier, reserved).pipe(|v| {
                    if matches!(op, ';' | '?' | '&') {
                        named_field(&spec.name, v, op)
                    } else {
                        v
                    }
                }));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(op, ';' | '?' | '&') {
                            named_field(&spec.name, encoded, op)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let encoded = items.iter().map(|item| encode(item, reserved)).collect::<Vec<_>>().join(",");
                    fields.push(if matches!(op, ';' | '?' | '&') {
                        named_field(&spec.name, encoded, op)
                    } else {
                        encoded
                    });
                }
            }
            Value::Map(entries) => {
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(key, _)| key == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }
                if entries.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, val) in entries {
                        let k = encode(key, reserved);
                        let v = encode(val, reserved);
                        fields.push(if op == ';' && v.is_empty() {
                            k
                        } else {
                            format!("{k}={v}")
                        });
                    }
                } else {
                    let components = entries
                        .iter()
                        .flat_map(|(key, val)| [encode(key, reserved), encode(val, reserved)])
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(op, ';' | '?' | '&') {
                        named_field(&spec.name, components, op)
                    } else {
                        components
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let joined = fields.join(&separator);
    Ok(match op {
        '#' | '.' | '/' | ';' | '?' | '&' => format!("{op}{joined}"),
        _ => joined,
    })
}

trait Pipe: Sized {
    fn pipe<T>(self, f: impl FnOnce(Self) -> T) -> T {
        f(self)
    }
}
impl<T> Pipe for T {}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut literal_start = 0;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(offset) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + offset;
                if let Some(inner) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + inner });
                }
                let expr = parse_expression(template, open, close)?;
                output.push_str(&template[literal_start..open]);
                output.push_str(&expand_expression(&expr, vars)?);
                i = close + 1;
                literal_start = i;
            }
            _ => i += 1,
        }
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
