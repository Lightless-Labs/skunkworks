#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    modifier: Modifier,
    pos: usize,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone)]
struct Expression {
    op: Option<char>,
    specs: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;

    let op = if content_start < end
        && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let op = bytes[content_start] as char;
        content_start += 1;
        Some(op)
    } else if content_start < end
        && matches!(bytes[content_start], b'=' | b'!' | b'@' | b'$' | b'|' )
    {
        return Err(Error::UnsupportedOperator { pos: content_start });
    } else {
        None
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut piece_start = content_start;

    loop {
        let piece_end = bytes[piece_start..end]
            .iter()
            .position(|&byte| byte == b',')
            .map(|offset| piece_start + offset)
            .unwrap_or(end);

        if piece_start == piece_end {
            return Err(Error::EmptyVariable { pos: piece_start });
        }

        let piece = &bytes[piece_start..piece_end];
        let modifier_offset = piece.iter().position(|&byte| byte == b'*' || byte == b':');
        let name_end = modifier_offset
            .map(|offset| piece_start + offset)
            .unwrap_or(piece_end);

        if name_end == piece_start {
            return Err(Error::InvalidName { pos: piece_start });
        }

        let name_bytes = &bytes[piece_start..name_end];
        let invalid_name_pos = if !is_name_start(name_bytes[0]) {
            Some(piece_start)
        } else {
            name_bytes
                .iter()
                .enumerate()
                .skip(1)
                .find(|(_, byte)| !is_name_continue(**byte))
                .map(|(offset, _)| piece_start + offset)
        };

        if let Some(pos) = invalid_name_pos {
            return Err(Error::InvalidName { pos });
        }

        let name = template[piece_start..name_end].to_owned();
        let modifier = if let Some(offset) = modifier_offset {
            let modifier_pos = piece_start + offset;
            let suffix = &bytes[modifier_pos..piece_end];

            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.len() >= 2
                && suffix.len() <= 5
                && suffix[0] == b':'
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut value = 0usize;
                for &digit in &suffix[1..] {
                    value = value * 10 + usize::from(digit - b'0');
                }
                if (1..=9999).contains(&value) {
                    Modifier::Prefix(value)
                } else {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        } else {
            Modifier::None
        };

        if seen.iter().any(|prior| prior == &name) {
            return Err(Error::DuplicateVariable { pos: piece_start });
        }

        seen.push(name.clone());
        specs.push(Specifier {
            name,
            modifier,
            pos: piece_start,
        });

        if piece_end == end {
            break;
        }
        piece_start = piece_end + 1;
    }

    Ok(Expression { op, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%'
            && index + 2 < bytes.len()
            && bytes[index + 1].is_ascii_hexdigit()
            && bytes[index + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[index + 1] as char).to_ascii_uppercase());
            output.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        let unreserved =
            byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let is_reserved = reserved
            && matches!(
                byte,
                b':' | b'/'
                    | b'?'
                    | b'#'
                    | b'['
                    | b']'
                    | b'@'
                    | b'!'
                    | b'$'
                    | b'&'
                    | b'\''
                    | b'('
                    | b')'
                    | b'*'
                    | b'+'
                    | b','
                    | b';'
                    | b'='
            );

        if unreserved || is_reserved {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

fn binding<'a>(vars: &'a [(&str, Value)], name: &str) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| value)
}

fn encode_scalar(value: &str, modifier: Modifier, reserved: bool) -> String {
    let selected = match modifier {
        Modifier::Prefix(count) => value.chars().take(count).collect::<String>(),
        Modifier::None | Modifier::Explode => value.to_owned(),
    };
    encode(&selected, reserved)
}

fn named_field(name: &str, value: String, op: char) -> String {
    if op == ';' && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.op.unwrap_or('\0');
    let reserved = matches!(op, '+' | '#');
    let separator = match op {
        '.' => ".",
        '/' => "/",
        ';' => ";",
        '?' | '&' => "&",
        _ => ",",
    };

    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let Some(value) = binding(vars, &spec.name) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(value) => {
                let encoded = encode_scalar(value, spec.modifier, reserved);
                fields.push(if matches!(op, ';' | '?' | '&') {
                    named_field(&spec.name, encoded, op)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(op, ';' | '?' | '&') {
                            named_field(&spec.name, encoded, op)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let encoded = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(op, ';' | '?' | '&') {
                        named_field(&spec.name, encoded, op)
                    } else {
                        encoded
                    });
                }
            }
            Value::Map(entries) => {
                for index in 0..entries.len() {
                    if entries[..index]
                        .iter()
                        .any(|(key, _)| key == &entries[index].0)
                    {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }

                if entries.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        fields.push(if op == ';' && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else {
                    let components = entries
                        .iter()
                        .flat_map(|(key, value)| {
                            [encode(key, reserved), encode(value, reserved)]
                        })
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(op, ';' | '?' | '&') {
                        named_field(&spec.name, components, op)
                    } else {
                        components
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let joined = fields.join(separator);
    Ok(match op {
        '#' | '.' | '/' | ';' | '?' | '&' => format!("{op}{joined}"),
        _ => joined,
    })
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut literal_start = 0;
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open = index;
                let Some(offset) = bytes[index + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: open });
                };

                let close = index + 1 + offset;
                if let Some(inner_offset) = bytes[index + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + inner_offset,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                output.push_str(&template[literal_start..open]);
                output.push_str(&expand_expression(&expression, vars)?);
                index = close + 1;
                literal_start = index;
            }
            _ => index += 1,
        }
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}
