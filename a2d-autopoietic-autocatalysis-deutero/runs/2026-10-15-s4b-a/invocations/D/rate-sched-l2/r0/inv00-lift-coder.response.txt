#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        if let Some(i) = bytes.iter().position(|b| b.is_ascii_whitespace() && *b != b' ') {
            return Err(CronError::Layout(i));
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, b)| (*b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let sep = spaces[0];
        if sep == 0 || sep + 1 == bytes.len() {
            return Err(CronError::Layout(sep));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(&text[..sep], 0, 59, &mut minutes)?;
        parse_field(&text[sep + 1..], sep + 1, 23, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_number(fragment: &str, start: usize, max: usize) -> Result<usize, CronError> {
    let b = fragment.as_bytes();

    if fragment.is_empty() {
        return Err(CronError::Syntax(start));
    }
    if let Some(i) = b.iter().position(|c| !c.is_ascii_digit()) {
        return Err(CronError::Syntax(start + i));
    }
    if b.len() > 1 && b[0] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut n = 0usize;
    for &digit in b {
        n = match n.checked_mul(10).and_then(|v| v.checked_add((digit - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
    }
    if n > max {
        return Err(CronError::Bounds(start));
    }
    Ok(n)
}

fn parse_field(field: &str, offset: usize, max: usize, selected: &mut [bool]) -> Result<(), CronError> {
    let mut item_start = 0usize;

    loop {
        let item_end = field[item_start..]
            .find(',')
            .map(|i| item_start + i)
            .unwrap_or(field.len());
        let item = &field[item_start..item_end];
        let abs_start = offset + item_start;

        if item.is_empty() {
            return Err(CronError::Syntax(abs_start));
        }

        let slash = item.find('/');
        let base_end = slash.unwrap_or(item.len());
        if let Some(slash_pos) = slash {
            if let Some(second) = item[slash_pos + 1..].find('/') {
                return Err(CronError::Syntax(abs_start + slash_pos + 1 + second));
            }
        }

        let base = &item[..base_end];
        let is_star = base == "*";
        let dash = base.find('-');

        let (start, end, dash_abs) = if is_star {
            (0, max, None)
        } else if let Some(dash_pos) = dash {
            let left = parse_number(&base[..dash_pos], abs_start, max)?;
            let right = parse_number(
                &base[dash_pos + 1..],
                abs_start + dash_pos + 1,
                max,
            )?;
            if left > right {
                return Err(CronError::Descending(abs_start + dash_pos));
            }
            (left, right, Some(abs_start + dash_pos))
        } else {
            if slash.is_some() {
                return Err(CronError::Syntax(abs_start + slash.unwrap()));
            }
            let n = parse_number(base, abs_start, max)?;
            (n, n, None)
        };

        let step = if let Some(slash_pos) = slash {
            if !is_star && dash_abs.is_none() {
                return Err(CronError::Syntax(abs_start + slash_pos));
            }
            let step_start = abs_start + slash_pos + 1;
            let step_text = &item[slash_pos + 1..];
            let step_max = if max == 59 { 60 } else { 24 };
            let value = parse_number(step_text, step_start, step_max)
                .map_err(|e| match e {
                    CronError::Bounds(_) => CronError::Step(step_start),
                    other => other,
                })?;
            if value == 0 {
                return Err(CronError::Step(step_start));
            }
            value
        } else {
            1
        };

        let mut n = start;
        loop {
            selected[n] = true;
            if end - n < step {
                break;
            }
            n += step;
        }

        if item_end == field.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let now_is_boundary = now % 60_000 == 0;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let mut matching = Vec::new();
            if first_minute <= last_minute {
                for minute in first_minute..=last_minute {
                    if job.cron.matches(minute) {
                        matching.push(minute * 60_000);
                    }
                }
            }

            match job.policy {
                Policy::All => {
                    for at in matching {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Once => {
                    if let Some(at) = matching.last().copied() {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now_is_boundary && matching.last().copied() == Some(now) {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
