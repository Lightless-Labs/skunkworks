#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
}

#[derive(Debug)]
struct Expression {
    operator: Operator,
    names: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut expression_index = 0;
    let mut cursor = 0;

    while cursor < template.len() {
        if template.as_bytes()[cursor] == b'{' {
            let expression = &expressions[expression_index];
            expression_index += 1;

            let start = cursor + 1;
            let end = template[start..]
                .find('}')
                .map(|offset| start + offset)
                .expect("template was validated");

            let mut fields = Vec::new();
            for name in &expression.names {
                if let Some(value) = lookup(name, vars) {
                    fields.push(encode(value, expression.operator));
                }
            }

            if !fields.is_empty() {
                if expression.operator == Operator::Fragment {
                    output.push('#');
                }
                output.push_str(&fields.join(","));
            }

            cursor = end + 1;
        } else {
            let ch = template[cursor..]
                .chars()
                .next()
                .expect("cursor is on a character boundary");
            output.push(ch);
            cursor += ch.len_utf8();
        }
    }

    Ok(output)
}

fn parse_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[cursor + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(offset) => cursor + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[cursor + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + offset,
                    });
                }

                expressions.push(parse_expression(template, open, close)?);
                cursor = close + 1;
            }
            _ => {
                let ch = template[cursor..]
                    .chars()
                    .next()
                    .expect("cursor is on a character boundary");
                cursor += ch.len_utf8();
            }
        }
    }

    Ok(expressions)
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let mut content_start = open + 1;
    let operator = match template.as_bytes()[content_start] {
        b'+' => {
            content_start += 1;
            Operator::Reserved
        }
        b'#' => {
            content_start += 1;
            Operator::Fragment
        }
        b'.' | b'/' | b';' | b'?' | b'&' => {
            return Err(Error::UnsupportedOperator {
                pos: content_start,
            });
        }
        _ => Operator::Simple,
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let content = &template[content_start..close];
    let mut names = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut spec_start = content_start;

    for specifier in content.split(',') {
        if specifier.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        validate_name(specifier, spec_start)?;
        if !seen.insert(specifier) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        names.push(specifier.to_owned());
        spec_start += specifier.len() + 1;
    }

    Ok(Expression { operator, names })
}

fn validate_name(name: &str, start: usize) -> Result<(), Error> {
    for (index, byte) in name.bytes().enumerate() {
        let valid = if index == 0 {
            byte.is_ascii_alphabetic() || byte == b'_'
        } else {
            byte.is_ascii_alphanumeric() || byte == b'_'
        };
        if !valid {
            return Err(Error::InvalidName { pos: start + index });
        }
    }
    Ok(())
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn encode(value: &str, operator: Operator) -> String {
    let mut output = String::new();
    let bytes = value.as_bytes();
    let mut cursor = 0;

    while cursor < bytes.len() {
        let byte = bytes[cursor];

        if byte == b'%' && cursor + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[cursor + 1]), hex_value(bytes[cursor + 2]))
            {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                cursor += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (operator != Operator::Simple && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        cursor += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    b"0123456789ABCDEF"[value as usize] as char
}
