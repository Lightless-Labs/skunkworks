```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let segments = parse_path(head)?;

        let mut best: Option<(usize, Match)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &segments) {
                let replace = match &best {
                    None => true,
                    Some((best_id, _)) => {
                        let best_route = &self.routes[*best_id];
                        route.rank > best_route.rank
                            || (route.rank == best_route.rank && route.id < *best_id)
                    }
                };
                if replace {
                    best = Some((
                        route.id,
                        Match {
                            route: route.id,
                            params,
                        },
                    ));
                }
            }
        }

        best.map(|(_, matched)| matched).ok_or(Error::NotFound)
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let parts = split_segments(input).map_err(|at| Error::InvalidPattern { at })?;
    let mut parsed = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (index, (start, raw)) in parts.iter().enumerate() {
        if raw.starts_with(':') || raw.starts_with('*') {
            let is_wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if is_wildcard && index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            parsed.push(if is_wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            parsed.push(Segment::Static(decoded));
        }
    }

    Ok(parsed)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let parts = split_segments(input).map_err(|at| Error::InvalidPath { at })?;
    let mut parsed = Vec::with_capacity(parts.len());
    for (start, raw) in parts {
        parsed.push(
            decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })?,
        );
    }
    Ok(parsed)
}

fn split_segments(input: &str) -> Result<Vec<(usize, &str)>, usize> {
    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(i);
                }
                return Err(i);
            }
            if start == i {
                return Err(start);
            }
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(bytes.len() - 1);
    }
    parts.push((start, &input[start..]));
    Ok(parts)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = hex_value(bytes[i + 1]).ok_or(offset + i)?;
            let lo = hex_value(bytes[i + 2]).ok_or(offset + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(byte) {
                return Err(offset + i);
            }
            decoded.push(byte);
            i += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segments are ASCII"))
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
```
