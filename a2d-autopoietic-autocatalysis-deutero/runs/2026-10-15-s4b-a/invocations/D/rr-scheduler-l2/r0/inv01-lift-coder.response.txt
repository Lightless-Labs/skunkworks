#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completions = vec![None; jobs.len()];
    let mut ready = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut time = 0u64;
    let mut event_index = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let index = events[event_index];
            event_index += 1;
            if jobs[index].ticks == 0 {
                completions[index] = Some(u64::from(jobs[index].arrival));
                completed_count += 1;
            } else {
                ready.push_back((index, jobs[index].ticks));
            }
        }

        let Some((index, mut remaining)) = ready.pop_front() else {
            if event_index >= events.len() {
                break;
            }
            let next_time = u64::from(jobs[events[event_index]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
            continue;
        };

        if let Some(last_id) = last_dispatched {
            if last_id != jobs[index].id {
                time += u64::from(switch_cost);
            }
        }

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arrived = events[event_index];
            event_index += 1;
            if jobs[arrived].ticks == 0 {
                completions[arrived] = Some(u64::from(jobs[arrived].arrival));
                completed_count += 1;
            } else {
                ready.push_back((arrived, jobs[arrived].ticks));
            }
        }

        let start = time;
        let consumed = remaining.min(quantum);
        time += u64::from(consumed);
        remaining -= consumed;
        let finished = remaining == 0;
        runs.push(Run {
            id: jobs[index].id,
            start,
            end: time,
            finished,
        });

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arrived = events[event_index];
            event_index += 1;
            if jobs[arrived].ticks == 0 {
                completions[arrived] = Some(u64::from(jobs[arrived].arrival));
                completed_count += 1;
            } else {
                ready.push_back((arrived, jobs[arrived].ticks));
            }
        }

        if finished {
            completions[index] = Some(time);
            completed_count += 1;
        } else {
            ready.push_back((index, remaining));
        }

        last_dispatched = Some(jobs[index].id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completions[index].unwrap_or(0)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
