fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())

fn valid_route_name(s: &str) -> bool {
    let mut bytes = s.bytes();

        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static(raw: &str) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_path(
    input: &str,
    pattern: bool,
) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut parts: Vec<(usize, &str)> = Vec::new();
    let mut start = 1usize;
    for (i, b) in bytes.iter().enumerate().skip(1) {
        if *b == b'/' {
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = bytes.len() - usize::from(trailing);
    parts.push((start, &input[start..end]));

    if trailing {
        parts.pop();
    }

    let mut segments = Vec::with_capacity(parts.len());
    let mut seen = std::collections::HashSet::new();

    for (at, raw) in parts {
        if raw.is_empty() {
            return Err(at);
        }

        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(at);
            }
            if marker == b'*' {
                // A wildcard must be the final nonempty segment.
                if input[at + raw.len()..].contains('/') {
                    return Err(at);
                }
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            let decoded = decode_static(raw).map_err(|offset| at + offset)?;
            segments.push(Segment::Static(decoded));
        }
    }

    if pattern {
        if let Some(pos) = segments.iter().position(|s| matches!(s, Segment::Wildcard(_))) {
            if pos + 1 != segments.len() {
                // The wildcard's own segment start takes precedence over later parsing.
                let wildcard_at = parts_start_for_wildcard(input, pos);
                return Err(wildcard_at);
            }
        }
    }

    Ok((segments, trailing))
}

fn parts_start_for_wildcard(input: &str, segment_index: usize) -> usize {
    let mut current = 0usize;
    let mut start = 1usize;
    for (i, b) in input.as_bytes().iter().enumerate().skip(1) {
        if *b == b'/' {
            if current == segment_index {
                return start;
            }
            current += 1;
            start = i + 1;
        }
    }
    start
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true).map_err(pattern_error)?;

        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let mut rank = Vec::new();
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => {
                    rank.push(0);
                    break;
                }
            }
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }

        for route in &self.routes {
            if route.method != method {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if route.segments.len() != segments.len() {
                continue;
            }
            let same_shape = route.segments.iter().zip(&segments).all(|(a, b)| match (a, b) {
                (Segment::Static(x), Segment::Static(y)) => x == y,
                (Segment::Param(_), Segment::Param(_)) => true,
                (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                _ => false,
            });
            if same_shape {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, trailing) =
            parse_path(head, false).map_err(path_error)?;

        let mut path_candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &request_segments) {
                path_candidates.push((route, params));
            }
        }

        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let has_head = path_candidates.iter().any(|(r, _)| r.method == "HEAD");
        let effective_method = if method == "HEAD" && !has_head {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = path_candidates
            .iter()
            .filter(|(r, _)| r.method == effective_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (route, _) in &path_candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_owned());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                a.rank
                    .cmp(&b.rank)
                    .then_with(|| b.id.cmp(&a.id))
            })
            .expect("eligible routes are nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && winner.trailing != trailing
        {
            if winner.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let captures: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|s| match s {
                Segment::Param(n) | Segment::Wildcard(n) => Some(n.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &captures {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let extra = supplied
            .keys()
            .filter(|key| !captures.contains(&key.as_ref()))
            .min_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            if let Segment::Param(n) = segment {
                let value = supplied[n.as_str()];
                if !value.is_ascii()
                    || value.bytes().any(|b| !(0x20..=0x7e).contains(&b))
                    || value.is_empty()
                    || value.contains('/')
                {
                    return Err(Error::InvalidValue { name: n.clone() });
                }
            } else if let Segment::Wildcard(n) = segment {
                let value = supplied[n.as_str()];
                if !value.is_ascii()
                    || value.bytes().any(|b| !(0x20..=0x7e).contains(&b))
                    || (!value.is_empty()
                        && (value.starts_with('/')
                            || value.ends_with('/')
                            || value.split('/').any(|part| part.is_empty())))
                {
                    return Err(Error::InvalidValue { name: n.clone() });
                }
            }
        }

        let mut output = String::new();
        for segment in &route.segments {
            output.push('/');
            match segment {
                Segment::Static(value) => output.push_str(&encode_component(value)),
                Segment::Param(name) => output.push_str(&encode_component(supplied[name.as_str()])),
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part));
                        }
                    }
                }
            }
        }

        if output.is_empty() {
            output.push('/');
        } else if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn match_segments(pattern: &[Segment], request: &[Segment]) -> Option<Vec<(String, String)>> {
    let wildcard = pattern.last().and_then(|s| match s {
        Segment::Wildcard(name) => Some(name.as_str()),
        _ => None,
    });

    let fixed_len = if wildcard.is_some() {
        pattern.len() - 1
    } else {
        pattern.len()
    };
    if request.len() < fixed_len || (wildcard.is_none() && request.len() != pattern.len()) {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match (&pattern[i], &request[i]) {
            (Segment::Static(a), Segment::Static(b)) if a == b => {}
            (Segment::Param(name), Segment::Static(value)) => {
                params.push((name.clone(), value.clone()));
            }
            _ => return None,
        }
    }

    if let Some(name) = wildcard {
        let rest = request[fixed_len..]
            .iter()
            .map(|s| match s {
                Segment::Static(value) => value.as_str(),
                _ => "",
            })
            .collect::<Vec<_>>();
        params.push((name.to_owned(), rest.join("/")));
    }
    Some(params)
}

fn encode_component(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);

            {
                return Err(Error::Conflict { with: route.id });

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, trailing) = parse_path(head)?;

        let mut candidates = Vec::new();
        for route in &self.routes {
            if let Some(params) = match_route(route, &request_segments) {
                let slash_ok = self.policy != SlashPolicy::Strict || route.trailing == trailing;
                if slash_ok {
                    candidates.push((route, params));
                }
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = Vec::new();
            for (route, _) in &candidates {
                if !allowed.contains(&route.method) {
                    allowed.push(route.method.clone());
                }
                if route.method == "GET" && !allowed.iter().any(|m| m == "HEAD") {
                    allowed.push("HEAD".to_owned());
                }
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank
                .cmp(&a.rank)
                .then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible[0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != trailing
            && !route.segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };
        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|route| route.name == name).ok_or(Error::UnknownName)?;

        let mut supplied: Vec<(&str, &str)> = Vec::new();
        for &(key, value) in params {
            if supplied.iter().any(|(old, _)| *old == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_owned(),
                });
            }
            supplied.push((key, value));
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => {
                    if !capture_names.iter().any(|old: &String| old == name) {
                        capture_names.push(name.clone());
                    }
                }
                Segment::Static(_) => {}
            }
        }
        for key in &capture_names {
            if !supplied.iter().any(|(given, _)| *given == key) {
                return Err(Error::MissingParam { name: key.clone() });
            }
        }

        let mut extras: Vec<&str> = supplied
            .iter()
            .filter(|(key, _)| !capture_names.iter().any(|name| name == key))
            .map(|(key, _)| *key)
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let (name, wildcard) = match segment {
                Segment::Param(name) => (name, false),
                Segment::Wildcard(name) => (name, true),
                Segment::Static(_) => continue,
            };
            let value = supplied.iter().find(|(key, _)| *key == name).unwrap().1;
            if !valid_build_value(value, wildcard) {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if let Segment::Wildcard(name) = segment {
                let value = supplied.iter().find(|(key, _)| *key == name).unwrap().1;
                if value.is_empty() {
                    continue;
                }
                for component in value.split('/') {
                    if !first {
                        output.push('/');
                    }
                    encode_component(component, &mut output);
                    first = false;
                }
            } else {
                if !first {
                    output.push('/');
                }
                match segment {
                    Segment::Static(value) => encode_component(value, &mut output),
                    Segment::Param(name) => {
                        let value = supplied.iter().find(|(key, _)| *key == name).unwrap().1;
                        encode_component(value, &mut output);
                    }
                    Segment::Wildcard(_) => unreachable!(),
                }
                first = false;
            }
        }
        if route.trailing && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut end = bytes.len();
    let trailing = bytes[end - 1] == b'/';
    if trailing {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = Vec::new();
    let mut start = 1;
    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(Error::InvalidPattern { at: start });
        }
        let raw = &input[start..stop];
        let first = bytes[start];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if names.iter().any(|old: &String| old == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' && stop < end {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            if first == b':' {
                segments.push(Segment::Param(name.to_owned()));
            } else {
                segments.push(Segment::Wildcard(name.to_owned()));
            }
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }
        start = stop + 1;
    }
    Ok((segments, trailing))
}

fn parse_path(input: &str) -> Result<(Vec<String>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut end = bytes.len();
    let trailing = bytes[end - 1] == b'/';
    if trailing {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut start = 1;
    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(&input[start..stop], start, false)?);
        start = stop + 1;
    }
    Ok((segments, trailing))
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            let (Some(hi), Some(lo)) = (hex_upper(bytes[i + 1]), hex_upper(bytes[i + 2])) else {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            };
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            decoded.push(value);
            i += 3;
        } else if unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: offset + i }
            } else {
                Error::InvalidPath { at: offset + i }
            });
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    if a.len() != b.len() {
        return false;
    }
    a.iter().zip(b).all(|(left, right)| match (left, right) {
        (Segment::Static(x), Segment::Static(y)) => x == y,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;
    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if request.get(i)? != value {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                let value = request.get(i)?;
                params.push((name.clone(), value.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                let value = request[i..].join("/");
                params.push((name.clone(), value));
                i = request.len();
            }
        }
    }
    if i != request.len() {
        return None;
    }
    Some(params)
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
        return false;
    }
    if wildcard {
        if value.is_empty() {
            return true;
        }
        value
            .split('/')
            .all(|part| !part.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn encode_component(value: &str, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for b in value.bytes() {
        if unreserved(b) {
            output.push(b as char);
        } else {
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
}

fn decode_segment(raw: &str) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {

        if b == b'%' {

        }
    }

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut end = input.len();
    let trailing = input.as_bytes().last() == Some(&b'/');
    if trailing {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut seen = BTreeSet::new();
    let mut start = 1;
    let mut wildcard_seen = false;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|n| start + n)
            .unwrap_or(end);
        if next == start {
            return Err(invalid(start));
        }

        let raw = &input[start..next];
        let segment = if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            if wildcard_seen {
                return Err(invalid(start));
            }
            let is_wildcard = raw.starts_with('*');
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_string()) {
                return Err(invalid(start));
            }
            if is_wildcard {
                wildcard_seen = true;
                if next != end {
                    return Err(invalid(start));
                }
                Segment::Wildcard(name.to_string())
            } else {
                Segment::Param(name.to_string())
            }
        } else {
            if wildcard_seen {
                return Err(invalid(start));
            }
            let decoded = decode_segment(raw).map_err(|at| invalid(start + at))?;
            Segment::Static(decoded)
        };
        segments.push(segment);
        start = next + 1;
    }

    Ok((segments, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn rank_for(segments: &[Segment]) -> Vec<u8> {

            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),

        }

fn parse_request(input: &str) -> Result<(Vec<String>, bool), Error> {
    let cut = input
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(input.len());
    let head = &input[..cut];
    let (segments, trailing) = parse_path(head, false)?;
    let decoded = segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(value) => Ok(value),
            _ => unreachable!(),
        })
        .collect::<Result<Vec<_>, Error>>()?;
    Ok((decoded, trailing))
}

fn route_matches(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));

    if let Some(wi) = wildcard_index {
        if request.len() < wi {
            return None;
        }
        for i in 0..wi {
            match &route.segments[i] {
                Segment::Static(s) if s == &request[i] => {}
                Segment::Param(_) => {}
                _ => return None,
            }
        }
        let mut params = Vec::new();
        for i in 0..wi {
            if let Segment::Param(name) = &route.segments[i] {
                params.push((name.clone(), request[i].clone()));
            }
        }
        if let Segment::Wildcard(name) = &route.segments[wi] {
            params.push((name.clone(), request[wi..].join("/")));
        }
        Some(params)
    } else {
        if route.segments.len() != request.len() {
            return None;
        }
        let mut params = Vec::new();
        for (pattern, value) in route.segments.iter().zip(request) {
            match pattern {
                Segment::Static(s) if s == value => {}
                Segment::Param(name) => params.push((name.clone(), value.clone())),
                _ => return None,
            }
        }
        Some(params)
    }
}

            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let route = Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            rank: rank_for(&segments),
            segments,
            trailing,
        };
        for existing in &self.routes {
            if shape_conflicts(existing, &route, self.policy) {
                return Err(Error::Conflict { with: existing.id });
            }
        }
        self.names.insert(name.to_string(), id);
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (request, trailing) = parse_request(path)?;
        let path_candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| {
                if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                    return None;
                }
                route_matches(route, &request).map(|params| (route, params))
            })
            .collect();

        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && path_candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = path_candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &path_candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (route, params) = eligible[0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !request.is_empty()
            && route.trailing != trailing
        {
            let cut = path
                .char_indices()
                .find(|(_, c)| *c == '?' || *c == '#')
                .map(|(i, _)| i)
                .unwrap_or(path.len());
            let mut head = path[..cut].to_string();
            if route.trailing {
                head.push('/');
            } else if head.ends_with('/') {
                head.pop();
            }
            Some(head)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(&id) = self.names.get(name) else {
            return Err(Error::UnknownName);
        };
        let route = &self.routes[id];

        let mut supplied = BTreeMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let required_set: BTreeSet<&str> = required.iter().copied().collect();
        if let Some(extra) = supplied.keys().find(|key| !required_set.contains(**key)) {
            return Err(Error::ExtraParam {
                name: (*extra).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = supplied[name.as_str()];
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || (!value.is_empty()
                            && value.split('/').any(|part| part.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if !first {
                        output.push('/');
                    }
                    encode_component(value.as_bytes(), &mut output);
                }
                Segment::Param(name) => {
                    if !first {
                        output.push('/');
                    }
                    encode_component(supplied[name.as_str()].as_bytes(), &mut output);
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        if !first {
                            output.push('/');
                        }
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            encode_component(part.as_bytes(), &mut output);
                        }
                    }
                }
            }
            first = false;
        }

        if route.segments.is_empty() {
            return Ok("/".to_string());
        }
        if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn encode_component(bytes: &[u8], out: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in bytes {
        if unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
}

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())

fn valid_route_name(s: &str) -> bool {
    let mut bytes = s.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')

    let mut bytes = s.bytes();

        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static(raw: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            let fail = || {
                if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                }
            };
            if i + 2 >= bytes.len() {
                return Err(fail());
            }
            let hi = hex_upper(bytes[i + 1]).ok_or_else(fail)?;
            let lo = hex_upper(bytes[i + 2]).ok_or_else(fail)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(fail());
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: base + i }
            } else {
                Error::InvalidPath { at: base + i }
            });
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut start = 1usize;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(invalid(start));
        }
        let raw = &input[start..stop];

        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(invalid(start));
            }
            if wildcard && stop != end {
                return Err(invalid(start));
            }
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            segments.push(Segment::Static(decode_static(raw, start, pattern)?));
        }
        if stop == end {
            break;
        }
        start = stop + 1;
    }

    Ok((segments, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let mut rank = Vec::new();
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });

        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        let route = Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        };
        if let Some(existing) = self.routes.iter().find(|r| shape_conflicts(&route, r, self.policy)) {
            return Err(Error::Conflict { with: existing.id });
        }
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head, false)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            let mut captures = Vec::new();
            let mut matched = true;
            let mut i = 0usize;
            for segment in &route.segments {
                match segment {
                    Segment::Static(expected) => {
                        if request_segments.get(i).map(|s| match s {
                            Segment::Static(v) => v == expected,
                            _ => false,
                        }) != Some(true) {
                            matched = false;
                            break;
                        }
                        i += 1;
                    }
                    Segment::Param(name) => {
                        match request_segments.get(i) {
                            Some(Segment::Static(value)) => {
                                captures.push((name.clone(), value.clone()));
                                i += 1;
                            }
                            _ => {
                                matched = false;
                                break;
                            }
                        }
                    }
                    Segment::Wildcard(name) => {
                        let mut tail = Vec::new();
                        while i < request_segments.len() {
                            match &request_segments[i] {
                                Segment::Static(v) => tail.push(v.clone()),
                                _ => {
                                    matched = false;
                                    break;
                                }
                            }
                            i += 1;
                        }
                        if !matched {
                            break;
                        }
                        captures.push((name.clone(), tail.join("/")));
                    }
                }
            }
            if matched && i == request_segments.len() {
                candidates.push((route, captures));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" && candidates.iter().any(|(r, _)| r.method == "HEAD") {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };
        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(r, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_iter().collect() });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (winner, params) = eligible[0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && winner.trailing != request_trailing
        {
            if winner.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::<&str, &str>::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam { name: (*key).to_owned() });
            }
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(n) | Segment::Wildcard(n) => capture_names.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &capture_names {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam { name: (*key).to_owned() });
            }
        }
        let extra = supplied.keys().filter(|k| !capture_names.contains(k)).min();
        if let Some(key) = extra {
            return Err(Error::ExtraParam { name: (*key).to_owned() });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(n) => {
                    let value = supplied[n.as_str()];
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Wildcard(n) => {
                    let value = supplied[n.as_str()];
                    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || (!value.is_empty()
                            && value.split('/').any(|part| part.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            let mut pieces: Vec<String> = Vec::new();
            match segment {
                Segment::Static(value) => pieces.push(encode_component(value)),
                Segment::Param(name) => pieces.push(encode_component(supplied[name.as_str()])),
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        pieces.extend(value.split('/').map(encode_component));
                    }
                }
            }
            for piece in pieces {
                if !first {
                    output.push('/');
                }
                output.push_str(&piece);
                first = false;
            }
        }
        if route.trailing && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}
