use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
    names: BTreeMap<String, usize>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut end = input.len();
    let trailing = input.as_bytes().last() == Some(&b'/');
    if trailing {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut seen = BTreeSet::new();
    let mut start = 1;
    let mut wildcard_seen = false;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|n| start + n)
            .unwrap_or(end);
        if next == start {
            return Err(invalid(start));
        }

        let raw = &input[start..next];
        let segment = if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            if wildcard_seen {
                return Err(invalid(start));
            }
            let is_wildcard = raw.starts_with('*');
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_string()) {
                return Err(invalid(start));
            }
            if is_wildcard {
                wildcard_seen = true;
                if next != end {
                    return Err(invalid(start));
                }
                Segment::Wildcard(name.to_string())
            } else {
                Segment::Param(name.to_string())
            }
        } else {
            if wildcard_seen {
                return Err(invalid(start));
            }
            let decoded = decode_segment(raw).map_err(|at| invalid(start + at))?;
            Segment::Static(decoded)
        };
        segments.push(segment);
        start = next + 1;
    }

    Ok((segments, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn rank_for(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn parse_request(input: &str) -> Result<(Vec<String>, bool), Error> {
    let cut = input
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(input.len());
    let head = &input[..cut];
    let (segments, trailing) = parse_path(head, false)?;
    let decoded = segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(value) => Ok(value),
            _ => unreachable!(),
        })
        .collect::<Result<Vec<_>, Error>>()?;
    Ok((decoded, trailing))
}

fn route_matches(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));

    if let Some(wi) = wildcard_index {
        if request.len() < wi {
            return None;
        }
        for i in 0..wi {
            match &route.segments[i] {
                Segment::Static(s) if s == &request[i] => {}
                Segment::Param(_) => {}
                _ => return None,
            }
        }
        let mut params = Vec::new();
        for i in 0..wi {
            if let Segment::Param(name) = &route.segments[i] {
                params.push((name.clone(), request[i].clone()));
            }
        }
        if let Segment::Wildcard(name) = &route.segments[wi] {
            params.push((name.clone(), request[wi..].join("/")));
        }
        Some(params)
    } else {
        if route.segments.len() != request.len() {
            return None;
        }
        let mut params = Vec::new();
        for (pattern, value) in route.segments.iter().zip(request) {
            match pattern {
                Segment::Static(s) if s == value => {}
                Segment::Param(name) => params.push((name.clone(), value.clone())),
                _ => return None,
            }
        }
        Some(params)
    }
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
            names: BTreeMap::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        if self.names.contains_key(name) {
            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let route = Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            rank: rank_for(&segments),
            segments,
            trailing,
        };
        for existing in &self.routes {
            if shape_conflicts(existing, &route, self.policy) {
                return Err(Error::Conflict { with: existing.id });
            }
        }
        self.names.insert(name.to_string(), id);
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (request, trailing) = parse_request(path)?;
        let path_candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| {
                if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                    return None;
                }
                route_matches(route, &request).map(|params| (route, params))
            })
            .collect();

        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && path_candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = path_candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &path_candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (route, params) = eligible[0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !request.is_empty()
            && route.trailing != trailing
        {
            let cut = path
                .char_indices()
                .find(|(_, c)| *c == '?' || *c == '#')
                .map(|(i, _)| i)
                .unwrap_or(path.len());
            let mut head = path[..cut].to_string();
            if route.trailing {
                head.push('/');
            } else if head.ends_with('/') {
                head.pop();
            }
            Some(head)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(&id) = self.names.get(name) else {
            return Err(Error::UnknownName);
        };
        let route = &self.routes[id];

        let mut supplied = BTreeMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let required_set: BTreeSet<&str> = required.iter().copied().collect();
        if let Some(extra) = supplied.keys().find(|key| !required_set.contains(**key)) {
            return Err(Error::ExtraParam {
                name: (*extra).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = supplied[name.as_str()];
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || (!value.is_empty()
                            && value.split('/').any(|part| part.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if !first {
                        output.push('/');
                    }
                    encode_component(value.as_bytes(), &mut output);
                }
                Segment::Param(name) => {
                    if !first {
                        output.push('/');
                    }
                    encode_component(supplied[name.as_str()].as_bytes(), &mut output);
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        if !first {
                            output.push('/');
                        }
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            encode_component(part.as_bytes(), &mut output);
                        }
                    }
                }
            }
            first = false;
        }

        if route.segments.is_empty() {
            return Ok("/".to_string());
        }
        if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn encode_component(bytes: &[u8], out: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in bytes {
        if unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
}
