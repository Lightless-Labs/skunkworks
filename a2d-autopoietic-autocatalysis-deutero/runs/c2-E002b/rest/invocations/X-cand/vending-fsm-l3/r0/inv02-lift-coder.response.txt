```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended {
        id: String,
        price: u16,
        change: Vec<u16>,
    },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    indices: HashMap<String, usize>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut indices = HashMap::new();
        for (index, product) in products.iter().enumerate() {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if indices.contains_key(&product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
            indices.insert(product.id.clone(), index);
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            indices,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let denomination_index = DENOMINATIONS
            .iter()
            .position(|&denomination| denomination == coin)
            .ok_or(Error::BadCoin { coin })?;

        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        let _ = denomination_index;
        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.push_event(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.indices.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        self.push_event(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self.indices[&id];
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change = self.make_change(amount)?;
        let mut proposed_tubes = self.tubes;

        for (tube_index, denomination) in DENOMINATIONS.iter().enumerate() {
            let used = change
                .iter()
                .filter(|&&coin| coin == *denomination)
                .count() as u16;
            proposed_tubes[tube_index] -= used;
        }

        for &coin in &self.escrow {
            let tube_index = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            proposed_tubes[tube_index] += 1;
        }

        for (tube_index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[tube_index],
                });
            }
        }

        self.tubes = proposed_tubes;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.push_event(EventKind::Vended { id, price, change });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.push_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = *self
            .indices
            .get(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.push_event(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = *self
            .indices
            .get(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = self.products[index].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = new_stock as u16;
        self.push_event(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn push_event(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }

    fn make_change(&self, amount: u16) -> Result<Vec<u16>, Error> {
        if amount == 0 {
            return Ok(Vec::new());
        }

        let mut remaining = amount;
        let mut greedy = Vec::new();
        for index in (0..DENOMINATIONS.len()).rev() {
            let denomination = DENOMINATIONS[index];
            let count = self.tubes[index].min(remaining / denomination);
            greedy.extend(std::iter::repeat_n(denomination, count as usize));
            remaining -= count * denomination;
        }
        if remaining == 0 {
            return Ok(greedy);
        }

        let mut best: Option<Vec<u16>> = None;
        let mut counts = [0u16; 4];
        Self::search_change(
            3,
            amount as i32,
            &self.tubes,
            &mut counts,
            &mut best,
        );

        best.map(|counts| {
            let mut coins = Vec::new();
            for index in (0..DENOMINATIONS.len()).rev() {
                coins.extend(std::iter::repeat_n(
                    DENOMINATIONS[index],
                    counts[index] as usize,
                ));
            }
            coins
        })
        .ok_or(Error::NoChange { amount })
    }

    fn search_change(
        index: isize,
        remaining: i32,
        tubes: &[u16; 4],
        counts: &mut [u16; 4],
        best: &mut Option<Vec<u16>>,
    ) {
        if remaining == 0 {
            if best
                .as_ref()
                .is_none_or(|current| Self::better_counts(counts, current))
            {
                *best = Some(*counts);
            }
            return;
        }
        if index < 0 {
            return;
        }

        let i = index as usize;
        let denomination = DENOMINATIONS[i] as i32;
        let max_count = tubes[i].min((remaining / denomination) as u16);
        for count in 0..=max_count {
            counts[i] = count;
            Self::search_change(
                index - 1,
                remaining - count as i32 * denomination,
                tubes,
                counts,
                best,
            );
        }
        counts[i] = 0;
    }

    fn better_counts(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
        let candidate_total: u32 = candidate.iter().map(|&count| count as u32).sum();
        let current_total: u32 = current.iter().map(|&count| count as u32).sum();
        if candidate_total != current_total {
            return candidate_total < current_total;
        }

        for index in (0..4).rev() {
            if candidate[index] != current[index] {
                return candidate[index] > current[index];
            }
        }
        false
    }
}
```
