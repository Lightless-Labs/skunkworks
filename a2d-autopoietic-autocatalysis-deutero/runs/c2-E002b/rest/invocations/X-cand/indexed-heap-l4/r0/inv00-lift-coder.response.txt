```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    capacity: Option<usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::with_capacity_inner(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_capacity_inner(Some(capacity))
    }

    fn with_capacity_inner(capacity: Option<usize>) -> Self {
        Self {
            entries: HashMap::new(),
            capacity,
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn canonical_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }
        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }

        Ok(input[start..end].to_ascii_lowercase())
    }

    fn compare_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn best_to_worst(&self) -> Vec<Entry> {
        let mut items: Vec<Entry> = self.entries.values().cloned().collect();
        items.sort_by(Self::compare_entries);
        items
    }

    fn worst_to_best(&self) -> Vec<Entry> {
        let mut items = self.best_to_worst();
        items.reverse();
        items
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut ordered = self.best_to_worst();
            ordered.sort_by_key(|entry| entry.age);
            for (age, entry) in ordered.into_iter().enumerate() {
                if let Some(current) = self.entries.get_mut(&entry.item.key) {
                    current.age = age as u128;
                }
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let canonical = Self::canonical_key(key)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        let item = Item {
            key: canonical.clone(),
            priority,
            value: value.to_owned(),
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(item));
        }

        if let Some(capacity) = self.capacity {
            if self.entries.len() >= capacity {
                let worst = self.worst_to_best().into_iter().next().unwrap();
                let proposal = Entry {
                    item: item.clone(),
                    age: self.next_age,
                };
                if Self::compare_entries(&proposal, &worst) != std::cmp::Ordering::Less {
                    return Ok(Admission::Discarded(item));
                }
                self.entries.remove(&worst.item.key);
                let age = self.fresh_age();
                self.entries.insert(canonical, Entry { item, age });
                return Ok(Admission::Evicted(worst.item));
            }
        }

        let age = self.fresh_age();
        self.entries.insert(canonical, Entry { item, age });
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_to_worst().into_iter().next().map(|entry| entry.item)
    }

    pub fn pop(&mut self) -> Option<Item> {
        let best = self.best_to_worst().into_iter().next()?;
        self.entries.remove(&best.item.key);
        Some(best.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = Self::canonical_key(key)?;
        Ok(self.entries.get(&canonical).map(|entry| entry.item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let canonical = Self::canonical_key(key)?;
        let current = self.entries.get(&canonical)
            .ok_or_else(|| Error::Missing { key: canonical.clone() })?
            .item.priority;
        if priority >= current {
            return Err(Error::WrongDirection { current, requested: priority });
        }
        self.entries.get_mut(&canonical).unwrap().item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let canonical = Self::canonical_key(key)?;
        let current = self.entries.get(&canonical)
            .ok_or_else(|| Error::Missing { key: canonical.clone() })?
            .item.priority;
        if priority <= current {
            return Err(Error::WrongDirection { current, requested: priority });
        }
        self.entries.get_mut(&canonical).unwrap().item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let canonical = Self::canonical_key(key)?;
        let entry = self.entries.get_mut(&canonical)
            .ok_or_else(|| Error::Missing { key: canonical.clone() })?;
        Ok(std::mem::replace(&mut entry.item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let canonical = Self::canonical_key(key)?;
        self.entries.remove(&canonical)
            .map(|entry| entry.item)
            .ok_or(Error::Missing { key: canonical })
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = Vec::new();
        if let Some(limit) = capacity {
            while self.entries.len() > limit {
                let worst = self.worst_to_best().into_iter().next().unwrap();
                self.entries.remove(&worst.item.key);
                evicted.push(worst.item);
            }
        }
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.best_to_worst()
            .into_iter()
            .map(|entry| entry.item)
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source = other.best_to_worst();
        if policy == CollisionPolicy::Reject {
            if let Some(entry) = source.iter().find(|entry| self.entries.contains_key(&entry.item.key)) {
                return Err(Error::Collision { key: entry.item.key.clone() });
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for entry in source {
            let key = entry.item.key.clone();
            if let Some(receiver) = self.entries.get_mut(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        receiver.item.priority = entry.item.priority;
                        receiver.item.value = entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                self.entries.insert(key, Entry { item: entry.item, age });
                report.moved += 1;
            }
        }

        other.entries.clear();

        if let Some(limit) = self.capacity {
            while self.entries.len() > limit {
                let worst = self.worst_to_best().into_iter().next().unwrap();
                self.entries.remove(&worst.item.key);
                report.evicted.push(worst.item);
            }
        }

        Ok(report)
    }
}
```
