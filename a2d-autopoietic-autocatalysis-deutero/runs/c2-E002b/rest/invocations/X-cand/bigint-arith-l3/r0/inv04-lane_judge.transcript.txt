pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_offsets = Vec::new();
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            if !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
            digit_offsets.push(index);
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        if digit_offsets.len() > 1 && bytes[digit_offsets[0]] == b'0' {
            return Err(Error::LeadingZero {
                index: digit_offsets[0],
            });
        }

        if digit_offsets.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_offsets[max_digits],
            });
        }

        let mut digits = Vec::with_capacity(digit_offsets.len());
        for &index in digit_offsets.iter().rev() {
            digits.push(bytes[index] - b'0');
        }
        let mut value = Self { negative, digits };
        value.normalize();
        Ok(value)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let actual = self.digits.len().max(1);
        let width = actual.max(min_digits);
        let padding = width - actual;
        let mut out = String::with_capacity(width + width / 2 + usize::from(self.negative));
        if self.negative {
            out.push('-');

                out.push('_');
            }
            let digit = if pos < padding {
                0
            } else {
                self.digits[pos - padding]
            };
            out.push(char::from(b'0' + digit));
        }
        out
    }

            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),

        } else if self.negative {
            -1
        } else {

        }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            for (j, &b) in rhs.digits.iter().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }

        let mut digits = Vec::with_capacity(accum.len() + 1);
        let mut carry = 0u32;
        for value in accum {
            let total = value + carry;
            digits.push((total % 10) as u8);
            carry = total / 10;
        }
        while carry != 0 {
            digits.push((carry % 10) as u8);
            carry /= 10;
        }

        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            digits,
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        match cmp_magnitude(&self.digits, &rhs.digits) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                let quotient = Self {
                    negative: self.negative ^ rhs.negative,
                    digits: vec![1],
                };
                return Ok((quotient, Self::zero()));
            }
            Ordering::Greater => {}
        }

        let mut quotient_digits = vec![0u8; self.digits.len()];
        let mut remainder = vec![0u8];

        for i in (0..self.digits.len()).rev() {
            remainder.insert(0, self.digits[i]);
            trim_magnitude(&mut remainder);

            let mut qdigit = 0u8;
            while cmp_magnitude(&remainder, &rhs.digits) != Ordering::Less {
                remainder = sub_magnitude(&remainder, &rhs.digits);
                qdigit += 1;
            }
            quotient_digits[i] = qdigit;
        }

        let mut quotient = Self {
            negative: self.negative ^ rhs.negative,
            digits: quotient_digits,
        };
        quotient.normalize();

        let mut remainder_value = Self {
            negative: self.negative,
            digits: remainder,
        };
        remainder_value.normalize();

        Ok((quotient, remainder_value))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        trim_magnitude(&mut self.digits);
        if self.is_zero() {
            self.negative = false;
        }
    }
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}

fn cmp_magnitude(lhs: &[u8], rhs: &[u8]) -> Ordering {
    match lhs.len().cmp(&rhs.len()) {
        Ordering::Equal => {
            for (&a, &b) in lhs.iter().rev().zip(rhs.iter().rev()) {
                match a.cmp(&b) {
                    Ordering::Equal => {}
                    ordering => return ordering,
                }
            }
            Ordering::Equal
        }
        ordering => ordering,
    }
}

fn add_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(lhs.len().max(rhs.len()) + 1);
    let mut carry = 0u8;
    for i in 0..lhs.len().max(rhs.len()) {
        let sum = lhs.get(i).copied().unwrap_or(0) + rhs.get(i).copied().unwrap_or(0) + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        result.push(carry);
    }
    result
}

fn sub_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_magnitude(lhs, rhs) != Ordering::Less);
    let mut result = Vec::with_capacity(lhs.len());
    let mut borrow = 0i8;

    for i in 0..lhs.len() {
        let mut digit = i16::from(lhs[i]) - i16::from(borrow as u8);
        let subtrahend = i16::from(rhs.get(i).copied().unwrap_or(0));
        if digit < subtrahend {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push((digit - subtrahend) as u8);
    }

    trim_magnitude(&mut result);
    result
}

#[derive(Clone, Debug, PartialEq, Eq)]

    negative: bool,
    // Little-endian decimal digits; zero is represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_count = 0usize;
        let mut first_digit_index = None;

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            if !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
            digit_count += 1;
            if first_digit_index.is_none() {
                first_digit_index = Some(index);
            }
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let first_index = first_digit_index.expect("nonempty magnitude has a digit");
        if digit_count > 1 && bytes[first_index] == b'0' {
            return Err(Error::LeadingZero { index: first_index });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte.is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }

        let mut result = Self { negative, digits };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude = self.digits.len().max(min_digits).max(1);
        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }

        let padding = magnitude - self.digits.len();
        for position in (0..magnitude).rev() {
            if group > 0 && position + 1 < magnitude && (position + 1) % group == 0 {
                out.push('_');
            }
            if position >= self.digits.len() {
                out.push('0');
            } else {
                out.push((b'0' + self.digits[position]) as char);
            }
        }

        // The loop above groups according to position from the left only when
        // the width is divisible by group. Rebuild grouping from the right.
        if group > 0 {
            let mut plain = String::with_capacity(magnitude);
            for position in (0..magnitude).rev() {
                if position >= self.digits.len() {
                    plain.push('0');
                } else {
                    plain.push((b'0' + self.digits[position]) as char);
                }
            }
            let mut grouped = String::with_capacity(magnitude + magnitude / group);
            for (i, ch) in plain.chars().enumerate() {
                if i > 0 && (magnitude - i) % group == 0 {
                    grouped.push('_');
                }
                grouped.push(ch);
            }
            if self.negative && !self.is_zero() {
                out.clear();
                out.push('-');
            } else {
                out.clear();
            }
            out.push_str(&grouped);
        }

        let _ = padding;
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (-1, 0 | 1) => Ordering::Less,
            (0 | 1, -1) => Ordering::Greater,
            (0, 0) => Ordering::Equal,
            (1, 1) => cmp_magnitude(&self.digits, &rhs.digits),
            (-1, -1) => cmp_magnitude(&rhs.digits, &self.digits),
            _ => unreachable!(),
        }
    }

                negative: self.negative,
                digits: add_magnitude(&self.digits, &rhs.digits),

                        negative: rhs.negative,
                        digits: sub_magnitude(&rhs.digits, &self.digits),
                    };
                    result.normalize();
                    result
                }
            }
        }
    }

        result

        result.normalize();
        result
    }

        }

        self.digits.len() == 1 && self.digits[0] == 0

            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

        Ordering::Equal => {
            for (&x, &y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(&y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = carry
            + a.get(i).copied().unwrap_or(0)
            + b.get(i).copied().unwrap_or(0);
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        result.push(carry);
    }
    result
}

fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i16;
    for i in 0..a.len() {
        let mut digit = a[i] as i16 - borrow - b.get(i).copied().unwrap_or(0) as i16;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(digit as u8);
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn div_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    let mut remainder = vec![0u8];
    let mut quotient_be = Vec::with_capacity(dividend.len());

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        trim_magnitude(&mut remainder);

        let mut qdigit = 0u8;
        while cmp_magnitude(&remainder, divisor) != Ordering::Less {
            remainder = sub_magnitude(&remainder, divisor);
            qdigit += 1;
        }
        quotient_be.push(qdigit);
    }

    while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
        quotient_be.remove(0);
    }
    let quotient = quotient_be.into_iter().rev().collect();
    (quotient, remainder)
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian decimal digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' || bytes[0] == b'-' {
            1
        } else {
            0
        };
        if start == bytes.len() && start == 1 {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        // Find the leftmost underscore not surrounded by digits.
        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let index = bytes[start..]
                .iter()
                .enumerate()
                .filter(|(_, b)| **b != b'_')
                .nth(excess_number - 1)
                .map(|(offset, _)| start + offset)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in bytes[start..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        let negative = start == 1 && bytes[0] == b'-' && !is_zero_digits(&digits);
        Ok(Self { negative, digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let padding = width - self.digits.len();
        let mut out = String::with_capacity(
            width + if group > 0 { (width - 1) / group } else { 0 } + usize::from(self.negative),
        );

        if self.negative {
            out.push('-');
        }
        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }
            let digit = if position < padding {
                0
            } else {
                self.digits[width - 1 - position]
            };
            out.push((b'0' + digit) as char);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_magnitude(
                self.negative,
                add_magnitude(&self.digits, &rhs.digits),
            )
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => Self::from_magnitude(
                    self.negative,
                    sub_magnitude(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_magnitude(
                    rhs.negative,
                    sub_magnitude(&rhs.digits, &self.digits),
                ),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::from_magnitude(
            self.negative != rhs.negative,
            mul_magnitude(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient, remainder) = div_rem_magnitude(&self.digits, &rhs.digits);
        Ok((
            Self::from_magnitude(self.negative != rhs.negative, quotient),
            Self::from_magnitude(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_magnitude(negative: bool, digits: Vec<u8>) -> Self {
        let digits = normalize(digits);

    }

        if digit < 0 {

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = vec![0u8; a.len() + b.len()];
    for (i, &da) in a.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &db) in b.iter().enumerate() {
            let index = i + j;
            let value = result[index] as u16 + da as u16 * db as u16 + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
        }
        let mut index = i + b.len();
        while carry > 0 {
            if index == result.len() {
                result.push(0);
            }
            let value = result[index] as u16 + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
            index += 1;
        }
    }
    normalize(result)
}

fn mul_digit(a: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }
    let mut result = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u16;
    for &d in a {
        let value = d as u16 * digit as u16 + carry;
        result.push((value % 10) as u8);
        carry = value / 10;
    }
    while carry > 0 {
        result.push((carry % 10) as u8);
        carry /= 10;
    }
    normalize(result)
}

// Long division of nonnegative magnitudes.
fn div_rem_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_magnitude(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0u8];

    for &digit in dividend.iter().rev() {
        if is_zero_digits(&remainder) {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        remainder = normalize(remainder);

        let mut qdigit = 0u8;
        for candidate in (1..=9).rev() {
            let product = mul_digit(divisor, candidate);
            if cmp_magnitude(&product, &remainder) != Ordering::Greater {
                qdigit = candidate;
                remainder = sub_magnitude(&remainder, &product);
                break;
            }
        }
        quotient_be.push(qdigit);
    }

    quotient_be.reverse();
    (normalize(quotient_be), normalize(remainder))
}

                break;
            }

            return Err(Error::InvalidDigit { index });

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())

                separator_error = Some(index);
                break;
            }

            return Err(Error::Separator { index });

                }

                digits.push(byte - b'0');
            }

        quotient_be.push(quotient_digit);
    }

    let mut quotient: Vec<u8> = quotient_be.into_iter().rev().collect();
    trim_magnitude(&mut quotient);
    trim_magnitude(&mut remainder);
    (quotient, remainder)
}

fn trim_magnitude(value: &mut Vec<u8>) {
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
}
