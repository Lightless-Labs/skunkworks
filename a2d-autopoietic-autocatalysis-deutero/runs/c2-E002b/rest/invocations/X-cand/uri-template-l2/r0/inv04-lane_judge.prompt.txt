    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let operator = match first {
                    Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
                        let op = bytes[content_start] as char;
                        if op != '+' && op != '#' {
                            return Err(Error::UnsupportedOperator { pos: content_start });
                        }
                        content_start += 1;
                        Some(op)
                    }
                    _ => None,
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut seen = Vec::<String>::new();
                let mut spec_start = content_start;

                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                        return Err(Error::EmptyVariable { pos: spec_start });

                        return Err(Error::InvalidName {

                        });

                        return Err(Error::DuplicateVariable { pos: spec_start });

                    seen.push(name.clone());
                    variables.push((name, spec_start));

                        break;

                    spec_start = spec_end + 1;
                }

                    start,
                    end: close + 1,
                    operator,
                    variables,
                });
                i = close + 1;
            }
            _ => i += 1,
        }

        output.push_str(&template[literal_start..expression.start]);

            }
            output.push_str(&fields.join(","));
        }

        literal_start = expression.end;
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression {
    operator: Option<char>,
    names: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        let operator = expression.operator;
        let mut fields = Vec::new();

        for name in &expression.names {
            if let Some(value) = lookup(vars, name) {
                fields.push(encode(value, matches!(operator, Some('+') | Some('#'))));
            }
        }

        if !fields.is_empty() {
            if operator == Some('#') {
                output.push('#');
            }
            let separator = if operator == Some(',') { "," } else { "," };
            output.push_str(&fields.join(separator));
        }

        cursor = expression.end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

struct ParsedExpression {
    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<String>,
}

fn parse(template: &str) -> Result<Vec<ParsedExpression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let close = i + 1 + relative_close;

                if let Some(relative_open) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_open,
                    });
                }

                let mut content_start = start + 1;
                let mut operator = None;
                if content_start < close && is_operator(bytes[content_start]) {
                    let op = bytes[content_start] as char;
                    if op != '+' && op != '#' {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    operator = Some(op);
                    content_start += 1;
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|relative| spec_start + relative)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &template[spec_start..spec_end];
                    validate_name(spec, spec_start)?;
                    if seen.iter().any(|name| name == spec) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    let owned = spec.to_owned();
                    seen.push(owned.clone());
                    names.push(owned);

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(ParsedExpression {
                    start,
                    end: close + 1,
                    operator,
                    names,
                });
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn validate_name(name: &str, start: usize) -> Result<(), Error> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Err(Error::EmptyVariable { pos: start });
    }

    if !is_name_start(bytes[0]) {
        return Err(Error::InvalidName { pos: start });
    }

    for (offset, &byte) in bytes.iter().enumerate().skip(1) {
        if !is_name_continue(byte) {
            return Err(Error::InvalidName { pos: start + offset });
        }
    }
    Ok(())
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' {
            if i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit() {
                output.push('%');
                output.push((bytes[i + 1] as char).to_ascii_uppercase());
                output.push((bytes[i + 2] as char).to_ascii_uppercase());
                i += 3;
                continue;
            }
            output.push_str("%25");
            i += 1;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        i += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {

}

        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, String)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        let mut fields = Vec::new();

        for (_, name) in &expression.names {
            if let Some(value) = lookup(vars, name) {
                fields.push(encode(value, expression.operator != Operator::Simple));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Operator::Fragment {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let start = cursor;
                let close = match bytes[start + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => start + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[start + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + offset,
                    });
                }

                let mut content_start = start + 1;
                let mut operator = Operator::Simple;
                if content_start < close {
                    operator = match bytes[content_start] {
                        b'+' => {
                            content_start += 1;
                            Operator::Reserved
                        }
                        b'#' => {
                            content_start += 1;
                            Operator::Fragment
                        }
                        b'.' | b'/' | b';' | b'?' | b'&' => {
                            return Err(Error::UnsupportedOperator { pos: content_start });
                        }
                        _ => Operator::Simple,
                    };
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let name_bytes = &bytes[spec_start..spec_end];
                    for (index, &byte) in name_bytes.iter().enumerate() {
                        let valid = if index == 0 {
                            byte.is_ascii_alphabetic() || byte == b'_'
                        } else {
                            byte.is_ascii_alphanumeric() || byte == b'_'
                        };
                        if !valid {
                            return Err(Error::InvalidName {
                                pos: spec_start + index,
                            });
                        }
                    }

                    let name = template[spec_start..spec_end].to_owned();
                    if seen.iter().any(|existing| existing == &name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name.clone());
                    names.push((spec_start, name));

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end: close,
                    operator,
                    names,
                });
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            let high = bytes[index + 1];
            let low = bytes[index + 2];
            if high.is_ascii_hexdigit() && low.is_ascii_hexdigit() {
                output.push('%');
                output.push((high as char).to_ascii_uppercase());
                output.push((low as char).to_ascii_uppercase());
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
