use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_sep_missing(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut nums = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if pos == start {
            return Err(if is_sep_missing(bytes.get(pos).copied()) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        let field = &bytes[start..pos];
        if field.len() > 1 && field[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut value = 0u64;
        for &d in field {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((d - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        nums[i] = value;

        if i < 2 {
            match bytes.get(pos).copied() {
                Some(b'.') => pos += 1,
                b if is_sep_missing(b) => return Err(VersionError::MissingCore { at: pos }),
                _ => return Err(VersionError::InvalidChar { at: pos }),
            }
        }
    }

    if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if bytes.get(pos) == Some(&b'-') {
        pos += 1;
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());
        parse_identifiers(s, pos, end, true, &mut pre)?;
        pos = end;
    }

    if bytes.get(pos) == Some(&b'+') {
        pos += 1;
        parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if is_pre
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(())
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_ids(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric(x);
                    let yn = is_numeric(y);
                    let ord = match (xn, yn) {
                        (true, true) => cmp_numeric_ids(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    comparisons: Vec<Comparison>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Comparison {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < b.len() {
        while p < b.len() && ws(b[p]) {
            p += 1;
        }
        if p == b.len() {
            break;
        }
        let start = p;
        while p < b.len() && !ws(b[p]) {
            p += 1;
        }
        out.push(Token {
            text: s[start..p].to_string(),
            start,
        });
    }
    out
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
    k: usize,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_sign = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    if has_sign || (fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f))) {
        let v = parse_full_operand(s, at)?;
        return Ok(Partial {
            nums: vec![v.major, v.minor, v.patch],
            k: 3,
            full: Some(v),
        });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if is_wildcard(f) {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen
            || !f.bytes().all(|b| b.is_ascii_digit())
            || (f.len() > 1 && f.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }
        let mut n = 0u64;
        for b in f.bytes() {
            n = n
                .checked_mul(10)
                .and_then(|x| x.checked_add((b - b'0') as u64))
                .ok_or(RangeError::Syntax { at })?;
        }
        nums.push(n);
    }
    let k = nums.len();
    Ok(Partial {
        nums,
        full: None,
        k,
    })
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build: Vec::new(),
    }
}

fn padded(p: &Partial) -> Version {
    endpoint(
        *p.nums.first().unwrap_or(&0),
        *p.nums.get(1).unwrap_or(&0),
        *p.nums.get(2).unwrap_or(&0),
        p.full.as_ref().map(|v| v.pre.clone()).unwrap_or_default(),
    )
}

fn increment_bound(
    major: u64,
    minor: u64,
    patch: u64,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut m = major;
    let mut n = minor;
    let mut p = patch;
    match field {
        0 => m = m.checked_add(1).ok_or(RangeError::Syntax { at })?,
        1 => n = n.checked_add(1).ok_or(RangeError::Syntax { at })?,
        _ => p = p.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    if field == 0 {
        n = 0;
        p = 0;
    } else if field == 1 {
        p = 0;
    }
    Ok(endpoint(m, n, p, vec!["0".to_string()]))
}

fn push_cmp(arm: &mut Arm, op: Op, version: Version, explicit: bool) {
    if explicit && !version.pre.is_empty() {
        arm.anchors.push(version_tuple(&version));
    }
    arm.comparisons.push(Comparison { op, version });
}

fn expand_partial(
    arm: &mut Arm,
    p: Partial,
    at: usize,
    kind: char,
) -> Result<(), RangeError> {
    let k = p.k;
    if kind == '^' || kind == '~' {
        if k == 0 {
            return Err(RangeError::Syntax { at });
        }
        let lower = padded(&p);
        push_cmp(arm, Op::Ge, lower.clone(), p.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
        let (m, n, patch) = version_tuple(&lower);
        let field = if kind == '^' {
            if m > 0 || k == 1 {
                0
            } else if n > 0 || k == 2 {
                1
            } else {
                2
            }
        } else if k == 1 {
            0
        } else {
            1
        };
        let upper = increment_bound(m, n, patch, field, at)?;
        push_cmp(arm, Op::Lt, upper, false);
    } else if p.full.is_some() {
        push_cmp(arm, Op::Eq, p.full.unwrap(), true);
    } else if k == 0 {
        // Wildcard-only bare operands match all versions.
    } else if k == 3 {
        let v = padded(&p);
        push_cmp(arm, Op::Eq, v, false);
    } else {
        let lower = padded(&p);
        push_cmp(arm, Op::Ge, lower, false);
        let field = if k == 1 { 0 } else { 1 };
        let (m, n, patch) = version_tuple(&padded(&p));
        let upper = increment_bound(m, n, patch, field, at)?;
        push_cmp(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_comparator_token(arm: &mut Arm, t: &Token) -> Result<(), RangeError> {
    let (op, op_len) = if t.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if t.text.starts_with("<=") {
        (Op::Le, 2)
    } else if t.text.starts_with('>') {
        (Op::Gt, 1)
    } else if t.text.starts_with('<') {
        (Op::Lt, 1)
    } else if t.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        return Err(RangeError::Syntax { at: t.start });
    };
    let operand = &t.text[op_len..];
    let at = t.start + op_len;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: t.start });
    }
    let v = parse_full_operand(operand, at)?;
    push_cmp(arm, op, v, true);
    Ok(())
}

fn parse_arm(tokens: &[Token]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };
    if tokens.is_empty() {
        return Ok(arm);
    }

    if let Some(minus) = tokens.iter().position(|t| t.text == "-") {
        if tokens.len() != 3 || minus != 1 {
            return Err(RangeError::Syntax {
                at: tokens[minus].start,
            });
        }
        let lower = parse_partial(&tokens[0].text, tokens[0].start)?;
        let upper = parse_partial(&tokens[2].text, tokens[2].start)?;
        if lower.k == 0 {
            return Err(RangeError::Syntax {
                at: tokens[0].start,
            });
        }
        if upper.k == 0 {
            return Err(RangeError::Syntax {
                at: tokens[2].start,
            });
        }

        let lo = padded(&lower);
        push_cmp(
            &mut arm,
            Op::Ge,
            lo,
            lower.full.as_ref().is_some_and(|v| !v.pre.is_empty()),
        );
        if let Some(v) = upper.full {
            push_cmp(&mut arm, Op::Le, v, true);
        } else if upper.k == 3 {
            push_cmp(&mut arm, Op::Le, padded(&upper), false);
        } else {
            let base = padded(&upper);
            push_cmp(&mut arm, Op::Lt, increment_bound(
                base.major,
                base.minor,
                base.patch,
                if upper.k == 1 { 0 } else { 1 },
                tokens[2].start,
            )?, false);
        }
        return Ok(arm);
    }

    for t in tokens {
        let text = t.text.as_str();
        if text.starts_with('^') || text.starts_with('~') {
            let kind = text.as_bytes()[0] as char;
            let operand = &text[1..];
            let at = t.start + 1;
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: t.start });
            }
            let p = parse_partial(operand, at)?;
            expand_partial(&mut arm, p, at, kind)?;
        } else if text.starts_with('<')
            || text.starts_with('>')
            || text.starts_with('=')
        {
            parse_comparator_token(&mut arm, t)?;
        } else if text.starts_with(',')
            || text.starts_with('(')
            || text.starts_with(')')
            || text == "&"
        {
            return Err(RangeError::Syntax { at: t.start });
        } else if text.starts_with('*') || text.starts_with('x') || text.starts_with('X') {
            let p = parse_partial(text, t.start)?;
            expand_partial(&mut arm, p, t.start, ' ');
        } else if text.as_bytes().first().is_some_and(|b| b.is_ascii_digit()) {
            let p = parse_partial(text, t.start)?;
            expand_partial(&mut arm, p, t.start, ' ');
        } else {
            return Err(RangeError::Syntax { at: t.start });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut groups: Vec<Vec<Token>> = vec![Vec::new()];
    let mut separators = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            separators.push(tokens[i].start);
            groups.push(Vec::new());
        } else {
            groups.last_mut().unwrap().push(tokens[i].clone());
        }
        i += 1;
    }

    let mut arms = Vec::new();
    for (idx, group) in groups.iter().enumerate() {
        if group.is_empty() {
            let at = if idx < separators.len() {
                separators[idx]
            } else {
                s.len()
            };
            return Err(RangeError::Empty { at });
        }
        arms.push(parse_arm(group)?);
    }
    Ok(Range { arms })
}

fn comparison_matches(v: &Version, c: &Comparison) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.contains(&version_tuple(v))
        {
            return false;
        }
        arm.comparisons.iter().all(|c| comparison_matches(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (want_max && ord == Ordering::Greater)
                    || (!want_max && ord == Ordering::Less)
                {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
