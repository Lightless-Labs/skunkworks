use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(b: u8) -> bool {
    b == b'.' || b == b'-' || b == b'+'
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let value = s[start..pos]
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;
        core[field] = value;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let start = pos;
                while pos < bytes.len() && bytes[pos] != b'+' {
                    pos += 1;
                }
                pre = parse_identifiers(s, start, pos, true)?;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, pos, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut item_start = start;

    loop {
        let mut i = item_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == item_start {
            return Err(VersionError::EmptyIdentifier { at: item_start });
        }

        let item = &s[item_start..i];
        if prerelease
            && item.bytes().all(|b| b.is_ascii_digit())
            && item.len() > 1
            && item.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: item_start });
        }

        out.push(item.to_string());
        if i == end {
            break;
        }
        item_start = i + 1;
        if item_start == end {
            return Err(VersionError::EmptyIdentifier { at: item_start });
        }
    }

    Ok(out)
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_strings(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric_identifier(x);
                    let yn = is_numeric_identifier(y);
                    let ord = match (xn, yn) {
                        (true, true) => compare_numeric_strings(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    admission_anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
enum Check {
    Compare { op: Op, version: Version },
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

impl Op {
    fn test(self, ordering: Ordering) -> bool {
        match self {
            Op::Eq => ordering == Ordering::Equal,
            Op::Lt => ordering == Ordering::Less,
            Op::Le => ordering != Ordering::Greater,
            Op::Gt => ordering == Ordering::Greater,
            Op::Ge => ordering != Ordering::Less,
        }
    }
}

#[derive(Clone, Copy, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens<'a>(s: &'a str, base: usize) -> Vec<Token<'a>> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut pos = 0;
    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }
        let start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        out.push(Token {
            text: &s[start..pos],
            start: base + start,
        });
    }
    out
}

fn version_core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint_version(major: u64, minor: u64, patch: u64, pre: Vec<String>, build: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build,
    }
}

fn upper_bound(
    major: u64,
    minor: u64,
    patch: u64,
    at: usize,
    bump: usize,
) -> Result<Version, RangeError> {
    let (m, n, p) = match bump {
        0 => (major.checked_add(1), Some(0), Some(0)),
        1 => (Some(major), minor.checked_add(1), Some(0)),
        _ => (Some(major), Some(minor), patch.checked_add(1)),
    };
    match (m, n, p) {
        (Some(m), Some(n), Some(p)) => Ok(endpoint_version(m, n, p, vec!["0".into()], vec![])),
        _ => Err(RangeError::Syntax { at }),
    }
}

fn full_version_operand(token: Token<'_>, operand: &str, operand_start: usize) -> Result<Version, RangeError> {
    parse_version(operand).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })
}

fn add_check(arm: &mut Arm, op: Op, version: Version) {
    if !version.pre.is_empty() {
        arm.admission_anchors.push(version_core(&version));
    }
    arm.checks.push(Check::Compare { op, version });
}

fn parse_partial(
    token: Token<'_>,
    operand: &str,
    operand_start: usize,
) -> Result<(Vec<Option<u64>>, usize, bool, Option<Version>), RangeError> {
    let bytes = operand.as_bytes();
    let fullish = bytes.contains(&b'-')
        || bytes.contains(&b'+')
        || (operand.split('.').count() >= 3
            && !operand
                .split('.')
                .any(|f| matches!(f, "*" | "x" | "X")));

    if fullish {
        let v = full_version_operand(token, operand, operand_start)?;
        return Ok((
            vec![Some(v.major), Some(v.minor), Some(v.patch)],
            3,
            true,
            Some(v),
        ));
    }

    let fields: Vec<&str> = operand.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: token.start });
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for field in &fields {
        if matches!(*field, "*" | "x" | "X") {
            wildcard_seen = true;
            nums.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: token.start });
            }
            match field.parse::<u64>() {
                Ok(n) => nums.push(Some(n)),
                Err(_) => return Err(RangeError::Syntax { at: token.start }),
            }
        }
    }

    let numeric_count = nums.iter().take_while(|x| x.is_some()).count();
    let had_wildcard = nums.iter().any(|x| x.is_none());
    let mut padded = nums.clone();
    while padded.len() < 3 {
        padded.push(None);
    }
    Ok((padded, numeric_count, false, None))
}

fn padded_tuple(fields: &[Option<u64>]) -> (u64, u64, u64) {
    (
        fields.first().and_then(|x| *x).unwrap_or(0),
        fields.get(1).and_then(|x| *x).unwrap_or(0),
        fields.get(2).and_then(|x| *x).unwrap_or(0),
    )
}

fn parse_expanding_operand(
    arm: &mut Arm,
    token: Token<'_>,
    kind: char,
    operand: &str,
    operand_start: usize,
) -> Result<(), RangeError> {
    let (fields, k, is_full, full) = parse_partial(token, operand, operand_start)?;
    if k == 0 {
        if kind == '^' || kind == '~' {
            return Err(RangeError::Syntax { at: token.start });
        }
        return Ok(());
    }

    let (major, minor, patch) = padded_tuple(&fields);
    let lower = if is_full {
        full.unwrap()
    } else {
        endpoint_version(major, minor, patch, vec![], vec![])
    };

    if kind == '^' || kind == '~' {
        add_check(arm, Op::Ge, lower);
        let bump = if kind == '^' {
            if major > 0 || k == 1 {
                0
            } else if minor > 0 || k == 2 {
                1
            } else {
                2
            }
        } else if k == 1 {
            0
        } else {
            1
        };
        let upper = upper_bound(major, minor, patch, token.start, bump)?;
        add_check(arm, Op::Lt, upper);
    } else if is_full {
        add_check(arm, Op::Eq, lower);
    } else if k == 1 || k == 2 {
        add_check(arm, Op::Ge, lower);
        let upper = upper_bound(major, minor, patch, token.start, if k == 1 { 0 } else { 1 })?;
        add_check(arm, Op::Lt, upper);
    }
    Ok(())
}

fn parse_comparator(token: Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let text = token.text;
    let (op, skip) = if text.starts_with(">=") {
        (Op::Ge, 2)
    } else if text.starts_with("<=") {
        (Op::Le, 2)
    } else if text.starts_with('>') {
        (Op::Gt, 1)
    } else if text.starts_with('<') {
        (Op::Lt, 1)
    } else if text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &text[skip..];
    let operand_start = token.start + skip;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = full_version_operand(token, operand, operand_start)?;
    add_check(arm, op, v);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    let minus_idx = tokens.iter().position(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || minus_idx != 1 {
        return Err(RangeError::Syntax {
            at: tokens[minus_idx].start,
        });
    }

    let lower_token = tokens[0];
    let upper_token = tokens[2];

    let (lower_fields, lower_k, lower_full, lower_v) =
        parse_partial(lower_token, lower_token.text, lower_token.start)?;
    let (upper_fields, upper_k, upper_full, upper_v) =
        parse_partial(upper_token, upper_token.text, upper_token.start)?;

    if lower_k == 0 {
        return Err(RangeError::Syntax {
            at: lower_token.start,
        });
    }
    if upper_k == 0 {
        return Err(RangeError::Syntax {
            at: upper_token.start,
        });
    }

    let (lm, ln, lp) = padded_tuple(&lower_fields);
    let lower = if lower_full {
        lower_v.unwrap()
    } else {
        endpoint_version(lm, ln, lp, vec![], vec![])
    };
    add_check(arm, Op::Ge, lower);

    if upper_full {
        add_check(arm, Op::Le, upper_v.unwrap());
    } else {
        let (um, un, up) = padded_tuple(&upper_fields);
        let upper = upper_bound(
            um,
            un,
            up,
            upper_token.start,
            if upper_k == 1 { 0 } else { 1 },
        )?;
        add_check(arm, Op::Lt, upper);
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut pos = 0usize;
    let mut saw_any_token = false;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token = &s[token_start..pos];

        if token == "||" {
            let arm_slice = &s[arm_start..token_start];
            let tokens = split_tokens(arm_slice, arm_start);
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: token_start });
            }
            let mut arm = Arm {
                checks: Vec::new(),
                admission_anchors: Vec::new(),
            };
            parse_arm(&tokens, &mut arm)?;
            arms.push(arm);
            arm_start = pos;
            saw_any_token = false;
        } else {
            saw_any_token = true;
        }
    }

    if !saw_any_token && arms.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let tail = &s[arm_start..];
    let tokens = split_tokens(tail, arm_start);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let mut arm = Arm {
        checks: Vec::new(),
        admission_anchors: Vec::new(),
    };
    parse_arm(&tokens, &mut arm)?;
    arms.push(arm);

    Ok(Range { arms })
}

fn parse_arm(tokens: &[Token<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    if let Some(minus) = tokens.iter().find(|t| t.text == "-") {
        let _ = minus;
        return parse_hyphen_arm(tokens, arm);
    }

    for token in tokens {
        let text = token.text;
        if text.starts_with('^') || text.starts_with('~') {
            let kind = text.as_bytes()[0] as char;
            let operand = &text[1..];
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: token.start });
            }
            parse_expanding_operand(arm, *token, kind, operand, token.start + 1)?;
        } else if text.starts_with('<')
            || text.starts_with('>')
            || text.starts_with('=')
            || text.starts_with('!')
        {
            if text.starts_with('!') {
                return Err(RangeError::Syntax { at: token.start });
            }
            parse_comparator(*token, arm)?;
        } else if text.starts_with('*') || text.starts_with('x') || text.starts_with('X') {
            parse_expanding_operand(arm, *token, 'b', text, token.start)?;
        } else if text.as_bytes().first().is_some_and(|b| b.is_ascii_digit()) {
            parse_expanding_operand(arm, *token, 'b', text, token.start)?;
        } else {
            return Err(RangeError::Syntax { at: token.start });
        }
    }
    Ok(())
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm
                .admission_anchors
                .iter()
                .any(|core| *core == version_core(v))
        {
            return false;
        }
        arm.checks.iter().all(|check| match check {
            Check::Compare { op, version } => op.test(compare_versions(v, version)),
        })
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(current) => {
                let ordering = compare_versions(version, &parsed[current]);
                if (maximum && ordering == Ordering::Greater)
                    || (!maximum && ordering == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }

    Ok(best.map(|index| candidates[index].clone()))
}
