    pub patch: u64,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let text = std::str::from_utf8(digits).unwrap();
    text.parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut id_start = start;
    let mut i = start;
    loop {
        if i == end || bytes[i] == b'.' {
            if i == id_start {
                return Err(VersionError::EmptyIdentifier { at: id_start });
            }
            if prerelease {
                let part = &bytes[id_start..i];
                if part.iter().all(u8::is_ascii_digit)
                    && part.len() > 1
                    && part[0] == b'0'
                {
                    return Err(VersionError::LeadingZero { at: id_start });
                }
            }
            out.push(s[id_start..i].to_string());
            if i == end {
                break;
            }
            id_start = i + 1;
        } else if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let minor = parse_core_field(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let patch = parse_core_field(s, &mut pos)?;

    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() {
        match bytes[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let plus = bytes[pre_start..]
                    .iter()
                    .position(|&b| b == b'+')
                    .map(|n| pre_start + n)
                    .unwrap_or(s.len());
                pre = parse_identifiers(s, pre_start, plus, true)?;
                pos = plus;
                if pos < s.len() {
                    build = parse_identifiers(s, pos + 1, s.len(), false)?;
                }
            }
            b'+' => {
                build = parse_identifiers(s, pos + 1, s.len(), false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

fn range_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')

fn tokens<'a>(s: &'a str, base: usize) -> Vec<Token<'a>> {
    let mut out = Vec::new();
    let bytes = s.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && range_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !range_ws(bytes[i]) {
            i += 1;
        }
        out.push(Token {
            text: &s[start..i],
            start: base + start,
        });
    }
    out
}

fn parse_partial(token: Token<'_>) -> Result<Partial, RangeError> {

            .map(|v| Partial {

                full: Some(v),
                wildcard: false,

                at: token.start,
                error,
            });
    }
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if fields.len() >= 3 && !has_wild {
        return parse_version(s)
            .map(|v| Partial {
                nums: vec![v.major, v.minor, v.patch],
                full: Some(v),
                wildcard: false,
            })
            .map_err(|error| RangeError::Version {
                at: token.start,
                error,
            });
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(syntax(token.start));
    }
    let mut nums = Vec::new();
    let mut saw_wild = false;
    for field in fields {
        if matches!(field, "*" | "x" | "X") {
            saw_wild = true;
        } else {

                || !field.bytes().all(|b| b.is_ascii_digit())

            }

            nums.push(n);
        }
    }
    Ok(Partial {
        wildcard: nums.is_empty(),
        nums,
        full: None,
    })
}

}

    if lo.wildcard {
        return Err(syntax(ts[0].start));
    }
    if hi.wildcard {
        return Err(syntax(ts[2].start));
    }
    let lower = if let Some(v) = lo.full.clone() {
        v
    } else {
        version_at(padded(&lo), Vec::new())
    };
    add_check(arm, Op::Ge, lower, true);
    if let Some(v) = hi.full {
        add_check(arm, Op::Le, v, true);
    } else {
        let tuple = padded(&hi);
        let upper = if hi.nums.len() == 1 {
            version_at(
                (tuple.0.checked_add(1).ok_or_else(|| syntax(ts[2].start))?, 0, 0),
                vec!["0".to_string()],
            )
        } else {
            version_at(
                (
                    tuple.0,
                    tuple.1.checked_add(1).ok_or_else(|| syntax(ts[2].start))?,
                    0,
                ),
                vec!["0".to_string()],
            )
        };
        add_check(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_arm(ts: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        checks: Vec::new(),
        anchors: Vec::new(),
    };
    if ts.iter().any(|t| t.text == "-") {
        parse_hyphen_arm(&mut arm, ts)?;
        return Ok(arm);
    }
    for token in ts {
        let first = token.text.as_bytes().first().copied();
        match first {
            Some(b'^') => add_caret_tilde(&mut arm, *token, true)?,
            Some(b'~') => add_caret_tilde(&mut arm, *token, false)?,
            Some(b'<') | Some(b'>') | Some(b'=') => add_explicit(&mut arm, *token)?,
            Some(b'0'..=b'9') | Some(b'*') | Some(b'x') | Some(b'X') => {
                add_partial_bare(&mut arm, *token)?
            }
            _ => return Err(syntax(token.start)),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pieces: Vec<(usize, usize)> = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && range_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !range_ws(bytes[i]) {
            i += 1;
        }
        pieces.push((start, i));
    }
    if pieces.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut previous_sep = None;
    for (start, end) in pieces {
        let part = &s[start..end];
        if part == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            let ts = current;
            arms.push(parse_arm(&ts)?);
            current = Vec::new();
            previous_sep = Some(start);
        } else {
            let t = Token {
                text: part,
                start,
            };
            if part.contains("||") {
                // A glued separator remains part of its operand and is parsed normally.
            }
            current.push(t);
            previous_sep = None;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    let _ = previous_sep;
    Ok(Range { arms })
}

fn check_arm(v: &Version, arm: &Arm) -> bool {
    if v.pre.len() > 0 && !arm.anchors.iter().any(|&c| c == core(v)) {
        return false;
    }
    arm.checks.iter().all(|c| {
        let ord = compare_versions(v, &c.version);
        match c.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| check_arm(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        match parse_version(candidate) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for (index, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(prev) = best {
            let ord = compare_versions(v, &parsed[prev]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(index);
            }
        } else {
            best = Some(index);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(b: u8) -> bool {
    b == b'.' || b == b'-' || b == b'+'
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let value = s[start..pos]
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;
        core[field] = value;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let start = pos;
                while pos < bytes.len() && bytes[pos] != b'+' {
                    pos += 1;
                }
                pre = parse_identifiers(s, start, pos, true)?;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, pos, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut item_start = start;

    loop {
        let mut i = item_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == item_start {
            return Err(VersionError::EmptyIdentifier { at: item_start });
        }

        let item = &s[item_start..i];
        if prerelease
            && item.bytes().all(|b| b.is_ascii_digit())
            && item.len() > 1
            && item.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: item_start });
        }

        out.push(item.to_string());
        if i == end {
            break;
        }
        item_start = i + 1;
        if item_start == end {
            return Err(VersionError::EmptyIdentifier { at: item_start });
        }
    }

    Ok(out)
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_strings(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric_identifier(x);
                    let yn = is_numeric_identifier(y);
                    let ord = match (xn, yn) {
                        (true, true) => compare_numeric_strings(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },

#[derive(Clone, Debug)]
pub struct Range {

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    admission_anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
enum Check {
    Compare { op: Op, version: Version },
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

impl Op {
    fn test(self, ordering: Ordering) -> bool {
        match self {
            Op::Eq => ordering == Ordering::Equal,
            Op::Lt => ordering == Ordering::Less,
            Op::Le => ordering != Ordering::Greater,
            Op::Gt => ordering == Ordering::Greater,
            Op::Ge => ordering != Ordering::Less,
        }
    }
}

#[derive(Clone, Copy, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens<'a>(s: &'a str, base: usize) -> Vec<Token<'a>> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut pos = 0;
    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }
        let start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        out.push(Token {
            text: &s[start..pos],
            start: base + start,
        });
    }
    out
}

fn version_core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint_version(major: u64, minor: u64, patch: u64, pre: Vec<String>, build: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build,
    }
}

fn upper_bound(
    major: u64,
    minor: u64,
    patch: u64,
    at: usize,
    bump: usize,
) -> Result<Version, RangeError> {
    let (m, n, p) = match bump {
        0 => (major.checked_add(1), Some(0), Some(0)),
        1 => (Some(major), minor.checked_add(1), Some(0)),
        _ => (Some(major), Some(minor), patch.checked_add(1)),
    };
    match (m, n, p) {
        (Some(m), Some(n), Some(p)) => Ok(endpoint_version(m, n, p, vec!["0".into()], vec![])),
        _ => Err(RangeError::Syntax { at }),
    }
}

fn full_version_operand(token: Token<'_>, operand: &str, operand_start: usize) -> Result<Version, RangeError> {
    parse_version(operand).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })
}

fn add_check(arm: &mut Arm, op: Op, version: Version) {
    if !version.pre.is_empty() {
        arm.admission_anchors.push(version_core(&version));
    }
    arm.checks.push(Check::Compare { op, version });
}

fn parse_partial(
    token: Token<'_>,
    operand: &str,
    operand_start: usize,
) -> Result<(Vec<Option<u64>>, usize, bool, Option<Version>), RangeError> {
    let bytes = operand.as_bytes();
    let fullish = bytes.contains(&b'-')
        || bytes.contains(&b'+')
        || (operand.split('.').count() >= 3
            && !operand
                .split('.')
                .any(|f| matches!(f, "*" | "x" | "X")));

    if fullish {
        let v = full_version_operand(token, operand, operand_start)?;
        return Ok((
            vec![Some(v.major), Some(v.minor), Some(v.patch)],
            3,
            true,
            Some(v),
        ));
    }

    let fields: Vec<&str> = operand.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: token.start });
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for field in &fields {
        if matches!(*field, "*" | "x" | "X") {
            wildcard_seen = true;
            nums.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: token.start });
            }
            match field.parse::<u64>() {
                Ok(n) => nums.push(Some(n)),
                Err(_) => return Err(RangeError::Syntax { at: token.start }),
            }
        }
    }

    let numeric_count = nums.iter().take_while(|x| x.is_some()).count();
    let had_wildcard = nums.iter().any(|x| x.is_none());
    let mut padded = nums.clone();
    while padded.len() < 3 {
        padded.push(None);
    }
    Ok((padded, numeric_count, false, None))
}

fn padded_tuple(fields: &[Option<u64>]) -> (u64, u64, u64) {
    (
        fields.first().and_then(|x| *x).unwrap_or(0),
        fields.get(1).and_then(|x| *x).unwrap_or(0),
        fields.get(2).and_then(|x| *x).unwrap_or(0),
    )
}

fn parse_expanding_operand(
    arm: &mut Arm,
    token: Token<'_>,
    kind: char,
    operand: &str,
    operand_start: usize,
) -> Result<(), RangeError> {
    let (fields, k, is_full, full) = parse_partial(token, operand, operand_start)?;
    if k == 0 {
        if kind == '^' || kind == '~' {
            return Err(RangeError::Syntax { at: token.start });
        }
        return Ok(());
    }

    let (major, minor, patch) = padded_tuple(&fields);
    let lower = if is_full {
        full.unwrap()
    } else {
        endpoint_version(major, minor, patch, vec![], vec![])
    };

    if kind == '^' || kind == '~' {
        add_check(arm, Op::Ge, lower);
        let bump = if kind == '^' {
            if major > 0 || k == 1 {
                0
            } else if minor > 0 || k == 2 {
                1
            } else {
                2
            }
        } else if k == 1 {
            0
        } else {
            1
        };
        let upper = upper_bound(major, minor, patch, token.start, bump)?;
        add_check(arm, Op::Lt, upper);
    } else if is_full {
        add_check(arm, Op::Eq, lower);
    } else if k == 1 || k == 2 {
        add_check(arm, Op::Ge, lower);
        let upper = upper_bound(major, minor, patch, token.start, if k == 1 { 0 } else { 1 })?;
        add_check(arm, Op::Lt, upper);
    }
    Ok(())
}

fn parse_comparator(token: Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let text = token.text;
    let (op, skip) = if text.starts_with(">=") {
        (Op::Ge, 2)
    } else if text.starts_with("<=") {
        (Op::Le, 2)
    } else if text.starts_with('>') {
        (Op::Gt, 1)
    } else if text.starts_with('<') {
        (Op::Lt, 1)
    } else if text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &text[skip..];
    let operand_start = token.start + skip;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = full_version_operand(token, operand, operand_start)?;
    add_check(arm, op, v);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    let minus_idx = tokens.iter().position(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || minus_idx != 1 {
        return Err(RangeError::Syntax {
            at: tokens[minus_idx].start,
        });
    }

    let lower_token = tokens[0];
    let upper_token = tokens[2];

    let (lower_fields, lower_k, lower_full, lower_v) =
        parse_partial(lower_token, lower_token.text, lower_token.start)?;
    let (upper_fields, upper_k, upper_full, upper_v) =
        parse_partial(upper_token, upper_token.text, upper_token.start)?;

    if lower_k == 0 {
        return Err(RangeError::Syntax {
            at: lower_token.start,
        });
    }
    if upper_k == 0 {
        return Err(RangeError::Syntax {
            at: upper_token.start,
        });
    }

    let (lm, ln, lp) = padded_tuple(&lower_fields);
    let lower = if lower_full {
        lower_v.unwrap()
    } else {
        endpoint_version(lm, ln, lp, vec![], vec![])
    };
    add_check(arm, Op::Ge, lower);

    if upper_full {
        add_check(arm, Op::Le, upper_v.unwrap());
    } else {
        let (um, un, up) = padded_tuple(&upper_fields);
        let upper = upper_bound(
            um,
            un,
            up,
            upper_token.start,
            if upper_k == 1 { 0 } else { 1 },
        )?;
        add_check(arm, Op::Lt, upper);
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut pos = 0usize;
    let mut saw_any_token = false;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token = &s[token_start..pos];

        if token == "||" {
            let arm_slice = &s[arm_start..token_start];
            let tokens = split_tokens(arm_slice, arm_start);
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: token_start });
            }
            let mut arm = Arm {
                checks: Vec::new(),
                admission_anchors: Vec::new(),
            };
            parse_arm(&tokens, &mut arm)?;
            arms.push(arm);
            arm_start = pos;
            saw_any_token = false;
        } else {
            saw_any_token = true;
        }
    }

    if !saw_any_token && arms.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let tail = &s[arm_start..];
    let tokens = split_tokens(tail, arm_start);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let mut arm = Arm {
        checks: Vec::new(),
        admission_anchors: Vec::new(),
    };
    parse_arm(&tokens, &mut arm)?;
    arms.push(arm);

    Ok(Range { arms })
}

fn parse_arm(tokens: &[Token<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    if let Some(minus) = tokens.iter().find(|t| t.text == "-") {
        let _ = minus;
        return parse_hyphen_arm(tokens, arm);
    }

    for token in tokens {
        let text = token.text;
        if text.starts_with('^') || text.starts_with('~') {
            let kind = text.as_bytes()[0] as char;
            let operand = &text[1..];
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: token.start });
            }
            parse_expanding_operand(arm, *token, kind, operand, token.start + 1)?;
        } else if text.starts_with('<')
            || text.starts_with('>')
            || text.starts_with('=')
            || text.starts_with('!')
        {
            if text.starts_with('!') {
                return Err(RangeError::Syntax { at: token.start });
            }
            parse_comparator(*token, arm)?;
        } else if text.starts_with('*') || text.starts_with('x') || text.starts_with('X') {
            parse_expanding_operand(arm, *token, 'b', text, token.start)?;
        } else if text.as_bytes().first().is_some_and(|b| b.is_ascii_digit()) {
            parse_expanding_operand(arm, *token, 'b', text, token.start)?;
        } else {
            return Err(RangeError::Syntax { at: token.start });
        }
    }
    Ok(())
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm
                .admission_anchors
                .iter()
                .any(|core| *core == version_core(v))
        {
            return false;
        }
        arm.checks.iter().all(|check| match check {
            Check::Compare { op, version } => op.test(compare_versions(v, version)),
        })
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]

    pub error: VersionError,

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,

    extrema(candidates, r, true)
}

    r: &Range,

fn extrema(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(current) => {
                let ordering = compare_versions(version, &parsed[current]);
                if (maximum && ordering == Ordering::Greater)
                    || (!maximum && ordering == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }

    Ok(best.map(|index| candidates[index].clone()))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_sep_missing(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut nums = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if pos == start {
            return Err(if is_sep_missing(bytes.get(pos).copied()) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        let field = &bytes[start..pos];
        if field.len() > 1 && field[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut value = 0u64;
        for &d in field {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((d - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        nums[i] = value;

        if i < 2 {
            match bytes.get(pos).copied() {
                Some(b'.') => pos += 1,
                b if is_sep_missing(b) => return Err(VersionError::MissingCore { at: pos }),
                _ => return Err(VersionError::InvalidChar { at: pos }),
            }
        }
    }

    if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if bytes.get(pos) == Some(&b'-') {
        pos += 1;
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());
        parse_identifiers(s, pos, end, true, &mut pre)?;
        pos = end;
    }

    if bytes.get(pos) == Some(&b'+') {
        pos += 1;
        parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if is_pre
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(())
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_ids(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric(x);
                    let yn = is_numeric(y);
                    let ord = match (xn, yn) {
                        (true, true) => cmp_numeric_ids(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < b.len() {
        while p < b.len() && ws(b[p]) {
            p += 1;
        }
        if p == b.len() {
            break;
        }
        let start = p;
        while p < b.len() && !ws(b[p]) {
            p += 1;
        }
        out.push(Token {
            text: s[start..p].to_string(),
            start,
        });
    }
    out
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
    k: usize,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_sign = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    if has_sign || (fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f))) {
        let v = parse_full_operand(s, at)?;
        return Ok(Partial {
            nums: vec![v.major, v.minor, v.patch],
            k: 3,
            full: Some(v),
        });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if is_wildcard(f) {
            wildcard_seen = true;
            continue;
        }
        if wildcard_seen
            || !f.bytes().all(|b| b.is_ascii_digit())
            || (f.len() > 1 && f.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }
        let mut n = 0u64;
        for b in f.bytes() {
            n = n
                .checked_mul(10)
                .and_then(|x| x.checked_add((b - b'0') as u64))
                .ok_or(RangeError::Syntax { at })?;
        }
        nums.push(n);
    }
    let k = nums.len();
    Ok(Partial {
        nums,
        full: None,
        k,
    })
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build: Vec::new(),
    }
}

fn padded(p: &Partial) -> Version {
    endpoint(
        *p.nums.first().unwrap_or(&0),
        *p.nums.get(1).unwrap_or(&0),
        *p.nums.get(2).unwrap_or(&0),
        p.full.as_ref().map(|v| v.pre.clone()).unwrap_or_default(),
    )
}

fn increment_bound(
    major: u64,
    minor: u64,
    patch: u64,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut m = major;
    let mut n = minor;
    let mut p = patch;
    match field {
        0 => m = m.checked_add(1).ok_or(RangeError::Syntax { at })?,
        1 => n = n.checked_add(1).ok_or(RangeError::Syntax { at })?,
        _ => p = p.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    if field == 0 {
        n = 0;
        p = 0;
    } else if field == 1 {
        p = 0;
    }
    Ok(endpoint(m, n, p, vec!["0".to_string()]))
}

fn push_cmp(arm: &mut Arm, op: Op, version: Version, explicit: bool) {
    if explicit && !version.pre.is_empty() {
        arm.anchors.push(version_tuple(&version));
    }
    arm.comparisons.push(Comparison { op, version });
}

fn expand_partial(
    arm: &mut Arm,
    p: Partial,
    at: usize,
    kind: char,
) -> Result<(), RangeError> {
    let k = p.k;
    if kind == '^' || kind == '~' {
        if k == 0 {
            return Err(RangeError::Syntax { at });
        }
        let lower = padded(&p);
        push_cmp(arm, Op::Ge, lower.clone(), p.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
        let (m, n, patch) = version_tuple(&lower);
        let field = if kind == '^' {
            if m > 0 || k == 1 {
                0
            } else if n > 0 || k == 2 {
                1
            } else {
                2
            }
        } else if k == 1 {
            0
        } else {
            1
        };
        let upper = increment_bound(m, n, patch, field, at)?;
        push_cmp(arm, Op::Lt, upper, false);
    } else if p.full.is_some() {
        push_cmp(arm, Op::Eq, p.full.unwrap(), true);
    } else if k == 0 {
        // Wildcard-only bare operands match all versions.
    } else if k == 3 {
        let v = padded(&p);
        push_cmp(arm, Op::Eq, v, false);
    } else {
        let lower = padded(&p);
        push_cmp(arm, Op::Ge, lower, false);
        let field = if k == 1 { 0 } else { 1 };
        let (m, n, patch) = version_tuple(&padded(&p));
        let upper = increment_bound(m, n, patch, field, at)?;
        push_cmp(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_comparator_token(arm: &mut Arm, t: &Token) -> Result<(), RangeError> {
    let (op, op_len) = if t.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if t.text.starts_with("<=") {
        (Op::Le, 2)
    } else if t.text.starts_with('>') {
        (Op::Gt, 1)
    } else if t.text.starts_with('<') {
        (Op::Lt, 1)
    } else if t.text.starts_with('=') {
        (Op::Eq, 1)
    } else {
        return Err(RangeError::Syntax { at: t.start });
    };
    let operand = &t.text[op_len..];
    let at = t.start + op_len;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: t.start });
    }
    let v = parse_full_operand(operand, at)?;
    push_cmp(arm, op, v, true);
    Ok(())
}

fn parse_arm(tokens: &[Token]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };
    if tokens.is_empty() {
        return Ok(arm);
    }

    if let Some(minus) = tokens.iter().position(|t| t.text == "-") {
        if tokens.len() != 3 || minus != 1 {
            return Err(RangeError::Syntax {
                at: tokens[minus].start,
            });
        }
        let lower = parse_partial(&tokens[0].text, tokens[0].start)?;
        let upper = parse_partial(&tokens[2].text, tokens[2].start)?;
        if lower.k == 0 {
            return Err(RangeError::Syntax {
                at: tokens[0].start,
            });
        }
        if upper.k == 0 {
            return Err(RangeError::Syntax {
                at: tokens[2].start,
            });
        }

        let lo = padded(&lower);
        push_cmp(
            &mut arm,
            Op::Ge,
            lo,
            lower.full.as_ref().is_some_and(|v| !v.pre.is_empty()),
        );
        if let Some(v) = upper.full {
            push_cmp(&mut arm, Op::Le, v, true);
        } else if upper.k == 3 {
            push_cmp(&mut arm, Op::Le, padded(&upper), false);
        } else {
            let base = padded(&upper);
            push_cmp(&mut arm, Op::Lt, increment_bound(
                base.major,
                base.minor,
                base.patch,
                if upper.k == 1 { 0 } else { 1 },
                tokens[2].start,
            )?, false);
        }
        return Ok(arm);
    }

    for t in tokens {
        let text = t.text.as_str();
        if text.starts_with('^') || text.starts_with('~') {
            let kind = text.as_bytes()[0] as char;
            let operand = &text[1..];
            let at = t.start + 1;
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: t.start });
            }
            let p = parse_partial(operand, at)?;
            expand_partial(&mut arm, p, at, kind)?;
        } else if text.starts_with('<')
            || text.starts_with('>')
            || text.starts_with('=')
        {
            parse_comparator_token(&mut arm, t)?;
        } else if text.starts_with(',')
            || text.starts_with('(')
            || text.starts_with(')')
            || text == "&"
        {
            return Err(RangeError::Syntax { at: t.start });
        } else if text.starts_with('*') || text.starts_with('x') || text.starts_with('X') {
            let p = parse_partial(text, t.start)?;
            expand_partial(&mut arm, p, t.start, ' ');
        } else if text.as_bytes().first().is_some_and(|b| b.is_ascii_digit()) {
            let p = parse_partial(text, t.start)?;
            expand_partial(&mut arm, p, t.start, ' ');
        } else {
            return Err(RangeError::Syntax { at: t.start });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut groups: Vec<Vec<Token>> = vec![Vec::new()];
    let mut separators = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            separators.push(tokens[i].start);
            groups.push(Vec::new());
        } else {
            groups.last_mut().unwrap().push(tokens[i].clone());
        }
        i += 1;
    }

    let mut arms = Vec::new();
    for (idx, group) in groups.iter().enumerate() {
        if group.is_empty() {
            let at = if idx < separators.len() {
                separators[idx]
            } else {
                s.len()
            };
            return Err(RangeError::Empty { at });
        }
        arms.push(parse_arm(group)?);
    }
    Ok(Range { arms })
}

fn comparison_matches(v: &Version, c: &Comparison) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.contains(&version_tuple(v))
        {
            return false;
        }
        arm.comparisons.iter().all(|c| comparison_matches(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (want_max && ord == Ordering::Greater)
                    || (!want_max && ord == Ordering::Less)
                {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
