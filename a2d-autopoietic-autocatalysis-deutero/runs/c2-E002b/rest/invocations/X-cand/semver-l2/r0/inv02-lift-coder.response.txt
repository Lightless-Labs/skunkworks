use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_ident_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            if !valid_ident_byte(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let ident = &s[id_start..i];
        if pre && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.len() > 1
            && ident.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(ident.to_owned());

        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }

    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let n = parse_core(s, &mut pos)?;
        if _ == 0 {
            // The loop variable is used only to keep the required parse ordering.
            let _ = n;
        }
    }

    // Reparse in a straightforward sequence, preserving the same left-to-right
    // error precedence.
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;

    if p < s.len() && !matches!(s.as_bytes()[p], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: p });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let pre_start = p + 1;
        let plus = s[pre_start..].find('+').map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        p = pre_end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let build_start = p + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
        p = s.len();
    }
    if p != s.len() {
        return Err(VersionError::InvalidChar { at: p });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    let an = numeric_identifier(a);
    let bn = numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => {
            let aa = a.trim_start_matches('0');
            let bb = b.trim_start_matches('0');
            let aa = if aa.is_empty() { "0" } else { aa };
            let bb = if bb.is_empty() { "0" } else { bb };
            aa.len().cmp(&bb.len()).then_with(|| aa.as_bytes().cmp(bb.as_bytes()))
        }
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_identifier(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    predicates: Vec<Predicate>,
    admission: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Predicate {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

impl Predicate {
    fn accepts(&self, v: &Version) -> bool {
        let ord = compare_versions(v, &self.version);
        match self.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    }
}

#[derive(Clone, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str, start: usize, end: usize) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && is_ws(bytes[i]) {
            i += 1;
        }
        if i == end {
            break;
        }
        let a = i;
        while i < end && !is_ws(bytes[i]) {
            i += 1;
        }
        out.push(Token {
            text: &s[a..i],
            start: a,
        });
    }
    out
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
}

impl Partial {
    fn numeric_count(&self) -> usize {
        self.fields.iter().take_while(|f| f.is_some()).count()
    }

    fn padded(&self) -> Version {
        Version {
            major: self.fields.first().and_then(|x| *x).unwrap_or(0),
            minor: self.fields.get(1).and_then(|x| *x).unwrap_or(0),
            patch: self.fields.get(2).and_then(|x| *x).unwrap_or(0),
            pre: Vec::new(),
            build: Vec::new(),
        }
    }

    fn lower(&self) -> Version {
        self.full.clone().unwrap_or_else(|| self.padded())
    }

    fn core(&self) -> (u64, u64, u64) {
        let p = self.padded();
        (p.major, p.minor, p.patch)
    }
}

fn wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn parse_partial(
    s: &str,
    token: &Token<'_>,
    operand: &str,
    operand_start: usize,
) -> Result<Partial, RangeError> {
    let has_suffix = operand.contains('-') || operand.contains('+');
    let fields_text: Vec<&str> = operand.split('.').collect();
    let full_form = has_suffix || (fields_text.len() >= 3 && !fields_text.iter().any(|f| wildcard(f)));
    if full_form {
        let v = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            full: Some(v),
        });
    }

    if fields_text.is_empty() || fields_text.len() > 3 || fields_text.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: token.start });
    }

    let mut fields = Vec::new();
    let mut saw_wildcard = false;
    for f in fields_text {
        if wildcard(f) {
            saw_wildcard = true;
            fields.push(None);
        } else {
            if saw_wildcard
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at: token.start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: token.start })?;
            fields.push(Some(n));
        }
    }
    if fields.is_empty() {
        return Err(RangeError::Syntax { at: token.start });
    }
    Ok(Partial { fields, full: None })
}

fn add_predicate(arm: &mut Arm, op: Op, version: Version, explicit: bool) {
    if explicit && !version.pre.is_empty() {
        arm.admission.push((version.major, version.minor, version.patch));
    }
    arm.predicates.push(Predicate { op, version });
}

fn bump_upper(
    mut v: Version,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let target = match field {
        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,
    };
    *target = target
        .checked_add(1)
        .ok_or(RangeError::Syntax { at })?;
    if field <= 1 {
        v.minor = if field == 0 { 0 } else { v.minor };
    }
    if field <= 2 {
        v.patch = if field <= 1 { 0 } else { v.patch };
    }
    v.pre = vec!["0".to_owned()];
    v.build.clear();
    Ok(v)
}

fn expand_partial(
    arm: &mut Arm,
    partial: Partial,
    token_start: usize,
) -> Result<(), RangeError> {
    let k = partial.numeric_count();
    if k == 0 {
        return Ok(());
    }
    let lower = partial.lower();
    let core = partial.core();
    if let Some(ref full) = partial.full {
        if !full.pre.is_empty() {
            arm.admission.push(core);
        }
        add_predicate(arm, Op::Eq, lower, false);
        return Ok(());
    }
    add_predicate(arm, Op::Ge, lower, false);
    if k < 3 {
        let upper = bump_upper(partial.padded(), k - 1, token_start)?;
        add_predicate(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_explicit_comparator(
    arm: &mut Arm,
    token: &Token<'_>,
) -> Result<(), RangeError> {
    let (op, skip) = if token.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.text.starts_with("<=") {
        (Op::Le, 2)
    } else if token.text.starts_with('>') {
        (Op::Gt, 1)
    } else if token.text.starts_with('<') {
        (Op::Lt, 1)
    } else {
        (Op::Eq, 1)
    };
    let operand_start = token.start + skip;
    let operand = &token.text[skip..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = parse_version(operand).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })?;
    add_predicate(arm, op, v, true);
    Ok(())
}

fn parse_arm(s: &str, tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        predicates: Vec::new(),
        admission: Vec::new(),
    };
    if let Some(minus) = tokens.iter().find(|t| t.text == "-") {
        if tokens.len() != 3 || tokens[1].text != "-" {
            return Err(RangeError::Syntax { at: minus.start });
        }
        let left = parse_partial(s, &tokens[0], tokens[0].text, tokens[0].start)?;
        let right = parse_partial(s, &tokens[2], tokens[2].text, tokens[2].start)?;
        if left.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: tokens[0].start });
        }
        if right.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: tokens[2].start });
        }

        let lower = left.lower();
        if let Some(ref v) = left.full {
            if !v.pre.is_empty() {
                arm.admission.push((v.major, v.minor, v.patch));
            }
        } else if !lower.pre.is_empty() {
            arm.admission.push((lower.major, lower.minor, lower.patch));
        }
        add_predicate(&mut arm, Op::Ge, lower, false);

        if let Some(upper) = right.full {
            if !upper.pre.is_empty() {
                arm.admission.push((upper.major, upper.minor, upper.patch));
            }
            add_predicate(&mut arm, Op::Le, upper, false);
        } else {
            let k = right.numeric_count();
            let upper = if k >= 3 {
                right.padded()
            } else {
                bump_upper(right.padded(), k - 1, tokens[2].start)?
            };
            if k >= 3 {
                add_predicate(&mut arm, Op::Le, upper, false);
            } else {
                add_predicate(&mut arm, Op::Lt, upper, false);
            }
        }
        return Ok(arm);
    }

    for token in tokens {
        if token.text.starts_with('^') || token.text.starts_with('~') {
            let caret = token.text.starts_with('^');
            let operand = &token.text[1..];
            let operand_start = token.start + 1;
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: token.start });
            }
            let partial = parse_partial(s, token, operand, operand_start)?;
            let k = partial.numeric_count();
            if k == 0 {
                return Err(RangeError::Syntax { at: token.start });
            }
            let lower = partial.lower();
            if let Some(ref v) = partial.full {
                if !v.pre.is_empty() {
                    arm.admission.push((v.major, v.minor, v.patch));
                }
            }
            add_predicate(&mut arm, Op::Ge, lower, false);

            let field = if caret {
                if partial.padded().major > 0 || k == 1 {
                    0
                } else if partial.padded().minor > 0 || k == 2 {
                    1
                } else {
                    2
                }
            } else if k == 1 {
                0
            } else {
                1
            };
            let upper = bump_upper(partial.padded(), field, token.start)?;
            add_predicate(&mut arm, Op::Lt, upper, false);
        } else if matches!(token.text.as_bytes().first(), Some(b'<') | Some(b'>') | Some(b'=')) {
            parse_explicit_comparator(&mut arm, token)?;
        } else if token.text.starts_with(',')
            || token.text.starts_with('(')
            || token.text.starts_with(')')
            || token.text == "&"
        {
            return Err(RangeError::Syntax { at: token.start });
        } else {
            let first = token.text.as_bytes()[0];
            if !(first.is_ascii_digit() || wildcard(token.text)) {
                return Err(RangeError::Syntax { at: token.start });
            }
            let partial = parse_partial(s, token, token.text, token.start)?;
            expand_partial(&mut arm, partial, token.start)?;
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    let mut saw_any = false;

    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let tokens_start = i;
        let mut end = i;
        let mut separator = None;
        while end < bytes.len() {
            if is_ws(bytes[end]) {
                let mut j = end;
                while j < bytes.len() && is_ws(bytes[j]) {
                    j += 1;
                }
                if j + 1 < bytes.len() && &s[j..j + 2] == "||" {
                    separator = Some((j, j + 2));
                    break;
                }
                end = j;
                continue;
            }
            end += 1;
        }

        let arm_end = separator.map(|x| x.0).unwrap_or(bytes.len());
        let tokens = tokenize(s, tokens_start, arm_end);
        if tokens.is_empty() {
            return Err(RangeError::Empty { at: arm_start });
        }
        arms.push(parse_arm(s, &tokens)?);
        saw_any = true;

        if let Some((sep_start, after)) = separator {
            if after < bytes.len() && !is_ws(bytes[after]) {
                // A glued separator remains part of its token and is parsed as such.
            }
            let mut next = after;
            while next < bytes.len() && is_ws(bytes[next]) {
                next += 1;
            }
            if next == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
            if next + 1 < bytes.len() && &s[next..next + 2] == "||" {
                return Err(RangeError::Empty { at: next });
            }
            arm_start = sep_start;
            i = next;
        } else {
            break;
        }
    }

    if !saw_any {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm
                .admission
                .iter()
                .any(|&(a, b, c)| (v.major, v.minor, v.patch) == (a, b, c))
        {
            return false;
        }
        arm.predicates.iter().all(|p| p.accepts(v))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, text) in candidates.iter().enumerate() {
        let v = parse_version(text).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
