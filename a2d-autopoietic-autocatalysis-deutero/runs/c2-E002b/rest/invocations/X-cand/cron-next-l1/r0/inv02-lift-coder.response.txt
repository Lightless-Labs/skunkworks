#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_star: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &allowed)| allowed.then_some(value as u8))
            .collect()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
    width: u32,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if step {
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (min, max) = domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field, selected: &mut [bool]) -> Result<bool, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let width = max - min + 1;
    let exact_star = bytes == b"*";

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_start) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some(slash + 1)),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        enum Base {
            Star,
            Single(usize, usize),
            Range(usize, usize, usize, usize),
        }

        let parsed_base = if base == b"*" {
            Base::Star
        } else {
            match hyphens.as_slice() {
                [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                    Base::Single(0, base.len())
                }
                [hyphen]
                    if *hyphen > 0
                        && *hyphen + 1 < base.len()
                        && base[..*hyphen].iter().all(u8::is_ascii_digit)
                        && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
                {
                    Base::Range(0, *hyphen, *hyphen + 1, base.len())
                }
                _ => {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
            }
        };

        let (start, end) = match parsed_base {
            Base::Star => (min, max),
            Base::Single(a, b) => {
                let n = parse_number(base, a, b, field, false, width)?;
                (n, if step_start.is_some() { max } else { n })
            }
            Base::Range(a, b, c, d) => {
                let start = parse_number(base, a, b, field, false, width)?;
                let end = parse_number(base, c, d, field, false, width)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + c,
                    });
                }
                (start, end)
            }
        };

        let step = match step_start {
            None => None,
            Some(step_start) => {
                if step_start == item.len()
                    || !item[step_start..].iter().all(u8::is_ascii_digit)
                {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                Some(parse_number(
                    item,
                    step_start,
                    item.len(),
                    field,
                    true,
                    width,
                )?)
            }
        };

        match step {
            None => {
                for value in start..=end {
                    selected[(value - min) as usize] = true;
                }
            }
            Some(step) => {
                let mut value = start;
                loop {
                    selected[(value - min) as usize] = true;
                    match value.checked_add(step) {
                        Some(next) if next <= end => value = next,
                        _ => break,
                    }
                }
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(exact_star)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 31],
        vec![false; 12],
        vec![false; 7],
    ];
    let mut day_star = [false; 2];

    for index in 0..5 {
        let is_star = parse_field(fields[index], field_kinds[index], &mut selected[index])?;
        if index == 2 {
            day_star[0] = is_star;
        } else if index == 4 {
            day_star[1] = is_star;
        }
    }

    Ok(Cron { selected, day_star })
}
