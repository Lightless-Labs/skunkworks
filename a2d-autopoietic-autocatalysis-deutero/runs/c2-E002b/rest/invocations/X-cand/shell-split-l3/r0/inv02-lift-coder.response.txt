#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    Bare,
    Single { start: usize },
    Double { start: usize },
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_literal(word: &mut Word, c: char, wildcard_active: bool) {
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, wildcard_active: bool) {
    for c in value.chars() {
        append_literal(word, c, wildcard_active);
    }
}

fn find_braced_end(chars: &[char], start: usize) -> Option<usize> {
    let mut i = start;
    while i < chars.len() {
        if chars[i] == '\\' {
            i += 1;
            if i < chars.len() {
                i += 1;
            }
        } else if chars[i] == '}' {
            return Some(i);
        } else {
            i += 1;
        }
    }
    None
}

fn decode_default(chars: &[char]) -> String {
    let mut output = String::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i] == '\\' && i + 1 < chars.len() {
            output.push(chars[i + 1]);
            i += 2;
        } else {
            output.push(chars[i]);
            i += 1;
        }
    }
    output
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let chars: Vec<char> = input.chars().collect();
    let mut offsets = Vec::with_capacity(chars.len() + 1);
    let mut byte_pos = 0;
    for c in &chars {
        offsets.push(byte_pos);
        byte_pos += c.len_utf8();
    }
    offsets.push(byte_pos);

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::Bare;
    let mut i = 0;

    while i < chars.len() {
        let c = chars[i];
        let pos = offsets[i];

        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::Bare;
                } else {
                    word.text.push(c);
                }
                i += 1;
            }
            Quote::Double { .. } => {
                match c {
                    '"' => {
                        quote = Quote::Bare;
                        i += 1;
                    }
                    '\\' => {
                        if i + 1 >= chars.len() {
                            return Err(SplitError::TrailingEscape { pos });
                        }
                        let next = chars[i + 1];
                        if matches!(next, '"' | '\\' | '$' | '\n') {
                            if next != '\n' {
                                word.text.push(next);
                            }
                            i += 2;
                        } else {
                            word.text.push('\\');
                            i += 1;
                        }
                    }
                    '$' => {
                        if i + 1 < chars.len() && chars[i + 1] == '(' {
                            return Err(SplitError::CommandSubstitution { pos });
                        }
                        if i + 1 < chars.len() && chars[i + 1] == '{' {
                            let end = match find_braced_end(&chars, i + 2) {
                                Some(end) => end,
                                None => {
                                    return Err(SplitError::UnterminatedExpansion { pos });
                                }
                            };
                            let content = &chars[i + 2..end];
                            let name_end = content
                                .iter()
                                .position(|&ch| !is_name_continue(ch))
                                .unwrap_or(content.len());
                            if name_end == 0 || !is_name_start(content[0]) {
                                return Err(SplitError::InvalidExpansion { pos });
                            }
                            let name: String = content[..name_end].iter().collect();
                            let selected = if name_end == content.len() {
                                lookup(&name, env).unwrap_or("")
                            } else if name_end + 1 < content.len()
                                && content[name_end] == ':'
                                && content[name_end + 1] == '-'
                            {
                                let value = lookup(&name, env).unwrap_or("");
                                if value.is_empty() {
                                    let decoded = decode_default(&content[name_end + 2..]);
                                    append_expansion(&mut word, &decoded, false);
                                    active = true;
                                    i = end + 1;
                                    continue;
                                }
                                value
                            } else {
                                return Err(SplitError::InvalidExpansion { pos });
                            };
                            append_expansion(&mut word, selected, false);
                            active = true;
                            i = end + 1;
                        } else if i + 1 < chars.len() && is_name_start(chars[i + 1]) {
                            let mut end = i + 2;
                            while end < chars.len() && is_name_continue(chars[end]) {
                                end += 1;
                            }
                            let name: String = chars[i + 1..end].iter().collect();
                            append_expansion(&mut word, lookup(&name, env).unwrap_or(""), false);
                            active = true;
                            i = end;
                        } else {
                            word.text.push('$');
                            i += 1;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos }),
                    _ => {
                        word.text.push(c);
                        i += 1;
                    }
                }
            }
            Quote::Bare => {
                match c {
                    '\'' => {
                        active = true;
                        quote = Quote::Single { start: pos };
                        i += 1;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double { start: pos };
                        i += 1;
                    }
                    '\\' => {
                        if i + 1 >= chars.len() {
                            return Err(SplitError::TrailingEscape { pos });
                        }
                        let next = chars[i + 1];
                        if next == '\n' {
                            i += 2;
                        } else {
                            active = true;
                            append_literal(&mut word, next, false);
                            i += 2;
                        }
                    }
                    '$' => {
                        if i + 1 < chars.len() && chars[i + 1] == '(' {
                            return Err(SplitError::CommandSubstitution { pos });
                        }
                        if i + 1 < chars.len() && chars[i + 1] == '{' {
                            let end = match find_braced_end(&chars, i + 2) {
                                Some(end) => end,
                                None => {
                                    return Err(SplitError::UnterminatedExpansion { pos });
                                }
                            };
                            let content = &chars[i + 2..end];
                            let name_end = content
                                .iter()
                                .position(|&ch| !is_name_continue(ch))
                                .unwrap_or(content.len());
                            if name_end == 0 || !is_name_start(content[0]) {
                                return Err(SplitError::InvalidExpansion { pos });
                            }
                            let name: String = content[..name_end].iter().collect();
                            let selected = if name_end == content.len() {
                                lookup(&name, env).unwrap_or("")
                            } else if name_end + 1 < content.len()
                                && content[name_end] == ':'
                                && content[name_end + 1] == '-'
                            {
                                let value = lookup(&name, env).unwrap_or("");
                                if value.is_empty() {
                                    let decoded = decode_default(&content[name_end + 2..]);
                                    append_expansion(&mut word, &decoded, true);
                                    active = true;
                                    i = end + 1;
                                    continue;
                                }
                                value
                            } else {
                                return Err(SplitError::InvalidExpansion { pos });
                            };
                            append_expansion(&mut word, selected, true);
                            active = true;
                            i = end + 1;
                        } else if i + 1 < chars.len() && is_name_start(chars[i + 1]) {
                            let mut end = i + 2;
                            while end < chars.len() && is_name_continue(chars[end]) {
                                end += 1;
                            }
                            let name: String = chars[i + 1..end].iter().collect();
                            append_expansion(&mut word, lookup(&name, env).unwrap_or(""), true);
                            active = true;
                            i = end;
                        } else {
                            active = true;
                            word.text.push('$');
                            i += 1;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos }),
                    ch if is_separator(ch) => {
                        if active {
                            words.push(word);
                            word = Word {
                                text: String::new(),
                                glob: Vec::new(),
                            };
                            active = false;
                        }
                        i += 1;
                    }
                    _ => {
                        active = true;
                        append_literal(&mut word, c, true);
                        i += 1;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single { start } => return Err(SplitError::UnterminatedSingle { pos: start }),
        Quote::Double { start } => return Err(SplitError::UnterminatedDouble { pos: start }),
        Quote::Bare => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
