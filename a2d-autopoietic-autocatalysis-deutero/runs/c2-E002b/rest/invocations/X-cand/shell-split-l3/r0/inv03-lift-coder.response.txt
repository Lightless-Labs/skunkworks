#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_literal(word: &mut Word, c: char, wildcard_active: bool) {
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_atomic(word: &mut Word, value: &str, wildcard_active: bool) {
    for c in value.chars() {
        append_literal(word, c, wildcard_active);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().expect("valid character boundary");
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let escaped_end = next_i + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$') {
                        append_literal(&mut word, escaped, false);
                        i = escaped_end;
                    } else if escaped == '\n' {
                        i = escaped_end;
                    } else {
                        append_literal(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let end = j + ch.len_utf8();
                            if ch == '\\' {
                                if end < input.len() {
                                    let skipped = input[end..].chars().next().unwrap();
                                    j = end + skipped.len_utf8();
                                } else {
                                    j = end;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = end;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default_text) = if let Some(marker) = contents.find(":-") {
                            (&contents[..marker], Some(&contents[marker + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_continue)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        active = true;
                        let value = match (lookup(env, name), default_text) {
                            (Some(v), Some(default)) if !v.is_empty() => v,
                            (Some(v), None) => v,
                            (_, Some(default)) => {
                                let mut decoded = String::new();
                                let mut k = 0usize;
                                while k < default.len() {
                                    let ch = default[k..].chars().next().unwrap();
                                    let end = k + ch.len_utf8();
                                    if ch == '\\' && end < default.len() {
                                        let escaped = default[end..].chars().next().unwrap();
                                        decoded.push(escaped);
                                        k = end + escaped.len_utf8();
                                    } else {
                                        decoded.push(ch);
                                        k = end;
                                    }
                                }
                                append_atomic(&mut word, &decoded, false);
                                i = close + 1;
                                continue;
                            }
                            (None, None) => "",
                        };
                        append_atomic(&mut word, value, false);
                        i = close + 1;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            active = true;
                            if let Some(value) = lookup(env, name) {
                                append_atomic(&mut word, value, false);
                            }
                            i = end;
                        } else {
                            append_literal(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_pos = i;
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_pos = i;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let escaped_end = next_i + escaped.len_utf8();
                    if escaped != '\n' {
                        active = true;
                        append_literal(&mut word, escaped, false);
                    }
                    i = escaped_end;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let end = j + ch.len_utf8();
                            if ch == '\\' {
                                if end < input.len() {
                                    let skipped = input[end..].chars().next().unwrap();
                                    j = end + skipped.len_utf8();
                                } else {
                                    j = end;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = end;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let (name, default_text) = if let Some(marker) = contents.find(":-") {
                            (&contents[..marker], Some(&contents[marker + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_continue)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        active = true;
                        let value = match (lookup(env, name), default_text) {
                            (Some(v), Some(_)) if !v.is_empty() => Some(v),
                            (Some(v), None) => Some(v),
                            (_, Some(default)) => {
                                let mut decoded = String::new();
                                let mut k = 0usize;
                                while k < default.len() {
                                    let ch = default[k..].chars().next().unwrap();
                                    let end = k + ch.len_utf8();
                                    if ch == '\\' && end < default.len() {
                                        let escaped = default[end..].chars().next().unwrap();
                                        decoded.push(escaped);
                                        k = end + escaped.len_utf8();
                                    } else {
                                        decoded.push(ch);
                                        k = end;
                                    }
                                }
                                append_atomic(&mut word, &decoded, true);
                                i = close + 1;
                                continue;
                            }
                            (None, None) => Some(""),
                        };
                        append_atomic(&mut word, value.unwrap_or(""), true);
                        i = close + 1;
                    } else if next_i < input.len() {
                        let first = input[next_i..].chars().next().unwrap();
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            active = true;
                            if let Some(value) = lookup(env, name) {
                                append_atomic(&mut word, value, true);
                            }
                            i = end;
                        } else {
                            active = true;
                            append_literal(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        active = true;
                        append_literal(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    active = true;
                    append_literal(&mut word, c, matches!(c, '*' | '?' | '['));
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
