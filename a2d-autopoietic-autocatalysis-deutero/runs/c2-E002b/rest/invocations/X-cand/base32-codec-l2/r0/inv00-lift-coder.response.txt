#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let symbols = alphabet_bytes(alphabet);
    let data_len = (input.len() * 8 + 4) / 5;
    let total_len = match padding {
        Padding::Omitted => data_len,
        Padding::Required => (data_len + 7) / 8 * 8,
    };
    let mut output = String::with_capacity(total_len);

    for group_start in (0..input.len() * 8).step_by(5) {
        let mut value = 0u8;
        for bit_offset in 0..5 {
            value <<= 1;
            let bit_index = group_start + bit_offset;
            if bit_index < input.len() * 8 {
                let byte = input[bit_index / 8];
                value |= (byte >> (7 - bit_index % 8)) & 1;
            }
        }
        output.push(symbols[value as usize] as char);
    }

    while output.len() < total_len {
        output.push('=');
    }
    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let symbols = alphabet_bytes(alphabet);

    // Lexical validation checks the entire input before considering padding.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && !symbols.contains(&byte) {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let first_padding = bytes.iter().position(|&byte| byte == b'=');
    let p = first_padding.unwrap_or(bytes.len());

    match padding {
        Padding::Omitted if first_padding.is_some() => {
            return Err(DecodeError::Padding { offset: p });
        }
        Padding::Required => {
            if let Some(index) = first_padding {
                if let Some(offset) = bytes[index..].iter().position(|&byte| byte != b'=') {
                    return Err(DecodeError::Padding {
                        offset: index + offset,
                    });
                }
            }
        }
        Padding::Omitted => {}
    }

    let d = p;
    let residue = d % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength {
            offset: bytes.len(),
        });
    }

    if padding == Padding::Required {
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => unreachable!(),
        };
        if bytes.len() % 8 != 0 || bytes.len() - p != expected_padding {
            return Err(DecodeError::Padding { offset: p });
        }
    }

    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };

    if unused_bits != 0 {
        let last_value = symbols
            .iter()
            .position(|&symbol| symbol == bytes[d - 1])
            .expect("lexical validation guarantees alphabet membership");
        if last_value & ((1usize << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset: d - 1 });
        }
    }

    let mut output = Vec::with_capacity(d * 5 / 8);
    let mut accumulator = 0u32;
    let mut bit_count = 0u32;

    for &byte in &bytes[..d] {
        let value = symbols
            .iter()
            .position(|&symbol| symbol == byte)
            .expect("lexical validation guarantees alphabet membership") as u32;
        accumulator = (accumulator << 5) | value;
        bit_count += 5;

        if bit_count >= 8 {
            bit_count -= 8;
            output.push((accumulator >> bit_count) as u8);
            accumulator &= (1u32 << bit_count) - 1;
        }
    }

    Ok(output)
}
