#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = hex_value(bytes[i + 1]).ok_or(offset + i)?;
            let lo = hex_value(bytes[i + 2]).ok_or(offset + i)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn split_path<'a>(path: &'a str, error: fn(usize) -> Error) -> Result<(Vec<(&'a str, usize)>, bool), Error> {
    if !path.starts_with('/') {
        return Err(error(0));
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let mut segments = Vec::new();
    let mut start = 1;
    let bytes = path.as_bytes();

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if i == start {
                    return Err(error(start));
                }
                return Ok((segments_with_last(path, start, i, segments), true));
            }
            if i == start {
                return Err(error(start));
            }
            segments.push((&path[start..i], start));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(error(start));
    }
    segments.push((&path[start..], start));
    Ok((segments, false))
}

fn segments_with_last<'a>(
    path: &'a str,
    start: usize,
    end: usize,
    mut segments: Vec<(&'a str, usize)>,
) -> Vec<(&'a str, usize)> {
    segments.push((&path[start..end], start));
    segments
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (raw_segments, trailing_slash) =
            split_path(pattern, |at| Error::InvalidPattern { at })?;

        let mut segments = Vec::with_capacity(raw_segments.len());
        let mut names = std::collections::HashSet::new();
        let mut rank = Vec::with_capacity(raw_segments.len() + 1);

        for (index, (raw, offset)) in raw_segments.iter().enumerate() {
            if let Some(name) = raw.strip_prefix(':') {
                if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                    return Err(Error::InvalidPattern { at: *offset });
                }
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            } else if let Some(name) = raw.strip_prefix('*') {
                if index + 1 != raw_segments.len() {
                    return Err(Error::InvalidPattern { at: *offset });
                }
                if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                    return Err(Error::InvalidPattern { at: *offset });
                }
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            } else {
                let decoded = decode_segment(raw, *offset)
                    .map_err(|at| Error::InvalidPattern { at })?;
                segments.push(Segment::Static(decoded));
                rank.push(2);
            }
        }

        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];

        let (raw_segments, trailing_slash) =
            split_path(head, |at| Error::InvalidPath { at })?;
        let mut decoded_segments = Vec::with_capacity(raw_segments.len());
        for (raw, offset) in &raw_segments {
            decoded_segments.push(
                decode_segment(raw, *offset).map_err(|at| Error::InvalidPath { at })?,
            );
        }

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != trailing_slash
            {
                continue;
            }

            let mut params = Vec::new();
            let mut path_index = 0;
            let mut matched = true;

            for segment in &route.segments {
                match segment {
                    Segment::Static(expected) => {
                        if decoded_segments.get(path_index) != Some(expected) {
                            matched = false;
                            break;
                        }
                        path_index += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = decoded_segments.get(path_index) {
                            params.push((name.clone(), value.clone()));
                            path_index += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = decoded_segments[path_index..].join("/");
                        params.push((name.clone(), value));
                        path_index = decoded_segments.len();
                    }
                }
            }

            if matched && path_index == decoded_segments.len() {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            let has_head = candidates
                .iter()
                .any(|(route, _)| route.method == "HEAD");
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (route, _))| {
                    (route.method == if has_head { "HEAD" } else { "GET" }).then_some(i)
                })
                .collect()
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (route, _))| (route.method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect::<Vec<_>>();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut best = eligible[0];
        for &i in &eligible[1..] {
            if candidates[i].0.rank > candidates[best].0.rank {
                best = i;
            }
        }

        let (route, params) = &candidates[best];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing_slash != trailing_slash
        {
            if route.trailing_slash {
                Some(format!("{head}/"))
            } else {
                Some(head[..head.len() - 1].to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }
}
