#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head)?;

        let mut candidates = Vec::new();
        for (idx, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some(params) = match_route(&route.segments, &request_segments) {
                candidates.push((idx, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_has_candidate = method == "HEAD"
            && candidates
                .iter()
                .any(|(idx, _)| self.routes[*idx].method == "HEAD");
        let effective_method = if method == "HEAD" && !head_has_candidate {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(idx, _)| self.routes[*idx].method == effective_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(idx, _)| self.routes[*idx].method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut winner: Option<(usize, Vec<(String, String)>)> = None;
        for (idx, params) in eligible {
            match &winner {
                None => winner = Some((*idx, params.clone())),
                Some((best_idx, _)) => {
                    let candidate_rank = &self.routes[*idx].rank;
                    let best_rank = &self.routes[*best_idx].rank;
                    if candidate_rank > best_rank {
                        winner = Some((*idx, params.clone()));
                    }
                }
            }
        }

        let (idx, params) = winner.expect("eligible candidates are nonempty");
        let route = &self.routes[idx];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && route.trailing != request_trailing
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: idx,
            params,
            redirect,
        })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    parse_segments(pattern, true)
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    let (segments, trailing) = parse_segments(path, false)?;
    let decoded = segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(v) => v,
            _ => unreachable!(),
        })
        .collect();
    Ok((decoded, trailing))
}

fn parse_segments(
    input: &str,
    pattern: bool,
) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };

    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let mut start = 1usize;
    let mut wildcard_seen = false;

    while start < end {
        let next = bytes[start..end]
            .iter()
            .position(|b| *b == b'/')
            .map(|n| start + n)
            .unwrap_or(end);

        if next == start {
            return Err(invalid(start));
        }

        if wildcard_seen {
            return Err(invalid(start));
        }

        let raw = &input[start..next];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let is_wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];
            if !valid_capture_name(name) || names.iter().any(|n| n == name) {
                return Err(invalid(start));
            }
            names.push(name.to_string());

            if is_wildcard && next != end {
                return Err(invalid(start));
            }
            if is_wildcard {
                wildcard_seen = true;
                segments.push(Segment::Wildcard(name.to_string()));
            } else {
                segments.push(Segment::Param(name.to_string()));
            }
        } else {
            let decoded = decode_static(raw, start).map_err(invalid)?;
            segments.push(Segment::Static(decoded));
        }

        if next == end {
            break;
        }
        start = next + 1;
    }

    Ok((segments, trailing))
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    match bytes.next() {
        Some(b'a'..=b'z') => bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'),
        _ => false,
    }
}

fn decode_static(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(offset + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(offset + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(offset + i);
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if request.get(i)? != value {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(i)?.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                let value = request[i..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    if i == request.len() {
        Some(params)
    } else {
        None
    }
}
