    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,

fn field_bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn domain_width(field: Field) -> u32 {
    let (min, max) = field_bounds(field);
    max - min + 1
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if step {
        if value == 0 || value > domain_width(field) {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = field_bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(value)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[&[u8]] = match field {
        Field::Month => &[
            b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN", b"JUL", b"AUG", b"SEP", b"OCT",

        ],
        Field::DayOfWeek => &[b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT"],
        _ => return None,
    };

    names.iter().position(|name| {
        name.len() == bytes.len()
            && name
                .iter()
                .zip(bytes.iter())
                .all(|(a, b)| a.eq_ignore_ascii_case(b))
    }).map(|index| {
        if field == Field::Month {
            index as u32 + 1
        } else {
            index as u32
        }
    })
}

fn parse_endpoint(
    bytes: &[u8],
    field: Field,
    offset: usize,
) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        return parse_number(bytes, field, offset, false);
    }
    if let Some(value) = name_value(field, bytes) {
        return Ok(value);
    }
    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        let step_part = &item[slash + 1..];
        if step_part.is_empty() || !step_part.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        (&item[..slash], Some(step_part), Some(item_offset + slash + 1))
    } else {
        (item, None, None)
    };

    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    if is_star && !hyphens.is_empty() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }
    if hyphens.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let range = hyphens.first().copied();
    if is_star && range.is_some() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (start, end, end_offset) = if is_star {
        let (min, max) = field_bounds(field);
        (min, max, item_offset)
    } else if let Some(hyphen) = range {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        let start = parse_endpoint(&base[..hyphen], field, item_offset)?;
        let end = parse_endpoint(&base[hyphen + 1..], field, item_offset + hyphen + 1)?;
        (start, end, item_offset + hyphen + 1)
    } else {
        let start = parse_endpoint(base, field, item_offset)?;
        if step_bytes.is_some() {
            let (_, max) = field_bounds(field);
            (start, max, item_offset)
        } else {
            (start, start, item_offset)
        }
    };

    if range.is_some() && start > end {
        return Err(ParseError::ReversedRange {
            field,
            offset: end_offset,
        });
    }

    let step = if let (Some(bytes), Some(offset)) = (step_bytes, step_offset) {
        Some(parse_number(bytes, field, offset, true)?)
    } else {
        None
    };
    let step = step.unwrap_or(1);

    let mut value = start;
    while value <= end {
        let (min, _) = field_bounds(field);
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = field_bounds(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut offset = 0usize;

    for item in text.split(',') {
        parse_item(item.as_bytes(), offset, field, &mut selected)?;
        offset += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    for (piece, field) in pieces.iter().zip(fields) {
        selected.push(parse_field(piece, field)?);
    }

    Ok(Cron {
        selected: [
            selected.remove(0),
            selected.remove(0),
            selected.remove(0),
            selected.remove(0),
            selected.remove(0),
        ],
        day_of_month_unrestricted: pieces[2] == "*",
        day_of_week_unrestricted: pieces[4] == "*",
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]

    InvalidDateTime { component: DatePart },
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_days(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_days(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += month_days(at.year, month) as u32;
    }
    days += at.day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn selected_value(cron: &Cron, field: Field, value: u8) -> bool {
    let (min, _) = field_bounds(field);
    cron.selected[field_index(field)][(value as u32 - min) as usize]
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = field_bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    fn day_matches(&self, at: CivilDateTime) -> bool {
        let dom = selected_value(self, Field::DayOfMonth, at.day);
        let dow = selected_value(self, Field::DayOfWeek, weekday(at));

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_days(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(selected_value(self, Field::Minute, at.minute)
            && selected_value(self, Field::Hour, at.hour)
            && selected_value(self, Field::Month, at.month)
            && self.day_matches(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;

        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    if current.day < month_days(current.year, current.month) {
                        current.day += 1;
                    } else {
                        current.day = 1;
                        if current.month < 12 {
                            current.month += 1;
                        } else {
                            current.month = 1;
                            current.year += 1;
                        }
                    }
                }
            }

            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

    field: Field,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {

        return Err(invalid_syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {

        if value == 0 || value > width(field) {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (lo, hi) = domain(field);
        if value < lo || value > hi {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }

    Ok(value)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    if bytes.len() != 3 {
        return None;
    }
    let upper = [
        bytes[0].to_ascii_uppercase(),
        bytes[1].to_ascii_uppercase(),
        bytes[2].to_ascii_uppercase(),
    ];
    names.iter().position(|name| name.as_bytes() == upper).map(|i| {
        if field == Field::Month {
            (i + 1) as u32
        } else {
            i as u32
        }
    })
}

fn parse_endpoint(
    bytes: &[u8],
    field: Field,
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        return Ok((parse_number(bytes, field, offset, false)?, true));
    }
    if let Some(value) = name_value(field, bytes) {
        return Ok((value, false));
    }
    Err(invalid_syntax(field, offset))
}

fn parse_item(
    item: &[u8],
    field: Field,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }
    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, item_offset)
    };
    if base.is_empty() || step_bytes.is_some_and(|s| s.is_empty()) {
        return Err(invalid_syntax(field, item_offset));
    }

    let star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    if star && !hyphens.is_empty() {
        return Err(invalid_syntax(field, item_offset));
    }
    if !star && hyphens.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }

    let (start_bytes, end_bytes, start_offset, end_offset) = if star {
        (None, None, item_offset, item_offset)
    } else if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(invalid_syntax(field, item_offset));
        }
        (
            Some(&base[..hyphen]),
            Some(&base[hyphen + 1..]),
            item_offset,
            item_offset + hyphen + 1,
        )
    } else {
        (Some(base), None, item_offset, item_offset)
    };

    let step = if let Some(step_bytes) = step_bytes {
        if !step_bytes.iter().all(u8::is_ascii_digit) {
            return Err(invalid_syntax(field, item_offset));
        }
        Some(parse_number(step_bytes, field, step_offset, true)?)
    } else {
        None
    };

    let (lo, hi) = domain(field);
    let (start, end) = match (start_bytes, end_bytes) {
        (None, None) => (lo, hi),
        (Some(sb), Some(eb)) => {
            let (start, start_numeric) = parse_endpoint(sb, field, start_offset)?;
            let (end, end_numeric) = parse_endpoint(eb, field, end_offset)?;
            if start_numeric && start > hi {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start_offset,
                });
            }
            if end_numeric && end > hi {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_offset,
                });
            }
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (start, end)
        }
        (Some(sb), None) => {
            let (start, numeric) = parse_endpoint(sb, field, start_offset)?;
            if numeric && start > hi {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start_offset,
                });
            }
            if step.is_some() {
                (start, hi)
            } else {
                (start, start)
            }
        }
        _ => return Err(invalid_syntax(field, item_offset)),
    };

    let stride = step.unwrap_or(1);
    let mut value = start;
    loop {
        selected[(value - lo) as usize] = true;
        if value > end.saturating_sub(stride) {
            break;
        }
        value += stride;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, max) = domain(field);
    let mut selected = vec![false; (max - domain(field).0 + 1) as usize];
    let mut item_start = 0;
    for (i, part) in bytes
        .split(|&b| b == b',')
        .enumerate()
    {
        let _ = i;
        parse_item(part, field, item_start, &mut selected)?;
        item_start += part.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut parsed: Vec<Vec<bool>> = Vec::with_capacity(5);
    for (i, part) in parts.iter().enumerate() {
        parsed.push(parse_field(part, FIELDS[i])?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().unwrap(),
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (lo, _) = domain(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some((i as u32 + lo) as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let minute = self.selected[0][at.minute as usize];
        let hour = self.selected[1][at.hour as usize];
        let month = self.selected[3][at.month as usize - 1];
        if !(minute && hour && month) {
            return false;
        }

        let dom = self.selected[2][at.day as usize - 1];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                let last = at.day == month_length(at.year, at.month);
                if last {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            advance_minute(&mut current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) {
    if at.minute < 59 {
        at.minute += 1;
        return;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return;
    }
    at.hour = 0;
    if at.day < month_length(at.year, at.month) {
        at.day += 1;
        return;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return;
    }
    at.month = 1;
    at.year += 1;
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,

fn field_domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_decimal(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax_error(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(u32::from(byte - b'0')))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[(&[u8], u32)] = match field {
        Field::Month => &[
            (b"JAN", 1), (b"FEB", 2), (b"MAR", 3), (b"APR", 4),
            (b"MAY", 5), (b"JUN", 6), (b"JUL", 7), (b"AUG", 8),
            (b"SEP", 9), (b"OCT", 10), (b"NOV", 11), (b"DEC", 12),
        ],
        Field::DayOfWeek => &[
            (b"SUN", 0), (b"MON", 1), (b"TUE", 2), (b"WED", 3),
            (b"THU", 4), (b"FRI", 5), (b"SAT", 6),
        ],
        _ => return None,
    };
    names.iter().find_map(|(name, value)| {
        (bytes.len() == name.len()
            && bytes
                .iter()
                .zip(name.iter())
                .all(|(a, b)| a.to_ascii_uppercase() == *b))
        .then_some(*value)
    })
}

fn parse_endpoint(
    bytes: &[u8],
    field: Field,
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        let value = parse_decimal(bytes, field, offset)?;
        let (min, max) = field_domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok((value, true))
    } else if let Some(value) = name_value(field, bytes) {
        Ok((value, false))
    } else {
        Err(syntax_error(field, offset))
    }
}

fn parse_item(
    item: &[u8],
    field: Field,
    offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(syntax_error(field, offset));
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), Some(offset + slash + 1))
    } else {
        (item, None, None)
    };
    if base.is_empty() || step_bytes.is_some_and(|s| s.is_empty()) {
        return Err(syntax_error(field, offset));
    }

    let (start, end, numeric_start, numeric_end, end_offset) = if base == b"*" {
        let (min, max) = field_domain(field);
        (min, max, None, None, None)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();
        if hyphens.len() > 1 {
            return Err(syntax_error(field, offset));
        }
        if let Some(&hyphen) = hyphens.first() {
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if left.is_empty() || right.is_empty() || left == b"*" || right == b"*" {
                return Err(syntax_error(field, offset));
            }
            let left_offset = offset;
            let right_offset = offset + hyphen + 1;

            // Check lexical validity of both endpoints before numeric validation.
            let left_is_atom = left.iter().all(u8::is_ascii_digit)
                || name_value(field, left).is_some();
            let right_is_atom = right.iter().all(u8::is_ascii_digit)
                || name_value(field, right).is_some();
            if !left_is_atom || !right_is_atom {
                return Err(syntax_error(field, offset));
            }

            let (a, a_numeric) = parse_endpoint(left, field, left_offset)?;
            let (b, b_numeric) = parse_endpoint(right, field, right_offset)?;
            if a > b {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: right_offset,
                });
            }
            (a, b, Some(a_numeric), Some(b_numeric), Some(right_offset))
        } else {
            let (a, is_numeric) = parse_endpoint(base, field, offset)?;
            let (_, max) = field_domain(field);
            (a, max, Some(is_numeric), None, None)
        }
    };

    let step = if let Some(bytes) = step_bytes {
        let step_offset = step_offset.unwrap();
        if !bytes.iter().all(u8::is_ascii_digit) {
            return Err(syntax_error(field, offset));
        }
        let value = parse_decimal(bytes, field, step_offset)?;
        let (_, max) = field_domain(field);
        if value == 0 || value > max + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        Some(value)
    } else {
        None
    };

    // Numeric endpoints have already been validated; these bindings also make
    // the intended validation ordering explicit.
    let _ = (numeric_start, numeric_end, end_offset);

    let increment = step.unwrap_or(1);
    let stepped = step.is_some();
    if stepped || base == b"*" || base.contains(&b'-') {
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }
    } else {
        selected[start as usize] = true;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = field_domain(field);
    let mut selected = vec![false; (max + 1) as usize];
    let mut start = 0usize;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(&bytes[start..i], field, start, &mut selected)?;
            start = i + 1;
        }
    }
    parse_item(&bytes[start..], field, start, &mut selected)?;

    Ok((min..=max)
        .filter(|&value| selected[value as usize])
        .map(|value| value as u8)
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut allowed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for (index, field) in FIELDS.into_iter().enumerate() {
        allowed[index] = parse_field(pieces[index], field)?;
    }

    Ok(Cron {
        allowed,
        unrestricted_days: [pieces[2] == "*", pieces[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        }]
        .clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        if after == max_datetime() {
            return Ok(None);
        }

        let mut current = after;
        loop {
            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    let month_len = days_in_month(current.year, current.month);
                    if current.day < month_len {
                        current.day += 1;
                    } else {
                        current.day = 1;
                        if current.month < 12 {
                            current.month += 1;
                        } else {
                            current.month = 1;
                            current.year += 1;
                            if current.year > 2399 {
                                return Ok(None);
                            }
                        }
                    }
                }
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
            if current == max_datetime() {
                return Ok(None);
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.allowed[0].contains(&at.minute)
            || !self.allowed[1].contains(&at.hour)
            || !self.allowed[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.allowed[2].contains(&at.day);
        let dow = self.allowed[4].contains(&weekday(at.year, at.month, at.day));
        match self.unrestricted_days {
            [true, true] => true,
            [false, true] => dom,
            [true, false] => dow,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

use std::str;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_star: bool,
    day_of_week_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn parse_number(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(err(field, offset, 1));
    }
    let mut value: u32 = 0;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| err(field, offset, 2))?;
    }
    Ok(value)
}

fn parse_atom(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(err(field, offset, 0));
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        let n = parse_number(bytes, field, offset)?;
        let (lo, hi) = bounds(field);
        if n < lo as u32 || n > hi as u32 {
            return Err(err(field, offset, 3));
        }
        return Ok(n);
    }

    if bytes.iter().all(u8::is_ascii_alphabetic) {
        if let Some(table) = names(field) {
            if let Ok(s) = str::from_utf8(bytes) {
                for (i, name) in table.iter().enumerate() {
                    if s.eq_ignore_ascii_case(name) {
                        return Ok((i as u32) + if field == Field::Month { 1 } else { 0 });
                    }
                }
            }
        }
    }
    Err(err(field, offset, 0))
}

fn parse_field(
    text: &str,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = bounds(field);
    let width = (max - min + 1) as u32;
    let mut item_start = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }
        let ib = item.as_bytes();

        let slash_positions: Vec<usize> = ib
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();
        if slash_positions.len() > 1 {
            return Err(err(field, item_start, 0));
        }

        let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&ib[..slash], Some(&ib[slash + 1..]), item_start + slash + 1)
        } else {
            (ib, None, 0)
        };

        if step_bytes == Some(&[][..]) {
            return Err(err(field, item_start, 0));
        }

        let mut hyphens = Vec::new();
        for (i, &b) in base.iter().enumerate() {
            if b == b'-' {
                hyphens.push(i);
            }
        }
        if base.is_empty()
            || (base == b"*" && !hyphens.is_empty())
            || (base != b"*" && (hyphens.len() > 1 || hyphens.first() == Some(&0) || hyphens.last() == Some(&(base.len() - 1))))
        {
            return Err(err(field, item_start, 0));
        }
        if base != b"*" && hyphens.is_empty() {
            // Atom form.
            if !atom_lexically_valid(base, field) {
                return Err(err(field, item_start, 0));
            }
        } else if base != b"*" {
            let h = hyphens[0];
            if !atom_lexically_valid(&base[..h], field)
                || !atom_lexically_valid(&base[h + 1..], field)
            {
                return Err(err(field, item_start, 0));
            }
        }

        if let Some(step) = step_bytes {
            if !step.iter().all(u8::is_ascii_digit) {
                return Err(err(field, item_start, 0));
            }
        }

        let (start, end, end_offset) = if base == b"*" {
            (min as u32, max as u32, item_start)
        } else if hyphens.is_empty() {
            let n = parse_atom(base, field, item_start)?;
            if step_bytes.is_some() {
                (n, max as u32, item_start)
            } else {
                (n, n, item_start)
            }
        } else {
            let h = hyphens[0];
            let left = parse_atom(&base[..h], field, item_start)?;
            let right_offset = item_start + h + 1;
            let right = parse_atom(&base[h + 1..], field, right_offset)?;
            if left > right {
                return Err(err(field, right_offset, 4));
            }
            (left, right, right_offset)
        };

        let step = if let Some(step) = step_bytes {
            let n = parse_number(step, field, step_offset)?;
            if n == 0 || n > width {
                return Err(err(field, step_offset, 5));
            }
            n
        } else {
            1
        };

        let _ = end_offset;
        let mut value = start;
        while value <= end {
            selected[(value - min as u32) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }
    Ok(())
}

fn atom_lexically_valid(bytes: &[u8], field: Field) -> bool {
    if bytes.iter().all(u8::is_ascii_digit) {
        return true;
    }
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return false;
    }
    if let Some(table) = names(field) {
        if let Ok(s) = str::from_utf8(bytes) {
            return table.iter().any(|name| s.eq_ignore_ascii_case(name));
        }
    }
    false
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (lo, hi) = bounds(FIELDS[i]);
        vec![false; (hi - lo + 1) as usize]
    });

    for i in 0..5 {
        parse_field(parts[i], FIELDS[i], &mut selected[i])?;
    }

    Ok(Cron {
        selected,
        day_of_month_star: parts[2] == "*",
        day_of_week_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (lo, _) = bounds(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| if yes { Some(lo + i as u8) } else { None })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let minute_ok = self.selected[0][at.minute as usize];
        let hour_ok = self.selected[1][at.hour as usize];
        let month_ok = self.selected[3][at.month as usize - 1];
        if !(minute_ok && hour_ok && month_ok) {
            return false;
        }

        let dom = self.selected[2][at.day as usize - 1];
        let dow_value = weekday(at.year, at.month, at.day);
        let dow = self.selected[4][dow_value as usize];

        match (self.day_of_month_star, self.day_of_week_star) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        if after.year == 2399 && after.month == 12 && after.day == 31 && after.hour == 23
            && after.minute == 59
        {
            return Ok(None);
        }

        let mut y = after.year;
        let mut m = after.month;
        let mut d = after.day;
        let mut h = after.hour;
        let mut min = after.minute;

        loop {
            if min < 59 {
                min += 1;
            } else {
                min = 0;
                if h < 23 {
                    h += 1;
                } else {
                    h = 0;
                    if d < days_in_month(y, m) {
                        d += 1;
                    } else {
                        d = 1;
                        if m < 12 {
                            m += 1;
                        } else {
                            m = 1;
                            if y == 2399 {
                                return Ok(None);
                            }
                            y += 1;
                        }
                    }
                }
            }
            let candidate = CivilDateTime {
                year: y,
                month: m,
                day: d,
                hour: h,
                minute: min,
            };
            if self.matches_valid(candidate) {
                return Ok(Some(candidate));
            }
            if y == 2399 && m == 12 && d == 31 && h == 23 && min == 59 {
                return Ok(None);
            }
        }
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days: i32 = 0;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as i32;
    }
    days += day as i32 - 1;
    ((6 + days) % 7) as u8
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if !(1..=days_in_month(at.year, at.month)).contains(&at.day) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}
