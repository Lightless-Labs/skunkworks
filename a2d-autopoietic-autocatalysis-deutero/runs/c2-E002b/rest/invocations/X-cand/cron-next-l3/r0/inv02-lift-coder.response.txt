#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn width(field: Field) -> u32 {
    let (lo, hi) = domain(field);
    hi - lo + 1
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(invalid_syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if step {
        if value == 0 || value > width(field) {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (lo, hi) = domain(field);
        if value < lo || value > hi {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }

    Ok(value)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    if bytes.len() != 3 {
        return None;
    }
    let upper = [
        bytes[0].to_ascii_uppercase(),
        bytes[1].to_ascii_uppercase(),
        bytes[2].to_ascii_uppercase(),
    ];
    names.iter().position(|name| name.as_bytes() == upper).map(|i| {
        if field == Field::Month {
            (i + 1) as u32
        } else {
            i as u32
        }
    })
}

fn parse_endpoint(
    bytes: &[u8],
    field: Field,
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        return Ok((parse_number(bytes, field, offset, false)?, true));
    }
    if let Some(value) = name_value(field, bytes) {
        return Ok((value, false));
    }
    Err(invalid_syntax(field, offset))
}

fn parse_item(
    item: &[u8],
    field: Field,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }
    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, item_offset)
    };
    if base.is_empty() || step_bytes.is_some_and(|s| s.is_empty()) {
        return Err(invalid_syntax(field, item_offset));
    }

    let star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    if star && !hyphens.is_empty() {
        return Err(invalid_syntax(field, item_offset));
    }
    if !star && hyphens.len() > 1 {
        return Err(invalid_syntax(field, item_offset));
    }

    let (start_bytes, end_bytes, start_offset, end_offset) = if star {
        (None, None, item_offset, item_offset)
    } else if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(invalid_syntax(field, item_offset));
        }
        (
            Some(&base[..hyphen]),
            Some(&base[hyphen + 1..]),
            item_offset,
            item_offset + hyphen + 1,
        )
    } else {
        (Some(base), None, item_offset, item_offset)
    };

    let step = if let Some(step_bytes) = step_bytes {
        if !step_bytes.iter().all(u8::is_ascii_digit) {
            return Err(invalid_syntax(field, item_offset));
        }
        Some(parse_number(step_bytes, field, step_offset, true)?)
    } else {
        None
    };

    let (lo, hi) = domain(field);
    let (start, end) = match (start_bytes, end_bytes) {
        (None, None) => (lo, hi),
        (Some(sb), Some(eb)) => {
            let (start, start_numeric) = parse_endpoint(sb, field, start_offset)?;
            let (end, end_numeric) = parse_endpoint(eb, field, end_offset)?;
            if start_numeric && start > hi {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start_offset,
                });
            }
            if end_numeric && end > hi {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_offset,
                });
            }
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (start, end)
        }
        (Some(sb), None) => {
            let (start, numeric) = parse_endpoint(sb, field, start_offset)?;
            if numeric && start > hi {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start_offset,
                });
            }
            if step.is_some() {
                (start, hi)
            } else {
                (start, start)
            }
        }
        _ => return Err(invalid_syntax(field, item_offset)),
    };

    let stride = step.unwrap_or(1);
    let mut value = start;
    loop {
        selected[(value - lo) as usize] = true;
        if value > end.saturating_sub(stride) {
            break;
        }
        value += stride;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, max) = domain(field);
    let mut selected = vec![false; (max - domain(field).0 + 1) as usize];
    let mut item_start = 0;
    for (i, part) in bytes
        .split(|&b| b == b',')
        .enumerate()
    {
        let _ = i;
        parse_item(part, field, item_start, &mut selected)?;
        item_start += part.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut parsed: Vec<Vec<bool>> = Vec::with_capacity(5);
    for (i, part) in parts.iter().enumerate() {
        parsed.push(parse_field(part, FIELDS[i])?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().unwrap(),
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (lo, _) = domain(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some((i as u32 + lo) as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let minute = self.selected[0][at.minute as usize];
        let hour = self.selected[1][at.hour as usize];
        let month = self.selected[3][at.month as usize - 1];
        if !(minute && hour && month) {
            return false;
        }

        let dom = self.selected[2][at.day as usize - 1];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                let last = at.day == month_length(at.year, at.month);
                if last {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            advance_minute(&mut current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) {
    if at.minute < 59 {
        at.minute += 1;
        return;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return;
    }
    at.hour = 0;
    if at.day < month_length(at.year, at.month) {
        at.day += 1;
        return;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return;
    }
    at.month = 1;
    at.year += 1;
}
