#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    let separator = match separators.as_slice() {
        [] => return Err(EmailError::MissingAt),
        [separator] => *separator,
        [_, second, ..] => return Err(EmailError::MultipleAt { at: *second }),
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let allowed_special = b"!#$%&'*+-/=?^_`{|}~";
    let mut previous_was_dot = false;
    for (index, &byte) in local.iter().enumerate() {
        if byte == b'.' {
            if index == 0 || index + 1 == local.len() || previous_was_dot {
                return Err(EmailError::LocalDot { at: index });
            }
            previous_was_dot = true;
        } else {
            previous_was_dot = false;
            if !byte.is_ascii_alphanumeric() && !allowed_special.contains(&byte) {
                return Err(EmailError::LocalChar { at: index });
            }
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_end = if has_root_dot {
        input.len() - 1
    } else {
        input.len()
    };
    let domain = &bytes[domain_start..domain_end];

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut start = 0;
    for index in 0..=domain.len() {
        if index == domain.len() || domain[index] == b'.' {
            let label = &domain[start..index];
            let absolute_start = domain_start + start;
            if label.is_empty() {
                let at = if index == domain.len() && has_root_dot {
                    input.len() - 1
                } else {
                    absolute_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }

            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }

            for (offset, &byte) in label.iter().enumerate() {
                if !byte.is_ascii_alphanumeric() && byte != b'-' {
                    return Err(EmailError::DomainChar {
                        at: absolute_start + offset,
                    });
                }
            }

            label_starts.push((start, index));
            start = index + 1;
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = label_starts[label_starts.len() - 1];
    if domain[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let normalized_domain = domain
        .iter()
        .map(|byte| byte.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: String::from_utf8(local.to_vec()).expect("input was checked as ASCII"),
        domain: normalized_domain,
        quoted: false,
    })
}
