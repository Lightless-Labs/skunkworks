fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'

    byte.is_ascii_alphanumeric() || byte == b'_'

fn parse_specifier<'a>(raw: &'a str, start: usize) -> Result<Specifier<'a>, Error> {
    let bytes = raw.as_bytes();
    let modifier_at = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(bytes.len());

    if name_end == 0 {
        return Err(Error::InvalidName { pos: start });
    }

    if !is_name_start(bytes[0]) {
        return Err(Error::InvalidName { pos: start });
    }
    for (i, &byte) in bytes[..name_end].iter().enumerate().skip(1) {
        if !is_name_continue(byte) {
            return Err(Error::InvalidName { pos: start + i });
        }
    }

    let modifier = if let Some(i) = modifier_at {
        let suffix = &bytes[i..];
        match suffix {
            [b'*'] => Modifier::Explode,
            [b':', first @ b'1'..=b'9', rest @ ..]
                if (0..=3).contains(&rest.len())
                    && rest.iter().all(u8::is_ascii_digit) =>
            {
                let mut n = (first - b'0') as usize;
                for &digit in rest {
                    n = n * 10 + (digit - b'0') as usize;
                }
                if (1..=9999).contains(&n) {
                    Modifier::Prefix(n)
                } else {
                    return Err(Error::InvalidModifier { pos: start + i });
                }
            }
            _ => return Err(Error::InvalidModifier { pos: start + i }),
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name: &raw[..name_end],
        name_pos: start,
        start,
        modifier,
    })
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let content_start = open + 1;
    let content = &template[content_start..close];
    let mut operator = Operator::Simple;
    let mut body_start = content_start;

    if let Some(&first) = content.as_bytes().first() {
        let recognized = match first {
            b'+' => Some(Operator::Reserved),
            b'#' => Some(Operator::Fragment),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Query),
            b'&' => Some(Operator::Continuation),
            _ => None,
        };
        if let Some(op) = recognized {
            operator = op;
            body_start += 1;
        } else if !is_name_start(first) {
            return Err(Error::InvalidName { pos: content_start });
        }
    }

    if body_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let body = &template[body_start..close];
    let mut specs = Vec::new();
    let mut offset = body_start;
    let mut seen = Vec::<&str>::new();

    for part in body.split(',') {
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: offset });
        }
        let spec = parse_specifier(part, offset)?;
        if seen.iter().any(|&name| name == spec.name) {
            return Err(Error::DuplicateVariable { pos: spec.start });
        }
        seen.push(spec.name);
        specs.push(spec);
        offset += part.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$'
                    | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn encode_prefix(value: &str, count: usize, reserved: bool) -> String {
    let prefix: String = value.chars().take(count).collect();
    encode(&prefix, reserved)
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if matches!(operator, Operator::Semicolon) && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn add_value_fields(
    spec: &Specifier<'_>,
    operator: Operator,
    value: &Value,
    fields: &mut Vec<String>,
) -> Result<(), Error> {
    let reserved = operator.reserved();

    if let Modifier::Prefix(_) = spec.modifier {
        if !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.start });
        }
    }

    match value {
        Value::Scalar(text) => {
            let encoded = match spec.modifier {
                Modifier::Prefix(count) => encode_prefix(text, count, reserved),
                Modifier::None | Modifier::Explode => encode(text, reserved),
            };
            fields.push(if operator.named() {
                named_field(spec.name, encoded, operator)
            } else {
                encoded
            });
        }
        Value::List(items) => {
            let exploded = matches!(spec.modifier, Modifier::Explode);
            if exploded {
                for item in items {
                    let encoded = encode(item, reserved);
                    fields.push(if operator.named() {
                        named_field(spec.name, encoded, operator)
                    } else {
                        encoded
                    });
                }
            } else if !items.is_empty() {
                let joined = items
                    .iter()
                    .map(|item| encode(item, reserved))
                    .collect::<Vec<_>>()
                    .join(",");
                fields.push(if operator.named() {
                    named_field(spec.name, joined, operator)
                } else {
                    joined
                });
            }
        }
        Value::Map(entries) => {
            for i in 0..entries.len() {
                if entries[..i].iter().any(|(key, _)| key == &entries[i].0) {
                    return Err(Error::DuplicateKey { pos: spec.start });
                }
            }

            let exploded = matches!(spec.modifier, Modifier::Explode);
            if exploded {
                for (key, value) in entries {
                    let encoded_key = encode(key, reserved);
                    let encoded_value = encode(value, reserved);
                    let field = if matches!(operator, Operator::Semicolon)
                        && encoded_value.is_empty()
                    {
                        encoded_key
                    } else {
                        format!("{encoded_key}={encoded_value}")
                    };
                    fields.push(field);
                }
            } else if !entries.is_empty() {
                let mut components = Vec::with_capacity(entries.len() * 2);
                for (key, value) in entries {
                    components.push(encode(key, reserved));
                    components.push(encode(value, reserved));
                }
                let joined = components.join(",");
                fields.push(if operator.named() {
                    named_field(spec.name, joined, operator)
                } else {
                    joined
                });
            }
        }
    }

    Ok(())
}

fn expand_expression(
    expression: &Expression<'_>,
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let mut fields = Vec::new();

    for spec in &expression.specs {
        if let Some(value) = selected_value(spec.name, vars) {
            add_value_fields(spec, expression.operator, value, &mut fields)?;
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut result = String::new();
    if let Some(prefix) = expression.operator.prefix() {
        result.push(prefix);
    }
    let separator = expression.operator.separator();
    for (index, field) in fields.iter().enumerate() {
        if index > 0 {
            result.push(separator);
        }
        result.push_str(field);
    }
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;

                if let Some(relative_nested) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_nested,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (open, close, expression) in &expressions {
        output.push_str(&template[cursor..*open]);
        output.push_str(&expand_expression(expression, vars)?);
        cursor = close + 1;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}

            Operator::Ampersand
        }
        Some(b'=') | Some(b',') | Some(b'!') | Some(b'@') | Some(b'|') => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => Operator::Simple,
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specifiers = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = content_start;

    loop {
        let end = bytes[start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| start + offset)
            .unwrap_or(close);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let modifier_start = (start..end)
            .find(|&i| bytes[i] == b'*' || bytes[i] == b':')
            .unwrap_or(end);

        if modifier_start == start {
            return Err(Error::InvalidName { pos: start });
        }

        let name_bytes = &bytes[start..modifier_start];
        let mut valid_name = true;
        let mut bad_name_pos = start;
        for (i, &b) in name_bytes.iter().enumerate() {
            if (i == 0 && !is_name_start(b)) || (i > 0 && !is_name_continue(b)) {
                valid_name = false;
                bad_name_pos = start + i;
                break;
            }
        }
        if !valid_name {
            return Err(Error::InvalidName { pos: bad_name_pos });
        }

        let name = template[start..modifier_start].to_owned();
        let modifier = if modifier_start == end {
            Modifier::None
        } else {
            let suffix = &bytes[modifier_start..end];
            let parsed = if suffix == b"*" {
                Some(Modifier::Explode)
            } else if suffix.len() >= 2
                && suffix[0] == b':'
                && suffix.len() <= 5
                && suffix[1..].iter().all(u8::is_ascii_digit)
                && suffix[1] != b'0'
            {
                let mut value = 0usize;
                let mut okay = true;
                for &digit in &suffix[1..] {
                    value = value * 10 + (digit - b'0') as usize;
                    if value > 9999 {
                        okay = false;
                    }
                }
                if okay {
                    Some(Modifier::Prefix(value))
                } else {
                    None
                }
            } else {
                None
            };
            parsed.ok_or(Error::InvalidModifier { pos: modifier_start })?
        };

        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(name.clone());
        specifiers.push(Specifier {
            name,
            pos: start,
            modifier,
        });

        if end == close {
            break;
        }
        start = end + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];

        if b == b'%' && i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
            || (reserved && is_reserved(b))
        {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn encoded_value(value: &str, modifier: Modifier, reserved: bool) -> String {
    let selected = match modifier {
        Modifier::Prefix(n) => value.chars().take(n).collect::<String>(),
        _ => value.to_owned(),
    };
    encode(&selected, reserved)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expression.operator;
    let reserved = op.reserved();
    let named = op.named_fields();
    let mut fields = Vec::<String>::new();

    for spec in &expression.specifiers {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        match (value, spec.modifier) {
            (Value::Scalar(s), modifier) => {
                let encoded = encoded_value(s, modifier, reserved);
                fields.push(if named {
                    named_field(&spec.name, encoded, op)
                } else {
                    encoded
                });
            }
            (Value::List(_), Modifier::Prefix(_)) | (Value::Map(_), Modifier::Prefix(_)) => {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
            (Value::List(items), modifier) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if named {
                            named_field(&spec.name, encoded, op)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if named {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
            (Value::Map(entries), modifier) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|existing| *existing == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                if matches!(modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let field = if op == Operator::Semicolon && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if named {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = op.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&op.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;

                if let Some(relative_open) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_open,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                output.push_str(&expand_expression(&expression, vars)?);
                cursor = close + 1;
            }
            _ => {
                let start = cursor;
                while cursor < bytes.len() && bytes[cursor] != b'{' && bytes[cursor] != b'}' {
                    cursor += 1;
                }
                output.push_str(&template[start..cursor]);
            }
        }
    }

    Ok(output)
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;

    let operator = if content_start < close {
        match bytes[content_start] {
            b'+' => {
                content_start += 1;
                Operator::Plus
            }
            b'#' => {
                content_start += 1;
                Operator::Hash
            }
            b'.' => {
                content_start += 1;
                Operator::Dot
            }
            b'/' => {
                content_start += 1;
                Operator::Slash
            }
            b';' => {
                content_start += 1;
                Operator::Semicolon
            }
            b'?' => {
                content_start += 1;
                Operator::Query
            }
            b'&' => {
                content_start += 1;
                Operator::Ampersand
            }
            b'!' | b'@' | b'=' | b',' | b':' | b'*' | b'%' | b'[' | b']'
            | b'(' | b')' | b' ' | b'\t' | b'\r' | b'\n' => {
                return Err(Error::UnsupportedOperator { pos: content_start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut used_names = Vec::<String>::new();
    let mut spec_start = content_start;

    loop {
        let mut spec_end = spec_start;
        while spec_end < close && bytes[spec_end] != b',' {
            spec_end += 1;
        }

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let mut modifier_at = spec_end;
        for i in spec_start..spec_end {
            if bytes[i] == b'*' || bytes[i] == b':' {
                modifier_at = i;
                break;
            }
        }

        if modifier_at == spec_start {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..modifier_at];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(offset) = name_bytes.iter().enumerate().skip(1).find_map(|(i, b)| {
            if is_name_continue(*b) { None } else { Some(i) }
        }) {
            return Err(Error::InvalidName { pos: spec_start + offset });
        }

        let modifier = if modifier_at == spec_end {
            Modifier::None
        } else {
            let suffix = &bytes[modifier_at..spec_end];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.first() == Some(&b':') {
                let digits = &suffix[1..];
                let valid = (1..=4).contains(&digits.len())
                    && digits[0].is_ascii_digit()
                    && digits[0] != b'0'
                    && digits.iter().all(u8::is_ascii_digit);
                if !valid {
                    return Err(Error::InvalidModifier { pos: modifier_at });
                }
                let mut value = 0usize;
                for digit in digits {
                    value = value * 10 + (digit - b'0') as usize;
                }
                if !(1..=9999).contains(&value) {
                    return Err(Error::InvalidModifier { pos: modifier_at });
                }
                Modifier::Prefix(value)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
        };

        let name = template[spec_start..modifier_at].to_string();
        if used_names.iter().any(|n| n == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        used_names.push(name.clone());
        specs.push(Specifier {
            name,
            pos: spec_start,
            modifier,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
        if spec_start == close {
            return Err(Error::EmptyVariable { pos: close });
        }
    }

    Ok(Expression { operator, specs })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                if let Some(relative_open) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_open,
                    });
                }
                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                const HEX: &[u8; 16] = b"0123456789ABCDEF";
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

    if omit_equals {
        name.to_string()
    } else {

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::<String>::new();
    let reserved = matches!(expression.operator, Operator::Plus | Operator::Hash);

    for spec in &expression.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) {
            match value {
                Value::Scalar(_) => {}
                Value::List(_) | Value::Map(_) => {
                    return Err(Error::TypeMismatch { pos: spec.pos });
                }
            }
        }

        match value {
            Value::Scalar(scalar) => {
                let sliced;
                let source = match spec.modifier {
                    Modifier::Prefix(count) => {
                        sliced = scalar.chars().take(count).collect::<String>();
                        sliced.as_str()
                    }
                    _ => scalar.as_str(),
                };
                let encoded = encode(source, reserved);
                fields.push(match expression.operator {
                    Operator::Semicolon | Operator::Query | Operator::Ampersand => {
                        encode_named(&spec.name, encoded, expression.operator)
                    }
                    _ => encoded,
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(match expression.operator {
                            Operator::Semicolon | Operator::Query | Operator::Ampersand => {
                                encode_named(&spec.name, encoded, expression.operator)
                            }
                            _ => encoded,
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(match expression.operator {
                        Operator::Semicolon | Operator::Query | Operator::Ampersand => {
                            encode_named(&spec.name, joined, expression.operator)
                        }
                        _ => joined,
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|existing| *existing == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }
                if entries.is_empty() {
                    continue;
                }

                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        if expression.operator == Operator::Semicolon && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(match expression.operator {
                        Operator::Semicolon | Operator::Query | Operator::Ampersand => {
                            encode_named(&spec.name, joined, expression.operator)
                        }
                        _ => joined,
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match expression.operator {
        Operator::Simple | Operator::Plus => (",", ""),
        Operator::Hash => (",", "#"),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("&", "?"),
        Operator::Ampersand => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (open, close, expression) in expressions {
        output.push_str(&template[cursor..open]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = close + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse_expression(content: &str, base: usize, close: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (op, offset) = if let Some(&first) = bytes.first() {
        let recognized = match first {
            b'+' => Some(Operator::Reserved),
            b'#' => Some(Operator::Fragment),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Query),
            b'&' => Some(Operator::Continue),
            _ => None,
        };
        if let Some(op) = recognized {
            (op, 1)
        } else {
            (Operator::Simple, 0)
        }
    } else {
        (Operator::Simple, 0)
    };

    if offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = offset;

    loop {
        let end = content[start..]
            .find(',')
            .map(|relative| start + relative)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::EmptyVariable { pos: base + start });
        }

        let spec = &content[start..end];
        let spec_bytes = spec.as_bytes();
        let modifier_at = spec_bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_at.unwrap_or(spec_bytes.len());

        if name_end == 0 {
            return Err(Error::InvalidName { pos: base + start });
        }

        for (i, &byte) in spec_bytes[..name_end].iter().enumerate() {
            let valid = if i == 0 {
                is_name_start(byte)
            } else {
                is_name_continue(byte)
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: base + start + i,
                });
            }
        }

        let modifier = if let Some(mod_index) = modifier_at {
            let suffix = &spec_bytes[mod_index..];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.first() == Some(&b':') {
                let digits = &suffix[1..];
                if digits.is_empty()
                    || digits.len() > 4
                    || digits[0] == b'0'
                    || !digits.iter().all(u8::is_ascii_digit)
                {
                    return Err(Error::InvalidModifier {
                        pos: base + start + mod_index,
                    });
                }
                let mut number = 0usize;
                for &digit in digits {
                    number = number * 10 + usize::from(digit - b'0');
                }
                if !(1..=9999).contains(&number) {
                    return Err(Error::InvalidModifier {
                        pos: base + start + mod_index,
                    });
                }
                Modifier::Prefix(number)
            } else {
                return Err(Error::InvalidModifier {
                    pos: base + start + mod_index,
                });
            }
        } else {
            Modifier::None
        };

        let name = spec[..name_end].to_owned();
        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: base + start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            pos: base + start,
            modifier,
        });

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::EmptyVariable { pos: base + start });
        }
    }

    Ok(Expression { op, specs })
}

fn parse_template(template: &str) -> Result<Vec<Part>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut literal_start = 0;
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                if literal_start < index {
                    parts.push(Part::Literal(template[literal_start..index].to_owned()));
                }
                let open = index;
                let close = match bytes[index + 1..].iter().position(|&byte| byte == b'}') {
                    Some(relative) => index + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(relative) = bytes[index + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + relative,
                    });
                }

                let content_start = index + 1;
                let content = &template[content_start..close];
                let expression = parse_expression(content, content_start, close)?;
                parts.push(Part::Expression(expression));
                index = close + 1;
                literal_start = index;
            }
            _ => index += 1,
        }
    }

    if literal_start < bytes.len() {
        parts.push(Part::Literal(template[literal_start..].to_owned()));
    }
    Ok(parts)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%'
            && index + 2 < bytes.len()
            && let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
        {
            result.push('%');
            result.push(hex_digit(high));
            result.push(hex_digit(low));
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(hex_digit(byte >> 4));
            result.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    result
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn scalar_field(name: &str, value: String, op: Operator) -> String {
    match op {
        Operator::Semicolon => {
            if value.is_empty() {
                name.to_owned()
            } else {
                format!("{name}={value}")
            }
        }
        Operator::Query | Operator::Continue => format!("{name}={value}"),
        _ => value,
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(expression.op, Operator::Reserved | Operator::Fragment);
    let explode_separator = match expression.op {
        Operator::Dot => ".",
        Operator::Slash => "/",
        Operator::Semicolon => ";",
        Operator::Query | Operator::Continue => "&",
        _ => ",",
    };

    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(text) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(count) => text.chars().take(count).collect::<String>(),
                    _ => text.clone(),
                };
                fields.push(scalar_field(&spec.name, encode(&selected, reserved), expression.op));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if spec.modifier == Modifier::Explode {
                    for item in items {
                        fields.push(scalar_field(&spec.name, encode(item, reserved), expression.op));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(scalar_field(&spec.name, joined, expression.op));
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|existing| *existing == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                if spec.modifier == Modifier::Explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        if expression.op == Operator::Semicolon && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    fields.push(scalar_field(
                        &spec.name,
                        components.join(","),
                        expression.op,
                    ));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expression.op {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continue => ("&", "&"),
    };

    let _ = explode_separator;
    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let parts = parse_template(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(&text),
            Part::Expression(expression) => {
                output.push_str(&expand_expression(&expression, vars)?);
            }
        }
    }

    Ok(output)
}
