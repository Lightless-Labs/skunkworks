use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }
        if pos == start {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut n = 0u64;
        for &b in &bytes[start..pos] {
            n = n
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        core[i] = n;

        if i < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let (pre, build);
    if pos == bytes.len() {
        pre = Vec::new();
        build = Vec::new();
    } else if bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        if let Some(p) = plus {
            build = parse_identifiers(s, p + 1, bytes.len(), false)?;
        } else {
            build = Vec::new();
        }
    } else if bytes[pos] == b'+' {
        pre = Vec::new();
        build = parse_identifiers(s, pos + 1, bytes.len(), false)?;
    } else {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut part = start;
    loop {
        let mut i = part;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == part {
            return Err(VersionError::EmptyIdentifier { at: part });
        }
        let numeric = bytes[part..i].iter().all(|&b| is_digit(b));
        if prerelease && numeric && i - part > 1 && bytes[part] == b'0' {
            return Err(VersionError::LeadingZero { at: part });
        }
        out.push(s[part..i].to_string());
        if i == end {
            break;
        }
        part = i + 1;
    }
    Ok(out)
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_identifier(x);
        let yn = numeric_identifier(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a)
        .cmp(&core(b))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum CmpOp {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CmpOp,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    comps: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone)]
struct Tok {
    text: String,
    start: usize,
}

fn tokens(s: &str, base: usize) -> Vec<Tok> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0usize;
    while i < b.len() {
        while i < b.len() && ws(b[i]) {
            i += 1;
        }
        if i == b.len() {
            break;
        }
        let st = i;
        while i < b.len() && !ws(b[i]) {
            i += 1;
        }
        out.push(Tok {
            text: s[st..i].to_string(),
            start: base + st,
        });
    }
    out
}

fn version_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn parse_full_comparator(tok: &Tok) -> Result<Comparator, RangeError> {
    let t = tok.text.as_str();
    let (op, n) = if t.starts_with(">=") {
        (CmpOp::Ge, 2)
    } else if t.starts_with("<=") {
        (CmpOp::Le, 2)
    } else if t.starts_with('>') {
        (CmpOp::Gt, 1)
    } else if t.starts_with('<') {
        (CmpOp::Lt, 1)
    } else if t.starts_with('=') {
        (CmpOp::Eq, 1)
    } else {
        (CmpOp::Eq, 0)
    };
    let operand = &t[n..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let v = version_at(operand, tok.start + n)?;
    let anchor = !v.pre.is_empty();
    Ok(Comparator { op, version: v, anchor })
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
    numeric_count: usize,
}

fn partial_or_full(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        let v = version_at(s, at)?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            numeric_count: 3,
            full: Some(v),
        });
    }

    let raw: Vec<&str> = s.split('.').collect();
    let full_candidate = raw.len() >= 3 && !raw.iter().any(|x| matches!(*x, "*" | "x" | "X"));
    if full_candidate {
        let v = version_at(s, at)?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            numeric_count: 3,
            full: Some(v),
        });
    }

    if raw.is_empty() || raw.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for field in raw {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
            fields.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(is_digit)
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            fields.push(Some(n));
        }
    }
    let numeric_count = fields.iter().take_while(|x| x.is_some()).count();
    Ok(Partial {
        fields,
        full: None,
        numeric_count,
    })
}

fn padded(p: &Partial) -> Version {
    let mut vals = [0u64; 3];
    for (i, f) in p.fields.iter().enumerate().take(3) {
        if let Some(n) = f {
            vals[i] = *n;
        }
    }
    Version {
        major: vals[0],
        minor: vals[1],
        patch: vals[2],
        pre: p.full.as_ref().map(|v| v.pre.clone()).unwrap_or_default(),
        build: p.full.as_ref().map(|v| v.build.clone()).unwrap_or_default(),
    }
}

fn bound_version(core: (u64, u64, u64)) -> Version {
    Version {
        major: core.0,
        minor: core.1,
        patch: core.2,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn increment_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut c = [v.major, v.minor, v.patch];
    c[field] = c[field]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at })?;
    for x in c.iter_mut().skip(field + 1) {
        *x = 0;
    }
    Ok(Version {
        major: c[0],
        minor: c[1],
        patch: c[2],
        pre: vec!["0".to_string()],
        build: Vec::new(),
    })
}

fn add_cmp(arm: &mut Arm, op: CmpOp, version: Version, anchor: bool) {
    if anchor {
        arm.anchors.push(core(&version));
    }
    arm.comps.push(Comparator {
        op,
        version,
        anchor,
    });
}

fn parse_partial_token(
    tok: &Tok,
    mode: u8,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let (body, body_at) = if mode == 1 || mode == 2 {
        (&tok.text[1..], tok.start + 1)
    } else {
        (tok.text.as_str(), tok.start)
    };
    let p = partial_or_full(body, body_at)?;
    let k = p.numeric_count;
    if mode != 0 && k == 0 {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let lower = padded(&p);
    match mode {
        0 => {
            if let Some(v) = p.full {
                let anchor = !v.pre.is_empty();
                add_cmp(arm, CmpOp::Eq, v, anchor);
            } else if k == 1 || k == 2 {
                add_cmp(arm, CmpOp::Ge, lower.clone(), false);
                let upper = increment_upper(&lower, k - 1, tok.start)?;
                add_cmp(arm, CmpOp::Lt, upper, false);
            }
        }
        1 => {
            add_cmp(
                arm,
                CmpOp::Ge,
                lower.clone(),
                p.full.as_ref().is_some_and(|v| !v.pre.is_empty()),
            );
            let field = if lower.major > 0 || k == 1 {
                0
            } else if lower.minor > 0 || k == 2 {
                1
            } else {
                2
            };
            let upper = increment_upper(&lower, field, tok.start)?;
            add_cmp(arm, CmpOp::Lt, upper, false);
        }
        2 => {
            add_cmp(
                arm,
                CmpOp::Ge,
                lower.clone(),
                p.full.as_ref().is_some_and(|v| !v.pre.is_empty()),
            );
            let field = if k == 1 { 0 } else { 1 };
            let upper = increment_upper(&lower, field, tok.start)?;
            add_cmp(arm, CmpOp::Lt, upper, false);
        }
        _ => unreachable!(),
    }
    Ok(())
}

fn parse_hyphen_arm(ts: &[Tok], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = ts.iter().position(|t| t.text == "-").unwrap();
    if ts.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax { at: ts[minus].start });
    }
    let lo = partial_or_full(&ts[0].text, ts[0].start)?;
    let hi = partial_or_full(&ts[2].text, ts[2].start)?;
    if lo.numeric_count == 0 {
        return Err(RangeError::Syntax { at: ts[0].start });
    }
    if hi.numeric_count == 0 {
        return Err(RangeError::Syntax { at: ts[2].start });
    }
    let lower = padded(&lo);
    add_cmp(
        arm,
        CmpOp::Ge,
        lower,
        lo.full.as_ref().is_some_and(|v| !v.pre.is_empty()),
    );
    if let Some(v) = hi.full {
        let anchor = !v.pre.is_empty();
        add_cmp(arm, CmpOp::Le, v, anchor);
    } else {
        let upper = padded(&hi);
        if hi.numeric_count == 1 || hi.numeric_count == 2 {
            let up = increment_upper(&upper, hi.numeric_count - 1, ts[2].start)?;
            add_cmp(arm, CmpOp::Lt, up, false);
        } else {
            add_cmp(arm, CmpOp::Le, upper, false);
        }
    }
    Ok(())
}

fn parse_arm(ts: &[Tok]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comps: Vec::new(),
        anchors: Vec::new(),
    };
    if let Some(minus) = ts.iter().position(|t| t.text == "-") {
        let _ = minus;
        parse_hyphen_arm(ts, &mut arm)?;
        return Ok(arm);
    }

    for t in ts {
        let first = t.text.as_bytes().first().copied().unwrap_or(0);
        if first == b'<' || first == b'>' || first == b'=' {
            arm.comps.push(parse_full_comparator(t)?);
        } else if first == b'^' {
            parse_partial_token(t, 1, &mut arm)?;
        } else if first == b'~' {
            parse_partial_token(t, 2, &mut arm)?;
        } else if first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X') {
            parse_partial_token(t, 0, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: t.start });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;
    let mut saw_content = false;

    while i < bytes.len() {
        if bytes[i..].starts_with(b"||")
            && (i == 0 || ws(bytes[i - 1]))
            && (i + 2 == bytes.len() || ws(bytes[i + 2]))
        {
            let segment = &s[segment_start..i];
            let ts = tokens(segment, segment_start);
            if ts.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(&ts)?);
            saw_content = true;
            i += 2;
            segment_start = i;
        } else {
            i += 1;
        }
    }

    let segment = &s[segment_start..];
    let ts = tokens(segment, segment_start);
    if ts.is_empty() {
        if !saw_content {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&ts)?);
    Ok(Range { arms })
}

fn comparator_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        CmpOp::Eq => ord == Ordering::Equal,
        CmpOp::Lt => ord == Ordering::Less,
        CmpOp::Le => ord != Ordering::Greater,
        CmpOp::Gt => ord == Ordering::Greater,
        CmpOp::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comps.iter().all(|c| comparator_matches(v, c)) {
        return false;
    }
    if include_prerelease || v.pre.is_empty() {
        return true;
    }
    arm.anchors.iter().any(|&a| a == core(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for i in 0..candidates.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

fn add_core(x: (u64, u64, u64), field: usize) -> Option<(u64, u64, u64)> {
    let mut a = [x.0, x.1, x.2];
    a[field] = a[field].checked_add(1)?;
    for v in a.iter_mut().skip(field + 1) {
        *v = 0;
    }
    Some((a[0], a[1], a[2]))
}

fn next_core(x: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if let Some(p) = x.2.checked_add(1) {
        Some((x.0, x.1, p))
    } else if let Some(m) = x.1.checked_add(1) {
        Some((x.0, m, 0))
    } else {
        x.0.checked_add(1).map(|a| (a, 0, 0))
    }
}

fn arm_has_stable_interval(arm: &Arm) -> bool {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.comps {
        match c.op {
            CmpOp::Eq => {
                lower = Some((c.version.clone(), true));
                upper = Some((c.version.clone(), true));
            }
            CmpOp::Gt | CmpOp::Ge => {
                let inclusive = matches!(c.op, CmpOp::Ge);
                if lower.as_ref().is_none_or(|(v, inc)| {
                    compare_versions(&c.version, v) == Ordering::Greater
                        || (compare_versions(&c.version, v) == Ordering::Equal && !inclusive && *inc)
                }) {
                    lower = Some((c.version.clone(), inclusive));
                }
            }
            CmpOp::Lt | CmpOp::Le => {
                let inclusive = matches!(c.op, CmpOp::Le);
                if upper.as_ref().is_none_or(|(v, inc)| {
                    compare_versions(&c.version, v) == Ordering::Less
                        || (compare_versions(&c.version, v) == Ordering::Equal && !inclusive && *inc)
                }) {
                    upper = Some((c.version.clone(), inclusive));
                }
            }
        }
    }
    match (lower, upper) {
        (Some((l, li)), Some((u, ui))) => {
            let ord = compare_versions(&l, &u);
            ord == Ordering::Less || (ord == Ordering::Equal && li && ui && l.pre.is_empty())
        }
        _ => true,
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut both = aa.clone();
            both.comps.extend(bb.comps.clone());
            both.anchors.extend(bb.anchors.clone());
            if !arm_has_stable_interval(&both) {
                continue;
            }
            if include_prerelease {
                return true;
            }
            if aa.anchors.iter().any(|x| bb.anchors.contains(x)) {
                let c = bound_version((0, 0, 0));
                if arm_matches(&c, &both, false) {
                    return true;
                }
                for anchor in aa.anchors.iter().filter(|x| bb.anchors.contains(x)) {
                    let v = Version {
                        major: anchor.0,
                        minor: anchor.1,
                        patch: anchor.2,
                        pre: vec!["0".to_string()],
                        build: Vec::new(),
                    };
                    if arm_matches(&v, &both, false) {
                        return true;
                    }
                }
            }
            // Stable versions are dense enough to check boundary tuples and their successors.
            let mut candidates = vec![bound_version((0, 0, 0))];
            for c in both.comps.iter() {
                let x = core(&c.version);
                candidates.push(bound_version(x));
                candidates.push(Version {
                    major: x.0,
                    minor: x.1,
                    patch: x.2,
                    pre: vec!["0".to_string()],
                    build: Vec::new(),
                });
                if let Some(n) = next_core(x) {
                    candidates.push(bound_version(n));
                }
            }
            if candidates.iter().any(|v| {
                v.pre.is_empty() && arm_matches(v, &both, false)
            }) {
                return true;
            }
        }
    }
    false
}

fn stripped_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn canonical_arm(arm: &Arm) -> Option<String> {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.comps {
        match c.op {
            CmpOp::Eq => {
                lower = Some((c.version.clone(), true));
                upper = Some((c.version.clone(), true));
            }
            CmpOp::Gt | CmpOp::Ge => {
                let inc = matches!(c.op, CmpOp::Ge);
                if lower.as_ref().is_none_or(|(v, old_inc)| {
                    compare_versions(&c.version, v) == Ordering::Greater
                        || (compare_versions(&c.version, v) == Ordering::Equal && !inc && *old_inc)
                }) {
                    lower = Some((c.version.clone(), inc));
                }
            }
            CmpOp::Lt | CmpOp::Le => {
                let inc = matches!(c.op, CmpOp::Le);
                if upper.as_ref().is_none_or(|(v, old_inc)| {
                    compare_versions(&c.version, v) == Ordering::Less
                        || (compare_versions(&c.version, v) == Ordering::Equal && !inc && *old_inc)
                }) {
                    upper = Some((c.version.clone(), inc));
                }
            }
        }
    }

    if !arm_has_stable_interval(arm)
        && !arm.anchors.iter().any(|a| {
            let v = Version {
                major: a.0,
                minor: a.1,
                patch: a.2,
                pre: vec!["0".to_string()],
                build: Vec::new(),
            };
            arm_matches(&v, arm, false)
        })
    {
        return None;
    }

    let l = lower.map_or("*".to_string(), |(mut v, inc)| {
        v.build.clear();
        format!("{}{}", if inc { ">=" } else { ">" }, stripped_version(&v))
    });
    let u = upper.map_or("*".to_string(), |(mut v, inc)| {
        v.build.clear();
        format!("{}{}", if inc { "<=" } else { "<" }, stripped_version(&v))
    });
    let mut anchors = arm.anchors.clone();
    anchors.sort_unstable();
    anchors.dedup();
    let anch = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors
            .iter()
            .map(|a| format!("{}.{}.{}", a.0, a.1, a.2))
            .collect::<Vec<_>>()
            .join(",")
    };
    Some(format!("[{};{};pre={}]", l, u, anch))
}

pub fn canonical(r: &Range) -> String {
    let mut arms: Vec<String> = r.arms.iter().filter_map(canonical_arm).collect();
    arms.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    arms.dedup();
    if arms.is_empty() {
        "EMPTY".to_string()
    } else {
        arms.join(" || ")
    }
}
