use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_missing_core_byte(b: u8) -> bool {
    b == b'.' || b == b'-' || b == b'+'
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            if pos == bytes.len() || is_missing_core_byte(bytes[pos]) {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &b in &bytes[start..pos] {
            value = match value
                .checked_mul(10)
                .and_then(|n| n.checked_add((b - b'0') as u64))
            {
                Some(n) => n,
                None => return Err(VersionError::Overflow { at: start }),
            };
        }
        core[field] = value;

        if field < 2 {
            if pos >= bytes.len() || bytes[pos] != b'.' {
                if pos >= bytes.len() || is_missing_core_byte(bytes[pos]) {
                    return Err(VersionError::MissingCore { at: pos });
                }
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let plus = bytes[pos..].iter().position(|&b| b == b'+').map(|i| pos + i);
        let end = plus.unwrap_or(bytes.len());
        parse_identifiers(s, pre_start, end, true, &mut pre)?;
        pos = end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    if start == end {
        return Err(VersionError::EmptyIdentifier { at: start });
    }

    let mut id_start = start;
    let mut i = start;
    loop {
        if i == end || bytes[i] == b'.' {
            if i == id_start {
                return Err(VersionError::EmptyIdentifier { at: id_start });
            }
            let id = &s[id_start..i];
            if prerelease
                && id.len() > 1
                && id.as_bytes()[0] == b'0'
                && id.bytes().all(|b| b.is_ascii_digit())
            {
                return Err(VersionError::LeadingZero { at: id_start });
            }
            out.push(id.to_string());
            if i == end {
                break;
            }
            id_start = i + 1;
            i += 1;
            continue;
        }

        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || b == b'-') {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(())
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let xn = is_numeric_id(x);
                let yn = is_numeric_id(y);
                let c = match (xn, yn) {
                    (true, true) => cmp_numeric_id(x, y),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    (false, false) => x.as_bytes().cmp(y.as_bytes()),
                };
                if c != Ordering::Equal {
                    return c;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    comps: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, start: usize, end: usize) -> Vec<Token> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && is_ws(b[i]) {
            i += 1;
        }
        if i >= end {
            break;
        }
        let ts = i;
        while i < end && !is_ws(b[i]) {
            i += 1;
        }
        out.push(Token {
            text: s[ts..i].to_string(),
            start: ts,
        });
    }
    out
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn full_version_for_operand(
    text: &str,
    at: usize,
) -> Result<Version, RangeError> {
    parse_version(text).map_err(|error| RangeError::Version { at, error })
}

fn token_is_numeric_start(t: &str) -> bool {
    t.as_bytes()
        .first()
        .map_or(false, |b| b.is_ascii_digit())
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
    numeric_count: usize,
}

fn parse_partial_operand(
    text: &str,
    at: usize,
) -> Result<Partial, RangeError> {
    let b = text.as_bytes();
    if text.contains('-') || text.contains('+') {
        let v = full_version_for_operand(text, at)?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            full: Some(v),
            numeric_count: 3,
        });
    }

    let fields_s: Vec<&str> = text.split('.').collect();
    let has_wildcard = fields_s
        .iter()
        .any(|f| matches!(*f, "*" | "x" | "X"));
    if fields_s.len() >= 3 && !has_wildcard {
        let v = full_version_for_operand(text, at)?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            full: Some(v),
            numeric_count: 3,
        });
    }

    if fields_s.is_empty() || fields_s.len() > 3 {
        return Err(RangeError::Syntax { at });
    }

    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    let mut numeric_count = 0;
    for field in fields_s {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
            fields.push(None);
            continue;
        }

        if wildcard_seen
            || !field.bytes().all(|b| b.is_ascii_digit())
            || (field.len() > 1 && field.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }

        let mut value = 0u64;
        for ch in field.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((ch - b'0') as u64))
                .ok_or(RangeError::Syntax { at })?;
        }
        fields.push(Some(value));
        numeric_count += 1;
    }

    while fields.len() < 3 {
        fields.push(None);
    }

    Ok(Partial {
        fields,
        full: None,
        numeric_count,
    })
}

fn padded_version(p: &Partial) -> Version {
    Version {
        major: p.fields.first().copied().flatten().unwrap_or(0),
        minor: p.fields.get(1).copied().flatten().unwrap_or(0),
        patch: p.fields.get(2).copied().flatten().unwrap_or(0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_version(
    major: u64,
    minor: u64,
    patch: u64,
    at: usize,
) -> Result<Version, RangeError> {
    Ok(Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    })
}

fn increment_tuple(
    major: u64,
    minor: u64,
    patch: u64,
    which: usize,
    at: usize,
) -> Result<Version, RangeError> {
    match which {
        0 => Ok(Version {
            major: major.checked_add(1).ok_or(RangeError::Syntax { at })?,
            minor: 0,
            patch: 0,
            pre: vec!["0".to_string()],
            build: Vec::new(),
        }),
        1 => Ok(Version {
            major,
            minor: minor.checked_add(1).ok_or(RangeError::Syntax { at })?,
            patch: 0,
            pre: vec!["0".to_string()],
            build: Vec::new(),
        }),
        _ => Ok(Version {
            major,
            minor,
            patch: patch.checked_add(1).ok_or(RangeError::Syntax { at })?,
            pre: vec!["0".to_string()],
            build: Vec::new(),
        }),
    }
}

fn push_comparator(arm: &mut Arm, op: Op, version: Version, explicit: bool) {
    let anchor = explicit && !version.pre.is_empty();
    if anchor {
        arm.anchors.push(core_tuple(&version));
    }
    arm.comps.push(Comparator {
        op,
        version,
        anchor,
    });
}

fn parse_explicit_comparator(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let b = token.text.as_bytes();
    let (op, n) = if b.starts_with(b">=") {
        (Op::Ge, 2)
    } else if b.starts_with(b"<=") {
        (Op::Le, 2)
    } else if b.starts_with(b">") {
        (Op::Gt, 1)
    } else if b.starts_with(b"<") {
        (Op::Lt, 1)
    } else if b.starts_with(b"=") {
        (Op::Eq, 1)
    } else {
        return Err(RangeError::Syntax { at: token.start });
    };

    let operand = &token.text[n..];
    let operand_at = token.start + n;
    if operand.is_empty() || !token_is_numeric_start(operand) {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = full_version_for_operand(operand, operand_at)?;
    push_comparator(arm, op, v, true);
    Ok(())
}

fn parse_bare_token(token: &Token, arm: &mut Arm) -> Result<(), RangeError> {
    let p = parse_partial_operand(&token.text, token.start)?;
    if let Some(v) = p.full {
        push_comparator(arm, Op::Eq, v, true);
        return Ok(());
    }

    match p.numeric_count {
        0 => Ok(()),
        1 => {
            let lo = padded_version(&p);
            push_comparator(arm, Op::Ge, lo, false);
            let hi = increment_tuple(p.fields[0].unwrap(), 0, 0, 0, token.start)?;
            arm.comps.push(Comparator {
                op: Op::Lt,
                version: hi,
                anchor: false,
            });
            Ok(())
        }
        2 => {
            let lo = padded_version(&p);
            push_comparator(arm, Op::Ge, lo, false);
            let hi = increment_tuple(
                p.fields[0].unwrap(),
                p.fields[1].unwrap(),
                0,
                1,
                token.start,
            )?;
            arm.comps.push(Comparator {
                op: Op::Lt,
                version: hi,
                anchor: false,
            });
            Ok(())
        }
        _ => {
            let v = padded_version(&p);
            push_comparator(arm, Op::Eq, v, false);
            Ok(())
        }
    }
}

fn parse_caret_tilde(
    token: &Token,
    arm: &mut Arm,
    caret: bool,
) -> Result<(), RangeError> {
    let operand = &token.text[1..];
    let operand_at = token.start + 1;
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let p = parse_partial_operand(operand, operand_at)?;
    if p.numeric_count == 0 {
        return Err(RangeError::Syntax { at: token.start });
    }

    let lo = p.full.clone().unwrap_or_else(|| padded_version(&p));
    let (major, minor, patch) = core_tuple(&lo);
    push_comparator(arm, Op::Ge, lo.clone(), p.full.is_some() && !lo.pre.is_empty());

    let upper = if caret {
        let k = p.numeric_count;
        if major > 0 || k == 1 {
            increment_tuple(major, 0, 0, 0, token.start)?
        } else if minor > 0 || k == 2 {
            increment_tuple(major, minor, 0, 1, token.start)?
        } else {
            increment_tuple(major, minor, patch, 2, token.start)?
        }
    } else if p.numeric_count == 1 {
        increment_tuple(major, 0, 0, 0, token.start)?
    } else {
        increment_tuple(major, minor, 0, 1, token.start)?
    };

    arm.comps.push(Comparator {
        op: Op::Lt,
        version: upper,
        anchor: false,
    });
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token], arm: &mut Arm) -> Result<(), RangeError> {
    let minus_at = tokens
        .iter()
        .find(|t| t.text == "-")
        .map(|t| t.start)
        .unwrap_or(0);
    if tokens.len() != 3 || tokens[1].text != "-" {
        return Err(RangeError::Syntax { at: minus_at });
    }

    let lower = parse_partial_operand(&tokens[0].text, tokens[0].start)?;
    let upper = parse_partial_operand(&tokens[2].text, tokens[2].start)?;

    if lower.numeric_count == 0 {
        return Err(RangeError::Syntax {
            at: tokens[0].start,
        });
    }
    if upper.numeric_count == 0 {
        return Err(RangeError::Syntax {
            at: tokens[2].start,
        });
    }

    let low_v = lower.full.clone().unwrap_or_else(|| padded_version(&lower));
    push_comparator(
        arm,
        Op::Ge,
        low_v.clone(),
        lower.full.is_some() && !low_v.pre.is_empty(),
    );

    if let Some(v) = upper.full {
        push_comparator(arm, Op::Le, v, true);
    } else {
        let hi = match upper.numeric_count {
            1 => increment_tuple(upper.fields[0].unwrap(), 0, 0, 0, tokens[2].start)?,
            2 => increment_tuple(
                upper.fields[0].unwrap(),
                upper.fields[1].unwrap(),
                0,
                1,
                tokens[2].start,
            )?,
            _ => {
                let v = padded_version(&upper);
                push_comparator(arm, Op::Le, v, false);
                return Ok(());
            }
        };
        arm.comps.push(Comparator {
            op: Op::Lt,
            version: hi,
            anchor: false,
        });
    }
    Ok(())
}

fn parse_arm(tokens: &[Token]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comps: Vec::new(),
        anchors: Vec::new(),
    };

    if tokens.iter().any(|t| t.text == "-") {
        parse_hyphen_arm(tokens, &mut arm)?;
        return Ok(arm);
    }

    for token in tokens {
        let first = token.text.as_bytes().first().copied();
        match first {
            Some(b'<') | Some(b'>') | Some(b'=') => {
                parse_explicit_comparator(token, &mut arm)?;
            }
            Some(b'^') => parse_caret_tilde(token, &mut arm, true)?,
            Some(b'~') => parse_caret_tilde(token, &mut arm, false)?,
            Some(b'*') | Some(b'x') | Some(b'X') => {
                parse_bare_token(token, &mut arm)?;
            }
            Some(b) if b.is_ascii_digit() => parse_bare_token(token, &mut arm)?,
            _ => return Err(RangeError::Syntax { at: token.start }),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut i = 0usize;
    let mut saw_any = false;

    while i < bytes.len() {
        if bytes[i] == b'|' && i + 1 < bytes.len() && bytes[i + 1] == b'|' {
            let left_tokens = split_tokens(s, arm_start, i);
            if left_tokens.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            let arm = parse_arm(&left_tokens)?;
            arms.push(arm);
            saw_any = true;
            i += 2;
            while i < bytes.len() && is_ws(bytes[i]) {
                i += 1;
            }
            arm_start = i;
            continue;
        }
        i += 1;
    }

    let tokens = split_tokens(s, arm_start, bytes.len());
    if tokens.is_empty() {
        if saw_any {
            return Err(RangeError::Empty { at: bytes.len() });
        }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(&tokens)?);

    Ok(Range { arms })
}

fn cmp_one(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_admits(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if v.pre.is_empty() || include_prerelease {
        return true;
    }
    let tuple = core_tuple(v);
    arm.anchors.iter().any(|a| *a == tuple)
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    arm_admits(v, arm, include_prerelease) && arm.comps.iter().all(|c| cmp_one(v, c))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(&parsed[i], &parsed[j]);
                if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }

    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

#[derive(Clone)]
struct Interval {
    lower: Option<(Version, bool)>,
    upper: Option<(Version, bool)>,
}

fn strongest_bounds(arm: &Arm) -> Option<Interval> {
    let mut interval = Interval {
        lower: None,
        upper: None,
    };
    for c in &arm.comps {
        match c.op {
            Op::Eq => {
                let low = (c.version.clone(), true);
                let high = (c.version.clone(), true);
                interval.lower = choose_lower(interval.lower, Some(low));
                interval.upper = choose_upper(interval.upper, Some(high));
            }
            Op::Gt => {
                interval.lower =
                    choose_lower(interval.lower, Some((c.version.clone(), false)));
            }
            Op::Ge => {
                interval.lower =
                    choose_lower(interval.lower, Some((c.version.clone(), true)));
            }
            Op::Lt => {
                interval.upper =
                    choose_upper(interval.upper, Some((c.version.clone(), false)));
            }
            Op::Le => {
                interval.upper =
                    choose_upper(interval.upper, Some((c.version.clone(), true)));
            }
        }
    }
    Some(interval)
}

fn choose_lower(
    a: Option<(Version, bool)>,
    b: Option<(Version, bool)>,
) -> Option<(Version, bool)> {
    match (a, b) {
        (None, x) | (x, None) => x,
        (Some(x), Some(y)) => {
            let ord = compare_versions(&x.0, &y.0);
            if ord == Ordering::Greater {
                Some(x)
            } else if ord == Ordering::Less {
                Some(y)
            } else {
                Some((x.0, x.1 && y.1))
            }
        }
    }
}

fn choose_upper(
    a: Option<(Version, bool)>,
    b: Option<(Version, bool)>,
) -> Option<(Version, bool)> {
    match (a, b) {
        (None, x) | (x, None) => x,
        (Some(x), Some(y)) => {
            let ord = compare_versions(&x.0, &y.0);
            if ord == Ordering::Less {
                Some(x)
            } else if ord == Ordering::Greater {
                Some(y)
            } else {
                Some((x.0, x.1 && y.1))
            }
        }
    }
}

fn strip_build(mut v: Version) -> Version {
    v.build.clear();
    v
}

fn version_display(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn arm_to_canonical(arm: &Arm) -> String {
    let interval = strongest_bounds(arm).unwrap();
    let lower = interval.lower.map(|(v, inclusive)| {
        format!(
            "{}{}",
            if inclusive { ">=" } else { ">" },
            version_display(&strip_build(v))
        )
    });
    let upper = interval.upper.map(|(v, inclusive)| {
        format!(
            "{}{}",
            if inclusive { "<=" } else { "<" },
            version_display(&strip_build(v))
        )
    });
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let anchor_text = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors
            .iter()
            .map(|(a, b, c)| format!("{a}.{b}.{c}"))
            .collect::<Vec<_>>()
            .join(",")
    };
    format!(
        "[{};{};pre={}]",
        lower.unwrap_or_else(|| "*".to_string()),
        upper.unwrap_or_else(|| "*".to_string()),
        anchor_text
    )
}

pub fn canonical(r: &Range) -> String {
    let mut strings = r.arms.iter().map(arm_to_canonical).collect::<Vec<_>>();
    strings.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    strings.dedup();
    if strings.is_empty() {
        "EMPTY".to_string()
    } else {
        strings.join(" || ")
    }
}

fn compare_core(a: (u64, u64, u64), b: (u64, u64, u64)) -> Ordering {
    a.cmp(&b)
}

fn core_successor(t: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if t.2 < u64::MAX {
        Some((t.0, t.1, t.2 + 1))
    } else if t.1 < u64::MAX {
        Some((t.0, t.1 + 1, 0))
    } else if t.0 < u64::MAX {
        Some((t.0 + 1, 0, 0))
    } else {
        None
    }
}

fn admits_tuple(arm: &Arm, tuple: (u64, u64, u64), include: bool) -> bool {
    include || arm.anchors.iter().any(|a| *a == tuple)
}

fn tuple_version(tuple: (u64, u64, u64), prerelease: bool) -> Version {
    Version {
        major: tuple.0,
        minor: tuple.1,
        patch: tuple.2,
        pre: if prerelease {
            vec!["0".to_string()]
        } else {
            Vec::new()
        },
        build: Vec::new(),
    }
}

fn check_candidate_pair(
    a: &Arm,
    b: &Arm,
    v: &Version,
    include: bool,
) -> bool {
    arm_matches(v, a, include) && arm_matches(v, b, include)
}

fn intersects_arms(a: &Arm, b: &Arm, include: bool) -> bool {
    let ia = strongest_bounds(a).unwrap();
    let ib = strongest_bounds(b).unwrap();
    let lower = choose_lower(ia.lower, ib.lower);
    let upper = choose_upper(ia.upper, ib.upper);

    if let (Some((lo, lo_inc)), Some((hi, hi_inc))) = (&lower, &upper) {
        match compare_versions(lo, hi) {
            Ordering::Greater => return false,
            Ordering::Equal if !(*lo_inc && *hi_inc) => return false,
            _ => {}
        }
    }

    let mut candidates = Vec::<Version>::new();
    for bound in [lower.as_ref(), upper.as_ref()].into_iter().flatten() {
        let v = strip_build(bound.0.clone());
        candidates.push(v.clone());
        let tuple = core_tuple(&v);
        candidates.push(tuple_version(tuple, true));
        candidates.push(tuple_version(tuple, false));
        if let Some(next) = core_successor(tuple) {
            candidates.push(tuple_version(next, true));
            candidates.push(tuple_version(next, false));
        }
    }
    for tuple in a.anchors.iter().chain(b.anchors.iter()) {
        candidates.push(tuple_version(*tuple, true));
        candidates.push(tuple_version(*tuple, false));
    }
    candidates.push(tuple_version((0, 0, 0), true));
    candidates.push(tuple_version((0, 0, 0), false));

    for v in &candidates {
        if check_candidate_pair(a, b, v, include) {
            return true;
        }
    }

    if include {
        return interval_has_stable_witness(&lower, &upper);
    }

    let anchors_a = a.anchors.iter().copied().collect::<Vec<_>>();
    let anchors_b = b.anchors.iter().copied().collect::<Vec<_>>();
    for t in anchors_a {
        if anchors_b.contains(&t) {
            let pre = tuple_version(t, true);
            if check_candidate_pair(a, b, &pre, false) {
                return true;
            }
        }
    }
    interval_has_stable_witness(&lower, &upper)
}

fn interval_has_stable_witness(
    lower: &Option<(Version, bool)>,
    upper: &Option<(Version, bool)>,
) -> bool {
    if let (Some((lo, lo_inc)), Some((hi, hi_inc))) = (lower, upper) {
        let ord = compare_versions(lo, hi);
        if ord == Ordering::Greater || (ord == Ordering::Equal && !(*lo_inc && *hi_inc)) {
            return false;
        }
        if ord == Ordering::Equal {
            return lo.pre.is_empty();
        }
        let lo_core = core_tuple(lo);
        let hi_core = core_tuple(hi);
        if lo_core < hi_core {
            let stable = tuple_version(lo_core, false);
            return compare_versions(&stable, lo) != Ordering::Less
                && compare_versions(&stable, hi) != Ordering::Greater;
        }
        return false;
    }

    if let Some((lo, inclusive)) = lower {
        if lo.pre.is_empty() {
            return true;
        }
        if !inclusive || !lo.pre.is_empty() {
            if let Some(next) = core_successor(core_tuple(lo)) {
                return true;
            }
        }
        return true;
    }

    true
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    a.arms.iter().any(|left| {
        b.arms
            .iter()
            .any(|right| intersects_arms(left, right, include_prerelease))
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    fn pr(s: &str) -> Range {
        parse_range(s).unwrap()
    }

    #[test]
    fn public_version_comparators() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(
            compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")),
            Ordering::Less
        );
        let r = pr(">=2.1.0 <3.0.0");
        assert!(matches(&pv("2.5.0"), &r));
        assert!(!matches(&pv("3.0.0"), &r));
        assert!(matches(
            &pv("2.1.3-rc"),
            &pr(">=2.1.3-beta <2.1.3")
        ));
    }

    #[test]
    fn public_expanded_extrema() {
        for s in ["^2.1.0", "~2.5.0", "2.5.x", "2.1 - 2.8"] {
            assert!(matches(&pv("2.5.1"), &pr(s)));
        }
        assert!(matches(&pv("4.0.0"), &pr("2.x || 4.0.0")));
        let cs = vec![
            "2.2.0".to_string(),
            "2.8.0".to_string(),
            "3.0.0".to_string(),
        ];
        let r = pr("2.x");
        assert_eq!(max_satisfying(&cs, &r), Ok(Some("2.8.0".into())));
        assert_eq!(min_satisfying(&cs, &r), Ok(Some("2.2.0".into())));
    }

    #[test]
    fn public_algebra() {
        let r = pr(">=2.1.0 <2.8.0");
        assert!(intersects(&r, &pr("2.5.x"), false));
        assert_eq!(canonical(&r), "[>=2.1.0;<2.8.0;pre=-]");
        assert!(matches_with(&pv("2.5.0-rc"), &r, true));
    }

    #[test]
    fn public_canonical_empty() {
        assert_eq!(canonical(&pr(">=2.0.0 <2.0.0")), "EMPTY");
        assert!(!intersects(&pr("2.x"), &pr("3.x"), true));
    }
}
