    } else {
        0

        return Err(BaseError::Empty { operand });

        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let mut digits = Vec::new();
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        } else if let Some(value) = digit_value(byte) {
            digits.push(value);
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in value.iter_mut() {
        let product = (*limb as u16) * radix as u16 + carry;
        *limb = (product & 0xff) as u8;
        carry = product >> 8;
    }
    while carry != 0 {
        value.push((carry & 0xff) as u8);
        carry >>= 8;
    }
}

fn add_values(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for index in 0..left.len().max(right.len()) {
        let sum = left.get(index).copied().unwrap_or(0) as u16
            + right.get(index).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum & 0xff) as u8);
        carry = sum >> 8;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn to_digits(value: &[u8], radix: u8) -> Vec<u8> {
    let mut number = value.to_vec();
    while number.last() == Some(&0) {
        number.pop();
    }
    if number.is_empty() {
        return vec![0];
    }

    let mut digits = Vec::new();
    while !number.is_empty() {
        let mut remainder = 0u16;
        for limb in number.iter_mut().rev() {
            let current = (remainder << 8) | *limb as u16;
            *limb = (current / radix as u16) as u8;
            remainder = current % radix as u16;
        }
        digits.push(remainder as u8);
        while number.last() == Some(&0) {
            number.pop();
        }
    }
    digits.reverse();
    digits
}

fn format_value(value: &[u8], radix: u8, style: BaseStyle) -> String {
    let digits = to_digits(value, radix);
    let padding = style.min_width.saturating_sub(digits.len());
    let total_len = padding + digits.len();

    let mut output = String::with_capacity(total_len + total_len / style.group.max(1));
    for index in 0..total_len {
        if style.group != 0 && index != 0 && (total_len - index) % style.group == 0 {

    }
    output
}

            index: payload_start + offset,
        });

        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn to_big_endian_bytes(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        let mut carry = digit as u16;
        for byte in value.iter_mut().rev() {
            let n = *byte as u16 * radix as u16 + carry;
            *byte = (n & 0xff) as u8;
            carry = n >> 8;
        }
        while carry != 0 {
            value.insert(0, (carry & 0xff) as u8);
            carry >>= 8;
        }
    }
    let first_nonzero = value.iter().position(|&b| b != 0).unwrap_or(value.len() - 1);
    value.drain(0..first_nonzero);
    value
}

fn add_bytes(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry != 0 {
        let a = if i > 0 {
            i -= 1;
            left[i] as u16
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            right[j] as u16
        } else {
            0
        };
        let sum = a + b + carry;
        result.push((sum & 0xff) as u8);
        carry = sum >> 8;
    }

    result.reverse();
    result
}

fn render(mut value: Vec<u8>, radix: u8, style: BaseStyle) -> String {
    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };

    let mut digits = Vec::new();
    if value.iter().all(|&b| b == 0) {
        digits.push(b'0');
    } else {
        while value.iter().any(|&b| b != 0) {
            let mut remainder = 0u16;
            for byte in value.iter_mut() {
                let current = (remainder << 8) | *byte as u16;
                *byte = (current / radix as u16) as u8;
                remainder = current % radix as u16;
            }
            digits.push(alphabet[remainder as usize]);
            let first_nonzero = value.iter().position(|&b| b != 0).unwrap_or(value.len());
            value.drain(0..first_nonzero);
            if value.is_empty() {
                break;
            }
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let mut padded = vec![b'0'; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    if style.group == 0 || digits.len() <= style.group {
        return String::from_utf8(digits).expect("ASCII digits");
    }

    let mut output = Vec::with_capacity(digits.len() + digits.len() / style.group);
    for (index, digit) in digits.iter().enumerate() {
        if index > 0 && (digits.len() - index) % style.group == 0 {
            output.push(b'_');
        }
        output.push(*digit);
    }
    String::from_utf8(output).expect("ASCII digits")
}

fn accumulate_in_base(value: &mut Vec<u8>, multiplier: u8, addend: u8, radix: u8) {
    let mut carry = addend as u16;
    for digit in value.iter_mut() {
        let current = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (current % radix as u16) as u8;
        carry = current / radix as u16;
    }
    while carry != 0 {
        value.push((carry % radix as u16) as u8);
        carry /= radix as u16;
    }
}

    result

    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    Ok(format_digits(convert_digits(&digits, from, to), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value: Vec<u8> = left_digits.into_iter().rev().collect();
    let right_value: Vec<u8> = right_digits.into_iter().rev().collect();
    Ok(format_digits(add_digits(&left_value, &right_value, base), style))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash_index) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash_index] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash_index + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let mut digits = Vec::new();

    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => return Err(BaseError::InvalidDigit { operand, index }),
        }
    }

    for index in payload_start..bytes.len() {
        if bytes[index] == b'_' {
            let left_is_digit = index > payload_start && digit_value(bytes[index - 1]).is_some();
            let right_is_digit =
                index + 1 < bytes.len() && digit_value(bytes[index + 1]).is_some();
            if !left_is_digit || !right_is_digit {
                return Err(BaseError::Separator { operand, index });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(number: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for digit in number.iter_mut() {
        let value = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (value % 10) as u8;
        carry = value / 10;
    }
    while carry != 0 {
        number.push((carry % 10) as u8);
        carry /= 10;
    }
}

fn to_decimal(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut decimal = vec![0];
    for &digit in digits {
        multiply_add(&mut decimal, radix, digit);
    }
    while decimal.len() > 1 && decimal.last() == Some(&0) {
        decimal.pop();
    }
    decimal
}

fn divide_small(number: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for digit in number.iter_mut().rev() {
        let value = remainder * 10 + *digit as u16;
        *digit = (value / divisor as u16) as u8;
        remainder = value % divisor as u16;
    }
    while number.len() > 1 && number.last() == Some(&0) {
        number.pop();
    }
    remainder as u8
}

fn render(mut decimal: Vec<u8>, radix: u8, style: BaseStyle) -> String {
    let mut reversed = Vec::new();
    if decimal.iter().all(|&digit| digit == 0) {
        reversed.push(0);
    } else {
        while decimal.iter().any(|&digit| digit != 0) {
            reversed.push(divide_small(&mut decimal, radix));
        }
    }

    let mut chars: Vec<char> = reversed
        .into_iter()
        .rev()
        .map(|digit| match digit {
            0..=9 => (b'0' + digit) as char,
            _ if style.upper => (b'A' + digit - 10) as char,
            _ => (b'a' + digit - 10) as char,
        })
        .collect();

    if chars.len() < style.min_width {
        let mut padded = vec!['0'; style.min_width - chars.len()];
        padded.extend(chars);
        chars = padded;
    }

    if style.group == 0 || chars.len() <= style.group {
        return chars.into_iter().collect();
    }

    let mut output = String::with_capacity(chars.len() + (chars.len() - 1) / style.group);
    for (index, ch) in chars.iter().enumerate() {
        if index != 0 && (chars.len() - index) % style.group == 0 {
            output.push('_');
        }
        output.push(*ch);
    }
    output
}

fn add_decimal(left: Vec<u8>, right: Vec<u8>) -> Vec<u8> {
    let mut sum = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u8;
    for index in 0..left.len().max(right.len()) {
        let value = left.get(index).copied().unwrap_or(0)
            + right.get(index).copied().unwrap_or(0)
            + carry;
        sum.push(value % 10);
        carry = value / 10;
    }
    if carry != 0 {
        sum.push(carry);
    }
    sum
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    Ok(render(to_decimal(&digits, from), to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_decimal = to_decimal(&left_digits, base);
    let right_decimal = to_decimal(&right_digits, base);
    Ok(render(add_decimal(left_decimal, right_decimal), base, style))
}
