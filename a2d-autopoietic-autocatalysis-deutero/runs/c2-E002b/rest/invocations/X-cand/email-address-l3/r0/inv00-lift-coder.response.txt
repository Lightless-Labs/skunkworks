#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::new();
    let mut positions = Vec::new();
    let mut runs: Vec<Run> = Vec::new();
    let mut i = 0usize;
    let mut quote_open = false;
    let mut quote_escape = false;
    let mut quote_closed = false;

    while i < bytes.len() {
        let b = bytes[i];

        if quote_open {
            clean.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                quote_open = false;
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if clean.is_empty() && !quote_closed && b == b'"' {
            quote_open = true;
            clean.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' || b == b'\r' || b == b'\n' {
            let run_start = i;
            loop {
                let before = i;
                while i < bytes.len() && (bytes[i] == b' ' || bytes[i] == b'\t') {
                    i += 1;
                }
                if i < bytes.len() && (bytes[i] == b'\r' || bytes[i] == b'\n') {
                    if bytes[i] == b'\r'
                        && i + 2 < bytes.len()
                        && bytes[i + 1] == b'\n'
                        && (bytes[i + 2] == b' ' || bytes[i + 2] == b'\t')
                    {
                        i += 2;
                        while i < bytes.len() && (bytes[i] == b' ' || bytes[i] == b'\t') {
                            i += 1;
                        }
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if i < bytes.len() && bytes[i] == b'(' {
                    let outer = i;
                    let mut depth = 0usize;
                    while i < bytes.len() {
                        let c = bytes[i];
                        if c == b'\\' {
                            if i + 1 >= bytes.len()
                                || !(0x20..=0x7e).contains(&bytes[i + 1])
                            {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                            continue;
                        }
                        if c == b'\r' || c == b'\n' {
                            if c == b'\r'
                                && i + 2 < bytes.len()
                                && bytes[i + 1] == b'\n'
                                && (bytes[i + 2] == b' ' || bytes[i + 2] == b'\t')
                            {
                                i += 2;
                                while i < bytes.len()
                                    && (bytes[i] == b' ' || bytes[i] == b'\t')
                                {
                                    i += 1;
                                }
                                continue;
                            }
                            return Err(EmailError::BadFold { at: i });
                        }
                        if c == b'(' {
                            depth += 1;
                            if depth > 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            i += 1;
                            continue;
                        }
                        if c == b')' {
                            depth -= 1;
                            i += 1;
                            if depth == 0 {
                                break;
                            }
                            continue;
                        }
                        if !(0x20..=0x7e).contains(&c) && c != b'\t' {
                            return Err(EmailError::CommentChar { at: i });
                        }
                        i += 1;
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                } else {
                    break;
                }
                if i == before {
                    break;
                }
                if i >= bytes.len()
                    || !(bytes[i] == b' '
                        || bytes[i] == b'\t'
                        || bytes[i] == b'('
                        || bytes[i] == b'\r'
                        || bytes[i] == b'\n')
                {
                    break;
                }
            }
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        clean.push(b);
        positions.push(i);
        i += 1;
    }

    let cleaned = String::from_utf8(clean.clone()).unwrap();
    for run in &runs {
        let left = positions.iter().rposition(|&p| p < run.start);
        let right = positions.iter().position(|&p| p >= run.end);
        let legal_left = left.map_or(true, |k| clean[k] == b'@');
        let legal_right = right.map_or(true, |k| clean[k] == b'@');
        if !legal_left && !legal_right {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1usize;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: positions[0] });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut ats = Vec::new();
    for j in 0..clean.len() {
        if clean[j] == b'@' && !(quote_end.is_some() && j <= protected_end) {
            ats.push(j);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[ats[1]],
        });
    }

    let at_index = ats[0];
    if at_index == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[at_index],
        });
    }

    let (local, quoted) = if let Some(qend) = quote_end {
        if qend + 1 != at_index {
            let pos = positions.get(qend + 1).copied().unwrap_or(input.len());
            return Err(EmailError::TrailingLocal { at: pos });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < qend {
            let b = clean[j];
            if b == b'\\' {
                if j + 1 >= qend {
                    return Err(EmailError::BadEscape { at: positions[j] });
                }
                let next = clean[j + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: positions[j] });
                }
                decoded.push(next);
                origins.push(positions[j]);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: positions[j] });
                }
                decoded.push(b);
                origins.push(positions[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[qend],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let local_bytes = &clean[..at_index];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for j in 0..local_bytes.len() {
            let b = local_bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: positions[j] });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: positions[j] });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = at_index + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(positions[domain_end]);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: positions.get(domain_start).copied().unwrap_or(input.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for j in domain_start..domain_end {
        if clean[j] == b'.' {
            if j == start {
                return Err(EmailError::EmptyLabel { at: positions[j] });
            }
            labels.push((start, j));
            start = j + 1;
        }
    }
    if start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: root_dot.unwrap_or(input.len()),
        });
    }
    labels.push((start, domain_end));

    for &(begin, end) in &labels {
        if end - begin > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[begin + 63],
            });
        }
        if clean[begin] == b'-' {
            return Err(EmailError::LabelHyphen { at: positions[begin] });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for j in begin..end {
            let b = clean[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: positions[j] });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: positions[domain_start],
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = cleaned[domain_start..domain_end]
        .to_ascii_lowercase();
    Ok(Address {
        local,
        domain,
        quoted,
    })
}
