#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
    retained_before: Option<usize>,
    retained_after: Option<usize>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let bytes = input.as_bytes();
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::<u8>::new();
    let mut map = Vec::<usize>::new();
    let mut runs = Vec::<Run>::new();
    let mut i = 0usize;
    let mut quote_prefix = false;
    let mut quote_open = None;

    while i < bytes.len() {
        if quote_prefix {
            let b = bytes[i];
            clean.push(b);
            map.push(i);
            i += 1;
            if b == b'\\' {
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    map.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                quote_prefix = false;
            }
            continue;
        }

        let b = bytes[i];
        if clean.is_empty() && b == b'"' {
            quote_prefix = true;
            quote_open = Some(i);
            clean.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        let is_construct = is_hws(b) || b == b'(' || b == b'\r' || b == b'\n';
        if is_construct {
            let start = i;
            let mut depth = 0usize;
            let mut outer_open = None;

            loop {
                if i >= bytes.len() {
                    break;
                }
                let c = bytes[i];

                if is_hws(c) {
                    i += 1;
                    continue;
                }

                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if depth == 0 {
                    if c != b'(' {
                        break;
                    }
                    outer_open = Some(i);
                    depth = 1;
                    i += 1;
                    continue;
                }

                if c == b'\\' {
                    let slash = i;
                    if i + 1 >= bytes.len()
                        || !(0x20..=0x7e).contains(&bytes[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: slash });
                    }
                    i += 2;
                    continue;
                }

                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if c == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                    continue;
                }

                if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 {
                        outer_open = None;
                    }
                    continue;
                }

                if !(0x20..=0x7e).contains(&c) && c != b'\t' {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }

            if depth != 0 {
                return Err(EmailError::UnclosedComment {
                    at: outer_open.unwrap_or(start),
                });
            }
            runs.push(Run {
                start,
                end: i,
                retained_before: clean.len().checked_sub(1),
                retained_after: None,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    if quote_prefix {
        let _ = quote_open;
    }

    let mut retained_at_positions = Vec::new();
    for (idx, &b) in clean.iter().enumerate() {
        if b == b'@' {
            retained_at_positions.push(idx);
        }
    }

    for run in &mut runs {
        run.retained_after = (run.end..bytes.len())
            .find_map(|_| None);
        let before = run.retained_before;
        let after = map.iter().position(|&p| p >= run.end);
        let legal = before.is_none()
            || after.is_none()
            || before.is_some_and(|x| clean.get(x) == Some(&b'@'))
            || after.is_some_and(|x| clean.get(x) == Some(&b'@'));
        if !legal {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = None;
    if clean.first() == Some(&b'"') {
        let mut j = 1usize;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                protected_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if protected_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: map[0] });
        }
    }

    let mut separators = Vec::new();
    let mut j = 0usize;
    while j < clean.len() {
        if Some(j) == protected_end {
            j += 1;
            continue;
        }
        if clean[j] == b'@' {
            separators.push(j);
        }
        j += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: map[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: map[sep] });
    }
    let local_bytes = &clean[..sep];
    let local_map = &map[..sep];

    let (local, quoted) = if local_bytes[0] == b'"' {
        let close = protected_end.unwrap_or(0);
        if close >= sep || close + 1 != sep {
            let at = if close + 1 < sep {
                map[close + 1]
            } else {
                map[sep]
            };
            return Err(EmailError::TrailingLocal { at });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut k = 1usize;
        while k < close {
            let b = local_bytes[k];
            if b == b'\\' {
                if k + 1 >= close
                    || !matches!(local_bytes[k + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape { at: local_map[k] });
                }
                decoded.push(local_bytes[k + 1]);
                origins.push(local_map[k]);
                k += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: local_map[k] });
                }
                decoded.push(b);
                origins.push(local_map[k]);
                k += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: local_map[close] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: local_map[64] });
        }
        for k in 0..local_bytes.len() {
            let b = local_bytes[k];
            if b == b'.' {
                if k == 0
                    || k + 1 == local_bytes.len()
                    || local_bytes[k - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: local_map[k] });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar { at: local_map[k] });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start_idx = sep + 1;
    let domain_start = map.get(domain_start_idx).copied().unwrap_or(input.len());
    if domain_start_idx >= clean.len() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let mut domain_end = clean.len();
    let mut root_dot = None;
    if clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(map[domain_end]);
    }
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_end - domain_start_idx > 253 {
        return Err(EmailError::DomainTooLong {
            at: map[domain_start_idx + 253],
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ranges = Vec::new();
    let mut start = domain_start_idx;
    for k in domain_start_idx..=domain_end {
        if k == domain_end || clean[k] == b'.' {
            if start == k {
                let at = if k == domain_end {
                    root_dot.unwrap_or(input.len())
                } else {
                    map[k]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            label_starts.push(start);
            label_ranges.push((start, k));
            start = k + 1;
        }
    }

    for &(a, b) in &label_ranges {
        if b - a > 63 {
            return Err(EmailError::LabelTooLong { at: map[a + 63] });
        }
        if clean[a] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[a] });
        }
        if clean[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[b - 1] });
        }
        for k in a..b {
            let c = clean[k];
            if !(c.is_ascii_alphanumeric() || c == b'-') {
                return Err(EmailError::DomainChar { at: map[k] });
            }
        }
    }

    if label_ranges.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (last_a, last_b) = *label_ranges.last().unwrap();
    if clean[last_a..last_b].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[last_a] });
    }

    let domain = clean[domain_start_idx..domain_end]
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    let _ = retained_at_positions;
    let _ = label_starts;

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
