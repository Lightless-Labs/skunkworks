#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut end = None;

        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    end = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }

        match end {
            Some(index) => Some(index),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }

        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    let (local, quoted) = if local_bytes[0] == b'"' {
        let end = quoted_end.expect("quoted local has a closing quote");
        if end + 1 != separator {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;

        while j < end {
            if local_bytes[j] == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape { at: j });
                }
                match local_bytes[j + 1] {
                    b'"' | b'\\' | b'@' => {
                        decoded.push(local_bytes[j + 1]);
                        origins.push(j);
                        j += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: j }),
                }
            } else {
                let byte = local_bytes[j];
                if !(0x20..=0x7e).contains(&byte) {
                    return Err(EmailError::QuotedChar { at: j });
                }
                decoded.push(byte);
                origins.push(j);
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (
            String::from_utf8(decoded).expect("decoded bytes are ASCII"),
            true,
        )
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (index, &byte) in local_bytes.iter().enumerate() {
            if byte == b'.' {
                if index == 0
                    || index + 1 == local_bytes.len()
                    || local_bytes[index - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: index });
                }
            } else if !byte.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
            {
                return Err(EmailError::LocalChar { at: index });
            }
        }

        (
            String::from_utf8(local_bytes.to_vec()).expect("input was checked as ASCII"),
            false,
        )
    };

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_bytes = &raw_domain[..domain_len];
    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0;

    for index in 0..=domain_len {
        if index == domain_len || domain_bytes[index] == b'.' {
            labels.push((label_start, &domain_bytes[label_start..index]));
            label_start = index + 1;
        }
    }

    for (label_index, (start, label)) in labels.iter().enumerate() {
        if label.is_empty() {
            let at = if label_index + 1 == labels.len() && has_root_dot {
                input.len() - 1
            } else {
                domain_start + start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + start + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + start,
            });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + start + label.len() - 1,
            });
        }

        for (offset, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + start + offset,
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let domain = domain_bytes
        .iter()
        .map(|byte| byte.to_ascii_lowercase())
        .collect::<Vec<_>>();
    let domain = String::from_utf8(domain).expect("domain bytes are ASCII");

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
