#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let ch = template[cursor..].chars().next().unwrap();

        if ch == '}' {
            return Err(Error::UnexpectedClose { pos: cursor });
        }
        if ch != '{' {
            cursor += ch.len_utf8();
            continue;
        }

        let open = cursor;
        let content_start = open + 1;
        let Some(relative_close) = template[content_start..].find('}') else {
            return Err(Error::Unclosed { pos: open });
        };
        let close = content_start + relative_close;
        let content = &template[content_start..close];

        if let Some(inner) = content.find('{') {
            return Err(Error::NestedOpen {
                pos: content_start + inner,
            });
        }

        let (operator, names_start) = match content.as_bytes().first().copied() {
            Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
                let op_pos = content_start;
                return Err(Error::UnsupportedOperator { pos: op_pos });
            }
            Some(_) => (None, content_start),
            None => (None, content_start),
        };
        let _ = operator;

        if names_start == close {
            return Err(Error::EmptyExpression { pos: close });
        }

        let names = &template[names_start..close];
        let mut seen: Vec<&str> = Vec::new();
        let mut start = 0;

        for part in names.split(',') {
            let part_start = names_start + start;
            if part.is_empty() {
                return Err(Error::EmptyVariable { pos: part_start });
            }

            let bytes = part.as_bytes();
            let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
            let valid_rest = |b: u8| b.is_ascii_alphanumeric() || b == b'_';

            if !valid_first(bytes[0]) {
                return Err(Error::InvalidName { pos: part_start });
            }
            if let Some(index) = bytes.iter().enumerate().skip(1).find_map(|(i, &b)| {
                if valid_rest(b) { None } else { Some(i) }
            }) {
                return Err(Error::InvalidName {
                    pos: part_start + index,
                });
            }
            if seen.contains(&part) {
                return Err(Error::DuplicateVariable { pos: part_start });
            }
            seen.push(part);
            start += part.len() + 1;
        }

        expressions.push((open, close + 1, names.to_owned()));
        cursor = close + 1;
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for (open, end, names) in expressions {
        output.push_str(&template[literal_start..open]);
        let mut fields = Vec::new();

        for name in names.split(',') {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(encode_value(value));
            }
        }

        if !fields.is_empty() {
            output.push_str(&fields.join(","));
        }
        literal_start = end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode_value(value: &str) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if is_unreserved(byte) {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(hex_digit(byte >> 4));
            out.push(hex_digit(byte & 0x0f));
        }
        i += 1;
    }

    out
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_digit(nibble: u8) -> char {
    match nibble {
        0..=9 => (b'0' + nibble) as char,
        _ => (b'A' + nibble - 10) as char,
    }
}
