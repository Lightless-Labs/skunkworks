#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);

        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => rank.push(0),
            }
        }

        if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parts = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(&route.segments, &parts) {
                let replace = match &best {
                    None => true,
                    Some((best_id, _)) => {
                        route.rank > self.routes[*best_id].rank
                            || (route.rank == self.routes[*best_id].rank && route_id < *best_id)
                    }
                };
                if replace {
                    best = Some((route_id, params));
                }
            }
        }

        best.map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

enum ParseError {
    Invalid(usize),
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    let raw = split_and_validate_head(input).map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::with_capacity(raw.len());
    let mut names = std::collections::HashSet::new();

    for (i, (start, text)) in raw.iter().enumerate() {
        let first = text.as_bytes()[0];
        if first == b':' || first == b'*' {
            let name = &text[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if first == b'*' {
                if i + 1 != raw.len() {
                    return Err(Error::InvalidPattern { at: *start });
                }
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            let decoded = decode_static(text, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok(segments)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    let raw = split_and_validate_head(input).map_err(|at| Error::InvalidPath { at })?;
    raw.iter()
        .map(|(start, text)| decode_static(text, *start).map_err(|at| Error::InvalidPath { at }))
        .collect()
}

fn split_and_validate_head(input: &str) -> Result<Vec<(usize, &str)>, usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut raw = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(i);
            }
            if i == start {
                return Err(start);
            }
            raw.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if start == bytes.len() {
        return Err(bytes.len() - 1);
    }
    raw.push((start, &input[start..]));
    Ok(raw)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_static(text: &str, base: usize) -> Result<String, usize> {
    let bytes = text.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(hi) = upper_hex(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(lo) = upper_hex(bytes[i + 2]) else {
                return Err(base + i);
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = path.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    (path_index == path.len()).then_some(params)
}
