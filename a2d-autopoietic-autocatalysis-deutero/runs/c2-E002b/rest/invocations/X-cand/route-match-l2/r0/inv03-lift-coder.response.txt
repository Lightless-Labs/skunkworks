#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);

        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => rank.push(0),
            }
        }
        if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let segments = parse_path(&path[..head_end])?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            let Some(params) = match_route(&route.segments, &segments) else {
                continue;
            };

            let replace = match &best {
                None => true,
                Some((best_id, _)) => {
                    route.rank > self.routes[*best_id].rank
                        || (route.rank == self.routes[*best_id].rank && route_id < *best_id)
                }
            };
            if replace {
                best = Some((route_id, params));
            }
        }

        best.map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut raw_segments = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                break;
            }
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            raw_segments.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    if start < bytes.len() {
        raw_segments.push((start, &input[start..]));
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();

    for (at, raw) in raw_segments {
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            segments.push(Segment::Static(decode_segment(raw, at, true)?));
        }
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    if let Some((at, _)) = raw_segments_for_wildcard_check(input).into_iter().find(|(at, raw)| {
        raw.starts_with('*') && *at != input.rfind('/').unwrap_or(0) + 1
    }) {
        return Err(Error::InvalidPattern { at });
    }

    Ok(segments)
}

fn raw_segments_for_wildcard_check(input: &str) -> Vec<(usize, &str)> {
    if input == "/" {
        return Vec::new();
    }
    let end = if input.ends_with('/') {
        input.len() - 1
    } else {
        input.len()
    };
    let mut out = Vec::new();
    let mut start = 1;
    for i in 1..end {
        if input.as_bytes()[i] == b'/' {
            out.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    if start <= end {
        out.push((start, &input[start..end]));
    }
    out
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                break;
            }
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            segments.push(decode_segment(&input[start..i], start, false)
                .map_err(|e| match e {
                    Error::InvalidPattern { at } | Error::InvalidPath { at } => Error::InvalidPath { at },
                    Error::NotFound => Error::NotFound,
                })?);
            start = i + 1;
        }
    }

    if start < bytes.len() {
        segments.push(decode_segment(&input[start..], start, false)
            .map_err(|e| match e {
                Error::InvalidPattern { at } | Error::InvalidPath { at } => Error::InvalidPath { at },
                Error::NotFound => Error::NotFound,
            })?);
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }
    Ok(segments)
}

fn decode_segment(raw: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            let fail = || {
                if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                }
            };
            if i + 2 >= bytes.len() {
                return Err(fail());
            }
            let hi = upper_hex(bytes[i + 1]).ok_or_else(fail)?;
            let lo = upper_hex(bytes[i + 2]).ok_or_else(fail)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(fail());
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: base + i }
            } else {
                Error::InvalidPath { at: base + i }
            });
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), path[path_index..].join("/")));
                return Some(params);
            }
        }
    }

    (path_index == path.len()).then_some(params)
}
