#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn clean_cfws(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut run_start: Option<usize> = None;

    let quoted_prefix = input.first() == Some(&b'"');
    let mut in_quote = false;
    let mut quote_closed = false;

    while i < input.len() {
        if quoted_prefix && !quote_closed {
            if !in_quote {
                in_quote = true;
            }
            let b = input[i];
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' && i > 0 {
                in_quote = false;
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if comment_depth > 0 {
            let b = input[i];
            if b == b'\\' {
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let b = input[i];
        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }
        if is_hws(b) {
            if run_start.is_none() {
                run_start = Some(i);
            }
            i += 1;
            continue;
        }
        if b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            comment_depth = 1;
            outer_comment = i;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if let Some(start) = run_start.take() {
            runs.push(Run { start, end: i });
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if let Some(start) = run_start {
        runs.push(Run {
            start,
            end: input.len(),
        });
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn mapped(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn local_atom_valid(local: &[u8]) -> bool {
    if local.is_empty() || local.len() > 64 || local[0] == b'.' || local[local.len() - 1] == b'.' {
        return false;
    }
    let mut previous_dot = false;
    for &b in local {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else {
            previous_dot = false;
            if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return false;
            }
        }
    }
    true
}

fn parse_cleaned(
    input: &str,
    cleaned: &Cleaned,
    allow_single_label: bool,
) -> Result<(Address, usize), EmailError> {
    let bytes = &cleaned.bytes;
    let input_len = input.len();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    for run in &cleaned.runs {
        let left_at = bytes
            .iter()
            .enumerate()
            .find(|(_, b)| **b == b'@')
            .map(|(i, _)| i);
        let legal = if let Some(at) = left_at {
            let run_before = run.end <= mapped(cleaned, at, input_len);
            let run_after = run.start > mapped(cleaned, at, input_len);
            run_before || run_after
        } else {
            false
        };
        let at_edges = run.start == 0 || run.end == input_len;
        if !legal && !at_edges {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    let mut quoted_close = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quoted_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: mapped(cleaned, 0, input_len),
            });
        }
    }

    let protected_end = quoted_close.map_or(0, |i| i + 1);
    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !(protected_end > 0 && i < protected_end) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: mapped(cleaned, ats[1], input_len),
        });
    }

    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: mapped(cleaned, at, input_len),
        });
    }

    let raw_local = &bytes[..at];
    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let mut quoted = false;

    if raw_local.first() == Some(&b'"') {
        let close = quoted_close.expect("opening quote has closing quote");
        if close + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: mapped(cleaned, close + 1, input_len),
            });
        }
        quoted = true;
        let mut i = 1;
        while i < close {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= close
                    || !matches!(bytes[i + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: mapped(cleaned, i, input_len),
                    });
                }
                local.push(bytes[i + 1]);
                local_origins.push(mapped(cleaned, i, input_len));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: mapped(cleaned, i, input_len),
                    });
                }
                local.push(b);
                local_origins.push(mapped(cleaned, i, input_len));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: mapped(cleaned, close, input_len),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        if raw_local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: mapped(cleaned, 64, input_len),
            });
        }
        let mut previous_dot = false;
        for (i, &b) in raw_local.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == raw_local.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: mapped(cleaned, i, input_len),
                    });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !b.is_ascii_alphanumeric()
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
                {
                    return Err(EmailError::LocalChar {
                        at: mapped(cleaned, i, input_len),
                    });
                }
            }
            local.push(b);
            local_origins.push(mapped(cleaned, i, input_len));
        }
    }

    let domain_start = at + 1;
    if domain_start == bytes.len() {
        return Err(EmailError::EmptyDomain {
            at: input_len,
        });
    }
    let mut domain_end = bytes.len();
    let root_position;
    if bytes[domain_end - 1] == b'.' {
        root_position = mapped(cleaned, domain_end - 1, input_len);
        domain_end -= 1;
    } else {
        root_position = input_len;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: mapped(cleaned, domain_start, input_len),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: mapped(cleaned, domain_start + 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for i in domain_start..domain_end {
        if bytes[i] == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }
    labels.push((label_start, domain_end));

    for (start, end) in &labels {
        if start == end {
            let position = if *start == domain_end && root_position != input_len {
                root_position
            } else {
                mapped(cleaned, *start, input_len)
            };
            return Err(EmailError::EmptyLabel { at: position });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: mapped(cleaned, start + 63, input_len),
            });
        }
        if bytes[*start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(cleaned, *start, input_len),
            });
        }
        if bytes[*end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(cleaned, end - 1, input_len),
            });
        }
        for i in *start..*end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: mapped(cleaned, i, input_len),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(cleaned, domain_start, input_len),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: mapped(cleaned, last_start, input_len),
        });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    let address = Address {
        local: String::from_utf8(local).expect("validated ASCII local"),
        domain: String::from_utf8(domain).expect("validated ASCII domain"),
        quoted,
    };
    Ok((address, local_origins.first().copied().unwrap_or(input_len)))
}

fn parse_with_policy(
    input: &str,
    policy: &Policy,
    allow_single_label: bool,
) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = clean_cfws(raw)?;
    for run in &cleaned.runs {
        let separator = cleaned.bytes.iter().position(|&b| b == b'@');
        let edge = run.start == 0 || run.end == input.len();
        let adjacent_at = separator.is_some_and(|at_index| {
            let at_pos = cleaned.positions[at_index];
            run.end <= at_pos || run.start > at_pos
        });
        if !edge && !adjacent_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    let (mut address, local_origin) =
        parse_cleaned(input, &cleaned, allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local = address
            .local
            .bytes()
            .map(|b| b.to_ascii_lowercase() as char)
            .collect();
    }

    let mut normalized_origin = local_origin;
    if policy.strip_subaddress {
        if let Some(plus) = address.local.find('+') {
            if plus == 0 {
                let origin = if address.quoted {
                    // Find the corresponding decoded plus origin.
                    let raw_at = cleaned.bytes.iter().position(|&b| b == b'@').unwrap();
                    let mut i = 1;
                    let mut decoded = 0usize;
                    let mut found = input.len();
                    while i < raw_at {
                        if cleaned.bytes[i] == b'\\' {
                            if decoded == 0 {
                                found = cleaned.positions[i];
                            }
                            if cleaned.bytes.get(i + 1) == Some(&b'+') {
                                found = cleaned.positions[i];
                            }
                            i += 2;
                        } else {
                            if decoded == 0 {
                                found = cleaned.positions[i];
                            }
                            if cleaned.bytes[i] == b'+' {
                                found = cleaned.positions[i];
                            }
                            i += 1;
                        }
                        decoded += 1;
                    }
                    found
                } else {
                    local_origin
                };
                return Err(EmailError::EmptyNormalizedLocal { at: origin });
            }
            address.local.truncate(plus);
        }
    }

    let valid_atom = local_atom_valid(address.local.as_bytes());
    if !valid_atom {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    let _ = normalized_origin;
    Ok(address)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), false)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, policy.allow_single_label)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| {
                        local == &address.local && domain == &address.domain
                    })
                {
                    duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((
                        address.local.clone(),
                        address.domain.clone(),
                        index,
                    ));
                    addresses.push(address);
                    source_indices.push(index);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
