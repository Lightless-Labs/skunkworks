#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::new();
    let mut positions = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected_quote = false;
    let mut quote_escaped = false;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        let b = input[i];

        if protected_quote {
            out.push(b);
            positions.push(i);
            if quote_escaped {
                quote_escaped = false;
            } else if b == b'\\' {
                quote_escaped = true;
            } else if b == b'"' {
                protected_quote = false;
            }
            i += 1;
            continue;
        }

        if comment_depth > 0 {
            if b == b'\\' {
                let next = input.get(i + 1).copied();
                if !next.is_some_and(|c| (0x20..=0x7e).contains(&c)) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let is_cfws = is_hws(b) || b == b'(' || b == b'\r' || b == b'\n';
        if is_cfws {
            if run_start.is_none() {
                run_start = Some(i);
            }
            if b == b'(' {
                comment_depth = 1;
                outer_comment = i;
                i += 1;
            } else if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else {
                i += 1;
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if b == b'"' && out.is_empty() {
            protected_quote = true;
        }

        if let Some(start) = run_start.take() {
            runs.push((start, i));
        }
        out.push(b);
        positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if let Some(start) = run_start {
        runs.push((start, input.len()));
    }

    Ok(Cleaned {
        bytes: out,
        positions,
        runs,
    })
}

fn pos(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

fn validate_unquoted(local: &[u8], clean: &Cleaned, input_len: usize) -> Result<(), EmailError> {
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong {
            at: pos(clean, 64, input_len),
        });
    }

    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot {
                    at: pos(clean, i, input_len),
                });
            }
        } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
            return Err(EmailError::LocalChar {
                at: pos(clean, i, input_len),
            });
        }
    }
    Ok(())
}

fn parse_with_policy(input: &str, policy: Option<&Policy>) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let clean = cleanup(raw)?;
    let bytes = &clean.bytes;

    for &(start, end) in &clean.runs {
        let before = clean.positions.iter().position(|&p| p >= start);
        let after = clean.positions.iter().position(|&p| p >= end);
        let left = before.and_then(|i| i.checked_sub(1)).map(|i| bytes[i]);
        let right = after.map(|i| bytes[i]);
        let at_boundary = before.is_none()
            || after.is_none()
            || left == Some(b'@')
            || right == Some(b'@');
        if !at_boundary {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        let mut closed = None;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                closed = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match closed {
            Some(end) => quote_end = Some(end),
            None => {
                return Err(EmailError::UnclosedQuote {
                    at: pos(&clean, 0, raw.len()),
                });
            }
        }
    }

    let mut ats = Vec::new();
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'@' && !quote_end.is_some_and(|end| i <= end) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(&clean, ats[1], raw.len()),
        });
    }

    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&clean, at, raw.len()),
        });
    }

    let (mut local, mut quoted, local_origins) = if let Some(end) = quote_end {
        if end + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: pos(&clean, end + 1, raw.len()),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < end {
            let b = bytes[i];
            if b == b'\\' {
                let Some(&escaped) = bytes.get(i + 1).filter(|_| i + 1 < end) else {
                    return Err(EmailError::BadEscape {
                        at: pos(&clean, i, raw.len()),
                    });
                };
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: pos(&clean, i, raw.len()),
                    });
                }
                decoded.push(escaped);
                origins.push(pos(&clean, i, raw.len()));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(&clean, i, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(pos(&clean, i, raw.len()));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&clean, end, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        (decoded, true, origins)
    } else {
        let part = &bytes[..at];
        validate_unquoted(part, &clean, raw.len())?;
        (
            part.to_vec(),
            false,
            (0..part.len()).map(|i| pos(&clean, i, raw.len())).collect(),
        )
    };

    let domain_start = at + 1;
    let raw_domain = &bytes[domain_start..];
    let mut domain_end = bytes.len();
    let root_dot = if raw_domain.last() == Some(&b'.') {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&clean, domain_start, raw.len()),
        });
    }
    let domain = &bytes[domain_start..domain_end];
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&clean, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    for &(start, end) in &labels {
        if start == end {
            let error_index = if start == domain.len() {
                root_dot.unwrap_or(domain_start + start)
            } else {
                domain_start + start
            };
            return Err(EmailError::EmptyLabel {
                at: pos(&clean, error_index, raw.len()),
            });
        }
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&clean, domain_start + start + 63, raw.len()),
            });
        }
        if domain[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&clean, domain_start + start, raw.len()),
            });
        }
        if domain[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&clean, domain_start + end - 1, raw.len()),
            });
        }
        for i in start..end {
            let b = domain[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar {
                    at: pos(&clean, domain_start + i, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !policy.is_some_and(|p| p.allow_single_label) {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&clean, domain_start, raw.len()),
        });
    }

    let &(tld_start, tld_end) = labels.last().expect("nonempty domain has a label");
    if domain[tld_start..tld_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(&clean, domain_start + tld_start, raw.len()),
        });
    }

    let mut final_local = String::from_utf8(local.clone()).expect("validated ASCII local");
    let mut final_quoted = quoted;

    if let Some(policy) = policy {
        if policy.local_case == LocalCase::Lower {
            for byte in &mut local {
                byte.make_ascii_lowercase();
            }
        }

        if policy.strip_subaddress {
            if let Some(plus) = local.iter().position(|&b| b == b'+') {
                if plus == 0 {
                    return Err(EmailError::EmptyNormalizedLocal {
                        at: local_origins.get(plus).copied().unwrap_or(pos(&clean, 0, raw.len())),
                    });
                }
                local.truncate(plus);
            }
        }

        let is_dot_atom = validate_unquoted(&local, &clean, raw.len()).is_ok();
        if !is_dot_atom {
            final_quoted = true;
        } else if policy.unquote_safe {
            final_quoted = false;
        }
        final_local = String::from_utf8(local).expect("validated ASCII local");
    }

    let final_domain = domain
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local: final_local,
        domain: final_domain,
        quoted: final_quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, None)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, Some(policy))
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((i, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[i] = Some(first_index);
                } else {
                    seen.insert(key, i);
                    source_indices.push(i);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
