#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Debug, Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<(usize, Change)>,
}

fn lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        Vec::new()
    } else {
        input[..input.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect()
    }
}

fn edit_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    #[derive(Clone)]
    enum Op {
        Keep,
        Delete,
        Insert(usize),
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut ops = Vec::new();
    while i < n && j < m {
        if old[i] == new[j] {
            ops.push(Op::Keep);
            i += 1;
            j += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins || (del == ins && (i + j) % 2 == 0) {
                ops.push(Op::Delete);
                i += 1;
            } else {
                ops.push(Op::Insert(j));
                j += 1;
            }
        }
    }
    while i < n {
        ops.push(Op::Delete);
        i += 1;
    }
    while j < m {
        ops.push(Op::Insert(j));
        j += 1;
    }

    let mut changes = Vec::new();
    let mut base_pos = 0usize;
    let mut k = 0usize;
    while k < ops.len() {
        if matches!(ops[k], Op::Keep) {
            base_pos += 1;
            k += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0usize;
        let mut replacement = Vec::new();
        while k < ops.len() && !matches!(ops[k], Op::Keep) {
            match ops[k] {
                Op::Delete => {
                    deleted += 1;
                    base_pos += 1;
                }
                Op::Insert(index) => replacement.push(new[index].clone()),
                Op::Keep => unreachable!(),
            }
            k += 1;
        }
        changes.push(Change {
            start,
            end: start + deleted,
            replacement,
        });
    }
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;
    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut out = Vec::new();
    let mut cursor = start;
    for change in changes {
        if change.start < cursor || change.end > end {
            continue;
        }
        out.extend(base[cursor..change.start].iter().cloned());
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    out.extend(base[cursor..end].iter().cloned());
    out
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_changes(&base_lines, &left_lines);
    let right_changes = edit_changes(&base_lines, &right_lines);

    let mut all_changes: Vec<(usize, Change)> = left_changes
        .iter()
        .cloned()
        .map(|c| (0, c))
        .chain(right_changes.iter().cloned().map(|c| (1, c)))
        .collect();

    let mut parent: Vec<usize> = (0..all_changes.len()).collect();
    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            let p = parent[x];
            parent[x] = parent[p];
            x = parent[x];
        }
        x
    }
    for i in 0..all_changes.len() {
        for j in i + 1..all_changes.len() {
            if interacts(&all_changes[i].1, &all_changes[j].1) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<Vec<(usize, Change)>> = Vec::new();
    let mut group_roots = Vec::new();
    for (idx, item) in all_changes.drain(..).enumerate() {
        let r = root(&mut parent, idx);
        let pos = match group_roots.iter().position(|&x| x == r) {
            Some(pos) => pos,
            None => {
                group_roots.push(r);
                groups.push(Vec::new());
                groups.len() - 1
            }
        };
        groups[pos].push(item);
    }

    let mut components: Vec<Component> = groups
        .into_iter()
        .map(|changes| {
            let start = changes.iter().map(|(_, c)| c.start).min().unwrap();
            let end = changes.iter().map(|(_, c)| c.end).max().unwrap();
            Component {
                start,
                end,
                changes,
            }
        })
        .collect();

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        append_lines(&mut output, &base_lines[base_cursor..component.start]);

        let mut l_changes = Vec::new();
        let mut r_changes = Vec::new();
        for (side, change) in component.changes {
            if side == 0 {
                l_changes.push(change);
            } else {
                r_changes.push(change);
            }
        }
        l_changes.sort_by_key(|c| (c.start, c.end));
        r_changes.sort_by_key(|c| (c.start, c.end));

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &l_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &r_changes,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_line = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("<<<<<<< LEFT {}", ordinal));
                    append_lines(&mut output, &left_projection);
                    output.push(format!("||||||| BASE {},{}", base_line, base_len));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {}", ordinal));
                    let output_len = output.len() - output_start;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    append_lines(&mut output, &base_lines[base_cursor..]);
    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
