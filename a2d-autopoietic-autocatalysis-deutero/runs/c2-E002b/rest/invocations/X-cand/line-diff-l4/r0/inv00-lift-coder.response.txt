#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Clone)]
struct Component {
    start: usize,
    end: usize,
    left: Vec<Change>,
    right: Vec<Change>,
}

fn split_lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let m = old.len();
    let n = new.len();
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut ops: Vec<(bool, String)> = Vec::new();
    let (mut i, mut j) = (0, 0);

    while i < m || j < n {
        if i == m {
            ops.push((false, new[j].clone()));
            j += 1;
        } else if j == n {
            ops.push((true, old[i].clone()));
            i += 1;
        } else if old[i] == new[j] {
            ops.push((true, old[i].clone()));
            i += 1;
            j += 1;
        } else {
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            if delete_len > insert_len
                || (delete_len == insert_len && (i + j) % 2 == 0)
            {
                ops.push((false, old[i].clone()));
                i += 1;
            } else {
                ops.push((false, new[j].clone()));
                j += 1;
            }
        }
    }

    // Keep rows need to be distinguished from inserts and deletes.
    // Rebuild with explicit operation kinds using the same canonical walk.
    let mut changes = Vec::new();
    let (mut i, mut j, mut base_pos) = (0, 0, 0);

    while i < m || j < n {
        if i < m && j < n && old[i] == new[j] {
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0;
        let mut replacement = Vec::new();

        while i < m || j < n {
            if i < m && j < n && old[i] == new[j] {
                break;
            }
            if i == m {
                replacement.push(new[j].clone());
                j += 1;
            } else if j == n {
                i += 1;
                base_pos += 1;
                deleted += 1;
            } else {
                let delete_len = lcs[i + 1][j];
                let insert_len = lcs[i][j + 1];
                if delete_len > insert_len
                    || (delete_len == insert_len && (i + j) % 2 == 0)
                {
                    i += 1;
                    base_pos += 1;
                    deleted += 1;
                } else {
                    replacement.push(new[j].clone());
                    j += 1;
                }
            }
        }

        changes.push(Change {
            start,
            end: start + deleted,
            replacement,
        });
    }

    let _ = ops;
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;

    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn make_components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let total = left.len() + right.len();
    let mut parent: Vec<usize> = (0..total).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    fn union(parent: &mut [usize], a: usize, b: usize) {
        let ra = find(parent, a);
        let rb = find(parent, b);
        if ra != rb {
            parent[rb] = ra;
        }
    }

    let all: Vec<(bool, &Change)> = left
        .iter()
        .map(|c| (true, c))
        .chain(right.iter().map(|c| (false, c)))
        .collect();

    for i in 0..total {
        for j in i + 1..total {
            if interacts(all[i].1, all[j].1) {
                union(&mut parent, i, j);
            }
        }
    }

    let mut groups: Vec<(usize, Vec<usize>)> = Vec::new();
    for i in 0..total {
        let root = find(&mut parent, i);
        if let Some((_, indices)) = groups.iter_mut().find(|(r, _)| *r == root) {
            indices.push(i);
        } else {
            groups.push((root, vec![i]));
        }
    }

    let mut components = Vec::new();
    for (_, indices) in groups {
        let mut start = usize::MAX;
        let mut end = 0;
        let mut component = Component {
            start: 0,
            end: 0,
            left: Vec::new(),
            right: Vec::new(),
        };
        for idx in indices {
            let (is_left, change) = all[idx];
            start = start.min(change.start);
            end = end.max(change.end);
            if is_left {
                component.left.push(change.clone());
            } else {
                component.right.push(change.clone());
            }
        }
        component.start = start;
        component.end = end;
        component.left.sort_by_key(|c| (c.start, c.end));
        component.right.sort_by_key(|c| (c.start, c.end));
        components.push(component);
    }

    components.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));
    components
}

fn project(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;

    for change in changes {
        if cursor < change.start {
            result.extend(base[cursor..change.start].iter().cloned());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_changes(&base_lines, &left_lines);
    let right_changes = edit_changes(&base_lines, &right_lines);
    let components = make_components(&left_changes, &right_changes);

    let mut output: Vec<String> = Vec::new();
    let mut cursor = 0;

    for component in components {
        if cursor < component.start {
            output.extend(base_lines[cursor..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.left,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.right,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = 1 + output
                        .iter()
                        .filter(|line| line.starts_with("<<<<<<< LEFT "))
                        .count();
                    let output_start = output.len();
                    let base_start_display = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());
                    output.push(format!(
                        "||||||| BASE {base_start_display},{}",
                        component.end - component.start
                    ));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    let output_len = 4
                        + left_projection.len()
                        + base_projection.len()
                        + right_projection.len();
                    // Store records after output is serialized below.
                    let _ = (output_start, output_len);
                }
            }
        }

        cursor = component.end;
    }

    if cursor < base_lines.len() {
        output.extend(base_lines[cursor..].iter().cloned());
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    let mut regions = Vec::new();
    let mut line_index = 0;
    let mut component_index = 0;
    for component in &components {
        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.left,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.right,
        );

        if left_projection != right_projection
            && left_projection != base_projection
            && right_projection != base_projection
            && resolution == Resolution::Mark
        {
            let ordinal = regions.len() + 1;
            let output_start = locate_marker(&output, ordinal, line_index);
            let output_len =
                4 + left_projection.len() + base_projection.len() + right_projection.len();
            regions.push(Conflict {
                ordinal,
                base_start: component.start,
                base_len: component.end - component.start,
                output_start,
                output_len,
            });
        }

        let added = if left_projection == right_projection {
            left_projection.len()
        } else if left_projection == base_projection {
            right_projection.len()
        } else if right_projection == base_projection {
            left_projection.len()
        } else if resolution == Resolution::PreferLeft {
            left_projection.len()
        } else if resolution == Resolution::PreferRight {
            right_projection.len()
        } else {
            4 + left_projection.len() + base_projection.len() + right_projection.len()
        };
        line_index += component.start.saturating_sub(if component_index == 0 {
            0
        } else {
            components[component_index - 1].end
        });
        line_index += added;
        component_index += 1;
    }

    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}

fn locate_marker(output: &[String], ordinal: usize, _hint: usize) -> usize {
    let needle = format!("<<<<<<< LEFT {ordinal}");
    output
        .iter()
        .position(|line| line == &needle)
        .unwrap_or(0)
}
