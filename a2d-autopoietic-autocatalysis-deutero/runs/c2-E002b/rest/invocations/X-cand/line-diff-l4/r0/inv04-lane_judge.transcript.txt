fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut ops = Vec::new();
    let (mut i, mut j) = (0, 0);
    while i < n || j < m {
        if i == n {
            ops.push(Op::Insert);
            j += 1;
        } else if j == m {
            ops.push(Op::Delete);
            i += 1;
        } else if old[i] == new[j] {
            ops.push(Op::Keep);
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            };
            if delete {
                ops.push(Op::Delete);
                i += 1;
            } else {
                ops.push(Op::Insert);
                j += 1;
            }
        }
    }

    let mut changes = Vec::new();
    let (mut old_i, mut new_i, mut k) = (0, 0, 0);
    while k < ops.len() {
        if matches!(ops[k], Op::Keep) {
            old_i += 1;
            new_i += 1;
            k += 1;
            continue;
        }

        let start = old_i;
        let mut deleted = 0;
        let mut replacement = Vec::new();
        while k < ops.len() && !matches!(ops[k], Op::Keep) {
            match ops[k] {
                Op::Delete => {
                    deleted += 1;
                    old_i += 1;
                }
                Op::Insert => {
                    replacement.push(new[new_i].clone());
                    new_i += 1;
                }
                Op::Keep => unreachable!(),
            }
            k += 1;
        }
        changes.push(Change {
            start,
            end: start + deleted,
            replacement,
        });
    }
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change], ids: &[usize]) -> Vec<String> {
    if ids.is_empty() {
        return base[start..end].to_vec();
    }

    let mut out = Vec::new();
    let mut cursor = start;
    for &id in ids {
        let change = &changes[id];
        if change.start > cursor {
            out.extend_from_slice(&base[cursor..change.start]);
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        out.extend_from_slice(&base[cursor..end]);
    }
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);
    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let mut nodes: Vec<(bool, usize)> = (0..left_changes.len())
        .map(|i| (true, i))
        .chain((0..right_changes.len()).map(|i| (false, i)))
        .collect();
    let mut parent: Vec<usize> = (0..nodes.len()).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    for a in 0..nodes.len() {
        for b in a + 1..nodes.len() {
            let (a_left, a_id) = nodes[a];
            let (b_left, b_id) = nodes[b];
            let ca = if a_left {
                &left_changes[a_id]
            } else {
                &right_changes[a_id]
            };
            let cb = if b_left {
                &left_changes[b_id]
            } else {
                &right_changes[b_id]
            };
            if interacts(ca, cb) {
                let ra = find(&mut parent, a);
                let rb = find(&mut parent, b);
                if ra != rb {
                    parent[rb] = ra;
                }
            }
        }
    }

    let mut groups: Vec<Vec<usize>> = Vec::new();
    for i in 0..nodes.len() {
        let root = find(&mut parent, i);
        if let Some(group) = groups.iter_mut().find(|g| {
            let first = g[0];
            find(&mut parent, first) == root
        }) {
            group.push(i);
        } else {
            groups.push(vec![i]);
        }
    }

    let mut components = Vec::new();
    for group in groups {
        let mut start = usize::MAX;
        let mut end = 0;
        let mut left_ids = Vec::new();
        let mut right_ids = Vec::new();
        for node in group {
            let (is_left, id) = nodes[node];
            let change = if is_left {
                &left_changes[id]
            } else {
                &right_changes[id]
            };
            start = start.min(change.start);
            end = end.max(change.end);
            if is_left {
                left_ids.push(id);
            } else {
                right_ids.push(id);
            }
        }
        left_ids.sort_unstable();
        right_ids.sort_unstable();
        components.push(Component {
            start,
            end,
            left: left_ids,
            right: right_ids,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0;

    for component in components {
        if base_cursor < component.start {
            output.extend_from_slice(&base_lines[base_cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
            &component.left,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
            &component.right,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let region_len = component.end - component.start;
                    let base_line_number = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection);
                    output.push(format!(
                        "||||||| BASE {base_line_number},{region_len}"
                    ));
                    output.extend(base_projection);
                    output.push("=======".to_owned());
                    output.extend(right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len = 4
                        + output_start
                        .checked_sub(output_start)
                        .unwrap_or(0);
                    let block_len = output.len() - output_start;
                    let _ = output_len;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len: block_len,
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        output.extend_from_slice(&base_lines[base_cursor..]);
    }

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_changes(&base_lines, &left_lines);
    let right_changes = edit_changes(&base_lines, &right_lines);

    let mut all_changes: Vec<(usize, Change)> = left_changes
        .iter()
        .cloned()
        .map(|c| (0, c))
        .chain(right_changes.iter().cloned().map(|c| (1, c)))
        .collect();

    let mut parent: Vec<usize> = (0..all_changes.len()).collect();
    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            let p = parent[x];
            parent[x] = parent[p];
            x = parent[x];
        }
        x
    }
    for i in 0..all_changes.len() {
        for j in i + 1..all_changes.len() {
            if interacts(&all_changes[i].1, &all_changes[j].1) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<Vec<(usize, Change)>> = Vec::new();
    let mut group_roots = Vec::new();
    for (idx, item) in all_changes.drain(..).enumerate() {
        let r = root(&mut parent, idx);
        let pos = match group_roots.iter().position(|&x| x == r) {
            Some(pos) => pos,
            None => {
                group_roots.push(r);
                groups.push(Vec::new());
                groups.len() - 1
            }
        };
        groups[pos].push(item);
    }

    let mut components: Vec<Component> = groups
        .into_iter()
        .map(|changes| {
            let start = changes.iter().map(|(_, c)| c.start).min().unwrap();
            let end = changes.iter().map(|(_, c)| c.end).max().unwrap();
            Component {
                start,
                end,
                changes,
            }
        })
        .collect();

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        append_lines(&mut output, &base_lines[base_cursor..component.start]);

        let mut l_changes = Vec::new();
        let mut r_changes = Vec::new();
        for (side, change) in component.changes {
            if side == 0 {
                l_changes.push(change);
            } else {
                r_changes.push(change);
            }
        }
        l_changes.sort_by_key(|c| (c.start, c.end));
        r_changes.sort_by_key(|c| (c.start, c.end));

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &l_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &r_changes,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_line = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("<<<<<<< LEFT {}", ordinal));
                    append_lines(&mut output, &left_projection);
                    output.push(format!("||||||| BASE {},{}", base_line, base_len));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {}", ordinal));
                    let output_len = output.len() - output_start;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    append_lines(&mut output, &base_lines[base_cursor..]);
    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn edit_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let (mut base_pos, mut in_change) = (0usize, false);
    let mut changes = Vec::new();
    let mut start = 0usize;
    let mut end = 0usize;
    let mut replacement = Vec::new();

    let finish_change = |changes: &mut Vec<Change>,
                         in_change: &mut bool,
                         start: usize,
                         end: usize,
                         replacement: &mut Vec<String>| {
        if *in_change {
            changes.push(Change {
                start,
                end,
                replacement: std::mem::take(replacement),
            });
            *in_change = false;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            finish_change(
                &mut changes,
                &mut in_change,
                start,
                end,
                &mut replacement,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            if !in_change {
                in_change = true;
                start = base_pos;
                end = base_pos;
            }
            replacement.push(new[j].clone());
            j += 1;
        } else if j == m {
            if !in_change {
                in_change = true;
                start = base_pos;
                end = base_pos;
            }
            i += 1;
            base_pos += 1;
            end = base_pos;
        } else {
            let delete_lcs = lcs[i + 1][j];
            let insert_lcs = lcs[i][j + 1];
            let delete = if delete_lcs > insert_lcs {
                true
            } else if insert_lcs > delete_lcs {
                false
            } else {
                (i + j) % 2 == 0
            };

            if !in_change {
                in_change = true;
                start = base_pos;
                end = base_pos;
            }

            if delete {
                i += 1;
                base_pos += 1;
                end = base_pos;
            } else {
                replacement.push(new[j].clone());
                j += 1;
            }
        }
    }

    finish_change(
        &mut changes,
        &mut in_change,
        start,
        end,
        &mut replacement,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_positive = a.start < a.end;
    let b_positive = b.start < b.end;

    match (a_positive, b_positive) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn make_components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let mut nodes: Vec<(bool, usize)> = Vec::new();
    nodes.extend((0..left.len()).map(|i| (true, i)));
    nodes.extend((0..right.len()).map(|i| (false, i)));

    let mut parent: Vec<usize> = (0..nodes.len()).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    fn union(parent: &mut [usize], a: usize, b: usize) {
        let ra = find(parent, a);
        let rb = find(parent, b);
        if ra != rb {
            parent[rb] = ra;
        }
    }

    for i in 0..nodes.len() {
        for j in i + 1..nodes.len() {
            let (left_i, idx_i) = nodes[i];
            let (left_j, idx_j) = nodes[j];
            if left_i == left_j {
                continue;
            }
            let a = if left_i { &left[idx_i] } else { &right[idx_i] };
            let b = if left_j { &left[idx_j] } else { &right[idx_j] };
            if interacts(a, b) {
                union(&mut parent, i, j);
            }
        }
    }

    let mut groups: Vec<(usize, Vec<usize>)> = Vec::new();
    for node in 0..nodes.len() {
        let root = find(&mut parent, node);
        if let Some((_, members)) = groups.iter_mut().find(|(r, _)| *r == root) {
            members.push(node);
        } else {
            groups.push((root, vec![node]));
        }
    }

    let mut components = Vec::new();
    for (_, members) in groups {
        let mut component = Component {
            start: usize::MAX,
            end: 0,
            left_changes: Vec::new(),
            right_changes: Vec::new(),
        };

        for node in members {
            let (is_left, index) = nodes[node];
            let change = if is_left {
                left[index].clone()
            } else {
                right[index].clone()
            };
            component.start = component.start.min(change.start);
            component.end = component.end.max(change.end);
            if is_left {
                component.left_changes.push(change);
            } else {
                component.right_changes.push(change);
            }
        }

        component.left_changes.sort_by_key(|c| (c.start, c.end));
        component.right_changes.sort_by_key(|c| (c.start, c.end));
        components.push(component);
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start < a.end).cmp(&(b.start < b.end)))
            .then_with(|| a.end.cmp(&b.end))
    });
    components
}

fn project(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut output = Vec::new();
    let mut pos = start;

    for change in changes {
        if change.start > pos {
            output.extend(base[pos..change.start].iter().cloned());
        }
        output.extend(change.replacement.iter().cloned());
        pos = change.end;
    }

    if pos < end {
        output.extend(base[pos..end].iter().cloned());
    }

    output
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    if !base.is_empty() && !base.ends_with('\n') {
        return Err(MergeError::MissingFinalLf {
            side: MergeSide::Base,
        });
    }
    if !left.is_empty() && !left.ends_with('\n') {
        return Err(MergeError::MissingFinalLf {
            side: MergeSide::Left,
        });
    }
    if !right.is_empty() && !right.ends_with('\n') {
        return Err(MergeError::MissingFinalLf {
            side: MergeSide::Right,
        });
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_changes(&base_lines, &left_lines);
    let right_changes = edit_changes(&base_lines, &right_lines);
    let components = make_components(&left_changes, &right_changes);

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_pos = 0usize;

    for component in components {
        if base_pos < component.start {
            output.extend(base_lines[base_pos..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.left_changes,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &component.right_changes,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());

                    let region_len = component.end - component.start;
                    let display_start = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("||||||| BASE {display_start},{region_len}"));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_pos = component.end;
    }

    if base_pos < base_lines.len() {
        output.extend(base_lines[base_pos..].iter().cloned());
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
