pub enum Error {

    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependencies in input order before other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dep) in task.deps.iter().enumerate() {
            if dep >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dep,
                });
            }
            if dep == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dep) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dep,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &cap)) in task.needs.iter().zip(capacity).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: cap,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        order.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }
    if order.len() != n {
        let removed = {
            let mut removed = vec![false; n];
            for &id in &order {
                removed[id] = true;
            }
            removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u64; resource_count];
    let mut worker_available = vec![0u64; worker_count];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count, resource_count
    ));

    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks due now, in task ID order.
        let mut due: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        due.sort_by_key(|(_, task, _)| *task);

        for (finish, task_id, worker) in due {
            if let Some(pos) = running
                .iter()
                .position(|entry| *entry == (finish, task_id, worker))
            {
                running.swap_remove(pos);
            }
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);

            for r in 0..resource_count {
                used[r] -= tasks[task_id].needs[r] as u64;
            }
            append_event(&mut trace, now, "end", task_id, worker, &used);

            for &child in &successors[task_id] {
                remaining_deps[child] -= 1;
            }
        }

        // Admit one task at a time, reevaluating after every start/completion.
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_deps[id] == 0
                        && fits(&tasks[id].needs, &used, capacity)
                })
                .collect();
            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });
            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for r in 0..resource_count {
                used[r] += tasks[task_id].needs[r] as u64;
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            append_event(&mut trace, now, "start", task_id, worker, &used);

            if finish == now {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for r in 0..resource_count {
                    used[r] -= tasks[task_id].needs[r] as u64;
                }
                append_event(&mut trace, now, "end", task_id, worker, &used);
                for &child in &successors[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the next finish, release, or worker availability event.
        let mut next = None::<u64>;
        for &(finish, _, _) in &running {
            if finish > now {
                next = Some(next.map_or(finish, |v| v.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |v| v.min(tasks[id].release)));
            }
        }

        // A worker expiry matters only if some task could otherwise be admitted
        // once its release and dependencies allow it.
        for &expiry in &worker_available {
            if expiry > now {
                let has_ready_task = (0..n).any(|id| {
                    !started[id]
                        && tasks[id].release <= expiry
                        && remaining_deps[id] == 0
                        && fits(&tasks[id].needs, &used, capacity)
                });
                if has_ready_task {
                    next = Some(next.map_or(expiry, |v| v.min(expiry)));
                }
            }
        }

        if let Some(next_time) = next {
            now = next_time;
        } else {
            // With an acyclic graph, this can only occur if a task is waiting
            // on an event at u64::MAX. Move there if it is a valid future event.
            let future = (0..n)
                .filter(|&id| !started[id] && tasks[id].release > now)
                .map(|id| tasks[id].release)
                .chain(running.iter().map(|(finish, _, _)| *finish))
                .chain(worker_available.iter().copied().filter(|&t| t > now))
                .min();
            if let Some(t) = future {
                now = t;
            } else {
                break;
            }
        }
    }

    for (id, slot) in runs.iter().enumerate() {
        if slot.is_none() {
            // This is unreachable for a valid acyclic graph under the event rules.
            let _ = id;
        }
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs
        .iter()
        .zip(used.iter().zip(capacity))
        .all(|(&need, (&in_use, &cap))| (need as u64) <= (cap as u64) - in_use)
}

fn append_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    use std::fmt::Write;

    let _ = write!(trace, "{}|{}|{}|{}|", time, kind, task, worker);
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, amount) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            let _ = write!(trace, "{}", amount);
        }
    }
    trace.push('\n');
}

                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &cap)) in task.needs.iter().zip(capacity).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: cap,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    let mut topo = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        topo.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if topo.len() != n {
        let mut removed = vec![false; n];
        for &id in &topo {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u64; resource_count];
    let mut remaining_preds: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_free_at = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut trace = format!("workers={};resources={}\n", worker_count, resource_count);
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    loop {
        // Complete all tasks ending now in ascending task ID order.
        let mut ending: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        ending.sort_by_key(|(_, task, _)| *task);

        if !ending.is_empty() {
            running.retain(|(finish, _, _)| *finish != now);
            for (finish, task_id, worker) in ending {
                for r in 0..resource_count {
                    used[r] -= tasks[task_id].needs[r] as u64;
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);

                for &child in &children[task_id] {
                    remaining_preds[child] -= 1;
                }

                append_trace_event(
                    &mut trace,
                    now,
                    "end",
                    task_id,
                    worker,
                    &used,
                    resource_count,
                );
            }
        }

        if completed_count == n {
            break;
        }

        // Start one task at a time, reevaluating after every admission.
        let mut admitted = false;
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_preds[id] == 0
                        && fits(&tasks[id].needs, &used, capacity)
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_free_at[w] <= now)
                .expect("admission requires an available worker");

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let worker_free = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_free_at[worker] = worker_free;
            for r in 0..resource_count {
                used[r] += tasks[task_id].needs[r] as u64;
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            append_trace_event(
                &mut trace,
                now,
                "start",
                task_id,
                worker,
                &used,
                resource_count,
            );

            if finish == now {
                // Immediate completion of a zero-duration task.
                for r in 0..resource_count {
                    used[r] -= tasks[task_id].needs[r] as u64;
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for &child in &children[task_id] {
                    remaining_preds[child] -= 1;
                }
                append_trace_event(
                    &mut trace,
                    now,
                    "end",
                    task_id,
                    worker,
                    &used,
                    resource_count,
                );
            } else {
                running.push((finish, task_id, worker));
            }
            admitted = true;
            break;
        }

        if admitted {
            continue;
        }

        // Advance to the next strictly later event.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |t| t.min(release)));
            }
        }
        for &free_at in &worker_free_at {
            if free_at > now {
                next_time = Some(next_time.map_or(free_at, |t| t.min(free_at)));
            }
        }

        now = next_time.expect("unfinished tasks must have a future event");
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs.iter().enumerate().all(|(r, &need)| {
        (need as u64) <= (capacity[r] as u64 - used[r])
    })
}

fn append_trace_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
    resource_count: usize,
) {
    use std::fmt::Write;

    let _ = write!(trace, "{}|{}|{}|{}|", time, kind, task, worker);
    if resource_count == 0 {
        trace.push('-');
    } else {
        for (index, amount) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            let _ = write!(trace, "{}", amount);
        }
    }
    trace.push('\n');
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let task_count = tasks.len();

    // Dependency validation takes precedence over every other check.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and indegrees without changing the input.
    let mut successors = vec![Vec::<usize>::new(); task_count];
    let mut indegree = vec![0usize; task_count];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            successors[dependency].push(task_id);
        }
    }

    // Kahn traversal using a sorted available set. Newly available smaller
    // IDs are selected before any larger IDs already waiting.
    let mut available = Vec::<usize>::new();
    for task_id in 0..task_count {
        if indegree[task_id] == 0 {
            available.push(task_id);
        }
    }

    let mut order = Vec::with_capacity(task_count);
    while !available.is_empty() {
        let task_id = available.remove(0);
        order.push(task_id);

        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let insertion = available
                    .binary_search(&child)
                    .unwrap_or_else(|position| position);
                available.insert(insertion, child);
            }
        }
    }

    if order.len() != task_count {
        let mut removed = vec![false; task_count];
        for &task_id in &order {
            removed[task_id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..task_count).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; task_count];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let resource_count = capacity.len();
    let mut used = vec![0u64; resource_count];
    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        task_count
    ];
    let mut started = vec![false; task_count];
    let mut completed = vec![false; task_count];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();

    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<(u64, usize, usize)>::new(); // finish, task, worker
    let mut trace = format!("workers={worker_count};resources={resource_count}\n");
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < task_count {
        // Complete all tasks finishing now in ascending task ID.
        running.sort_unstable_by_key(|&(finish, task_id, _)| (finish, task_id));
        let mut keep = Vec::with_capacity(running.len());
        for (finish, task_id, worker) in running.drain(..) {
            if finish == now {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for resource in 0..resource_count {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }
                append_trace_event(&mut trace, now, "end", task_id, worker, &used);
                for &child in &successors[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                keep.push((finish, task_id, worker));
            }
        }
        running = keep;

        // Start one task at a time, reconsidering eligibility and ranking
        // after each start and each immediate zero-duration completion.
        loop {
            let mut candidates: Vec<usize> = (0..task_count)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= now
                        && remaining_deps[task_id] == 0
                        && tasks[task_id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r])
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now)
                .expect("a task may start only when a worker is available");

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Run {
                start: now,
                finish,
                worker,
            };

            for resource in 0..resource_count {
                used[resource] += u64::from(tasks[task_id].needs[resource]);
            }
            append_trace_event(&mut trace, now, "start", task_id, worker, &used);

            if finish == now {
                // Immediate completion occurs before choosing another start.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for resource in 0..resource_count {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }
                append_trace_event(&mut trace, now, "end", task_id, worker, &used);
                for &child in &successors[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == task_count {
            break;
        }

        // Move to the earliest event that can change eligibility or worker
        // availability. All candidate times are strictly later than `now`.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |old| old.min(time)));
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for task_id in 0..task_count {
            if !started[task_id] && tasks[task_id].release > now {
                consider(tasks[task_id].release);
            }
        }
        for (worker, &available_at) in worker_available.iter().enumerate() {
            let worker_is_busy = running.iter().any(|&(_, _, running_worker)| running_worker == worker);
            if worker_is_busy && available_at > now {
                consider(available_at);
            }
        }

        now = next_time.expect("unfinished tasks always have a future event");
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn append_trace_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    use std::fmt::Write;

    let _ = write!(trace, "{time}|{kind}|{task}|{worker}|");
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, amount) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            let _ = write!(trace, "{amount}");
        }
    }
    trace.push('\n');
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before any other validation.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
            successors[value].push(task_id);
            indegree[task_id] += 1;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Resource validation is task-major and resource-minor.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut remaining_indegree = indegree.clone();
    let mut available = std::collections::BinaryHeap::new();
    for (id, &degree) in remaining_indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &successors[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u64; resource_count];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors = indegree;
    let mut runs: Vec<Option<Run>> = vec![None; n];

    #[derive(Clone)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut running = Vec::<Running>::new();
    let mut worker_available_at = vec![0u64; worker_count];
    let mut trace = format!("workers={worker_count};resources={resource_count}\n");
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut done_count = 0usize;

    while done_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut completing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, entry)| (entry.finish == now).then_some(index))
            .collect();
        completing.sort_by_key(|&index| running[index].task);

        for index in completing.into_iter().rev() {
            let entry = running.remove(index);
            for resource in 0..resource_count {
                used[resource] -= u64::from(tasks[entry.task].needs[resource]);
            }
            completed[entry.task] = true;
            done_count += 1;
            makespan = makespan.max(entry.finish);
            append_event(&mut trace, now, "end", entry.task, entry.worker, &used);
            for &child in &successors[entry.task] {
                remaining_predecessors[child] -= 1;
            }
        }

        // Start one task at a time, reevaluating eligibility after each start.
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_predecessors[id] == 0
                        && fits(&tasks[id].needs, &used, capacity)
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&id| worker_available_at[id] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available_at[worker] = available_at;
            for resource in 0..resource_count {
                used[resource] += u64::from(tasks[task_id].needs[resource]);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            append_event(&mut trace, now, "start", task_id, worker, &used);

            if tasks[task_id].duration == 0 {
                // Immediate completion precedes the next task selection.
                for resource in 0..resource_count {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }
                completed[task_id] = true;
                done_count += 1;
                makespan = makespan.max(finish);
                append_event(&mut trace, now, "end", task_id, worker, &used);
                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if done_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for entry in &running {
            consider(entry.finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for worker in 0..worker_count {
            if worker_available_at[worker] > now {
                consider(worker_available_at[worker]);
            }
        }

        now = next_time.expect("unfinished tasks must have a future event");
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task started"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs.iter().enumerate().all(|(resource, &need)| {
        u64::from(need) <= u64::from(capacity[resource]) - used[resource]
    })
}

fn append_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    use std::fmt::Write;

    let _ = write!(trace, "{time}|{kind}|{task}|{worker}|");
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, amount) in used.iter().enumerate() {
            if index > 0 {
                trace.push(',');
            }
            let _ = write!(trace, "{amount}");
        }
    }
    trace.push('\n');
}
