#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];
const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];
const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn append_group(n: u16, out: &mut String) {
    let mut rest = n;
    if rest >= 100 {
        out.push_str(SMALL[(rest / 100) as usize]);
        out.push_str(" hundred");
        rest %= 100;
        if rest != 0 {
            out.push_str(" and ");
        }
    }
    if rest >= 20 {
        out.push_str(TENS[(rest / 10 - 2) as usize]);
        rest %= 10;
        if rest != 0 {
            out.push('-');
            out.push_str(SMALL[rest as usize]);
        }
    } else if rest != 0 {
        out.push_str(SMALL[rest as usize]);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut have_higher = false;
    for index in (1..7).rev() {
        let n = groups[index];
        if n == 0 {
            continue;
        }
        if !out.is_empty() && !(out == "minus ") {
            out.push(' ');
        }
        append_group(n, &mut out);
        out.push(' ');
        out.push_str(SCALES[index - 1]);
        have_higher = true;
    }

    let units = groups[0];
    if units != 0 {
        if have_higher {
            if units < 100 {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        append_group(units, &mut out);
    }

    Ok(out)
}

fn dictionary_value(word: &str) -> Option<u128> {
    if word == "and" {
        return Some(0);
    }
    if word == "hundred" {
        return Some(100);
    }
    if let Some(i) = SMALL.iter().position(|&w| w == word) {
        return Some(i as u128);
    }
    if let Some(i) = TENS.iter().position(|&w| w == word) {
        return Some(((i + 2) * 10) as u128);
    }
    for (i, &scale) in SCALES.iter().enumerate() {
        if word == scale {
            return Some(1000u128.pow((i + 1) as u32));
        }
    }
    None
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, &(word, index)) in tokens.iter().enumerate() {
        if word == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index });
            }
        } else if dictionary_value(word).is_none() {
            return Err(WordsError::InvalidWord { index });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for &(word, _) in arithmetic {
        match word {
            "and" | "zero" => {}
            "hundred" => group *= 100,
            "thousand" => {
                total += group * 1_000;
                group = 0;
            }
            "million" => {
                total += group * 1_000_000;
                group = 0;
            }
            "billion" => {
                total += group * 1_000_000_000;
                group = 0;
            }
            "trillion" => {
                total += group * 1_000_000_000_000;
                group = 0;
            }
            "quadrillion" => {
                total += group * 1_000_000_000_000_000;
                group = 0;
            }
            "quintillion" => {
                total += group * LIMIT;
                group = 0;
            }
            _ => group += dictionary_value(word).unwrap_or(0),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(value)?;
    if canonical != input {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(input.as_bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }
    Ok(value)
}
