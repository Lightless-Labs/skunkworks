#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut n = magnitude;
    for group in &mut groups {
        *group = (n % 1000) as u16;
        n /= 1000;
    }

    let mut parts: Vec<String> = Vec::new();
    for index in (0..groups.len()).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        let mut part = spell_group(group);
        if index > 0 {
            part.push(' ');
            part.push_str(SCALES[index - 1]);
        }
        parts.push(part);
    }

    let units = groups[0];
    let join_with_and = units > 0
        && units < 100
        && groups[1..].iter().any(|&group| group != 0);

    let mut output = String::new();
    let last = parts.len() - 1;
    for (index, part) in parts.into_iter().enumerate() {
        if index > 0 {
            if join_with_and && index == last {
                output.push_str(" and ");
            } else {
                output.push(' ');
            }
        }
        output.push_str(&part);
    }

    if value < 0 {
        Ok(format!("minus {output}"))
    } else {
        Ok(output)
    }
}

fn spell_group(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }

    if n < 100 {
        let tens = (n / 10) as usize - 2;
        let units = n % 10;
        if units == 0 {
            TENS[tens].to_string()
        } else {
            format!("{}-{}", TENS[tens], SMALL[units as usize])
        }
    } else {
        let hundreds = n / 100;
        let remainder = n % 100;
        let mut result = format!("{} hundred", SMALL[hundreds as usize]);
        if remainder != 0 {
            result.push_str(" and ");
            result.push_str(&spell_group(remainder));
        }
        result
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (index, &(token, position)) in tokens.iter().enumerate() {
        if token == "minus" && index == 0 {
            continue;
        }
        if !is_dictionary_word(token) {
            return Err(WordsError::InvalidWord { index: position });
        }
    }

    let negative = tokens.first().is_some_and(|(token, _)| *token == "minus");
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(token, _) in arithmetic_tokens {
        match token {
            "and" => {}
            "zero" => {}
            "hundred" => group *= 100,
            "thousand" => {
                total += group * 1_000;
                group = 0;
            }
            "million" => {
                total += group * 1_000_000;
                group = 0;
            }
            "billion" => {
                total += group * 1_000_000_000;
                group = 0;
            }
            "trillion" => {
                total += group * 1_000_000_000_000;
                group = 0;
            }
            "quadrillion" => {
                total += group * 1_000_000_000_000_000;
                group = 0;
            }
            "quintillion" => {
                total += group * 1_000_000_000_000_000_000;
                group = 0;
            }
            _ => {
                group += word_value(token).expect("dictionary scan checked every token");
            }
        }

        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let mismatch = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(value)
}

fn is_dictionary_word(token: &str) -> bool {
    token == "and"
        || token == "hundred"
        || token == "thousand"
        || token == "million"
        || token == "billion"
        || token == "trillion"
        || token == "quadrillion"
        || token == "quintillion"
        || SMALL.contains(&token)
        || TENS.contains(&token)
}

fn word_value(token: &str) -> Option<u128> {
    SMALL
        .iter()
        .position(|&word| word == token)
        .filter(|&value| value > 0)
        .map(|value| value as u128)
        .or_else(|| {
            TENS
                .iter()
                .position(|&word| word == token)
                .map(|value| (value as u128 + 2) * 10)
        })
}
