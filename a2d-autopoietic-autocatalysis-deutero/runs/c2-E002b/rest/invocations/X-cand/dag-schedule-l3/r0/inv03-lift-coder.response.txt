#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(
    tasks: &[Task],
    workers: usize,
    turnaround: u64,
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before checking worker count or computing
    // anything else.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Construct successor lists and indegrees. Their order is immaterial;
    // available IDs are always selected by minimum ID.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    let mut remaining_indegree = indegree.clone();
    let mut available: Vec<usize> = (0..n)
        .filter(|&id| remaining_indegree[id] == 0)
        .collect();
    let mut topo = Vec::with_capacity(n);

    while !available.is_empty() {
        let min_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(min_pos);
        topo.push(id);

        for &child in &successors[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if topo.len() != n {
        let removed: Vec<bool> = {
            let mut flags = vec![false; n];
            for &id in &topo {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available_at = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // finish, task, worker
    let mut events = Vec::new();
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every task finishing now, in task ID order.
        let mut finishing: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for entry in running.drain(..) {
            if entry.0 == now {
                finishing.push(entry);
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;
        finishing.sort_by_key(|entry| entry.1);

        for &(finish, task_id, worker_id) in &finishing {
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
        }

        loop {
            let worker_id = match (0..workers)
                .find(|&id| worker_available_at[id] <= now)
            {
                Some(id) => id,
                None => break,
            };

            let selected = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && tasks[id].deps.iter().all(|&dep| completed[dep])
                })
                .min_by(|&a, &b| {
                    tails[b]
                        .cmp(&tails[a])
                        .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                        .then_with(|| a.cmp(&b))
                });

            let task_id = match selected {
                Some(id) => id,
                None => break,
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(
                Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                },
            )?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available_at[worker_id] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |current| current.min(release)));
            }
        }
        for &available_at in &worker_available_at {
            if available_at > now {
                next_time = Some(
                    next_time.map_or(available_at, |current| current.min(available_at)),
                );
            }
        }

        // If tasks remain, some future completion, release, or cooldown must
        // enable progress. The fallback also prevents a non-advancing loop.
        now = next_time.unwrap_or_else(|| {
            running
                .iter()
                .map(|entry| entry.0)
                .filter(|&time| time > now)
                .min()
                .unwrap_or(now)
        });
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task must start exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
