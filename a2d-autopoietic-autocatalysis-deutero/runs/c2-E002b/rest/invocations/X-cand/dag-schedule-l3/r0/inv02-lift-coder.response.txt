```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(
    tasks: &[Task],
    workers: usize,
    turnaround: u64,
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before checking anything else.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    // Kahn traversal with the smallest available task ID selected each time.
    let mut available: Vec<usize> = (0..n).filter(|&i| indegree[i] == 0).collect();
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, id)| *id)
            .map(|(pos, _)| pos)
            .unwrap();
        let task_id = available.swap_remove(smallest_pos);
        order.push(task_id);

        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let removed: Vec<bool> = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom-level tails in reverse topological order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running: Vec<(usize, usize, u64)> = Vec::new(); // task, worker, finish
    let mut events = Vec::new();
    let mut makespan = 0u64;
    let mut now = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Finish all positive-duration work completing now, ordered by task ID.
        let mut finishing: Vec<(usize, usize)> = running
            .iter()
            .filter(|&&(_, _, finish)| finish == now)
            .map(|&(task, worker, _)| (task, worker))
            .collect();
        finishing.sort_by_key(|&(task, _)| task);

        for (task, worker) in finishing {
            events.push(Event {
                time: now,
                task,
                worker,
                kind: EventKind::Finish,
            });
            completed[task] = true;
            completed_count += 1;
            makespan = makespan.max(now);
            running.retain(|&(running_task, _, _)| running_task != task);
        }

        loop {
            let worker = worker_available
                .iter()
                .enumerate()
                .filter(|&(_, &free_at)| free_at <= now)
                .map(|(id, _)| id)
                .min();

            let Some(worker) = worker else {
                break;
            };

            let mut eligible: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= now
                        && tasks[task_id]
                            .deps
                            .iter()
                            .all(|&dep| completed[dep])
                })
                .collect();

            if eligible.is_empty() {
                break;
            }

            eligible.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });
            let task_id = eligible[0];

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let worker_free = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            worker_available[worker] = worker_free;
            events.push(Event {
                time: now,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
            } else {
                running.push((task_id, worker, finish));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest event that can change eligibility or capacity.
        let mut next_time: Option<u64> = None;
        for &(_, _, finish) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for (task_id, task) in tasks.iter().enumerate() {
            if !started[task_id] && task.release > now {
                next_time = Some(next_time.map_or(task.release, |t| t.min(task.release)));
            }
        }
        for &free_at in &worker_available {
            if free_at > now {
                next_time = Some(next_time.map_or(free_at, |t| t.min(free_at)));
            }
        }

        // With a valid acyclic graph, an unfinished task always has a future
        // release, completion, or worker-availability event.
        now = next_time.expect("unfinished schedule must have a future event");
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}
```
