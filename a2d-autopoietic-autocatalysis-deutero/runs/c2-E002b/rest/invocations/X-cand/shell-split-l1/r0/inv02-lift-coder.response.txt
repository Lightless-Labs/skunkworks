#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(std::mem::take(&mut word));
                    active = false;
                }
            }
            '\'' => {
                active = true;
                let mut closed = false;
                for (_, quoted) in chars.by_ref() {
                    if quoted == '\'' {
                        closed = true;
                        break;
                    }
                    word.push(quoted);
                }
                if !closed {
                    return Err(SplitError::UnterminatedSingle { pos });
                }
            }
            '"' => {
                active = true;
                let mut closed = false;

                while let Some((inner_pos, quoted)) = chars.next() {
                    match quoted {
                        '"' => {
                            closed = true;
                            break;
                        }
                        '\\' => match chars.peek().copied() {
                            None => return Err(SplitError::TrailingEscape { pos: inner_pos }),
                            Some((_, next)) if matches!(next, '"' | '\\' | '$' | '\n') => {
                                chars.next();
                                if next != '\n' {
                                    word.push(next);
                                }
                            }
                            Some(_) => word.push('\\'),
                        },
                        _ => word.push(quoted),
                    }
                }

                if !closed {
                    return Err(SplitError::UnterminatedDouble { pos });
                }
            }
            '\\' => {
                let Some((_, next)) = chars.next() else {
                    return Err(SplitError::TrailingEscape { pos });
                };
                if next != '\n' {
                    active = true;
                    word.push(next);
                }
            }
            _ => {
                active = true;
                word.push(ch);
            }
        }
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
