#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_sqrt(n: u128) -> u128 {
    // Integer square root via binary search; the search bounds are safe for
    // all values produced by valid coordinates.
    let mut lo = 0u128;
    let mut hi = n.min(u64::MAX as u128) + 1;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }

    let lower_square = lo * lo;
    let upper = lo + 1;
    let upper_square = upper * upper;
    if n - lower_square >= upper_square - n {
        upper
    } else {
        lo
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let raw_len = points.len();
    if raw_len > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: raw_len });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if raw_len >= 2 && points[raw_len - 1] == points[0] {
        raw_len - 1
    } else {
        raw_len
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0u64;
    let mut longest_edge = None;
    let mut longest_key: Option<(u64, Point, Point, usize)> = None;

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];

        signed_area2 += (a.x as i128) * (b.y as i128)
            - (a.y as i128) * (b.x as i128);

        let dx = (b.x as i128 - a.x as i128).unsigned_abs();
        let dy = (b.y as i128 - a.y as i128).unsigned_abs();
        let d2 = dx * dx + dy * dy;
        let milli = rounded_sqrt(d2 * 1_000_000) as u64;
        edge_milli.push(milli);
        perimeter_milli += milli;

        let (low, high) = if a <= b { (a, b) } else { (b, a) };
        let key = (milli, low, high, i);
        if longest_key
            .as_ref()
            .map_or(true, |current| {
                key.0 > current.0
                    || (key.0 == current.0
                        && (key.1, key.2, key.3) < (current.1, current.2, current.3))
            })
        {
            longest_key = Some(key);
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
