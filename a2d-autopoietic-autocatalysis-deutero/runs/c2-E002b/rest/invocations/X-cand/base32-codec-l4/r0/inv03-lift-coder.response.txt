#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut all = Vec::with_capacity(self.pending.len() + input.len());
        all.extend_from_slice(&self.pending);
        all.extend_from_slice(input);

        let complete_len = (all.len() / 5) * 5;
        let mut output = String::new();
        for chunk in all[..complete_len].chunks_exact(5) {
            encode_bytes(chunk, self.config.alphabet, &mut output);
        }

        self.pending.clear();
        self.pending.extend_from_slice(&all[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let symbol_count = encode_bytes(&self.pending, self.config.alphabet, &mut output);
            if self.config.padding == Padding::Required {
                while (symbol_count + output_padding_count(&output)) % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn output_padding_count(output: &str) -> usize {
    output.bytes().rev().take_while(|&b| b == b'=').count()
}

fn encode_bytes(input: &[u8], alphabet: Alphabet, output: &mut String) -> usize {
    let chars: &[u8; 32] = match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    };

    let mut accumulator: u32 = 0;
    let mut bits = 0usize;
    let mut count = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | byte as u32;
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            output.push(chars[((accumulator >> bits) & 31) as usize] as char);
            count += 1;
        }
    }

    if bits > 0 {
        output.push(chars[((accumulator << (5 - bits)) & 31) as usize] as char);
        count += 1;
    }

    count
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    pending_bits: u32,
    bit_count: usize,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            pending_bits: 0,
            bit_count: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for byte in input.bytes() {
            let offset = next.position;
            let classified = classify(byte, next.config.alphabet)
                .ok_or(StreamError::InvalidByte { offset, byte })?;

            match classified {
                Classified::Data(value, canonical) => {
                    if next.config.canonical && !canonical {
                        return Err(StreamError::NonCanonical {
                            offset,
                            expected: byte.to_ascii_uppercase(),
                        });
                    }
                    if next.first_padding_offset.is_some() {
                        return Err(StreamError::Padding { offset });
                    }

                    next.data_count += 1;
                    next.last_data_offset = Some(offset);
                    next.pending_bits = (next.pending_bits << 5) | value as u32;
                    next.bit_count += 5;
                    while next.bit_count >= 8 {
                        next.bit_count -= 8;
                        output.push(((next.pending_bits >> next.bit_count) & 0xff) as u8);
                        next.pending_bits &= (1u32 << next.bit_count).wrapping_sub(1);
                    }
                }
                Classified::Padding => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }
                    if next.first_padding_offset.is_none() {
                        let (expected, legal) = expected_padding(next.data_count % 8);
                        if !legal || expected == 0 {
                            return Err(StreamError::Padding { offset });
                        }
                        next.first_padding_offset = Some(offset);
                        next.expected_padding = expected;
                    }
                    if next.padding_count >= next.expected_padding {
                        return Err(StreamError::Padding { offset });
                    }
                    next.padding_count += 1;
                }
            }

            next.position += 1;
        }

        self.copy_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let offset = self.position;
        let (expected, legal) = expected_padding(self.data_count % 8);
        if !legal {
            return Err(StreamError::InvalidLength { offset });
        }

        if self.config.padding == Padding::Required && self.padding_count != expected {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(offset),
            });
        }

        let unused = match self.data_count % 8 {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused > 0 && (self.pending_bits & ((1u32 << unused) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(offset),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

#[derive(Clone, Copy)]
enum Classified {
    Data(u8, bool),
    Padding,
}

impl Decoder {
    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            pending_bits: self.pending_bits,
            bit_count: self.bit_count,
            last_data_offset: self.last_data_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }

    fn copy_from(&mut self, other: Self) {
        self.position = other.position;
        self.data_count = other.data_count;
        self.pending_bits = other.pending_bits;
        self.bit_count = other.bit_count;
        self.last_data_offset = other.last_data_offset;
        self.first_padding_offset = other.first_padding_offset;
        self.padding_count = other.padding_count;
        self.expected_padding = other.expected_padding;
        self.closed = other.closed;
    }
}

fn classify(byte: u8, alphabet: Alphabet) -> Option<Classified> {
    if byte == b'=' {
        return Some(Classified::Padding);
    }

    let upper = byte.to_ascii_uppercase();
    let value = match alphabet {
        Alphabet::Standard => match upper {
            b'A'..=b'Z' => upper - b'A',
            b'2'..=b'7' => upper - b'2' + 26,
            _ => return None,
        },
        Alphabet::Hex => match upper {
            b'0'..=b'9' => upper - b'0',
            b'A'..=b'V' => upper - b'A' + 10,
            _ => return None,
        },
    };

    Some(Classified::Data(value, byte == upper))
}

fn expected_padding(residue: usize) -> (usize, bool) {
    match residue {
        0 => (0, true),
        2 => (6, true),
        4 => (4, true),
        5 => (3, true),
        7 => (1, true),
        _ => (0, false),
    }
}
