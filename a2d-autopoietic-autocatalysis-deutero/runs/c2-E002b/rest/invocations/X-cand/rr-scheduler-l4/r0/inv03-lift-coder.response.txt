```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let count = jobs.len();
    if count == 0 {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(count * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: 0,
            index,
        });
    }
    events.sort_unstable();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut current_level: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut burst_position = vec![0usize; count];
    let mut remaining_cpu: Vec<u32> = jobs.iter().map(|job| job.bursts[0]).collect();
    let mut completion = vec![0u64; count];
    let mut completed = 0usize;

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut event_pos = 0usize;
    let mut last_dispatch_id: Option<u32> = None;

    while completed < count {
        while event_pos < events.len() && events[event_pos].time <= now {
            let event = events[event_pos];
            event_pos += 1;
            if event.kind == 0 {
                queues[jobs[event.index].level].push_back(event.index);
            } else {
                queues[current_level[event.index]].push_back(event.index);
            }
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_pos >= events.len() {
                break;
            }
            let next_time = events[event_pos].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = jobs[index].id;
        if last_dispatch_id.is_some_and(|previous| previous != id) {
            now += switch_cost as u64;
        }

        while event_pos < events.len() && events[event_pos].time <= now {
            let event = events[event_pos];
            event_pos += 1;
            if event.kind == 0 {
                queues[jobs[event.index].level].push_back(event.index);
            } else {
                queues[current_level[event.index]].push_back(event.index);
            }
        }

        let consumed = remaining_cpu[index].min(quanta[level]);
        let start = now;
        now += consumed as u64;
        let end = now;
        remaining_cpu[index] -= consumed;

        let burst_finished = remaining_cpu[index] == 0;
        let job_finished = burst_finished && burst_position[index] + 1 == jobs[index].bursts.len() - 1;

        runs.push(Run {
            id,
            start,
            end,
            finished: job_finished,
            level,
        });

        while event_pos < events.len() && events[event_pos].time <= end {
            let event = events[event_pos];
            event_pos += 1;
            if event.kind == 0 {
                queues[jobs[event.index].level].push_back(event.index);
            } else {
                queues[current_level[event.index]].push_back(event.index);
            }
        }

        if !burst_finished {
            current_level[index] = (level + 1).min(quanta.len() - 1);
            queues[current_level[index]].push_back(index);
        } else if job_finished {
            completion[index] = end;
            completed += 1;
        } else {
            burst_position[index] += 2;
            remaining_cpu[index] = jobs[index].bursts[burst_position[index]];
            let io_delay = jobs[index].bursts[burst_position[index] - 1] as u64;
            current_level[index] = level.saturating_sub(1);
            events.push(Event {
                time: end + io_delay,
                kind: 1,
                index,
            });
            events[event_pos..].sort_unstable();
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(count);
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;
    for (index, job) in jobs.iter().enumerate() {
        let turnaround = completion[index] - job.arrival as u64;
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion: completion[index],
            turnaround,
            waiting,
        });
    }

    let divisor = count as u64;
    let mean_waiting_x100 = (total_waiting * 100 + divisor / 2) / divisor;
    let mean_turnaround_x100 = (total_turnaround * 100 + divisor / 2) / divisor;
    let end = completion.iter().copied().max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}
```
