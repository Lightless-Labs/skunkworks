#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct SimJob {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    level: usize,
    burst_index: usize,
    remaining_cpu: u32,
    io_dispatch_level: usize,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty()
            || job.bursts.len() > 31
            || job.bursts.len() % 2 == 0
        {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut sim_jobs: Vec<SimJob> = jobs
        .iter()
        .map(|job| SimJob {
            id: job.id,
            arrival: job.arrival as u64,
            bursts: job.bursts.clone(),
            level: job.level,
            burst_index: 0,
            remaining_cpu: job.bursts[0],
            io_dispatch_level: job.level,
        })
        .collect();

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: 0,
            index,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut next_event = 0usize;
    let mut ready: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completion = vec![0u64; jobs.len()];
    let mut completed = 0usize;
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;

    while completed < jobs.len() {
        admit_events(
            time,
            &events,
            &mut next_event,
            &mut ready,
            &mut sim_jobs,
            quanta.len(),
        );

        let selected_level = ready.iter().position(|queue| !queue.is_empty());

        if selected_level.is_none() {
            if next_event >= events.len() {
                break;
            }
            let next_time = events[next_event].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        }

        let level = selected_level.unwrap();
        let job_index = ready[level].pop_front().unwrap();
        let selected_id = sim_jobs[job_index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != selected_id) {
            time += switch_cost as u64;
        }

        admit_events(
            time,
            &events,
            &mut next_event,
            &mut ready,
            &mut sim_jobs,
            quanta.len(),
        );

        let consumed = sim_jobs[job_index]
            .remaining_cpu
            .min(quanta[level]);
        let start = time;
        time += consumed as u64;
        let end = time;
        sim_jobs[job_index].remaining_cpu -= consumed;

        let burst_finished = sim_jobs[job_index].remaining_cpu == 0;
        let final_burst = burst_finished
            && sim_jobs[job_index].burst_index + 1 == sim_jobs[job_index].bursts.len() - 1;

        runs.push(Run {
            id: selected_id,
            start,
            end,
            finished: final_burst,
            level,
        });

        // Schedule a wake before admission at the endpoint so it is ordered
        // with external arrivals by (timestamp, kind, input index).
        if burst_finished && !final_burst {
            let next_io = sim_jobs[job_index].burst_index + 1;
            let wake_time = end + sim_jobs[job_index].bursts[next_io] as u64;
            sim_jobs[job_index].io_dispatch_level = level;
            events.push(Event {
                time: wake_time,
                kind: 1,
                index: job_index,
            });
            events[next_event..].sort_by_key(|event| (event.time, event.kind, event.index));
        }

        admit_events(
            end,
            &events,
            &mut next_event,
            &mut ready,
            &mut sim_jobs,
            quanta.len(),
        );

        if final_burst {
            completion[job_index] = end;
            completed += 1;
        } else if burst_finished {
            let next_io = sim_jobs[job_index].burst_index + 1;
            sim_jobs[job_index].burst_index += 2;
            sim_jobs[job_index].remaining_cpu =
                sim_jobs[job_index].bursts[sim_jobs[job_index].burst_index];
            let _ = next_io;
        } else {
            sim_jobs[job_index].level = (level + 1).min(quanta.len() - 1);
            ready[sim_jobs[job_index].level].push_back(job_index);
        }

        last_dispatch_id = Some(selected_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let turnaround = completion[index] - job.arrival as u64;
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&length| length as u64)
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&length| length as u64)
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion: completion[index],
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}

use std::collections::VecDeque;

fn admit_events(
    through: u64,
    events: &[Event],
    next_event: &mut usize,
    ready: &mut [VecDeque<usize>],
    jobs: &mut [SimJob],
    _level_count: usize,
) {
    while *next_event < events.len() && events[*next_event].time <= through {
        let event = events[*next_event];
        *next_event += 1;

        if event.kind == 0 {
            let index = event.index;
            let level = jobs[index].level;
            ready[level].push_back(index);
        } else {
            let index = event.index;
            let level = jobs[index].io_dispatch_level.saturating_sub(1);
            jobs[index].level = level;
            ready[level].push_back(index);
        }
    }
}
