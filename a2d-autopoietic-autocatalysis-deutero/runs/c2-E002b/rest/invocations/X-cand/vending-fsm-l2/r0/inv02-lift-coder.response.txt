#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(_) = self.products.iter().find(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let Some(product) = self.products.iter().find(|p| &p.id == selected_id) else {
            return Err(Error::NoSelection);
        };
        let id = product.id.clone();
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change_counts = choose_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for i in 0..4 {
            proposed_tubes[i] -= change_counts[i];
            let deposited = self
                .escrow
                .iter()
                .filter(|coin| **coin == DENOMINATIONS[i])
                .count() as u16;
            let total = proposed_tubes[i] + deposited;
            if total > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
            proposed_tubes[i] = total;
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change_counts[i] {
                change.push(DENOMINATIONS[i]);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let mut coins = self.escrow.clone();
        coins.reverse();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(coins)
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;

    for i in (0..4).rev() {
        let coin = DENOMINATIONS[i];
        let take = tubes[i].min(remaining / coin);
        greedy[i] = take;
        remaining -= take * coin;
    }

    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<([u16; 4], u16)> = None;
    search_change(
        3,
        amount,
        &tubes,
        &mut [0; 4],
        0,
        &mut best,
    );
    best.map(|(counts, _)| counts)
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    total_coins: u16,
    best: &mut Option<([u16; 4], u16)>,
) {
    if remaining == 0 {
        let replace = match best {
            None => true,
            Some((best_counts, best_total)) => {
                total_coins < *best_total
                    || (total_coins == *best_total
                        && (0..4).rev().find_map(|i| {
                            if counts[i] != best_counts[i] {
                                Some(counts[i] > best_counts[i])
                            } else {
                                None
                            }
                        }) == Some(true))
            }
        };
        if replace {
            *best = Some((*counts, total_coins));
        }
        return;
    }
    if index < 0 {
        return;
    }
    if let Some((_, best_total)) = best {
        if total_coins >= *best_total {
            return;
        }
    }

    let i = index as usize;
    let coin = DENOMINATIONS[i];
    let max_take = tubes[i].min(remaining / coin);
    for take in (0..=max_take).rev() {
        counts[i] = take;
        search_change(
            index - 1,
            remaining - take * coin,
            tubes,
            counts,
            total_coins + take,
            best,
        );
    }
    counts[i] = 0;
}
