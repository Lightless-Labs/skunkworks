#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn validate_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let content_start = start + 1;

    if let Some(relative) = bytes[content_start..end].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let operator = match bytes.get(content_start).copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            Some(bytes[content_start] as char)
        }
        _ => None,
    };

    let variables_start = content_start + usize::from(operator.is_some());
    if variables_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut variables = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut entry_start = variables_start;

    loop {
        let entry_end = bytes[entry_start..end]
            .iter()
            .position(|&b| b == b',')
            .map_or(end, |relative| entry_start + relative);

        if entry_start == entry_end {
            return Err(Error::EmptyVariable { pos: entry_start });
        }

        let name_bytes = &bytes[entry_start..entry_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: entry_start });
        }

        if let Some(relative) = name_bytes[1..]
            .iter()
            .position(|&byte| !is_name_continue(byte))
        {
            return Err(Error::InvalidName {
                pos: entry_start + 1 + relative,
            });
        }

        let name = template[entry_start..entry_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: entry_start });
        }
        seen.push(name.clone());
        variables.push((name, entry_start));

        if entry_end == end {
            break;
        }
        entry_start = entry_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        variables,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let start = index;
                let Some(relative_close) = bytes[index + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = index + 1 + relative_close;
                expressions.push(validate_expression(template, start, end)?);
                index = end + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if hex_value(bytes[index + 1]).is_some() && hex_value(bytes[index + 2]).is_some() {
                output.push('%');
                output.push(bytes[index + 1].to_ascii_uppercase() as char);
                output.push(bytes[index + 2].to_ascii_uppercase() as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let mut fields = Vec::<String>::new();

    for (name, _) in &expression.variables {
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, matches!(operator, Some('+' | '#')));
            let field = match operator {
                Some(';' | '?' | '&') if operator == Some(';') && encoded.is_empty() => {
                    name.clone()
                }
                Some(';' | '?' | '&') => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        Some('#') => ("#", ","),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("?", "&"),
        Some('&') => ("&", "&"),
        _ => ("", ","),
    };

    format!("{prefix}{}", fields.join(separator))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        output.push_str(&expand_expression(&expression, vars));
        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
