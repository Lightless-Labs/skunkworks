#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

struct Expression {
    operator: Operator,
    variables: Vec<(String, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, start) = match bytes.first().copied() {
        Some(b'+') => (Operator::Reserved, 1),
        Some(b'#') => (Operator::Fragment, 1),
        Some(b'.') => (Operator::Dot, 1),
        Some(b'/') => (Operator::Slash, 1),
        Some(b';') => (Operator::Semicolon, 1),
        Some(b'?') => (Operator::Query, 1),
        Some(b'&') => (Operator::Continuation, 1),
        Some(b'=' | b'!' | b'@' | b'|' | b'[' | b']') => {
            return Err(Error::UnsupportedOperator { pos: content_pos });
        }
        _ => (Operator::Simple, 0),
    };

    if start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut variables = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut spec_start = start;

    for spec_end in (start..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if spec_start == spec_end {
            return Err(Error::EmptyVariable {
                pos: content_pos + spec_start,
            });
        }

        let spec = &content[spec_start..spec_end];
        let spec_bytes = spec.as_bytes();
        if !is_name_start(spec_bytes[0]) {
            return Err(Error::InvalidName {
                pos: content_pos + spec_start,
            });
        }
        for (offset, byte) in spec_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(*byte) {
                return Err(Error::InvalidName {
                    pos: content_pos + spec_start + offset,
                });
            }
        }

        if !seen.insert(spec) {
            return Err(Error::DuplicateVariable {
                pos: content_pos + spec_start,
            });
        }

        variables.push((spec.to_owned(), content_pos + spec_start));
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, variables })
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(HEX[high as usize] as char);
                out.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = matches!(
            byte,
            b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
                | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
        );

        if unreserved || (reserved && reserved_byte) {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|(binding_name, value)| (*binding_name == name).then_some(*value))
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();
    let reserved = matches!(expression.operator, Operator::Reserved | Operator::Fragment);

    for (name, _) in &expression.variables {
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, reserved);
            let field = match expression.operator {
                Operator::Semicolon if encoded.is_empty() => name.clone(),
                Operator::Semicolon | Operator::Query | Operator::Continuation => {
                    format!("{name}={encoded}")
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };

    format!("{prefix}{}", fields.join(separator))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&byte| byte == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;

                if let Some(relative_inner) = bytes[open + 1..close].iter().position(|&byte| byte == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_inner,
                    });
                }

                let content = &template[open + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (start, end, expression) in &expressions {
        output.push_str(&template[cursor..*start]);
        output.push_str(&expand_expression(expression, vars));
        cursor = *end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    variables: Vec<Variable>,
}

#[derive(Debug)]
struct Variable {
    name: String,
    start: usize,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + offset,
        });
    }

    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(
            bytes[content_start],
            b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
        )
    {
        let op = bytes[content_start];
        content_start += 1;
        Some(op)
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map_or(close, |offset| spec_start + offset);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        if !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let mut index = spec_start + 1;
        while index < spec_end && is_name_continue(bytes[index]) {
            index += 1;
        }
        if index != spec_end {
            return Err(Error::InvalidName { pos: index });
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|used| used == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        variables,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let Some(offset) = bytes[index + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: index });
                };
                let close = index + 1 + offset;
                expressions.push(parse_expression(template, index, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + value - 10
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = Vec::with_capacity(bytes.len());

    let mut index = 0;
    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%'
            && index + 2 < bytes.len()
            && let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
        {
            output.extend_from_slice(&[b'%', hex_digit(high), hex_digit(low)]);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte);
        } else {
            output.extend_from_slice(&[b'%', hex_digit(byte >> 4), hex_digit(byte & 0x0f)]);
        }
        index += 1;
    }

    String::from_utf8(output).expect("percent-encoded output is ASCII")
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let reserved = matches!(operator, Some(b'+') | Some(b'#'));
    let separator = match operator {
        Some(b'.') => b'.',
        Some(b'/') => b'/',
        Some(b';') => b';',
        Some(b'?') | Some(b'&') => b'&',
        _ => b',',
    };

    let mut fields = Vec::<String>::new();

    for variable in &expression.variables {
        let value = vars
            .iter()
            .rev()
            .find(|(name, _)| *name == variable.name)
            .map(|(_, value)| *value);

        if let Some(value) = value {
            let encoded = encode(value, reserved);
            let field = match operator {
                Some(b';') if encoded.is_empty() => variable.name.clone(),
                Some(b';') | Some(b'?') | Some(b'&') => {
                    format!("{}={}", variable.name, encoded)
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut output = String::new();
    match operator {
        Some(b'#') => output.push('#'),
        Some(b'.') => output.push('.'),
        Some(b'/') => output.push('/'),
        Some(b';') => output.push(';'),
        Some(b'?') => output.push('?'),
        Some(b'&') => output.push('&'),
        _ => {}
    }

    for (index, field) in fields.iter().enumerate() {
        if index != 0 {
            output.push(separator as char);
        }
        output.push_str(field);
    }
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        output.push_str(&expand_expression(&expression, vars));
        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },

    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

fn validate_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let content_start = start + 1;

    if let Some(relative) = bytes[content_start..end].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let operator = match bytes.get(content_start).copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            Some(bytes[content_start] as char)
        }
        _ => None,
    };

    let variables_start = content_start + usize::from(operator.is_some());
    if variables_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut variables = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut entry_start = variables_start;

    loop {
        let entry_end = bytes[entry_start..end]
            .iter()
            .position(|&b| b == b',')
            .map_or(end, |relative| entry_start + relative);

        if entry_start == entry_end {
            return Err(Error::EmptyVariable { pos: entry_start });
        }

        let name_bytes = &bytes[entry_start..entry_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: entry_start });
        }

        if let Some(relative) = name_bytes[1..]
            .iter()
            .position(|&byte| !is_name_continue(byte))
        {
            return Err(Error::InvalidName {
                pos: entry_start + 1 + relative,
            });
        }

        let name = template[entry_start..entry_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: entry_start });
        }
        seen.push(name.clone());
        variables.push((name, entry_start));

        if entry_end == end {
            break;
        }
        entry_start = entry_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        variables,
    })
}

        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if hex_value(bytes[index + 1]).is_some() && hex_value(bytes[index + 2]).is_some() {
                output.push('%');
                output.push(bytes[index + 1].to_ascii_uppercase() as char);
                output.push(bytes[index + 2].to_ascii_uppercase() as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let mut fields = Vec::<String>::new();

    for (name, _) in &expression.variables {
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, matches!(operator, Some('+' | '#')));
            let field = match operator {
                Some(';' | '?' | '&') if operator == Some(';') && encoded.is_empty() => {
                    name.clone()
                }
                Some(';' | '?' | '&') => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        Some('#') => ("#", ","),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("?", "&"),
        Some('&') => ("&", "&"),
        _ => ("", ","),
    };

    format!("{prefix}{}", fields.join(separator))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        output.push_str(&expand_expression(&expression, vars));
        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
