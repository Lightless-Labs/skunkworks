#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn append_text(word: &mut Word, text: &str, wildcards_active: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn scalar_at(input: &str, pos: usize) -> Option<(char, usize)> {
    input.get(pos..)?.chars().next().map(|c| (c, pos + c.len_utf8()))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let (c, next) = scalar_at(text, i).unwrap();
        if c == '\\' {
            if let Some((escaped, end)) = scalar_at(text, next) {
                result.push(escaped);
                i = end;
            } else {
                result.push('\\');
                i = next;
            }
        } else {
            result.push(c);
            i = next;
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let (c, next) = scalar_at(input, i).unwrap();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_text(&mut word, &input[i..next], false);
                }
                i = next;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next;
                } else if c == '\\' {
                    let slash_pos = i;
                    if let Some((following, end)) = scalar_at(input, next) {
                        if matches!(following, '"' | '\\' | '$' | '\n') {
                            if following != '\n' {
                                append_text(&mut word, &input[next..end], false);
                            }
                            i = end;
                        } else {
                            append_text(&mut word, "\\", false);
                            i = next;
                        }
                    } else {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next..].starts_with('{') {
                        let dollar = i;
                        let content_start = next + 1;
                        let mut cursor = content_start;
                        let mut close = None;
                        while cursor < input.len() {
                            let (inner, end) = scalar_at(input, cursor).unwrap();
                            if inner == '\\' {
                                if let Some((_, after)) = scalar_at(input, end) {
                                    cursor = after;
                                } else {
                                    cursor = end;
                                }
                            } else if inner == '}' {
                                close = Some(cursor);
                                break;
                            } else {
                                cursor = end;
                            }
                        }
                        let Some(close_pos) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close_pos];
                        let (name_end, name) = {
                            let mut p = 0;
                            let mut chars = contents.char_indices();
                            let Some((_, first)) = chars.next() else {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            };
                            if !is_name_start(first) {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            p += first.len_utf8();
                            while p < contents.len() {
                                let ch = contents[p..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    p += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            (p, &contents[..p])
                        };
                        let selected = if name_end == contents.len() {
                            env_value(env, name).unwrap_or("")
                        } else if contents[name_end..].starts_with(":-") {
                            let default_start = name_end + 2;
                            let value = env_value(env, name).unwrap_or("");
                            if value.is_empty() {
                                // The closing brace was found using escaped-scalar rules;
                                // decode only the selected default data.
                                let default = decode_default(&contents[default_start..]);
                                append_text(&mut word, &default, false);
                                i = close_pos + 1;
                                active = true;
                                continue;
                            } else {
                                value
                            }
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        };
                        append_text(&mut word, selected, false);
                        active = true;
                        i = close_pos + 1;
                    } else if let Some((following, end)) = scalar_at(input, next) {
                        if is_name_start(following) {
                            let mut name_end = end;
                            while name_end < input.len() {
                                let ch = input[name_end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    name_end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next..name_end];
                            append_text(&mut word, env_value(env, name).unwrap_or(""), false);
                            active = true;
                            i = name_end;
                        } else {
                            append_text(&mut word, "$", false);
                            active = true;
                            i = next;
                        }
                    } else {
                        append_text(&mut word, "$", false);
                        active = true;
                        i = next;
                    }
                } else {
                    append_text(&mut word, &input[i..next], false);
                    i = next;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next;
                } else if c == '\\' {
                    let slash_pos = i;
                    if let Some((following, end)) = scalar_at(input, next) {
                        if following != '\n' {
                            append_text(&mut word, &input[next..end], false);
                            active = true;
                        }
                        i = end;
                    } else {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next..].starts_with('{') {
                        let dollar = i;
                        let content_start = next + 1;
                        let mut cursor = content_start;
                        let mut close = None;
                        while cursor < input.len() {
                            let (inner, end) = scalar_at(input, cursor).unwrap();
                            if inner == '\\' {
                                if let Some((_, after)) = scalar_at(input, end) {
                                    cursor = after;
                                } else {
                                    cursor = end;
                                }
                            } else if inner == '}' {
                                close = Some(cursor);
                                break;
                            } else {
                                cursor = end;
                            }
                        }
                        let Some(close_pos) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close_pos];
                        let mut p = 0;
                        let first = contents.chars().next().ok_or(
                            SplitError::InvalidExpansion { pos: dollar },
                        )?;
                        if !is_name_start(first) {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        p += first.len_utf8();
                        while p < contents.len() {
                            let ch = contents[p..].chars().next().unwrap();
                            if is_name_continue(ch) {
                                p += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &contents[..p];
                        let selected = if p == contents.len() {
                            env_value(env, name).unwrap_or("")
                        } else if contents[p..].starts_with(":-") {
                            let default_start = p + 2;
                            let value = env_value(env, name).unwrap_or("");
                            if value.is_empty() {
                                let default = decode_default(&contents[default_start..]);
                                append_text(&mut word, &default, true);
                                active = true;
                                i = close_pos + 1;
                                continue;
                            } else {
                                value
                            }
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        };
                        append_text(&mut word, selected, true);
                        active = true;
                        i = close_pos + 1;
                    } else if let Some((following, end)) = scalar_at(input, next) {
                        if is_name_start(following) {
                            let mut name_end = end;
                            while name_end < input.len() {
                                let ch = input[name_end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    name_end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next..name_end];
                            append_text(&mut word, env_value(env, name).unwrap_or(""), true);
                            active = true;
                            i = name_end;
                        } else {
                            append_text(&mut word, "$", false);
                            active = true;
                            i = next;
                        }
                    } else {
                        append_text(&mut word, "$", false);
                        active = true;
                        i = next;
                    }
                } else {
                    append_text(&mut word, &input[i..next], matches!(c, '*' | '?' | '['));
                    active = true;
                    i = next;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates: Vec<(u8, String)> = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((0, candidate));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((1, candidate));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((2, candidate));
        }

        let candidate = if let Some((_, candidate)) = candidates
            .into_iter()
            .min_by_key(|(priority, candidate)| (candidate.len(), *priority))
        {
            candidate
        } else {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push('\'');
                    fallback.push('\\');
                    fallback.push('\'');
                    fallback.push('\'');
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            fallback
        };
        encoded.push(candidate);
    }
    Ok(encoded.join(" "))
}
