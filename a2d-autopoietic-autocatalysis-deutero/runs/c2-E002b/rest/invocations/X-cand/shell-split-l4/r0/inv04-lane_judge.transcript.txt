fn append_text(word: &mut Word, text: &str, glob_active: bool) {
    for c in text.chars() {
        let offset = word.text.len();

        }
    }

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%') {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((0usize, candidate));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((1usize, candidate));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((2usize, candidate));
        }

        let result = if candidates.is_empty() {
            fallback_candidate(word)
        } else {
            candidates
                .into_iter()
                .min_by_key(|(priority, candidate)| (candidate.len(), *priority))
                .unwrap()
                .1
        };
        encoded.push(result);
    }

    Ok(encoded.join(" "))
}

    CommandSubstitution { pos: usize },
}

    Nul { word: usize, pos: usize },

    None,
    Single(usize),
    Double(usize),

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn append_text(word: &mut Word, text: &str, wildcards_active: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn scalar_at(input: &str, pos: usize) -> Option<(char, usize)> {
    input.get(pos..)?.chars().next().map(|c| (c, pos + c.len_utf8()))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let (c, next) = scalar_at(text, i).unwrap();
        if c == '\\' {
            if let Some((escaped, end)) = scalar_at(text, next) {
                result.push(escaped);
                i = end;
            } else {
                result.push('\\');
                i = next;
            }
        } else {
            result.push(c);
            i = next;
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let (c, next) = scalar_at(input, i).unwrap();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_text(&mut word, &input[i..next], false);
                }
                i = next;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next;
                } else if c == '\\' {
                    let slash_pos = i;
                    if let Some((following, end)) = scalar_at(input, next) {
                        if matches!(following, '"' | '\\' | '$' | '\n') {
                            if following != '\n' {
                                append_text(&mut word, &input[next..end], false);
                            }
                            i = end;
                        } else {
                            append_text(&mut word, "\\", false);
                            i = next;
                        }
                    } else {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next..].starts_with('{') {
                        let dollar = i;
                        let content_start = next + 1;
                        let mut cursor = content_start;
                        let mut close = None;
                        while cursor < input.len() {
                            let (inner, end) = scalar_at(input, cursor).unwrap();
                            if inner == '\\' {
                                if let Some((_, after)) = scalar_at(input, end) {
                                    cursor = after;
                                } else {
                                    cursor = end;
                                }
                            } else if inner == '}' {
                                close = Some(cursor);
                                break;
                            } else {
                                cursor = end;
                            }
                        }
                        let Some(close_pos) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close_pos];
                        let (name_end, name) = {
                            let mut p = 0;
                            let mut chars = contents.char_indices();
                            let Some((_, first)) = chars.next() else {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            };
                            if !is_name_start(first) {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            p += first.len_utf8();
                            while p < contents.len() {
                                let ch = contents[p..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    p += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            (p, &contents[..p])
                        };
                        let selected = if name_end == contents.len() {
                            env_value(env, name).unwrap_or("")
                        } else if contents[name_end..].starts_with(":-") {
                            let default_start = name_end + 2;
                            let value = env_value(env, name).unwrap_or("");
                            if value.is_empty() {
                                // The closing brace was found using escaped-scalar rules;
                                // decode only the selected default data.
                                let default = decode_default(&contents[default_start..]);
                                append_text(&mut word, &default, false);
                                i = close_pos + 1;
                                active = true;
                                continue;
                            } else {
                                value
                            }
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        };
                        append_text(&mut word, selected, false);
                        active = true;
                        i = close_pos + 1;
                    } else if let Some((following, end)) = scalar_at(input, next) {
                        if is_name_start(following) {
                            let mut name_end = end;
                            while name_end < input.len() {
                                let ch = input[name_end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    name_end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next..name_end];
                            append_text(&mut word, env_value(env, name).unwrap_or(""), false);
                            active = true;
                            i = name_end;
                        } else {
                            append_text(&mut word, "$", false);
                            active = true;
                            i = next;
                        }
                    } else {
                        append_text(&mut word, "$", false);
                        active = true;
                        i = next;
                    }
                } else {
                    append_text(&mut word, &input[i..next], false);
                    i = next;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next;
                } else if c == '\\' {
                    let slash_pos = i;
                    if let Some((following, end)) = scalar_at(input, next) {
                        if following != '\n' {
                            append_text(&mut word, &input[next..end], false);
                            active = true;
                        }
                        i = end;
                    } else {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next..].starts_with('{') {
                        let dollar = i;
                        let content_start = next + 1;
                        let mut cursor = content_start;
                        let mut close = None;
                        while cursor < input.len() {
                            let (inner, end) = scalar_at(input, cursor).unwrap();
                            if inner == '\\' {
                                if let Some((_, after)) = scalar_at(input, end) {
                                    cursor = after;
                                } else {
                                    cursor = end;
                                }
                            } else if inner == '}' {
                                close = Some(cursor);
                                break;
                            } else {
                                cursor = end;
                            }
                        }
                        let Some(close_pos) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close_pos];
                        let mut p = 0;
                        let first = contents.chars().next().ok_or(
                            SplitError::InvalidExpansion { pos: dollar },
                        )?;
                        if !is_name_start(first) {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        p += first.len_utf8();
                        while p < contents.len() {
                            let ch = contents[p..].chars().next().unwrap();
                            if is_name_continue(ch) {
                                p += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &contents[..p];
                        let selected = if p == contents.len() {
                            env_value(env, name).unwrap_or("")
                        } else if contents[p..].starts_with(":-") {
                            let default_start = p + 2;
                            let value = env_value(env, name).unwrap_or("");
                            if value.is_empty() {
                                let default = decode_default(&contents[default_start..]);
                                append_text(&mut word, &default, true);
                                active = true;
                                i = close_pos + 1;
                                continue;
                            } else {
                                value
                            }
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        };
                        append_text(&mut word, selected, true);
                        active = true;
                        i = close_pos + 1;
                    } else if let Some((following, end)) = scalar_at(input, next) {
                        if is_name_start(following) {
                            let mut name_end = end;
                            while name_end < input.len() {
                                let ch = input[name_end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    name_end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next..name_end];
                            append_text(&mut word, env_value(env, name).unwrap_or(""), true);
                            active = true;
                            i = name_end;
                        } else {
                            append_text(&mut word, "$", false);
                            active = true;
                            i = next;
                        }
                    } else {
                        append_text(&mut word, "$", false);
                        active = true;
                        i = next;
                    }
                } else {
                    append_text(&mut word, &input[i..next], matches!(c, '*' | '?' | '['));
                    active = true;
                    i = next;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates: Vec<(u8, String)> = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((0, candidate));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((1, candidate));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((2, candidate));
        }

        let candidate = if let Some((_, candidate)) = candidates
            .into_iter()
            .min_by_key(|(priority, candidate)| (candidate.len(), *priority))
        {
            candidate
        } else {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push('\'');
                    fallback.push('\\');
                    fallback.push('\'');
                    fallback.push('\'');
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            fallback
        };
        encoded.push(candidate);
    }
    Ok(encoded.join(" "))
}

    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))

fn find_braced_end(input: &str, start: usize) -> Option<usize> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next()?;
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next()?;
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        i += c.len_utf8();
        if c == '\\' && i < text.len() {
            let next = text[i..].chars().next().unwrap();
            i += next.len_utf8();
            out.push(next);
        } else {
            out.push(c);
        }
    }
    out
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if input[after_dollar..].starts_with('{') {
        let content_start = after_dollar + 1;
        let end = find_braced_end(input, content_start)
            .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..end];

        let mut name_end = 0;
        for c in contents.chars() {
            if name_end == 0 {
                if !is_name_start(c) {
                    return Err(SplitError::InvalidExpansion { pos: dollar });
                }
            } else if !is_name_continue(c) {
                break;
            }
            name_end += c.len_utf8();
        }
        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let rest = &contents[name_end..];
        let value = if rest.is_empty() {
            lookup(env, name).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            match lookup(env, name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((end + 1, value))
    } else {
        let mut i = after_dollar;
        let first = input[i..].chars().next().unwrap();
        if !is_name_start(first) {
            return Ok((after_dollar, "$".to_string()));
        }
        i += first.len_utf8();
        while i < input.len() {
            let c = input[i..].chars().next().unwrap();
            if !is_name_continue(c) {
                break;
            }
            i += c.len_utf8();
        }
        Ok((i, lookup(env, &input[after_dollar..i]).unwrap_or("").to_string()))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::Bare;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::Bare;
                } else {
                    append(&mut current, &input[i..i + n], false);
                }
                i += n;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::Bare;
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let next_len = next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append(&mut current, &input[i..i + next_len], false);
                        }
                        i += next_len;
                    } else {
                        append(&mut current, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_pos = i + n;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_pos..].starts_with('{')
                        || input[next_pos..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, i, env)?;
                        append(&mut current, &value, false);
                        active = true;
                        i = end;
                    } else {
                        append(&mut current, "$", false);
                        i += n;
                    }
                } else {
                    append(&mut current, &input[i..i + n], false);
                    i += n;
                }
            }
            Quote::Bare => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' || c == '"' {
                    active = true;
                    quote_start = i;
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let next_len = next.len_utf8();
                    if next == '\n' {
                        i += next_len;
                    } else {
                        append(&mut current, &input[i..i + next_len], false);
                        active = true;
                        i += next_len;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_pos = i + n;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_pos..].starts_with('{')
                        || input[next_pos..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, i, env)?;
                        append(&mut current, &value, true);
                        active = true;
                        i = end;
                    } else {
                        append(&mut current, "$", true);
                        active = true;
                        i += n;
                    }
                } else {
                    append(&mut current, &input[i..i + n], true);
                    active = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::Bare => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

            out.push('\\');
            out.push(c);

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        let mut best: Option<(usize, String)> = None;
        for candidate in candidates.into_iter().flatten() {
            let length = candidate.len();
            if best.as_ref().is_none_or(|(best_len, _)| length < *best_len) {
                best = Some((length, candidate));
            }
        }

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()

    is_name_start(c) || c.is_ascii_digit()

fn env_value<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {

fn append_text(word: &mut Word, text: &str, glob_active: bool) {
    for c in text.chars() {
        if glob_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn find_braced_end(input: &str, start: usize) -> Option<usize> {
    let mut it = input[start..].char_indices().peekable();
    while let Some((offset, c)) = it.next() {
        if c == '\\' {
            it.next();
        } else if c == '}' {
            return Some(start + offset);
        }
    }
    None
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let len = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    current.text.push(c);
                }
                i += len;
            }
            Quote::Double(_) => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        i += len;
                    }
                    '\\' => {
                        let esc_pos = i;
                        i += 1;
                        if i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: esc_pos });
                        }
                        let next = input[i..].chars().next().unwrap();
                        let next_len = next.len_utf8();
                        if matches!(next, '"' | '\\' | '$' | '\n') {
                            if next != '\n' {
                                current.text.push(next);
                            }
                            i += next_len;
                        } else {
                            current.text.push('\\');
                        }
                    }
                    '$' => {
                        let dollar = i;
                        let after = i + 1;
                        if after < input.len() {
                            let next = input[after..].chars().next().unwrap();
                            if next == '(' {
                                return Err(SplitError::CommandSubstitution { pos: dollar });
                            }
                            if next == '{' {
                                let content_start = after + 1;
                                let Some(close) = find_braced_end(input, content_start) else {
                                    return Err(SplitError::UnterminatedExpansion { pos: dollar });
                                };
                                let content = &input[content_start..close];
                                let (name, default) = parse_braced_content(content)
                                    .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                                active = true;
                                if let Some(default_text) = default {
                                    match env_value(name, env) {
                                        Some(value) if !value.is_empty() => {
                                            append_text(&mut current, value, false);
                                        }
                                        _ => {
                                            let decoded = decode_default(default_text);
                                            append_text(&mut current, &decoded, false);
                                        }
                                    }
                                } else if let Some(value) = env_value(name, env) {
                                    append_text(&mut current, value, false);
                                }
                                i = close + 1;
                            } else if is_name_start(next) {
                                let name_start = after;
                                let mut end = after + next.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if !is_name_char(ch) {
                                        break;
                                    }
                                    end += ch.len_utf8();
                                }
                                active = true;
                                let name = &input[name_start..end];
                                if let Some(value) = env_value(name, env) {
                                    append_text(&mut current, value, false);
                                }
                                i = end;
                            } else {
                                current.text.push('$');
                                i += 1;
                            }
                        } else {
                            current.text.push('$');
                            i += 1;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                    _ => {
                        current.text.push(c);
                        i += len;
                    }
                }
            }
            Quote::None => {
                match c {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(current);
                            current = Word {
                                text: String::new(),
                                glob: Vec::new(),
                            };
                            active = false;
                        }
                        i += len;
                    }
                    '\'' => {
                        active = true;
                        quote = Quote::Single(i);
                        i += len;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double(i);
                        i += len;
                    }
                    '\\' => {
                        let esc_pos = i;
                        i += 1;
                        if i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: esc_pos });
                        }
                        let next = input[i..].chars().next().unwrap();
                        let next_len = next.len_utf8();
                        if next == '\n' {
                            i += next_len;
                        } else {
                            active = true;
                            current.text.push(next);
                            i += next_len;
                        }
                    }
                    '$' => {
                        let dollar = i;
                        let after = i + 1;
                        if after < input.len() {
                            let next = input[after..].chars().next().unwrap();
                            if next == '(' {
                                return Err(SplitError::CommandSubstitution { pos: dollar });
                            }
                            if next == '{' {
                                let content_start = after + 1;

                                    return Err(SplitError::UnterminatedExpansion { pos: dollar });
                                };
                                let content = &input[content_start..close];
                                let (name, default) = parse_braced_content(content)
                                    .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                                active = true;
                                if let Some(default_text) = default {
                                    match env_value(name, env) {
                                        Some(value) if !value.is_empty() => {
                                            append_text(&mut current, value, true);
                                        }
                                        _ => {
                                            let decoded = decode_default(default_text);
                                            append_text(&mut current, &decoded, true);
                                        }
                                    }
                                } else if let Some(value) = env_value(name, env) {
                                    append_text(&mut current, value, true);
                                }
                                i = close + 1;
                            } else if is_name_start(next) {
                                let name_start = after;
                                let mut end = after + next.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if !is_name_char(ch) {
                                        break;
                                    }
                                    end += ch.len_utf8();
                                }
                                active = true;
                                let name = &input[name_start..end];
                                if let Some(value) = env_value(name, env) {
                                    append_text(&mut current, value, true);
                                }
                                i = end;
                            } else {
                                active = true;
                                current.text.push('$');
                                i += 1;
                            }
                        } else {
                            active = true;
                            current.text.push('$');
                            i += 1;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                    _ => {
                        active = true;
                        if matches!(c, '*' | '?' | '[') {
                            current.glob.push(current.text.len());
                        }
                        current.text.push(c);
                        i += len;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn parse_braced_content(content: &str) -> Option<(&str, Option<&str>)> {
    let mut name_end = 0;
    for (idx, c) in content.char_indices() {
        if (idx == 0 && is_name_start(c)) || (idx > 0 && is_name_char(c)) {
            name_end = idx + c.len_utf8();
        } else {
            break;
        }
    }
    if name_end == 0 {
        return None;
    }

    let name = &content[..name_end];
    let rest = &content[name_end..];
    if rest.is_empty() {
        Some((name, None))
    } else if let Some(default) = rest.strip_prefix(":-") {
        Some((name, Some(default)))
    } else {
        None
    }
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut encoded = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%') {
            encoded.push(c);
        } else {
            encoded.push('\\');
            encoded.push(c);
        }
    }
    Some(encoded)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut encoded = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            encoded.push('\\');
        }
        encoded.push(c);
    }
    encoded.push('"');
    Some(encoded)
}

fn fallback_candidate(word: &str) -> String {
    let mut encoded = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            encoded.push_str("'\\''");
        } else {
            encoded.push(c);
        }
    }
    encoded.push('\'');
    encoded
}

fn encode_word(word: &str) -> String {
    let candidates = [
        bare_candidate(word).map(|s| (0usize, s)),
        single_candidate(word).map(|s| (1usize, s)),
        double_candidate(word).map(|s| (2usize, s)),
    ];

    candidates
        .into_iter()
        .flatten()
        .min_by_key(|(priority, value)| (value.len(), *priority))
        .map(|(_, value)| value)
        .unwrap_or_else(|| fallback_candidate(word))
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    Ok(words
        .iter()
        .map(|word| encode_word(word))
        .collect::<Vec<_>>()
        .join(" "))
}
