const MAX_COORD: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-MAX_COORD..=MAX_COORD).contains(&p.x)

}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.first() == points.last() {
        points.len() - 1
    } else {
        points.len()
    };
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    Ok(points[..effective_len].to_vec())

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let squared = (dx * dx + dy * dy) as u128 * 1_000_000u128;

    // Integer square root, followed by nearest-integer rounding. The square
    // root's fractional part is at least one half exactly when 4*N >= (2*r+1)^2.
    let mut low = 0u128;
    let mut high = 2_828_427_125u128;
    while low < high {
        let mid = low + (high - low + 1) / 2;
        if mid <= squared / mid.max(1) {
            low = mid;
        } else {
            high = mid - 1;
        }
    }
    let root = low;
    let threshold = 2 * root + 1;
    let rounded = if 4 * squared >= threshold * threshold {
        root + 1
    } else {
        root
    };
    rounded as u64
}

fn perimeter_milli(points: &[Point]) -> u64 {
    if points.len() < 2 {
        return 0;
    }
    (0..points.len())
        .map(|i| rounded_edge_milli(points[i], points[(i + 1) % points.len()]))
        .sum()
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed = signed_area2(&polygon);
    let mut edge_milli = Vec::with_capacity(polygon.len());
    for i in 0..polygon.len() {
        edge_milli.push(rounded_edge_milli(
            polygon[i],
            polygon[(i + 1) % polygon.len()],
        ));
    }
    let perimeter_milli = edge_milli.iter().copied().sum();

    let longest_edge = (0..polygon.len()).max_by(|&i, &j| {
        edge_milli[i]
            .cmp(&edge_milli[j])
            .then_with(|| {
                let mut pi = (polygon[i], polygon[(i + 1) % polygon.len()]);
                let mut pj = (polygon[j], polygon[(j + 1) % polygon.len()]);
                if pi.1 < pi.0 {
                    pi = (pi.1, pi.0);
                }
                if pj.1 < pj.0 {
                    pj = (pj.1, pj.0);
                }
                pj.cmp(&pi)
            })
            .then_with(|| j.cmp(&i))
    });

    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: match signed.cmp(&0) {
            std::cmp::Ordering::Less => Orientation::Clockwise,
            std::cmp::Ordering::Equal => Orientation::Degenerate,
            std::cmp::Ordering::Greater => Orientation::CounterClockwise,
        },
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn classify_one(polygon: &[Point], q: Point) -> Location {
    for i in 0..polygon.len() {
        if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
            return Location::Boundary { edge: i };
        }
    }

    let mut inside = false;
    for i in 0..polygon.len() {
        let a = polygon[i];
        let b = polygon[(i + 1) % polygon.len()];
        if (a.y > q.y) != (b.y > q.y) {
            let side = cross(a, b, q);
            if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                inside = !inside;
            }
        }
    }
    if inside {
        Location::Inside
    } else {
        Location::Outside
    }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }
    Ok(queries
        .iter()
        .copied()
        .map(|query| classify_one(&polygon, query))
        .collect())
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique = Vec::<(Point, usize)>::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = unique.iter_mut().find(|entry| entry.0 == point) {
            existing.1 = existing.1.min(index);
        } else {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|entry| entry.0);

    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|entry| entry.0).collect();
        let source_indices: Vec<usize> = unique.iter().map(|entry| entry.1).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|entry| cross(unique[0].0, unique[1].0, entry.0) == 0);

    let hull_indices = if all_collinear {
        if mode == Collinear::Include {
            (0..unique.len()).collect()
        } else {
            vec![0, unique.len() - 1]
        }
    } else {
        let mut lower = Vec::<usize>::new();
        for i in 0..unique.len() {
            while lower.len() >= 2 {
                let n = lower.len();
                let turn = cross(unique[lower[n - 2]].0, unique[lower[n - 1]].0, unique[i].0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                lower.pop();
            }
            lower.push(i);
        }

        let mut upper = Vec::<usize>::new();
        for i in (0..unique.len()).rev() {
            while upper.len() >= 2 {
                let n = upper.len();
                let turn = cross(unique[upper[n - 2]].0, unique[upper[n - 1]].0, unique[i].0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                upper.pop();
            }
            upper.push(i);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_indices.iter().map(|&i| unique[i].0).collect();
    let source_indices: Vec<usize> = hull_indices.iter().map(|&i| unique[i].1).collect();
    let area = if vertices.len() >= 3 {
        signed_area2(&vertices).unsigned_abs()
    } else {
        0
    };
    let perimeter = perimeter_milli(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli: perimeter,
    })
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {

        return Err(GeomError::TooManyVertices { len: points.len() });
    }

        }
    }

    }

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_sqrt(n: u128) -> u64 {
    let mut lo = 0u128;
    let mut hi = 1u128;
    while hi * hi < n {
        hi *= 2;
    }
    while lo + 1 < hi {
        let mid = (lo + hi) / 2;
        if mid * mid < n {
            lo = mid;
        } else {
            hi = mid;
        }
    }

    // Compare n against the square of the half-integer between lo and hi:
    // (lo + 0.5)^2, represented after multiplying by four.
    if 4 * n >= (2 * lo + 1) * (2 * lo + 1) {
        (lo + 1) as u64
    } else {
        lo as u64
    }
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    rounded_sqrt(squared_distance(a, b) * 1_000_000)
}

fn perimeter_and_longest(points: &[Point]) -> (Vec<u64>, u64, Option<usize>) {
    if points.is_empty() {
        return (Vec::new(), 0, None);
    }

    let n = points.len();
    let mut lengths = Vec::with_capacity(n);
    let mut perimeter = 0u64;
    let mut longest: Option<usize> = None;

    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        let length = edge_length_milli(a, b);
        lengths.push(length);
        perimeter += length;

        let replace = match longest {
            None => true,
            Some(best) if length > lengths[best] => true,
            Some(best) if length < lengths[best] => false,
            Some(best) => {
                let (low, high) = if a <= b { (a, b) } else { (b, a) };
                let ba = points[best];
                let bb = points[(best + 1) % n];
                let (best_low, best_high) = if ba <= bb { (ba, bb) } else { (bb, ba) };
                (low, high, i) < (best_low, best_high, best)
            }
        };
        if replace {
            longest = Some(i);
        }
    }

    (lengths, perimeter, longest)
}

        area2: signed.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if on_segment(a, b, query) {
                boundary = Some(i);
                break;
            }
            if (a.y > query.y) != (b.y > query.y) {
                let determinant = cross(a, b, query);
                if (b.y > a.y && determinant > 0) || (b.y < a.y && determinant < 0) {
                    inside = !inside;
                }
            }
        }

        result.push(match boundary {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }
    Ok(result)
}

fn hull_cross(
    points: &[(Point, usize)],
    a: usize,
    b: usize,
    c: usize,
) -> i128 {
    cross(points[a].0, points[b].0, points[c].0)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut sorted: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = sorted.iter_mut().find(|entry| entry.0 == point) {
            existing.1 = existing.1.min(index);
        } else {
            sorted.push((point, index));
        }
    }
    sorted.sort_by_key(|entry| entry.0);

    if sorted.len() <= 1 {
        let vertices: Vec<Point> = sorted.iter().map(|entry| entry.0).collect();
        let source_indices: Vec<usize> = sorted.iter().map(|entry| entry.1).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let mut all_collinear = true;
    for i in 2..sorted.len() {
        if hull_cross(&sorted, 0, 1, i) != 0 {
            all_collinear = false;
            break;
        }
    }

    let hull_indices: Vec<usize> = if all_collinear {
        match mode {
            Collinear::Include => (0..sorted.len()).collect(),
            Collinear::Exclude => vec![0, sorted.len() - 1],
        }
    } else {
        let mut lower: Vec<usize> = Vec::new();
        for i in 0..sorted.len() {
            while lower.len() >= 2 {
                let n = lower.len();
                let turn = hull_cross(&sorted, lower[n - 2], lower[n - 1], i);
                let remove = match mode {
                    Collinear::Exclude => turn <= 0,
                    Collinear::Include => turn < 0,
                };
                if remove {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push(i);
        }

        let mut upper: Vec<usize> = Vec::new();
        for i in (0..sorted.len()).rev() {
            while upper.len() >= 2 {
                let n = upper.len();
                let turn = hull_cross(&sorted, upper[n - 2], upper[n - 1], i);
                let remove = match mode {
                    Collinear::Exclude => turn <= 0,
                    Collinear::Include => turn < 0,
                };
                if remove {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push(i);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_indices.iter().map(|&i| sorted[i].0).collect();
    let source_indices: Vec<usize> = hull_indices.iter().map(|&i| sorted[i].1).collect();
    let signed = signed_area2(&vertices);
    let area2 = signed.unsigned_abs();
    let (_, perimeter_milli, _) = perimeter_and_longest(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (q.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (q.x as i128 - a.x as i128)

fn shoelace(points: &[Point]) -> i128 {
    if points.is_empty() {
        return 0;

    (0..points.len())
        .map(|i| cross_points(points[i], points[(i + 1) % points.len()]))
        .sum()
}

fn integer_sqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128 - a.x as i128).unsigned_abs();
    let dy = (b.y as i128 - a.y as i128).unsigned_abs();
    let squared = dx * dx + dy * dy;
    let scaled = squared * 1_000_000;
    let floor = integer_sqrt(scaled);
    // Compare against the exact square of the half-integer threshold:
    // (floor + 1/2)^2 = floor^2 + floor + 1/4.
    let floor_squared = floor * floor;
    let remainder = scaled - floor_squared;
    let rounded = if remainder * 4 >= floor * 4 + 1 {
        floor + 1
    } else {
        floor
    };
    rounded as u64
}

        (b, a)

fn polygon_measures(points: &[Point]) -> Measures {
    let signed_area2 = shoelace(points);
    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..points.len() {
        let length = rounded_edge_milli(points[i], points[(i + 1) % points.len()]);
        edge_milli.push(length);
        perimeter_milli += length;

        longest_edge = match longest_edge {
            None => Some(i),
            Some(current) if length > edge_milli[current] => Some(i),
            Some(current) if length == edge_milli[current] => {
                let pair = endpoint_pair(points[i], points[(i + 1) % points.len()]);
                let current_pair = endpoint_pair(
                    points[current],
                    points[(current + 1) % points.len()],
                );
                if pair < current_pair || (pair == current_pair && i < current) {
                    Some(i)
                } else {
                    Some(current)
                }
            }
            Some(current) => Some(current),
        };
    }

    Measures {
        signed_area2,
        area2: abs_i128(signed_area2),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(existing, _)| existing == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    let mut vertices: Vec<(Point, usize)> = if unique.len() <= 1 {
        unique
    } else {
        let all_collinear = unique
            .iter()
            .skip(2)
            .all(|&(point, _)| cross(unique[0].0, unique[1].0, point) == 0);

        if all_collinear {
            match mode {
                Collinear::Include => unique,
                Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
            }
        } else {
            let mut lower: Vec<(Point, usize)> = Vec::new();
            for &item in &unique {
                while lower.len() >= 2 {
                    let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, item.0);
                    let remove = match mode {
                        Collinear::Exclude => turn <= 0,
                        Collinear::Include => turn < 0,
                    };
                    if remove {
                        lower.pop();
                    } else {
                        break;
                    }
                }
                lower.push(item);
            }

            let mut upper: Vec<(Point, usize)> = Vec::new();
            for &item in unique.iter().rev() {
                while upper.len() >= 2 {
                    let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, item.0);
                    let remove = match mode {
                        Collinear::Exclude => turn <= 0,
                        Collinear::Include => turn < 0,
                    };
                    if remove {
                        upper.pop();
                    } else {
                        break;
                    }
                }
                upper.push(item);
            }

            lower.pop();
            upper.pop();
            lower.extend(upper);
            lower
        }
    };

    // Monotone-chain construction starts at the lexicographic minimum. The
    // result has no repeated terminal point.
    let hull_points: Vec<Point> = vertices.iter().map(|&(point, _)| point).collect();
    let source_indices: Vec<usize> = vertices.iter().map(|&(_, index)| index).collect();
    let signed = shoelace(&hull_points);

    Ok(Hull {
        vertices: hull_points,
        source_indices,
        area2: abs_i128(signed),
        perimeter_milli: hull_perimeter(
            &vertices.iter().map(|&(point, _)| point).collect::<Vec<_>>(),
        ),
    })
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_sqrt(n: u128) -> u128 {
    if n == 0 {
        return 0;
    }

    let mut low = 0u128;
    let mut high = 1u128;
    while high <= n / high {
        high *= 2;
    }

    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid <= n / mid {
            low = mid;
        } else {
            high = mid;
        }
    }

    // Compare n with (low + 1/2)^2 without fractions:
    // 4n >= (2low + 1)^2 rounds upward, including exact ties.
    let threshold = 2 * low + 1;
    if 4 * n >= threshold * threshold {
        low + 1
    } else {
        low
    }
}

fn edge_mill(a: Point, b: Point) -> u64 {
    rounded_sqrt(squared_distance(a, b) * 1_000_000) as u64
}

fn polygon_signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum

fn polygon_perimeter(points: &[Point]) -> (Vec<u64>, u64, Option<usize>) {
    let n = points.len();
    let mut lengths = Vec::with_capacity(n);
    let mut perimeter = 0u64;
    let mut longest: Option<usize> = None;

    for i in 0..n {
        let length = edge_mill(points[i], points[(i + 1) % n]);
        lengths.push(length);
        perimeter += length;

        let should_replace = match longest {
            None => true,
            Some(best) => {
                if length > lengths[best] {
                    true
                } else if length < lengths[best] {
                    false
                } else {
                    let mut edge = (points[i], points[(i + 1) % n]);
                    if edge.1 < edge.0 {
                        edge = (edge.1, edge.0);
                    }
                    let mut best_edge = (points[best], points[(best + 1) % n]);
                    if best_edge.1 < best_edge.0 {
                        best_edge = (best_edge.1, best_edge.0);
                    }
                    edge < best_edge || (edge == best_edge && i < best)
                }
            }
        };
        if should_replace {
            longest = Some(i);
        }
    }

    (lengths, perimeter, longest)
}

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(existing, _)| existing == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    let distinct: Vec<Point> = unique.iter().map(|&(point, _)| point).collect();
    let (vertices, indices) = if distinct.len() <= 1 {
        (distinct, unique.iter().map(|&(_, index)| index).collect())
    } else if all_collinear(&distinct) {
        if mode == Collinear::Include {
            (
                distinct,
                unique.iter().map(|&(_, index)| index).collect(),
            )
        } else {
            let chosen = if distinct.len() == 2 {
                vec![0, 1]
            } else {
                vec![0, distinct.len() - 1]
            };
            (
                chosen.iter().map(|&i| distinct[i]).collect(),
                chosen.iter().map(|&i| unique[i].1).collect(),
            )
        }
    } else {
        let mut lower: Vec<usize> = Vec::new();
        for i in 0..distinct.len() {
            while lower.len() >= 2 {
                let n = lower.len();
                let turn = cross(distinct[lower[n - 2]], distinct[lower[n - 1]], distinct[i]);
                let remove = match mode {
                    Collinear::Exclude => turn <= 0,
                    Collinear::Include => turn < 0,
                };
                if remove {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push(i);
        }

        let mut upper: Vec<usize> = Vec::new();
        for i in (0..distinct.len()).rev() {
            while upper.len() >= 2 {
                let n = upper.len();
                let turn = cross(distinct[upper[n - 2]], distinct[upper[n - 1]], distinct[i]);
                let remove = match mode {
                    Collinear::Exclude => turn <= 0,
                    Collinear::Include => turn < 0,
                };
                if remove {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push(i);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);

        // The monotone chain is already counterclockwise and begins at the
        // lexicographically smallest point.
        let vertices: Vec<Point> = lower.iter().map(|&i| distinct[i]).collect();
        let indices: Vec<usize> = lower.iter().map(|&i| unique[i].1).collect();
        (vertices, indices)
    };

    let area2 = hull_area2(&vertices);
    let (_, perimeter_milli, _) = polygon_perimeter(&vertices);

    Ok(Hull {
        vertices,
        source_indices: indices,
        area2,
        perimeter_milli,
    })
}
