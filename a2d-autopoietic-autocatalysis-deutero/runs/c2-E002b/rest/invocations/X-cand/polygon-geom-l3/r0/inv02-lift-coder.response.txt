#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const MAX_COORD: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-MAX_COORD..=MAX_COORD).contains(&p.x)
        && (-MAX_COORD..=MAX_COORD).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.first() == points.last() {
        points.len() - 1
    } else {
        points.len()
    };
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    Ok(points[..effective_len].to_vec())
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let squared = (dx * dx + dy * dy) as u128 * 1_000_000u128;

    // Integer square root, followed by nearest-integer rounding. The square
    // root's fractional part is at least one half exactly when 4*N >= (2*r+1)^2.
    let mut low = 0u128;
    let mut high = 2_828_427_125u128;
    while low < high {
        let mid = low + (high - low + 1) / 2;
        if mid <= squared / mid.max(1) {
            low = mid;
        } else {
            high = mid - 1;
        }
    }
    let root = low;
    let threshold = 2 * root + 1;
    let rounded = if 4 * squared >= threshold * threshold {
        root + 1
    } else {
        root
    };
    rounded as u64
}

fn perimeter_milli(points: &[Point]) -> u64 {
    if points.len() < 2 {
        return 0;
    }
    (0..points.len())
        .map(|i| rounded_edge_milli(points[i], points[(i + 1) % points.len()]))
        .sum()
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed = signed_area2(&polygon);
    let mut edge_milli = Vec::with_capacity(polygon.len());
    for i in 0..polygon.len() {
        edge_milli.push(rounded_edge_milli(
            polygon[i],
            polygon[(i + 1) % polygon.len()],
        ));
    }
    let perimeter_milli = edge_milli.iter().copied().sum();

    let longest_edge = (0..polygon.len()).max_by(|&i, &j| {
        edge_milli[i]
            .cmp(&edge_milli[j])
            .then_with(|| {
                let mut pi = (polygon[i], polygon[(i + 1) % polygon.len()]);
                let mut pj = (polygon[j], polygon[(j + 1) % polygon.len()]);
                if pi.1 < pi.0 {
                    pi = (pi.1, pi.0);
                }
                if pj.1 < pj.0 {
                    pj = (pj.1, pj.0);
                }
                pj.cmp(&pi)
            })
            .then_with(|| j.cmp(&i))
    });

    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: match signed.cmp(&0) {
            std::cmp::Ordering::Less => Orientation::Clockwise,
            std::cmp::Ordering::Equal => Orientation::Degenerate,
            std::cmp::Ordering::Greater => Orientation::CounterClockwise,
        },
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn classify_one(polygon: &[Point], q: Point) -> Location {
    for i in 0..polygon.len() {
        if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
            return Location::Boundary { edge: i };
        }
    }

    let mut inside = false;
    for i in 0..polygon.len() {
        let a = polygon[i];
        let b = polygon[(i + 1) % polygon.len()];
        if (a.y > q.y) != (b.y > q.y) {
            let side = cross(a, b, q);
            if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                inside = !inside;
            }
        }
    }
    if inside {
        Location::Inside
    } else {
        Location::Outside
    }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }
    Ok(queries
        .iter()
        .copied()
        .map(|query| classify_one(&polygon, query))
        .collect())
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique = Vec::<(Point, usize)>::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = unique.iter_mut().find(|entry| entry.0 == point) {
            existing.1 = existing.1.min(index);
        } else {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|entry| entry.0);

    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|entry| entry.0).collect();
        let source_indices: Vec<usize> = unique.iter().map(|entry| entry.1).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|entry| cross(unique[0].0, unique[1].0, entry.0) == 0);

    let hull_indices = if all_collinear {
        if mode == Collinear::Include {
            (0..unique.len()).collect()
        } else {
            vec![0, unique.len() - 1]
        }
    } else {
        let mut lower = Vec::<usize>::new();
        for i in 0..unique.len() {
            while lower.len() >= 2 {
                let n = lower.len();
                let turn = cross(unique[lower[n - 2]].0, unique[lower[n - 1]].0, unique[i].0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                lower.pop();
            }
            lower.push(i);
        }

        let mut upper = Vec::<usize>::new();
        for i in (0..unique.len()).rev() {
            while upper.len() >= 2 {
                let n = upper.len();
                let turn = cross(unique[upper[n - 2]].0, unique[upper[n - 1]].0, unique[i].0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                upper.pop();
            }
            upper.push(i);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_indices.iter().map(|&i| unique[i].0).collect();
    let source_indices: Vec<usize> = hull_indices.iter().map(|&i| unique[i].1).collect();
    let area = if vertices.len() >= 3 {
        signed_area2(&vertices).unsigned_abs()
    } else {
        0
    };
    let perimeter = perimeter_milli(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli: perimeter,
    })
}
