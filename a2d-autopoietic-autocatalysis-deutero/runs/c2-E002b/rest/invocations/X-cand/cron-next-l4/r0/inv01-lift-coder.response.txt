use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [BTreeSet<u8>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Clone, Copy)]
struct Domain {
    min: u8,
    max: u8,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::InvalidSyntax { field, offset },
    }
}

fn names(field: Field) -> Option<&'static [(&'static str, u8)]> {
    match field {
        Field::Month => Some(&[
            ("JAN", 1), ("FEB", 2), ("MAR", 3), ("APR", 4),
            ("MAY", 5), ("JUN", 6), ("JUL", 7), ("AUG", 8),
            ("SEP", 9), ("OCT", 10), ("NOV", 11), ("DEC", 12),
        ]),
        Field::DayOfWeek => Some(&[
            ("SUN", 0), ("MON", 1), ("TUE", 2), ("WED", 3),
            ("THU", 4), ("FRI", 5), ("SAT", 6),
        ]),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum Atom {
    Number(u8),
    Name(u8),
}

impl Atom {
    fn value(self) -> u8 {
        match self {
            Atom::Number(v) | Atom::Name(v) => v,
        }
    }
}

fn parse_atom(
    text: &str,
    field: Field,
    offset: usize,
) -> Result<Atom, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    if text.bytes().all(|b| b.is_ascii_digit()) {
        if text.len() > 1 && text.as_bytes()[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }

        let mut value: u32 = 0;
        for b in text.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let d = domain(field);
        if value < d.min as u32 || value > d.max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok(Atom::Number(value as u8));
    }

    if text.bytes().all(|b| b.is_ascii_alphabetic()) {
        if let Some(table) = names(field) {
            for &(name, value) in table {
                if text.eq_ignore_ascii_case(name) {
                    return Ok(Atom::Name(value));
                }
            }
        }
    }

    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_step(text: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if text.len() > 1 && text.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value: u32 = 0;
    for b in text.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    let d = domain(field);
    let width = (d.max - d.min + 1) as u32;
    if value == 0 || value > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_item(
    item: &str,
    field: Field,
    item_offset: usize,
    out: &mut BTreeSet<u8>,
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, b)| (*b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
        (
            &item[..slash],
            Some(&item[slash + 1..]),
            Some(item_offset + slash + 1),
        )
    } else {
        (item, None, None)
    };

    let d = domain(field);
    let (start, end, range_end_offset, is_star) = if base == "*" {
        (d.min, d.max, item_offset, true)
    } else {
        let hyphens: Vec<usize> = base
            .as_bytes()
            .iter()
            .enumerate()
            .filter_map(|(i, b)| (*b == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }

        if let Some(&hyphen) = hyphens.first() {
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if left.is_empty() || right.is_empty() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_offset,
                });
            }
            let a = parse_atom(left, field, item_offset)?;
            let b = parse_atom(right, field, item_offset + hyphen + 1)?;
            let av = a.value();
            let bv = b.value();
            if av > bv {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_offset + hyphen + 1,
                });
            }
            (av, bv, item_offset + hyphen + 1, false)
        } else {
            let a = parse_atom(base, field, item_offset)?;
            let av = a.value();
            let end = if step_text.is_some() { d.max } else { av };
            (av, end, item_offset, false)
        }
    };

    let step = match (step_text, step_offset) {
        (Some(text), Some(offset)) => Some(parse_step(text, field, offset)?),
        _ => None,
    };

    if let Some(step) = step {
        let mut value = start as u32;
        while value <= end as u32 {
            out.insert(value as u8);
            value += step;
        }
    } else {
        for value in start..=end {
            out.insert(value);
        }
    }

    let _ = (range_end_offset, is_star);
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<BTreeSet<u8>, ParseError> {
    for (offset, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = BTreeSet::new();
    let mut offset = 0;
    for item in text.split(',') {
        parse_item(item, field, offset, &mut selected)?;
        offset += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    if input.as_bytes().first() == Some(&b'@') {
        let expansion = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        return parse_cron(expansion);
    }

    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    for (i, field) in FIELDS.into_iter().enumerate() {
        selected[i] = parse_field(pieces[i], field)?;
    }

    Ok(Cron {
        selected,
        unrestricted_dom: pieces[2] == "*",
        unrestricted_dow: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].iter().copied().collect()
    }

    fn date_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.selected[2].contains(&at.day);
        let dow = self.selected[4].contains(&weekday(at.year, at.month, at.day));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn time_matches(&self, at: CivilDateTime) -> bool {
        self.selected[0].contains(&at.minute)
            && self.selected[1].contains(&at.hour)
            && self.selected[3].contains(&at.month)
            && self.date_matches(at)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.time_matches(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut date = (after.year, after.month, after.day);
        let mut first_day = true;

        loop {
            let (year, month, day) = date;
            let date_start = CivilDateTime {
                year,
                month,
                day,
                hour: 0,
                minute: 0,
            };
            if self.date_matches(date_start) && self.selected[3].contains(&month) {
                let min_time = if first_day
                    && year == after.year
                    && month == after.month
                    && day == after.day
                {
                    after.hour as u16 * 60 + after.minute as u16 + 1
                } else {
                    0
                };

                for &hour in &self.selected[1] {
                    for &minute in &self.selected[0] {
                        let total = hour as u16 * 60 + minute as u16;
                        if total >= min_time {
                            return Ok(Some(CivilDateTime {
                                year,
                                month,
                                day,
                                hour,
                                minute,
                            }));
                        }
                    }
                }
            }

            match next_date(year, month, day) {
                Some(next) => {
                    date = next;
                    first_day = false;
                }
                None => return Ok(None),
            }
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut date = (before.year, before.month, before.day);
        let mut first_day = true;

        loop {
            let (year, month, day) = date;
            let date_start = CivilDateTime {
                year,
                month,
                day,
                hour: 0,
                minute: 0,
            };
            if self.date_matches(date_start) && self.selected[3].contains(&month) {
                let max_time = if first_day {
                    before.hour as i16 * 60 + before.minute as i16 - 1
                } else {
                    24 * 60 - 1
                };

                if max_time >= 0 {
                    for &hour in self.selected[1].iter().rev() {
                        for &minute in self.selected[0].iter().rev() {
                            let total = hour as i16 * 60 + minute as i16;
                            if total <= max_time {
                                return Ok(Some(CivilDateTime {
                                    year,
                                    month,
                                    day,
                                    hour,
                                    minute,
                                }));
                            }
                        }
                    }
                }
            }

            match previous_date(year, month, day) {
                Some(prev) => {
                    date = prev;
                    first_day = false;
                }
                None => return Ok(None),
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;

        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        if start == end {
            return Ok(FireWindow {
                fires: Vec::new(),
                truncated: false,
            });
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(next) = self.next_fire(cursor).expect("validated datetime")? {
            if next >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(next);
            cursor = next;
        }

        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let fields = [
            canonical_field(&self.selected[0], domain(Field::Minute), false),
            canonical_field(&self.selected[1], domain(Field::Hour), false),
            canonical_field(&self.selected[2], domain(Field::DayOfMonth), self.unrestricted_dom),
            canonical_field(&self.selected[3], domain(Field::Month), false),
            canonical_field(&self.selected[4], domain(Field::DayOfWeek), self.unrestricted_dow),
        ];
        fields.join(" ")
    }
}

fn canonical_field(values: &BTreeSet<u8>, d: Domain, unrestricted: bool) -> String {
    if unrestricted {
        return "*".to_string();
    }
    let full_len = (d.max - d.min + 1) as usize;
    if values.len() == full_len {
        if matches!(d.min, 1) && d.max == 31 {
            return "*/1".to_string();
        }
        if d.min == 0 && d.max == 6 {
            return "*/1".to_string();
        }
        return "*".to_string();
    }
    values
        .iter()
        .map(u8::to_string)
        .collect::<Vec<_>>()
        .join(",")
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days: u32 = 0;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn next_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day < days_in_month(year, month) {
        Some((year, month, day + 1))
    } else if month < 12 {
        Some((year, month + 1, 1))
    } else if year < 2399 {
        Some((year + 1, 1, 1))
    } else {
        None
    }
}

fn previous_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day > 1 {
        Some((year, month, day - 1))
    } else if month > 1 {
        let prev_month = month - 1;
        Some((year, prev_month, days_in_month(year, prev_month)))
    } else if year > 2000 {
        Some((year - 1, 12, 31))
    } else {
        None
    }
}
