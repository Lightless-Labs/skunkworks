#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Debug, Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
            })
            .collect();

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let segments = parse_path_head(&path[..head_end])?;

        let mut best: Option<(usize, Vec<u8>, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matched = true;

            for (pattern_segment, path_segment) in route.segments.iter().zip(&segments) {
                match pattern_segment {
                    Segment::Static(value) if value == path_segment => {}
                    Segment::Static(_) => {
                        matched = false;
                        break;
                    }
                    Segment::Param(name) => params.push((name.clone(), path_segment.clone())),
                }
            }

            if matched {
                let replace = best
                    .as_ref()
                    .map(|(_, rank, _)| route.rank > *rank)
                    .unwrap_or(true);
                if replace {
                    best = Some((route_id, route.rank.clone(), params));
                }
            }
        }

        match best {
            Some((route, _, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    parse_segments(input, true).map_err(|at| Error::InvalidPattern { at })
}

fn parse_path_head(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    let raw_segments = parse_segments(input, false)
        .map_err(|at| Error::InvalidPath { at })?;

    raw_segments
        .into_iter()
        .map(|(_, start, raw)| decode_static(raw, start).map_err(|at| Error::InvalidPath { at }))
        .collect()
}

fn parse_segments<'a>(
    input: &'a str,
    is_pattern: bool,
) -> Result<Vec<(Segment, usize, &'a str)>, usize> {
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut out = Vec::new();
    let mut start = 1usize;

    while start < input.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|offset| start + offset)
            .unwrap_or(input.len());

        if end == start {
            return Err(start);
        }

        let raw = &input[start..end];
        let kind = if is_pattern && raw.starts_with(':') {
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(start);
            }
            Segment::Param(name.to_owned())
        } else if is_pattern && raw.starts_with('*') {
            return Err(start);
        } else {
            let decoded = decode_static(raw, start)?;
            Segment::Static(decoded)
        };

        out.push((kind, start, raw));

        if end == input.len() {
            break;
        }
        start = end + 1;
        if start == input.len() {
            return Err(end);
        }
    }

    let mut names = std::collections::HashSet::new();
    if is_pattern {
        for (segment, start, _) in &out {
            if let Segment::Param(name) = segment {
                if !names.insert(name.clone()) {
                    return Err(*start);
                }
            }
        }
    }

    Ok(out)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_static(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = uppercase_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = uppercase_hex(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            decoded.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn uppercase_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}
