use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut core = [0u64; 3];

    for (index, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            return Err(match bytes.get(pos) {
                None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut parsed = 0u64;
        for &digit in &bytes[start..pos] {
            parsed = parsed
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *value = parsed;

        if index < 2 {
            if bytes.get(pos) != Some(&b'.') {
                return Err(match bytes.get(pos) {
                    None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                    _ => VersionError::InvalidChar { at: pos },
                });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if bytes.get(pos) == Some(&b'-') {
        pos += 1;
        let pre_start = pos;
        let plus = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pos + offset)
            .unwrap_or(bytes.len());

        pre = parse_identifiers(bytes, pre_start, plus, true)?;

        if plus < bytes.len() {
            pos = plus + 1;
            build = parse_identifiers(bytes, pos, bytes.len(), false)?;
        } else {
            pos = bytes.len();
        }
    } else if bytes.get(pos) == Some(&b'+') {
        pos += 1;
        build = parse_identifiers(bytes, pos, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    check_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut identifier_start = start;

    loop {
        let identifier_end = bytes[identifier_start..end]
            .iter()
            .position(|&b| b == b'.')
            .map(|offset| identifier_start + offset)
            .unwrap_or(end);

        if identifier_start == identifier_end {
            return Err(VersionError::EmptyIdentifier {
                at: identifier_start,
            });
        }

        for index in identifier_start..identifier_end {
            let byte = bytes[index];
            if !(byte.is_ascii_alphanumeric() || byte == b'-') {
                return Err(VersionError::InvalidChar { at: index });
            }
        }

        if check_numeric_leading_zero
            && bytes[identifier_start..identifier_end]
                .iter()
                .all(u8::is_ascii_digit)
            && identifier_end - identifier_start > 1
            && bytes[identifier_start] == b'0'
        {
            return Err(VersionError::LeadingZero {
                at: identifier_start,
            });
        }

        result.push(
            std::str::from_utf8(&bytes[identifier_start..identifier_end])
                .expect("identifier bytes are ASCII")
                .to_owned(),
        );

        if identifier_end == end {
            break;
        }
        identifier_start = identifier_end + 1;
    }

    Ok(result)
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.iter().zip(b) {
        let left_numeric = left.bytes().all(|byte| byte.is_ascii_digit());
        let right_numeric = right.bytes().all(|byte| byte.is_ascii_digit());

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => left
                .len()
                .cmp(&right.len())
                .then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: ComparatorOp,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum ComparatorOp {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, operand_offset) = if token.starts_with(">=") {
            (ComparatorOp::GreaterEqual, 2)
        } else if token.starts_with("<=") {
            (ComparatorOp::LessEqual, 2)
        } else if token.starts_with('>') {
            (ComparatorOp::Greater, 1)
        } else if token.starts_with('<') {
            (ComparatorOp::Less, 1)
        } else if token.starts_with('=') {
            (ComparatorOp::Equal, 1)
        } else if token.as_bytes().first().is_some_and(u8::is_ascii_digit) {
            (ComparatorOp::Equal, 0)
        } else {
            return Err(RangeError::Syntax { at: token_start });
        };

        let operand_start = token_start + operand_offset;
        if operand_offset >= token.len()
            || !token.as_bytes()[operand_offset].is_ascii_digit()
        {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

fn is_range_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted = v.pre.is_empty()
            || arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && core_tuple(v) == core_tuple(&comparator.version)
            });

        admitted
            && arm.iter().all(|comparator| {
                let ordering = compare_versions(v, &comparator.version);
                match comparator.op {
                    ComparatorOp::Equal => ordering == Ordering::Equal,
                    ComparatorOp::Less => ordering == Ordering::Less,
                    ComparatorOp::LessEqual => ordering != Ordering::Greater,
                    ComparatorOp::Greater => ordering == Ordering::Greater,
                    ComparatorOp::GreaterEqual => ordering != Ordering::Less,
                }
            })
    })
}

fn core_tuple(version: &Version) -> (u64, u64, u64) {
    (version.major, version.minor, version.patch)
}
