use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, at: usize) -> Option<u8> {
    s.as_bytes().get(at).copied()
}

fn missing_core_byte(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while matches!(byte(s, *pos), Some(b'0'..=b'9')) {
        *pos += 1;
    }

    if *pos == start {
        return if missing_core_byte(byte(s, start)) {
            Err(VersionError::MissingCore { at: start })
        } else {
            Err(VersionError::InvalidChar { at: start })
        };
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in digits.as_bytes() {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && byte(s, i) != Some(b'.') {
            let b = byte(s, i).unwrap();
            if !matches!(b, b'0'..=b'9' | b'a'..=b'z' | b'A'..=b'Z' | b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let identifier = &s[ident_start..i];
        if prerelease
            && identifier.len() > 1
            && identifier.as_bytes().iter().all(u8::is_ascii_digit)
            && identifier.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(identifier.to_owned());

        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;

    if byte(s, pos) != Some(b'.') {
        return if missing_core_byte(byte(s, pos)) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if byte(s, pos) != Some(b'.') {
        return if missing_core_byte(byte(s, pos)) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    if !matches!(byte(s, pos), None | Some(b'-' | b'+')) {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if byte(s, pos) == Some(b'-') {
        let pre_start = pos + 1;
        let build_pos = s[pre_start..].find('+').map(|i| pre_start + i);
        let pre_end = build_pos.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;

        if byte(s, pos) == Some(b'+') {
            pos += 1;
            build = parse_identifiers(s, pos, s.len(), false)?;
            pos = s.len();
        }
    } else if byte(s, pos) == Some(b'+') {
        pos += 1;
        build = parse_identifiers(s, pos, s.len(), false)?;
        pos = s.len();
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let ordering = compare_identifiers(left, right);
                if ordering != Ordering::Equal {
                    return ordering;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_space(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        let (operator, operator_len) = match bytes[pos] {
            b'=' => (Operator::Equal, 1),
            b'<' if bytes.get(pos + 1) == Some(&b'=') => (Operator::LessEqual, 2),
            b'<' => (Operator::Less, 1),
            b'>' if bytes.get(pos + 1) == Some(&b'=') => (Operator::GreaterEqual, 2),
            b'>' => (Operator::Greater, 1),
            _ => (Operator::Equal, 0),
        };

        let operand_start = token_start + operator_len;
        if operand_start >= bytes.len() || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let mut token_end = operand_start;
        while token_end < bytes.len() && !is_range_space(bytes[token_end]) {
            token_end += 1;
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { operator, version });
        pos = token_end;
    }

    if comparators.is_empty() {
        Err(RangeError::Empty { at: 0 })
    } else {
        Ok(Range { comparators })
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let candidate_core = (v.major, v.minor, v.patch);

    if !v.pre.is_empty()
        && !r.comparators.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && (
                    comparator.version.major,
                    comparator.version.minor,
                    comparator.version.patch,
                ) == candidate_core
        })
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Equal => ordering == Ordering::Equal,
            Operator::Less => ordering == Ordering::Less,
            Operator::LessEqual => ordering != Ordering::Greater,
            Operator::Greater => ordering == Ordering::Greater,
            Operator::GreaterEqual => ordering != Ordering::Less,
        }
    })
}
