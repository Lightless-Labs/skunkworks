#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<(usize, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(relative) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: start + 1 + relative,
        });
    }

    let mut content_start = start + 1;
    let operator = match bytes.get(content_start).copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            let op = bytes[content_start] as char;
            content_start += 1;
            if op != '+' && op != '#' {
                return Err(Error::UnsupportedOperator {
                    pos: content_start - 1,
                });
            }
            Some(op)
        }
        _ => None,
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..end]
            .iter()
            .position(|&b| b == b',')
            .map_or(end, |offset| spec_start + offset);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        if !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let mut cursor = spec_start + 1;
        while cursor < spec_end && is_name_continue(bytes[cursor]) {
            cursor += 1;
        }
        if cursor != spec_end {
            return Err(Error::InvalidName { pos: cursor });
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        names.push((spec_start, spec_end));

        if spec_end == end {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        names,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let start = cursor;
                let Some(relative) = bytes[start + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = start + 1 + relative;
                expressions.push(parse_expression(template, start, end)?);
                cursor = end + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte,
        b'a'..=b'f' => byte - b'a' + b'A',
        _ => byte,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn push_percent_encoded(output: &mut String, byte: u8) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    output.push('%');
    output.push(HEX[(byte >> 4) as usize] as char);
    output.push(HEX[(byte & 0x0f) as usize] as char);
}

fn encode_value(value: &str, allow_reserved: bool, output: &mut String) {
    let bytes = value.as_bytes();
    let mut cursor = 0;

    while cursor < bytes.len() {
        let byte = bytes[cursor];

        if byte == b'%' && cursor + 2 < bytes.len()
            && is_hex(bytes[cursor + 1])
            && is_hex(bytes[cursor + 2])
        {
            output.push('%');
            output.push(hex_value(bytes[cursor + 1]) as char);
            output.push(hex_value(bytes[cursor + 2]) as char);
            cursor += 3;
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
            cursor += 1;
        } else {
            push_percent_encoded(output, byte);
            cursor += 1;
        }
    }
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(binding, value)| (binding == name).then_some(value))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut literal_cursor = 0;

    for expression in expressions {
        output.push_str(&template[literal_cursor..expression.start]);

        let allow_reserved = matches!(expression.operator, Some('+' | '#'));
        let mut fields = Vec::new();

        for &(name_start, name_end) in &expression.names {
            let name = &template[name_start..name_end];
            if let Some(value) = lookup(name, vars) {
                let mut encoded = String::new();
                encode_value(value, allow_reserved, &mut encoded);
                fields.push(encoded);
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            for (index, field) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                output.push_str(field);
            }
        }

        literal_cursor = expression.end + 1;
    }

    output.push_str(&template[literal_cursor..]);
    Ok(output)
}
