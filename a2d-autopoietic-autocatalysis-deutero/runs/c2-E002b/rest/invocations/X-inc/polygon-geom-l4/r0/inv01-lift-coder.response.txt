#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_ITEMS: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<(Vec<Point>, Vec<usize>), GeomError> {
    if points.len() > MAX_ITEMS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut len = points.len();
    if len >= 2 && points[len - 1] == points[0] {
        len -= 1;
    }
    if len < 3 {
        return Err(GeomError::TooFewVertices { len });
    }

    Ok((points[..len].to_vec(), (0..len).collect()))
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0_i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = (b.x as i128 - a.x as i128) as u128;
    let dy = (b.y as i128 - a.y as i128) as u128;
    dx * dx + dy * dy
}

fn integer_sqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }

    let mut x = n;
    let mut y = x / 2 + 1;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let n = squared_distance(a, b) * 1_000_000;
    let root = integer_sqrt(n);
    let remainder = n - root * root;
    let rounded = if remainder > root { root + 1 } else { root };
    rounded as u64
}

fn endpoint_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b {
        (a, b)
    } else {
        (b, a)
    }
}

fn measures_for(points: &[Point]) -> Measures {
    let signed_area2 = shoelace(points);
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..points.len() {
        let length = rounded_edge_milli(points[i], points[(i + 1) % points.len()]);
        edge_milli.push(length);
        perimeter_milli += length;

        match longest_edge {
            None => longest_edge = Some(i),
            Some(best) => {
                let pair = endpoint_pair(points[i], points[(i + 1) % points.len()]);
                let best_pair =
                    endpoint_pair(points[best], points[(best + 1) % points.len()]);
                if length > edge_milli[best]
                    || (length == edge_milli[best]
                        && (pair < best_pair || (pair == best_pair && i < best)))
                {
                    longest_edge = Some(i);
                }
            }
        }
    }

    Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let (vertices, _) = prepare_polygon(points)?;
    Ok(measures_for(&vertices))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn classify_one(polygon: &[Point], q: Point) -> Location {
    let n = polygon.len();
    for i in 0..n {
        if on_segment(polygon[i], polygon[(i + 1) % n], q) {
            return Location::Boundary { edge: i };
        }
    }

    let mut inside = false;
    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];
        if (a.y > q.y) != (b.y > q.y) {
            let c = cross(a, b, q);
            if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                inside = !inside;
            }
        }
    }

    if inside {
        Location::Inside
    } else {
        Location::Outside
    }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let (polygon, _) = prepare_polygon(points)?;

    if queries.len() > MAX_ITEMS {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }

    Ok(queries
        .iter()
        .map(|&query| classify_one(&polygon, query))
        .collect())
}

fn between(a: Point, b: Point, p: Point) -> bool {
    p.x >= a.x.min(b.x)
        && p.x <= a.x.max(b.x)
        && p.y >= a.y.min(b.y)
        && p.y <= a.y.max(b.y)
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let o1 = cross(a, b, c);
    let o2 = cross(a, b, d);
    let o3 = cross(c, d, a);
    let o4 = cross(c, d, b);

    if o1 == 0 && between(a, b, c) {
        return true;
    }
    if o2 == 0 && between(a, b, d) {
        return true;
    }
    if o3 == 0 && between(c, d, a) {
        return true;
    }
    if o4 == 0 && between(c, d, b) {
        return true;
    }

    (o1 < 0) != (o2 < 0) && (o3 < 0) != (o4 < 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let (vertices, _) = prepare_polygon(points)?;
    let n = vertices.len();

    for i in 0..n {
        for j in (i + 1)..n {
            if vertices[i] == vertices[j] {
                return Err(GeomError::Duplicate {
                    first: i,
                    second: j,
                });
            }
        }
    }

    let area = shoelace(&vertices);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let current = vertices[i];
        let next = vertices[(i + 1) % n];
        if cross(prev, current, next) == 0 {
            let ax = prev.x as i128 - current.x as i128;
            let ay = prev.y as i128 - current.y as i128;
            let bx = next.x as i128 - current.x as i128;
            let by = next.y as i128 - current.y as i128;
            if ax * bx + ay * by > 0 {
                return Err(GeomError::AdjacentOverlap { vertex: i });
            }
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                vertices[i],
                vertices[(i + 1) % n],
                vertices[j],
                vertices[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let min_index = (0..n).min_by_key(|&i| vertices[i]).unwrap();
    let direction: isize = if area > 0 { 1 } else { -1 };
    let mut canonical_vertices = Vec::with_capacity(n);
    let mut source_indices = Vec::with_capacity(n);
    for step in 0..n {
        let index =
            (min_index as isize + direction * step as isize).rem_euclid(n as isize) as usize;
        canonical_vertices.push(vertices[index]);
        source_indices.push(index);
    }

    let measures = measures_for(&canonical_vertices);
    Ok(Validated {
        vertices: canonical_vertices,
        source_indices,
        measures,
    })
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_ITEMS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (i, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(existing, _)| *existing == point) {
            unique.push((point, i));
        }
    }
    unique.sort_by_key(|(point, _)| *point);

    let vertices_with_sources: Vec<(Point, usize)> = match unique.len() {
        0 => Vec::new(),
        1 => unique.clone(),
        _ => {
            let all_collinear = unique
                .iter()
                .skip(2)
                .all(|(point, _)| cross(unique[0].0, unique[1].0, *point) == 0);

            if all_collinear {
                match mode {
                    Collinear::Include => unique.clone(),
                    Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
                }
            } else {
                let mut lower: Vec<(Point, usize)> = Vec::new();
                for &item in &unique {
                    while lower.len() >= 2 {
                        let m = lower.len();
                        let turn = cross(lower[m - 2].0, lower[m - 1].0, item.0);
                        let remove = match mode {
                            Collinear::Exclude => turn <= 0,
                            Collinear::Include => turn < 0,
                        };
                        if remove {
                            lower.pop();
                        } else {
                            break;
                        }
                    }
                    lower.push(item);
                }

                let mut upper: Vec<(Point, usize)> = Vec::new();
                for &item in unique.iter().rev() {
                    while upper.len() >= 2 {
                        let m = upper.len();
                        let turn = cross(upper[m - 2].0, upper[m - 1].0, item.0);
                        let remove = match mode {
                            Collinear::Exclude => turn <= 0,
                            Collinear::Include => turn < 0,
                        };
                        if remove {
                            upper.pop();
                        } else {
                            break;
                        }
                    }
                    upper.push(item);
                }

                lower.pop();
                upper.pop();
                lower.extend(upper);
                lower
            }
        }
    };

    let vertices: Vec<Point> = vertices_with_sources.iter().map(|(p, _)| *p).collect();
    let source_indices: Vec<usize> = vertices_with_sources.iter().map(|(_, i)| *i).collect();

    let area2 = if vertices.len() >= 3 {
        shoelace(&vertices).unsigned_abs()
    } else {
        0
    };

    let perimeter_milli = if vertices.len() < 2 {
        0
    } else {
        (0..vertices.len())
            .map(|i| rounded_edge_milli(vertices[i], vertices[(i + 1) % vertices.len()]))
            .sum()
    };

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}
