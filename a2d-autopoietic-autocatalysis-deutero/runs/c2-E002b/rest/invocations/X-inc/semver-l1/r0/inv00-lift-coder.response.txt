use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_ascii_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    if start >= bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
        return Err(VersionError::MissingCore { at: start });
    }
    if !is_ascii_digit(bytes[start]) {
        return Err(VersionError::InvalidChar { at: start });
    }

    while *pos < bytes.len() && is_ascii_digit(bytes[*pos]) {
        *pos += 1;
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }

    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    pos: &mut usize,
    stop_at_plus: bool,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut identifiers = Vec::new();

    loop {
        let start = *pos;
        while *pos < bytes.len()
            && bytes[*pos] != b'.'
            && !(stop_at_plus && bytes[*pos] == b'+')
        {
            if !valid_identifier_char(bytes[*pos]) {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let identifier = &s[start..*pos];
        if prerelease
            && identifier.len() > 1
            && identifier.starts_with('0')
            && identifier.bytes().all(is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: start });
        }
        identifiers.push(identifier.to_owned());

        if *pos >= bytes.len() || (stop_at_plus && bytes[*pos] == b'+') {
            break;
        }

        // The only other stopping byte here is a dot.
        *pos += 1;
        if *pos >= bytes.len() || (stop_at_plus && bytes[*pos] == b'+') {
            return Err(VersionError::EmptyIdentifier { at: *pos });
        }
    }

    Ok(identifiers)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;

    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;

    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;

    if pos < s.len() && !matches!(s.as_bytes()[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        pos += 1;
        pre = parse_identifiers(s, &mut pos, true, true)?;
    }

    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, &mut pos, false, false)?;
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_ascii_digit)
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (left, right) in a.iter().zip(b.iter()) {
        let left_numeric = is_numeric_identifier(left);
        let right_numeric = is_numeric_identifier(right);

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => compare_numeric_identifiers(left, right),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n') {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, operand_offset) = if let Some(rest) = token.strip_prefix(">=") {
            (Op::GreaterEqual, token.len() - rest.len())
        } else if let Some(rest) = token.strip_prefix("<=") {
            (Op::LessEqual, token.len() - rest.len())
        } else if let Some(rest) = token.strip_prefix('>') {
            (Op::Greater, token.len() - rest.len())
        } else if let Some(rest) = token.strip_prefix('<') {
            (Op::Less, token.len() - rest.len())
        } else if let Some(rest) = token.strip_prefix('=') {
            (Op::Equal, token.len() - rest.len())
        } else {
            (Op::Equal, 0)
        };

        let operand_start = token_start + operand_offset;
        let operand = &s[operand_start..token_end];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && (v.major, v.minor, v.patch)
                        == (
                            comparator.version.major,
                            comparator.version.minor,
                            comparator.version.patch,
                        )
            })
        {
            return false;
        }

        arm.iter().all(|comparator| {
            let ordering = compare_versions(v, &comparator.version);
            match comparator.op {
                Op::Equal => ordering == Ordering::Equal,
                Op::Less => ordering == Ordering::Less,
                Op::LessEqual => ordering != Ordering::Greater,
                Op::Greater => ordering == Ordering::Greater,
                Op::GreaterEqual => ordering != Ordering::Less,
            }
        })
    })
}
