#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(&str, u128); 6] = [
    ("thousand", 1_000),
    ("million", 1_000_000),
    ("billion", 1_000_000_000),
    ("trillion", 1_000_000_000_000),
    ("quadrillion", 1_000_000_000_000_000),
    ("quintillion", 1_000_000_000_000_000_000),
];

fn append_under_thousand(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_under_hundred(remainder, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

fn append_under_hundred(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let negative = value < 0;
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut out = String::new();
    if negative {
        out.push_str("minus ");
    }

    let mut higher_present = false;
    for index in (1..7).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }
        if !out.is_empty() && !(out == "minus ") {
            out.push(' ');
        } else if out == "minus " {
            // The sign already includes its required following space.
        }
        append_under_thousand(group, &mut out);
        out.push(' ');
        out.push_str(SCALES[index - 1].0);
        higher_present = true;
    }

    let units = groups[0];
    if units != 0 {
        if higher_present {
            out.push_str(if units < 100 { " and " } else { " " });
        } else if negative {
            // The sign's trailing space is already in the output.
        }
        append_under_thousand(units, &mut out);
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }
    for (value, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(value as u128);
        }
    }
    for (index, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(((index + 2) * 10) as u128);
        }
    }
    None
}

fn scale_value(word: &str) -> Option<u128> {
    SCALES.iter().find_map(|(spelling, value)| {
        if word == *spelling {
            Some(*value)
        } else {
            None
        }
    })
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, &(word, position)) in tokens.iter().enumerate() {
        let valid = word == "minus"
            || word == "and"
            || word == "hundred"
            || word_value(word).is_some()
            || scale_value(word).is_some();
        if !valid || (word == "minus" && i != 0) {
            return Err(WordsError::InvalidWord { index: position });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic_start = if negative { 1 } else { 0 };
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(word, _) in &tokens[arithmetic_start..] {
        if word == "and" {
            // No arithmetic effect.
        } else if word == "hundred" {
            group = group.checked_mul(100).ok_or(WordsError::Range)?;
        } else if let Some(scale) = scale_value(word) {
            total = total
                .checked_add(group.checked_mul(scale).ok_or(WordsError::Range)?)
                .ok_or(WordsError::Range)?;
            group = 0;
        } else if let Some(value) = word_value(word) {
            group = group.checked_add(value).ok_or(WordsError::Range)?;
        }

        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(value)
}
