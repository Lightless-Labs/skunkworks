    dollar: usize,

                        .chars()
                        .next()
                        .expect("character follows backslash");
                    cursor = next + escaped.len_utf8();

                    cursor = next;
                }

                break;
            } else {
                cursor = next;
            }

            .chars()
            .take_while(|&ch| is_name_continue(ch))
            .map(char::len_utf8)
            .sum::<usize>();

        let name = &contents[..name_len];
        let valid_name = !name.is_empty()
            && name.chars().next().is_some_and(is_name_start);

            None

            Some(&contents[name_len + 2..])

            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

            (Some(value), Some(default)) if !value.is_empty() => value.to_string(),

            (Some(value), None) => value.to_string(),
            (None, None) => String::new(),

            end += ch.len_utf8();
        }
        let name = &input[after_dollar..end];
        Ok((end, Some(lookup(env, name).unwrap_or("").to_string())))
    } else {
        Ok((after_dollar, None))
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let ch = text[i..].chars().next().expect("valid character boundary");
        let next = i + ch.len_utf8();
        if ch == '\\' && next < text.len() {
            let escaped = text[next..]
                .chars()
                .next()
                .expect("character follows backslash");
            out.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            out.push(ch);
            i = next;
        }
    }
    out
}
