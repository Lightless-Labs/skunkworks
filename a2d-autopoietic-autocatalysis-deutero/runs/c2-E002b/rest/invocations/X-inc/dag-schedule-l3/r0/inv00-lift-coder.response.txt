#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before performing any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Store successor lists and indegrees. Dependency list ordering does not
    // affect scheduling; successor lists are sorted for deterministic traversal.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }
    for children in &mut successors {
        children.sort_unstable();
    }

    // Kahn traversal always removes the smallest currently available ID.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        topo.push(task_id);
        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if topo.len() != n {
        let removed: Vec<bool> = {
            let mut seen = vec![false; n];
            for &task_id in &topo {
                seen[task_id] = true;
            }
            seen
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in topo.iter().rev() {
        let max_child_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running = Vec::<(u64, usize, usize)>::new(); // finish, task, worker
    let mut completed_count = 0usize;
    let mut current_time = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all running tasks finishing now, ordered by task ID.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|&(finish, _, _)| finish == current_time)
            .collect();
        finishing.sort_by_key(|&(_, task_id, _)| task_id);

        for &(finish, task_id, worker_id) in &finishing {
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
        }
        if !finishing.is_empty() {
            running.retain(|&(finish, _, _)| finish != current_time);
        }

        loop {
            let worker_id = match worker_available
                .iter()
                .position(|&available_at| available_at <= current_time)
            {
                Some(id) => id,
                None => break,
            };

            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > current_time {
                    continue;
                }
                if tasks[task_id]
                    .deps
                    .iter()
                    .any(|&dep| !completed[dep])
                {
                    continue;
                }

                let is_better = match best {
                    None => true,
                    Some(other) => {
                        tails[task_id] > tails[other]
                            || (tails[task_id] == tails[other]
                                && (tasks[task_id].priority > tasks[other].priority
                                    || (tasks[task_id].priority == tasks[other].priority
                                        && task_id < other)))
                    }
                };
                if is_better {
                    best = Some(task_id);
                }
            }

            let task_id = match best {
                Some(id) => id,
                None => break,
            };

            let finish = current_time.checked_add(tasks[task_id].duration).ok_or(
                Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                },
            )?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker_id] = available_at;
            runs[task_id] = Some(Run {
                start: current_time,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time: current_time,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if finish == current_time {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest future event that can change eligibility.
        let next_finish = running
            .iter()
            .map(|&(finish, _, _)| finish)
            .filter(|&time| time > current_time)
            .min();
        let next_release = (0..n)
            .filter(|&task_id| !started[task_id] && tasks[task_id].release > current_time)
            .map(|task_id| tasks[task_id].release)
            .min();
        let next_worker = worker_available
            .iter()
            .copied()
            .filter(|&time| time > current_time)
            .min();

        let next_time = [next_finish, next_release, next_worker]
            .into_iter()
            .flatten()
            .min();

        match next_time {
            Some(time) => current_time = time,
            None => {
                // With an acyclic graph, an unfinished task must eventually
                // become eligible; this branch is only a defensive safeguard.
                return Err(Error::Cycle {
                    blocked: (0..n).filter(|&task_id| !completed[task_id]).collect(),
                });
            }
        }
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}
