use std::str;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[&[u8]] = match field {
        Field::Month => &[
            b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN", b"JUL", b"AUG", b"SEP", b"OCT",
            b"NOV", b"DEC",
        ],
        Field::DayOfWeek => &[b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT"],
        _ => return None,
    };
    names.iter().position(|name| {
        name.len() == bytes.len()
            && name
                .iter()
                .zip(bytes)
                .all(|(a, b)| a.eq_ignore_ascii_case(b))
    }).map(|i| {
        if field == Field::Month {
            (i + 1) as u32
        } else {
            i as u32
        }
    })
}

fn parse_u32_number(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_endpoint(bytes: &[u8], field: Field, offset: usize) -> Result<(u32, bool), ParseError> {
    if !bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit) {
        let n = parse_u32_number(bytes, field, offset)?;
        let (min, max) = bounds(field);
        if n < min || n > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok((n, true));
    }
    if let Some(n) = name_value(field, bytes) {
        return Ok((n, false));
    }
    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_field(
    text: &str,
    field: Field,
    unrestricted: &mut bool,
) -> Result<Vec<bool>, ParseError> {
    let (min, max) = bounds(field);
    let mut selected = vec![false; (max + 1) as usize];

    for (offset, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    *unrestricted = matches!(field, Field::DayOfMonth | Field::DayOfWeek)
        && text.as_bytes() == b"*";

    let mut start = 0usize;
    loop {
        let end = text.as_bytes()[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(text.len());
        let item = &text.as_bytes()[start..end];
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: start });
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), start + slash + 1)
        } else {
            (item, None, 0)
        };

        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();
        if hyphens.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        if base.is_empty()
            || (step_bytes.is_some_and(|s| s.is_empty()))
            || (base == b"*" && !hyphens.is_empty())
        {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        let (left, right) = if let Some(&h) = hyphens.first() {
            if base[..h].is_empty() || base[h + 1..].is_empty()
                || base[..h] == b"*"
                || base[h + 1..] == b"*"
            {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            (&base[..h], Some(&base[h + 1..]))
        } else {
            (base, None)
        };

        // Confirm all base pieces are valid atoms before doing numeric validation.
        let valid_atom = |s: &[u8]| {
            (!s.is_empty() && s.iter().all(u8::is_ascii_digit))
                || name_value(field, s).is_some()
        };
        let wildcard = left == b"*" && right.is_none();
        if (!wildcard && !valid_atom(left))
            || right.is_some_and(|r| !valid_atom(r))
            || step_bytes.is_some_and(|s| !s.iter().all(u8::is_ascii_digit))
        {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }

        let (from, to, start_offset, end_offset) = if wildcard {
            (min, max, start, start)
        } else {
            let (a, a_numeric) = parse_endpoint(left, field, start)?;
            let left_offset = start;
            let (b, right_offset) = if let Some(r) = right {
                let ro = start + (r.as_ptr() as usize - item.as_ptr() as usize);
                let (n, _) = parse_endpoint(r, field, ro)?;
                (n, ro)
            } else {
                (a, left_offset)
            };
            if let Some(_) = right {
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: right_offset,
                    });
                }
            }
            let _ = a_numeric;
            (a, b, left_offset, right_offset)
        };

        let step = if let Some(s) = step_bytes {
            let n = parse_u32_number(s, field, step_offset)?;
            let width = max - min + 1;
            if n == 0 || n > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            n
        } else {
            0
        };

        if step == 0 {
            for n in from..=to {
                selected[n as usize] = true;
            }
        } else {
            let mut n = from;
            loop {
                selected[n as usize] = true;
                if n > to.saturating_sub(step) {
                    break;
                }
                n += step;
            }
        }

        let _ = (start_offset, end_offset);
        if end == text.len() {
            break;
        }
        start = end + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };
    let pieces: Vec<&str> = expanded.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }
    let mut unrestricted = [false; 2];
    let mut fields = Vec::with_capacity(5);
    for (i, piece) in pieces.iter().enumerate() {
        let mut flag = false;
        let parsed = parse_field(piece, FIELDS[i], &mut flag)?;
        if i == 2 {
            unrestricted[0] = flag;
        } else if i == 4 {
            unrestricted[1] = flag;
        }
        fields.push(parsed);
    }
    let mut it = fields.into_iter();
    Ok(Cron {
        selected: [
            it.next().unwrap(),
            it.next().unwrap(),
            it.next().unwrap(),
            it.next().unwrap(),
            it.next().unwrap(),
        ],
        unrestricted,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = field_index(field);
        self.selected[i]
            .iter()
            .enumerate()
            .filter_map(|(n, &yes)| yes.then_some(n as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }
        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];
        match (self.unrestricted[0], self.unrestricted[1]) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            if let Some(next) = increment_minute(current) {
                current = next;
            } else {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut current = before;
        loop {
            if let Some(prev) = decrement_minute(current) {
                current = prev;
            } else {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut current = start;
        loop {
            if current >= end {
                return Ok(FireWindow {
                    fires,
                    truncated: false,
                });
            }
            if self.matches_valid(current) {
                if fires.len() == cap {
                    return Ok(FireWindow {
                        fires,
                        truncated: true,
                    });
                }
                fires.push(current);
            }
            match increment_minute(current) {
                Some(next) => current = next,
                None => {
                    return Ok(FireWindow {
                        fires,
                        truncated: false,
                    })
                }
            }
        }
    }

    pub fn canonical(&self) -> String {
        let mut parts = Vec::new();
        for i in 0..5 {
            let field = FIELDS[i];
            let (min, max) = bounds(field);
            let all = (min..=max).all(|n| self.selected[i][n as usize]);
            if i == 2 && self.unrestricted[0] {
                parts.push("*".to_string());
            } else if i == 4 && self.unrestricted[1] {
                parts.push("*".to_string());
            } else if all {
                if i == 2 || i == 4 {
                    parts.push("*/1".to_string());
                } else {
                    parts.push("*".to_string());
                }
            } else {
                parts.push(
                    (min..=max)
                        .filter(|&n| self.selected[i][n as usize])
                        .map(|n| n.to_string())
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        parts.join(" ")
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut count = 0u32;
    for y in 2000..year {
        count += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        count += days_in_month(year, m) as u32;
    }
    count += day as u32 - 1;
    ((6 + count) % 7) as u8
}

fn increment_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        return Some(CivilDateTime { minute: at.minute + 1, ..at });
    }
    if at.hour < 23 {
        return Some(CivilDateTime {
            minute: 0,
            hour: at.hour + 1,
            ..at
        });
    }
    if at.day < days_in_month(at.year, at.month) {
        return Some(CivilDateTime {
            minute: 0,
            hour: 0,
            day: at.day + 1,
            ..at
        });
    }
    if at.month < 12 {
        return Some(CivilDateTime {
            minute: 0,
            hour: 0,
            day: 1,
            month: at.month + 1,
            ..at
        });
    }
    if at.year < 2399 {
        return Some(CivilDateTime {
            minute: 0,
            hour: 0,
            day: 1,
            month: 1,
            year: at.year + 1,
        });
    }
    None
}

fn decrement_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute > 0 {
        return Some(CivilDateTime { minute: at.minute - 1, ..at });
    }
    if at.hour > 0 {
        return Some(CivilDateTime {
            minute: 59,
            hour: at.hour - 1,
            ..at
        });
    }
    if at.day > 1 {
        return Some(CivilDateTime {
            minute: 59,
            hour: 23,
            day: at.day - 1,
            ..at
        });
    }
    if at.month > 1 {
        let month = at.month - 1;
        return Some(CivilDateTime {
            minute: 59,
            hour: 23,
            day: days_in_month(at.year, month),
            month,
            ..at
        });
    }
    if at.year > 2000 {
        let year = at.year - 1;
        return Some(CivilDateTime {
            minute: 59,
            hour: 23,
            day: 31,
            month: 12,
            year,
        });
    }
    None
}
