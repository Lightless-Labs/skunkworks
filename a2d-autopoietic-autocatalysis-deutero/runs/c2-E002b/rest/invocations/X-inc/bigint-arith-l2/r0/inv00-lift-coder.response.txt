use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented as [0] with negative=false.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over every separator error.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if !byte.is_ascii_digit() && byte != b'_' {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut first_bad_separator = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                first_bad_separator = Some(index);
                break;
            }
        }
        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess = None;
        let mut digits = Vec::new();

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte.is_ascii_digit() {
                if first_digit.is_none() {
                    first_digit = Some((byte, index));
                }
                digit_count += 1;
                if digit_count == max_digits.saturating_add(1) {
                    first_excess = Some(index);
                }
                digits.push(byte - b'0');
            }
        }

        if digit_count > 1 {
            let (digit, index) = first_digit.expect("nonempty magnitude");
            if digit == b'0' {
                return Err(Error::LeadingZero { index });
            }
        }
        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.expect("first excess digit exists"),
            });
        }

        digits.reverse();
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let is_zero = digits.iter().all(|&digit| digit == 0);
        Ok(Self {
            negative: negative && !is_zero,
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1).max(min_digits);
        let pad = magnitude_len.saturating_sub(self.digits.len());

        let mut result = String::new();
        if self.negative {
            result.push('-');
        }

        for position in 0..magnitude_len {
            if group > 0 && position > 0 && (magnitude_len - position) % group == 0 {
                result.push('_');
            }
            let digit = if position < pad {
                0
            } else {
                self.digits[self.digits.len() - 1 - (position - pad)]
            };
            result.push(char::from(b'0' + digit));
        }
        result
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_magnitude(self, rhs),
            (true, true) => Self::cmp_magnitude(rhs, self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = Self::add_magnitudes(&self.digits, &rhs.digits);
            Self::from_parts(self.negative, digits)
        } else {
            match Self::cmp_magnitude(self, rhs) {
                Ordering::Greater => {
                    Self::from_parts(self.negative, Self::sub_magnitudes(&self.digits, &rhs.digits))
                }
                Ordering::Less => {
                    Self::from_parts(rhs.negative, Self::sub_magnitudes(&rhs.digits, &self.digits))
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let index = i + j;
                let value = result[index] as u16 + a as u16 * b as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
            }

            let mut index = i + rhs.digits.len();
            while carry > 0 {
                if index == result.len() {
                    result.push(0);
                }
                let value = result[index] as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
                index += 1;
            }
        }

        Self::from_parts(self.negative != rhs.negative, result)
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn cmp_magnitude(lhs: &Self, rhs: &Self) -> Ordering {
        match lhs.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => lhs.digits.iter().rev().cmp(rhs.digits.iter().rev()),
            ordering => ordering,
        }
    }

    fn add_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len().max(rhs.len()) + 1);
        let mut carry = 0u8;
        for index in 0..lhs.len().max(rhs.len()) {
            let sum = lhs.get(index).copied().unwrap_or(0)
                + rhs.get(index).copied().unwrap_or(0)
                + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }
        if carry > 0 {
            result.push(carry);
        }
        result
    }

    // Requires lhs magnitude >= rhs magnitude.
    fn sub_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len());
        let mut borrow = 0i8;
        for (index, &left) in lhs.iter().enumerate() {
            let mut digit = left as i8 - borrow - rhs.get(index).copied().unwrap_or(0) as i8;
            if digit < 0 {
                digit += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(digit as u8);
        }
        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result
    }
}
