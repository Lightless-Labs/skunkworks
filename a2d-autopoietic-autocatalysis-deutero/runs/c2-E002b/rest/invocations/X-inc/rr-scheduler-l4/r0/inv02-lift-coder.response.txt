```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

#[derive(Clone, Debug)]
struct JobState {
    level: usize,
    burst_index: usize,
    remaining: u32,
    completion: Option<u64>,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        0
    } else {
        (total * 100 + count as u64 / 2) / count as u64
    }
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: EventKind::Arrival,
            index,
        })
        .collect();

    let mut states: Vec<JobState> = jobs
        .iter()
        .map(|job| JobState {
            level: job.level,
            burst_index: 0,
            remaining: job.bursts[0],
            completion: None,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut event_pos = 0usize;
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completed = 0usize;

    while completed < jobs.len() {
        events[event_pos..].sort_by_key(|event| {
            (
                event.time,
                match event.kind {
                    EventKind::Arrival => 0u8,
                    EventKind::Wake => 1u8,
                },
                event.index,
            )
        });

        while event_pos < events.len() && events[event_pos].time <= time {
            let event = events[event_pos];
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake => {
                    queues[states[event.index].level].push_back(event.index);
                }
            }
            event_pos += 1;
        }

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            if event_pos >= events.len() {
                break;
            }

            let next_time = events[event_pos].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let job_index = queues[level].pop_front().expect("selected queue is nonempty");
        let job_id = jobs[job_index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != job_id) {
            time += u64::from(switch_cost);
        }

        while event_pos < events.len() && events[event_pos].time <= time {
            let event = events[event_pos];
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake => {
                    queues[states[event.index].level].push_back(event.index);
                }
            }
            event_pos += 1;
        }

        let run_start = time;
        let consumed = states[job_index].remaining.min(quanta[level]);
        time += u64::from(consumed);
        let run_end = time;
        states[job_index].remaining -= consumed;

        let burst_finished = states[job_index].remaining == 0;
        let is_final_burst =
            burst_finished && states[job_index].burst_index + 1 == jobs[job_index].bursts.len();

        runs.push(Run {
            id: job_id,
            start: run_start,
            end: run_end,
            finished: is_final_burst,
            level,
        });

        if burst_finished && !is_final_burst {
            let io_position = states[job_index].burst_index + 1;
            events.push(Event {
                time: run_end + u64::from(jobs[job_index].bursts[io_position]),
                kind: EventKind::Wake,
                index: job_index,
            });
        }

        events[event_pos..].sort_by_key(|event| {
            (
                event.time,
                match event.kind {
                    EventKind::Arrival => 0u8,
                    EventKind::Wake => 1u8,
                },
                event.index,
            )
        });

        while event_pos < events.len() && events[event_pos].time <= run_end {
            let event = events[event_pos];
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push_back(event.index),
                EventKind::Wake => {
                    queues[states[event.index].level].push_back(event.index);
                }
            }
            event_pos += 1;
        }

        if is_final_burst {
            states[job_index].completion = Some(run_end);
            completed += 1;
        } else if burst_finished {
            let io_position = states[job_index].burst_index + 1;
            states[job_index].burst_index += 2;
            states[job_index].remaining = jobs[job_index].bursts[states[job_index].burst_index];
            states[job_index].level = level.saturating_sub(1);
            let _ = io_position;
        } else {
            states[job_index].level = (level + 1).min(quanta.len() - 1);
            queues[states[job_index].level].push_back(job_index);
        }

        last_dispatch_id = Some(job_id);
    }

    let end = states
        .iter()
        .filter_map(|state| state.completion)
        .max()
        .unwrap_or(0);

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap_or(0);
        let turnaround = completion - u64::from(job.arrival);
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&value| u64::from(value))
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&value| u64::from(value))
            .sum();
        let waiting = turnaround - cpu - io;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end,
    })
}
```
