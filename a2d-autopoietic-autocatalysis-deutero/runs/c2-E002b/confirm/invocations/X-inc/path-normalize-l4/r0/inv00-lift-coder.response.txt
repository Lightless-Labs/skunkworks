#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone, Copy)]
struct Component<'a> {
    text: &'a str,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<Component<'_>> {
    path.as_bytes()
        .split(|&b| b == b'/')
        .enumerate()
        .filter_map(|(start, _)| {
            let bytes = path.as_bytes();
            if start >= bytes.len() {
                return None;
            }
            None
        })
        .collect::<Vec<_>>()
}

// Return each nonempty slash-delimited component and its byte offset.
fn raw_components(path: &str) -> Vec<Component<'_>> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push(Component {
            text: &path[start..i],
            at: start,
        });
    }
    result
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    raw_components(path)
        .last()
        .is_some_and(|c| c.text == "." || c.text == "..")
}

fn render_absolute(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return "/".to_string();
    }
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn render_normalized(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if stack.is_empty() {
            return "/".to_string();
        }
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce_plain<'a>(
    parts: impl IntoIterator<Item = Component<'a>>,
    absolute: bool,
) -> Result<Vec<String>, PathError> {
    let mut stack: Vec<String> = Vec::new();
    for part in parts {
        match part.text {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|s| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: part.at });
                } else {
                    stack.push("..".to_string());
                }
            }
            name => stack.push(name.to_string()),
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let stack = reduce_plain(raw_components(path), absolute)?;
    Ok(render_normalized(&stack, absolute, intent))
}

fn validated_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => Ok(raw_components(&normalized)
            .into_iter()
            .map(|c| c.text.to_string())
            .collect()),
        Err(_) => Err(error),
    }
}

fn expand_tilde<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, 0));
    }
    let bytes = path.as_bytes();
    if bytes.len() == 1 || bytes[1] == b'/' {
        if home.is_none() {
            return Err(PathError::MissingHome);
        }
        return Ok((true, if bytes.len() > 1 { 2 } else { 1 }));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

fn resolve_core(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, consumed) = expand_tilde(path, home)?;

    let absolute = !tilde && path.starts_with('/');
    let mut stack = if absolute {
        Vec::new()
    } else if tilde {
        home.unwrap().to_vec()
    } else {
        cwd.to_vec()
    };

    let remainder = &path[consumed..];
    for part in raw_components(remainder) {
        let at = part.at + consumed;
        match part.text {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            name => stack.push(name.to_string()),
        }
    }
    Ok(render_absolute(&stack, intent))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(validated_context(h, PathError::InvalidHome)?),
        None => None,
    };
    resolve_core(path, &cwd_parts, home_parts.as_deref())
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        let canonical = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !canonical || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(first) = links[..index].iter().position(|prev| prev.path == link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct WorkItem {
    component: String,
    at: usize,
    source: Source,
}

#[derive(Clone, Copy)]
enum Source {
    Context,
    Request,
    Target,
}

fn make_work(path: &str, offset: usize, source: Source) -> Vec<WorkItem> {
    raw_components(&path[offset..])
        .into_iter()
        .map(|part| WorkItem {
            component: part.text.to_string(),
            at: part.at + offset,
            source,
        })
        .collect()
}

fn expand_with_links(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, consumed) = expand_tilde(path, home)?;

    let absolute = !tilde && path.starts_with('/');
    let mut stack: Vec<String> = if absolute {
        Vec::new()
    } else if tilde {
        home.unwrap().to_vec()
    } else {
        cwd.to_vec()
    };

    let mut work = Vec::new();
    if !absolute && !tilde {
        for component in cwd {
            work.push(WorkItem {
                component: component.clone(),
                at: 0,
                source: Source::Context,
            });
        }
    } else if tilde {
        for component in home.unwrap() {
            work.push(WorkItem {
                component: component.clone(),
                at: 0,
                source: Source::Context,
            });
        }
    }
    work.extend(make_work(path, consumed, Source::Request));

    // Context is already represented in stack; traverse it from root for link lookup.
    if !absolute && !tilde {
        stack.clear();
    } else if tilde {
        stack.clear();
    }

    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;
    let mut cursor = 0usize;

    while cursor < work.len() {
        let item = work[cursor].clone();
        cursor += 1;

        match item.component.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    let at = match item.source {
                        Source::Context => 0,
                        Source::Request | Source::Target => item.at,
                    };
                    return Err(PathError::Escape { at });
                }
                continue;
            }
            _ => {}
        }

        stack.push(item.component);
        let matched = render_absolute(&stack, false);
        if let Some(index) = links.iter().position(|link| link.path == matched) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);

            // Remove the link name; relative targets start at its parent.
            stack.pop();
            let target = &links[index].target;
            if target.starts_with('/') {
                stack.clear();
            }
            let mut target_work = make_work(target, 0, Source::Target);
            target_work.extend(work[cursor..].iter().cloned());
            work.truncate(cursor);
            work.extend(target_work);

            // Mark this expansion's completion after its target work has drained.
            // A sentinel is encoded as an empty component with a synthetic source.
            work.insert(
                cursor,
                WorkItem {
                    component: String::new(),
                    at: index,
                    source: Source::Context,
                },
            );
            continue;
        }
    }

    Ok(render_absolute(&stack, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(validated_context(h, PathError::InvalidHome)?),
        None => None,
    };
    validate_links(links)?;
    expand_with_links(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        max_expansions,
    )
}

fn absolute_components(path: &str) -> Vec<String> {
    raw_components(path)
        .into_iter()
        .map(|c| c.text.to_string())
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_string()));
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

fn resolved_with_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links(path, cwd, home, links, max_expansions)
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolved_with_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolved_with_links(target, cwd, home, links, max_expansions)?;
    let base_parts = absolute_components(&resolved_base);
    let target_parts = absolute_components(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probed) = probe else {
            continue;
        };
        let probed_parts = absolute_components(&probed);
        if probed_parts.len() > target_parts.len()
            || probed_parts
                .iter()
                .zip(target_parts.iter())
                .any(|(a, b)| a != b)
        {
            continue;
        }
        let suffix = &target_parts[probed_parts.len()..];
        let mut candidate = route(&base_parts, &absolute_components(&link.path));
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::<String>::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) else {
            continue;
        };
        if absolute_components(&resolved) == target_parts {
            valid.push(candidate);
        }
    }

    valid
        .into_iter()
        .min_by(|a, b| {
            a.len()
                .cmp(&b.len())
                .then_with(|| a.as_bytes().cmp(b.as_bytes()))
        })
        .ok_or(PathError::Escape { at: 0 })
}
