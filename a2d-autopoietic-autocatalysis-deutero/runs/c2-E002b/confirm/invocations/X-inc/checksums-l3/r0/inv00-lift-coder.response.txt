#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc32_state: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc32_state: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc32_state: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc32_state ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc32_state = crc32_byte(self.crc32_state, byte);
            self.adler_a = (self.adler_a + u32::from(byte)) % 65521;
            self.adler_b = (self.adler_b + self.adler_a) % 65521;
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        let mut adler = 0u32;
        for offset in 1..9 {
            crc = (crc << 4) | u32::from(hex_digit(bytes[offset], offset)?);
        }
        for offset in 11..19 {
            adler = (adler << 4) | u32::from(hex_digit(bytes[offset], offset)?);
        }
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_digit(byte: u8, offset: usize) -> Result<u8, SnapshotError> {
    match byte {
        b'0'..=b'9' => Ok(byte - b'0'),
        b'a'..=b'f' => Ok(byte - b'a' + 10),
        _ => Err(SnapshotError::Hex { offset }),
    }
}

fn crc32_byte(mut state: u32, byte: u8) -> u32 {
    state ^= u32::from(byte);
    for _ in 0..8 {
        let low_bit = state & 1;
        state >>= 1;
        if low_bit != 0 {
            state ^= 0xedb8_8320;
        }
    }
    state
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut state = 0xffff_ffff;
    for &byte in data {
        state = crc32_byte(state, byte);
    }
    state ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut preceding_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < preceding_end {
            return Err(SpanError::Overlap { span: index });
        }
        preceding_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_byte(a: &mut u32, b: &mut u32, byte: u8, high: &mut Option<u8>) {
    if let Some(first) = high.take() {
        let word = (u16::from(first) << 8) | u16::from(byte);
        *a = (*a + u32::from(word)) % 65535;
        *b = (*b + *a) % 65535;
    } else {
        *high = Some(byte);
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut high = None;
    for &byte in data {
        fletcher32_byte(&mut a, &mut b, byte, &mut high);
    }
    if let Some(first) = high {
        let word = u16::from(first) << 8;
        a = (a + u32::from(word)) % 65535;
        b = (b + a) % 65535;
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut high = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            fletcher32_byte(&mut a, &mut b, byte, &mut high);
        }
    }
    if let Some(first) = high {
        let word = u16::from(first) << 8;
        a = (a + u32::from(word)) % 65535;
        b = (b + a) % 65535;
    }
    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut register = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                register ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let top_bit = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if top_bit != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            register
        }
        Crc16::X25 => {
            for &byte in data {
                register ^= u16::from(byte);
                for _ in 0..8 {
                    let low_bit = register & 1;
                    register >>= 1;
                    if low_bit != 0 {
                        register ^= 0x8408;
                    }
                }
            }
            register ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut stream = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_high = None;
    let mut crc16_register = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);

        for &byte in selected {
            f16_a = (f16_a + u32::from(byte)) % 255;
            f16_b = (f16_b + f16_a) % 255;

            fletcher32_byte(&mut f32_a, &mut f32_b, byte, &mut f32_high);

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_register ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        let top_bit = crc16_register & 0x8000;
                        crc16_register = crc16_register.wrapping_shl(1);
                        if top_bit != 0 {
                            crc16_register ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_register ^= u16::from(byte);
                    for _ in 0..8 {
                        let low_bit = crc16_register & 1;
                        crc16_register >>= 1;
                        if low_bit != 0 {
                            crc16_register ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(first) = f32_high {
        let word = u16::from(first) << 8;
        f32_a = (f32_a + u32::from(word)) % 65535;
        f32_b = (f32_b + f32_a) % 65535;
    }

    if variant == Crc16::X25 {
        crc16_register ^= 0xffff;
    }

    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_register,
    })
}
