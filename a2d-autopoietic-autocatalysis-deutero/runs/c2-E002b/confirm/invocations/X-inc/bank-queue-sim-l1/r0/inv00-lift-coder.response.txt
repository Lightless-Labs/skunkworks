#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut queue: std::collections::VecDeque<usize> =
        std::collections::VecDeque::new();
    let mut next_arrival = 0usize;
    let mut finished: Option<(u64, usize)> = None;
    let mut previous_finish: Option<u64> = None;
    let mut completed = 0usize;

    while completed < customers.len() {
        let arrival_time = arrivals
            .get(next_arrival)
            .map(|&i| customers[i].arrival);
        let finish_time = finished.map(|(time, _)| time);

        let time = match (arrival_time, finish_time) {
            (Some(a), Some(f)) => a.min(f),
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (None, None) => break,
        };

        if let Some((f, index)) = finished {
            if f == time {
                events.push(Event {
                    time,
                    customer: customers[index].id,
                    kind: Kind::Finish,
                });
                finished = None;
                previous_finish = Some(time);
                completed += 1;
            }
        }

        while let Some(&index) = arrivals.get(next_arrival) {
            let customer = &customers[index];
            if customer.arrival != time {
                break;
            }
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            queue.push_back(index);
            next_arrival += 1;
        }

        if finished.is_none() {
            if let Some(index) = queue.pop_front() {
                let customer = &customers[index];
                let wake = previous_finish.map_or(true, |f| time > f);
                let after_service = time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    },
                )?;
                let finish = after_service
                    .checked_add(if wake { wake_cost } else { 0 })
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    })?;

                events.push(Event {
                    time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == time {
                    events.push(Event {
                        time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(time);
                    completed += 1;

                    while let Some(index) = queue.pop_front() {
                        let customer = &customers[index];
                        let wake = false;
                        let after_service =
                            time.checked_add(customer.service).ok_or(
                                Error::TimeOverflow {
                                    customer: customer.id,
                                    time,
                                },
                            )?;
                        let finish = after_service
                            .checked_add(0)
                            .ok_or(Error::TimeOverflow {
                                customer: customer.id,
                                time,
                            })?;

                        events.push(Event {
                            time,
                            customer: customer.id,
                            kind: Kind::Start { finish, wake },
                        });

                        if finish == time {
                            events.push(Event {
                                time,
                                customer: customer.id,
                                kind: Kind::Finish,
                            });
                            previous_finish = Some(time);
                            completed += 1;
                        } else {
                            finished = Some((finish, index));
                            break;
                        }
                    }
                } else {
                    finished = Some((finish, index));
                }
            }
        }
    }

    Ok(events)
}
