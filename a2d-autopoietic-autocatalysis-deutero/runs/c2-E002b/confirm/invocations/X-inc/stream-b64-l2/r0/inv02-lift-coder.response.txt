#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

fn encode_triplet(a: u8, b: Option<u8>, c: Option<u8>, output: &mut String) {
    let b_value = b.unwrap_or(0);
    let c_value = c.unwrap_or(0);

    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b_value >> 4)) as usize] as char);

    if b.is_some() {
        output.push(ALPHABET[(((b_value & 0x0f) << 2) | (c_value >> 6)) as usize] as char);
    } else {
        output.push('=');
    }

    if c.is_some() {
        output.push(ALPHABET[(c_value & 0x3f) as usize] as char);
    } else {
        output.push('=');
    }
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                encode_triplet(
                    self.tail[0],
                    Some(self.tail[1]),
                    Some(input[index - 1]),
                    &mut output,
                );
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            encode_triplet(
                input[index],
                Some(input[index + 1]),
                Some(input[index + 2]),
                &mut output,
            );
            index += 3;
        }

        let remaining = input.len() - index;
        self.tail_len = remaining;
        if remaining > 0 {
            self.tail[0] = input[index];
        }
        if remaining > 1 {
            self.tail[1] = input[index + 1];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => encode_triplet(self.tail[0], None, None, &mut output),
            2 => encode_triplet(self.tail[0], Some(self.tail[1]), None, &mut output),
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    pending_padding: Option<usize>,
    terminal: bool,
    offset: usize,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            pending_padding: None,
            terminal: false,
            offset: 0,
            finished: false,
            error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.error = Some(error.clone());
        Err(error)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let physical_offset = self.offset;
            self.offset += 1;

            if self.terminal {
                return self.fail(Error::BadPadding {
                    offset: physical_offset,
                });
            }

            if self.pending_padding.is_some() {
                if byte != b'=' {
                    return self.fail(Error::BadPadding {
                        offset: physical_offset,
                    });
                }

                let first = digit(self.group[0]).expect("validated digit");
                let second = digit(self.group[1]).expect("validated digit");
                output.push((first << 2) | (second >> 4));
                self.len = 0;
                self.pending_padding = None;
                self.terminal = true;
                continue;
            }

            if self.len < 2 && byte == b'=' {
                return self.fail(Error::BadPadding {
                    offset: physical_offset,
                });
            }

            if byte == b'=' && self.len == 2 {
                self.group[2] = b'=';
                self.offsets[2] = physical_offset;
                self.len = 3;
                self.pending_padding = Some(physical_offset);
                continue;
            }

            if byte == b'=' && self.len == 3 {
                let first = digit(self.group[0]).expect("validated digit");
                let second = digit(self.group[1]).expect("validated digit");
                let third = digit(self.group[2]).expect("validated digit");
                output.push((first << 2) | (second >> 4));
                output.push((second << 4) | (third >> 2));
                self.len = 0;
                self.terminal = true;
                continue;
            }

            let value = match digit(byte) {
                Some(value) => value,
                None => {
                    return self.fail(Error::InvalidByte {
                        offset: physical_offset,
                        byte,
                    });
                }
            };

            // A pending position-two '=' occupies group slot 2. The next
            // '=' completes xx==; an alphabet digit is handled above as an
            // immediate padding error.
            let slot = if self.pending_padding.is_some() { 3 } else { self.len };
            self.group[slot] = value;
            self.offsets[slot] = physical_offset;
            self.len += 1;

            if self.len == 4 {
                let a = self.group[0];
                let b = self.group[1];
                let c = self.group[2];
                let d = self.group[3];
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
                self.len = 0;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_padding {
            return self.fail(Error::BadPadding { offset });
        }

        if self.len != 0 {
            return self.fail(Error::Truncated {
                offset: self.offsets[0],
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }
}
