#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
}

struct Line {
    indent: usize,
    pieces: Vec<Piece>,
}

impl Line {
    fn content_len(&self) -> usize {
        self.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn is_empty(&self) -> bool {
        self.pieces.is_empty()
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_side = |j: usize| {
            j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
        };
        if i == 0 || !valid_side(i - 1) || !valid_side(i + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut segments: Vec<Vec<char>> = Vec::new();
    let mut current = Vec::new();
    let mut boundaries = Vec::new();
    i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let count = i - start;
            if count >= 2 {
                segments.push(std::mem::take(&mut current));
                boundaries.push(count - 1);
            } else {
                current.push('\n');
            }
        } else {
            current.push(normalized[i]);
            i += 1;
        }
    }
    segments.push(current);

    let mut output = Vec::new();
    for (segment_index, segment) in segments.iter().enumerate() {
        let words = parse_words(segment);
        let lines = layout_paragraph(&words, options);
        output.extend(lines);
        if segment_index < boundaries.len() {
            output.extend(std::iter::repeat(String::new()).take(boundaries[segment_index]));
        }
    }

    Ok(output)
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut breaks = Vec::new();

    for &c in segment {
        if is_separator(c) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    breaks: std::mem::take(&mut breaks),
                });
            }
        } else if c == '~' {
            breaks.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    if !chars.is_empty() {
        words.push(Word { chars, breaks });
    }
    words
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut lines: Vec<Line> = Vec::new();
    let mut line = Line {
        indent: options.first_indent,
        pieces: Vec::new(),
    };

    for word in words {
        let mut offset = 0;
        while offset < word.chars.len() {
            let capacity = options.width - line.indent;
            let current_len = line.content_len();
            let room = capacity.saturating_sub(current_len);
            let remaining = word.chars.len() - offset;
            let gap = if line.is_empty() { 0 } else { 1 };

            if remaining + gap <= room {
                if gap != 0 {
                    line.pieces.push(Piece {
                        text: String::new(),
                    });
                }
                append_word_suffix(&mut line, word, offset, word.chars.len());
                offset = word.chars.len();
                continue;
            }

            let prefix_room = room.saturating_sub(gap);
            let chosen = word
                .breaks
                .iter()
                .copied()
                .filter(|&b| b > offset && b < word.chars.len())
                .filter(|&b| b - offset + 1 <= prefix_room)
                .last();

            if let Some(end) = chosen {
                let used = gap + (end - offset) + 1;
                let unused = room.saturating_sub(used);
                let accept = if line.is_empty() {
                    true
                } else {
                    let baseline = capacity.saturating_sub(current_len);
                    unused * unused + options.hyphen_cost < baseline * baseline
                };
                if accept {
                    if gap != 0 {
                        line.pieces.push(Piece {
                            text: String::new(),
                        });
                    }
                    append_word_suffix(&mut line, word, offset, end);
                    line.pieces.last_mut().unwrap().text.push('-');
                    lines.push(line);
                    line = Line {
                        indent: options.rest_indent,
                        pieces: Vec::new(),
                    };
                    offset = end;
                    continue;
                }
            }

            if !line.is_empty() {
                lines.push(line);
                line = Line {
                    indent: options.rest_indent,
                    pieces: Vec::new(),
                };
                continue;
            }

            let capacity = options.width - line.indent;
            if remaining > capacity {
                append_word_suffix(&mut line, word, offset, offset + capacity);
                lines.push(line);
                line = Line {
                    indent: options.rest_indent,
                    pieces: Vec::new(),
                };
                offset += capacity;
            } else {
                append_word_suffix(&mut line, word, offset, word.chars.len());
                offset = word.chars.len();
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    let mut rendered = Vec::with_capacity(lines.len());
    for (line_index, line) in lines.iter().enumerate() {
        let content_len = line.content_len();
        let gaps = line.pieces.len().saturating_sub(1);
        let justify = options.justify && line_index + 1 < lines.len() && gaps > 0;
        let extra = if justify {
            options.width - line.indent - content_len
        } else {
            0
        };
        let each = if gaps > 0 { extra / gaps } else { 0 };
        let remainder = if gaps > 0 { extra % gaps } else { 0 };
        let start_gap = if gaps > 0 { line_index % gaps } else { 0 };

        let mut text = " ".repeat(line.indent);
        for (piece_index, piece) in line.pieces.iter().enumerate() {
            if piece_index > 0 {
                let gap_index = piece_index - 1;
                let bonus = usize::from(
                    remainder > 0 && (gap_index + gaps - start_gap) % gaps < remainder,
                );
                text.push_str(&" ".repeat(1 + each + bonus));
            }
            text.push_str(&piece.text);
        }
        rendered.push(text);
    }

    rendered
}

fn append_word_suffix(line: &mut Line, word: &Word, start: usize, end: usize) {
    let text: String = word.chars[start..end].iter().collect();
    line.pieces.push(Piece { text });
}
