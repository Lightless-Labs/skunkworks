#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
}

struct Route {
    segments: Vec<Segment>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { segments, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if route.segments.len() != request_segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_segment, request_segment) in
                route.segments.iter().zip(request_segments.iter())
            {
                match pattern_segment {
                    Segment::Static(value) if value == request_segment => {}
                    Segment::Static(_) => {
                        matches = false;
                        break;
                    }
                    Segment::Param(name) => {
                        params.push((name.clone(), request_segment.clone()));
                    }
                }
            }

            if !matches {
                continue;
            }

            let replace = match best {
                None => true,
                Some((current, _)) => route_is_more_specific(route, current),
            };

            if replace {
                best = Some((route, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn route_is_more_specific(candidate: &Route, current: &Route) -> bool {
    for (a, b) in candidate.segments.iter().zip(current.segments.iter()) {
        let rank_a = match a {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
        };
        let rank_b = match b {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
        };
        if rank_a != rank_b {
            return rank_a > rank_b;
        }
    }
    candidate.id < current.id
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut segments = Vec::new();
    let mut seen_names = std::collections::HashSet::new();
    let mut start = 1;

    for end in 1..=bytes.len() {
        if end == bytes.len() || bytes[end] == b'/' {
            if end == start {
                return Err(Error::InvalidPattern { at: start });
            }

            let raw = &pattern[start..end];
            let first = bytes[start];

            if first == b':' {
                let name = &raw[1..];
                if !valid_capture_name(name) || !seen_names.insert(name.to_string()) {
                    return Err(Error::InvalidPattern { at: start });
                }
                segments.push(Segment::Param(name.to_string()));
            } else if first == b'*' {
                return Err(Error::InvalidPattern { at: start });
            } else {
                let decoded = decode_static_segment(raw, start)
                    .map_err(|at| Error::InvalidPattern { at })?;
                segments.push(Segment::Static(decoded));
            }

            if end < bytes.len() {
                start = end + 1;
            }
        }
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;

    for end in 1..=bytes.len() {
        if end == bytes.len() || bytes[end] == b'/' {
            if end == start {
                return Err(Error::InvalidPath { at: start });
            }

            let decoded = decode_static_segment(&path[start..end], start)
                .map_err(|at| Error::InvalidPath { at })?;
            segments.push(decoded);

            if end < bytes.len() {
                start = end + 1;
            }
        }
    }

    if bytes.last() == Some(&b'/') {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_static_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }

            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}
