#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
    level: usize,
}

#[derive(Clone, Debug)]
struct JobState {
    burst_position: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut states: Vec<JobState> = jobs
        .iter()
        .map(|job| JobState {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: EventKind::Arrival,
            index,
            level: job.level,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed = 0usize;

    while completed < jobs.len() {
        admit_events(&mut events, time, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if events.is_empty() {
                break;
            }
            let next_time = events.iter().map(|event| event.time).min().unwrap();
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let job_index = queues[level].pop_front().unwrap();
        let job = &jobs[job_index];

        if last_dispatch_id.is_some_and(|last_id| last_id != job.id) {
            time += u64::from(switch_cost);
        }
        admit_events(&mut events, time, &mut queues);

        let quantum = quanta[level];
        let consumed = states[job_index].remaining.min(quantum);
        let start = time;
        time += u64::from(consumed);
        let end = time;
        let burst_finished = consumed == states[job_index].remaining;
        let is_final_burst = burst_finished && states[job_index].burst_position + 1 == job.bursts.len();
        let finished = is_final_burst;

        runs.push(Run {
            id: job.id,
            start,
            end,
            finished,
            level,
        });

        states[job_index].remaining -= consumed;
        admit_events(&mut events, end, &mut queues);

        if finished {
            states[job_index].completion = Some(end);
            completed += 1;
        } else if burst_finished {
            let io_position = states[job_index].burst_position + 1;
            let wake_time = end + u64::from(job.bursts[io_position]);
            states[job_index].burst_position += 2;
            states[job_index].remaining = job.bursts[states[job_index].burst_position];
            let promoted_level = level.saturating_sub(1);
            events.push(Event {
                time: wake_time,
                kind: EventKind::Wake,
                index: job_index,
                level: promoted_level,
            });
            admit_events(&mut events, end, &mut queues);
        } else {
            states[job_index].level = (level + 1).min(quanta.len() - 1);
            queues[states[job_index].level].push_back(job_index);
        }

        last_dispatch_id = Some(job.id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap_or(time);
        let turnaround = completion - u64::from(job.arrival);
        let cpu_and_io: u64 = job.bursts.iter().map(|&burst| u64::from(burst)).sum();
        let waiting = turnaround - cpu_and_io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}

use std::collections::VecDeque;

fn admit_events(events: &mut Vec<Event>, time: u64, queues: &mut [VecDeque<usize>]) {
    events.sort_by_key(|event| {
        let kind_order = match event.kind {
            EventKind::Arrival => 0u8,
            EventKind::Wake => 1u8,
        };
        (event.time, kind_order, event.index)
    });

    let mut admitted = Vec::new();
    let mut pending = Vec::new();
    for event in events.drain(..) {
        if event.time <= time {
            admitted.push(event);
        } else {
            pending.push(event);
        }
    }
    *events = pending;

    for event in admitted {
        queues[event.level].push_back(event.index);
    }
}
