```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.num as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        Self::from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = (self.num as i128) * (rhs.den as i128);
        let d = (self.den as i128) * (rhs.num as i128);
        Self::from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        Self::from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }

        let num_start = 0;
        let num_digits_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_digits_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let mut slash = None;
        let mut den_digits_start = 0;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            den_digits_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_digits_start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let num_token = &bytes[num_start..num_digits_start + (if negative { 1 } else { 0 }) + (num_digits_start..)
            .take_while(|&j| j < bytes.len() && bytes[j].is_ascii_digit())
            .count()];
        let num_digit_count = num_token.len() - usize::from(negative);
        if num_digit_count >= 2 && bytes[num_digits_start] == b'0' {
            return Err(parse_error(num_digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_token == b"-0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        let denominator_token = slash.map(|_| &bytes[den_digits_start..]);
        if let Some(token) = denominator_token {
            if token.len() >= 2 && token[0] == b'0' {
                return Err(parse_error(den_digits_start + 1, ParseKind::NonCanonical));
            }
        }

        let num_mag = parse_magnitude(&num_token[usize::from(negative)..]);
        let num_limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag > num_limit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let den_mag = if let Some(token) = denominator_token {
            let value = parse_magnitude(token);
            if value > i64::MAX as u128 {
                return Err(parse_error(den_digits_start, ParseKind::OutOfRange));
            }
            value as u64
        } else {
            1
        };

        if denominator_token.is_some() && den_mag == 0 {
            return Err(parse_error(den_digits_start, ParseKind::ZeroDenominator));
        }

        if let Some(slash_at) = slash {
            if den_mag == 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag as u128) > 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        Ok(Self {
            num: signed_num,
            den: den_mag as i64,
        })
    }

    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;

        if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }
}

fn parse_magnitude(bytes: &[u8]) -> u128 {
    bytes.iter().fold(0u128, |acc, &b| {
        acc.saturating_mul(10).saturating_add((b - b'0') as u128)
    })
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
