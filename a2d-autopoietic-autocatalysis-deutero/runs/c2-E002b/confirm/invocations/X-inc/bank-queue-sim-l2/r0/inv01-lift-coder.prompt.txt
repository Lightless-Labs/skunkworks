    ZeroServerId { index: usize },

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|s| ServerState {
            id: s.id,
            opens: s.opens,
            overhead: s.overhead,
            opened: false,
            busy_until: None,
            idle_since: s.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut arrival_index = 0usize;
    let mut completed = 0usize;
    let mut time = 0u64;

    loop {
        // Finish events first, ordered by ascending server ID.
        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (s.busy_until == Some(time)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| states[i].id);
        for i in finishing {
            // Every occupied server has a customer; locate the active one
            // through the scheduled finish record maintained below.
            let server_id = states[i].id;
            let customer_id = active_customer_for_server(&events, server_id, time);
            events.push(Event::Finish {
                time,
                customer: customer_id,
                server: server_id,
            });
            states[i].busy_until = None;
            states[i].idle_since = time;
            completed += 1;
        }

        // Open events next, ordered by descending server ID.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.opens == time).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time,
                server: states[i].id,
            });
        }

        // Add this tick's arrivals in descending customer ID order.
        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            let customer = arrivals[arrival_index];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer);
            arrival_index += 1;
        }

        // Dispatch repeatedly, recalculating the best server after each start.
        while !waiting.is_empty() {
            let candidate = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = candidate else {
                break;
            };

            let customer = waiting.remove(0);
            let server_id = states[server_index].id;
            let finish = customer
                .service
                .checked_add(time)
                .and_then(|t| t.checked_add(states[server_index].overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
                completed += 1;
            } else {
                states[server_index].busy_until = Some(finish);
                // Record the active customer association as an internal marker.
                events.push(Event::Start {
                    time: finish,
                    customer: customer.id,
                    server: server_id,
                    finish,
                });
                // The marker is removed before returning; it is only used to
                // recover the customer ID at the scheduled Finish phase.
            }
        }

        if completed == customers.len() && states.iter().all(|s| s.opened) {
            break;
        }

        let next_arrival = arrivals.get(arrival_index).map(|c| c.arrival);
        let next_open = states
            .iter()
            .filter(|s| !s.opened)
            .map(|s| s.opens)
            .min();
        let next_finish = states.iter().filter_map(|s| s.busy_until).min();

        let next = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&t| t > time)
            .min();

        match next {
            Some(t) => time = t,
            None => break,
        }
    }

    // Remove the internal markers used to retain active customer IDs.
    events.retain(|event| {
        !matches!(
            event,
            Event::Start {
                time: marker_time,
                finish,
                ..
            } if marker_time == finish && marker_time > 0
        )
    });

    Ok(events)
}

fn active_customer_for_server(events: &[Event], server: u32, finish_time: u64) -> u32 {
    events
        .iter()
        .rev()
        .find_map(|event| match event {
            Event::Start {
                customer,
                server: start_server,
                finish,
                ..
            } if *start_server == server && *finish == finish_time => Some(*customer),
            _ => None,
        })
        .unwrap_or(0)
}
