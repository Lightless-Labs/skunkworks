#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn push_text(word: &mut Word, text: &str, active_glob: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn first_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    push_text(&mut current, &input[i..next_i], false);
                }
                i = next_i;
            }
            Quote::Double(open) => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let n = input[next_i..].chars().next().unwrap();
                        let after_n = next_i + n.len_utf8();
                        if matches!(n, '"' | '\\' | '$') {
                            push_text(&mut current, &input[next_i..after_n], false);
                            i = after_n;
                        } else if n == '\n' {
                            i = after_n;
                        } else {
                            push_text(&mut current, "\\", false);
                            i = next_i;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                    '$' => {
                        if input[next_i..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if input[next_i..].starts_with('{') {
                            let dollar = i;
                            let content_start = next_i + 1;
                            let mut j = content_start;
                            let mut close = None;
                            while j < input.len() {
                                let ch = input[j..].chars().next().unwrap();
                                let after = j + ch.len_utf8();
                                if ch == '\\' {
                                    if after < input.len() {
                                        let escaped = input[after..].chars().next().unwrap();
                                        j = after + escaped.len_utf8();
                                    } else {
                                        j = after;
                                    }
                                } else if ch == '}' {
                                    close = Some(j);
                                    break;
                                } else {
                                    j = after;
                                }
                            }
                            let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                            let body = &input[content_start..close];
                            let (name, default) = if let Some(separator) = body.find(":-") {
                                (&body[..separator], Some(&body[separator + 2..]))
                            } else {
                                (body, None)
                            };
                            if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                                || !name.chars().all(is_name_char)
                            {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            let value = match default {
                                Some(text) => match first_env(env, name) {
                                    Some(v) if !v.is_empty() => v,
                                    _ => {
                                        let decoded = decode_default(text);
                                        push_text(&mut current, &decoded, false);
                                        active = true;
                                        i = close + 1;
                                        continue;
                                    }
                                },
                                None => first_env(env, name).unwrap_or(""),
                            };
                            push_text(&mut current, value, false);
                            active = true;
                            i = close + 1;
                        } else if next_i < input.len() {
                            let n = input[next_i..].chars().next().unwrap();
                            if is_name_start(n) {
                                let mut end = next_i + n.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if !is_name_char(ch) {
                                        break;
                                    }
                                    end += ch.len_utf8();
                                }
                                let name = &input[next_i..end];
                                push_text(&mut current, first_env(env, name).unwrap_or(""), false);
                                active = true;
                                i = end;
                            } else {
                                push_text(&mut current, "$", false);
                                active = true;
                                i = next_i;
                            }
                        } else {
                            push_text(&mut current, "$", false);
                            active = true;
                            i = next_i;
                        }
                    }
                    _ => {
                        push_text(&mut current, &input[i..next_i], false);
                        i = next_i;
                    }
                }
                let _ = open;
            }
            Quote::None => {
                match c {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(current);
                            current = Word {
                                text: String::new(),
                                glob: Vec::new(),
                            };
                            active = false;
                        }
                        i = next_i;
                    }
                    '\'' => {
                        active = true;
                        quote = Quote::Single(i);
                        i = next_i;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double(i);
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let n = input[next_i..].chars().next().unwrap();
                        let after_n = next_i + n.len_utf8();
                        if n == '\n' {
                            i = after_n;
                        } else {
                            push_text(&mut current, &input[next_i..after_n], false);
                            active = true;
                            i = after_n;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                    '$' => {
                        if input[next_i..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if input[next_i..].starts_with('{') {
                            let dollar = i;
                            let content_start = next_i + 1;
                            let mut j = content_start;
                            let mut close = None;
                            while j < input.len() {
                                let ch = input[j..].chars().next().unwrap();
                                let after = j + ch.len_utf8();
                                if ch == '\\' {
                                    if after < input.len() {
                                        let escaped = input[after..].chars().next().unwrap();
                                        j = after + escaped.len_utf8();
                                    } else {
                                        j = after;
                                    }
                                } else if ch == '}' {
                                    close = Some(j);
                                    break;
                                } else {
                                    j = after;
                                }
                            }
                            let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                            let body = &input[content_start..close];
                            let (name, default) = if let Some(separator) = body.find(":-") {
                                (&body[..separator], Some(&body[separator + 2..]))
                            } else {
                                (body, None)
                            };
                            if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                                || !name.chars().all(is_name_char)
                            {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            let value = match default {
                                Some(text) => match first_env(env, name) {
                                    Some(v) if !v.is_empty() => v,
                                    _ => {
                                        let decoded = decode_default(text);
                                        push_text(&mut current, &decoded, true);
                                        active = true;
                                        i = close + 1;
                                        continue;
                                    }
                                },
                                None => first_env(env, name).unwrap_or(""),
                            };
                            push_text(&mut current, value, true);
                            active = true;
                            i = close + 1;
                        } else if next_i < input.len() {
                            let n = input[next_i..].chars().next().unwrap();
                            if is_name_start(n) {
                                let mut end = next_i + n.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if !is_name_char(ch) {
                                        break;
                                    }
                                    end += ch.len_utf8();
                                }
                                let name = &input[next_i..end];
                                push_text(&mut current, first_env(env, name).unwrap_or(""), true);
                                active = true;
                                i = end;
                            } else {
                                push_text(&mut current, "$", false);
                                active = true;
                                i = next_i;
                            }
                        } else {
                            push_text(&mut current, "$", false);
                            active = true;
                            i = next_i;
                        }
                    }
                    _ => {
                        push_text(&mut current, &input[i..next_i], true);
                        active = true;
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(open) => return Err(SplitError::UnterminatedSingle { pos: open }),
        Quote::Double(open) => return Err(SplitError::UnterminatedDouble { pos: open }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}
