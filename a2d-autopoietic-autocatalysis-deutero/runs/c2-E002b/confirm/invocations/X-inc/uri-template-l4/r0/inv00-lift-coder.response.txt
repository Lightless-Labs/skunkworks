#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Question | Self::Ampersand => '&',
        }
    }

    fn prefix(self) -> char {
        match self {
            Self::Fragment => '#',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Question => '?',
            Self::Ampersand => '&',
            Self::Simple | Self::Reserved => '\0',
        }
    }

    fn named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Question | Self::Ampersand)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }

    fn semicolon(self) -> bool {
        self == Self::Semicolon
    }
}

#[derive(Clone)]
struct Specifier<'a> {
    name: &'a str,
    pos: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

fn name_char_first(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn name_char_rest(b: u8) -> bool {
    name_char_first(b) || b.is_ascii_digit()
}

fn parse_specifier<'a>(text: &'a str, start: usize) -> Result<Specifier<'a>, Error> {
    if text.is_empty() {
        return Err(Error::EmptyVariable { pos: start });
    }

    let bytes = text.as_bytes();
    let modifier_at = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(bytes.len());
    if name_end == 0 {
        return Err(Error::InvalidName { pos: start });
    }

    if !name_char_first(bytes[0]) {
        return Err(Error::InvalidName { pos: start });
    }
    for (i, &b) in bytes[..name_end].iter().enumerate().skip(1) {
        if !name_char_rest(b) {
            return Err(Error::InvalidName { pos: start + i });
        }
    }

    let modifier = if let Some(i) = modifier_at {
        let suffix = &bytes[i..];
        if suffix == b"*" {
            Modifier::Explode
        } else if suffix.len() >= 2
            && suffix[0] == b':'
            && suffix.len() <= 5
            && suffix[1].is_ascii_digit()
            && suffix[1] != b'0'
            && suffix[1..].iter().all(u8::is_ascii_digit)
        {
            let mut n = 0usize;
            for &b in &suffix[1..] {
                n = n * 10 + (b - b'0') as usize;
            }
            if (1..=9999).contains(&n) {
                Modifier::Prefix(n)
            } else {
                return Err(Error::InvalidModifier { pos: start + i });
            }
        } else {
            return Err(Error::InvalidModifier { pos: start + i });
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name: &text[..name_end],
        pos: start,
        modifier,
    })
}

fn validate_template<'a>(template: &'a str) -> Result<Vec<(usize, usize, Operator, Vec<Specifier<'a>>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };
                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                let mut operator = Operator::Simple;
                if content_start < close {
                    operator = match bytes[content_start] {
                        b'+' => Operator::Reserved,
                        b'#' => Operator::Fragment,
                        b'.' => Operator::Dot,
                        b'/' => Operator::Slash,
                        b';' => Operator::Semicolon,
                        b'?' => Operator::Question,
                        b'&' => Operator::Ampersand,
                        b'=' | b',' | b'!' | b'@' | b'|' => {
                            return Err(Error::UnsupportedOperator { pos: content_start });
                        }
                        _ => Operator::Simple,
                    };
                    if operator != Operator::Simple {
                        content_start += 1;
                    }
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let content = &template[content_start..close];
                let mut specs = Vec::new();
                let mut local_start = 0;
                let mut seen: Vec<&str> = Vec::new();
                loop {
                    let next_comma = content.as_bytes()[local_start..]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| local_start + offset);
                    let end = next_comma.unwrap_or(content.len());
                    let spec = parse_specifier(&content[local_start..end], content_start + local_start)?;
                    if seen.iter().any(|&name| name == spec.name) {
                        return Err(Error::DuplicateVariable { pos: spec.pos });
                    }
                    seen.push(spec.name);
                    specs.push(spec);
                    match next_comma {
                        Some(comma) => local_start = comma + 1,
                        None => break,
                    }
                }

                expressions.push((open, close + 1, operator, specs));
                i = close + 1;
            }
            _ => i += 1,
        }
    }
    Ok(expressions)
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || (reserved
                && matches!(
                    b,
                    b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                        | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
                ));
        if safe {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn prefix_chars(value: &str, count: usize) -> &str {
    match value.char_indices().nth(count) {
        Some((index, _)) => &value[..index],
        None => value,
    }
}

fn scalar_field(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, operator.reserved());
    if operator.named() {
        if operator.semicolon() && encoded.is_empty() {
            name.to_owned()
        } else {
            format!("{name}={encoded}")
        }
    } else {
        encoded
    }
}

fn expand_expression(
    operator: Operator,
    specs: &[Specifier<'_>],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let mut fields = Vec::new();

    for spec in specs {
        let value = match selected_value(spec.name, vars) {
            Some(value) => value,
            None => continue,
        };

        if let Modifier::Prefix(count) = spec.modifier {
            match value {
                Value::Scalar(_) => {}
                Value::List(_) | Value::Map(_) => {
                    return Err(Error::TypeMismatch { pos: spec.pos });
                }
            }
            if let Value::Scalar(s) = value {
                let shortened = prefix_chars(s, count);
                fields.push(scalar_field(spec.name, shortened, operator));
            }
            continue;
        }

        let explode = matches!(spec.modifier, Modifier::Explode);
        match value {
            Value::Scalar(s) => fields.push(scalar_field(spec.name, s, operator)),
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if explode {
                    for item in items {
                        fields.push(scalar_field(spec.name, item, operator));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, operator.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    if operator.named() {
                        fields.push(if operator.semicolon() && joined.is_empty() {
                            spec.name.to_owned()
                        } else {
                            format!("{}={joined}", spec.name)
                        });
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(key, _)| key == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }

                if explode {
                    for (key, value) in entries {
                        let key = encode(key, operator.reserved());
                        let value = encode(value, operator.reserved());
                        if operator.semicolon() && value.is_empty() {
                            fields.push(key);
                        } else {
                            fields.push(format!("{key}={value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, operator.reserved()));
                        components.push(encode(value, operator.reserved()));
                    }
                    let joined = components.join(",");
                    if operator.named() {
                        fields.push(if operator.semicolon() && joined.is_empty() {
                            spec.name.to_owned()
                        } else {
                            format!("{}={joined}", spec.name)
                        });
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    let prefix = operator.prefix();
    if prefix != '\0' {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, operator, specs) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(operator, &specs, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
