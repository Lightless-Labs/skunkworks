pub fn encode(input: &[u8], checked: bool) -> String {
    let mut output = String::new();
    let mut remainder = 0usize;
    let mut accumulator = 0u32;
    let mut bits = 0u8;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let value = ((accumulator >> bits) & 0x1f) as usize;
            output.push(DATA_ALPHABET[value] as char);
            remainder = (remainder * 32 + value) % 37;
        }
    }

    if bits != 0 {
        let value = ((accumulator << (5 - bits)) & 0x1f) as usize;
        output.push(DATA_ALPHABET[value] as char);

    }

    if checked {
        output.push(CHECK_ALPHABET[remainder] as char);
    }

    output
}
