#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut field_is_quoted = false;
    let mut quote_opening = 0usize;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        match byte {
            b'\r' if bytes.get(pos + 1) != Some(&b'\n') && !field_is_quoted => {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: pos,
                });
            }
            b'"' if field_is_quoted => {
                if bytes.get(pos + 1) == Some(&b'"') {
                    field.push('"');
                    pos += 2;
                } else {
                    field_is_quoted = false;
                    pos += 1;

                    if pos < bytes.len()
                        && bytes[pos] != b','
                        && bytes[pos] != b'\n'
                        && bytes[pos] != b'\r'
                    {
                        return Err(CsvError::TrailingByte {
                            record: record_index,
                            byte: pos,
                        });
                    }
                }
            }
            b'"' if field.is_empty() => {
                field_is_quoted = true;
                quote_opening = pos;
                pos += 1;
            }
            b'"' => {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
            b',' if !field_is_quoted => {
                record.push(std::mem::take(&mut field));
                pos += 1;
            }
            b'\n' if !field_is_quoted => {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                pos += 1;
            }
            b'\r' if !field_is_quoted => {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                pos += 2;
            }
            _ => {
                let ch = input[pos..].chars().next().expect("valid UTF-8");
                field.push(ch);
                pos += ch.len_utf8();
            }
        }
    }

    if field_is_quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_opening,
        });
    }

    if !field.is_empty() || !record.is_empty() || bytes.last() == Some(&b',') {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
