#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion_times = vec![None; jobs.len()];
    let mut ready = std::collections::VecDeque::new();
    let mut event_cursor = 0;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();

    while completion_times.iter().any(Option::is_none) {
        while event_cursor < events.len()
            && u64::from(jobs[events[event_cursor]].arrival) <= time
        {
            let index = events[event_cursor];
            if jobs[index].ticks == 0 {
                completion_times[index] = Some(u64::from(jobs[index].arrival));
            } else {
                ready.push_back(index);
            }
            event_cursor += 1;
        }

        if let Some(index) = ready.pop_front() {
            let id = jobs[index].id;
            if last_dispatched.is_some_and(|previous| previous != id) {
                time += u64::from(switch_cost);
            }

            while event_cursor < events.len()
                && u64::from(jobs[events[event_cursor]].arrival) <= time
            {
                let arriving = events[event_cursor];
                if jobs[arriving].ticks == 0 {
                    completion_times[arriving] = Some(u64::from(jobs[arriving].arrival));
                } else {
                    ready.push_back(arriving);
                }
                event_cursor += 1;
            }

            let start = time;
            let consumed = remaining[index].min(quantum);
            time += u64::from(consumed);
            remaining[index] -= consumed;
            let finished = remaining[index] == 0;

            runs.push(Run {
                id,
                start,
                end: time,
                finished,
            });

            while event_cursor < events.len()
                && u64::from(jobs[events[event_cursor]].arrival) <= time
            {
                let arriving = events[event_cursor];
                if jobs[arriving].ticks == 0 {
                    completion_times[arriving] = Some(u64::from(jobs[arriving].arrival));
                } else {
                    ready.push_back(arriving);
                }
                event_cursor += 1;
            }

            if finished {
                completion_times[index] = Some(time);
            } else {
                ready.push_back(index);
            }
            last_dispatched = Some(id);
        } else if event_cursor < events.len() {
            let next_time = u64::from(jobs[events[event_cursor]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
        }
    }

    let completion = jobs
        .iter()
        .zip(completion_times)
        .map(|(job, completed)| (job.id, completed.expect("all jobs completed")))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
