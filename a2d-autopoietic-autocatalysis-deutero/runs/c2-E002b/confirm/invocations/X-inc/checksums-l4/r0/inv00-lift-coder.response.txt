#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

fn crc32_update_raw(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % 65_521;
        *b = (*b + *a) % 65_521;
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_update_raw(0xffff_ffff, data) ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index != 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update_raw(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc_raw = crc32_update_raw(self.crc_raw, data);
        adler_update(&mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let crc = parse_hex(bytes, 1, 9)?;
        let adler = parse_hex(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

fn parse_hex(bytes: &[u8], start: usize, end: usize) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => bytes[offset] - b'0',
            b'a'..=b'f' => bytes[offset] - b'a' + 10,
            _ => return Err(SnapshotError::Hex { offset }),
        };
        value = (value << 4) | digit as u32;
    }
    Ok(value)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

fn fletcher32_update(a: &mut u32, b: &mut u32, pending: &mut Option<u8>, data: &[u8]) {
    let mut index = 0;
    if let Some(high) = pending.take() {
        if let Some(&low) = data.first() {
            let word = ((high as u32) << 8) | low as u32;
            *a = (*a + word) % 65_535;
            *b = (*b + *a) % 65_535;
            index = 1;
        } else {
            *pending = Some(high);
            return;
        }
    }
    while index + 1 < data.len() {
        let word = ((data[index] as u32) << 8) | data[index + 1] as u32;
        *a = (*a + word) % 65_535;
        *b = (*b + *a) % 65_535;
        index += 2;
    }
    if index < data.len() {
        *pending = Some(data[index]);
    }
}

fn fletcher32_finish(a: u32, b: u32, pending: Option<u8>) -> u32 {
    let (mut a, mut b) = (a, b);
    if let Some(high) = pending {
        let word = (high as u32) << 8;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    (b << 16) | a
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (0, 0);
    let mut pending = None;
    fletcher32_update(&mut a, &mut b, &mut pending, data);
    fletcher32_finish(a, b, pending)
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let (mut a, mut b) = (0, 0);
    let mut pending = None;
    for span in spans {
        fletcher32_update(
            &mut a,
            &mut b,
            &mut pending,
            &data[span.start..span.end],
        );
    }
    Ok(fletcher32_finish(a, b, pending))
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc: u16 = if variant == Crc16::False { 0xffff } else { 0 };
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000 != 0;
                    crc = crc.wrapping_shl(1);
                    if top {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc: u16 = 0xffff;
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1 != 0;
                    crc >>= 1;
                    if low {
                        crc ^= 0x8408;
                    }
                }
            }
            crc ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let (mut adler_a, mut adler_b) = (1, 0);
    let (mut f16_a, mut f16_b) = (0u32, 0u32);
    let (mut f32_a, mut f32_b) = (0u32, 0u32);
    let mut f32_pending = None;

    let mut c16 = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update_raw(crc, selected);
        adler_update(&mut adler_a, &mut adler_b, selected);

        for &byte in selected {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;
        }

        fletcher32_update(&mut f32_a, &mut f32_b, &mut f32_pending, selected);

        match variant {
            Crc16::Xmodem | Crc16::False => {
                for &byte in selected {
                    c16 ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = c16 & 0x8000 != 0;
                        c16 = c16.wrapping_shl(1);
                        if top {
                            c16 ^= 0x1021;
                        }
                    }
                }
            }
            Crc16::X25 => {
                for &byte in selected {
                    c16 ^= byte as u16;
                    for _ in 0..8 {
                        let low = c16 & 1 != 0;
                        c16 >>= 1;
                        if low {
                            c16 ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b << 8) | f16_a) as u16,
        fletcher32: fletcher32_finish(f32_a, f32_b, f32_pending),
        crc16: c16,
    })
}

fn crc32_zero_byte(mut value: u32) -> u32 {
    for _ in 0..8 {
        let low = value & 1;
        value >>= 1;
        if low != 0 {
            value ^= 0xedb8_8320;
        }
    }
    value
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut result = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            result ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    result
}

fn gf2_matrix_square(matrix: &[u32; 32]) -> [u32; 32] {
    let mut square = [0u32; 32];
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
    square
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut operator = [0u32; 32];
    for bit in 0..32 {
        operator[bit] = crc32_zero_byte(1u32 << bit);
    }

    let mut result = left;
    let mut length = right_len;
    while length != 0 {
        if length & 1 != 0 {
            result = gf2_matrix_times(&operator, result);
        }
        length >>= 1;
        if length != 0 {
            operator = gf2_matrix_square(&operator);
        }
    }
    result ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn selected_length(spans: &[Span]) -> usize {
    spans.iter().map(|span| span.end - span.start).sum()
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let hashes = checksums_spans(data, spans)?;
    let len = selected_length(spans);
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len as u64, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let len = parse_hex(bytes, 4, 20).map_err(|error| match error {
        SnapshotError::Hex { offset } => VerifyError::Hex { offset },
        _ => VerifyError::Shape,
    })? as u64;
    let crc = parse_hex(bytes, 25, 33).map_err(|error| match error {
        SnapshotError::Hex { offset } => VerifyError::Hex { offset },
        _ => VerifyError::Shape,
    })?;
    let adler = parse_hex(bytes, 40, 48).map_err(|error| match error {
        SnapshotError::Hex { offset } => VerifyError::Hex { offset },
        _ => VerifyError::Shape,
    })?;

    let actual_len = selected_length(spans) as u64;
    if len != actual_len {
        return Err(VerifyError::Length);
    }
    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
