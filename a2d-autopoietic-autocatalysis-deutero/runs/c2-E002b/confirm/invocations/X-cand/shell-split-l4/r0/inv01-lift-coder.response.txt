#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }

    out
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;
    let mut close = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, c)| if !is_name_continue(c) { Some(i) } else { None })
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    let mut name_chars = name.chars();
    if !name_chars.next().is_some_and(is_name_start)
        || !name_chars.all(is_name_continue)
    {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let value = if name_end == contents.len() {
        env_value(env, name).unwrap_or("").to_string()
    } else if contents[name_end..].starts_with(":-") {
        let default = &contents[name_end + 2..];
        match env_value(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut cursor = 0usize;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let width = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                    cursor += width;
                } else {
                    current.text.push(c);
                    cursor += width;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    cursor += width;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    cursor += width;
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            current.text.push(next);
                        }
                        cursor += next.len_utf8();
                    } else {
                        current.text.push('\\');
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    let after = cursor + width;
                    if input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[after..].starts_with('{') {
                        let (value, next) = parse_braced(input, cursor, env)?;
                        append_expansion(&mut current, &value, false);
                        active = true;
                        cursor = next;
                    } else if after < input.len()
                        && input[after..].chars().next().is_some_and(is_name_start)
                    {
                        let mut end = after;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[after..end];
                        let value = env_value(env, name).unwrap_or("");
                        append_expansion(&mut current, value, false);
                        active = true;
                        cursor = end;
                    } else {
                        current.text.push('$');
                        cursor += width;
                    }
                } else {
                    current.text.push(c);
                    cursor += width;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    cursor += width;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = cursor;
                    active = true;
                    cursor += width;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = cursor;
                    active = true;
                    cursor += width;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    cursor += width;
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    cursor += next.len_utf8();
                    if next != '\n' {
                        append_literal(&mut current, next, false);
                        active = true;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    let after = cursor + width;
                    if input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[after..].starts_with('{') {
                        let (value, next) = parse_braced(input, cursor, env)?;
                        append_expansion(&mut current, &value, true);
                        active = true;
                        cursor = next;
                    } else if after < input.len()
                        && input[after..].chars().next().is_some_and(is_name_start)
                    {
                        let mut end = after;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[after..end];
                        let value = env_value(env, name).unwrap_or("");
                        append_expansion(&mut current, value, true);
                        active = true;
                        cursor = end;
                    } else {
                        current.text.push('$');
                        active = true;
                        cursor += width;
                    }
                } else {
                    append_literal(&mut current, c, true);
                    active = true;
                    cursor += width;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }

    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate, 0u8));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate, 1u8));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate, 2u8));
        }

        let candidate = if candidates.is_empty() {
            fallback_candidate(word)
        } else {
            candidates
                .into_iter()
                .min_by_key(|(candidate, priority)| (candidate.len(), *priority))
                .unwrap()
                .0
        };
        encoded.push(candidate);
    }

    Ok(encoded.join(" "))
}
