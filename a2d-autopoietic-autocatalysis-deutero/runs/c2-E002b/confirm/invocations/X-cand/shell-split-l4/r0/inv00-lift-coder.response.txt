use std::fmt::Write;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn append(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
        .unwrap_or("")
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn expand_braced(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(String, usize), SplitError> {
    let after = dollar + 2;
    let mut i = after;
    let mut close = None;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next().unwrap();
                i += next.len_utf8();
            }
        } else if c == '}' {
            close = Some(i);
            break;
        } else {
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[after..close];

    let name_end = content
        .char_indices()
        .find_map(|(idx, c)| (!is_name_continue(c)).then_some(idx))
        .unwrap_or(content.len());

    if name_end == 0 || !is_name_start(content.chars().next().unwrap()) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &content[..name_end];
    let rest = &content[name_end..];
    let value = if rest.is_empty() {
        lookup(env, name).to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        let found = lookup(env, name);
        if found.is_empty() {
            decode_default(default)
        } else {
            found.to_string()
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut mode = Mode::Bare;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match mode {
            Mode::Single(_) => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    append(&mut word, &input[i..i + n], false);
                }
                i += n;
            }
            Mode::Double(open) => {
                if c == '"' {
                    mode = Mode::Bare;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append(&mut word, &input[i..i + next.len_utf8()], false);
                        }
                        i += next.len_utf8();
                    } else {
                        append(&mut word, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_i = i + n;
                    if next_i < input.len() {
                        let next = input[next_i..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        } else if next == '{' {
                            let (value, end) = expand_braced(input, i, env)?;
                            append(&mut word, &value, false);
                            active = true;
                            i = end;
                        } else if is_name_start(next) {
                            let mut end = next_i + next.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            append(&mut word, lookup(env, &input[next_i..end]), false);
                            active = true;
                            i = end;
                        } else {
                            append(&mut word, "$", false);
                            i += n;
                        }
                    } else {
                        append(&mut word, "$", false);
                        i += n;
                    }
                } else {
                    append(&mut word, &input[i..i + n], false);
                    i += n;
                }
                let _ = open;
            }
            Mode::Bare => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    active = true;
                    mode = Mode::Single(i);
                    i += n;
                } else if c == '"' {
                    active = true;
                    mode = Mode::Double(i);
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        append(&mut word, &input[i..i + next.len_utf8()], false);
                        active = true;
                    }
                    i += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_i = i + n;
                    if next_i < input.len() {
                        let next = input[next_i..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        } else if next == '{' {
                            let (value, end) = expand_braced(input, i, env)?;
                            append(&mut word, &value, true);
                            active = true;
                            i = end;
                        } else if is_name_start(next) {
                            let mut end = next_i + next.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            append(&mut word, lookup(env, &input[next_i..end]), true);
                            active = true;
                            i = end;
                        } else {
                            append(&mut word, "$", true);
                            active = true;
                            i += n;
                        }
                    } else {
                        append(&mut word, "$", true);
                        active = true;
                        i += n;
                    }
                } else {
                    append(&mut word, &input[i..i + n], true);
                    active = true;
                    i += n;
                }
            }
        }
    }

    match mode {
        Mode::Single(open) => return Err(SplitError::UnterminatedSingle { pos: open }),
        Mode::Double(open) => return Err(SplitError::UnterminatedDouble { pos: open }),
        Mode::Bare => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        let best = candidates
            .into_iter()
            .flatten()
            .min_by_key(|candidate| candidate.len());

        let result = if let Some(best) = best {
            best
        } else {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push_str("'\\''");
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            fallback
        };
        encoded.push(result);
    }

    let mut out = String::new();
    for (i, item) in encoded.iter().enumerate() {
        if i != 0 {
            out.push(' ');
        }
        out.push_str(item);
    }
    Ok(out)
}
