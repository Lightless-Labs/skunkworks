#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_wildcard(c: char) -> bool {
    matches!(c, '*' | '?' | '[')
}

fn append_literal(word: &mut Word, c: char, active_wildcard: bool) {
    if active_wildcard && is_wildcard(c) {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, active_wildcard: bool) {
    for c in value.chars() {
        append_literal(word, c, active_wildcard);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut chars = text.chars().peekable();

    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                decoded.push(next);
            } else {
                decoded.push('\\');
            }
        } else {
            decoded.push(c);
        }
    }

    decoded
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    let mut chars = input[after_dollar..].char_indices();

    if let Some((_, first)) = chars.clone().next() {
        if is_name_start(first) {
            let mut end = after_dollar;
            for (offset, c) in chars {
                if !is_name_continue(c) {
                    break;
                }
                end = after_dollar + offset + c.len_utf8();
            }
            let name = &input[after_dollar..end];
            return Ok((end, lookup(env, name).unwrap_or("").to_string()));
        }
    }

    if input[after_dollar..].starts_with('{') {
        let content_start = after_dollar + 1;
        let mut scan = content_start;
        let mut close = None;

        while scan < input.len() {
            let c = input[scan..].chars().next().unwrap();
            if c == '\\' {
                scan += c.len_utf8();
                if scan < input.len() {
                    let next = input[scan..].chars().next().unwrap();
                    scan += next.len_utf8();
                }
            } else if c == '}' {
                close = Some(scan);
                break;
            } else {
                scan += c.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .find_map(|(offset, c)| (!is_name_continue(c)).then_some(offset))
            .unwrap_or(contents.len());

        let name = &contents[..name_end];
        if name.is_empty() || !name.chars().next().is_some_and(is_name_start) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let suffix = &contents[name_end..];
        let value = if suffix.is_empty() {
            lookup(env, name).unwrap_or("").to_string()
        } else if let Some(default) = suffix.strip_prefix(":-") {
            match lookup(env, name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((close + 1, value));
    }

    Ok((after_dollar, "$".to_string()))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_literal(&mut word, next, false);
                        }
                        i = after_next;
                    } else {
                        append_literal(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let (end, value) = parse_expansion(input, i, env)?;
                    if end == next_i && value == "$" {
                        append_literal(&mut word, '$', false);
                    } else {
                        active = true;
                        append_expansion(&mut word, &value, false);
                    }
                    i = end;
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_start = i;
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_start = i;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        append_literal(&mut word, next, false);
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let (end, value) = parse_expansion(input, i, env)?;
                    if end == next_i && value == "$" {
                        active = true;
                        append_literal(&mut word, '$', true);
                    } else {
                        active = true;
                        append_expansion(&mut word, &value, true);
                    }
                    i = end;
                } else {
                    active = true;
                    append_literal(&mut word, c, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        return None;
    }
    let mut out = String::with_capacity(word.len() + 2);
    out.push('\'');
    out.push_str(word);
    out.push('\'');
    Some(out)
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::with_capacity(word.len() + 2);
    out.push('"');
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::new();
    out.push('\'');
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        let mut best: Option<(usize, String)> = None;
        for (priority, candidate) in candidates.into_iter().enumerate() {
            if let Some(candidate) = candidate {
                let len = candidate.len();
                if best.as_ref().is_none_or(|(best_len, _)| len < *best_len) {
                    best = Some((len, candidate));
                }
                // Equal lengths preserve the earlier candidate priority.
                let _ = priority;
            }
        }
        let chosen = match best {
            Some((_, value)) => value,
            None => fallback_candidate(word),
        };
        encoded.push(chosen);
    }

    Ok(encoded.join(" "))
}
