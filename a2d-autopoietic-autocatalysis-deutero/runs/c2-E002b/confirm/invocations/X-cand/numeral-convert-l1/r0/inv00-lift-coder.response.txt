#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }

    if value == 0 {
        return Ok(match case {
            RomanCase::Upper => "N".to_owned(),
            RomanCase::Lower => "n".to_owned(),
        });
    }

    const TOKENS: &[(u32, &str)] = &[
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    let mut remainder = value;
    let mut output = String::new();

    for &(amount, token) in TOKENS {
        while remainder >= amount {
            output.push_str(token);
            remainder -= amount;
        }
    }

    if matches!(case, RomanCase::Lower) {
        output.make_ascii_lowercase();
    }

    Ok(output)
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    // Validate every byte first so InvalidByte takes precedence over case errors.
    for (index, byte) in input.bytes().enumerate() {
        if !matches!(
            byte,
            b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
                | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n'
        ) {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let mut case = None;
    for (index, byte) in input.bytes().enumerate() {
        let byte_case = if byte.is_ascii_uppercase() {
            Some(RomanCase::Upper)
        } else if byte.is_ascii_lowercase() {
            Some(RomanCase::Lower)
        } else {
            None
        };

        if let Some(byte_case) = byte_case {
            if let Some(ref first_case) = case {
                if *first_case != byte_case {
                    return Err(RomanError::MixedCase { index });
                }
            } else {
                case = Some(byte_case);
            }
        }
    }

    let upper = input.to_ascii_uppercase();

    if upper == "N" {
        return Ok(0);
    }
    if upper.contains('N') {
        return Err(RomanError::NonCanonical { index: 0 });
    }

    fn symbol_value(byte: u8) -> u32 {
        match byte {
            b'I' => 1,
            b'V' => 5,
            b'X' => 10,
            b'L' => 50,
            b'C' => 100,
            b'D' => 500,
            b'M' => 1000,
            _ => 0,
        }
    }

    let bytes = upper.as_bytes();
    let mut provisional = 0u32;
    let mut index = 0;

    while index < bytes.len() {
        let current = symbol_value(bytes[index]);
        if index + 1 < bytes.len() {
            let next = symbol_value(bytes[index + 1]);
            if next > current {
                provisional = provisional.saturating_add(next - current).min(4000);
                index += 2;
                continue;
            }
        }
        provisional = provisional.saturating_add(current).min(4000);
        index += 1;
    }

    if provisional > 3999 {
        return Err(RomanError::Range);
    }

    let input_case = case.unwrap_or(RomanCase::Upper);
    let canonical = roman_encode(provisional, input_case).expect("value is in range");

    let canonical_bytes = canonical.as_bytes();
    let input_bytes = input.as_bytes();
    let common_len = input_bytes.len().min(canonical_bytes.len());

    for index in 0..common_len {
        if input_bytes[index] != canonical_bytes[index] {
            return Err(RomanError::NonCanonical { index });
        }
    }

    if input_bytes.len() != canonical_bytes.len() {
        return Err(RomanError::NonCanonical { index: common_len });
    }

    Ok(provisional)
}
