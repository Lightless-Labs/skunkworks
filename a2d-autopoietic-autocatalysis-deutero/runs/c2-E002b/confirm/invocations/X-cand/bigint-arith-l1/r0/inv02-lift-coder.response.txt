```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Canonical, little-endian base-10 digits. Zero is represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let magnitude_start = if bytes[0] == b'+' { 1 } else { 0 };
        if magnitude_start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan all bytes before considering separator or numeric validation.
        if let Some(index) = bytes[magnitude_start..]
            .iter()
            .enumerate()
            .find_map(|(offset, &byte)| {
                if byte != b'_' && !byte.is_ascii_digit() {
                    Some(magnitude_start + offset)
                } else {
                    None
                }
            })
        {
            return Err(Error::InvalidDigit { index });
        }

        // Underscores must have an ASCII digit on each side.
        if let Some(index) = (magnitude_start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == magnitude_start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let first_digit_index = magnitude_start;
        let first_digit = (magnitude_start..bytes.len())
            .find(|&i| bytes[i] != b'_')
            .expect("nonempty magnitude with valid bytes");
        let digit_count = bytes[magnitude_start..]
            .iter()
            .filter(|&&byte| byte != b'_')
            .count();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_digit_number = max_digits;
            let index = bytes[magnitude_start..]
                .iter()
                .enumerate()
                .filter(|(_, byte)| **byte != b'_')
                .nth(excess_digit_number)
                .map(|(offset, _)| magnitude_start + offset)
                .expect("digit count exceeds limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[magnitude_start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(min_digits).max(1);
        let padding = width - self.digits.len();
        let mut out = String::with_capacity(width + width.saturating_sub(1) / group.max(1));

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }

            if position < padding {
                out.push('0');
            } else {
                let index_from_left = position - padding;
                let digit = self.digits[self.digits.len() - 1 - index_from_left];
                out.push((b'0' + digit) as char);
            }
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => self.digits.iter().rev().cmp(rhs.digits.iter().rev()),
            ordering => ordering,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let length = self.digits.len().max(rhs.digits.len());
        let mut digits = Vec::with_capacity(length + 1);
        let mut carry = 0u8;

        for i in 0..length {
            let sum = self.digits.get(i).copied().unwrap_or(0)
                + rhs.digits.get(i).copied().unwrap_or(0)
                + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            digits.push(carry);
        }

        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i8;

        for i in 0..self.digits.len() {
            let mut difference = self.digits[i] as i8
                - rhs.digits.get(i).copied().unwrap_or(0) as i8
                - borrow;

            if difference < 0 {
                difference += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }

            digits.push(difference as u8);
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }
}
```
