pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Base-10 digits, most significant first; zero is represented as [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over every separator error.
        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        // A separator must have an ASCII digit on both sides.
        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let index = bytes[start..]
                .iter()
                .enumerate()
                .filter(|(_, b)| **b != b'_')
                .nth(excess_number - 1)
                .map(|(i, _)| start + i)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let digits: Vec<u8> = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| b - b'0')
            .collect();

        Ok(Self::from_digits(digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(min_digits).max(1);
        let padding = width - self.digits.len();
        let mut output = String::with_capacity(width + if group > 0 { width / group } else { 0 });

        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                output.push('_');
            }
            if i < padding {
                output.push('0');
            } else {
                output.push((b'0' + self.digits[i - padding]) as char);
            }
        }
        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => self.digits.cmp(&rhs.digits),
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = Vec::with_capacity(self.digits.len().max(rhs.digits.len()) + 1);
        let mut carry = 0u8;
        let mut left = self.digits.iter().rev();
        let mut right = rhs.digits.iter().rev();

        loop {
            match (left.next(), right.next()) {
                (None, None) => break,
                (a, b) => {
                    let sum = a.copied().unwrap_or(0) + b.copied().unwrap_or(0) + carry;
                    result.push(sum % 10);
                    carry = sum / 10;
                }
            }
        }

        if carry != 0 {
            result.push(carry);
        }
        result.reverse();
        Self::from_digits(result)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;
        let mut right = rhs.digits.iter().rev();

        for &digit in self.digits.iter().rev() {
            let mut value = digit as i16 - borrow - right.next().copied().unwrap_or(0) as i16;
            if value < 0 {
                value += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(value as u8);
        }

        result.reverse();
        Ok(Self::from_digits(result))
    }

    fn from_digits(mut digits: Vec<u8>) -> Self {
        let first_nonzero = digits.iter().position(|&digit| digit != 0);
        match first_nonzero {
            Some(index) => {
                if index > 0 {
                    digits.drain(..index);
                }
            }
            None => digits = vec![0],
        }
        Self { digits }
    }
}

}

pub struct Big {

}

impl Big {

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Find invalid bytes first, ignoring separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        // Then validate separators against their immediate magnitude neighbors.
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&byte| byte != b'_').count();
        let first_digit_index = (start..bytes.len())
            .find(|&index| bytes[index] != b'_')
            .expect("nonempty magnitude contains a digit");

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_digit_number = max_digits;
            let index = bytes[start..]
                .iter()
                .enumerate()
                .filter(|(_, byte)| **byte != b'_')
                .nth(excess_digit_number)
                .map(|(offset, _)| start + offset)
                .expect("digit count exceeds limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in &bytes[start..] {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }

        Ok(Self::from_digits(digits))
    }

        let width = min_digits.max(magnitude_len);
        let padding = width - self.digits.len();

        let mut output = String::with_capacity(width.saturating_add(width / 2));
        for position in 0..width {

            if position < padding {
                output.push('0');
            } else {
                output.push(char::from(b'0' + self.digits[position - padding]));
            }
        }
        output

            .cmp(&rhs.digits.len())

            let b = right.next().copied();
            if a.is_none() && b.is_none() && carry == 0 {
                break;
            }

            let sum = a.unwrap_or(0) + b.unwrap_or(0) + carry;
            result.push(sum % 10);
            carry = sum / 10;

                borrow = 0;

            .position(|&digit| digit != 0)
            .unwrap_or(digits.len().saturating_sub(1));
        if first_nonzero > 0 {
            digits.drain(..first_nonzero);
        }
        if digits.is_empty() {
            digits.push(0);
        }
        Self { digits }
    }

pub struct Big {
    // Little-endian base-10 digits, with no high-order zeroes except for zero itself.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);

            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())

            return Err(Error::InvalidDigit { index });
        }

        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'

                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())

            return Err(Error::Separator { index });
        }

        let first_digit = (start..bytes.len())
            .find(|&i| bytes[i].is_ascii_digit())
            .expect("nonempty magnitude with valid bytes and separators");

        let digit_count = bytes[start..]
            .iter()
            .filter(|&&byte| byte.is_ascii_digit())
            .count();

                .enumerate()
                .filter(|(_, byte)| byte.is_ascii_digit())
                .nth(excess_digit_number - 1)
                .map(|(offset, _)| start + offset)
                .expect("excess digit exists");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let digits = bytes[start..]
            .iter()
            .filter(|&&byte| byte.is_ascii_digit())

                borrow = 0;

            digits.push(difference as u8);
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }

        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let length = self.digits.len().max(rhs.digits.len());
        let mut digits = Vec::with_capacity(length + 1);
        let mut carry = 0u8;

        for i in 0..length {
            let sum = self.digits.get(i).copied().unwrap_or(0)
                + rhs.digits.get(i).copied().unwrap_or(0)
                + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            digits.push(carry);
        }

        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i8;

        for i in 0..self.digits.len() {
            let mut difference = self.digits[i] as i8
                - rhs.digits.get(i).copied().unwrap_or(0) as i8
                - borrow;

            if difference < 0 {
                difference += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }

            digits.push(difference as u8);
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }
}
