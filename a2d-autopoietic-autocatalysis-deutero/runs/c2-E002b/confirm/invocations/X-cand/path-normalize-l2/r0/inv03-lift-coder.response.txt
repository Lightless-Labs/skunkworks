#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .collect()
}

fn reduce_relative<'a>(
    stack: &mut Vec<&'a str>,
    parts: impl IntoIterator<Item = &'a str>,
    absolute: bool,
    offset: usize,
    original: &str,
) -> Result<(), PathError> {
    for part in parts {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| *last != "..") {
                    stack.pop();
                } else if absolute {
                    let at = original[offset..]
                        .find(part)
                        .map(|relative| offset + relative)
                        .unwrap_or(offset);
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..");
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

fn render(parts: &[&str], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };

    if absolute && parts.is_empty() {
        return result;
    }
    if !absolute && parts.is_empty() {
        return result;
    }
    if intent {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    for part in components(path) {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| *last != "..") {
                    stack.pop();
                } else if absolute {
                    let at = path.find("..").unwrap_or(0);
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..");
                }
            }
            _ => stack.push(part),
        }
    }

    Ok(render(&stack, absolute, intent))
}

fn validated_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return Err(());
    }

    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let mut cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(home) => Some(validated_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);
    let bytes = path.as_bytes();

    let (absolute, remainder, offset) = if bytes.first() == Some(&b'/') {
        (true, path, 0)
    } else if bytes.first() == Some(&b'~') {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let parts = home_parts.ok_or(PathError::MissingHome)?;
            cwd_parts = parts;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            (true, &path[consumed..], consumed)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (false, path, 0)
    };

    if absolute {
        if bytes.first() == Some(&b'/') {
            cwd_parts.clear();
        }
    }

    let mut stack: Vec<&str> = cwd_parts.iter().map(String::as_str).collect();
    let parts = components(remainder);

    for part in parts {
        match part {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    let at = path[offset..]
                        .find("..")
                        .map(|relative| offset + relative)
                        .unwrap_or(offset);
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(part),
        }
    }

    Ok(render(&stack, true, intent))
}
