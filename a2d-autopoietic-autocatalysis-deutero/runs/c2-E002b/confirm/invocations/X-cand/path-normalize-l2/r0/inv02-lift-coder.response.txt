#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str, start: usize) -> impl Iterator<Item = (usize, &str)> {
    path[start..]
        .split('/')
        .scan(start, |offset, part| {
            let at = *offset;
            *offset += part.len() + 1;
            Some((at, part))
        })
        .filter(|(_, part)| !part.is_empty())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/')
        .find(|part| !part.is_empty())
        .is_some_and(|part| part == "." || part == "..")
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };

    if absolute && parts.is_empty() {
        return "/".to_string();
    }
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce(
    parts: &mut Vec<String>,
    path: &str,
    start: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, part) in components(path, start) {
        match part {
            "." => {}
            ".." => {
                if parts.last().is_some_and(|last| last != "..") {
                    parts.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    parts.push("..".to_string());
                }
            }
            _ => parts.push(part.to_string()),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut parts = Vec::new();
    reduce(&mut parts, path, 0, absolute)?;
    Ok(render(&parts, absolute, intent))
}

fn validated_absolute(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => Ok(normalized
            .split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()),
        Err(_) => Err(error),
    }
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validated_absolute(cwd, PathError::InvalidCwd)?;

    let home_parts = match home {
        Some(home) => Some(validated_absolute(home, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let (absolute, mut parts, start) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_parts = home_parts.ok_or(PathError::MissingHome)?;
        let start = if bytes.len() > 1 { 2 } else { 1 };
        (true, home_parts, start)
    } else if path.starts_with('/') {
        (true, Vec::new(), 0)
    } else {
        (true, cwd_parts, 0)
    };

    let intent = directory_intent(path);
    reduce(&mut parts, path, start, absolute)?;
    Ok(render(&parts, true, intent))
}
