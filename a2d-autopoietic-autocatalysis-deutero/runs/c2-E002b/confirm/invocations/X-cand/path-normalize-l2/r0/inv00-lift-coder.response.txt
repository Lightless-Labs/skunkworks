#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();

    if let Some(at) = bytes.iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = bytes.iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = bytes.iter().position(|&b| (1..=31).contains(&b) || b == 127) {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/').find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn normalize_valid(path: &str) -> Result<String, PathError> {
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut result: Vec<&str> = Vec::new();

    for component in components(path) {
        match component {
            "." => {}
            ".." => {
                if result.last().is_some_and(|last| *last != "..") {
                    result.pop();
                } else if absolute {
                    let at = path.find(component).unwrap_or(0);
                    return Err(PathError::Escape { at });
                } else {
                    result.push("..");
                }
            }
            ordinary => result.push(ordinary),
        }
    }

    let mut output = if absolute {
        format!("/{}", result.join("/"))
    } else if result.is_empty() {
        ".".to_owned()
    } else {
        result.join("/")
    };

    if intent && output != "/" && output != "." {
        output.push('/');
    }

    Ok(output)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    normalize_valid(path)
}

fn validate_context(path: &str) -> Result<String, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    validate_raw(path).map_err(|_| ())?;
    normalize_valid(path).map_err(|_| ())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_normalized = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_normalized = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let (start, remainder) = if bytes.first() == Some(&b'/') {
        (Vec::new(), path)
    } else if bytes.first() == Some(&b'~') {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let home = home_normalized.ok_or(PathError::MissingHome)?;
            let remainder = if bytes.len() == 1 { "" } else { &path[2..] };
            (components(&home), remainder)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (components(&cwd_normalized), path)
    };

    let intent = directory_intent(path);
    let mut result: Vec<&str> = start;

    for (component, at) in remainder
        .match_indices(|c: char| c == '/')
        .map(|(at, _)| (&remainder[at..at], at))
        .collect::<Vec<_>>()
    {
        let _ = (component, at);
    }

    let base_offset = path.len() - remainder.len();
    let mut offset = 0;
    for component in remainder.split('/') {
        if component.is_empty() {
            offset += 1;
            continue;
        }
        match component {
            "." => {}
            ".." => {
                if result.is_empty() {
                    return Err(PathError::Escape { at: base_offset + offset });
                }
                result.pop();
            }
            ordinary => result.push(ordinary),
        }
        offset += component.len() + 1;
    }

    let mut output = format!("/{}", result.join("/"));
    if intent && output != "/" {
        output.push('/');
    }
    Ok(output)
}
