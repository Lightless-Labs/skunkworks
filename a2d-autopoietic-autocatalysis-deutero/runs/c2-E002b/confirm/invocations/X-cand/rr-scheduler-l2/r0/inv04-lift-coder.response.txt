#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion_times = vec![None; jobs.len()];
    let mut ready = std::collections::VecDeque::<usize>::new();
    let mut event_index = 0usize;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();

    let admit_through = |limit: u64,
                         event_index: &mut usize,
                         ready: &mut std::collections::VecDeque<usize>,
                         completion_times: &mut Vec<Option<u64>>| {
        while *event_index < events.len() {
            let job_index = events[*event_index];
            let job = &jobs[job_index];
            if u64::from(job.arrival) > limit {
                break;
            }
            *event_index += 1;
            if job.ticks == 0 {
                completion_times[job_index] = Some(u64::from(job.arrival));
            } else {
                ready.push_back(job_index);
            }
        }
    };

    while completion_times.iter().any(Option::is_none) {
        admit_through(time, &mut event_index, &mut ready, &mut completion_times);

        if ready.is_empty() {
            if event_index >= events.len() {
                break;
            }
            let next_time = u64::from(jobs[events[event_index]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        }

        let job_index = ready.pop_front().expect("ready queue is nonempty");
        let job = &jobs[job_index];

        if last_dispatched_id.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }

        admit_through(time, &mut event_index, &mut ready, &mut completion_times);

        let remaining = u64::from(job.ticks)
            - runs
                .iter()
                .filter(|run| run.id == job.id)
                .map(|run| run.end - run.start)
                .sum::<u64>();
        let consumed = remaining.min(u64::from(quantum));
        let start = time;
        time += consumed;
        let finished = consumed == remaining;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished,
        });

        admit_through(time, &mut event_index, &mut ready, &mut completion_times);

        if finished {
            completion_times[job_index] = Some(time);
        } else {
            ready.push_back(job_index);
        }
        last_dispatched_id = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion_times[index].unwrap_or(time)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
