use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn is_hex(b: u8) -> bool {
    b.is_ascii_hexdigit()
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'a'..=b'f' => b - b'a' + 10,
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn validate_component(
    component: Component,
    raw: &str,
    base_offset: usize,
) -> Result<(), ParseError> {
    decode_component(component, raw).map(|_| ()).map_err(|e| match e {
        CodecError::BadEscape { at } => ParseError::BadEscape {
            at: base_offset + at,
        },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 {
            at: base_offset + at,
        },
    })
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let rest_start = colon + 3;
    let authority_end = bytes[rest_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| rest_start + i)
        .unwrap_or(input.len());
    let authority = &input[rest_start..authority_end];
    let auth_bytes = authority.as_bytes();

    let ats: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: rest_start + ats[1],
        });
    }
    let (userinfo, userinfo_len, host_start_in_auth) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: rest_start });
        }
        (
            Some(authority[..at].to_owned()),
            at,
            at + 1,
        )
    } else {
        (None, 0, 0)
    };

    let hostport = &authority[host_start_in_auth..];
    let host_abs = rest_start + host_start_in_auth;
    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: host_abs }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_abs });
        }
        let suffix = &hostport[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], host_abs + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_abs + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port, true)
    } else {
        let split = hostport.find(':');
        let host_part = split.map(|i| &hostport[..i]).unwrap_or(hostport);
        if host_part.is_empty() {
            return Err(ParseError::InvalidHost { at: host_abs });
        }
        for (i, b) in host_part.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_abs + i });
            }
        }
        let parsed_port = if let Some(i) = split {
            Some(parse_port(&hostport[i + 1..], host_abs + i + 1)?)
        } else {
            None
        };
        (host_part.to_ascii_lowercase(), parsed_port, false)
    };

    let mut fragment = None;
    let before_fragment_end = if let Some(hash) = input[authority_end..].find('#') {
        let pos = authority_end + hash;
        fragment = Some(input[pos + 1..].to_owned());
        pos
    } else {
        input.len()
    };

    let mut query = None;
    let before_query_end = if let Some(qmark) = input[authority_end..before_fragment_end].find('?') {
        let pos = authority_end + qmark;
        query = Some(input[pos + 1..before_fragment_end].to_owned());
        pos
    } else {
        before_fragment_end
    };

    let path = if before_query_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..before_query_end].to_owned()
    };

    if let Some(ui) = &userinfo {
        validate_component(Component::UserInfo, ui, rest_start)?;
    }
    if before_query_end != authority_end {
        validate_component(
            Component::Path,
            &input[authority_end..before_query_end],
            authority_end,
        )?;
    }
    if let Some(q) = &query {
        let q_offset = input[authority_end..before_query_end]
            .find('?')
            .map(|i| authority_end + i + 1)
            .unwrap_or(before_query_end);
        validate_component(Component::Query, q, q_offset)?;
    }
    if let Some(f) = &fragment {
        let hash = input[authority_end..].find('#').unwrap() + authority_end;
        validate_component(Component::Fragment, f, hash + 1)?;
    }

    let _ = userinfo_len;
    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(port: &str, at: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }
    let bytes = port.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }
    let mut value: u32 = 0;
    for &b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

fn path_error_from_codec(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(path_error_from_codec)?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut terminal_navigation = false;

    for (i, raw) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(path_error_from_codec)?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if i + 1 == segments.len() {
            terminal_navigation = is_dot || is_dotdot;
        }
        if is_dot {
            continue;
        } else if is_dotdot {
            stack.pop();
        } else {
            stack.push((*raw).to_owned());
        }
    }

    if terminal_navigation {
        stack.push(String::new());
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn scan_invalid_byte(input: &str) -> Option<usize> {
    input
        .bytes()
        .enumerate()
        .find_map(|(i, b)| (!(0x21..=0x7e).contains(&b)).then_some(i))
}

fn split_reference_components(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let hash = reference.find('#');
    let frag = hash.map(|h| &reference[h + 1..]);
    let before_frag = hash.map(|h| &reference[..h]).unwrap_or(reference);
    let q = before_frag.find('?');
    let query = q.map(|i| &before_frag[i + 1..]);
    let path = q.map(|i| &before_frag[..i]).unwrap_or(before_frag);
    (path, query, frag)
}

fn path_from_resolved(url: &mut Url) -> Result<(), ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(())
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Some(at) = scan_invalid_byte(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    if !reference.is_empty() {
        let boundary = reference
            .bytes()
            .position(|b| matches!(b, b'/' | b'?' | b'#'))
            .unwrap_or(reference.len());
        if reference[..boundary].contains(':') {
            let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
            path_from_resolved(&mut url)?;
            return Ok(url);
        }
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        path_from_resolved(&mut url)?;
        return Ok(url);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference_components(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.clone(),
            if ref_query.is_some() {
                ref_query.map(str::to_owned)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..slash], ref_path)
        };
        (merged, ref_query.map(str::to_owned))
    };

    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        constructed.push_str(userinfo);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&selected_path);
    if let Some(q) = &selected_query {
        constructed.push('?');
        constructed.push_str(q);
    }
    if let Some(f) = ref_fragment {
        constructed.push('#');
        constructed.push_str(f);
    }

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    path_from_resolved(&mut url)?;
    Ok(url)
}
