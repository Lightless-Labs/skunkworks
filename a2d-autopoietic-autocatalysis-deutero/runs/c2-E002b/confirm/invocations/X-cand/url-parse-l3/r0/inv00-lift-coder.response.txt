#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if component_safe(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn codec_to_parse(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn validate_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    if let Some((i, _)) = port_text.bytes().enumerate().find(|(_, b)| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort {
            at: port_start + i,
        });
    }
    if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for b in port_text.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
        {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_abs = authority_start + hostport_start;
    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => {
                return Err(ParseError::InvalidIpv6 {
                    at: hostport_abs,
                })
            }
        };
        let interior = &hostport[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 {
                at: hostport_abs,
            });
        }
        let suffix = &hostport[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(validate_port(&suffix[1..], hostport_abs + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_abs + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port, true)
    } else {
        let split = hostport.find(':');
        let (host_text, port_text) = match split {
            Some(i) => (&hostport[..i], Some((&hostport[i + 1..], i + 1))),
            None => (hostport, None),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_abs });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_abs + i,
                });
            }
        }
        let parsed_port = match port_text {
            Some((p, relative_start)) => {
                Some(validate_port(p, hostport_abs + relative_start)?)
            }
            None => None,
        };
        (host_text.to_ascii_lowercase(), parsed_port, false)
    };

    let remainder_start = authority_end;
    let fragment_pos = bytes[remainder_start..]
        .iter()
        .position(|&b| b == b'#')
        .map(|i| remainder_start + i);
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());
    let query_pos = bytes[remainder_start..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| remainder_start + i);

    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path = if path_end == remainder_start {
        "/".to_owned()
    } else {
        input[remainder_start..path_end].to_owned()
    };
    let query = query_pos.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|f| input[f + 1..].to_owned());

    if let Some(ref value) = userinfo {
        decode_component(Component::UserInfo, value)
            .map_err(|e| codec_to_parse(e, authority_start))?;
    }
    if path != "/" || path_end != remainder_start {
        decode_component(Component::Path, &path)
            .map_err(|e| codec_to_parse(e, remainder_start))?;
    }
    if let Some(ref value) = query {
        decode_component(Component::Query, value)
            .map_err(|e| codec_to_parse(e, query_pos.unwrap() + 1))?;
    }
    if let Some(ref value) = fragment {
        decode_component(Component::Fragment, value)
            .map_err(|e| codec_to_parse(e, fragment_pos.unwrap() + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut final_navigation = false;

    for (i, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 + raw_segments[..i].iter().map(|s| s.len() + 1).sum::<usize>() },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 + raw_segments[..i].iter().map(|s| s.len() + 1).sum::<usize>() },
        })?;
        let navigation = decoded == "." || decoded == "..";
        final_navigation = i + 1 == raw_segments.len() && navigation;

        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push((*segment).to_owned());
        }
    }

    if final_navigation {
        stack.push(String::new());
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn scan_reference_bytes(reference: &str) -> Result<(), ParseError> {
    for (i, b) in reference.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let bytes = reference.as_bytes();
    let fragment_pos = bytes.iter().position(|&b| b == b'#');
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());
    let query_pos = bytes[..before_fragment_end]
        .iter()
        .position(|&b| b == b'?');
    let path_end = query_pos.unwrap_or(before_fragment_end);

    let path = &reference[..path_end];
    let query = query_pos.map(|q| &reference[q + 1..before_fragment_end]);
    let fragment = fragment_pos.map(|f| &reference[f + 1..]);
    (path, query, fragment)
}

fn normalize_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalize_url_path(base_url)?;

    scan_reference_bytes(reference).map_err(ResolveError::Reference)?;

    let first_delimiter = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);
    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if reference_query.is_some() {
                reference_query.map(str::to_owned)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], reference_path)
        };
        (merged, reference_query.map(str::to_owned))
    };

    let mut absolute = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        absolute.push_str(userinfo);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&selected_path);
    if let Some(query) = selected_query {
        absolute.push('?');
        absolute.push_str(&query);
    }
    if let Some(fragment) = reference_fragment {
        absolute.push('#');
        absolute.push_str(fragment);
    }

    let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    normalize_url_path(url)
}
