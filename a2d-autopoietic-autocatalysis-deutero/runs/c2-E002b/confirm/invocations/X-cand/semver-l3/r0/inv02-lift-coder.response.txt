use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn missing_core_byte(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    let mut i = start;
    while i < bytes.len() && bytes[i].is_ascii_digit() {
        i += 1;
    }
    if i == start {
        return if missing_core_byte(byte_at(s, start)) {
            Err(VersionError::MissingCore { at: start })
        } else {
            Err(VersionError::InvalidChar { at: start })
        };
    }

    let digits = &s[start..i];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    *pos = i;
    Ok(value)
}

fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut part = start;

    loop {
        let mut i = part;
        while i < end && bytes[i] != b'.' {
            if !is_ident_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == part {
            return Err(VersionError::EmptyIdentifier { at: part });
        }
        let ident = &s[part..i];
        if pre && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.len() > 1
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: part });
        }
        result.push(ident.to_string());

        if i == end {
            break;
        }
        part = i + 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        match byte_at(s, pos) {
            Some(b'.') => pos += 1,
            b if missing_core_byte(b) => {
                return Err(VersionError::MissingCore { at: pos });
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        let n = parse_core(s, &mut pos)?;
        if _ == 0 {
            // The first parsed field after major is minor.
            let minor = n;
            match byte_at(s, pos) {
                Some(b'.') => {
                    pos += 1;
                    let patch = parse_core(s, &mut pos)?;
                    return finish_version(s, pos, major, minor, patch);
                }
                b if missing_core_byte(b) => {
                    return Err(VersionError::MissingCore { at: pos });
                }
                _ => return Err(VersionError::InvalidChar { at: pos }),
            }
        }
    }
    unreachable!()
}

fn finish_version(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let id_start = pos + 1;
        let plus = bytes[id_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|n| id_start + n);
        let end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, id_start, end, true)?;
        pos = end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let id_start = pos + 1;
        build = parse_identifiers(s, id_start, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_ident(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_ident(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric_ident(x);
        let yn = is_numeric_ident(y);
        let ord = match (xn, yn) {
            (true, true) => compare_numeric_ident(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

pub enum RangeError {

    Syntax { at: usize },

    version: Version,

#[derive(Clone, Copy, Debug)]

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, base: usize) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) {
            i += 1;
        }
        out.push((base + start, &s[start..i]));
    }
    out
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
    numeric_count: usize,
}

fn operand_is_full(text: &str) -> bool {
    text.contains('-')
        || text.contains('+')
        || (text.split('.').count() >= 3
            && !text
                .split('.')
                .any(|f| matches!(f, "*" | "x" | "X")))
}

fn parse_partial(text: &str, offset: usize) -> Result<Partial, RangeError> {
    if operand_is_full(text) {
        return parse_version(text)
            .map(|v| Partial {
                fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
                numeric_count: 3,
                full: Some(v),
            })
            .map_err(|error| RangeError::Version { at: offset, error });
    }

    let parts: Vec<&str> = text.split('.').collect();
    if parts.is_empty() || parts.len() > 3 {
        return Err(RangeError::Syntax { at: offset });
    }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    let mut numeric_count = 0;
    for p in parts {
        if p.is_empty() {
            return Err(RangeError::Syntax { at: offset });
        }
        if matches!(p, "*" | "x" | "X") {
            wildcard_seen = true;
            fields.push(None);
        } else {
            if wildcard_seen
                || !p.bytes().all(|b| b.is_ascii_digit())
                || (p.len() > 1 && p.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: offset });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at: offset })?;
            fields.push(Some(n));
            numeric_count += 1;
        }
    }
    Ok(Partial {
        fields,
        full: None,
        numeric_count,
    })
}

fn padded(p: &Partial) -> Version {
    let vals: Vec<u64> = p.fields.iter().filter_map(|x| *x).collect();
    let (major, minor, patch) = match vals.as_slice() {
        [] => (0, 0, 0),
        [a] => (*a, 0, 0),
        [a, b] => (*a, *b, 0),
        [a, b, c, ..] => (*a, *b, *c),
    };
    p.full.clone().unwrap_or(Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    })
}

fn stable_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn prerelease_zero(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    }
}

fn bump(p: &Partial, mode: u8, offset: usize) -> Result<Version, RangeError> {
    let lo = padded(p);
    let k = p.numeric_count;
    let (mut ma, mut mi, mut pa) = (lo.major, lo.minor, lo.patch);

    let field = if mode == 1 {
        if ma > 0 || k == 1 {
            0
        } else if mi > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if mode == 2 {
        if k == 1 { 0 } else { 1 }
    } else if k == 1 {
        0
    } else {
        1
    };

    let result = match field {
        0 => ma.checked_add(1).map(|v| {
            ma = v;
            mi = 0;
            pa = 0;
        }),
        1 => mi.checked_add(1).map(|v| {
            mi = v;
            pa = 0;
        }),
        _ => pa.checked_add(1).map(|v| pa = v),
    };
    if result.is_none() {
        return Err(RangeError::Syntax { at: offset });
    }
    Ok(prerelease_zero(ma, mi, pa))
}

fn add_comparator(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push((v.major, v.minor, v.patch));
    }
    arm.comps.push(Comparator {
        op,
        version: v,
        anchor,
    });
}

fn parse_explicit(token: &str, start: usize) -> Result<Option<(Op, Version)>, RangeError> {
    let bytes = token.as_bytes();
    let (op, n) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else if token.starts_with('=') {
        (Op::Eq, 1)
    } else {
        return Ok(None);
    };
    if n >= bytes.len() || !bytes[n].is_ascii_digit() {
        return Err(RangeError::Syntax { at: start });
    }
    let operand = &token[n..];
    let v = parse_version(operand).map_err(|error| RangeError::Version {
        at: start + n,
        error,
    })?;
    Ok(Some((op, v)))
}

fn parse_hyphen_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let minus = tokens
        .iter()
        .find(|(_, t)| *t == "-")
        .map(|(i, _)| *i)
        .unwrap();
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: minus });
    }
    let (ls, lt) = tokens[0];
    let (us, ut) = tokens[2];
    let lower = parse_partial(lt, ls)?;
    let upper = parse_partial(ut, us)?;
    if lower.numeric_count == 0 {
        return Err(RangeError::Syntax { at: ls });
    }
    if upper.numeric_count == 0 {
        return Err(RangeError::Syntax { at: us });
    }

    let mut arm = Arm {
        comps: Vec::new(),
        anchors: Vec::new(),
    };
    let lv = padded(&lower);
    add_comparator(&mut arm, Op::Ge, lv.clone(), lower.full.is_some());
    if let Some(full) = upper.full {
        add_comparator(&mut arm, Op::Le, full, true);
    } else if upper.numeric_count == 3 {
        add_comparator(&mut arm, Op::Le, padded(&upper), false);
    } else {
        let hi = bump(&upper, 0, us)?;
        add_comparator(&mut arm, Op::Lt, hi, false);
    }
    Ok(arm)
}

fn parse_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    if let Some((_, _)) = tokens.iter().find(|(_, t)| *t == "-") {
        return parse_hyphen_arm(tokens);
    }

    let mut arm = Arm {
        comps: Vec::new(),
        anchors: Vec::new(),
    };

    for &(start, token) in tokens {
        if let Some((op, v)) = parse_explicit(token, start)? {
            add_comparator(&mut arm, op, v, true);
            continue;
        }

        if token.starts_with('^') || token.starts_with('~') {
            let mode = if token.as_bytes()[0] == b'^' { 1 } else { 2 };
            let operand = &token[1..];
            let offset = start + 1;
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit()
                && !matches!(operand.as_bytes()[0], b'*' | b'x' | b'X')
            {
                return Err(RangeError::Syntax { at: start });
            }
            let p = parse_partial(operand, offset)?;
            if p.numeric_count == 0 {
                return Err(RangeError::Syntax { at: start });
            }
            let lo = padded(&p);
            add_comparator(&mut arm, Op::Ge, lo.clone(), p.full.is_some());
            let hi = bump(&p, mode, offset)?;
            add_comparator(&mut arm, Op::Lt, hi, false);
            continue;
        }

        let first = token.as_bytes()[0];
        if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at: start });
        }
        let p = parse_partial(token, start)?;
        if let Some(full) = p.full {
            add_comparator(&mut arm, Op::Eq, full, true);
            continue;
        }

        let k = p.numeric_count;
        if k == 0 {
            continue;
        }
        let lo = padded(&p);
        if k == 3 {
            add_comparator(&mut arm, Op::Eq, lo, false);
        } else {
            add_comparator(&mut arm, Op::Ge, lo, false);
            let hi = bump(&p, 0, start)?;
            add_comparator(&mut arm, Op::Lt, hi, false);
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;
    let mut saw_non_ws = false;

    while i < bytes.len() {
        if bytes[i..].starts_with(b"||") {
            let before = &s[segment_start..i];
            let tokens = split_tokens(before, segment_start);
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(&tokens)?);
            let sep = i;
            i += 2;
            while i < bytes.len() && is_ws(bytes[i]) {
                i += 1;
            }
            segment_start = i;
            if i == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
            if sep > 0 && !is_ws(bytes[sep - 1]) || i < bytes.len() && !is_ws(bytes[i - 1]) {
                // A separator must be a standalone token.
                return Err(RangeError::Syntax { at: sep });
            }
            saw_non_ws = true;
            continue;
        }
        if !is_ws(bytes[i]) {
            saw_non_ws = true;
        }
        i += 1;
    }

    let tail = &s[segment_start..];
    let tokens = split_tokens(tail, segment_start);
    if tokens.is_empty() {
        if arms.is_empty() && !saw_non_ws {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: bytes.len() });
    }
    arms.push(parse_arm(&tokens)?);
    Ok(Range { arms })
}

fn cmp_satisfies(ord: Ordering, op: Op) -> bool {
    match op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn arm_admits(v: &Version, arm: &Arm) -> bool {
    !v.pre.is_empty() && arm.anchors.iter().any(|&t| t == version_tuple(v))
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty() && !include_prerelease && !arm_admits(v, arm) {
        return false;
    }
    arm.comps.iter().all(|c| {
        cmp_satisfies(compare_versions(v, &c.version), c.op)
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn normalize_bounds(arm: &Arm) -> (Option<Bound>, Option<Bound>, Vec<(u64, u64, u64)>) {
    let mut lower: Option<Bound> = None;
    let mut upper: Option<Bound> = None;
    for c in &arm.comps {
        match c.op {
            Op::Eq => {
                lower = choose_lower(
                    lower,
                    Bound {
                        version: c.version.clone(),
                        inclusive: true,
                    },
                );
                upper = choose_upper(
                    upper,
                    Bound {
                        version: c.version.clone(),
                        inclusive: true,
                    },
                );
            }
            Op::Gt | Op::Ge => {
                lower = choose_lower(
                    lower,
                    Bound {
                        version: c.version.clone(),
                        inclusive: matches!(c.op, Op::Ge),
                    },
                );
            }
            Op::Lt | Op::Le => {
                upper = choose_upper(
                    upper,
                    Bound {
                        version: c.version.clone(),
                        inclusive: matches!(c.op, Op::Le),
                    },
                );
            }
        }
    }
    (lower, upper, arm.anchors.clone())
}

fn choose_lower(old: Option<Bound>, new: Bound) -> Option<Bound> {
    match old {
        None => Some(new),
        Some(o) => match compare_versions(&new.version, &o.version) {
            Ordering::Greater => Some(new),
            Ordering::Less => Some(o),
            Ordering::Equal => Some(Bound {
                version: o.version,
                inclusive: o.inclusive && new.inclusive,
            }),
        },
    }
}

fn choose_upper(old: Option<Bound>, new: Bound) -> Option<Bound> {
    match old {
        None => Some(new),
        Some(o) => match compare_versions(&new.version, &o.version) {
            Ordering::Less => Some(new),
            Ordering::Greater => Some(o),
            Ordering::Equal => Some(Bound {
                version: o.version,
                inclusive: o.inclusive && new.inclusive,
            }),
        },
    }
}

fn version_no_build(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn core_string(t: (u64, u64, u64)) -> String {
    format!("{}.{}.{}", t.0, t.1, t.2)
}

fn arm_canonical(arm: &Arm) -> Option<String> {
    let (lower, upper, mut anchors) = normalize_bounds(arm);
    anchors.sort_unstable();
    anchors.dedup();

    // Check for an empty ordinary interval and account for anchored prereleases
    // at a single equality point.
    if let (Some(l), Some(u)) = (&lower, &upper) {
        let ord = compare_versions(&l.version, &u.version);
        if ord == Ordering::Greater || (ord == Ordering::Equal && !(l.inclusive && u.inclusive)) {
            return None;
        }
        if ord == Ordering::Equal {
            let v = &l.version;
            if !v.pre.is_empty() {
                if !anchors.contains(&version_tuple(v)) {
                    return None;
                }
            } else if !arm.comps.iter().all(|c| {
                cmp_satisfies(compare_versions(v, &c.version), c.op)
            }) {
                return None;
            }
        }
    }

    let lower_s = lower.map_or_else(|| "*".to_string(), |b| {
        format!("{}{}", if b.inclusive { ">=" } else { ">" }, version_no_build(&b.version))
    });
    let upper_s = upper.map_or_else(|| "*".to_string(), |b| {
        format!("{}{}", if b.inclusive { "<=" } else { "<" }, version_no_build(&b.version))
    });
    let anchor_s = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors
            .into_iter()
            .map(core_string)
            .collect::<Vec<_>>()
            .join(",")
    };
    Some(format!("[{};{};pre={}]", lower_s, upper_s, anchor_s))
}

pub fn canonical(r: &Range) -> String {
    let mut arms: Vec<String> = r.arms.iter().filter_map(arm_canonical).collect();
    arms.sort();
    arms.dedup();
    if arms.is_empty() {
        "EMPTY".to_string()
    } else {
        arms.join(" || ")
    }
}

fn next_core(v: &Version) -> Option<Version> {
    if v.patch < u64::MAX {
        Some(stable_version(v.major, v.minor, v.patch + 1))
    } else if v.minor < u64::MAX {
        Some(stable_version(v.major, v.minor + 1, 0))
    } else if v.major < u64::MAX {
        Some(stable_version(v.major + 1, 0, 0))
    } else {
        None
    }
}

fn least_prerelease(v: &Version) -> Version {
    prerelease_zero(v.major, v.minor, v.patch)
}

fn arm_has_stable_witness(arm: &Arm) -> bool {
    let (lower, upper, _) = normalize_bounds(arm);
    let mut candidate = stable_version(0, 0, 0);

    if let Some(l) = lower {
        if !l.version.pre.is_empty() {
            if let Some(n) = next_core(&l.version) {
                candidate = n;
            } else {
                return false;
            }
        } else {
            candidate = l.version.clone();
            if !l.inclusive {
                match next_core(&candidate) {
                    Some(n) => candidate = n,
                    None => return false,
                }
            }
        }
    }

    if let Some(u) = upper {
        let ord = compare_versions(&candidate, &u.version);
        if ord == Ordering::Greater || (ord == Ordering::Equal && !u.inclusive) {
            return false;
        }
    }
    arm.comps.iter().all(|c| cmp_satisfies(compare_versions(&candidate, &c.version), c.op))
}

fn arm_has_prerelease_witness(arm: &Arm, include_prerelease: bool) -> bool {
    let (lower, upper, anchors) = normalize_bounds(arm);
    let mut tuples = anchors;
    if let Some(l) = lower {
        tuples.push(version_tuple(&l.version));
        if !l.version.pre.is_empty() {
            tuples.push(version_tuple(&l.version));
        }
    }
    if let Some(u) = upper {
        tuples.push(version_tuple(&u.version));
    }
    tuples.sort_unstable();
    tuples.dedup();

    for tuple in tuples {
        let candidate = least_prerelease(&stable_version(tuple.0, tuple.1, tuple.2));
        if !include_prerelease && !arm_admits(&candidate, arm) {
            continue;
        }
        if arm.comps.iter().all(|c| cmp_satisfies(compare_versions(&candidate, &c.version), c.op)) {
            return true;
        }
    }

    // A prerelease bound can itself be a witness even when it is not -0.
    for c in &arm.comps {
        if !c.version.pre.is_empty()
            && (include_prerelease || arm_admits(&c.version, arm))
            && arm.comps.iter().all(|d| {
                cmp_satisfies(compare_versions(&c.version, &d.version), d.op)
            })
        {
            return true;
        }
    }
    false
}

fn arms_intersect(a: &Arm, b: &Arm, include_prerelease: bool) -> bool {
    let mut combined = Arm {
        comps: a.comps.clone(),
        anchors: a.anchors.clone(),
    };
    combined.comps.extend(b.comps.clone());

    if arm_has_stable_witness(&combined) {
        return true;
    }

    let mut tuples = a.anchors.clone();
    tuples.extend(b.anchors.iter().copied());
    if let Some((l, _, _)) = normalize_bounds(&combined).0.map(|x| {
        (x.version, x.inclusive, ())
    }) {
        tuples.push(version_tuple(&l));
    }
    if let Some((u, _, _)) = normalize_bounds(&combined).1.map(|x| {
        (x.version, x.inclusive, ())
    }) {
        tuples.push(version_tuple(&u));
    }
    tuples.sort_unstable();
    tuples.dedup();

    for t in tuples {
        let pre = least_prerelease(&stable_version(t.0, t.1, t.2));
        if !include_prerelease {
            if !arm_admits(&pre, a) || !arm_admits(&pre, b) {
                continue;
            }
        }
        if arm_matches(&pre, &combined, include_prerelease) {
            return true;
        }
    }

    for arm in [a, b] {
        for c in &arm.comps {
            let v = &c.version;
            if v.pre.is_empty() {
                continue;
            }
            if !include_prerelease && (!arm_admits(v, a) || !arm_admits(v, b)) {
                continue;
            }
            if combined.comps.iter().all(|d| cmp_satisfies(compare_versions(v, &d.version), d.op)) {
                return true;
            }
        }
    }

    false
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    a.arms
        .iter()
        .any(|x| b.arms.iter().any(|y| arms_intersect(x, y, include_prerelease)))
}
