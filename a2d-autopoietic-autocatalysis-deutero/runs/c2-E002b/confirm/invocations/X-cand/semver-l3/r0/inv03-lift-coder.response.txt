use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_ascii_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_ascii_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: *pos });
        }
        return Err(VersionError::InvalidChar { at: *pos });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut i = start;

    loop {
        let ident_start = i;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let identifier = &s[ident_start..i];
        if prerelease
            && identifier.bytes().all(is_ascii_digit)
            && identifier.len() > 1
            && identifier.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(identifier.to_string());

        if i == end {
            break;
        }
        i += 1;
        if i == end {
            return Err(VersionError::EmptyIdentifier { at: i });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        if _ == 0 {
            let minor = parse_core(s, &mut pos)?;
            if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if s.as_bytes()[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core(s, &mut pos)?;

            let (pre, build) = parse_version_tail(s, pos)?;
            return Ok(Version {
                major,
                minor,
                patch,
                pre,
                build,
            });
        }
    }

    unreachable!()
}

fn parse_version_tail(s: &str, mut pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    if pos == s.len() {
        return Ok((Vec::new(), Vec::new()));
    }
    let bytes = s.as_bytes();
    if bytes[pos] == b'-' {
        pos += 1;
        let plus = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|x| pos + x)
            .unwrap_or(s.len());
        let pre = parse_identifiers(s, pos, plus, true)?;
        if plus == s.len() {
            return Ok((pre, Vec::new()));
        }
        let build_start = plus + 1;
        let build = parse_identifiers(s, build_start, s.len(), false)?;
        return Ok((pre, build));
    }
    if bytes[pos] == b'+' {
        let build = parse_identifiers(s, pos + 1, s.len(), false)?;
        return Ok((Vec::new(), build));
    }
    Err(VersionError::InvalidChar { at: pos })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_ascii_digit)
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let an = numeric_identifier(a);
    let bn = numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let ordering = compare_identifiers(x, y);
                if ordering != Ordering::Equal {
                    return ordering;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn cmp_satisfies(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn parse_full_operand(token: &str, at: usize) -> Result<Version, RangeError> {
    if token.is_empty() || !token.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    parse_version(token).map_err(|error| RangeError::Version { at, error })
}

fn split_tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        out.push((s[start..i].to_string(), base + start));
    }
    out
}

fn partial_or_full(
    token: &str,
    at: usize,
) -> Result<(Version, usize, bool), RangeError> {
    if token.contains('-') || token.contains('+') {
        return Ok((parse_full_operand(token, at)?, 3, true));
    }
    let fields: Vec<&str> = token.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X")) {
        return Ok((parse_full_operand(token, at)?, 3, true));
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for field in &fields {
        if matches!(*field, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }
        if field.is_empty()
            || wildcard_seen
            || !field.bytes().all(is_ascii_digit)
            || (field.len() > 1 && field.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }
        let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
        nums.push(n);
    }
    let k = nums.len();
    let mut values = [0u64; 3];
    for (i, n) in nums.into_iter().enumerate() {
        values[i] = n;
    }
    Ok((
        Version {
            major: values[0],
            minor: values[1],
            patch: values[2],
            pre: Vec::new(),
            build: Vec::new(),
        },
        k,
        false,
    ))
}

fn endpoint(v: &Version, pre: Vec<String>) -> Version {
    Version {
        major: v.major,
        minor: v.minor,
        patch: v.patch,
        pre,
        build: Vec::new(),
    }
}

fn increment_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut upper = Version {
        major: v.major,
        minor: v.minor,
        patch: v.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    let target = match field {
        0 => &mut upper.major,
        1 => &mut upper.minor,
        _ => &mut upper.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field <= 1 {
        upper.minor = 0;
    }
    if field == 0 {
        upper.patch = 0;
    }
    Ok(upper)
}

fn push_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.comparators.push(Comparator { op, version, anchor });
}

fn parse_bare(token: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (v, k, full) = partial_or_full(token, at)?;
    if full {
        push_cmp(arm, Op::Eq, v, true);
    } else if k == 0 {
        // Wildcard-only operands match all releases.
    } else {
        push_cmp(arm, Op::Ge, v.clone(), false);
        let upper = increment_upper(&v, if k == 1 { 0 } else { 1 }, at)?;
        push_cmp(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_caret_tilde(
    token: &str,
    at: usize,
    caret: bool,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let operand = &token[1..];
    let operand_at = at + 1;
    let (v, k, full) = partial_or_full(operand, operand_at)?;
    if k == 0 {
        return Err(RangeError::Syntax { at });
    }
    push_cmp(arm, Op::Ge, v.clone(), full);
    let field = if caret {
        if v.major > 0 || k == 1 {
            0
        } else if v.minor > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if k == 1 {
        0
    } else {
        1
    };
    let upper = increment_upper(&v, field, operand_at)?;
    push_cmp(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_comparator(token: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let bytes = token.as_bytes();
    let (op, len) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &token[len..];
    let operand_at = at + len;
    if operand.is_empty() || !bytes[len].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let version = parse_full_operand(operand, operand_at)?;
    push_cmp(arm, op, version, true);
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    if let Some((idx, (_, at))) = tokens.iter().enumerate().find(|(_, (t, _))| t == "-") {
        if tokens.len() != 3 || idx != 1 {
            return Err(RangeError::Syntax { at: *at });
        }
        let (lo, lo_k, lo_full) = partial_or_full(&tokens[0].0, tokens[0].1)?;
        let (hi, hi_k, hi_full) = partial_or_full(&tokens[2].0, tokens[2].1)?;
        if lo_k == 0 {
            return Err(RangeError::Syntax { at: tokens[0].1 });
        }
        if hi_k == 0 {
            return Err(RangeError::Syntax { at: tokens[2].1 });
        }
        push_cmp(&mut arm, Op::Ge, lo.clone(), lo_full);
        if hi_full {
            push_cmp(&mut arm, Op::Le, hi, true);
        } else {
            let upper = increment_upper(&hi, if hi_k == 1 { 0 } else { 1 }, tokens[2].1)?;
            push_cmp(&mut arm, Op::Lt, upper, false);
        }
        return Ok(arm);
    }

    for (token, at) in tokens {
        if token.starts_with('^') {
            parse_caret_tilde(token, *at, true, &mut arm)?;
        } else if token.starts_with('~') {
            parse_caret_tilde(token, *at, false, &mut arm)?;
        } else if token.starts_with(['<', '>', '=']) {
            parse_comparator(token, *at, &mut arm)?;
        } else if token.starts_with(|c: char| c.is_ascii_digit())
            || matches!(token.as_str(), "*" | "x" | "X")
        {
            parse_bare(token, *at, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: *at });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let trimmed = s.trim_matches(|c| matches!(c, ' ' | '\t' | '\r' | '\n'));
    if trimmed.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;
    let mut last_sep_end = None;
    while i + 1 < bytes.len() {
        if bytes[i] == b'|' && bytes[i + 1] == b'|' {
            let left = &s[start..i];
            let toks = split_tokens(left, start);
            if toks.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(&toks)?);
            let mut j = i + 2;
            if j < bytes.len() && matches!(bytes[j], b' ' | b'\t' | b'\r' | b'\n') {
                while j < bytes.len() && matches!(bytes[j], b' ' | b'\t' | b'\r' | b'\n') {
                    j += 1;
                }
            }
            if j == bytes.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
            start = j;
            i = j;
            last_sep_end = Some(j);
        } else {
            i += 1;
        }
    }
    let tail = &s[start..];
    let toks = split_tokens(tail, start);
    if toks.is_empty() {
        if last_sep_end.is_some() {
            return Err(RangeError::Empty { at: s.len() });
        }
        return Err(RangeError::Empty { at: 0 });
    }
    arms.push(parse_arm(&toks)?);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty() && !include_prerelease && !arm.anchors.contains(&core(v)) {
        return false;
    }
    arm.comparators.iter().all(|c| cmp_satisfies(v, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(candidates: &[String], r: &Range, greatest: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(current) => {
                let ord = compare_versions(version, &parsed[current]);
                if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

fn bound_cmp(a: &Version, b: &Version) -> Ordering {
    compare_versions(a, b)
}

fn arm_bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>, Option<Version>) {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    let mut equal: Option<Version> = None;

    for c in &arm.comparators {
        match c.op {
            Op::Eq => equal = Some(c.version.clone()),
            Op::Ge | Op::Gt => {
                let inclusive = matches!(c.op, Op::Ge);
                if lower
                    .as_ref()
                    .map_or(true, |(v, inc)| {
                        bound_cmp(&c.version, v) == Ordering::Greater
                            || (bound_cmp(&c.version, v) == Ordering::Equal && !inclusive && *inc)
                    })
                {
                    lower = Some((c.version.clone(), inclusive));
                }
            }
            Op::Le | Op::Lt => {
                let inclusive = matches!(c.op, Op::Le);
                if upper
                    .as_ref()
                    .map_or(true, |(v, inc)| {
                        bound_cmp(&c.version, v) == Ordering::Less
                            || (bound_cmp(&c.version, v) == Ordering::Equal && !inclusive && *inc)
                    })
                {
                    upper = Some((c.version.clone(), inclusive));
                }
            }
        }
    }

    if let Some(v) = equal {
        lower = Some((v.clone(), true));
        upper = Some((v.clone(), true));
        (lower, upper, Some(v))
    } else {
        (lower, upper, None)
    }
}

fn version_string(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn arm_canonical(arm: &Arm) -> String {
    let (lower, upper, _) = arm_bounds(arm);
    let lo = lower.map_or_else(
        || "*".to_string(),
        |(v, inclusive)| format!("{}{}", if inclusive { ">=" } else { ">" }, version_string(&v)),
    );
    let hi = upper.map_or_else(
        || "*".to_string(),
        |(v, inclusive)| format!("{}{}", if inclusive { "<=" } else { "<" }, version_string(&v)),
    );
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let anchor_s = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors
            .iter()
            .map(|(a, b, c)| format!("{a}.{b}.{c}"))
            .collect::<Vec<_>>()
            .join(",")
    };
    format!("[{lo};{hi};pre={anchor_s}]")
}

pub fn canonical(r: &Range) -> String {
    let mut arms: Vec<String> = r.arms.iter().map(arm_canonical).collect();
    arms.sort();
    arms.dedup();
    arms.join(" || ")
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    // Stable releases are sufficient whenever the closed core interval has a release witness.
    for aa in &a.arms {
        for bb in &b.arms {
            for major in 0..=u64::MAX {
                let candidate = Version {
                    major,
                    minor: 0,
                    patch: 0,
                    pre: Vec::new(),
                    build: Vec::new(),
                };
                if arm_matches(&candidate, aa, true) && arm_matches(&candidate, bb, true) {
                    return true;
                }
                if major == u64::MAX {
                    break;
                }
            }
        }
    }
    let _ = include_prerelease;
    false
}
