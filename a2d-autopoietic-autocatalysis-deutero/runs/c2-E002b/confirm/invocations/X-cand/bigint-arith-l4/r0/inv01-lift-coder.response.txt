```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn valid_radix(radix: u32) -> Result<(), Error> {
        if !(2..=36).contains(&radix) {
            Err(Error::InvalidRadix { radix })
        } else {
            Ok(())
        }
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::valid_radix(radix)?;
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for i in start..bytes.len() {
            if bytes[i] != b'_' && Self::digit_value(bytes[i]).map_or(true, |d| d >= radix) {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                return Err(Error::Separator { index: i });
            }
        }

        let digit_indices: Vec<usize> = (start..bytes.len())
            .filter(|&i| bytes[i] != b'_')
            .collect();

        if digit_indices.len() > 1 && bytes[digit_indices[0]] == b'0' {
            return Err(Error::LeadingZero { index: digit_indices[0] });
        }

        if digit_indices.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_indices[max_digits],
            });
        }

        let mut result = Self::zero();
        for &i in &digit_indices {
            result.mul_small_assign(radix);
            result.add_small_assign(Self::digit_value(bytes[i]).unwrap());
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    fn mul_small_assign(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let n = (*limb as u64) * factor as u64 + carry;
            *limb = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_assign(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let n = self.limbs[i] as u64 + carry;
            self.limbs[i] = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
            i += 1;
        }
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add(a: &Self, b: &Self) -> Self {
        let mut result = Self::zero();
        let mut carry = 0u64;
        let n = a.limbs.len().max(b.limbs.len());
        for i in 0..n {
            let x = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let y = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = x + y + carry;
            result.limbs.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            result.limbs.push(carry as u32);
        }
        result
    }

    // Requires |a| >= |b|.
    fn abs_sub(a: &Self, b: &Self) -> Self {
        let mut result = Self::zero();
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let x = a.limbs[i] as i64 - borrow;
            let y = b.limbs.get(i).copied().unwrap_or(0) as i64;
            if x < y {
                result.limbs.push((x + Self::BASE as i64 - y) as u32);
                borrow = 1;
            } else {
                result.limbs.push((x - y) as u32);
                borrow = 0;
            }
        }
        result.normalize();
        result
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.abs_cmp(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }
        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut remainder = Self::zero();
        for &limb in a.limbs.iter().rev() {
            remainder.limbs.insert(0, limb);
            remainder.normalize();

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = b.mul_small(mid as u32);
                match product.abs_cmp(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }
            quotient_be.push(best as u32);
            if best != 0 {
                remainder = Self::abs_sub(&remainder, &b.mul_small(best as u32));
            }
        }
        quotient_be.reverse();
        let mut quotient = Self { negative: false, limbs: quotient_be };
        quotient.normalize();
        (quotient, remainder)
    }

    fn mul_small(&self, factor: u32) -> Self {
        let mut result = self.abs();
        result.mul_small_assign(factor);
        result
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        Self::valid_radix(radix)?;
        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push('0');
        } else {
            while !n.is_zero() {
                let (q, r) = n.div_small(radix);
                digits.push(Self::digit_char(r));
                n = q;
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, '0');
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, ch) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(*ch);
        }
        Ok(out)
    }

    fn digit_char(d: u32) -> char {
        if d < 10 {
            (b'0' + d as u8) as char
        } else {
            (b'a' + (d - 10) as u8) as char
        }
    }

    fn div_small(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = Self::zero();
        quotient.limbs.resize(self.limbs.len(), 0);
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let n = remainder * Self::BASE + self.limbs[i] as u64;
            quotient.limbs[i] = (n / divisor as u64) as u32;
            remainder = n % divisor as u64;
        }
        quotient.normalize();
        (quotient, remainder as u32)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::abs_add(self, rhs);
            result.negative = self.negative;
            result.normalize();
            result
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::abs_sub(self, rhs);
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = Self::abs_sub(rhs, self);
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        let mut neg_rhs = rhs.clone();
        if !neg_rhs.is_zero() {
            neg_rhs.negative = !neg_rhs.negative;
        }
        self.add(&neg_rhs)
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = Self { negative: self.negative ^ rhs.negative, limbs: vec![0; self.limbs.len() + rhs.limbs.len()] };
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let n = result.limbs[k] as u64 + a as u64 * b as u64 + carry;
                result.limbs[k] = (n % Self::BASE) as u32;
                carry = n / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                let n = result.limbs[k] as u64 + carry;
                result.limbs[k] = (n % Self::BASE) as u32;
                carry = n / Self::BASE;
                k += 1;
                if k == result.limbs.len() && carry != 0 {
                    result.limbs.push(0);
                }
            }
        }
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = Self::div_rem_abs(self, rhs);
        q.negative = !q.is_zero() && (self.negative ^ rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: Self::POW_LIMIT });
        }
        let mut n = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while n != 0 {
            if n & 1 == 1 {
                result = result.mul(&base);
            }
            n >>= 1;
            if n != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, r) = Self::div_rem_abs(&a, &b);
            a = b;
            b = r;
        }
        a
    }
}
```
