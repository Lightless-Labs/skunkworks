```rust
use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 1,000,000,000 limbs.
    limbs: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("limbs", &self.limbs)
            .finish()
    }
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn abs(&self) -> Self {
        Self { negative: false, limbs: self.limbs.clone() }
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len()).then_with(|| {
            self.limbs.iter().rev().cmp(rhs.limbs.iter().rev())
        })
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let x = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let y = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = x + y + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self { negative: false, limbs: out }
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let x = a.limbs[i] as i64;
            let y = b.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut d = x - y - borrow;
            if d < 0 {
                d += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(d as u32);
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn mul_small_abs(&mut self, m: u32) {
        if m == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let v = (*limb as u64) * (m as u64) + carry;
            *limb = (v % Self::BASE) as u32;
            carry = v / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_abs(&mut self, a: u32) {
        let mut carry = a as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let v = self.limbs[i] as u64 + carry;
            self.limbs[i] = (v % Self::BASE) as u32;
            carry = v / Self::BASE;
            i += 1;
        }
    }

    fn div_small_abs(&mut self, d: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let v = rem * Self::BASE + *limb as u64;
            *limb = (v / d as u64) as u32;
            rem = v % d as u64;
        }
        self.normalize();
        rem as u32
    }

    fn mul_abs(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut accum = vec![0u128; a.limbs.len() + b.limbs.len()];
        for (i, &x) in a.limbs.iter().enumerate() {
            for (j, &y) in b.limbs.iter().enumerate() {
                accum[i + j] += (x as u128) * (y as u128);
            }
        }
        let base = Self::BASE as u128;
        let mut limbs = Vec::with_capacity(accum.len() + 1);
        let mut carry = 0u128;
        for v in accum {
            let v = v + carry;
            limbs.push((v % base) as u32);
            carry = v / base;
        }
        while carry != 0 {
            limbs.push((carry % base) as u32);
            carry /= base;
        }
        let mut result = Self { negative: false, limbs };
        result.normalize();
        result
    }

    fn shift_add_base(&mut self, digit: u32) {
        if self.is_zero() {
            if digit != 0 {
                self.limbs.push(digit);
            }
        } else {
            self.limbs.insert(0, digit);
        }
    }

    // Positive magnitude division.
    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        if a.cmp_abs(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }
        let mut quotient = vec![0u32; a.limbs.len()];
        let mut rem = Self::zero();
        for i in (0..a.limbs.len()).rev() {
            rem.shift_add_base(a.limbs[i]);
            let mut lo = 0u64;
            let mut hi = Self::BASE - 1;
            while lo < hi {
                let mid = (lo + hi + 1) / 2;
                let mut product = b.abs();
                product.mul_small_abs(mid as u32);
                if product.cmp_abs(&rem) != Ordering::Greater {
                    lo = mid;
                } else {
                    hi = mid - 1;
                }
            }
            quotient[i] = lo as u32;
            if lo != 0 {
                let mut product = b.abs();
                product.mul_small_abs(lo as u32);
                rem = Self::sub_abs(&rem, &product);
            }
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        (q, rem)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            if !Self::digit_value(bytes[i]).is_some_and(|d| d < radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                separator = Some(i);
                break;
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let first_digit = start;
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }
        if digit_count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen > max_digits {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte != b'_' {
                value.mul_small_abs(radix);
                value.add_small_abs(Self::digit_value(byte).unwrap());
            }
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut value = self.abs();
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(0u32);
        } else {
            while !value.is_zero() {
                digits.push(value.div_small_abs(radix));
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut magnitude = String::new();
        for (i, d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                magnitude.push('_');
            }
            magnitude.push(if *d < 10 {
                (b'0' + *d as u8) as char
            } else {
                (b'a' + (*d - 10) as u8) as char
            });
        }
        if self.negative && !self.is_zero() {
            Ok(format!("-{magnitude}"))
        } else {
            Ok(magnitude)
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self::add_abs(self, rhs);
            out.negative = self.negative;
            out.normalize();
            out
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self::sub_abs(self, rhs);
                    out.negative = self.negative;
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self::sub_abs(rhs, self);
                    out.negative = rhs.negative;
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut out = Self::mul_abs(self, rhs);
        out.negative = !out.is_zero() && (self.negative != rhs.negative);
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = Self::div_rem_abs(self, rhs);
        q.negative = !q.is_zero() && (self.negative != rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 != 0 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, r) = Self::div_rem_abs(&a, &b);
            a = b;
            b = r;
        }
        a
    }
}
```
