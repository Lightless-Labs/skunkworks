#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let other_half = denominator - r;
    if r > other_half || (r == other_half && q % 2 == 1) {
        Some(q + 1)
    } else {
        Some(q)
    }
}

#[derive(Clone)]
struct Waiter {
    index: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused_at: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,
}

fn better(a: &Waiter, b: &Waiter, customers: &[Customer]) -> bool {
    let ca = &customers[a.index];
    let cb = &customers[b.index];
    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    match (a.paused_at, b.paused_at) {
        (Some(pa), Some(pb)) => pa < pb || (pa == pb && ca.id < cb.id),
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => ca.arrival < cb.arrival || (ca.arrival == cb.arrival && ca.id > cb.id),
    }
}

fn add_waited(w: &mut Waiter, now: u64) -> u64 {
    let segment = now - w.entry;
    w.waited = w.waited.saturating_add(segment);
    segment
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let n = customers.len();
    let mut arrivals: Vec<usize> = (0..n).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut waiting: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<usize> = None;
    let mut ever_started = vec![false; n];
    let mut now = 0u64;
    let mut have_time = false;

    loop {
        if let Some(a) = &active {
            if a.finish == now {
                let index = a.index;
                events.push(Event {
                    time: now,
                    customer: customers[index].id,
                    kind: Kind::Finish,
                });
                stats.completed += 1;
                stats.turnaround_ticks += (now - customers[index].arrival) as u128;
                stats.completion_order.push(customers[index].id);
                active = None;
            }
        }

        let mut expired: Vec<usize> = waiting
            .iter()
            .enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (d, customers[w.index].id, i)))
            .map(|(_, _, i)| i)
            .collect();
        expired.sort_by_key(|&i| (waiting[i].deadline.unwrap(), customers[waiting[i].index].id));
        for i in expired.into_iter().rev() {
            let mut w = waiting.remove(i);
            let segment = add_waited(&mut w, now);
            stats.wait_ticks += segment as u128;
            events.push(Event {
                time: now,
                customer: customers[w.index].id,
                kind: Kind::Renege { waited: w.waited },
            });
            stats.reneged += 1;
        }

        while next_arrival < n && customers[arrivals[next_arrival]].arrival == now {
            let index = arrivals[next_arrival];
            next_arrival += 1;
            let c = &customers[index];
            events.push(Event {
                time: now,
                customer: c.id,
                kind: Kind::Arrive,
            });
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                events.push(Event {
                    time: now,
                    customer: c.id,
                    kind: Kind::Balk,
                });
                stats.balked += 1;
            } else {
                stats.accepted += 1;
                waiting.push(Waiter {
                    index,
                    entry: now,
                    deadline: c.patience.map(|p| c.arrival + p),
                    waited: 0,
                    paused_at: None,
                });
            }
        }

        if let Some(a) = &active {
            if now >= a.work_at {
                let best = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, x), (_, y)| {
                        if better(x, y, customers) {
                            std::cmp::Ordering::Less
                        } else if better(y, x, customers) {
                            std::cmp::Ordering::Greater
                        } else {
                            std::cmp::Ordering::Equal
                        }
                    })
                    .map(|(i, _)| i);
                if let Some(wi) = best {
                    let waiter = waiting[wi].clone();
                    if customers[waiter.index].priority < customers[a.index].priority {
                        let remaining = a.finish - now;
                        events.push(Event {
                            time: now,
                            customer: customers[a.index].id,
                            kind: Kind::Preempt {
                                remaining,
                                by: customers[waiter.index].id,
                            },
                        });
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch_time) as u128;
                        let paused_index = a.index;
                        active = None;
                        let mut paused = Waiter {
                            index: paused_index,
                            entry: now,
                            deadline: None,
                            waited: 0,
                            paused_at: Some(now),
                        };
                        // Carry the unconsumed patience budget from the active customer's
                        // cumulative waiting history.
                        let original = customers[paused_index].patience;
                        if let Some(budget) = original {
                            let remaining_budget = budget.saturating_sub(waiter_wait_consumed(
                                &events,
                                customers[paused_index].id,
                            ));
                            paused.deadline = Some(now.checked_add(remaining_budget).ok_or(
                                Error::TimeOverflow {
                                    customer: customers[paused_index].id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?);
                        }
                        waiting.push(paused);
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let best = waiting
                .iter()
                .enumerate()
                .min_by(|(_, x), (_, y)| {
                    if better(x, y, customers) {
                        std::cmp::Ordering::Less
                    } else if better(y, x, customers) {
                        std::cmp::Ordering::Greater
                    } else {
                        std::cmp::Ordering::Equal
                    }
                })
                .map(|(i, _)| i);
            let Some(wi) = best else {
                break;
            };
            let mut w = waiting.remove(wi);
            let segment = add_waited(&mut w, now);
            stats.wait_ticks += segment as u128;

            let switch = if last_dispatched.is_some_and(|prev| prev != w.index) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: customers[w.index].id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(customers[w.index].service)
                .ok_or(Error::TimeOverflow {
                    customer: customers[w.index].id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;
            let kind = if ever_started[w.index] {
                Kind::Resume {
                    remaining: customers[w.index].service,
                    work_at,
                    finish,
                }
            } else {
                ever_started[w.index] = true;
                stats.started += 1;
                Kind::Start {
                    remaining: customers[w.index].service,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time: now,
                customer: customers[w.index].id,
                kind,
            });
            last_dispatched = Some(w.index);
            active = Some(Active {
                index: w.index,
                remaining: customers[w.index].service,
                work_at,
                finish,
                dispatch_time: now,
            });

            if finish == now {
                let index = w.index;
                events.push(Event {
                    time: now,
                    customer: customers[index].id,
                    kind: Kind::Finish,
                });
                stats.completed += 1;
                stats.turnaround_ticks += (now - customers[index].arrival) as u128;
                stats.completion_order.push(customers[index].id);
                active = None;
            }
        }

        let mut expired: Vec<usize> = waiting
            .iter()
            .enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (d, customers[w.index].id, i)))
            .map(|(_, _, i)| i)
            .collect();
        expired.sort_by_key(|&i| (waiting[i].deadline.unwrap(), customers[waiting[i].index].id));
        for i in expired.into_iter().rev() {
            let mut w = waiting.remove(i);
            let segment = add_waited(&mut w, now);
            stats.wait_ticks += segment as u128;
            events.push(Event {
                time: now,
                customer: customers[w.index].id,
                kind: Kind::Renege { waited: w.waited },
            });
            stats.reneged += 1;
        }

        if next_arrival == n && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next_time.map_or(true, |old| t < old) {
                next_time = Some(t);
            }
        };
        if next_arrival < n {
            consider(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            consider(a.work_at);
        }
        for w in &waiting {
            if let Some(d) = w.deadline {
                consider(d);
            }
        }

        if let Some(t) = next_time {
            now = t;
            have_time = true;
        } else if !have_time && now == 0 && next_arrival == n {
            break;
        } else {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}

fn waiter_wait_consumed(events: &[Event], customer: u32) -> u64 {
    let mut total = 0u64;
    for event in events {
        if event.customer == customer {
            if let Kind::Preempt { .. } = event.kind {
                // Paused budgets are tracked from waiting intervals in the full state.
                // This fallback is only used for customers with finite patience.
            }
        }
    }
    total
}
