fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_rank(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(e: &[Evidence]) -> Vec<Evidence> {
    let mut v = e.to_vec();
    v.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    v
}

fn evidence_lines(e: &[Evidence]) -> String {
    sorted_evidence(e)
        .iter()
        .map(|x| {
            format!(
                "\n  C:{} via {}",
                constraint_text(x.constraint),
                x.path.join(" -> ")
            )
        })
        .collect()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn feature_names(mut names: Vec<String>) -> Vec<String> {
    names.sort();
    names.dedup();
    names
}

fn root_token(i: usize, r: &Requirement) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        feature_names(r.features.clone()).join(",")
    )
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn target_matches(p: Platform, t: Target) -> bool {
    p == Platform::Any
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn dep_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        feature_names(dep.features.clone()).join(","),
        dep.gate.as_deref().unwrap_or("*"),
        platform_text(dep.platform)
    )
}

fn add_origin(active: &mut Active, package: &str, origin: Origin) {
    active
        .origins
        .entry(package.to_string())
        .or_default()
        .push(origin);
}

fn canonical_path_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut active = Active::default();
    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        add_origin(
            &mut active,
            &root.package,
            Origin {
                constraint: root.constraint,
                path: path.clone(),
            },
        );
        let current = active.paths.entry(root.package.clone()).or_insert(path.clone());
        if canonical_path_cmp(&path, current).is_lt() {
            *current = path;
        }
        active
            .features
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
    }

    loop {
        let before_origins = active.origins.clone();
        let before_features = active.features.clone();
        let before_paths = active.paths.clone();

        let package_names: Vec<String> = selected.keys().cloned().collect();
        for package_name in package_names {
            let Some(version) = selected.get(&package_name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&package_name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = active.features.get(&package_name).cloned().unwrap_or_default();
            let mut activated = requested.clone();
            let mut queue: Vec<String> = requested.into_iter().collect();
            while let Some(name) = queue.pop() {
                if let Some(feature) = release.features.iter().find(|f| f.name == name) {
                    for enabled in &feature.enables {
                        if activated.insert(enabled.clone()) {
                            queue.push(enabled.clone());
                        }
                    }
                }
            }
            active.features.entry(package_name.clone()).or_default().extend(activated.clone());

            let source_path = active.paths.get(&package_name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }
                let token = dep_token(&package_name, version, j, dep);
                let mut path = source_path.clone();
                path.push(token);
                add_origin(
                    &mut active,
                    &dep.package,
                    Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    },
                );
                let current = active.paths.entry(dep.package.clone()).or_insert(path.clone());
                if canonical_path_cmp(&path, current).is_lt() {
                    *current = path;
                }
                active
                    .features
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
            }
        }

        if active.origins == before_origins
            && active.features == before_features
            && active.paths == before_paths
        {
            break;
        }
    }
    active
}

fn canonical_requirements(active: &Active, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        active
            .origins
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| o.constraint)
            .collect(),
    )
}

fn package_evidence(active: &Active, package: &str) -> Vec<Evidence> {
    let mut result: Vec<Evidence> = active
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    result.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    result
}

fn dynamic_error(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> (Active, Option<ResolveError>) {
    let active = compute_active(catalog, roots, selected, target);

    if let Some(name) = active
        .origins
        .keys()
        .find(|name| !catalog.contains_key(*name))
    {
        return (
            active.clone(),
            Some(ResolveError::Missing {
                package: name.clone(),
                evidence: package_evidence(&active, name),
            }),
        );
    }

    for (name, version) in selected {
        if let Some(reqs) = active.origins.get(name) {
            if reqs.iter().any(|o| !matches(*version, o.constraint)) {
                return (
                    active.clone(),
                    Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical_requirements(&active, name),
                        evidence: package_evidence(&active, name),
                    }),
                );
            }
        }
    }

    for (name, names) in &active.features {
        if let Some(version) = selected.get(name) {
            if let Some(package) = catalog.get(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for feature in names {
                        if !release.features.iter().any(|f| f.name == *feature) {
                            return (
                                active.clone(),
                                Some(ResolveError::MissingFeature {
                                    package: name.clone(),
                                    selected: *version,
                                    feature: feature.clone(),
                                    path: active.paths.get(name).cloned().unwrap_or_default(),
                                }),
                            );
                        }
                    }
                }
            }
        }
    }

    for name in active.origins.keys() {
        if selected.contains_key(name) {
            continue;
        }
        let package = catalog.get(name).unwrap();
        let constraints = canonical_requirements(&active, name);
        let has_match = package.releases.iter().any(|release| {
            active.origins[name]
                .iter()
                .all(|origin| matches(release.version, origin.constraint))
        });
        if !has_match {
            return (
                active.clone(),
                Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints,
                    evidence: package_evidence(&active, name),
                }),
            );
        }
    }
    (active, None)
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>, Active)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (active, error) = dynamic_error(self.catalog, self.roots, selected, self.target);
        if let Some(error) = error {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active
            .origins
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(package_name) = next else {
            let selections: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_selections, _)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        let names: BTreeSet<String> = self
                            .catalog
                            .keys()
                            .chain(selected.keys())
                            .cloned()
                            .collect();
                        for name in names {
                            let a = selected.get(&name);
                            let b = best_selections
                                .iter()
                                .find(|s| s.package == name)
                                .map(|s| &s.version);
                            if a != b {
                                break a > b;
                            }
                        }
                        false
                    }
                }
            };
            if replace {
                self.best = Some((cost, selections, active));
            }
            return Ok(());
        };

        let package = self.catalog[&package_name];
        let constraints: Vec<Constraint> = active.origins[&package_name]
            .iter()
            .map(|o| o.constraint)
            .collect();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock) = self.locks.get(&package_name) {
            if let Some(i) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), version);
            let result = self.visit(selected);
            selected.remove(&package_name);
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.enable[{}]", base, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let locks_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: &locks_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    search.visit(&mut BTreeMap::new())?;

    if let Some((_, selected, active)) = search.best {
        let selected_map: BTreeMap<String, Version> = selected
            .iter()
            .map(|s| (s.package.clone(), s.version))
            .collect();
        let mut all_names: BTreeSet<String> = locks_map.keys().cloned().collect();
        all_names.extend(selected_map.keys().cloned());
        let mut changes = Vec::new();

        for package in all_names {
            let before = locks_map.get(&package).copied();
            let after = selected_map.get(&package).copied();
            if before == after {
                continue;
            }
            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let exists = catalog_map
                        .get(&package)
                        .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                    if !exists {
                        ChangeReason::Unavailable
                    } else if active
                        .origins
                        .get(&package)
                        .into_iter()
                        .flatten()
                        .any(|o| !matches(old, o.constraint))
                    {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
                (None, None) => continue,
            };
            changes.push(Change {
                package,
                before,
                after,
                reason,
            });
        }
        Ok(Resolution { selected, changes })
    } else {
        Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }))
    }
}
