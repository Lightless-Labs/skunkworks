```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

struct PendingJob {
    id: u32,
    remaining: u32,
    level: usize,
    completion_index: usize,
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut ready: Vec<VecDeque<PendingJob>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0_u64;
    let mut next_arrival = 0;
    let mut last_dispatched: Option<u32> = None;

    loop {
        while next_arrival < arrival_order.len()
            && u64::from(jobs[arrival_order[next_arrival]].arrival) <= now
        {
            let index = arrival_order[next_arrival];
            let job = &jobs[index];
            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
            } else {
                ready[job.level].push_back(PendingJob {
                    id: job.id,
                    remaining: job.ticks,
                    level: job.level,
                    completion_index: index,
                });
            }
            next_arrival += 1;
        }

        let selected_level = ready.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if next_arrival == arrival_order.len() {
                break;
            }

            let next_time = u64::from(jobs[arrival_order[next_arrival]].arrival);
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatched = None;
            }
            continue;
        };

        let mut selected = ready[level].pop_front().expect("selected queue is nonempty");

        if last_dispatched.is_some_and(|id| id != selected.id) {
            now += u64::from(switch_cost);
        }

        while next_arrival < arrival_order.len()
            && u64::from(jobs[arrival_order[next_arrival]].arrival) <= now
        {
            let index = arrival_order[next_arrival];
            let job = &jobs[index];
            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
            } else {
                ready[job.level].push_back(PendingJob {
                    id: job.id,
                    remaining: job.ticks,
                    level: job.level,
                    completion_index: index,
                });
            }
            next_arrival += 1;
        }

        let start = now;
        let consumed = selected.remaining.min(quanta[level]);
        now += u64::from(consumed);
        selected.remaining -= consumed;
        let finished = selected.remaining == 0;

        runs.push(Run {
            id: selected.id,
            start,
            end: now,
            finished,
            level,
        });

        while next_arrival < arrival_order.len()
            && u64::from(jobs[arrival_order[next_arrival]].arrival) <= now
        {
            let index = arrival_order[next_arrival];
            let job = &jobs[index];
            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
            } else {
                ready[job.level].push_back(PendingJob {
                    id: job.id,
                    remaining: job.ticks,
                    level: job.level,
                    completion_index: index,
                });
            }
            next_arrival += 1;
        }

        if finished {
            completion[selected.completion_index] = Some(now);
        } else {
            selected.level = (level + 1).min(quanta.len() - 1);
            ready[selected.level].push_back(selected);
        }

        last_dispatched = Some(runs.last().expect("just emitted a run").id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].expect("every job completes")))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}
```
