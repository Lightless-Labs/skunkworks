use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(b: u8) -> bool {
    matches!(b, b'.' | b'-' | b'+')
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            return Err(if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
                VersionError::MissingCore { at: start }
            } else {
                VersionError::InvalidChar { at: start }
            });
        }

        let digits = &bytes[start..pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut n = 0u64;
        for &d in digits {
            match n.checked_mul(10).and_then(|x| x.checked_add((d - b'0') as u64)) {
                Some(value) => n = value,
                None => return Err(VersionError::Overflow { at: start }),
            }
        }
        core[field] = n;

        if field < 2 {
            if pos >= bytes.len() || bytes[pos] != b'.' {
                return Err(if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
                    VersionError::MissingCore { at: pos }
                } else {
                    VersionError::InvalidChar { at: pos }
                });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..].iter().position(|&b| b == b'+')
            .map(|i| pre_start + i)
            .unwrap_or(bytes.len());
        parse_identifiers(s, pre_start, plus, true, &mut pre)?;
        pos = plus;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let start = pos + 1;
        parse_identifiers(s, start, bytes.len(), false, &mut build)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    check_numeric_zero: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let b = s.as_bytes();
    let mut id_start = start;
    loop {
        let mut i = id_start;
        while i < end && b[i] != b'.' {
            if !b[i].is_ascii_alphanumeric() && b[i] != b'-' {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if check_numeric_zero
            && b[id_start..i].iter().all(u8::is_ascii_digit)
            && i - id_start > 1
            && b[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(s[id_start..i].to_string());
        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(())
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_ids(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric_id(x);
        let yn = is_numeric_id(y);
        let order = match (xn, yn) {
            (true, true) => cmp_numeric_ids(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if order != Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

pub enum RangeError {

    Syntax { at: usize },
    Version { at: usize, error: VersionError },

#[derive(Clone, Debug)]
enum Op {

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,

pub struct Range {
    arms: Vec<Arm>,

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens_with_offsets(s: &str, start: usize, end: usize) -> Vec<(usize, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && whitespace(b[i]) {
            i += 1;
        }
        if i >= end {
            break;
        }
        let st = i;
        while i < end && !whitespace(b[i]) {
            i += 1;
        }
        out.push((st, i));
    }
    out
}

fn version_operand(s: &str, at: usize, end: usize) -> Result<Version, RangeError> {
    parse_version(&s[at..end]).map_err(|error| RangeError::Version { at, error })
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: vec!["0".to_string()], build: Vec::new() }
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: fields.first().copied().unwrap_or(0),
        minor: fields.get(1).copied().unwrap_or(0),
        patch: fields.get(2).copied().unwrap_or(0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn parse_partial(s: &str, at: usize, end: usize) -> Result<(Vec<u64>, usize, bool), RangeError> {
    let text = &s[at..end];
    if text.is_empty() {
        return Err(RangeError::Syntax { at });
    }
    let parts: Vec<&str> = text.split('.').collect();
    if parts.len() > 3 || parts.iter().any(|p| p.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut saw_wild = false;
    for p in &parts {
        if matches!(*p, "*" | "x" | "X") {
            saw_wild = true;
        } else {
            if saw_wild || !p.bytes().all(|b| b.is_ascii_digit()) || (p.len() > 1 && p.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok((nums, parts.len(), saw_wild))
}

fn is_fullish(s: &str, at: usize, end: usize) -> bool {
    let x = &s[at..end];
    x.contains('-') || x.contains('+') ||
        (x.split('.').count() >= 3 && !x.split('.').any(|p| matches!(p, "*" | "x" | "X")))
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(version_tuple(&version));
    }
    arm.cmps.push(Cmp { op, version, anchor });
}

fn bump_major(v: &Version, at: usize) -> Result<Version, RangeError> {
    let m = v.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
    Ok(endpoint(m, 0, 0))
}
fn bump_minor(v: &Version, at: usize) -> Result<Version, RangeError> {
    let m = v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
    Ok(endpoint(v.major, m, 0))
}
fn bump_patch(v: &Version, at: usize) -> Result<Version, RangeError> {
    let p = v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    Ok(endpoint(v.major, v.minor, p))
}

fn next_partial(v: &Version, k: usize, at: usize) -> Result<Version, RangeError> {
    if k == 1 { bump_major(v, at) } else { bump_minor(v, at) }
}

fn parse_flexible(s: &str, at: usize, end: usize) -> Result<(Version, usize, bool), RangeError> {
    if is_fullish(s, at, end) {
        let v = version_operand(s, at, end)?;
        return Ok((v, 3, true));
    }
    let (nums, _count, _wild) = parse_partial(s, at, end)?;
    Ok((padded(&nums), nums.len(), false))
}

fn parse_hyphen_arm(
    s: &str,
    toks: &[(usize, usize)],
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let minus = toks.iter().position(|&(a, b)| &s[a..b] == "-").unwrap();
    if toks.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax { at: toks[minus].0 });
    }
    let (la, lb) = toks[0];
    let (ua, ub) = toks[2];
    let (lower, lk, lf) = parse_flexible(s, la, lb)?;
    let (upper, uk, uf) = parse_flexible(s, ua, ub)?;
    if lk == 0 {
        return Err(RangeError::Syntax { at: la });
    }
    if uk == 0 {
        return Err(RangeError::Syntax { at: ua });
    }
    add_cmp(arm, Op::Ge, lower, lf);
    if uf {
        add_cmp(arm, Op::Le, upper, true);
    } else {
        let hi = next_partial(&upper, uk, ua)?;
        add_cmp(arm, Op::Lt, hi, false);
    }
    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let toks = tokens_with_offsets(s, start, end);
    if toks.is_empty() {
        return Err(RangeError::Empty { at: start });
    }
    let mut arm = Arm { cmps: Vec::new(), anchors: Vec::new() };

    if let Some(i) = toks.iter().position(|&(a, b)| &s[a..b] == "-") {
        if toks.len() != 3 || i != 1 {
            return Err(RangeError::Syntax { at: toks[i].0 });
        }
        parse_hyphen_arm(s, &toks, &mut arm)?;
        return Ok(arm);
    }

    for &(a, b) in &toks {
        let tok = &s[a..b];
        if tok.is_empty() {
            continue;
        }
        let first = tok.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            let (op, n) = if tok.starts_with("<=") { (Op::Le, 2) }
                else if tok.starts_with(">=") { (Op::Ge, 2) }
                else if tok.starts_with('<') { (Op::Lt, 1) }
                else if tok.starts_with('>') { (Op::Gt, 1) }
                else { (Op::Eq, 1) };
            let operand_at = a + n;
            if operand_at == b || !s.as_bytes()[operand_at].is_ascii_digit() {
                return Err(RangeError::Syntax { at: a });
            }
            let v = version_operand(s, operand_at, b)?;
            add_cmp(&mut arm, op, v, true);
            continue;
        }

        if first == b'^' || first == b'~' {
            let operand_at = a + 1;
            if operand_at == b {
                return Err(RangeError::Syntax { at: a });
            }
            let (lower, k, full) = parse_flexible(s, operand_at, b)?;
            if k == 0 {
                return Err(RangeError::Syntax { at: a });
            }
            add_cmp(&mut arm, Op::Ge, lower.clone(), full);
            let upper = if first == b'^' {
                if lower.major > 0 || k == 1 {
                    bump_major(&lower, operand_at)?
                } else if lower.minor > 0 || k == 2 {
                    bump_minor(&lower, operand_at)?
                } else {
                    bump_patch(&lower, operand_at)?
                }
            } else if k == 1 {
                bump_major(&lower, operand_at)?
            } else {
                bump_minor(&lower, operand_at)?
            };
            add_cmp(&mut arm, Op::Lt, upper, false);
            continue;
        }

        if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at: a });
        }
        let (v, k, full) = parse_flexible(s, a, b)?;
        if full {
            add_cmp(&mut arm, Op::Eq, v, true);
        } else if k == 0 {
            // Any all-wildcard partial denotes all versions.
        } else if k == 1 || k == 2 {
            add_cmp(&mut arm, Op::Ge, v.clone(), false);
            add_cmp(&mut arm, Op::Lt, next_partial(&v, k, a)?, false);
        } else {
            add_cmp(&mut arm, Op::Eq, v, false);
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut all = Vec::new();
    let mut i = 0;
    while i < b.len() && whitespace(b[i]) { i += 1; }
    if i == b.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arm_start = i;
    loop {
        let mut j = arm_start;
        while j < b.len() {
            if b[j..].starts_with(b"||") {
                break;
            }
            j += 1;
        }
        if j == b.len() {
            let end = b.len();
            let toks = tokens_with_offsets(s, arm_start, end);
            if toks.is_empty() {
                return Err(RangeError::Empty { at: end });
            }
            all.push(parse_arm(s, arm_start, end)?);
            break;
        }
        let sep = j;
        if sep > arm_start && !whitespace(b[sep - 1]) {
            // A glued separator remains part of its token.
            let mut k = sep + 2;
            while k < b.len() && !whitespace(b[k]) { k += 1; }
            // Continue searching after the glued token; it will be parsed as an operand.
            j = k;
            while j < b.len() && !b[j..].starts_with(b"||") { j += 1; }
            if j == b.len() {
                all.push(parse_arm(s, arm_start, b.len())?);
                break;
            }
            continue;
        }
        let mut after = sep + 2;
        if after < b.len() && !whitespace(b[after]) {
            // The separator token must be standalone.
            let mut k = after;
            while k < b.len() && !whitespace(b[k]) { k += 1; }
            let _ = k;
            all.push(parse_arm(s, arm_start, sep)?);
            return Err(RangeError::Syntax { at: sep });
        }
        let mut left_end = sep;
        while left_end > arm_start && whitespace(b[left_end - 1]) { left_end -= 1; }
        let toks = tokens_with_offsets(s, arm_start, left_end);
        if toks.is_empty() {
            return Err(RangeError::Empty { at: sep });
        }
        all.push(parse_arm(s, arm_start, left_end)?);
        while after < b.len() && whitespace(b[after]) { after += 1; }
        if after == b.len() {
            return Err(RangeError::Empty { at: b.len() });
        }
        if b[after..].starts_with(b"||") {
            return Err(RangeError::Empty { at: after });
        }
        arm_start = after;
    }

    Ok(Range { arms: all })
}

fn cmp_matches(v: &Version, c: &Cmp) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, a: &Arm, include_prerelease: bool) -> bool {
    if a.cmps.iter().any(|c| !cmp_matches(v, c)) {
        return false;
    }
    if !include_prerelease && !v.pre.is_empty() {
        let tuple = version_tuple(v);
        if !a.anchors.contains(&tuple) {
            return false;
        }
    }
    true
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (i, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index: i, error }),
        }
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug)]
enum Bound {
    Unbounded,
    Finite(Version, bool),
}

fn strongest_lower(a: &Arm) -> Bound {
    let mut best: Option<(Version, bool)> = None;
    for c in &a.cmps {
        let candidate = match c.op {
            Op::Eq => Some((c.version.clone(), true)),
            Op::Gt => Some((c.version.clone(), false)),
            Op::Ge => Some((c.version.clone(), true)),
            _ => None,
        };
        if let Some((v, inc)) = candidate {
            if let Some((old, old_inc)) = &best {
                let ord = compare_versions(&v, old);
                if ord == Ordering::Greater || (ord == Ordering::Equal && !inc && *old_inc) {
                    best = Some((v, inc));
                }
            } else {
                best = Some((v, inc));
            }
        }
    }
    best.map(|(v, i)| Bound::Finite(v, i)).unwrap_or(Bound::Unbounded)
}

fn strongest_upper(a: &Arm) -> Bound {
    let mut best: Option<(Version, bool)> = None;
    for c in &a.cmps {
        let candidate = match c.op {
            Op::Eq => Some((c.version.clone(), true)),
            Op::Lt => Some((c.version.clone(), false)),
            Op::Le => Some((c.version.clone(), true)),
            _ => None,
        };
        if let Some((v, inc)) = candidate {
            if let Some((old, old_inc)) = &best {
                let ord = compare_versions(&v, old);
                if ord == Ordering::Less || (ord == Ordering::Equal && !inc && *old_inc) {
                    best = Some((v, inc));
                }
            } else {
                best = Some((v, inc));
            }
        }
    }
    best.map(|(v, i)| Bound::Finite(v, i)).unwrap_or(Bound::Unbounded)
}

fn clean_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut strs = Vec::new();
    for a in &r.arms {
        let lower = strongest_lower(a);
        let upper = strongest_upper(a);
        let mut lo = None;
        let mut hi = None;
        let mut impossible = false;
        if let Bound::Finite(l, li) = &lower {
            if let Bound::Finite(u, ui) = &upper {
                match compare_versions(l, u) {
                    Ordering::Greater => impossible = true,
                    Ordering::Equal if !(*li && *ui) => impossible = true,
                    _ => {}
                }
            }
        }
        if impossible { continue; }
        match lower {
            Bound::Unbounded => {}
            Bound::Finite(v, inc) => lo = Some(format!("{}{}", if inc { ">=" } else { ">" }, clean_version(&v))),
        }
        match upper {
            Bound::Unbounded => {}
            Bound::Finite(v, inc) => hi = Some(format!("{}{}", if inc { "<=" } else { "<" }, clean_version(&v))),
        }
        let mut anchors = a.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors.iter().map(|(x,y,z)| format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")
        };
        strs.push(format!("[{};{};pre={}]", lo.unwrap_or_else(|| "*".into()), hi.unwrap_or_else(|| "*".into()), anchor_text));
    }
    strs.sort();
    strs.dedup();
    if strs.is_empty() { "EMPTY".into() } else { strs.join(" || ") }
}

}

fn validate_expr(
    package: &str,
    label: &str,
    expr: &SetExpr,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| SelectError::InvalidRange {
            package: package.to_string(),
            label: label.to_string(),
            path: path.clone(),
            error,
        }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(package, label, child, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, include)).unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|e| eval_expr(e, v, include)),
        SetExpr::Intersection(children) => children.iter().all(|e| eval_expr(e, v, include)),
    }
}

fn constraint_set_satisfiable(
    package: &Package,
    parsed: &[Version],
    indices: &[usize],
    include: bool,
) -> bool {
    parsed.iter().any(|v| {
        indices.iter().all(|&ci| eval_expr(&package.constraints[ci].expr, v, include))
    })
}

fn lex_labels(package: &Package, indices: &[usize]) -> Vec<String> {
    let mut labels: Vec<String> = indices.iter().map(|&i| package.constraints[i].label.clone()).collect();
    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    labels
}

fn find_conflict(
    package: &Package,
    parsed: &[Version],
    include: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if parsed.is_empty() { return Vec::new(); }
    for size in 1..=n {
        let mut best: Option<Vec<String>> = None;
        let mut cur = Vec::with_capacity(size);
        fn rec(
            package: &Package,
            parsed: &[Version],
            include: bool,
            size: usize,
            next: usize,
            cur: &mut Vec<usize>,
            best: &mut Option<Vec<String>>,
        ) {
            if cur.len() == size {
                if !constraint_set_satisfiable(package, parsed, cur, include) {
                    let labels = lex_labels(package, cur);
                    if best.as_ref().map_or(true, |b| labels.as_slice() < b.as_slice()) {
                        *best = Some(labels);
                    }
                }
                return;
            }
            let need = size - cur.len();
            if next + need > package.constraints.len() { return; }
            for i in next..=package.constraints.len() - need {
                cur.push(i);
                rec(package, parsed, include, size, i + 1, cur, best);
                cur.pop();
            }
        }
        rec(package, parsed, include, size, 0, &mut cur, &mut best);
        if let Some(v) = best { return v; }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }

    let mut all_versions = Vec::with_capacity(packages.len());
    for p in packages {
        let mut parsed = Vec::with_capacity(p.candidates.len());
        for (i, s) in p.candidates.iter().enumerate() {
            match parse_version(s) {
                Ok(v) => parsed.push(v),
                Err(error) => return Err(SelectError::InvalidCandidate {
                    package: p.name.clone(),
                    index: i,
                    error,
                }),
            }
        }
        all_versions.push(parsed);
    }

    for p in packages {
        for c in &p.constraints {
            validate_expr(&p.name, &c.label, &c.expr, &mut Vec::new())?;
        }
    }

    let mut result = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let parsed = &all_versions[pi];
        let qualifying: Vec<usize> = parsed.iter().enumerate().filter_map(|(i, v)| {
            if p.constraints.iter().all(|c| eval_expr(&c.expr, v, include_prerelease)) {
                Some(i)
            } else {
                None
            }
        }).collect();

        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: find_conflict(p, parsed, include_prerelease),
            });
        }

        let stable_exists = matches!(preference, Preference::StableFirst)
            && qualifying.iter().any(|&i| parsed[i].pre.is_empty());
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let vi = &parsed[i];
            let vb = &parsed[best];
            if stable_exists && (vi.pre.is_empty() != vb.pre.is_empty()) {
                if vi.pre.is_empty() { best = i; }
                continue;
            }
            let ord = compare_versions(vi, vb);
            let take = match preference {
                Preference::Lowest => ord == Ordering::Less,
                Preference::Highest | Preference::StableFirst => ord == Ordering::Greater,
            };
            if take { best = i; }
        }
        result.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(result)
}
