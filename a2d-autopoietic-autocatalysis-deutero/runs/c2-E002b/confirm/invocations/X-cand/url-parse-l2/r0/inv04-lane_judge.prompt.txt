    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

    UserInfo,
    Path,
    Query,
    Fragment,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            let input_at = origins.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: input_at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_component(
    component: Component,
    value: Option<&str>,
    base: usize,
) -> Result<(), ParseError> {
    if let Some(value) = value {
        match decode_component(component, value) {
            Ok(_) => Ok(()),
            Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
            Err(CodecError::InvalidUtf8 { at }) => {
                Err(ParseError::InvalidUtf8 { at: base + at })
            }
        }
    } else {
        Ok(())
    }
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (offset, byte) in port_text.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }

    let bytes = port_text.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: port_start })?;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(&authority[..at]), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_absolute_start = authority_start + hostport_start;

    let (host, port, ipv6) = if hostport.as_bytes().first() == Some(&b'[') {
        let opening = hostport_absolute_start;
        let closing = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(offset) => offset + 1,
            None => return Err(ParseError::InvalidIpv6 { at: opening }),
        };
        let interior = &hostport[1..closing];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: opening });
        }

        let suffix = &hostport[closing + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(
                &suffix[1..],
                hostport_absolute_start + closing + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_absolute_start + closing + 1,
            });
        };
        (interior, parsed_port, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&byte| byte == b':');
        let (host, parsed_port) = if let Some(split) = split {
            let host = &hostport[..split];
            let port = parse_port(
                &hostport[split + 1..],
                hostport_absolute_start + split + 1,
            )?;
            (host, Some(port))
        } else {
            (hostport, None)
        };

        if host.is_empty() {
            return Err(ParseError::InvalidHost {
                at: hostport_absolute_start,
            });
        }
        for (offset, byte) in host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_absolute_start + offset,
                });
            }
        }
        (host, parsed_port, false)
    };

    if !ipv6 && host.is_empty() {
        return Err(ParseError::InvalidHost {
            at: hostport_absolute_start,
        });
    }

    let fragment_marker = input[authority_end..].find('#').map(|n| authority_end + n);
    let before_fragment_end = fragment_marker.unwrap_or(input.len());
    let query_marker = input[authority_end..before_fragment_end]
        .find('?')
        .map(|n| authority_end + n);

    let path_end = query_marker.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/"
    } else {
        &input[authority_end..path_end]
    };

    let query = query_marker.map(|marker| &input[marker + 1..before_fragment_end]);
    let fragment = fragment_marker.map(|marker| &input[marker + 1..]);

    validate_component(Component::UserInfo, userinfo, authority_start)?;
    validate_component(
        Component::Path,
        if path == "/" && path_end == authority_end {
            None
        } else {
            Some(path)
        },
        authority_end,
    )?;
    validate_component(
        Component::Query,
        query,
        query_marker.map(|marker| marker + 1).unwrap_or(0),
    )?;
    validate_component(
        Component::Fragment,
        fragment,
        fragment_marker.map(|marker| marker + 1).unwrap_or(0),
    )?;

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo: userinfo.map(str::to_owned),
        host: host.to_ascii_lowercase(),
        port,
        path: path.to_owned(),
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = hex_value(bytes[i + 1]);
            let low = hex_value(bytes[i + 2]);
            match (high, low) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = origins.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

    }
    encoded
}

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }
    if at_positions.first() == Some(&0) {
        return Err(ParseError::InvalidUserInfo { at: authority_start });
    }

    let (userinfo, hostport_start) = if let Some(&at) = at_positions.first() {
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_bytes = hostport.as_bytes();
    let host_absolute_start = authority_start + hostport_start;

    let (host, port, ipv6) = if hostport_bytes.first() == Some(&b'[') {
        let close = match hostport_bytes.iter().position(|&b| b == b']') {
            Some(pos) => pos,
            None => return Err(ParseError::InvalidIpv6 { at: host_absolute_start }),
        };
        let interior = &hostport[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_absolute_start });
        }

        let suffix = &hostport_bytes[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix[0] == b':' {
            Some(parse_port(
                &hostport[close + 2..],
                host_absolute_start + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_absolute_start + close + 1,
            });
        };

        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport_bytes.iter().position(|&b| b == b':');
        let host_end = split.unwrap_or(hostport.len());
        let host_text = &hostport[..host_end];

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost {
                at: host_absolute_start,
            });
        }

        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_absolute_start + offset,
                });
            }
        }

        let port = if let Some(split_at) = split {
            Some(parse_port(
                &hostport[split_at + 1..],
                host_absolute_start + split_at + 1,
            )?)
        } else {
            None
        };

        (host_text.to_ascii_lowercase(), port, false)
    };

    let remaining = &input[authority_end..];
    let fragment_position = remaining.find('#');
    let before_fragment = fragment_position
        .map(|at| &remaining[..at])
        .unwrap_or(remaining);
    let fragment = fragment_position.map(|at| remaining[at + 1..].to_owned());

    let query_position = before_fragment.find('?');
    let path_text = query_position
        .map(|at| &before_fragment[..at])
        .unwrap_or(before_fragment);
    let query = query_position.map(|at| before_fragment[at + 1..].to_owned());
    let path = if path_text.is_empty() {
        "/".to_owned()
    } else {
        path_text.to_owned()
    };

    if let Some(userinfo_text) = userinfo.as_deref() {
        validate_component(
            Component::UserInfo,
            userinfo_text,
            authority_start,
        )?;
    }
    if !path_text.is_empty() {
        validate_component(
            Component::Path,
            path_text,
            authority_end,
        )?;
    }
    if let Some(query_text) = query.as_deref() {
        validate_component(
            Component::Query,
            query_text,
            authority_end + query_position.unwrap() + 1,
        )?;
    }
    if let Some(fragment_text) = fragment.as_deref() {
        validate_component(
            Component::Fragment,
            fragment_text,
            authority_end + fragment_position.unwrap() + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(text: &str, start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }

    for (offset, byte) in text.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + offset });
        }
    }

    if text.len() >= 2 && text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for byte in text.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|number| number.checked_add((byte - b'0') as u32))
        {
            Some(number) => number,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
    }

    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn validate_component(
    component: Component,
    text: &str,
    offset: usize,
) -> Result<(), ParseError> {
    match decode_component(component, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: offset + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: offset + at }),
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(err) => {
            let at = offsets
                .get(err.valid_up_to())
                .copied()
                .unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if is_safe(byte) || component_safe(component, byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
    }
    output
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| authority_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }
    let (userinfo, host_start_rel) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };
    let host_start = authority_start + host_start_rel;

    let (host, port, ipv6) = if authority_bytes.get(host_start_rel) == Some(&b'[') {
        let closing = authority_bytes[host_start_rel + 1..]
            .iter()
            .position(|&b| b == b']')
            .map(|n| host_start_rel + 1 + n);
        let close_rel = match closing {
            Some(pos) => pos,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &authority[host_start_rel + 1..close_rel];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix_rel = close_rel + 1;
        let parsed_port = if suffix_rel == authority.len() {
            None
        } else if authority_bytes[suffix_rel] == b':' {
            Some(parse_port(
                &authority[suffix_rel + 1..],
                authority_start + suffix_rel + 1,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: authority_start + suffix_rel,
            });
        };
        (
            interior.to_ascii_lowercase(),
            parsed_port,
            true,
        )
    } else {
        let colon_rel = authority_bytes[host_start_rel..]
            .iter()
            .position(|&b| b == b':')
            .map(|n| host_start_rel + n);
        let host_end_rel = colon_rel.unwrap_or(authority.len());
        let raw_host = &authority[host_start_rel..host_end_rel];
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, &byte) in raw_host.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        let parsed_port = match colon_rel {
            Some(port_colon) => Some(parse_port(
                &authority[port_colon + 1..],
                authority_start + port_colon + 1,
            )?),
            None => None,
        };
        (raw_host.to_ascii_lowercase(), parsed_port, false)
    };

    let after_authority = &input[authority_end..];
    let fragment_delim = after_authority.find('#');
    let before_fragment = fragment_delim
        .map(|i| &after_authority[..i])
        .unwrap_or(after_authority);
    let fragment = fragment_delim.map(|i| after_authority[i + 1..].to_owned());

    let query_delim = before_fragment.find('?');
    let path_raw = query_delim
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_delim.map(|i| before_fragment[i + 1..].to_owned());
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(info) = userinfo.as_deref() {
        validate_component(info, Component::UserInfo, input, authority_start)?;
    }
    if !path_raw.is_empty() {
        validate_component(path_raw, Component::Path, input, authority_end)?;
    }
    if let Some(q) = query.as_deref() {
        let query_start = authority_end + path_raw.len() + 1;
        validate_component(q, Component::Query, input, query_start)?;
    }
    if let Some(f) = fragment.as_deref() {
        let fragment_start = authority_end + after_authority.find('#').unwrap() + 1;
        validate_component(f, Component::Fragment, input, fragment_start)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn validate_component(
    value: &str,
    component: Component,
    _input: &str,
    base: usize,
) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

fn parse_port(port: &str, at: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    for (offset, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + offset });
        }
    }
    if port.len() >= 2 && port.starts_with('0') {
        return Err(ParseError::InvalidPort { at });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
        {
            Some(n) => n,
            None => return Err(ParseError::InvalidPort { at }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}

fn is_safe(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn component_safe(component: Component, byte: u8) -> bool {
    match component {
        Component::UserInfo => byte == b':',
        Component::Path => byte == b'/',
        Component::Query => matches!(byte, b'/' | b'?'),
        Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
    }
}
