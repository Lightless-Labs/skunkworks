#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobState {
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<Job>,
    states: Vec<JobState>,
    id_to_index: std::collections::HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    has_cycle: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut id_to_index = std::collections::HashMap::new();

        for (index, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if id_to_index.contains_key(&job.id) {
                return Err(Error::Duplicate(index));
            }
            id_to_index.insert(job.id.clone(), index);
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut indices = Vec::with_capacity(job.dependencies.len());
            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                match id_to_index.get(dependency) {
                    Some(&index) => indices.push(index),
                    None => {
                        return Err(Error::MissingDependency(job_index, dependency_index));
                    }
                }
            }
            dependencies.push(indices);
        }

        if has_cycle(&dependencies) {
            return Err(Error::Cycle);
        }

        let states = jobs
            .iter()
            .map(|_| JobState {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            id_to_index,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;

        if self.has_cycle
            && self.states.iter().any(|entry| {
                matches!(entry.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for entry in &mut self.states {
            entry.state = State::Waiting { ready_at: now };
            entry.attempts = 0;
        }

        self.has_cycle = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;

        for index in 0..self.jobs.len() {
            let ready_at = match self.states[index].state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };

            if !self.dependencies_succeeded(index) {
                continue;
            }

            let priority = self.effective_priority(index, ready_at, now);
            match selected {
                None => selected = Some((index, priority)),
                Some((best_index, best_priority))
                    if priority > best_priority
                        || (priority == best_priority && index < best_index) =>
                {
                    selected = Some((index, priority));
                }
                _ => {}
            }
        }

        self.last_time = now;

        if let Some((index, _)) = selected {
            let attempt = self.states[index].attempts + 1;
            self.states[index].attempts = attempt;
            self.states[index].state = State::Running { attempt };
            Ok(Some(Ticket {
                id: self.jobs[index].id.clone(),
                attempt,
            }))
        } else {
            Ok(None)
        }
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        let attempt = match self.states[index].state {
            State::Running { attempt } => attempt,
            _ => return Err(Error::NotRunning),
        };

        if success {
            self.states[index].state = State::Succeeded;
        } else if attempt == self.jobs[index].max_attempts {
            self.states[index].state = State::Failed;
        } else {
            let delay = self.retry_delay(attempt);
            self.states[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.propagate_blocked();
        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        Ok(self.states[index].state.clone())
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn dependencies_succeeded(&self, index: usize) -> bool {
        self.jobs[index]
            .dependencies
            .iter()
            .all(|id| {
                let dependency_index = self.id_to_index[id];
                matches!(self.states[dependency_index].state, State::Succeeded)
            })
    }

    fn effective_priority(&self, index: usize, ready_at: u64, now: u64) -> u8 {
        let base = self.jobs[index].priority;
        if self.aging_ms == 0 {
            return base;
        }
        let age = (now - ready_at) / self.aging_ms;
        (base as u64 + age).min(9) as u8
    }

    fn retry_delay(&self, attempt: u8) -> u64 {
        let mut delay = self.retry_base;
        for _ in 1..attempt {
            if delay >= self.retry_cap.saturating_add(1) / 2 {
                return self.retry_cap;
            }
            delay *= 2;
            if delay >= self.retry_cap {
                return self.retry_cap;
            }
        }
        delay.min(self.retry_cap)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();

            for index in 0..self.jobs.len() {
                if !matches!(self.states[index].state, State::Waiting { .. }) {
                    continue;
                }

                let blocked = self.jobs[index].dependencies.iter().any(|id| {
                    let dependency_index = self.id_to_index[id];
                    matches!(
                        self.states[dependency_index].state,
                        State::Failed | State::Blocked
                    )
                });

                if blocked {
                    newly_blocked.push(index);
                }
            }

            if newly_blocked.is_empty() {
                break;
            }

            for index in newly_blocked {
                self.states[index].state = State::Blocked;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 16
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(index: usize, dependencies: &[Vec<usize>], marks: &mut [u8]) -> bool {
        match marks[index] {
            1 => return true,
            2 => return false,
            _ => {}
        }

        marks[index] = 1;
        for &dependency in &dependencies[index] {
            if visit(dependency, dependencies, marks) {
                return true;
            }
        }
        marks[index] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    for index in 0..dependencies.len() {
        if visit(index, dependencies, &mut marks) {
            return true;
        }
    }
    false
}
