#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        output.push(ALPHABET[(a >> 2) as usize] as char);
        output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
        output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
        output.push(ALPHABET[(c & 0x3f) as usize] as char);
    }

    match chunks.remainder() {
        [a] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push_str("==");
        }
        [a, b] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        _ => {}
    }

    output
}

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut output = Vec::new();
    let mut group = [0u8; 4];
    let mut offsets = [0usize; 4];
    let mut len = 0usize;
    let mut padded = false;

    for (offset, &byte) in input.iter().enumerate() {
        if padded {
            return Err(Error::BadPadding { offset });
        }

        if len == 2 && group[2] == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        if byte == b'=' {
            match len {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    group[len] = byte;
                    offsets[len] = offset;
                    len += 1;
                }
                3 => {
                    group[len] = byte;
                    offsets[len] = offset;
                    len += 1;
                }
                _ => unreachable!(),
            }
        } else {
            let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
            group[len] = value;
            offsets[len] = offset;
            len += 1;
        }

        if len == 4 {
            match (group[2] == b'=', group[3] == b'=') {
                (true, true) => {
                    output.push((group[0] << 2) | (group[1] >> 4));
                    padded = true;
                }
                (false, true) => {
                    output.push((group[0] << 2) | (group[1] >> 4));
                    output.push((group[1] << 4) | (group[2] >> 2));
                    padded = true;
                }
                (false, false) => {
                    output.push((group[0] << 2) | (group[1] >> 4));
                    output.push((group[1] << 4) | (group[2] >> 2));
                    output.push((group[2] << 6) | group[3]);
                }
                (true, false) => unreachable!("pending padding is checked above"),
            }
            len = 0;
        }
    }

    if len == 0 {
        Ok(output)
    } else if len == 3 && group[2] == b'=' {
        Err(Error::BadPadding { offset: offsets[2] })
    } else {
        Err(Error::Truncated { offset: offsets[0] })
    }
}
