use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, name) = split_parent(&canonical);
        let parent = self.resolve(&parent)?;
        let destination = join(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_write(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = split_parent(&resolved);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {
                        self.entries.insert(resolved, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    _ => Err(Error::NotDir(parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => unreachable!(),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        let prefix = if resolved == "/" {
            "/".to_string()
        } else {
            format!("{}/", resolved)
        };
        let mut result = Vec::new();
        for (child, entry) in &self.entries {
            if child == &resolved || !child.starts_with(&prefix) {
                continue;
            }
            let rest = &child[prefix.len()..];
            if rest.is_empty() || rest.contains('/') {
                continue;
            }
            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
                Entry::Link(_) => Kind::Link,
            };
            result.push((rest.to_string(), kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = normalize(from, &self.cwd)?;
        let destination = normalize(to, &self.cwd)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent_raw, source_name) = split_parent(&source);
        let source_parent = self.resolve(&source_parent_raw)?;
        let source_path = join(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && contains(&source_path, &self.cwd) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent_raw, destination_name) = split_parent(&destination);
        let destination_parent = self.resolve(&destination_parent_raw)?;
        let destination_path = join(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && contains(&destination_path, &self.cwd)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && contains_strict(&source_path, &destination_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref existing) = destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination_path) {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        if destination_entry.is_some() {
            self.remove_subtree(&destination_path);
        }
        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| contains(&source_path, path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();
        for (path, _) in &moved {
            self.entries.remove(path);
        }
        for (old_path, entry) in moved {
            let suffix = &old_path[source_path.len()..];
            self.entries.insert(format!("{}{}", destination_path, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent_raw, name) = split_parent(&canonical);
        let parent = self.resolve(&parent_raw)?;
        let target = join(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && contains(&target, &self.cwd) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&target) && !recursive {
            return Err(Error::NotEmpty(target));
        }
        self.remove_subtree(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = normalize(link, &self.cwd)?;
        validate_target(target)?;
        if canonical_link == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent_raw, name) = split_parent(&canonical_link);
        let parent = self.resolve(&parent_raw)?;
        let link_path = join(&parent, &name);
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries
            .insert(link_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let (parent_raw, name) = split_parent(&canonical);
        let parent = self.resolve(&parent_raw)?;
        let target = join(&parent, &name);
        match self.entries.get(&target) {
            Some(Entry::Link(link)) => Ok(link.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(target)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical)?;
        let mut rows = Vec::new();
        for (entry_path, entry) in &self.entries {
            if contains(&resolved, entry_path) {
                let size = match entry {
                    Entry::Dir => 0,
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                };
                rows.push((entry_path.clone(), size));
            }
        }
        let total = rows.iter().map(|(_, size)| *size).sum();
        if let Some(row) = rows.iter_mut().find(|(entry_path, _)| entry_path == &resolved) {
            row.1 = total;
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    fn resolve(&self, path: &str) -> Result<String, Error> {
        self.resolve_internal(path, false)
    }

    fn resolve_write(&self, path: &str) -> Result<String, Error> {
        self.resolve_internal(path, true)
    }

    fn resolve_internal(&self, path: &str, create_final: bool) -> Result<String, Error> {
        let mut current = path.to_string();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            let components = components(&current);
            let mut physical = "/".to_string();
            let mut restarted = false;

            if components.is_empty() {
                return Ok("/".to_string());
            }

            for (index, component) in components.iter().enumerate() {
                physical = join(&physical, component);
                let entry = self.entries.get(&physical);
                let is_final = index + 1 == components.len();

                match entry {
                    Some(Entry::Link(target)) => {
                        if create_final && is_final {
                            // Write follows the final link; dangling final targets may be created.
                        }
                        if !visited.insert(physical.clone()) {
                            return Err(Error::Loop(physical));
                        }
                        if expansions == 8 {
                            return Err(Error::Depth(physical));
                        }
                        expansions += 1;

                        let link_parent = split_parent(&physical).0;
                        let target_base = if target.starts_with('/') {
                            "/".to_string()
                        } else {
                            link_parent
                        };
                        let target_path = normalize(target, &target_base)?;
                        let suffix = components[index + 1..].to_vec();
                        current = append_components(&target_path, &suffix);
                        restarted = true;
                        break;
                    }
                    Some(Entry::Dir) => {
                        if !is_final {
                            continue;
                        }
                        return Ok(physical);
                    }
                    Some(Entry::File(_)) => {
                        if !is_final {
                            return Err(Error::NotDir(physical));
                        }
                        return Ok(physical);
                    }
                    None => {
                        if create_final && is_final {
                            return Ok(physical);
                        }
                        return Err(Error::Missing(physical));
                    }
                }
            }

            if !restarted {
                return Ok(physical);
            }
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && contains(path, candidate))
    }

    fn remove_subtree(&mut self, path: &str) {
        let doomed: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| contains(path, candidate))
            .cloned()
            .collect();
        for candidate in doomed {
            self.entries.remove(&candidate);
        }
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(index) = path.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(Error::InvalidPath(index));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize(path: &str, base: &str) -> Result<String, Error> {
    validate_path(path)?;
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(base)
    };
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_string()),
        }
    }
    Ok(path_from_components(&parts))
}

fn validate_target(target: &str) -> Result<(), Error> {

        return Err(Error::InvalidTarget(0));
    }
    if let Some(index) = target.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(Error::InvalidTarget(index));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect()
}

fn path_from_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn append_components(base: &str, suffix: &[String]) -> String {
    let mut parts = components(base);
    parts.extend(suffix.iter().cloned());
    path_from_components(&parts)
}

fn split_parent(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_string(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_string(), path[1..].to_string()),
        Some(index) => (path[..index].to_string(), path[index + 1..].to_string()),
        None => ("/".to_string(), path.to_string()),
    }
}

fn join(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{}", name)
    } else {
        format!("{}/{}", parent, name)
    }
}

fn contains(parent: &str, candidate: &str) -> bool {
    if parent == "/" {
        return candidate.starts_with('/');
    }
    candidate == parent
        || candidate
            .strip_prefix(parent)
            .is_some_and(|suffix| suffix.starts_with('/'))
}

fn contains_strict(parent: &str, candidate: &str) -> bool {
    candidate != parent && contains(parent, candidate)
}

fn validate(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));

fn normalize_components<I>(components: I) -> String
where
    I: IntoIterator<Item = String>,
{
    let mut stack: Vec<String> = Vec::new();
    for component in components {
        match component.as_str() {
            "" | "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push(component),
        }
    }
    if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn normalize(path: &str, cwd: &str) -> String {
    let mut parts: Vec<String> = if path.starts_with('/') {
        Vec::new()
    } else {
        cwd.split('/')
            .filter(|s| !s.is_empty())
            .map(str::to_string)
            .collect()
    };
    parts.extend(path.split('/').map(str::to_string));
    normalize_components(parts)
}

fn parent(path: &str) -> String {
    if path == "/" {
        return "/".to_string();
    }
    match path.rfind('/') {
        Some(0) | None => "/".to_string(),
        Some(i) => path[..i].to_string(),
    }
}

fn components(path: &str) -> Vec<String> {

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let initial = normalize(path, &self.cwd);
        self.resolve_canonical(&initial, follow_final, create_final)
    }

    fn resolve_canonical(
        &self,
        initial: &str,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        let mut current = initial.to_string();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            let parts = components(&current);
            let mut resolved: Vec<String> = Vec::new();
            let mut restart = None;

            for (index, component) in parts.iter().enumerate() {
                let candidate_parts = {
                    let mut v = resolved.clone();
                    v.push(component.clone());
                    v
                };
                let candidate = join_components(&candidate_parts);
                let is_final = index + 1 == parts.len();

                let Some(entry) = self.entries.get(&candidate) else {
                    if is_final && create_final {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                };

                let should_follow = matches!(entry, Entry::Link(_)) && (!is_final || follow_final);
                if should_follow {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;
                    if expansions > 8 {
                        return Err(Error::Depth(candidate));
                    }

                    let Entry::Link(target) = entry else {
                        unreachable!()
                    };
                    let target_base = if target.starts_with('/') {
                        "/".to_string()
                    } else {
                        parent(&candidate)
                    };
                    let target_path = normalize(target, &target_base);
                    let mut expanded = target_path;
                    for rest in parts.iter().skip(index + 1) {
                        if expanded == "/" {
                            expanded.push_str(rest);
                        } else {
                            expanded.push('/');
                            expanded.push_str(rest);
                        }
                    }
                    current = normalize(&expanded, "/");
                    restart = Some(());
                    break;
                }

                if !is_final && !matches!(entry, Entry::Dir) {
                    return Err(Error::NotDir(candidate));
                }
                resolved.push(component.clone());
            }

            if restart.is_some() {
                continue;
            }
            return Ok(join_components(&resolved));
        }
    }

    fn resolve_parent(&self, canonical: &str) -> Result<(String, String), Error> {
        if canonical == "/" {
            return Ok(("/".to_string(), "".to_string()));
        }
        let p = parent(canonical);
        let resolved_parent = self.resolve_canonical(&p, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, basename(canonical).to_string())),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        let resolved = self.resolve_canonical(&canonical, true, true)?;
        let p = parent(&resolved);
        let resolved_parent = self.resolve_canonical(&p, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut result = Vec::new();
                for (key, entry) in &self.entries {
                    if key == &resolved {
                        continue;
                    }
                    if parent(key) == resolved {
                        result.push((basename(key).to_string(), entry.kind()));
                    }
                }
                result.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(result)
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        validate(from)?;
        validate(to)?;
        let source = normalize(from, &self.cwd);
        let destination = normalize(to, &self.cwd);
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = if source_parent == "/" {
            format!("/{}", source_name)
        } else {
            format!("{}/{}", source_parent, source_name)
        };
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = if destination_parent == "/" {
            format!("/{}", destination_name)
        } else {
            format!("{}/{}", destination_parent, destination_name)
        };

        if let Some(Entry::Dir) = self.entries.get(&destination_path) {
            if is_within(&self.cwd, &destination_path) {
                return Err(Error::Busy(destination_path));
            }
        }

        if source_path == destination_path {
            return Ok(());
        }
        if matches!(source_entry, Entry::Dir) && is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(dest_entry) = &destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.entries.keys().any(|k| k != &destination_path && is_within(k, &destination_path)) {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        if destination_entry.is_some() {
            self.entries.retain(|k, _| !is_within(k, &destination_path));
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(k, _)| is_within(k, &source_path))
            .map(|(k, v)| (k.clone(), v.clone()))
            .collect();

        self.entries.retain(|k, _| !is_within(k, &source_path));
        for (old, entry) in moving {
            let suffix = if old == source_path {
                String::new()
            } else {
                old[source_path.len()..].to_string()
            };
            self.entries.insert(format!("{}{}", destination_path, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        let entry = self
            .entries
            .get(&final_path)
            .cloned()
            .ok_or_else(|| Error::Missing(final_path.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &final_path) {
            return Err(Error::Busy(final_path));
        }
        if matches!(entry, Entry::Dir)
            && !recursive
            && self.entries.keys().any(|k| k != &final_path && is_within(k, &final_path))
        {
            return Err(Error::NotEmpty(final_path));
        }
        self.entries.retain(|k, _| !is_within(k, &final_path));
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        validate(link)?;
        let canonical = normalize(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut rows = Vec::new();
        for (key, entry) in &self.entries {
            if is_within(key, &resolved) {
                let size = match entry {
                    Entry::Dir => 0,
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                };
                rows.push((key.clone(), size));
            }
        }

        for i in (0..rows.len()).rev() {
            if matches!(self.entries.get(&rows[i].0), Some(Entry::Dir)) {
                let dir = rows[i].0.clone();
                let total = rows
                    .iter()
                    .filter(|(path, _)| path != &dir && is_within(path, &dir))
                    .map(|(_, size)| *size)
                    .sum();
                rows[i].1 = total;
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }
        Self::canonical(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        let i = path.rfind('/').unwrap();
        if i == 0 {
            "/".to_string()
        } else {
            path[..i].to_string()
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            String::new()
        } else {
            path.rsplit('/').next().unwrap().to_string()
        }
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn contains(parent: &str, child: &str) -> bool {
        parent == child
            || (parent == "/" && child.starts_with('/'))
            || (child.starts_with(parent)
                && child.as_bytes().get(parent.len()) == Some(&b'/'))
    }

    fn resolve(
        &self,
        path: &str,
        cwd: &str,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        let initial = Self::normalize(path, cwd);
        self.resolve_canonical(initial, follow_final, create_final)
    }

    fn resolve_canonical(
        &self,
        mut path: String,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            let parts = Self::components(&path);
            let mut prefix = "/".to_string();
            let mut restarted = false;

            for (i, component) in parts.iter().enumerate() {
                let next = Self::join(&prefix, component);
                let is_final = i + 1 == parts.len();
                let entry = self.entries.get(&next);

                if entry.is_none() {
                    if create_final && is_final {
                        return Ok(next);
                    }
                    return Err(Error::Missing(next));
                }

                if !is_final {
                    if let Some(Entry::Link(target)) = entry {
                        if !visited.insert(next.clone()) {
                            return Err(Error::Loop(next));
                        }
                        if expansions == 8 {
                            return Err(Error::Depth(next));
                        }
                        expansions += 1;

                        let target_path = Self::normalize(target, &prefix);
                        let suffix = &parts[i + 1..];
                        let mut expanded = Self::components(&target_path);
                        expanded.extend(suffix.iter().cloned());
                        path = Self::canonical(&expanded);
                        restarted = true;
                        break;
                    }
                    if !matches!(entry, Some(Entry::Dir)) {
                        return Err(Error::NotDir(next));
                    }
                } else if follow_final {
                    if let Some(Entry::Link(target)) = entry {
                        if !visited.insert(next.clone()) {
                            return Err(Error::Loop(next));
                        }
                        if expansions == 8 {
                            return Err(Error::Depth(next));
                        }
                        expansions += 1;

                        path = Self::normalize(target, &prefix);
                        restarted = true;
                        break;
                    }
                }

                prefix = next;
            }

            if restarted {
                continue;
            }
            return Ok(path);
        }
    }

    fn entry_kind(entry: &Entry) -> Kind {
        match entry {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                candidate.as_str() != path
                    && Self::parent(candidate) == path
                    && candidate.as_str() != "/"
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let parent = Self::parent(&canonical);
        let resolved_parent = self.resolve_canonical(parent.clone(), true, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        let destination = Self::join(&resolved_parent, &Self::basename(&canonical));
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let resolved = self.resolve(path, &self.cwd, true, true)?;
        match self.entries.get_mut(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let parent = Self::parent(&resolved);
                let resolved_parent = self.resolve_canonical(parent.clone(), true, false)?;
                if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
                    return Err(Error::NotDir(resolved_parent));
                }
                let actual = Self::join(&resolved_parent, &Self::basename(&resolved));
                self.entries
                    .insert(actual, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let resolved = self.resolve(path, &self.cwd, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(contents)) => Ok(contents.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let resolved = self.resolve(path, &self.cwd, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        let mut result: Vec<_> = self
            .children(&resolved)
            .into_iter()
            .map(|child| {
                (
                    Self::basename(&child),
                    Self::entry_kind(self.entries.get(&child).unwrap()),
                )
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let resolved = self.resolve(path, &self.cwd, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        Self::validate(to)?;
        let source = Self::normalize(from, &self.cwd);
        let destination = Self::normalize(to, &self.cwd);
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let source_parent = Self::parent(&source);
        let source_parent_resolved =
            self.resolve_canonical(source_parent, true, false)?;
        if !matches!(
            self.entries.get(&source_parent_resolved),
            Some(Entry::Dir)
        ) {
            return Err(Error::NotDir(source_parent_resolved));
        }
        let source_path = Self::join(
            &source_parent_resolved,
            &Self::basename(&source),
        );
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir)
            && Self::contains(&source_path, &self.cwd)
        {
            return Err(Error::Busy(source_path));
        }

        let destination_parent = Self::parent(&destination);
        let destination_parent_resolved =
            self.resolve_canonical(destination_parent, true, false)?;
        if !matches!(
            self.entries.get(&destination_parent_resolved),
            Some(Entry::Dir)
        ) {
            return Err(Error::NotDir(destination_parent_resolved));
        }
        let destination_path = Self::join(
            &destination_parent_resolved,
            &Self::basename(&destination),
        );

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::contains(&destination_path, &self.cwd)
        {
            return Err(Error::Busy(destination_path));
        }
        if source_path == destination_path {
            return Ok(());
        }
        if matches!(source_entry, Entry::Dir)
            && Self::contains(&source_path, &destination_path)
        {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }
        if let Some(ref target_entry) = destination_entry {
            match (&source_entry, target_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| Self::contains(&source_path, p))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if let Some(target) = destination_entry {
            if matches!(target, Entry::Dir) {
                self.entries.remove(&destination_path);
            } else {
                self.entries.remove(&destination_path);
            }
        }

        for (old, _) in &moving {
            self.entries.remove(old);
        }
        for (old, entry) in moving {
            let suffix = &old[source_path.len()..];
            self.entries
                .insert(format!("{destination_path}{suffix}"), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }
        let parent = Self::parent(&canonical);
        let resolved_parent = self.resolve_canonical(parent, true, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        let target = Self::join(&resolved_parent, &Self::basename(&canonical));
        let entry = self
            .entries
            .get(&target)
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains(&target, &self.cwd) {
            return Err(Error::Busy(target));
        }
        let descendants = self.children(&target);
        if matches!(entry, Entry::Dir) && !recursive && !descendants.is_empty() {
            return Err(Error::NotEmpty(target));
        }

        let remove_paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::contains(&target, p))
            .cloned()
            .collect();
        for p in remove_paths {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let link_path = Self::normalize(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let parent = Self::parent(&link_path);
        let resolved_parent = self.resolve_canonical(parent, true, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        let actual = Self::join(&resolved_parent, &Self::basename(&link_path));
        if self.entries.contains_key(&actual) {
            return Err(Error::Exists(actual));
        }
        self.entries
            .insert(actual, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let parent = Self::parent(&canonical);
        let resolved_parent = self.resolve_canonical(parent, true, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        let final_path = Self::join(&resolved_parent, &Self::basename(&canonical));
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let resolved = self.resolve(path, &self.cwd, true, false)?;
        let mut rows = Vec::new();
        let mut directory_totals = BTreeMap::<String, u64>::new();

        for (entry_path, entry) in self.entries.iter() {
            if !Self::contains(&resolved, entry_path) {
                continue;
            }
            let size = match entry {
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            rows.push((entry_path.clone(), size));
            if size != 0 {
                let mut current = Self::parent(entry_path);
                while Self::contains(&resolved, &current) {
                    *directory_totals.entry(current.clone()).or_default() += size;
                    if current == resolved {
                        break;
                    }
                    current = Self::parent(&current);
                }
            }
        }

        for (entry_path, entry) in self.entries.iter() {
            if matches!(entry, Entry::Dir) && Self::contains(&resolved, entry_path) {
                let total = directory_totals.get(entry_path).copied().unwrap_or(0);
                if let Some(row) = rows.iter_mut().find(|row| row.0 == *entry_path) {
                    row.1 = total;
                }
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate_path(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn normalize_from(path: &str, base: &str) -> String {
        let mut parts: Vec<&str> = if path.starts_with('/') {
            Vec::new()
        } else {
            base.split('/').filter(|s| !s.is_empty()).collect()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }

        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate_path(path)?;
        Ok(Self::normalize_from(path, &self.cwd))
    }

    fn parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        let index = path.rfind('/').unwrap();
        let parent = if index == 0 {
            "/".to_string()
        } else {
            path[..index].to_string()
        };
        (parent, path[index + 1..].to_string())
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || path
                .strip_prefix(dir)
                .map(|suffix| suffix.starts_with('/'))
                .unwrap_or(false)
    }

    fn kind(entry: &Entry) -> Kind {
        match entry {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }

    fn resolve_full(&self, path: &str, allow_missing_final: bool) -> Result<String, Error> {
        let mut pending: Vec<String> = path
            .split('/')
            .filter(|s| !s.is_empty())
            .map(str::to_string)
            .collect();
        let mut physical = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            let candidate = if physical.is_empty() {
                format!("/{}", component)
            } else {
                format!("/{}/{}", physical.join("/"), component)
            };

            let entry = self.entries.get(&candidate);
            let Some(entry) = entry else {
                if allow_missing_final && pending.is_empty() {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            };

            if let Entry::Link(target) = entry {
                if !visited.insert(candidate.clone()) {
                    return Err(Error::Loop(candidate));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(candidate));
                }
                expansions += 1;

                let link_parent = if physical.is_empty() {
                    "/".to_string()
                } else {
                    format!("/{}", physical.join("/"))
                };
                let mut expanded = Self::normalize_from(target, &link_parent);
                if !pending.is_empty() {
                    if expanded != "/" {
                        expanded.push('/');
                    }
                    expanded.push_str(&pending.join("/"));
                    expanded = Self::normalize_from(&expanded, "/");
                }
                pending = expanded
                    .split('/')
                    .filter(|s| !s.is_empty())
                    .map(str::to_string)
                    .collect();
                physical.clear();
                continue;
            }

            if !pending.is_empty() && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
            physical.push(component);
        }

        if physical.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", physical.join("/")))
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(path);
        let resolved = self.resolve_full(&parent, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Ok((resolved, name)),
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if path == "/" {
                    candidate
                        .strip_prefix('/')
                        .map(|rest| !rest.is_empty() && !rest.contains('/'))
                        .unwrap_or(false)
                } else {
                    candidate
                        .strip_prefix(path)
                        .and_then(|rest| rest.strip_prefix('/'))
                        .map(|rest| !rest.is_empty() && !rest.contains('/'))
                        .unwrap_or(false)
                }
            })
            .cloned()
            .collect()
    }

    fn subtree_size(&self, path: &str) -> u64 {
        match self.entries.get(path) {
            Some(Entry::File(data)) => data.len() as u64,
            Some(Entry::Link(target)) => target.len() as u64,
            Some(Entry::Dir) => self
                .children(path)
                .iter()
                .map(|child| self.subtree_size(child))
                .sum(),
            None => 0,
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let target = self.resolve_full(&canonical, true)?;
        if let Some(Entry::Dir) = self.entries.get(&target) {
            return Err(Error::IsDir(target));
        }
        if self.entries.contains_key(&target) {
            match self.entries.get_mut(&target) {
                Some(Entry::File(existing)) => existing.clear(),
                _ => {}
            }
            match self.entries.get_mut(&target) {
                Some(Entry::File(existing)) => existing.extend_from_slice(data),
                Some(_) => unreachable!(),
                None => unreachable!(),
            }
            return Ok(());
        }

        let (parent, _) = Self::parent(&target);
        let resolved_parent = self.resolve_full(&parent, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        self.entries
            .insert(target, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }

        let mut result = Vec::new();
        for child in self.children(&resolved) {
            let name = child.rsplit('/').next().unwrap().to_string();
            let kind = Self::kind(self.entries.get(&child).unwrap());
            result.push((name, kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&source)?;
        let src = if src_parent == "/" {
            format!("/{}", src_name)
        } else {
            format!("{}/{}", src_parent, src_name)
        };
        let source_entry = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&destination)?;
        let dst = if dst_parent == "/" {
            format!("/{}", dst_name)
        } else {
            format!("{}/{}", dst_parent, dst_name)
        };

        if matches!(self.entries.get(&dst), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let destination_entry = self.entries.get(&dst).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }

        if let Some(dest_entry) = &destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&dst).is_empty() {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &src))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(dest_entry) = destination_entry {
            if matches!(dest_entry, Entry::Dir) {
                let descendants = self
                    .entries
                    .keys()
                    .filter(|path| Self::is_within(path, &dst))
                    .cloned()
                    .collect::<Vec<_>>();
                for path in descendants {
                    self.entries.remove(&path);
                }
            } else {
                self.entries.remove(&dst);
            }
        }

        for (old, _) in &moved {
            self.entries.remove(old);
        }
        for (old, entry) in moved {
            let suffix = &old[src.len()..];
            self.entries.insert(format!("{}{}", dst, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        let entry = self
            .entries
            .get(&target)
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        let paths = if matches!(entry, Entry::Dir) && recursive {
            self.entries
                .keys()
                .filter(|candidate| Self::is_within(candidate, &target))
                .cloned()
                .collect::<Vec<_>>()
        } else {
            vec![target]
        };
        for p in paths {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = self.normalize(link)?;
        Self::validate_path(target).map_err(|error| match error {
            Error::InvalidPath(i) => Error::InvalidTarget(i),
            _ => unreachable!(),
        })?;

        let (parent, name) = self.resolve_parent(&canonical_link)?;
        let link_path = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries
            .insert(link_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = Self::parent(&canonical);
        let resolved_parent = self.resolve_full(&parent, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        let final_path = if resolved_parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", resolved_parent, name)
        };
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        let mut paths = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, &resolved))
            .cloned()
            .collect::<Vec<_>>();
        paths.sort();
        Ok(paths
            .into_iter()
            .map(|p| {
                let size = match self.entries.get(&p).unwrap() {
                    Entry::Dir => self.subtree_size(&p),
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                };
                (p, size)
            })
            .collect())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
