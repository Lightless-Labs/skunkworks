```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize_components<I>(components: I) -> String
where
    I: IntoIterator<Item = String>,
{
    let mut stack: Vec<String> = Vec::new();
    for component in components {
        match component.as_str() {
            "" | "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push(component),
        }
    }
    if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn normalize(path: &str, cwd: &str) -> String {
    let mut parts: Vec<String> = if path.starts_with('/') {
        Vec::new()
    } else {
        cwd.split('/')
            .filter(|s| !s.is_empty())
            .map(str::to_string)
            .collect()
    };
    parts.extend(path.split('/').map(str::to_string));
    normalize_components(parts)
}

fn parent(path: &str) -> String {
    if path == "/" {
        return "/".to_string();
    }
    match path.rfind('/') {
        Some(0) | None => "/".to_string(),
        Some(i) => path[..i].to_string(),
    }
}

fn basename(path: &str) -> &str {
    if path == "/" {
        ""
    } else {
        path.rsplit('/').next().unwrap_or("")
    }
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|s| !s.is_empty())
        .map(str::to_string)
        .collect()
}

fn join_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn is_within(path: &str, dir: &str) -> bool {
    dir == "/" || path == dir || path.starts_with(&format!("{}/", dir))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let initial = normalize(path, &self.cwd);
        self.resolve_canonical(&initial, follow_final, create_final)
    }

    fn resolve_canonical(
        &self,
        initial: &str,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        let mut current = initial.to_string();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            let parts = components(&current);
            let mut resolved: Vec<String> = Vec::new();
            let mut restart = None;

            for (index, component) in parts.iter().enumerate() {
                let candidate_parts = {
                    let mut v = resolved.clone();
                    v.push(component.clone());
                    v
                };
                let candidate = join_components(&candidate_parts);
                let is_final = index + 1 == parts.len();

                let Some(entry) = self.entries.get(&candidate) else {
                    if is_final && create_final {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                };

                let should_follow = matches!(entry, Entry::Link(_)) && (!is_final || follow_final);
                if should_follow {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;
                    if expansions > 8 {
                        return Err(Error::Depth(candidate));
                    }

                    let Entry::Link(target) = entry else {
                        unreachable!()
                    };
                    let target_base = if target.starts_with('/') {
                        "/".to_string()
                    } else {
                        parent(&candidate)
                    };
                    let target_path = normalize(target, &target_base);
                    let mut expanded = target_path;
                    for rest in parts.iter().skip(index + 1) {
                        if expanded == "/" {
                            expanded.push_str(rest);
                        } else {
                            expanded.push('/');
                            expanded.push_str(rest);
                        }
                    }
                    current = normalize(&expanded, "/");
                    restart = Some(());
                    break;
                }

                if !is_final && !matches!(entry, Entry::Dir) {
                    return Err(Error::NotDir(candidate));
                }
                resolved.push(component.clone());
            }

            if restart.is_some() {
                continue;
            }
            return Ok(join_components(&resolved));
        }
    }

    fn resolve_parent(&self, canonical: &str) -> Result<(String, String), Error> {
        if canonical == "/" {
            return Ok(("/".to_string(), "".to_string()));
        }
        let p = parent(canonical);
        let resolved_parent = self.resolve_canonical(&p, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, basename(canonical).to_string())),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        let resolved = self.resolve_canonical(&canonical, true, true)?;
        let p = parent(&resolved);
        let resolved_parent = self.resolve_canonical(&p, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut result = Vec::new();
                for (key, entry) in &self.entries {
                    if key == &resolved {
                        continue;
                    }
                    if parent(key) == resolved {
                        result.push((basename(key).to_string(), entry.kind()));
                    }
                }
                result.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(result)
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        validate(from)?;
        validate(to)?;
        let source = normalize(from, &self.cwd);
        let destination = normalize(to, &self.cwd);
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = if source_parent == "/" {
            format!("/{}", source_name)
        } else {
            format!("{}/{}", source_parent, source_name)
        };
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = if destination_parent == "/" {
            format!("/{}", destination_name)
        } else {
            format!("{}/{}", destination_parent, destination_name)
        };

        if let Some(Entry::Dir) = self.entries.get(&destination_path) {
            if is_within(&self.cwd, &destination_path) {
                return Err(Error::Busy(destination_path));
            }
        }

        if source_path == destination_path {
            return Ok(());
        }
        if matches!(source_entry, Entry::Dir) && is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(dest_entry) = &destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.entries.keys().any(|k| k != &destination_path && is_within(k, &destination_path)) {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        if destination_entry.is_some() {
            self.entries.retain(|k, _| !is_within(k, &destination_path));
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(k, _)| is_within(k, &source_path))
            .map(|(k, v)| (k.clone(), v.clone()))
            .collect();

        self.entries.retain(|k, _| !is_within(k, &source_path));
        for (old, entry) in moving {
            let suffix = if old == source_path {
                String::new()
            } else {
                old[source_path.len()..].to_string()
            };
            self.entries.insert(format!("{}{}", destination_path, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        let entry = self
            .entries
            .get(&final_path)
            .cloned()
            .ok_or_else(|| Error::Missing(final_path.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &final_path) {
            return Err(Error::Busy(final_path));
        }
        if matches!(entry, Entry::Dir)
            && !recursive
            && self.entries.keys().any(|k| k != &final_path && is_within(k, &final_path))
        {
            return Err(Error::NotEmpty(final_path));
        }
        self.entries.retain(|k, _| !is_within(k, &final_path));
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        validate(link)?;
        let canonical = normalize(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        validate(path)?;
        let canonical = normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let final_path = if p == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", p, name)
        };
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        validate(path)?;
        let resolved = self.resolve(path, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut rows = Vec::new();
        for (key, entry) in &self.entries {
            if is_within(key, &resolved) {
                let size = match entry {
                    Entry::Dir => 0,
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                };
                rows.push((key.clone(), size));
            }
        }

        for i in (0..rows.len()).rev() {
            if matches!(self.entries.get(&rows[i].0), Some(Entry::Dir)) {
                let dir = rows[i].0.clone();
                let total = rows
                    .iter()
                    .filter(|(path, _)| path != &dir && is_within(path, &dir))
                    .map(|(_, size)| *size)
                    .sum();
                rows[i].1 = total;
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}
```
