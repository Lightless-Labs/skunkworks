```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate_path(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn normalize_from(path: &str, base: &str) -> String {
        let mut parts: Vec<&str> = if path.starts_with('/') {
            Vec::new()
        } else {
            base.split('/').filter(|s| !s.is_empty()).collect()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }

        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate_path(path)?;
        Ok(Self::normalize_from(path, &self.cwd))
    }

    fn parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        let index = path.rfind('/').unwrap();
        let parent = if index == 0 {
            "/".to_string()
        } else {
            path[..index].to_string()
        };
        (parent, path[index + 1..].to_string())
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || path
                .strip_prefix(dir)
                .map(|suffix| suffix.starts_with('/'))
                .unwrap_or(false)
    }

    fn kind(entry: &Entry) -> Kind {
        match entry {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }

    fn resolve_full(&self, path: &str, allow_missing_final: bool) -> Result<String, Error> {
        let mut pending: Vec<String> = path
            .split('/')
            .filter(|s| !s.is_empty())
            .map(str::to_string)
            .collect();
        let mut physical = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            let candidate = if physical.is_empty() {
                format!("/{}", component)
            } else {
                format!("/{}/{}", physical.join("/"), component)
            };

            let entry = self.entries.get(&candidate);
            let Some(entry) = entry else {
                if allow_missing_final && pending.is_empty() {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            };

            if let Entry::Link(target) = entry {
                if !visited.insert(candidate.clone()) {
                    return Err(Error::Loop(candidate));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(candidate));
                }
                expansions += 1;

                let link_parent = if physical.is_empty() {
                    "/".to_string()
                } else {
                    format!("/{}", physical.join("/"))
                };
                let mut expanded = Self::normalize_from(target, &link_parent);
                if !pending.is_empty() {
                    if expanded != "/" {
                        expanded.push('/');
                    }
                    expanded.push_str(&pending.join("/"));
                    expanded = Self::normalize_from(&expanded, "/");
                }
                pending = expanded
                    .split('/')
                    .filter(|s| !s.is_empty())
                    .map(str::to_string)
                    .collect();
                physical.clear();
                continue;
            }

            if !pending.is_empty() && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
            physical.push(component);
        }

        if physical.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", physical.join("/")))
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(path);
        let resolved = self.resolve_full(&parent, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Ok((resolved, name)),
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if path == "/" {
                    candidate
                        .strip_prefix('/')
                        .map(|rest| !rest.is_empty() && !rest.contains('/'))
                        .unwrap_or(false)
                } else {
                    candidate
                        .strip_prefix(path)
                        .and_then(|rest| rest.strip_prefix('/'))
                        .map(|rest| !rest.is_empty() && !rest.contains('/'))
                        .unwrap_or(false)
                }
            })
            .cloned()
            .collect()
    }

    fn subtree_size(&self, path: &str) -> u64 {
        match self.entries.get(path) {
            Some(Entry::File(data)) => data.len() as u64,
            Some(Entry::Link(target)) => target.len() as u64,
            Some(Entry::Dir) => self
                .children(path)
                .iter()
                .map(|child| self.subtree_size(child))
                .sum(),
            None => 0,
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let target = self.resolve_full(&canonical, true)?;
        if let Some(Entry::Dir) = self.entries.get(&target) {
            return Err(Error::IsDir(target));
        }
        if self.entries.contains_key(&target) {
            match self.entries.get_mut(&target) {
                Some(Entry::File(existing)) => existing.clear(),
                _ => {}
            }
            match self.entries.get_mut(&target) {
                Some(Entry::File(existing)) => existing.extend_from_slice(data),
                Some(_) => unreachable!(),
                None => unreachable!(),
            }
            return Ok(());
        }

        let (parent, _) = Self::parent(&target);
        let resolved_parent = self.resolve_full(&parent, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        self.entries
            .insert(target, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }

        let mut result = Vec::new();
        for child in self.children(&resolved) {
            let name = child.rsplit('/').next().unwrap().to_string();
            let kind = Self::kind(self.entries.get(&child).unwrap());
            result.push((name, kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&source)?;
        let src = if src_parent == "/" {
            format!("/{}", src_name)
        } else {
            format!("{}/{}", src_parent, src_name)
        };
        let source_entry = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&destination)?;
        let dst = if dst_parent == "/" {
            format!("/{}", dst_name)
        } else {
            format!("{}/{}", dst_parent, dst_name)
        };

        if matches!(self.entries.get(&dst), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let destination_entry = self.entries.get(&dst).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }

        if let Some(dest_entry) = &destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&dst).is_empty() {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &src))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(dest_entry) = destination_entry {
            if matches!(dest_entry, Entry::Dir) {
                let descendants = self
                    .entries
                    .keys()
                    .filter(|path| Self::is_within(path, &dst))
                    .cloned()
                    .collect::<Vec<_>>();
                for path in descendants {
                    self.entries.remove(&path);
                }
            } else {
                self.entries.remove(&dst);
            }
        }

        for (old, _) in &moved {
            self.entries.remove(old);
        }
        for (old, entry) in moved {
            let suffix = &old[src.len()..];
            self.entries.insert(format!("{}{}", dst, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        let entry = self
            .entries
            .get(&target)
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        let paths = if matches!(entry, Entry::Dir) && recursive {
            self.entries
                .keys()
                .filter(|candidate| Self::is_within(candidate, &target))
                .cloned()
                .collect::<Vec<_>>()
        } else {
            vec![target]
        };
        for p in paths {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = self.normalize(link)?;
        Self::validate_path(target).map_err(|error| match error {
            Error::InvalidPath(i) => Error::InvalidTarget(i),
            _ => unreachable!(),
        })?;

        let (parent, name) = self.resolve_parent(&canonical_link)?;
        let link_path = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries
            .insert(link_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = Self::parent(&canonical);
        let resolved_parent = self.resolve_full(&parent, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        let final_path = if resolved_parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", resolved_parent, name)
        };
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical, false)?;
        let mut paths = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, &resolved))
            .cloned()
            .collect::<Vec<_>>();
        paths.sort();
        Ok(paths
            .into_iter()
            .map(|p| {
                let size = match self.entries.get(&p).unwrap() {
                    Entry::Dir => self.subtree_size(&p),
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                };
                (p, size)
            })
            .collect())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
