use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }

        if pos == start {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        let digits = &s[start..pos];
        if digits.len() > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        core[field] = digits
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let plus = bytes[pos..].iter().position(|&b| b == b'+').map(|i| pos + i);
        let pre_end = plus.unwrap_or(bytes.len());

        let mut id_start = pre_start;
        loop {
            let id_end = bytes[id_start..pre_end]
                .iter()
                .position(|&b| b == b'.')
                .map(|i| id_start + i)
                .unwrap_or(pre_end);

            if id_start == id_end {
                return Err(VersionError::EmptyIdentifier { at: id_start });
            }
            for i in id_start..id_end {
                if !is_identifier_char(bytes[i]) {
                    return Err(VersionError::InvalidChar { at: i });
                }
            }

            let ident = &s[id_start..id_end];
            if ident.bytes().all(is_digit) && ident.len() > 1 && ident.as_bytes()[0] == b'0' {
                return Err(VersionError::LeadingZero { at: id_start });
            }
            pre.push(ident.to_string());

            if id_end == pre_end {
                break;
            }
            id_start = id_end + 1;
        }

        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        let build_start = pos;
        let mut id_start = build_start;

        loop {
            let id_end = bytes[id_start..]
                .iter()
                .position(|&b| b == b'.')
                .map(|i| id_start + i)
                .unwrap_or(bytes.len());

            if id_start == id_end {
                return Err(VersionError::EmptyIdentifier { at: id_start });
            }
            for i in id_start..id_end {
                if !is_identifier_char(bytes[i]) {
                    return Err(VersionError::InvalidChar { at: i });
                }
            }
            build.push(s[id_start..id_end].to_string());

            if id_end == bytes.len() {
                break;
            }
            id_start = id_end + 1;
        }
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let an = is_numeric_identifier(a);
    let bn = is_numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let order = compare_identifiers(x, y);
                    if order != Ordering::Equal {
                        return order;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

impl Comparator {
    fn matches(&self, v: &Version) -> bool {
        let ord = compare_versions(v, &self.version);
        match self.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    }
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn make_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    numbers: Vec<u64>,
    full: Option<Version>,
}

impl Partial {
    fn numeric_count(&self) -> usize {
        self.numbers.len()
    }

    fn padded(&self) -> Version {
        if let Some(v) = &self.full {
            return v.clone();
        }
        make_version(
            *self.numbers.first().unwrap_or(&0),
            *self.numbers.get(1).unwrap_or(&0),
            *self.numbers.get(2).unwrap_or(&0),
        )
    }
}

fn parse_partial(token: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = token.bytes().any(|b| b == b'-' || b == b'+');
    let fields: Vec<&str> = token.split('.').collect();
    let has_full_core_shape = fields.len() >= 3
        && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));

    if has_suffix || has_full_core_shape {
        let version = parse_version(token).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            numbers: vec![version.major, version.minor, version.patch],
            full: Some(version),
        });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }

    let mut numbers = Vec::new();
    let mut saw_wildcard = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            saw_wildcard = true;
            continue;
        }
        if saw_wildcard
            || !field.bytes().all(is_digit)
            || (field.len() > 1 && field.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }
        let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
        numbers.push(n);
    }

    Ok(Partial {
        numbers,
        full: None,
    })
}

fn increment_upper(p: &Partial, kind: &str, at: usize) -> Result<Version, RangeError> {
    let nums = &p.numbers;
    let k = nums.len();
    let (mut major, mut minor, mut patch) = (
        *nums.first().unwrap_or(&0),
        *nums.get(1).unwrap_or(&0),
        *nums.get(2).unwrap_or(&0),
    );

    let index = match kind {
        "bare" => match k {
            1 => 0,
            2 => 1,
            _ => return Err(RangeError::Syntax { at }),
        },
        "caret" => {
            if major > 0 || k == 1 {
                0
            } else if minor > 0 || k == 2 {
                1
            } else {
                2
            }
        }
        "tilde" => {
            if k == 1 {
                0
            } else {
                1
            }
        }
        "hyphen" => match k {
            1 => 0,
            2 => 1,
            _ => return Err(RangeError::Syntax { at }),
        },
        _ => return Err(RangeError::Syntax { at }),
    };

    match index {
        0 => {
            major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            minor = 0;
            patch = 0;
        }
        1 => {
            minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            patch = 0;
        }
        _ => {
            patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
        }
    }
    Ok(upper_version(major, minor, patch))
}

fn add_comparator(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(tuple(&version));
    }
    arm.comparators.push(Comparator { op, version });
}

fn add_partial_token(arm: &mut Arm, token: &str, at: usize) -> Result<(), RangeError> {
    let p = parse_partial(token, at)?;
    let k = p.numeric_count();
    if let Some(full) = p.full {
        add_comparator(arm, Op::Eq, full, true);
    } else if k == 0 {
        return Ok(());
    } else if k == 3 {
        add_comparator(arm, Op::Ge, p.padded(), true);
    } else {
        add_comparator(arm, Op::Ge, p.padded(), false);
        add_comparator(arm, Op::Lt, increment_upper(&p, "bare", at)?, false);
    }
    Ok(())
}

fn add_caret_or_tilde(
    arm: &mut Arm,
    token: &str,
    at: usize,
    kind: &str,
) -> Result<(), RangeError> {
    let operand = &token[1..];
    let operand_at = at + 1;
    let p = parse_partial(operand, operand_at)?;
    if p.numeric_count() == 0 {
        return Err(RangeError::Syntax { at });
    }
    let lower = p.padded();
    add_comparator(arm, Op::Ge, lower.clone(), true);
    let upper = increment_upper(&p, kind, operand_at)?;
    add_comparator(arm, Op::Lt, upper, false);
    Ok(())
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens_with_offsets(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut pos = start;
    while pos < end {
        while pos < end && whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos >= end {
            break;
        }
        let token_start = pos;
        while pos < end && !whitespace(bytes[pos]) {
            pos += 1;
        }
        result.push((token_start, &s[token_start..pos]));
    }
    result
}

fn parse_comparator(token: &str, at: usize) -> Result<Comparator, RangeError> {
    let (op, prefix) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else if token.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };

    let operand = &token[prefix..];
    let operand_at = at + prefix;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let version = parse_version(operand)
        .map_err(|error| RangeError::Version { at: operand_at, error })?;
    Ok(Comparator { op, version })
}

fn parse_hyphen_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let minus_pos = tokens.iter().find(|(_, t)| *t == &"-").unwrap().0;
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: minus_pos });
    }

    let (lower_at, lower_token) = tokens[0];
    let (upper_at, upper_token) = tokens[2];
    let lower = parse_partial(lower_token, lower_at)?;
    let upper = parse_partial(upper_token, upper_at)?;
    if lower.numeric_count() == 0 {
        return Err(RangeError::Syntax { at: lower_at });
    }
    if upper.numeric_count() == 0 {
        return Err(RangeError::Syntax { at: upper_at });
    }

    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    add_comparator(&mut arm, Op::Ge, lower.padded(), true);

    if let Some(full) = upper.full {
        add_comparator(&mut arm, Op::Le, full, true);
    } else if upper.numeric_count() == 3 {
        add_comparator(&mut arm, Op::Le, upper.padded(), false);
    } else {
        let bound = increment_upper(&upper, "hyphen", upper_at)?;
        add_comparator(&mut arm, Op::Lt, bound, false);
    }
    Ok(arm)
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let tokens = tokens_with_offsets(s, start, end);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: start });
    }

    if tokens.iter().any(|(_, token)| *token == "-") {
        return parse_hyphen_arm(&tokens);
    }

    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    for (at, token) in tokens {
        if token.starts_with(">") || token.starts_with("<") || token.starts_with("=") {
            let comparator = parse_comparator(token, at)?;
            add_comparator(&mut arm, comparator.op, comparator.version, true);
        } else if token.starts_with('^') {
            add_caret_or_tilde(&mut arm, token, at, "caret")?;
        } else if token.starts_with('~') {
            add_caret_or_tilde(&mut arm, token, at, "tilde")?;
        } else if token
            .as_bytes()
            .first()
            .map(|b| b.is_ascii_digit() || matches!(*b, b'*' | b'x' | b'X'))
            .unwrap_or(false)
        {
            add_partial_token(&mut arm, token, at)?;
        } else {
            return Err(RangeError::Syntax { at });
        }
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    if bytes.iter().all(|&b| whitespace(b)) {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut pos = 0usize;
    let mut arm_start = 0usize;
    let mut saw_separator = false;

    while pos < bytes.len() {
        if pos + 1 < bytes.len()
            && bytes[pos] == b'|'
            && bytes[pos + 1] == b'|'
            && (pos == 0 || whitespace(bytes[pos - 1]))
            && (pos + 2 == bytes.len() || whitespace(bytes[pos + 2]))
        {
            let arm = parse_arm(s, arm_start, pos)?;
            arms.push(arm);
            saw_separator = true;
            pos += 2;
            while pos < bytes.len() && whitespace(bytes[pos]) {
                pos += 1;
            }
            if pos == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
            arm_start = pos;
            continue;
        }
        pos += 1;
    }

    let last = parse_arm(s, arm_start, bytes.len())?;
    arms.push(last);

    if !saw_separator && arms.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty() && !arm.anchors.iter().any(|&core| core == tuple(v)) {
            return false;
        }
        arm.comparators.iter().all(|c| c.matches(v))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, value) in candidates.iter().enumerate() {
        let version = parse_version(value).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(current) => {
                let order = compare_versions(version, &parsed[current]);
                if (want_max && order == Ordering::Greater)
                    || (!want_max && order == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }

    Ok(best.map(|index| candidates[index].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
