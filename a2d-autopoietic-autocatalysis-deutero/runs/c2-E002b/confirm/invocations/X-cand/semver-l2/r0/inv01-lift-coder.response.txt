use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_number(s: &str, start: usize) -> Result<(u64, usize), VersionError> {
    let bytes = s.as_bytes();
    let mut end = start;
    while end < bytes.len() && is_digit(bytes[end]) {
        end += 1;
    }
    if end == start {
        return Err(match bytes.get(start).copied() {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let n = s[start..end]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok((n, end))
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut pos = start;
    loop {
        let mut stop = pos;
        while stop < end && bytes[stop] != b'.' {
            stop += 1;
        }
        if stop == pos {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
        for i in pos..stop {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }
        if prerelease
            && stop - pos > 1
            && bytes[pos..stop].iter().all(|b| is_digit(*b))
            && bytes[pos] == b'0'
        {
            return Err(VersionError::LeadingZero { at: pos });
        }
        result.push(s[pos..stop].to_string());
        if stop == end {
            break;
        }
        pos = stop + 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let (major, p1) = parse_number(s, 0)?;
    if s.as_bytes().get(p1) != Some(&b'.') {
        return Err(match s.as_bytes().get(p1).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p1 },
            _ => VersionError::InvalidChar { at: p1 },
        });
    }

    let (minor, p2) = parse_number(s, p1 + 1)?;
    if s.as_bytes().get(p2) != Some(&b'.') {
        return Err(match s.as_bytes().get(p2).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p2 },
            _ => VersionError::InvalidChar { at: p2 },
        });
    }

    let (patch, p3) = parse_number(s, p2 + 1)?;
    let bytes = s.as_bytes();
    let mut pos = p3;
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let start = pos + 1;
        let plus = bytes[start..]
            .iter()
            .position(|b| *b == b'+')
            .map(|i| start + i)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(s, start, plus, true)?;
        pos = plus;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        build = parse_identifiers(s, pos + 1, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    let an = numeric_identifier(a);
    let bn = numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let c = compare_identifier(x, y);
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Clause>>,
}

#[derive(Clone, Debug)]
struct Clause {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str, offset: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && whitespace(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !whitespace(bytes[i]) {
            i += 1;
        }
        out.push((s[start..i].to_string(), offset + start));
    }
    out
}

fn is_wild(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
    at: usize,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let bytes = s.as_bytes();
    let fullish = s.contains('-')
        || s.contains('+')
        || (s.split('.').count() >= 3 && !s.split('.').any(is_wild));
    if fullish {
        let v = parse_version(s).map_err(|error| RangeError::Version { at, error })?;
        return Ok(Partial {
            nums: vec![v.major, v.minor, v.patch],
            full: Some(v),
            at,
        });
    }

    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if is_wild(f) {
            wildcard_seen = true;
        } else if !wildcard_seen && f.bytes().all(is_digit) {
            if f.len() > 1 && f.starts_with('0') {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(Partial { nums, full: None, at })
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build: Vec::new(),
    }
}

fn padded(p: &Partial) -> Version {
    let mut n = [0u64; 3];
    for (i, v) in p.nums.iter().enumerate() {
        n[i] = *v;
    }
    if let Some(v) = &p.full {
        v.clone()
    } else {
        endpoint(n[0], n[1], n[2], Vec::new())
    }
}

fn bump(p: &Partial, which: usize) -> Result<Version, RangeError> {
    let mut n = [0u64; 3];
    for (i, v) in p.nums.iter().enumerate() {
        n[i] = *v;
    }
    n[which] = n[which]
        .checked_add(1)
        .ok_or(RangeError::Syntax { at: p.at })?;
    for x in which + 1..3 {
        n[x] = 0;
    }
    Ok(endpoint(n[0], n[1], n[2], vec!["0".to_string()]))
}

fn clause(op: Op, version: Version, anchor: bool, out: &mut Vec<Clause>) {
    out.push(Clause {
        op,
        version,
        anchor,
    });
}

fn add_partial_bare(p: Partial, out: &mut Vec<Clause>) -> Result<(), RangeError> {
    if let Some(v) = p.full {
        clause(Op::Eq, v.clone(), !v.pre.is_empty(), out);
        return Ok(());
    }
    let k = p.nums.len();
    if k == 0 {
        return Ok(());
    }
    clause(Op::Ge, padded(&p), false, out);
    let upper = bump(&p, if k == 1 { 0 } else { 1 })?;
    clause(Op::Lt, upper, false, out);
    Ok(())
}

fn parse_full_comparator(
    token: &str,
    at: usize,
    out: &mut Vec<Clause>,
) -> Result<(), RangeError> {
    let (op, len) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand_at = at + len;
    let operand = &token[len..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let v = parse_version(operand)
        .map_err(|error| RangeError::Version { at: operand_at, error })?;
    let anchor = !v.pre.is_empty();
    clause(op, v, anchor, out);
    Ok(())
}

fn parse_caret_tilde(
    token: &str,
    at: usize,
    caret: bool,
    out: &mut Vec<Clause>,
) -> Result<(), RangeError> {
    let operand = &token[1..];
    let operand_at = at + 1;
    if operand.is_empty() {
        return Err(RangeError::Syntax { at });
    }
    let p = parse_partial(operand, operand_at)?;
    let k = p.nums.len();
    if k == 0 {
        return Err(RangeError::Syntax { at });
    }
    let lower = padded(&p);
    let anchor = !lower.pre.is_empty();
    clause(Op::Ge, lower, anchor, out);

    let bump_index = if caret {
        if p.nums[0] > 0 || k == 1 {
            0
        } else if p.nums.get(1).copied().unwrap_or(0) > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if k == 1 {
        0
    } else {
        1
    };
    clause(Op::Lt, bump(&p, bump_index)?, false, out);
    Ok(())
}

fn parse_non_hyphen(tokens: &[(String, usize)]) -> Result<Vec<Clause>, RangeError> {
    let mut out = Vec::new();
    for (token, at) in tokens {
        let b = token.as_bytes()[0];
        if matches!(b, b'<' | b'>' | b'=') {
            parse_full_comparator(token, *at, &mut out)?;
        } else if b == b'^' || b == b'~' {
            parse_caret_tilde(token, *at, b == b'^', &mut out)?;
        } else if b.is_ascii_digit() || is_wild(token) {
            let p = parse_partial(token, *at)?;
            add_partial_bare(p, &mut out)?;
        } else {
            return Err(RangeError::Syntax { at: *at });
        }
    }
    Ok(out)
}

fn parse_hyphen(tokens: &[(String, usize)], minus_at: usize) -> Result<Vec<Clause>, RangeError> {
    if tokens.len() != 3 || tokens[1].0 != "-" {
        return Err(RangeError::Syntax { at: minus_at });
    }
    let lower = parse_partial(&tokens[0].0, tokens[0].1)?;
    let upper = parse_partial(&tokens[2].0, tokens[2].1)?;
    if lower.nums.is_empty() {
        return Err(RangeError::Syntax { at: tokens[0].1 });
    }
    if upper.nums.is_empty() {
        return Err(RangeError::Syntax { at: tokens[2].1 });
    }

    let mut out = Vec::new();
    let lo = padded(&lower);
    clause(Op::Ge, lo.clone(), !lo.pre.is_empty(), &mut out);
    if let Some(v) = upper.full {
        clause(Op::Le, v.clone(), !v.pre.is_empty(), &mut out);
    } else {
        let upper_bound = bump(&upper, if upper.nums.len() == 1 { 0 } else { 1 })?;
        clause(Op::Lt, upper_bound, false, &mut out);
    }
    Ok(out)
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Vec<Clause>, RangeError> {
    let standalone_minus = tokens.iter().find(|(t, _)| t == "-").map(|(_, at)| *at);
    if let Some(at) = standalone_minus {
        return parse_hyphen(tokens, at);
    }
    parse_non_hyphen(tokens)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s, 0);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].0 == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].1 });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(tokens[i].clone());
            i += 1;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn clause_matches(v: &Version, c: &Clause) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &[Clause]) -> bool {
    if !v.pre.is_empty()
        && !arm.iter().any(|c| {
            c.anchor && version_tuple(&c.version) == version_tuple(v)
        })
    {
        return false;
    }
    arm.iter().all(|c| clause_matches(v, c))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let v = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
