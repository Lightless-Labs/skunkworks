```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(fail(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' { i = 1; }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
        if i == num_start {
            return Err(fail(i, ParseKind::InvalidByte));
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
            if i == den_start.unwrap() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let digit_start = usize::from(negative);
        if num_end - digit_start >= 2 && bytes[digit_start] == b'0' {
            return Err(fail(digit_start + 1, ParseKind::NonCanonical));
        }
        if negative && &bytes[..num_end] == b"-0" {
            return Err(fail(0, ParseKind::NonCanonical));
        }
        if let (Some(s), Some(e)) = (den_start, den_end) {
            if e - s >= 2 && bytes[s] == b'0' {
                return Err(fail(s + 1, ParseKind::NonCanonical));
            }
        }

        let nmag = parse_u128(&bytes[digit_start..num_end]);
        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag.map_or(true, |n| n > nlimit) {
            return Err(fail(0, ParseKind::OutOfRange));
        }
        let n = if negative {
            if nmag == Some(1u128 << 63) { i64::MIN } else { -(nmag.unwrap() as i64) }
        } else {
            nmag.unwrap() as i64
        };

        let Some((ds, de)) = den_start.zip(den_end) else {
            return Ok(Rational { num: n, den: 1 });
        };
        let dmag = parse_u128(&bytes[ds..de]);
        if dmag.map_or(true, |d| d > i64::MAX as u128) {
            return Err(fail(ds, ParseKind::OutOfRange));
        }
        let d = dmag.unwrap() as i64;
        if d == 0 {
            return Err(fail(ds, ParseKind::ZeroDenominator));
        }
        let slash = ds - 1;
        if d == 1 || n == 0 {
            return Err(fail(slash, ParseKind::NonCanonical));
        }
        if gcd_u128(n.unsigned_abs() as u128, d as u128) > 1 {
            return Err(fail(slash, ParseKind::NonCanonical));
        }
        Ok(Rational { num: n, den: d })
    }
}

fn parse_u128(bytes: &[u8]) -> Option<u128> {
    let mut n = 0u128;
    for &b in bytes {
        n = n.checked_mul(10)?.checked_add((b - b'0') as u128)?;
    }
    Some(n)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let lhs = self.data[r * self.cols + k];
                    let rr = rhs.data[k * rhs.cols + c];
                    let term = lhs.mul(&rr).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self, self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let augmented = Matrix { rows: n, cols: width, data };
        let red = eliminate(&augmented, n, width)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut inv = Vec::with_capacity(len);
        for r in 0..n {
            inv.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Matrix { rows: n, cols: n, data: inv })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

fn eliminate(matrix: &Matrix, pivot_cols: usize, update_cols: usize) -> Result<Reduction, MatrixError> {
    let mut work = matrix.clone();
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..pivot_cols {
        if p == work.rows { break; }
        let mut chosen: Option<usize> = None;
        for row in p..work.rows {
            let v = work.data[row * work.cols + col];
            if v.num == 0 { continue; }
            let key = (v.den, v.num.unsigned_abs(), row);
            if chosen.map_or(true, |old| {
                let o = work.data[old * work.cols + col];
                key < (o.den, o.num.unsigned_abs(), old)
            }) {
                chosen = Some(row);
            }
        }
        let Some(q) = chosen else { continue };
        if q != p {
            for c in 0..work.cols {
                work.data.swap(p * work.cols + c, q * work.cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = work.data[p * work.cols + col];
        let mut normalized = Vec::with_capacity(update_cols);
        for c in 0..update_cols {
            let original = work.data[p * work.cols + c];
            normalized.push(original.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                row: p, col: c, step: p, source
            })?);
        }
        for c in 0..update_cols {
            work.data[p * work.cols + c] = normalized[c];
        }

        for row in 0..work.rows {
            if row == p { continue; }
            let factor = work.data[row * work.cols + col];
            if factor.num == 0 { continue; }
            for c in 0..update_cols {
                let product = factor.mul(&normalized[c]).map_err(|source| MatrixError::Arithmetic {
                    row, col: c, step: p, source
                })?;
                let current = work.data[row * work.cols + c];
                work.data[row * work.cols + c] = current.sub(&product).map_err(|source| MatrixError::Arithmetic {
                    row, col: c, step: p, source
                })?;
            }
        }
        pivots.push(col);
        p += 1;
    }

    Ok(Reduction { matrix: work, pivots, swaps })
}
```
