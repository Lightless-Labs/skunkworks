#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Debug, Clone)]
enum Row {
    Context(String),
    Remove(String),
    Add(String),
}

#[derive(Debug, Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_decimal(bytes: &[u8], line: usize) -> Result<usize, ApplyError> {
    if bytes.is_empty()
        || (bytes.len() > 1 && bytes[0] == b'0')
        || !bytes.iter().all(u8::is_ascii_digit)
    {
        return Err(ApplyError::BadHunkHeader { line });
    }

    let mut value = 0usize;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as usize))
            .ok_or(ApplyError::BadHunkHeader { line })?;
        if value > 128 {
            return Err(ApplyError::BadHunkHeader { line });
        }
    }
    Ok(value)
}

fn parse_hunk_header(line_text: &str, line: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line };
    let bytes = line_text.as_bytes();
    if !bytes.starts_with(b"@@ -") || !bytes.ends_with(b" @@") {
        return Err(bad());
    }

    let inner = &bytes[4..bytes.len() - 3];
    let split = inner
        .windows(3)
        .position(|w| w == b" +")
        .ok_or_else(bad)?;
    let old = &inner[..split];
    let new = &inner[split + 2..];

    let comma_old = old.iter().position(|&b| b == b',').ok_or_else(bad)?;
    let comma_new = new.iter().position(|&b| b == b',').ok_or_else(bad)?;
    if old[comma_old + 1..].contains(&b',') || new[comma_new + 1..].contains(&b',') {
        return Err(bad());
    }

    let a = parse_decimal(&old[..comma_old], line)?;
    let b = parse_decimal(&old[comma_old + 1..], line)?;
    let c = parse_decimal(&new[..comma_new], line)?;
    let d = parse_decimal(&new[comma_new + 1..], line)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }

    Ok((a, b, c, d))
}

fn physical_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }
    let mut lines: Vec<&str> = text.split_terminator('\n').collect();
    if text.ends_with('\n') {
        // split_terminator already omits the final empty component.
    } else {
        lines.push("");
    }
    lines
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for (i, expected) in [(0, "--- before"), (1, "+++ after")] {
        if lines.get(i).copied() != Some(expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() <= 2 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut index = 2;
    let mut previous_old_end: Option<usize> = None;
    let mut preceding_delta: i64 = 0;

    while index < lines.len() {
        let header_line = index + 1;
        if !lines[index].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let (a, b, c, d) = parse_hunk_header(lines[index], header_line)?;
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        index += 1;
        let body_start = index;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while index < lines.len() && !lines[index].starts_with("@@") {
            let row_line = index + 1;
            let bytes = lines[index].as_bytes();
            if bytes.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &lines[index][1..];
            if payload.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            match bytes[0] {
                b' ' => {
                    rows.push(Row::Context(payload.to_owned()));
                    old_seen += 1;
                    new_seen += 1;
                }
                b'-' => {
                    rows.push(Row::Remove(payload.to_owned()));
                    old_seen += 1;
                }
                b'+' => {
                    rows.push(Row::Add(payload.to_owned()));
                    new_seen += 1;
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            }
            index += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| matches!(r, Row::Add(_) | Row::Remove(_))) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if let Some(end) = previous_old_end {
            if old_start < end {
                return Err(ApplyError::Misordered { line: header_line });
            }
        }
        let required_new = old_start as i64 + preceding_delta;
        if required_new < 0 || new_start as i64 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        let old_end = old_start + b;
        previous_old_end = Some(old_end);
        preceding_delta += d as i64 - b as i64;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });

        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }

        if index == body_start {
            // A header with an empty body is rejected by the no-change check above.
            unreachable!();
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn abs_diff_i64(a: i64, b: i64) -> i64 {
    if a >= b { a - b } else { b - a }
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }

    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source.split_terminator('\n').map(str::to_owned).collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i64 = 0;
    let mut drift: i64 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| matches!(r, Row::Context(_))).count();
        let trail = hunk.rows.iter().rev().take_while(|r| matches!(r, Row::Context(_))).count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i64 + net + drift;

        let mut best_rank: Option<(usize, i64)> = None;
        let mut best_starts = Vec::new();
        let mut best_choices: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    match row {
                        Row::Context(s) | Row::Remove(s) => old_payloads.push(s.as_str()),
                        Row::Add(_) => {}
                    }
                }
                if old_payloads.len() != hunk.old_count - l - r {
                    continue;
                }

                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_diff_i64(s as i64, expected);
                    if distance > options.max_offset as i64 {
                        continue;
                    }
                    if old_payloads
                        .iter()
                        .enumerate()
                        .all(|(i, payload)| working[s + l + i] == *payload)
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best_starts.clear();
                                best_choices.clear();
                            }
                            Some(current) if rank < current => {
                                best_rank = Some(rank);
                                best_starts.clear();
                                best_choices.clear();
                            }
                            Some(current) if rank > current => continue,
                            _ => {}
                        }
                        best_starts.push(s);
                        best_choices.push((s, l, r));
                    }
                }
            }
        }

        if best_choices.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_index + 1 });
        }

        best_starts.sort_unstable();
        best_starts.dedup();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let chosen_start = best_starts[0];
        let (_, chosen_l, chosen_r) = best_choices
            .into_iter()
            .filter(|(s, _, _)| *s == chosen_start)
            .min_by_key(|(_, l, _)| *l)
            .unwrap();

        let trimmed = &hunk.rows[chosen_l..hunk.rows.len() - chosen_r];
        let mut replacement = Vec::new();
        for row in trimmed {
            match row {
                Row::Context(s) | Row::Add(s) => replacement.push(s.clone()),
                Row::Remove(_) => {}
            }
        }
        let replace_start = chosen_start + chosen_l;
        let replace_end = chosen_start + hunk.old_count - chosen_r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i64 - hunk.old_count as i64;
        drift = chosen_start as i64 - (hunk.old_start as i64 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
