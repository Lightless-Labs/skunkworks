```rust
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended {
        id: String,
        price: u16,
        change: Vec<u16>,
    },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked {
        id: String,
        added: u16,
        stock: u16,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&d| d == coin)
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn product_index(products: &[Product], id: &str) -> Option<usize> {
    products.iter().position(|product| product.id == id)
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount as u32;
    let mut change = [0u16; 4];

    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index] as u32;
        let take = (tubes[index] as u32).min(remaining / denomination);
        change[index] = take as u16;
        remaining -= take * denomination;
    }

    if remaining == 0 {
        Some(change)
    } else {
        None
    }
}

fn better_change(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
    let candidate_count: u32 = candidate.iter().map(|&n| n as u32).sum();
    let current_count: u32 = current.iter().map(|&n| n as u32).sum();

    if candidate_count != current_count {
        return candidate_count < current_count;
    }

    for index in (0..4).rev() {
        if candidate[index] != current[index] {
            return candidate[index] > current[index];
        }
    }

    false
}

fn search_change(
    index: isize,
    remaining: u32,
    tubes: &[u16; 4],
    candidate: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if remaining == 0 {
        if best
            .as_ref()
            .is_none_or(|current| better_change(candidate, current))
        {
            *best = Some(*candidate);
        }
        return;
    }
    if index < 0 {
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i] as u32;
    let max_take = (tubes[i] as u32).min(remaining / denomination);

    for take in (0..=max_take).rev() {
        candidate[i] = take as u16;
        search_change(
            index - 1,
            remaining - take * denomination,
            tubes,
            candidate,
            best,
        );
    }
    candidate[i] = 0;
}

fn exact_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    if let Some(change) = greedy_change(amount, tubes) {
        return Some(change);
    }

    let mut best = None;
    let mut candidate = [0u16; 4];
    search_change(
        3,
        amount as u32,
        tubes,
        &mut candidate,
        &mut best,
    );
    best
}

fn descending_coins(counts: &[u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for index in 0..4 {
            if tubes[index] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if denomination_index(coin).is_none() {
            return Err(Error::BadCoin { coin });
        }

        if self.credit as u32 + coin as u32 > 100 {
            return Err(Error::CreditLimit);
        }

        let credit = self.credit + coin;
        self.escrow.push(coin);
        self.credit = credit;
        self.log(EventKind::Inserted { coin, credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if product_index(&self.products, id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = product_index(&self.products, &id).expect("selected catalog product");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change_counts =
            exact_change(amount, &self.tubes).ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for index in 0..4 {
            proposed_tubes[index] -= change_counts[index];
        }
        for &coin in &self.escrow {
            let index = denomination_index(coin).expect("escrow contains accepted coins");
            proposed_tubes[index] += 1;
        }
        for index in 0..4 {
            if proposed_tubes[index] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        let change = descending_coins(&change_counts);
        self.tubes = proposed_tubes;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = product_index(&self.products, id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;

        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = product_index(&self.products, id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;

        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let new_stock = self.products[index].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        let stock = new_stock as u16;
        self.products[index].stock = stock;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}
```
