#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const INPUT_LIMIT: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > INPUT_LIMIT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[effective.len() - 1] == effective[0] {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn dot_from_vertex(prev: Point, current: Point, next: Point) -> i128 {
    (prev.x as i128 - current.x as i128) * (next.x as i128 - current.x as i128)
        + (prev.y as i128 - current.y as i128) * (next.y as i128 - current.y as i128)
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let d2 = (dx * dx + dy * dy) as u128;
    let scaled = d2 * 1_000_000;
    let root = isqrt(scaled);
    let remainder = scaled - root * root;
    let rounded = root + u128::from(remainder >= root + 1);
    rounded as u64
}

fn measures_for(points: &[Point]) -> Measures {
    let signed = signed_area2(points);
    let mut edge_milli = Vec::with_capacity(points.len());
    for i in 0..points.len() {
        edge_milli.push(rounded_edge_milli(
            points[i],
            points[(i + 1) % points.len()],
        ));
    }
    let perimeter_milli = edge_milli.iter().copied().sum();

    let longest_edge = if edge_milli.is_empty() {
        None
    } else {
        let mut best = 0usize;
        for i in 1..edge_milli.len() {
            let endpoints_i = ordered_endpoints(points[i], points[(i + 1) % points.len()]);
            let endpoints_best = ordered_endpoints(
                points[best],
                points[(best + 1) % points.len()],
            );
            if edge_milli[i] > edge_milli[best]
                || (edge_milli[i] == edge_milli[best]
                    && (endpoints_i < endpoints_best
                        || (endpoints_i == endpoints_best && i < best)))
            {
                best = i;
            }
        }
        Some(best)
    };

    Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: if signed > 0 {
            Orientation::CounterClockwise
        } else if signed < 0 {
            Orientation::Clockwise
        } else {
            Orientation::Degenerate
        },
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

fn ordered_endpoints(a: Point, b: Point) -> (Point, Point) {
    if a <= b {
        (a, b)
    } else {
        (b, a)
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let effective = prepare_polygon(points)?;
    Ok(measures_for(&effective))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > INPUT_LIMIT {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if on_segment(a, b, query) {
                boundary = Some(i);
                break;
            }
        }

        if let Some(edge) = boundary {
            result.push(Location::Boundary { edge });
            continue;
        }

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > query.y) != (b.y > query.y) {
                let c = cross(a, b, query);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        result.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(result)
}

fn point_on_line_segment(a: Point, b: Point, p: Point) -> bool {
    on_segment(a, b, p)
}

fn hull_turn(a: Point, b: Point, c: Point) -> i128 {
    cross(a, b, c)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > INPUT_LIMIT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(existing, _)| existing == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    let make_hull = |vertices: Vec<(Point, usize)>| {
        let pts: Vec<Point> = vertices.iter().map(|&(point, _)| point).collect();
        let area = if pts.len() >= 3 {
            signed_area2(&pts).unsigned_abs()
        } else {
            0
        };
        let perimeter = if pts.is_empty() {
            0
        } else if pts.len() == 1 {
            0
        } else {
            (0..pts.len())
                .map(|i| rounded_edge_milli(pts[i], pts[(i + 1) % pts.len()]))
                .sum()
        };
        Hull {
            vertices: pts,
            source_indices: vertices.iter().map(|&(_, index)| index).collect(),
            area2: area,
            perimeter_milli: perimeter,
        }
    };

    if unique.len() <= 2 {
        return Ok(make_hull(unique));
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|&(p, _)| hull_turn(unique[0].0, unique[1].0, p) == 0);
    if all_collinear {
        return Ok(match mode {
            Collinear::Include => make_hull(unique),
            Collinear::Exclude => make_hull(vec![unique[0], *unique.last().unwrap()]),
        });
    }

    let mut lower: Vec<(Point, usize)> = Vec::new();
    for &(point, index) in &unique {
        while lower.len() >= 2 {
            let turn = hull_turn(lower[lower.len() - 2].0, lower[lower.len() - 1].0, point);
            if turn > 0 || (mode == Collinear::Include && turn == 0) {
                break;
            }
            lower.pop();
        }
        lower.push((point, index));
    }

    let mut upper: Vec<(Point, usize)> = Vec::new();
    for &(point, index) in unique.iter().rev() {
        while upper.len() >= 2 {
            let turn = hull_turn(upper[upper.len() - 2].0, upper[upper.len() - 1].0, point);
            if turn > 0 || (mode == Collinear::Include && turn == 0) {
                break;
            }
            upper.pop();
        }
        upper.push((point, index));
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);

    if mode == Collinear::Include {
        let mut boundary: Vec<(Point, usize)> = Vec::new();
        for i in 0..lower.len() {
            let a = lower[i].0;
            let b = lower[(i + 1) % lower.len()].0;
            if !boundary.iter().any(|&(p, _)| p == a) {
                boundary.push(lower[i]);
            }
            for &candidate in &unique {
                if candidate.0 != a
                    && candidate.0 != b
                    && point_on_line_segment(a, b, candidate.0)
                    && !boundary.iter().any(|&(p, _)| p == candidate.0)
                {
                    // Candidates are lexicographically sorted; sort along the directed edge.
                    let mut inserted = false;
                    for slot in 0..boundary.len() {
                        let current = boundary[slot].0;
                        if current == a {
                            continue;
                        }
                        if cross(a, b, candidate.0) == 0
                            && ((b.x >= a.x && candidate.0.x < current.x)
                                || (b.x < a.x && candidate.0.x > current.x)
                                || (b.x == a.x
                                    && ((b.y >= a.y && candidate.0.y < current.y)
                                        || (b.y < a.y && candidate.0.y > current.y)))
                        {
                            boundary.insert(slot, candidate);
                            inserted = true;
                            break;
                        }
                    }
                    if !inserted {
                        boundary.push(candidate);
                    }
                }
            }
        }
        lower = boundary;
    }

    Ok(make_hull(lower))
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);

    if ab_c == 0 && on_segment(a, b, c) {
        return true;
    }
    if ab_d == 0 && on_segment(a, b, d) {
        return true;
    }
    if cd_a == 0 && on_segment(c, d, a) {
        return true;
    }
    if cd_b == 0 && on_segment(c, d, b) {
        return true;
    }
    (ab_c > 0) != (ab_d > 0) && (cd_a > 0) != (cd_b > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let effective = prepare_polygon(points)?;
    let n = effective.len();

    for i in 0..n {
        for j in (i + 1)..n {
            if effective[i] == effective[j] {
                return Err(GeomError::Duplicate {
                    first: i,
                    second: j,
                });
            }
        }
    }

    let area = signed_area2(&effective);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = effective[(i + n - 1) % n];
        let current = effective[i];
        let next = effective[(i + 1) % n];
        if cross(prev, current, next) == 0 && dot_from_vertex(prev, current, next) > 0 {
            return Err(GeomError::AdjacentOverlap { vertex: i });
        }
    }

    for i in 0..n {
        let a = effective[i];
        let b = effective[(i + 1) % n];
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            let c = effective[j];
            let d = effective[(j + 1) % n];
            if segments_intersect(a, b, c, d) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let min_index = (0..n).min_by_key(|&i| effective[i]).unwrap();
    let mut indices = Vec::with_capacity(n);
    if area > 0 {
        for k in 0..n {
            indices.push((min_index + k) % n);
        }
    } else {
        for k in 0..n {
            indices.push((min_index + n - k) % n);
        }
    }

    let vertices: Vec<Point> = indices.iter().map(|&i| effective[i]).collect();
    let measures = measures_for(&vertices);
    Ok(Validated {
        vertices,
        source_indices: indices,
        measures,
    })
}
