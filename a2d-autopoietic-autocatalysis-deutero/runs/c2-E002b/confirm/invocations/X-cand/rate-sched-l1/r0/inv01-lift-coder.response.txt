#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u128,
    jobs: Vec<ScheduledJob>,
    states: std::collections::HashMap<String, JobState>,
}

struct ScheduledJob {
    job: Job,
    sequence: u128,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum JobState {
    Queued,
    Cancelled,
    Completed,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: Vec::new(),
            states: std::collections::HashMap::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.states.contains_key(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        let next_sequence = sequence.checked_add(1).expect("sequence space exhausted");

        self.jobs.push(ScheduledJob { job: job.clone(), sequence });
        self.states.insert(job.id, JobState::Queued);
        self.next_sequence = next_sequence;
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let state = self.states.get(id).copied().ok_or(Error::Unknown)?;
        match state {
            JobState::Queued => {
                self.states.insert(id.to_owned(), JobState::Cancelled);
                self.last_time = now;
                Ok(())
            }
            JobState::Cancelled => Err(Error::Cancelled),
            JobState::Completed => Err(Error::Completed),
        }
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let mut best: Option<(usize, u8, u128)> = None;
        for (index, scheduled) in self.jobs.iter().enumerate() {
            if self.states.get(&scheduled.job.id) != Some(&JobState::Queued)
                || scheduled.job.ready_at > now
            {
                continue;
            }

            let effective = effective_priority(&scheduled.job, now, self.aging_ms);
            match best {
                None => best = Some((index, effective, scheduled.sequence)),
                Some((_, best_priority, best_sequence))
                    if effective > best_priority
                        || (effective == best_priority && scheduled.sequence < best_sequence) =>
                {
                    best = Some((index, effective, scheduled.sequence));
                }
                _ => {}
            }
        }

        self.last_time = now;
        if let Some((index, effective_priority, _)) = best {
            let scheduled = self.jobs.remove(index);
            self.states.insert(scheduled.job.id.clone(), JobState::Completed);
            Ok(Some(Dispatch {
                id: scheduled.job.id,
                effective_priority,
            }))
        } else {
            Ok(None)
        }
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes.iter().all(|byte| {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-'
        })
}

fn effective_priority(job: &Job, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return job.priority;
    }

    let age_steps = (now - job.ready_at) / aging_ms;
    let room = (9 - job.priority) as u64;
    if age_steps >= room {
        9
    } else {
        job.priority + age_steps as u8
    }
}
