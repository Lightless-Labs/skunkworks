use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Byte {
    value: u8,
    pos: usize,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn cleanup(input: &[u8]) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let mut kept = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_mode = false;
    let mut first_retained = true;

    while i < input.len() {
        let b = input[i];

        if quote_mode {
            kept.push(Byte { value: b, pos: i });
            i += 1;
            if b == b'\\' {
                if i < input.len() {
                    kept.push(Byte { value: input[i], pos: i });
                    i += 1;
                }
            } else if b == b'"' {
                quote_mode = false;
            }
            continue;
        }

        if first_retained && b == b'"' {
            first_retained = false;
            quote_mode = true;
            kept.push(Byte { value: b, pos: i });
            i += 1;
            continue;
        }

        let run_start_i = i;
        let run_start_pos = i;
        let mut had_cfws = false;

        loop {
            if i >= input.len() {
                break;
            }
            let c = input[i];

            if is_hws(c) {
                had_cfws = true;
                i += 1;
                continue;
            }

            if c == b'\r' || c == b'\n' {
                if c == b'\r'
                    && i + 2 < input.len()
                    && input[i + 1] == b'\n'
                    && is_hws(input[i + 2])
                {
                    had_cfws = true;
                    i += 2;
                    while i < input.len() && is_hws(input[i]) {
                        i += 1;
                    }
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }

            if c == b'(' {
                had_cfws = true;
                let outer = i;
                let mut depth = 1usize;
                i += 1;
                while i < input.len() {
                    let x = input[i];
                    if x == b'\\' {
                        let slash = i;
                        if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                        continue;
                    }
                    if x == b'\r' || x == b'\n' {
                        if x == b'\r'
                            && i + 2 < input.len()
                            && input[i + 1] == b'\n'
                            && is_hws(input[i + 2])
                        {
                            i += 2;
                            while i < input.len() && is_hws(input[i]) {
                                i += 1;
                            }
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }
                    if x == b'(' {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        depth += 1;
                        i += 1;
                        continue;
                    }
                    if x == b')' {
                        depth -= 1;
                        i += 1;
                        if depth == 0 {
                            break;
                        }
                        continue;
                    }
                    if !is_printable(x) && x != b'\t' {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                }
                if depth != 0 {
                    return Err(EmailError::UnclosedComment { at: outer });
                }
                continue;
            }

            break;
        }

        if had_cfws {
            runs.push(Run {
                start: run_start_pos,
                end: i,
            });
            if i == run_start_i {
                unreachable!();
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        first_retained = false;
        kept.push(Byte { value: b, pos: i });
        i += 1;
    }

    Ok((kept, runs))
}

fn is_local_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn is_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if is_local_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_with_policy(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (bytes, runs) = cleanup(raw)?;
    let mut at_positions = Vec::new();
    for (i, byte) in bytes.iter().enumerate() {
        if byte.value == b'@' {
            at_positions.push(i);
        }
    }

    for run in &runs {
        let left_at = bytes
            .iter()
            .enumerate()
            .any(|(i, b)| b.value == b'@' && i + 1 == 0);
        let _ = left_at;
        let first_after = bytes.iter().position(|b| b.pos >= run.end);
        let last_before = bytes.iter().rposition(|b| b.pos < run.start);
        let boundary = first_after
            .map(|i| bytes[i].value == b'@')
            .unwrap_or(true)
            || last_before
                .map(|i| bytes[i].value == b'@')
                .unwrap_or(true);
        if !boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    let mut quoted_prefix = false;
    if bytes[0].value == b'"' {
        quoted_prefix = true;
        let mut i = 1;
        let mut found = false;
        while i < bytes.len() {
            if bytes[i].value == b'\\' {
                i += 2;
            } else if bytes[i].value == b'"' {
                protected_end = i;
                found = true;
                break;
            } else {
                i += 1;
            }
        }
        if !found {
            return Err(EmailError::UnclosedQuote { at: bytes[0].pos });
        }
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, b)| {
            if b.value == b'@' && (!quoted_prefix || i > protected_end) {
                Some(i)
            } else {
                None
            }
        })
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: bytes[separators[1]].pos,
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: bytes[sep].pos });
    }
    let local = &bytes[..sep];
    let domain = &bytes[sep + 1..];

    let (mut local_value, mut local_quoted, mut local_origins) = if local[0].value == b'"' {
        if local.len() < 2 || local.last().unwrap().value != b'"' {
            let closing = local.iter().position(|b| b.value == b'"');
            if let Some(q) = closing {
                return Err(EmailError::TrailingLocal {
                    at: local[q + 1].pos,
                });
            }
            return Err(EmailError::UnclosedQuote { at: local[0].pos });
        }

        let close = local.len() - 1;
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = local[i].value;
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: local[i].pos });
                }
                let next = local[i + 1].value;
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: local[i].pos });
                }
                decoded.push(next);
                origins.push(local[i].pos);
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: local[i].pos });
                }
                decoded.push(b);
                origins.push(local[i].pos);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: local[close].pos,
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (
            decoded,
            true,
            origins,
        )
    } else {
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: local[64].pos });
        }
        let mut previous_dot = false;
        for (i, b) in local.iter().enumerate() {
            if b.value == b'.' {
                if i == 0 || i + 1 == local.len() || previous_dot {
                    return Err(EmailError::LocalDot { at: b.pos });
                }
                previous_dot = true;
            } else {
                if !is_local_atom_char(b.value) {
                    return Err(EmailError::LocalChar { at: b.pos });
                }
                previous_dot = false;
            }
        }
        (
            local.iter().map(|b| b.value).collect(),
            false,
            local.iter().map(|b| b.pos).collect(),
        )
    };

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: input.len() });
    }
    let mut domain_end = domain.len();
    let root_pos = if domain.last().unwrap().value == b'.' {
        domain_end -= 1;
        Some(domain[domain_end].pos)
    } else {
        None
    };
    if domain_end == 0 {
        return Err(EmailError::EmptyDomain {
            at: domain[0].pos,
        });
    }
    if domain_end > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain[253].pos,
        });
    }

    let d = &domain[..domain_end];
    let mut labels = Vec::new();
    let mut start = 0usize;
    for i in 0..=d.len() {
        if i == d.len() || d[i].value == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }

    for (n, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let pos = if start < d.len() {
                d[start].pos
            } else if n > 0 {
                d[end - 1].pos
            } else {
                root_pos.unwrap_or(input.len())
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: d[start + 63].pos,
            });
        }
        if d[start].value == b'-' {
            return Err(EmailError::LabelHyphen { at: d[start].pos });
        }
        if d[end - 1].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: d[end - 1].pos,
            });
        }
        for b in &d[start..end] {
            if !b.value.is_ascii_alphanumeric() && b.value != b'-' {
                return Err(EmailError::DomainChar { at: b.pos });
            }
        }
    }

    if labels.len() < 2 && !policy.allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: domain[0].pos,
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if d[last_start..last_end]
        .iter()
        .all(|b| b.value.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: d[last_start].pos,
        });
    }

    let mut domain_value: String = d.iter().map(|b| b.value as char).collect();
    domain_value.make_ascii_lowercase();

    if policy.local_case == LocalCase::Lower {
        for b in &mut local_value {
            b.make_ascii_lowercase();
        }
    }

    if policy.strip_subaddress {
        if let Some(plus) = local_value.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: local_origins[plus],
                });
            }
            local_value.truncate(plus);
            local_origins.truncate(plus);
        }
    }

    let safe = is_dot_atom(&local_value);
    if !safe {
        local_quoted = true;
    } else if policy.unquote_safe {
        local_quoted = false;
    }

    Ok(Address {
        local: String::from_utf8(local_value).unwrap(),
        domain: domain_value,
        quoted: local_quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default())
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((i, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = seen.get(&key) {
                    duplicate_of[i] = Some(first);
                } else {
                    seen.insert(key, i);
                    source_indices.push(i);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
