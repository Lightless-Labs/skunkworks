use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut protected_quote = false;

    while i < input.len() {
        if protected_quote {
            let b = input[i];
            bytes.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' && i < input.len() {
                bytes.push(input[i]);
                positions.push(i);
                i += 1;
            } else if b == b'"' {
                protected_quote = false;
            }
            continue;
        }

        let b = input[i];

        if depth > 0 {
            if b == b'\\' {
                if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if b != b'\r'
                    || i + 2 >= input.len()
                    || input[i + 1] != b'\n'
                    || !is_hws(input[i + 2])
                {
                    return Err(EmailError::BadFold { at: i });
                }
                i += 3;
                while i < input.len() && is_hws(input[i]) {
                    i += 1;
                }
                continue;
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !is_printable(b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'(' || b == b'\r' || b == b'\n' {
            let start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if c != b'\r'
                        || i + 2 >= input.len()
                        || input[i + 1] != b'\n'
                        || !is_hws(input[i + 2])
                    {
                        return Err(EmailError::BadFold { at: i });
                    }
                    i += 3;
                    while i < input.len() && is_hws(input[i]) {
                        i += 1;
                    }
                    continue;
                }
                if c == b'(' {
                    if depth == 0 {
                        outer_open = i;
                    }
                    depth = 1;
                    i += 1;
                    while i < input.len() && depth > 0 {
                        let x = input[i];
                        if x == b'\\' {
                            if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if x == b'\r' || x == b'\n' {
                            if x != b'\r'
                                || i + 2 >= input.len()
                                || input[i + 1] != b'\n'
                                || !is_hws(input[i + 2])
                            {
                                return Err(EmailError::BadFold { at: i });
                            }
                            i += 3;
                            while i < input.len() && is_hws(input[i]) {
                                i += 1;
                            }
                        } else if x == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            i += 1;
                        } else if x == b')' {
                            depth -= 1;
                            i += 1;
                        } else if !is_printable(x) && x != b'\t' {
                            return Err(EmailError::CommentChar { at: i });
                        } else {
                            i += 1;
                        }
                    }
                    if depth > 0 {
                        return Err(EmailError::UnclosedComment { at: outer_open });
                    }
                    continue;
                }
                break;
            }
            runs.push((start, i));
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if bytes.is_empty() && b == b'"' {
            protected_quote = true;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn pos(clean: &Cleaned, idx: usize, input_len: usize) -> usize {
    clean.positions.get(idx).copied().unwrap_or(input_len)
}

fn local_dot_atom_valid(local: &[u8]) -> bool {
    if local.is_empty() || local[0] == b'.' || local[local.len() - 1] == b'.' {
        return false;
    }
    let mut previous_dot = false;
    for &b in local {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else {
            previous_dot = false;
            if !is_local_char(b) {
                return false;
            }
        }
    }
    true
}

fn is_local_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn parse_internal(
    input: &str,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let clean = cleanup(raw)?;

    for &(start, end) in &clean.runs {
        let legal = start == 0
            || end == raw.len()
            || (start > 0 && clean.bytes.get(start.saturating_sub(1)) == Some(&b'@'))
            || clean.bytes.get(start) == Some(&b'@')
            || clean.bytes.get(end) == Some(&b'@');
        if !legal {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if clean.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let bytes = &clean.bytes;
    let mut quote_end = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(&clean, 0, raw.len()),
            });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !(quote_end.is_some() && i <= protected_end) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(&clean, ats[1], raw.len()),
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&clean, sep, raw.len()),
        });
    }

    let local_slice = &bytes[..sep];
    let mut decoded = Vec::new();
    let mut origins = Vec::new();
    let quoted = local_slice.first() == Some(&b'"');

    if !quoted {
        if local_slice.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&clean, 64, raw.len()),
            });
        }
        for i in 0..local_slice.len() {
            let b = local_slice[i];
            if b == b'.'
                && (i == 0 || i + 1 == local_slice.len() || local_slice[i - 1] == b'.')
            {
                return Err(EmailError::LocalDot {
                    at: pos(&clean, i, raw.len()),
                });
            }
            if b != b'.' && !is_local_char(b) {
                return Err(EmailError::LocalChar {
                    at: pos(&clean, i, raw.len()),
                });
            }
        }
        decoded.extend_from_slice(local_slice);
        origins.extend((0..local_slice.len()).map(|i| pos(&clean, i, raw.len())));
    } else {
        let end = quote_end.expect("quoted prefix has a closing quote");
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&clean, end + 1, raw.len()),
            });
        }
        let mut i = 1;
        while i < end {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: pos(&clean, i, raw.len()),
                    });
                }
                let next = bytes[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: pos(&clean, i, raw.len()),
                    });
                }
                decoded.push(next);
                origins.push(pos(&clean, i, raw.len()));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(&clean, i, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(pos(&clean, i, raw.len()));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&clean, end, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    }

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let mut root_pos = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_pos = Some(pos(&clean, domain_end, raw.len()));
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&clean, domain_start, raw.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&clean, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || bytes[i] == b'.' {
            if start == i {
                let at = if i == domain_end {
                    root_pos.unwrap_or_else(|| pos(&clean, start, raw.len()))
                } else {
                    pos(&clean, i, raw.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((start, i));
            start = i + 1;
        }
    }

    for &(s, e) in &labels {
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&clean, s + 63, raw.len()),
            });
        }
        if bytes[s] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&clean, s, raw.len()),
            });
        }
        if bytes[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&clean, e - 1, raw.len()),
            });
        }
        for i in s..e {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: pos(&clean, i, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&clean, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(&clean, last_start, raw.len()),
        });
    }

    let local = String::from_utf8(decoded).expect("validated ASCII local");
    let domain = String::from_utf8(
        bytes[domain_start..domain_end]
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .expect("validated ASCII domain");
    Ok((
        Address {
            local,
            domain,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();
    let mut mapped_origins = origins;

    if policy.local_case == LocalCase::Lower {
        for b in &mut local {
            if b.is_ascii_uppercase() {
                *b = b.to_ascii_lowercase();
            }
        }
    }

    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: mapped_origins.first().copied().unwrap_or(input.len()),
                });
            }
            local.truncate(plus);
            mapped_origins.truncate(plus);
        }
    }

    let safe = local_dot_atom_valid(&local);
    address.local = String::from_utf8(local).expect("normalized local is ASCII");
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    addresses.push(address);
                    source_indices.push(index);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
