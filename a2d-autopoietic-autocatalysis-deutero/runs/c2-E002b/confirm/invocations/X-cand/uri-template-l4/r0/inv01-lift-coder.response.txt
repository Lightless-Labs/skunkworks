#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semi,
    Question,
    Amp,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier {
    name: String,
    modifier: Modifier,
    pos: usize,
}

struct Expression {
    op: Operator,
    specs: Vec<Specifier>,
}

fn name_char(c: u8) -> bool {
    c.is_ascii_alphanumeric() || c == b'_'
}

fn valid_name_start(c: u8) -> bool {
    c.is_ascii_alphabetic() || c == b'_'
}

fn parse_expression(content: &str, base: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (op, start) = if bytes.is_empty() {
        return Err(Error::EmptyExpression { pos: base });
    } else {
        match bytes[0] {
            b'+' => (Operator::Plus, 1),
            b'#' => (Operator::Hash, 1),
            b'.' => (Operator::Dot, 1),
            b'/' => (Operator::Slash, 1),
            b';' => (Operator::Semi, 1),
            b'?' => (Operator::Question, 1),
            b'&' => (Operator::Amp, 1),
            _ => (Operator::Simple, 0),
        }
    };

    if start == bytes.len() {
        return Err(Error::EmptyExpression { pos: base + bytes.len() });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut offset = start;

    for part in content[start..].split(',') {
        let part_bytes = part.as_bytes();
        let spec_pos = base + offset;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let modifier_index = part_bytes
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(part_bytes.len());
        let name_bytes = &part_bytes[..modifier_index];

        if name_bytes.is_empty() {
            return Err(Error::InvalidName { pos: spec_pos });
        }
        if !valid_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_pos });
        }
        if let Some(i) = name_bytes.iter().enumerate().skip(1).find_map(|(i, &b)| {
            if name_char(b) { None } else { Some(i) }
        }) {
            return Err(Error::InvalidName { pos: spec_pos + i });
        }

        let name = std::str::from_utf8(name_bytes)
            .expect("ASCII name bytes are valid UTF-8")
            .to_owned();

        let modifier = if modifier_index == part_bytes.len() {
            Modifier::None
        } else {
            let suffix = &part_bytes[modifier_index..];
            let parsed = if suffix == b"*" {
                Some(Modifier::Explode)
            } else if suffix.first() == Some(&b':') {
                let digits = &suffix[1..];
                if (1..=4).contains(&digits.len())
                    && digits[0].is_ascii_digit()
                    && digits[0] != b'0'
                    && digits.iter().all(u8::is_ascii_digit)
                {
                    let mut n = 0usize;
                    for &d in digits {
                        n = n * 10 + (d - b'0') as usize;
                    }
                    (n <= 9999).then_some(Modifier::Prefix(n))
                } else {
                    None
                }
            } else {
                None
            };
            parsed.ok_or(Error::InvalidModifier {
                pos: spec_pos + modifier_index,
            })?
        };

        if seen.iter().any(|n| n == &name) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            modifier,
            pos: spec_pos,
        });
        offset += part.len() + 1;
    }

    Ok(Expression { op, specs })
}

fn reserved(c: u8) -> bool {
    matches!(
        c,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn encode(input: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h1), Some(h2)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(HEX[h1 as usize] as char);
                out.push(HEX[h2 as usize] as char);
                i += 3;
                continue;
            }
        }

        if b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || (allow_reserved && reserved(b))
        {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn prefix_chars(s: &str, n: usize) -> String {
    s.chars().take(n).collect()
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn named_field(name: &str, value: String, op: Operator) -> String {
    if op == Operator::Semi && value.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + value.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&value);
        field
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let allow_reserved = matches!(expr.op, Operator::Plus | Operator::Hash);
    let separator = match expr.op {
        Operator::Simple | Operator::Plus | Operator::Hash => ',',
        Operator::Dot => '.',
        Operator::Slash => '/',
        Operator::Semi => ';',
        Operator::Question | Operator::Amp => '&',
    };
    let mut fields = Vec::<String>::new();

    for spec in &expr.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        match (&spec.modifier, value) {
            (Modifier::Prefix(_), Value::List(_) | Value::Map(_)) => {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
            _ => {}
        }

        match value {
            Value::Scalar(s) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(n) => prefix_chars(s, n),
                    Modifier::None | Modifier::Explode => s.clone(),
                };
                let encoded = encode(&selected, allow_reserved);
                if matches!(expr.op, Operator::Semi | Operator::Question | Operator::Amp) {
                    fields.push(named_field(&spec.name, encoded, expr.op));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = spec.modifier == Modifier::Explode;
                if exploded {
                    for item in items {
                        let encoded = encode(item, allow_reserved);
                        if matches!(expr.op, Operator::Semi | Operator::Question | Operator::Amp) {
                            fields.push(named_field(&spec.name, encoded, expr.op));
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if matches!(expr.op, Operator::Semi | Operator::Question | Operator::Amp) {
                        fields.push(named_field(&spec.name, joined, expr.op));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|&seen| seen == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }
                if spec.modifier == Modifier::Explode {
                    for (key, val) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(val, allow_reserved);
                        if expr.op == Operator::Semi && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, allow_reserved));
                        components.push(encode(val, allow_reserved));
                    }
                    let joined = components.join(",");
                    if matches!(expr.op, Operator::Semi | Operator::Question | Operator::Amp) {
                        fields.push(named_field(&spec.name, joined, expr.op));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut result = String::new();
    match expr.op {
        Operator::Hash => result.push('#'),
        Operator::Dot => result.push('.'),
        Operator::Slash => result.push('/'),
        Operator::Semi => result.push(';'),
        Operator::Question => result.push('?'),
        Operator::Amp => result.push('&'),
        Operator::Simple | Operator::Plus => {}
    }
    result.push_str(&fields.join(&separator.to_string()));
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<(usize, usize, Expression)>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;

                if let Some(relative_open) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_open,
                    });
                }

                let content = &template[open + 1..close];
                let parsed = parse_expression(content, open + 1)?;
                expressions.push((open, close + 1, parsed));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut result = String::new();
    let mut cursor = 0;
    for (start, end, expr) in expressions {
        result.push_str(&template[cursor..start]);
        result.push_str(&expand_expression(&expr, vars)?);
        cursor = end;
    }
    result.push_str(&template[cursor..]);
    Ok(result)
}
