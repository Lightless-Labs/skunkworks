#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug)]
struct Specifier {
    name: String,
    modifier: Modifier,
    pos: usize,
}

#[derive(Debug)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug)]
struct Expression {
    operator: char,
    specs: Vec<Specifier>,
}

fn name_byte(byte: u8, first: bool) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_' || (!first && byte.is_ascii_digit())
}

fn parse_specifier(text: &str, start: usize) -> Result<Specifier, Error> {
    if text.is_empty() {
        return Err(Error::EmptyVariable { pos: start });
    }

    let bytes = text.as_bytes();
    let modifier_at = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(bytes.len());

    if name_end == 0 {
        return Err(Error::InvalidName { pos: start });
    }

    for i in 0..name_end {
        if !name_byte(bytes[i], i == 0) {
            return Err(Error::InvalidName { pos: start + i });
        }
    }

    let modifier = if let Some(i) = modifier_at {
        let suffix = &text[i..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let valid = (1..=4).contains(&digits.len())
                && digits.as_bytes()[0].is_ascii_digit()
                && digits.as_bytes()[0] != b'0'
                && digits.bytes().all(|b| b.is_ascii_digit());
            let value = if valid {
                digits.parse::<usize>().ok()
            } else {
                None
            };
            match value {
                Some(n) if (1..=9999).contains(&n) => Modifier::Prefix(n),
                _ => return Err(Error::InvalidModifier { pos: start + i }),
            }
        } else {
            return Err(Error::InvalidModifier { pos: start + i });
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name: text[..name_end].to_owned(),
        modifier,
        pos: start,
    })
}

fn parse_expression(content: &str, start: usize, close: usize) -> Result<Expression, Error> {
    let mut operator = '\0';
    let mut body_start = 0;
    if let Some(first) = content.chars().next() {
        if "+#./;?&".contains(first) {
            operator = first;
            body_start = first.len_utf8();
        } else if first.is_ascii_punctuation() {
            return Err(Error::UnsupportedOperator { pos: start });
        }
    }

    let body = &content[body_start..];
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut offset = body_start;
    for part in body.split(',') {
        let spec = parse_specifier(part, start + offset)?;
        if seen.contains(&spec.name) {
            return Err(Error::DuplicateVariable { pos: spec.pos });
        }
        seen.insert(spec.name.clone());
        specs.push(spec);
        offset += part.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn reserved(byte: u8) -> bool {
    b":/?#[]@!$&'()*+,;=".contains(&byte)
}

fn encode(input: &str, allow_reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || (allow_reserved && reserved(b))
        {
            out.push(b as char);
            i += 1;
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
            i += 1;
        }
    }
    out
}

fn prefix_chars(value: &str, count: usize) -> String {
    value.chars().take(count).collect()
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: char) -> String {
    if operator == ';' && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_spec(spec: &Specifier, value: &Value, operator: char) -> Result<Vec<String>, Error> {
    let allow_reserved = matches!(operator, '+' | '#');
    let explode = matches!(spec.modifier, Modifier::Explode);
    let scalar_value = match &spec.modifier {
        Modifier::Prefix(n) => match value {
            Value::Scalar(s) => Some(prefix_chars(s, *n)),
            Value::List(_) | Value::Map(_) => {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
        },
        _ => None,
    };

    match value {
        Value::Scalar(original) => {
            let raw = scalar_value.as_deref().unwrap_or(original);
            let encoded = encode(raw, allow_reserved);
            Ok(vec![if matches!(operator, ';' | '?' | '&') {
                named_field(&spec.name, encoded, operator)
            } else {
                encoded
            }])
        }
        Value::List(items) => {
            if items.is_empty() {
                return Ok(Vec::new());
            }
            let encoded: Vec<String> = items.iter().map(|s| encode(s, allow_reserved)).collect();
            if explode {
                Ok(encoded
                    .into_iter()
                    .map(|item| {
                        if matches!(operator, ';' | '?' | '&') {
                            named_field(&spec.name, item, operator)
                        } else {
                            item
                        }
                    })
                    .collect())
            } else {
                let joined = encoded.join(",");
                Ok(vec![if matches!(operator, ';' | '?' | '&') {
                    named_field(&spec.name, joined, operator)
                } else {
                    joined
                }])
            }
        }
        Value::Map(entries) => {
            let mut keys = std::collections::HashSet::new();
            for (key, _) in entries {
                if !keys.insert(key) {
                    return Err(Error::DuplicateKey { pos: spec.pos });
                }
            }
            if entries.is_empty() {
                return Ok(Vec::new());
            }
            let encoded: Vec<(String, String)> = entries
                .iter()
                .map(|(key, value)| {
                    (encode(key, allow_reserved), encode(value, allow_reserved))
                })
                .collect();
            if explode {
                Ok(encoded
                    .into_iter()
                    .map(|(key, value)| {
                        if operator == ';' && value.is_empty() {
                            key
                        } else {
                            format!("{key}={value}")
                        }
                    })
                    .collect())
            } else {
                let components: Vec<String> = encoded
                    .into_iter()
                    .flat_map(|(key, value)| [key, value])
                    .collect();
                let joined = components.join(",");
                Ok(vec![if matches!(operator, ';' | '?' | '&') {
                    named_field(&spec.name, joined, operator)
                } else {
                    joined
                }])
            }
        }
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::new();
    for spec in &expression.specs {
        if let Some(value) = selected_value(&spec.name, vars) {
            fields.extend(expand_spec(spec, value, expression.operator)?);
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match expression.operator {
        '.' => (".", "."),
        '/' => ("/", "/"),
        ';' => (";", ";"),
        '?' => ("&", "?"),
        '&' => ("&", "&"),
        _ => (",", ""),
    };
    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = None;
                let mut j = i + 1;
                while j < bytes.len() {
                    if bytes[j] == b'}' {
                        close = Some(j);
                        break;
                    }
                    j += 1;
                }
                let close = match close {
                    Some(pos) => pos,
                    None => return Err(Error::Unclosed { pos: open }),
                };
                if let Some(inner) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner,
                    });
                }
                let content = &template[open + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for (start, end, expression) in expressions {
        out.push_str(&template[cursor..start]);
        out.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
