pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }

    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: target,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    let mut has_self_loop = vec![false; n];

    for &(source, target) in edges {
        adjacency[source].push(target);
        reverse[target].push(source);
        if source == target {
            has_self_loop[source] = true;
        }
    }

    // Iterative DFS finishing order, avoiding recursion on long paths.
    let mut visited = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for start in 0..n {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.last_mut() {
            if *next_index < adjacency[*vertex].len() {
                let next = adjacency[*vertex][*next_index];
                *next_index += 1;
                if !visited[next] {
                    visited[next] = true;
                    stack.push((next, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                order.push(finished);
            }
        }
    }

    let mut assigned = vec![false; n];
    let mut components = Vec::new();
    let mut component_cyclic = Vec::new();

    for &start in order.iter().rev() {
        if assigned[start] {
            continue;
        }

        assigned[start] = true;
        let mut members = Vec::new();
        let mut stack = vec![start];
        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &next in &reverse[vertex] {
                if !assigned[next] {
                    assigned[next] = true;
                    stack.push(next);
                }
            }
        }

        members.sort_unstable();
        let is_cyclic = members.len() > 1 || has_self_loop[members[0]];
        components.push(members);
        component_cyclic.push(is_cyclic);
    }

    let mut permutation: Vec<usize> = (0..components.len()).collect();
    permutation.sort_unstable_by(|&a, &b| {
        components[b]
            .len()
            .cmp(&components[a].len())
            .then_with(|| components[a][0].cmp(&components[b][0]))
    });

    let mut sorted_components = Vec::with_capacity(components.len());
    let mut sorted_cyclic = Vec::with_capacity(components.len());
    let mut component_of = vec![0usize; n];

    for (canonical_id, old_id) in permutation.into_iter().enumerate() {
        let members = std::mem::take(&mut components[old_id]);
        for &vertex in &members {
            component_of[vertex] = canonical_id;
        }
        sorted_components.push(members);
        sorted_cyclic.push(component_cyclic[old_id]);
    }

    Ok(Graph {
        components: sorted_components,
        component_of,
        cyclic: sorted_cyclic,
    })
}
