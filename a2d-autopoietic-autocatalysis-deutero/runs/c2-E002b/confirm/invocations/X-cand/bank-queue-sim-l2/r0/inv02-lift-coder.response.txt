```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy: Option<(u64, u32)>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<&Customer> = customers.iter().collect();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|s| ServerState {
            id: s.id,
            opens: s.opens,
            overhead: s.overhead,
            opened: false,
            busy: None,
            idle_since: s.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut customer_pos = 0usize;
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut time = 0u64;

    loop {
        // Finish events, ordered by ascending server ID.
        let mut finished: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                state
                    .busy
                    .filter(|(finish, _)| *finish == time)
                    .map(|(_, customer)| (state.id, i, customer))
            })
            .collect();
        finished.sort_by_key(|entry| entry.0);

        for (_, index, customer) in finished {
            let server_id = states[index].id;
            events.push(Event::Finish {
                time,
                customer,
                server: server_id,
            });
            states[index].busy = None;
            states[index].idle_since = time;
        }

        // Open events, ordered by descending server ID.
        let mut opening: Vec<(u32, usize)> = states
            .iter()
            .enumerate()
            .filter(|(_, state)| !state.opened && state.opens == time)
            .map(|(i, state)| (state.id, i))
            .collect();
        opening.sort_by(|a, b| b.0.cmp(&a.0));

        for (server_id, index) in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time,
                server: server_id,
            });
        }

        // Arrival events are already ordered by arrival time and descending ID.
        while customer_pos < ordered_customers.len()
            && ordered_customers[customer_pos].arrival == time
        {
            let customer = ordered_customers[customer_pos];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer);
            customer_pos += 1;
        }

        // Dispatch in FIFO order, recalculating server ranking each time.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting.remove(0);
            let server_id = states[server_index].id;
            let finish = customer
                .service
                .checked_add(time)
                .and_then(|t| t.checked_add(states[server_index].overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
            } else {
                states[server_index].busy = Some((finish, customer.id));
            }
        }

        if customer_pos == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.busy.is_none())
            && states.iter().all(|state| state.opened)
        {
            break;
        }

        let next_arrival = ordered_customers
            .get(customer_pos)
            .map(|c| c.arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states
            .iter()
            .filter_map(|state| state.busy.map(|(finish, _)| finish))
            .min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&candidate| candidate > time)
            .min();

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(events)
}
```
