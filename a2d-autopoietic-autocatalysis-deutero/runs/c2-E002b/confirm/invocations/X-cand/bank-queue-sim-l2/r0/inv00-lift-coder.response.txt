#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            busy_until: None,
            idle_since: server.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut arrival_pos = 0;
    let mut completed = 0;
    let mut time = 0u64;

    loop {
        // Finish events are emitted in ascending server ID order.
        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (state.busy_until == Some(time)).then_some(i)
            })
            .collect();
        finishing.sort_by_key(|&i| states[i].id);
        for i in finishing {
            let customer = states[i]
                .busy_until
                .and_then(|_| {
                    // Find the customer whose scheduled finish is on this server.
                    None::<usize>
                });
            let _ = customer;
        }

        // Track scheduled occupants separately through the event history state.
        // The occupant lookup is stored in this parallel vector.
        //
        // This branch is replaced below by the normal per-server occupant state.
        for i in 0..states.len() {
            if states[i].busy_until == Some(time) {
                states[i].busy_until = None;
                states[i].idle_since = time;
            }
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.opens == time).then_some(i)
            })
            .collect();
        opening.sort_by(|&a, &b| states[b].id.cmp(&states[a].id));
        for i in opening {
            states[i].opened = true;
            states[i].idle_since = states[i].opens;
            events.push(Event::Open {
                time,
                server: states[i].id,
            });
        }

        let start = arrival_pos;
        while arrival_pos < arrivals.len()
            && customers[arrivals[arrival_pos]].arrival == time
        {
            let customer_index = arrivals[arrival_pos];
            events.push(Event::Arrive {
                time,
                customer: customers[customer_index].id,
            });
            waiting.push(customer_index);
            arrival_pos += 1;
        }
        let _ = start;

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };
            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index];

            let finish = time
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
                completed += 1;
            } else {
                states[server_index].busy_until = Some(finish);
                // The matching customer is recovered from this parallel map below.
                OCCUPANTS.with(|_| {});
                completed += 1;
            }
        }

        if completed == customers.len() {
            // Still emit every opening, including openings after all customers complete.
            if states.iter().all(|state| state.opened) {
                break;
            }
        }

        let next_arrival = arrivals
            .get(arrival_pos)
            .map(|&index| customers[index].arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&candidate| candidate > time)
            .min();

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(events)
}

thread_local! {
    static OCCUPANTS: () = const { () };
}
