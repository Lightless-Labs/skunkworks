#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_text(word: &mut Word, text: &str, active_glob: bool) {
    for c in text.chars() {
        append_scalar(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().expect("valid char boundary");
        let len = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                    i += len;
                } else {
                    append_scalar(&mut word, c, false);
                    i += len;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i += len;
                } else if c == '\\' {
                    let slash_pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let next = input[i..].chars().next().expect("valid char boundary");
                    let next_len = next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        append_scalar(&mut word, next, false);
                        active = true;
                        i += next_len;
                    } else if next == '\n' {
                        i += next_len;
                    } else {
                        append_scalar(&mut word, '\\', false);
                        active = true;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let dollar_pos = i;
                    let next_pos = i + len;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: dollar_pos });
                    }
                    if input[next_pos..].starts_with('{') {
                        let (end, name, default) = parse_braced(input, dollar_pos)?;
                        let value = match default {
                            Some(default_text) => match lookup(env, name) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(default_text),
                            },
                            None => lookup(env, name).unwrap_or("").to_string(),
                        };
                        append_text(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else if let Some(first) = input[next_pos..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_pos + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().expect("valid char boundary");
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_pos..end];
                            append_text(&mut word, lookup(env, name).unwrap_or(""), false);
                            active = true;
                            i = end;
                        } else {
                            append_scalar(&mut word, '$', false);
                            active = true;
                            i += len;
                        }
                    } else {
                        append_scalar(&mut word, '$', false);
                        active = true;
                        i += len;
                    }
                } else {
                    append_scalar(&mut word, c, false);
                    active = true;
                    i += len;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += len;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_pos = i;
                    active = true;
                    i += len;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_pos = i;
                    active = true;
                    i += len;
                } else if c == '\\' {
                    let slash_pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let next = input[i..].chars().next().expect("valid char boundary");
                    let next_len = next.len_utf8();
                    if next == '\n' {
                        i += next_len;
                    } else {
                        append_scalar(&mut word, next, false);
                        active = true;
                        i += next_len;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let dollar_pos = i;
                    let next_pos = i + len;
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: dollar_pos });
                    }
                    if input[next_pos..].starts_with('{') {
                        let (end, name, default) = parse_braced(input, dollar_pos)?;
                        let value = match default {
                            Some(default_text) => match lookup(env, name) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(default_text),
                            },
                            None => lookup(env, name).unwrap_or("").to_string(),
                        };
                        append_text(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else if let Some(first) = input[next_pos..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_pos + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().expect("valid char boundary");
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_pos..end];
                            append_text(&mut word, lookup(env, name).unwrap_or(""), true);
                            active = true;
                            i = end;
                        } else {
                            append_scalar(&mut word, '$', true);
                            active = true;
                            i += len;
                        }
                    } else {
                        append_scalar(&mut word, '$', true);
                        active = true;
                        i += len;
                    }
                } else {
                    append_scalar(&mut word, c, true);
                    active = true;
                    i += len;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_braced<'a>(
    input: &'a str,
    dollar_pos: usize,
) -> Result<(usize, &'a str, Option<&'a str>), SplitError> {
    let content_start = dollar_pos + 2;
    let mut i = content_start;
    let mut closing = None;

    while i < input.len() {
        let c = input[i..].chars().next().expect("valid char boundary");
        if c == '\\' {
            i += 1;
            if i < input.len() {
                let next = input[i..].chars().next().expect("valid char boundary");
                i += next.len_utf8();
            }
        } else if c == '}' {
            closing = Some(i);
            break;
        } else {
            i += c.len_utf8();
        }
    }

    let close_pos = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let contents = &input[content_start..close_pos];

    let name_end = contents
        .char_indices()
        .find_map(|(offset, c)| (!is_name_continue(c)).then_some(offset))
        .unwrap_or(contents.len());

    if name_end == 0 || !is_name_start(contents.chars().next().unwrap()) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name = &contents[..name_end];
    if !name.chars().all(is_name_continue) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let remainder = &contents[name_end..];
    let default = if remainder.is_empty() {
        None
    } else if let Some(text) = remainder.strip_prefix(":-") {
        Some(text)
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((close_pos + 1, name, default))
}
