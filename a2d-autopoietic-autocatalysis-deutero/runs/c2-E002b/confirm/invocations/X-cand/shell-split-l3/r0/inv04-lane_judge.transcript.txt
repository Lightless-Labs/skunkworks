fn parse_braced<'a>(
    input: &'a str,
    dollar_pos: usize,
) -> Result<(usize, &'a str, Option<&'a str>), SplitError> {
    let content_start = dollar_pos + 2;
    let mut i = content_start;
    let mut closing = None;

    while i < input.len() {
        let c = input[i..].chars().next().expect("valid char boundary");
        if c == '\\' {
            i += 1;
            if i < input.len() {
                let next = input[i..].chars().next().expect("valid char boundary");
                i += next.len_utf8();
            }
        } else if c == '}' {
            closing = Some(i);
            break;
        } else {
            i += c.len_utf8();
        }
    }

    let close_pos = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let contents = &input[content_start..close_pos];

    let name_end = contents
        .char_indices()
        .find_map(|(offset, c)| (!is_name_continue(c)).then_some(offset))
        .unwrap_or(contents.len());

    if name_end == 0 || !is_name_start(contents.chars().next().unwrap()) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name = &contents[..name_end];
    if !name.chars().all(is_name_continue) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let remainder = &contents[name_end..];
    let default = if remainder.is_empty() {
        None
    } else if let Some(text) = remainder.strip_prefix(":-") {
        Some(text)
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((close_pos + 1, name, default))
}

fn first_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn braced_expansion(
    input: &str,
    chars: &[(usize, char)],
    start: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let dollar_pos = chars[start].0;
    let mut close = None;
    let mut i = start + 2;

    while i < chars.len() {
        if chars[i].1 == '\\' {
            i += 2;
        } else if chars[i].1 == '}' {
            close = Some(i);
            break;
        } else {
            i += 1;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let content_start = chars[start + 2].0;
    let content_end = chars[close].0;
    let content = &input[content_start..content_end];

    let content_chars: Vec<(usize, char)> = content.char_indices().collect();
    let name_len = content_chars
        .iter()
        .take_while(|(_, c)| is_name_continue(*c))
        .count();

    if name_len == 0 || !is_name_start(content_chars[0].1) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name_end = if name_len < content_chars.len() {
        content_chars[name_len].0
    } else {
        content.len()
    };
    let name = &content[..name_end];

    let selected = if name_end == content.len() {
        first_env(env, name).unwrap_or("").to_string()
    } else if content[name_end..].starts_with(":-") {
        let default_start = name_end + 2;
        match first_env(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => {
                let default = &content[default_start..];
                let mut decoded = String::new();
                let mut it = default.chars();
                while let Some(c) = it.next() {
                    if c == '\\' {
                        if let Some(next) = it.next() {
                            decoded.push(next);
                        } else {
                            decoded.push('\\');
                        }
                    } else {
                        decoded.push(c);
                    }
                }
                decoded
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((close + 1, selected))
}

                        }
                        i = end;
                    } else {
                        push_scalar(&mut word, '$', false);
                        i += 1;
                    }
                } else {
                    push_scalar(&mut word, c, false);
                    i += 1;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += 1;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_start = pos;
                    i += 1;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_start = pos;
                    i += 1;
                } else if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if next == '\n' {
                        i += 2;
                    } else {
                        active = true;
                        push_scalar(&mut word, next, false);
                        i += 2;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len() && chars[i + 1].1 == '{' {
                        active = true;
                        let (next_i, value) = braced_expansion(input, &chars, i, env)?;
                        push_string(&mut word, &value, true);
                        i = next_i;
                    } else if i + 1 < chars.len() && is_name_start(chars[i + 1].1) {
                        active = true;
                        let begin = i + 1;
                        let mut end = begin + 1;
                        while end < chars.len() && is_name_continue(chars[end].1) {
                            end += 1;
                        }
                        let name = &input[chars[begin].0..if end < chars.len() {
                            chars[end].0
                        } else {
                            input.len()
                        }];
                        if let Some(value) = first_env(env, name) {
                            push_string(&mut word, value, true);
                        }
                        i = end;
                    } else {
                        active = true;
                        push_scalar(&mut word, '$', true);
                        i += 1;
                    }
                } else {
                    active = true;
                    push_scalar(&mut word, c, true);
                    i += 1;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expansion_end(input: &str, start: usize) -> Result<usize, SplitError> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next().unwrap();
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Ok(i);
        } else {
            i += c.len_utf8();
        }
    }
    Err(SplitError::UnterminatedExpansion { pos: start - 2 })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();

    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                result.push(next);
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))

fn split_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_open = dollar + 2;
    let mut cursor = after_open;
    let mut closing = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if c == '}' {
            closing = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, c)| (!is_name_continue(c)).then_some(i))
        .unwrap_or(contents.len());

    if name_end == 0 || !contents[..name_end].chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let remainder = &contents[name_end..];

    let value = if remainder.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        if !default.is_empty() || remainder == ":-" {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut cursor = 0;

    while cursor < text.len() {
        let c = text[cursor..].chars().next().unwrap();
        cursor += c.len_utf8();
        if c == '\\' && cursor < text.len() {
            let next = text[cursor..].chars().next().unwrap();
            cursor += next.len_utf8();
            result.push(next);
        } else {
            result.push(c);
        }
    }

    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let pos = cursor;
        cursor += c.len_utf8();

        match quote {
            Quote::Single(opening) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append(&mut current, &c.to_string(), false);
                }
                let _ = opening;
            }
            Quote::Double(opening) => match c {
                '"' => quote = Quote::None,
                '\\' => {
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        cursor += next.len_utf8();
                        if next != '\n' {
                            append(&mut current, &next.to_string(), false);
                        }
                    } else {
                        append(&mut current, "\\", false);
                    }
                }
                '`' => return Err(SplitError::CommandSubstitution { pos }),
                '$' => {
                    if input[cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if input[cursor..].starts_with('{') {
                        let (next_cursor, value) = split_braced(input, pos, env)?;
                        cursor = next_cursor;
                        append(&mut current, &value, false);
                        active = true;
                    } else if cursor < input.len()
                        && input[cursor..].chars().next().is_some_and(is_name_start)
                    {
                        let start = cursor;
                        while cursor < input.len()
                            && input[cursor..]

                                .next()
                                .is_some_and(is_name_continue)

                            cursor += input[cursor..].chars().next().unwrap().len_utf8();
                        }
                        let name = &input[start..cursor];
                        append(&mut current, lookup(name, env).unwrap_or(""), false);
                        active = true;
                    } else {
                        append(&mut current, "$", false);
                    }
                }
                _ => append(&mut current, &c.to_string(), false),
            },
            Quote::None => match c {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                }
                '\'' => {
                    quote = Quote::Single(pos);
                    active = true;
                }

                    quote = Quote::Double(pos);
                    active = true;
                }
                '\\' => {
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    cursor += next.len_utf8();
                    if next != '\n' {
                        append(&mut current, &next.to_string(), false);
                        active = true;
                    }
                }
                '`' => return Err(SplitError::CommandSubstitution { pos }),
                '$' => {
                    if input[cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if input[cursor..].starts_with('{') {
                        let (next_cursor, value) = split_braced(input, pos, env)?;
                        cursor = next_cursor;
                        append(&mut current, &value, true);
                        active = true;
                    } else if cursor < input.len()
                        && input[cursor..].chars().next().is_some_and(is_name_start)

                        let start = cursor;

                            && input[cursor..]

                        }

                        append(&mut current, lookup(name, env).unwrap_or(""), true);
                        active = true;
                    } else {
                        append(&mut current, "$", true);
                        active = true;
                    }
                }
                _ => {
                    append(&mut current, &c.to_string(), matches!(c, '*' | '?' | '['));
                    active = true;
                }
            },
        }

        if let Quote::Single(opening) = quote {
            let _ = opening;
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

        words.push(current);
    }

}
