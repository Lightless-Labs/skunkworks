#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn push_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn push_string(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        push_scalar(word, c, active_glob);
    }
}

fn first_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn braced_expansion(
    input: &str,
    chars: &[(usize, char)],
    start: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let dollar_pos = chars[start].0;
    let mut close = None;
    let mut i = start + 2;

    while i < chars.len() {
        if chars[i].1 == '\\' {
            i += 2;
        } else if chars[i].1 == '}' {
            close = Some(i);
            break;
        } else {
            i += 1;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let content_start = chars[start + 2].0;
    let content_end = chars[close].0;
    let content = &input[content_start..content_end];

    let content_chars: Vec<(usize, char)> = content.char_indices().collect();
    let name_len = content_chars
        .iter()
        .take_while(|(_, c)| is_name_continue(*c))
        .count();

    if name_len == 0 || !is_name_start(content_chars[0].1) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name_end = if name_len < content_chars.len() {
        content_chars[name_len].0
    } else {
        content.len()
    };
    let name = &content[..name_end];

    let selected = if name_end == content.len() {
        first_env(env, name).unwrap_or("").to_string()
    } else if content[name_end..].starts_with(":-") {
        let default_start = name_end + 2;
        match first_env(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => {
                let default = &content[default_start..];
                let mut decoded = String::new();
                let mut it = default.chars();
                while let Some(c) = it.next() {
                    if c == '\\' {
                        if let Some(next) = it.next() {
                            decoded.push(next);
                        } else {
                            decoded.push('\\');
                        }
                    } else {
                        decoded.push(c);
                    }
                }
                decoded
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((close + 1, selected))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < chars.len() {
        let (pos, c) = chars[i];

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    push_scalar(&mut word, c, false);
                }
                i += 1;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i += 1;
                } else if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            push_scalar(&mut word, next, false);
                        }
                        i += 2;
                    } else {
                        push_scalar(&mut word, '\\', false);
                        i += 1;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len() && chars[i + 1].1 == '{' {
                        active = true;
                        let (next_i, value) = braced_expansion(input, &chars, i, env)?;
                        push_string(&mut word, &value, false);
                        i = next_i;
                    } else if i + 1 < chars.len() && is_name_start(chars[i + 1].1) {
                        active = true;
                        let begin = i + 1;
                        let mut end = begin + 1;
                        while end < chars.len() && is_name_continue(chars[end].1) {
                            end += 1;
                        }
                        let name = &input[chars[begin].0..if end < chars.len() {
                            chars[end].0
                        } else {
                            input.len()
                        }];
                        if let Some(value) = first_env(env, name) {
                            push_string(&mut word, value, false);
                        }
                        i = end;
                    } else {
                        push_scalar(&mut word, '$', false);
                        i += 1;
                    }
                } else {
                    push_scalar(&mut word, c, false);
                    i += 1;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += 1;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_start = pos;
                    i += 1;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_start = pos;
                    i += 1;
                } else if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if next == '\n' {
                        i += 2;
                    } else {
                        active = true;
                        push_scalar(&mut word, next, false);
                        i += 2;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len() && chars[i + 1].1 == '{' {
                        active = true;
                        let (next_i, value) = braced_expansion(input, &chars, i, env)?;
                        push_string(&mut word, &value, true);
                        i = next_i;
                    } else if i + 1 < chars.len() && is_name_start(chars[i + 1].1) {
                        active = true;
                        let begin = i + 1;
                        let mut end = begin + 1;
                        while end < chars.len() && is_name_continue(chars[end].1) {
                            end += 1;
                        }
                        let name = &input[chars[begin].0..if end < chars.len() {
                            chars[end].0
                        } else {
                            input.len()
                        }];
                        if let Some(value) = first_env(env, name) {
                            push_string(&mut word, value, true);
                        }
                        i = end;
                    } else {
                        active = true;
                        push_scalar(&mut word, '$', true);
                        i += 1;
                    }
                } else {
                    active = true;
                    push_scalar(&mut word, c, true);
                    i += 1;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
