#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut index = start;
    let mut record_index = 0usize;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        if !matches!(state, State::Quoted { .. }) && byte == b'\r' {
            if bytes.get(index + 1) == Some(&b'\n') {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                index += 2;
                state = State::FieldStart;
                continue;
            }
            return Err(CsvError::BareCr {
                record: record_index,
                byte: index,
            });
        }

        match state {
            State::FieldStart => match byte {
                b'"' => {
                    state = State::Quoted { opening: index };
                    index += 1;
                }
                b',' => {
                    record.push(String::new());
                    index += 1;
                }
                b'\n' => {
                    record.push(String::new());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    index += 1;
                    state = State::FieldStart;
                }
                _ => {
                    let character = input[index..].chars().next().unwrap();
                    field.push(character);
                    index += character.len_utf8();
                    state = State::Unquoted;
                }
            },
            State::Unquoted => match byte {
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: index,
                    });
                }
                b',' => {
                    record.push(std::mem::take(&mut field));
                    index += 1;
                    state = State::FieldStart;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    index += 1;
                    state = State::FieldStart;
                }
                _ => {
                    let character = input[index..].chars().next().unwrap();
                    field.push(character);
                    index += character.len_utf8();
                }
            },
            State::Quoted { opening } => {
                if byte == b'"' {
                    if bytes.get(index + 1) == Some(&b'"') {
                        field.push('"');
                        index += 2;
                    } else {
                        state = State::AfterQuote;
                        index += 1;
                    }
                } else {
                    let character = input[index..].chars().next().unwrap();
                    field.push(character);
                    index += character.len_utf8();
                    state = State::Quoted { opening };
                }
            }
            State::AfterQuote => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    index += 1;
                    state = State::FieldStart;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    index += 1;
                    state = State::FieldStart;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            },
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::FieldStart if !record.is_empty() => {
            record.push(String::new());
            records.push(record);
        }
        State::FieldStart => {}
        State::Unquoted | State::AfterQuote => {
            record.push(field);
            records.push(record);
        }
    }

    Ok(records)
}
