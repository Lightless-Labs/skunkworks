#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut now = 0u64;
    let mut occupied = false;
    let mut scheduled_finish = None;
    let mut previous_finish = None;
    let mut dispatched_before = false;

    loop {
        if let Some(finish) = scheduled_finish {
            if finish <= now {
                let customer_id = match events.last().map(|event: &Event| &event.kind) {
                    Some(Kind::Start { .. }) => events.last().unwrap().customer,
                    _ => unreachable!(),
                };
                events.push(Event {
                    time: finish,
                    customer: customer_id,
                    kind: Kind::Finish,
                });
                occupied = false;
                scheduled_finish = None;
                previous_finish = Some(finish);
                now = finish;
            }
        }

        if next_arrival < arrivals.len() && arrivals[next_arrival].arrival <= now {
            let tick = arrivals[next_arrival].arrival;
            now = tick;

            if let Some(finish) = scheduled_finish {
                if finish == tick {
                    let customer_id = match events.last().map(|event: &Event| &event.kind) {
                        Some(Kind::Start { .. }) => events.last().unwrap().customer,
                        _ => unreachable!(),
                    };
                    events.push(Event {
                        time: finish,
                        customer: customer_id,
                        kind: Kind::Finish,
                    });
                    occupied = false;
                    scheduled_finish = None;
                    previous_finish = Some(finish);
                }
            }

            while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == tick {
                let customer = arrivals[next_arrival];
                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Arrive,
                });
                waiting.push_back(customer);
                next_arrival += 1;
            }
        }

        if !occupied {
            if let Some(customer) = waiting.pop_front() {
                let wake = !dispatched_before || previous_finish.map_or(true, |finish| now > finish);
                let base_finish = now.checked_add(customer.service).ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time: now,
                })?;
                let finish = if wake {
                    base_finish.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    })?
                } else {
                    base_finish
                };

                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });
                dispatched_before = true;
                occupied = true;
                scheduled_finish = Some(finish);

                if finish == now {
                    events.push(Event {
                        time: finish,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    occupied = false;
                    scheduled_finish = None;
                    previous_finish = Some(finish);
                }
                continue;
            }
        }

        if next_arrival == arrivals.len() && !occupied && waiting.is_empty() {
            break;
        }

        if occupied {
            let finish = scheduled_finish.unwrap();
            if next_arrival < arrivals.len() && arrivals[next_arrival].arrival < finish {
                now = arrivals[next_arrival].arrival;
            } else {
                now = finish;
            }
        } else if next_arrival < arrivals.len() {
            now = arrivals[next_arrival].arrival;
        }
    }

    Ok(events)
}
