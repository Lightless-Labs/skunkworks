#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_value(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_value(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).expect("decoded segment is ASCII"))
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = input.ends_with('/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(end);
        if stop == start {
            return Err(invalid(start));
        }

        let raw = &input[start..stop];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(invalid(start));
            }
            if !names.insert(name.to_owned()) {
                return Err(invalid(start));
            }
            if marker == b'*' && stop != end {
                return Err(invalid(start));
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            match decode_segment(raw, start) {
                Ok(decoded) => segments.push(Segment::Static(decoded)),
                Err(at) => return Err(invalid(at)),
            }
        }
        start = stop + 1;
    }

    Ok((segments, trailing))
}

fn percent_encode(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_path(pattern, true)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let id = self.routes.len();
        let mut rank = Vec::new();
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => rank.push(0),
            }
        }
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        if !has_wildcard {
            rank.push(3);
        }

        for route in &self.routes {
            if route.method != method
                || (self.policy == SlashPolicy::Strict && route.trailing != trailing)
                || route.segments.len() != segments.len()
            {
                continue;
            }
            let same_shape = route.segments.iter().zip(&segments).all(|(a, b)| match (a, b) {
                (Segment::Static(x), Segment::Static(y)) => x == y,
                (Segment::Param(_), Segment::Param(_)) => true,
                (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                _ => false,
            });
            if same_shape {
                return Err(Error::Conflict { with: route.id });
            }
        }

        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head, false)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            let wildcard_index = route
                .segments
                .iter()
                .position(|s| matches!(s, Segment::Wildcard(_)));
            let fixed_len = wildcard_index.unwrap_or(route.segments.len());
            if request_segments.len() < fixed_len {
                continue;
            }
            if wildcard_index.is_none() && request_segments.len() != route.segments.len() {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;
            for (i, pattern_segment) in route.segments.iter().enumerate().take(fixed_len) {
                match (pattern_segment, &request_segments[i]) {
                    (Segment::Static(a), Segment::Static(b)) if a == b => {}
                    (Segment::Param(name), Segment::Static(value)) => {
                        params.push((name.clone(), value.clone()));
                    }
                    _ => {
                        matches = false;
                        break;
                    }
                }
            }
            if !matches {
                continue;
            }
            if let Some(wi) = wildcard_index {
                let Segment::Wildcard(name) = &route.segments[wi] else {
                    unreachable!()
                };
                let value = request_segments[wi..]
                    .iter()
                    .map(|s| match s {
                        Segment::Static(v) => v.as_str(),
                        _ => unreachable!(),
                    })
                    .collect::<Vec<_>>()
                    .join("/");
                params.push((name.clone(), value));
            }
            candidates.push((route, params));
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<_> = if method == "HEAD" {
            if candidates.iter().any(|(route, _)| route.method == "HEAD") {
                candidates
                    .into_iter()
                    .filter(|(route, _)| route.method == "HEAD")
                    .collect()
            } else {
                candidates
                    .into_iter()
                    .filter(|(route, _)| route.method == "GET")
                    .collect()
            }
        } else {
            candidates
                .into_iter()
                .filter(|(route, _)| route.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = self
                .routes
                .iter()
                .filter(|route| {
                    // Determine path candidacy independently of method.
                    let wi = route
                        .segments
                        .iter()
                        .position(|s| matches!(s, Segment::Wildcard(_)));
                    let fixed = wi.unwrap_or(route.segments.len());
                    request_segments.len() >= fixed
                        && (wi.is_some() || request_segments.len() == route.segments.len())
                        && (self.policy != SlashPolicy::Strict
                            || route.trailing == request_trailing)
                        && route.segments.iter().take(fixed).enumerate().all(|(i, s)| {
                            match (s, &request_segments[i]) {
                                (Segment::Static(a), Segment::Static(b)) => a == b,
                                (Segment::Param(_), Segment::Static(_)) => true,
                                _ => false,
                            }
                        })
                })
                .map(|route| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (route, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id)))
            .expect("eligible candidates are nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing != request_trailing
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for &(key, value) in params {
            if supplied.contains_key(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_owned(),
                });
            }
            supplied.insert(key, value);
        }

        let mut captures = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(key) | Segment::Wildcard(key) => captures.push(key),
                Segment::Static(_) => {}
            }
        }

        for key in &captures {
            if !supplied.contains_key(key.as_str()) {
                return Err(Error::MissingParam {
                    name: (*key).clone(),
                });
            }
        }

        let mut extra: Vec<&str> = supplied
            .keys()
            .filter(|key| !captures.iter().any(|capture| capture.as_str() == **key))
            .map(|key| key.as_str())
            .collect();
        extra.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extra.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let (key, wildcard) = match segment {
                Segment::Param(key) => (Some(key), false),
                Segment::Wildcard(key) => (Some(key), true),
                Segment::Static(_) => (None, false),
            };
            if let Some(key) = key {
                let value = supplied[key.as_str()];
                let valid_ascii = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
                let valid = if !valid_ascii {
                    false
                } else if wildcard {
                    value.is_empty()
                        || value
                            .split('/')
                            .all(|part| !part.is_empty())
                } else {
                    !value.is_empty() && !value.contains('/')
                };
                if !valid {
                    return Err(Error::InvalidValue { name: key.clone() });
                }
            }
        }

        let mut out = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if !first {
                out.push('/');
            }
            first = false;
            match segment {
                Segment::Static(value) => out.push_str(&percent_encode(value)),
                Segment::Param(key) => out.push_str(&percent_encode(supplied[key.as_str()])),
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    if !value.is_empty() {
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                out.push('/');
                            }
                            out.push_str(&percent_encode(part));
                        }
                    }
                }
            }
        }

        if route.trailing && out != "/" {
            out.push('/');
        }
        Ok(out)
    }
}
