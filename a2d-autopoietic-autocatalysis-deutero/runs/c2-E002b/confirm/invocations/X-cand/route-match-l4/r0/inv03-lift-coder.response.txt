#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method == method
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
                && same_shape(&route.segments, &segments)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        let rank = make_rank(&segments);
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (head, _) = path.split_once(['?', '#']).unwrap_or((path, ""));
        let parsed = parse_path(head)?;
        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();

        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &parsed.segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };

        let mut eligible: Vec<(&Route, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(r, _)| r.method == eligible_method)
            .map(|(r, p)| (*r, p.clone()))
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> =
                candidates.iter().map(|(r, _)| r.method.clone()).collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            rank_cmp(&b.rank, &a.rank).then_with(|| a.id.cmp(&b.id))
        });
        let (winner, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && winner.trailing != parsed.trailing
            && !parsed.segments.is_empty()
        {
            let mut value = head.to_string();
            if winner.trailing {
                value.push('/');
            } else if value.ends_with('/') {
                value.pop();
            }
            Some(value)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|r| r.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(n) | Segment::Wildcard(n) => required.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|k| !required.contains(k))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(n) => {
                    let value = supplied[n.as_str()];
                    if !valid_param_value(value) {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Wildcard(n) => {
                    let value = supplied[n.as_str()];
                    if !valid_wildcard_value(value) {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        if route.segments.is_empty() {
            return Ok("/".to_string());
        }

        let mut output = String::new();
        for segment in &route.segments {
            output.push('/');
            match segment {
                Segment::Static(s) => output.push_str(&encode_component(s)),
                Segment::Param(n) => output.push_str(&encode_component(supplied[n.as_str()])),
                Segment::Wildcard(n) => {
                    let value = supplied[n.as_str()];
                    if !value.is_empty() {
                        for (i, component) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(component));
                        }
                    } else {
                        output.pop();
                    }
                }
            }
        }
        if route.trailing && !output.is_empty() && !output.ends_with('/') {
            output.push('/');
        }
        if output.is_empty() {
            output.push('/');
        }
        Ok(output)
    }
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let (parts, trailing) = split_parts(pattern);
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();

    for (i, (part, start)) in parts.iter().enumerate() {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }
        if part.starts_with(':') || part.starts_with('*') {
            let marker = part.as_bytes()[0];
            let name = &part[1..];
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b'*' && i + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            });
        } else {
            segments.push(Segment::Static(decode_segment(part, *start, true)?));
        }
    }

    Ok((segments, trailing))
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let (parts, trailing) = split_parts(path);
    let mut segments = Vec::new();
    for (part, start) in parts {
        if part.is_empty() {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(part, start, false)?);
    }

    Ok(ParsedPath { segments, trailing })
}

fn split_parts(path: &str) -> (Vec<(&str, usize)>, bool) {
    let trailing = path.ends_with('/');
    let effective_end = if trailing { path.len() - 1 } else { path.len() };
    let mut parts = Vec::new();
    let bytes = path.as_bytes();
    let mut start = 1;
    let mut i = 1;
    while i <= effective_end {
        if i == effective_end || bytes[i] == b'/' {
            parts.push((&path[start..i], start));
            start = i + 1;
        }
        i += 1;
    }
    (parts, trailing)
}

fn decode_segment(segment: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            let Some(hi) = hi else {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            };
            let Some(lo) = lo else {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            };
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: offset + i }
            } else {
                Error::InvalidPath { at: offset + i }
            });
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    if a.len() != b.len() {
        return false;
    }
    a.iter().zip(b).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn rank_cmp(a: &[u8], b: &[u8]) -> std::cmp::Ordering {
    a.cmp(b)
}

fn match_segments(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let wildcard_index = pattern.iter().position(|s| matches!(s, Segment::Wildcard(_)));
    let prefix_len = wildcard_index.unwrap_or(pattern.len());

    if wildcard_index.is_none() && pattern.len() != path.len() {
        return None;
    }
    if wildcard_index.is_some() && path.len() < prefix_len {
        return None;
    }

    for i in 0..prefix_len {
        match &pattern[i] {
            Segment::Static(s) if s != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(n) => params.push((n.clone(), path[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }
    if let Some(i) = wildcard_index {
        let Segment::Wildcard(name) = &pattern[i] else {
            unreachable!()
        };
        params.push((name.clone(), path[i..].join("/")));
    }
    Some(params)
}

fn valid_param_value(value: &str) -> bool {
    !value.is_empty()
        && value
            .bytes()
            .all(|b| (0x20..=0x7e).contains(&b) && b != b'/')
}

fn valid_wildcard_value(value: &str) -> bool {
    if value.is_empty() {
        return true;
    }
    value
        .split('/')
        .all(|part| !part.is_empty() && part.bytes().all(|b| (0x20..=0x7e).contains(&b)))
}

fn encode_component(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}
