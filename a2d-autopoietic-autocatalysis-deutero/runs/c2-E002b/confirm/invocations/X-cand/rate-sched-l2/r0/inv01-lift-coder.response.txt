#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct FieldParseError(CronError);

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(&bytes[..split], 0, 60, &mut minutes)?;
        parse_field(&bytes[split + 1..], split + 1, 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn syntax_at(offset: usize) -> FieldParseError {
    FieldParseError(CronError::Syntax(offset))
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    max: usize,
) -> Result<usize, FieldParseError> {
    if start == end {
        return Err(syntax_at(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(syntax_at(i));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(FieldParseError(CronError::Number(start)));
    }

    let mut value = 0usize;
    for &b in &bytes[start..end] {
        let digit = (b - b'0') as usize;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => return Err(FieldParseError(CronError::Bounds(start))),
        };
    }

    if value > max {
        return Err(FieldParseError(CronError::Bounds(start)));
    }
    Ok(value)
}

fn parse_field(
    field: &[u8],
    base: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut item_start = 0;

    loop {
        let item_end = field[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(field.len());

        parse_item(field, base, item_start, item_end, max, selected)
            .map_err(|e| e.0)?;

        if item_end == field.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

fn parse_item(
    field: &[u8],
    base: usize,
    item_start: usize,
    item_end: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), FieldParseError> {
    if item_start == item_end {
        return Err(syntax_at(base + item_start));
    }

    let item = &field[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(syntax_at(base + item_start + slash_positions[1]));
    }

    let slash = slash_positions.first().copied();
    let base_end = slash.unwrap_or(item.len());

    let is_star = base_end == 1 && item[0] == b'*';
    let first_dash = item[..base_end].iter().position(|&b| b == b'-');

    let (start, end, dash_offset) = if is_star {
        (0, max, None)
    } else if let Some(dash) = first_dash {
        let left = parse_number(
            item,
            0,
            dash,
            max,
        )?;
        let right = parse_number(
            item,
            dash + 1,
            base_end,
            max,
        )?;
        if left > right {
            return Err(FieldParseError(CronError::Descending(
                base + item_start + dash,
            )));
        }
        (left, right, Some(dash))
    } else {
        if slash.is_some() {
            return Err(syntax_at(base + item_start + slash.unwrap()));
        }
        let value = parse_number(item, 0, base_end, max)?;
        (value, value, None)
    };

    let step = if let Some(slash_pos) = slash {
        let allowed = is_star || dash_offset.is_some();
        if !allowed {
            return Err(syntax_at(base + item_start + slash_pos));
        }

        let step_start = slash_pos + 1;
        if step_start == item.len() {
            return Err(syntax_at(base + item_start + step_start));
        }

        let parsed = parse_number(item, step_start, item.len(), usize::MAX)
            .map_err(|e| match e.0 {
                CronError::Bounds(_) => FieldParseError(CronError::Step(
                    base + item_start + step_start,
                )),
                other => FieldParseError(other),
            })?;

        let step_max = if max == 59 { 60 } else { 24 };
        if parsed == 0 || parsed > step_max {
            return Err(FieldParseError(CronError::Step(
                base + item_start + step_start,
            )));
        }
        parsed
    } else {
        1
    };

    let mut value = start;
    loop {
        selected[value] = true;
        if end - value < step {
            break;
        }
        value += step;
    }

    Ok(())
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let now_is_boundary = now % 60_000 == 0;

        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if first_minute <= last_minute {
            for job in &self.jobs {
                match job.policy {
                    Policy::All => {
                        let mut minute = first_minute;
                        while minute <= last_minute {
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                            if minute == u64::MAX {
                                break;
                            }
                            minute += 1;
                        }
                    }
                    Policy::Once => {
                        let mut minute = last_minute;
                        loop {
                            if minute < first_minute {
                                break;
                            }
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                                break;
                            }
                            if minute == 0 {
                                break;
                            }
                            minute -= 1;
                        }
                    }
                    Policy::Skip => {
                        if now_is_boundary
                            && last_minute >= first_minute
                            && job.cron.matches(last_minute)
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}

impl Default for Calendar {
    fn default() -> Self {
        Self::new()
    }
}
