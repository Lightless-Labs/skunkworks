#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

#[derive(Clone)]
struct Field {
    selected: Vec<bool>,
}

pub struct Cron {
    minutes: Field,
    hours: Field,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let minutes = parse_field(&bytes[..split], 0, 59)?;
        let hours = parse_field(&bytes[split + 1..], split + 1, 23)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes.selected[(minute % 60) as usize]
            && self.hours.selected[((minute / 60) % 24) as usize]
    }
}

fn parse_field(bytes: &[u8], offset: usize, max: usize) -> Result<Field, CronError> {
    let mut selected = vec![false; max + 1];
    let mut start = 0;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }

        parse_item(&bytes[start..end], offset + start, max, &mut selected)?;

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(Field { selected })
}

fn parse_item(
    item: &[u8],
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(CronError::Syntax(offset + slash_positions[1]));
    }

    let slash = slash_positions.first().copied();
    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];

    if let Some(slash_at) = slash {
        if base != b"*" && !base.contains(&b'-') {
            return Err(CronError::Syntax(offset + slash_at));
        }
    }

    let (range_start, range_end, dash_at) = if base == b"*" {
        (0, max, None)
    } else if let Some(dash) = base.iter().position(|&b| b == b'-') {
        let left = parse_number(&base[..dash], offset, max)?;
        let right = parse_number(&base[dash + 1..], offset + dash + 1, max)?;
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        (left, right, Some(dash))
    } else {
        let n = parse_number(base, offset, max)?;
        (n, n, None)
    };

    let step = if let Some(slash_at) = slash {
        let step_start = slash_at + 1;
        let step_value = parse_number(&item[step_start..], offset + step_start, usize::MAX)
            .map_err(|err| match err {
                CronError::Bounds(_) => CronError::Step(offset + step_start),
                other => other,
            })?;
        let limit = if max == 59 { 60 } else { 24 };
        if step_value == 0 || step_value > limit {
            return Err(CronError::Step(offset + step_start));
        }
        step_value
    } else {
        1
    };

    let _ = dash_at;
    let mut value = range_start;
    while value <= range_end {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_number(fragment: &[u8], offset: usize, max: usize) -> Result<usize, CronError> {
    if fragment.is_empty() {
        return Err(CronError::Syntax(offset));
    }

    for (i, &b) in fragment.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(offset + i));
        }
    }

    if fragment.len() > 1 && fragment[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in fragment {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as usize))
        {
            Some(v) => v,
            None => return Err(CronError::Bounds(offset)),
        };
        if value > max {
            return Err(CronError::Bounds(offset));
        }
    }

    Ok(value)
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    insertion: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let insertion = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            insertion,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll.map(|t| t / 60_000 + 1).unwrap_or(0);
        let upper = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if lower <= upper {
            for job in &self.jobs {
                match job.policy {
                    Policy::All => {
                        for minute in lower..=upper {
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.insertion,
                                ));
                            }
                        }
                    }
                    Policy::Once => {
                        if let Some(minute) = (lower..=upper)
                            .rev()
                            .find(|&minute| job.cron.matches(minute))
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.insertion,
                            ));
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0
                            && upper >= lower
                            && job.cron.matches(upper)
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.insertion,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
