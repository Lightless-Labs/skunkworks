#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

#[derive(Debug, Clone, Copy)]
struct Emitted {
    job: usize,
    at: u64,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved_ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut cron = Self {
            minutes: [false; 60],
            hours: [false; 24],
        };

        parse_field(
            &text[..split],
            0,
            59,
            &mut cron.minutes,
        )?;
        parse_field(
            &text[split + 1..],
            split + 1,
            23,
            &mut cron.hours,
        )?;

        Ok(cron)
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    max: u32,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = field.as_bytes();
    let mut start = 0;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }

        parse_item(&field[start..end], offset + start, max, selected)?;

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    max: u32,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();

    let slash = bytes.iter().position(|&b| b == b'/');
    if let Some(first_slash) = slash {
        if let Some(second) = bytes[first_slash + 1..]
            .iter()
            .position(|&b| b == b'/')
        {
            return Err(CronError::Syntax(offset + first_slash + 1 + second));
        }
    }

    let base_end = slash.unwrap_or(bytes.len());
    let base = &item[..base_end];

    let (range_start, range_end, dash) = if base == "*" {
        (0, max, None)
    } else if let Some(dash_pos) = base.as_bytes().iter().position(|&b| b == b'-') {
        let left = parse_number(&base[..dash_pos], offset, max)?;
        let right = parse_number(
            &base[dash_pos + 1..],
            offset + dash_pos + 1,
            max,
        )?;
        if left > right {
            return Err(CronError::Descending(offset + dash_pos));
        }
        (left, right, Some(dash_pos))
    } else {
        if slash.is_some() {
            return Err(CronError::Syntax(offset + slash.unwrap()));
        }
        let n = parse_number(base, offset, max)?;
        (n, n, None)
    };

    let step = if let Some(slash_pos) = slash {
        if base != "*" && dash.is_none() {
            return Err(CronError::Syntax(offset + slash_pos));
        }
        let step_start = slash_pos + 1;
        let value = parse_number(&item[step_start..], offset + step_start, u32::MAX)
            .map_err(|e| match e {
                CronError::Bounds(_) => CronError::Step(offset + step_start),
                other => other,
            })?;
        let step_max = if max == 59 { 60 } else { 24 };
        if value == 0 || value > step_max {
            return Err(CronError::Step(offset + step_start));
        }
        value
    } else {
        1
    };

    let mut n = range_start;
    while n <= range_end {
        selected[n as usize] = true;
        match n.checked_add(step) {
            Some(next) => n = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_number(fragment: &str, offset: usize, max: u32) -> Result<u32, CronError> {
    let bytes = fragment.as_bytes();

    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value: u32 = 0;
    for &b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
        {
            Some(v) => v,
            None => return Err(CronError::Bounds(offset)),
        };
    }
    if value > max {
        return Err(CronError::Bounds(offset));
    }
    Ok(value)
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved_ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved_ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        let owned_id = id.to_owned();
        self.reserved_ids.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let mut emitted = Vec::new();

        if first_minute <= last_minute {
            for (job_index, job) in self.jobs.iter().enumerate() {
                match job.policy {
                    Policy::All => {
                        let mut minute = first_minute;
                        while minute <= last_minute {
                            if job.cron.matches(minute) {
                                emitted.push(Emitted {
                                    job: job_index,
                                    at: minute * 60_000,
                                });
                            }
                            minute += 1;
                        }
                    }
                    Policy::Once => {
                        let mut minute = last_minute;
                        loop {
                            if minute < first_minute {
                                break;
                            }
                            if job.cron.matches(minute) {
                                emitted.push(Emitted {
                                    job: job_index,
                                    at: minute * 60_000,
                                });
                                break;
                            }
                            if minute == 0 {
                                break;
                            }
                            minute -= 1;
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0
                            && now / 60_000 >= first_minute
                            && job.cron.matches(now / 60_000)
                        {
                            emitted.push(Emitted {
                                job: job_index,
                                at: now,
                            });
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.at.cmp(&b.at)
                .then_with(|| self.jobs[b.job].priority.cmp(&self.jobs[a.job].priority))
                .then_with(|| self.jobs[a.job].order.cmp(&self.jobs[b.job].order))
        });

        let result = emitted
            .into_iter()
            .map(|entry| Occurrence {
                id: self.jobs[entry.job].id.clone(),
                at: entry.at,
            })
            .collect();

        self.last_poll = Some(now);
        Ok(result)
    }
}
