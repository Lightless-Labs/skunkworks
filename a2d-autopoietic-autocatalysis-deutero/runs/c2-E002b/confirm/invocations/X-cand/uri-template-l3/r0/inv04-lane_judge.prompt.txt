pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    variables: Vec<(usize, String)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let close = match bytes[pos + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => pos + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[pos + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: pos + 1 + offset });
                }

                let mut content_start = pos + 1;
                let operator = if content_start < close
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let op = bytes[content_start];
                    content_start += 1;
                    Some(op)
                } else if content_start < close
                    && matches!(bytes[content_start], b'!' | b'=' | b'@' | b'|' | b'$')
                {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                } else {
                    None
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut variables = Vec::new();
                let mut seen = std::collections::HashSet::new();
                let mut item_start = content_start;

                loop {
                    let item_end = bytes[item_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| item_start + offset)
                        .unwrap_or(close);

                    if item_start == item_end {
                        return Err(Error::EmptyVariable { pos: item_start });
                    }

                    let name_bytes = &bytes[item_start..item_end];
                    let valid_first = name_bytes[0].is_ascii_alphabetic() || name_bytes[0] == b'_';
                    if !valid_first {
                        return Err(Error::InvalidName { pos: item_start });
                    }
                    for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
                        if !(byte.is_ascii_alphanumeric() || byte == b'_') {
                            return Err(Error::InvalidName { pos: item_start + offset });
                        }
                    }

                    let name = std::str::from_utf8(name_bytes).unwrap().to_owned();
                    if !seen.insert(name.clone()) {
                        return Err(Error::DuplicateVariable { pos: item_start });
                    }
                    variables.push((item_start, name));

                    if item_end == close {
                        break;
                    }
                    item_start = item_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end: close + 1,
                    operator,
                    variables,
                });
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields = Vec::new();
        for (_, name) in &expression.variables {
            if let Some(value) = vars
                .iter()
                .rev()
                .find_map(|(binding_name, value)| (*binding_name == name).then_some(*value))
            {
                fields.push((name.as_str(), value));
            }
        }

        if !fields.is_empty() {
            match expression.operator {
                Some(b'.') => output.push('.'),
                Some(b'/') => output.push('/'),
                Some(b';') => output.push(';'),
                Some(b'?') => output.push('?'),
                Some(b'&') => output.push('&'),
                Some(b'#') => output.push('#'),
                _ => {}
            }

            let separator = match expression.operator {
                Some(b'.') => '.',
                Some(b'/') => '/',
                Some(b';') => ';',
                Some(b'?') | Some(b'&') => '&',
                _ => ',',
            };

            for (index, (name, value)) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(separator);
                }

                match expression.operator {
                    Some(b';') => {
                        let encoded = encode(value, false);
                        output.push_str(name);
                        if !encoded.is_empty() {
                            output.push('=');
                            output.push_str(&encoded);
                        }
                    }
                    Some(b'?') | Some(b'&') => {
                        output.push_str(name);
                        output.push('=');
                        output.push_str(&encode(value, false));
                    }
                    _ => output.push_str(&encode(
                        value,
                        matches!(expression.operator, Some(b'+') | Some(b'#')),
                    )),
                }
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == b'%' && pos + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[pos + 1]), hex_value(bytes[pos + 2])) {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                pos += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        pos += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    b"0123456789ABCDEF"[value as usize] as char
}

                    Some(op)

                    None

                        return Err(Error::DuplicateVariable { pos: spec_start });

                        break;

                    start: open,
                    end: close + 1,
                    operator,
                    variables,
                });
                i = close + 1;

            _ => i += 1,
        }

    let mut literal_start = 0;

        output.push_str(&template[literal_start..expression.start]);

                    _ => encoded,

            }

                Some('#') => output.push('#'),
                Some('.') => output.push('.'),
                Some('/') => output.push('/'),

                Some('?') => output.push('?'),
                Some('&') => output.push('&'),

                Some('/') => "/",
                Some(';') => ";",

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            result.push('%');
            result.push((bytes[i + 1] as char).to_ascii_uppercase());
            result.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;

            result.push(byte as char);

            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Operator,
    variables: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(expression, vars));
        cursor = end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let expression = parse_expression(template, start, close)?;
                expressions.push((start, close, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let content_start = open + 1;
    let content = &template[content_start..close];
    if content.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let first = content.as_bytes()[0];
    let (operator, vars_start) = match first {
        b'+' => (Operator::Reserved, content_start + 1),
        b'#' => (Operator::Fragment, content_start + 1),
        b'.' => (Operator::Dot, content_start + 1),
        b'/' => (Operator::Slash, content_start + 1),
        b';' => (Operator::Semicolon, content_start + 1),
        b'?' => (Operator::Query, content_start + 1),
        b'&' => (Operator::Continuation, content_start + 1),
        b'!' | b'=' | b'%' | b'@' | b'|' | b'^' => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => (Operator::Simple, content_start),
    };

    if vars_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let names = &template[vars_start..close];
    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut segment_start = vars_start;

    for segment in names.split(',') {
        if segment.is_empty() {
            return Err(Error::EmptyVariable { pos: segment_start });
        }

        let segment_bytes = segment.as_bytes();
        for (offset, &byte) in segment_bytes.iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: segment_start + offset,
                });
            }
        }

        if seen.iter().any(|&name| name == segment) {
            return Err(Error::DuplicateVariable { pos: segment_start });
        }

        seen.push(segment);
        variables.push((segment.to_owned(), segment_start));
        segment_start += segment.len() + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn expand_expression(expression: Expression, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();
    let reserved = matches!(expression.operator, Operator::Reserved | Operator::Fragment);

    for (name, _) in expression.variables {
        if let Some(value) = vars.iter().rev().find_map(|(key, value)| {
            if *key == name {
                Some(*value)
            } else {
                None
            }
        }) {
            let encoded = encode_value(value, reserved);
            let field = match expression.operator {
                Operator::Semicolon => {
                    if encoded.is_empty() {
                        name
                    } else {
                        format!("{name}={encoded}")
                    }
                }
                Operator::Query | Operator::Continuation => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };

    let mut result = String::from(prefix);
    result.push_str(&fields.join(separator));
    result
}

fn encode_value(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    variables: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open = index;
                let close = match bytes[index + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => index + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[index + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + offset,
                    });
                }

                let mut content_start = index + 1;
                let mut operator = None;
                if content_start < close
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let op = bytes[content_start] as char;
                    if !matches!(op, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    operator = Some(op);
                    content_start += 1;
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let content = &template[content_start..close];
                let mut parsed_vars = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut start = content_start;

                for specifier in content.split(',') {
                    if specifier.is_empty() {
                        return Err(Error::EmptyVariable { pos: start });
                    }

                    let spec_bytes = specifier.as_bytes();
                    let name_len = spec_bytes
                        .iter()
                        .position(|&b| !is_name_byte(b))
                        .unwrap_or(spec_bytes.len());

                    if name_len != spec_bytes.len() || name_len == 0 {
                        return Err(Error::InvalidName {
                            pos: start + name_len.min(spec_bytes.len()),
                        });
                    }

                    if seen.iter().any(|name| name == specifier) {
                        return Err(Error::DuplicateVariable { pos: start });
                    }

                    seen.push(specifier.to_string());
                    parsed_vars.push(specifier.to_string());
                    start += specifier.len() + 1;
                }

                expressions.push(Expression {
                    operator,
                    variables: parsed_vars,
                });
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    index = 0;

    while index < bytes.len() {
        if bytes[index] == b'{' {
            output.push_str(&template[cursor..index]);
            let close = index + 1 + bytes[index + 1..]
                .iter()
                .position(|&b| b == b'}')
                .expect("syntax validated");
            let expression = &expressions.remove(0);
            expand_expression(expression, vars, &mut output);
            cursor = close + 1;
            index = close + 1;
        } else {
            index += 1;
        }
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn is_name_byte(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)], output: &mut String) {
    let operator = expression.operator.unwrap_or('\0');
    let mut fields = Vec::new();

    for name in &expression.variables {
        if let Some(value) = vars.iter().rev().find(|(key, _)| *key == name).map(|(_, v)| *v) {
            let encoded = encode(value, matches!(operator, '+' | '#'));
            fields.push((name.as_str(), encoded));
        }
    }

    if fields.is_empty() {
        return;
    }

    match operator {
        '.' | '/' | ';' | '?' | '&' => output.push(operator),
        '#' => output.push('#'),
        _ => {}
    }

    let separator = match operator {
        '.' => ".",
        '/' => "/",
        ';' => ";",
        '?' | '&' => "&",
        _ => ",",
    };

    for (index, (name, value)) in fields.iter().enumerate() {
        if index > 0 {
            output.push_str(separator);
        }

        match operator {
            ';' => {
                output.push_str(name);
                if !value.is_empty() {
                    output.push('=');
                    output.push_str(value);
                }
            }
            '?' | '&' => {
                output.push_str(name);
                output.push('=');
                output.push_str(value);
            }
            _ => output.push_str(value),
        }
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%'
            && index + 2 < bytes.len()
            && is_hex(bytes[index + 1])
            && is_hex(bytes[index + 2])
        {
            result.push('%');
            result.push((bytes[index + 1] as char).to_ascii_uppercase());
            result.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(hex_digit(byte >> 4));
            result.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    result
}
