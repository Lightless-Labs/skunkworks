#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128).unsigned_abs();
    let dy = (a.y as i128 - b.y as i128).unsigned_abs();

    // Coordinates are validated before this function is called, so the scaled
    // squared distance is at most 8 * 10^24 and fits in u128.
    let scaled_squared = (dx * dx + dy * dy) * 1_000_000;

    // Find floor(sqrt(scaled_squared)) using binary search.
    let mut low = 0u128;
    let mut high = 4_000_000_000u128;
    while low < high {
        let mid = low + (high - low + 1) / 2;
        if mid * mid <= scaled_squared {
            low = mid;
        } else {
            high = mid - 1;
        }
    }

    // Round to nearest integer; an exact half tie rounds upward. Comparing
    // against the squared midpoint avoids floating point.
    let midpoint = 2 * low + 1;
    if 4 * scaled_squared >= midpoint * midpoint {
        (low + 1) as u64
    } else {
        low as u64
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = points.len()
        - usize::from(points.len() >= 2 && points.last() == points.first());
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    let polygon = &points[..effective_len];

    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for index in 0..effective_len {
        let a = polygon[index];
        let b = polygon[(index + 1) % effective_len];

        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let should_replace = match longest_edge {
            None => true,
            Some(current) => {
                if length != edge_milli[current] {
                    length > edge_milli[current]
                } else {
                    let current_a = polygon[current];
                    let current_b = polygon[(current + 1) % effective_len];
                    let candidate_pair = if a <= b { (a, b) } else { (b, a) };
                    let current_pair = if current_a <= current_b {
                        (current_a, current_b)
                    } else {
                        (current_b, current_a)
                    };
                    candidate_pair < current_pair
                        || (candidate_pair == current_pair && index < current)
                }
            }
        };
        if should_replace {
            longest_edge = Some(index);
        }
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
