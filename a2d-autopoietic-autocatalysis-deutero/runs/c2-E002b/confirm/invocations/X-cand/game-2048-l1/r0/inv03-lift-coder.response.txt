```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (!(2..=2_147_483_648).contains(&value) || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Box<dyn Iterator<Item = usize>> = match side {
        Side::Left => Box::new(0..row.len()),
        Side::Right => Box::new((0..row.len()).rev()),
    };

    let nonempty: Vec<usize> = indices.filter(|&i| row[i] != 0).collect();
    let mut outputs: Vec<(u32, Option<Merge>)> = Vec::with_capacity(nonempty.len());
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut pos = 0;

    while pos < nonempty.len() {
        let first = nonempty[pos];
        if pos + 1 < nonempty.len() && row[first] == row[nonempty[pos + 1]] {
            let second = nonempty[pos + 1];
            let Some(value) = row[first].checked_add(row[second]) else {
                return Err(Error::MergeOverflow { first, second });
            };
            let to = match side {
                Side::Left => outputs.len(),
                Side::Right => row.len() - 1 - outputs.len(),
            };
            let merge = Merge {
                from: [first, second],
                to,
                value,
            };
            score += u64::from(value);
            merges.push(merge.clone());
            outputs.push((value, Some(merge)));
            pos += 2;
        } else {
            outputs.push((row[first], None));
            pos += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    for (output_index, (value, _)) in outputs.iter().enumerate() {
        let to = match side {
            Side::Left => output_index,
            Side::Right => row.len() - 1 - output_index,
        };
        cells[to] = *value;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
```
