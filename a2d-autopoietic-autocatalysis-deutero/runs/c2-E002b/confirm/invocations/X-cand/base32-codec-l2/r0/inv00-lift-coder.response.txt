#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    if input.is_empty() {
        return String::new();
    }

    let symbols = alphabet_bytes(alphabet);
    let mut output = String::with_capacity((input.len() * 8 + 4) / 5 + 7);
    let mut accumulator: u32 = 0;
    let mut bits = 0;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let value = ((accumulator >> bits) & 0x1f) as usize;
            output.push(symbols[value] as char);
        }
    }

    if bits > 0 {
        let value = ((accumulator << (5 - bits)) & 0x1f) as usize;
        output.push(symbols[value] as char);
    }

    if padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

fn symbol_value(byte: u8, symbols: &[u8; 32]) -> Option<u8> {
    symbols.iter().position(|&symbol| symbol == byte).map(|v| v as u8)
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let symbols = alphabet_bytes(alphabet);

    // Pass 1: validate every lexical byte before checking any structural rule.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && symbol_value(byte, symbols).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let first_padding = bytes.iter().position(|&byte| byte == b'=');
    let p = first_padding.unwrap_or(bytes.len());

    // Pass 2: validate padding mode and placement.
    match padding {
        Padding::Omitted => {
            if first_padding.is_some() {
                return Err(DecodeError::Padding { offset: p });
            }
        }
        Padding::Required => {
            if let Some(start) = first_padding {
                if let Some(relative) = bytes[start..].iter().position(|&byte| byte != b'=') {
                    return Err(DecodeError::Padding {
                        offset: start + relative,
                    });
                }
            }
        }
    }

    let data_len = p;

    // Pass 3: validate the data-symbol length.
    let residue = data_len % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength {
            offset: bytes.len(),
        });
    }

    // Pass 4: validate Required padding shape.
    if padding == Padding::Required {
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => unreachable!(),
        };
        if bytes.len() - data_len != expected_padding || bytes.len() % 8 != 0 {
            return Err(DecodeError::Padding { offset: p });
        }
    }

    // Pass 5: validate unused low bits in the final symbol.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits > 0 {
        let final_value = symbol_value(bytes[data_len - 1], symbols).unwrap();
        if final_value & ((1u8 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits {
                offset: data_len - 1,
            });
        }
    }

    let mut output = Vec::with_capacity(data_len * 5 / 8);
    let mut accumulator: u32 = 0;
    let mut bits = 0;

    for &byte in &bytes[..data_len] {
        let value = u32::from(symbol_value(byte, symbols).unwrap());
        accumulator = (accumulator << 5) | value;
        bits += 5;

        if bits >= 8 {
            bits -= 8;
            output.push(((accumulator >> bits) & 0xff) as u8);
        }
    }

    Ok(output)
}
