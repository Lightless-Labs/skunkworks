#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Variable {
    name: String,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    variables: Vec<Variable>,
}

#[derive(Debug, Clone)]
enum Part {
    Literal(String),
    Expression(Expression),
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = parse(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(&text),
            Part::Expression(expression) => {
                let mut fields = Vec::new();

                for variable in expression.variables {
                    if let Some(value) = vars
                        .iter()
                        .rev()
                        .find(|(name, _)| *name == variable.name)
                        .map(|(_, value)| *value)
                    {
                        fields.push(encode(value, matches!(expression.operator, Some('+' | '#'))));
                    }
                }

                if fields.is_empty() {
                    continue;
                }

                if expression.operator == Some('#') {
                    output.push('#');
                }
                let separator = if expression.operator.is_some() {
                    ","
                } else {
                    ","
                };
                output.push_str(&fields.join(separator));
            }
        }
    }

    Ok(output)
}

fn parse(template: &str) -> Result<Vec<Part>, Error> {
    let mut parts = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let rest = &template[cursor..];
        let next_open = rest.find('{');
        let next_close = rest.find('}');

        match (next_open, next_close) {
            (None, Some(close)) => {
                return Err(Error::UnexpectedClose {
                    pos: cursor + close,
                });
            }
            (None, None) => {
                parts.push(Part::Literal(rest.to_owned()));
                break;
            }
            (Some(open), Some(close)) if close < open => {
                return Err(Error::UnexpectedClose {
                    pos: cursor + close,
                });
            }
            (Some(open), _) => {
                let open_pos = cursor + open;
                if open > 0 {
                    parts.push(Part::Literal(template[cursor..open_pos].to_owned()));
                }

                let content_start = open_pos + 1;
                let closing = match template[content_start..].find('}') {
                    Some(offset) => content_start + offset,
                    None => return Err(Error::Unclosed { pos: open_pos }),
                };

                if let Some(offset) = template[content_start..closing].find('{') {
                    return Err(Error::NestedOpen {
                        pos: content_start + offset,
                    });
                }

                parts.push(Part::Expression(parse_expression(
                    template,
                    content_start,
                    closing,
                )?));
                cursor = closing + 1;
            }
        }
    }

    Ok(parts)
}

fn parse_expression(
    template: &str,
    start: usize,
    end: usize,
) -> Result<Expression, Error> {
    let content = &template[start..end];
    let mut operator = None;
    let mut spec_start = start;

    if let Some(first) = content.chars().next() {
        if matches!(first, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            let operator_pos = start;
            if !matches!(first, '+' | '#') {
                return Err(Error::UnsupportedOperator { pos: operator_pos });
            }
            operator = Some(first);
            spec_start += first.len_utf8();
        }
    }

    if spec_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let specifiers = &template[spec_start..end];
    let mut variables = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut offset = spec_start;

    for specifier in specifiers.split(',') {
        if specifier.is_empty() {
            return Err(Error::EmptyVariable { pos: offset });
        }

        let name = specifier;
        let invalid = first_invalid_name_byte(name);
        if let Some(invalid_offset) = invalid {
            return Err(Error::InvalidName {
                pos: offset + invalid_offset,
            });
        }

        if seen.iter().any(|previous| previous == name) {
            return Err(Error::DuplicateVariable { pos: offset });
        }

        seen.push(name.to_owned());
        variables.push(Variable {
            name: name.to_owned(),
        });
        offset += specifier.len() + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn first_invalid_name_byte(name: &str) -> Option<usize> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Some(0);
    }

    let is_start = |byte: u8| byte.is_ascii_alphabetic() || byte == b'_';
    let is_continue = |byte: u8| is_start(byte) || byte.is_ascii_digit();

    if !is_start(bytes[0]) {
        return Some(0);
    }

    bytes
        .iter()
        .enumerate()
        .skip(1)
        .find_map(|(index, byte)| (!is_continue(*byte)).then_some(index))
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }

        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, String)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(relative) = bytes[open + 1..close].iter().position(|&byte| byte == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + relative,
        });
    }

    let mut content_start = open + 1;
    let operator = match bytes[content_start] {
        b'+' => {
            content_start += 1;
            Operator::Reserved
        }
        b'#' => {
            content_start += 1;
            Operator::Fragment
        }
        b'.' | b'/' | b';' | b'?' | b'&' => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => Operator::Simple,
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |relative| spec_start + relative);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &bytes[spec_start..spec_end];
        if !is_name_start(spec[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(relative) = spec.iter().skip(1).position(|&byte| !is_name_byte(byte)) {
            return Err(Error::InvalidName {
                pos: spec_start + 1 + relative,
            });
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|prior| prior == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push((spec_start, name));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        names,
    })
}

fn encode_value(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            let first = bytes[index + 1];
            let second = bytes[index + 2];
            if first.is_ascii_hexdigit() && second.is_ascii_hexdigit() {
                output.push('%');
                output.push((first as char).to_ascii_uppercase());
                output.push((second as char).to_ascii_uppercase());
                index += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$'
                    | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let Some(relative_close) = bytes[index + 1..].iter().position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: index });
                };
                let close = index + 1 + relative_close;
                expressions.push(parse_expression(template, index, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let reserved = expression.operator != Operator::Simple;
        let mut fields = Vec::new();

        for (_, name) in &expression.names {
            let mut value = None;
            for &(binding_name, binding_value) in vars {
                if binding_name == name {
                    value = Some(binding_value);
                }
            }
            if let Some(value) = value {
                fields.push(encode_value(value, reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Operator::Fragment {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<u8>,
    content: &'a str,
}

fn validate_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression<'_>, Error> {
    let inner_start = open + 1;
    let inner = &template[inner_start..close];

    if let Some(offset) = inner.as_bytes().iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: inner_start + offset,
        });
    }

    let bytes = inner.as_bytes();
    let operator = bytes.first().copied().filter(|b| {
        matches!(b, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    });

    if let Some(op) = operator {
        if op != b'+' && op != b'#' {
            return Err(Error::UnsupportedOperator { pos: inner_start });
        }
    }

    let content_start = inner_start + usize::from(operator.is_some());
    let content = &template[content_start..close];

    if content.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    for specifier in content.split(',') {
        if specifier.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = specifier.as_bytes();
        let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
        let valid_rest = |b: u8| b.is_ascii_alphanumeric() || b == b'_';

        if !valid_first(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        if let Some(offset) = name_bytes.iter().position(|&b| !valid_rest(b)) {
            if offset != 0 {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        if seen.contains(&specifier) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(specifier);

        spec_start += specifier.len() + 1;
    }

    Ok(Expression {
        start: content_start,
        end: close,
        operator,
        content,
    })
}

        for name in expression.content.split(',') {

                fields.push(encode_value(value, allow_reserved));
            }

            if expression.operator == Some(b'#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        cursor = after_close;
        let _ = expression.start;
        let _ = expression.end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<(usize, String)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        let fields: Vec<String> = expression
            .names
            .iter()
            .filter_map(|(_, name)| {
                vars.iter()
                    .rev()
                    .find(|(binding, _)| *binding == name)
                    .map(|(_, value)| encode(value, expression.operator))
            })
            .collect();

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }
        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let mut content_start = i + 1;
                let mut operator = None;
                if content_start < close
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let op = bytes[content_start] as char;
                    if op != '+' && op != '#' {
                        return Err(Error::UnsupportedOperator {
                            pos: content_start,
                        });
                    }
                    operator = Some(op);
                    content_start += 1;
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut spec_start = content_start;
                let mut seen = Vec::<String>::new();

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map_or(close, |offset| spec_start + offset);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let name_bytes = &bytes[spec_start..spec_end];
                    if !is_name_start(name_bytes[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
                        if !is_name_continue(byte) {
                            return Err(Error::InvalidName {
                                pos: spec_start + offset,
                            });
                        }
                    }

                    let name = std::str::from_utf8(name_bytes)
                        .expect("validated variable names are ASCII")
                        .to_owned();
                    if seen.iter().any(|previous| previous == &name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name.clone());
                    names.push((spec_start, name));

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end: close,
                    operator,
                    names,
                });
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn encode(value: &str, operator: Option<char>) -> String {
    let bytes = value.as_bytes();
    let mut result = String::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            result.push('%');
            result.push((bytes[i + 1] as char).to_ascii_uppercase());
            result.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if is_unreserved(byte) || (matches!(operator, Some('+') | Some('#')) && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(hex_digit(byte >> 4));
            result.push(hex_digit(byte & 0x0f));
        }
        i += 1;
    }

    result
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_digit(nibble: u8) -> char {
    match nibble {
        0..=9 => (b'0' + nibble) as char,
        _ => (b'A' + nibble - 10) as char,
    }
}
