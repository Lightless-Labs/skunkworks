#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Variable {
    name: String,
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    variables: Vec<Variable>,
}

#[derive(Debug, Clone)]
enum Part {
    Literal(String),
    Expression(Expression),
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = parse(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(&text),
            Part::Expression(expression) => {
                let mut fields = Vec::new();

                for variable in expression.variables {
                    if let Some(value) = vars
                        .iter()
                        .rev()
                        .find(|(name, _)| *name == variable.name)
                        .map(|(_, value)| *value)
                    {
                        fields.push(encode(value, matches!(expression.operator, Some('+' | '#'))));
                    }
                }

                if fields.is_empty() {
                    continue;
                }

                if expression.operator == Some('#') {
                    output.push('#');
                }
                let separator = if expression.operator.is_some() {
                    ","
                } else {
                    ","
                };
                output.push_str(&fields.join(separator));
            }
        }
    }

    Ok(output)
}

fn parse(template: &str) -> Result<Vec<Part>, Error> {
    let mut parts = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let rest = &template[cursor..];
        let next_open = rest.find('{');
        let next_close = rest.find('}');

        match (next_open, next_close) {
            (None, Some(close)) => {
                return Err(Error::UnexpectedClose {
                    pos: cursor + close,
                });
            }
            (None, None) => {
                parts.push(Part::Literal(rest.to_owned()));
                break;
            }
            (Some(open), Some(close)) if close < open => {
                return Err(Error::UnexpectedClose {
                    pos: cursor + close,
                });
            }
            (Some(open), _) => {
                let open_pos = cursor + open;
                if open > 0 {
                    parts.push(Part::Literal(template[cursor..open_pos].to_owned()));
                }

                let content_start = open_pos + 1;
                let closing = match template[content_start..].find('}') {
                    Some(offset) => content_start + offset,
                    None => return Err(Error::Unclosed { pos: open_pos }),
                };

                if let Some(offset) = template[content_start..closing].find('{') {
                    return Err(Error::NestedOpen {
                        pos: content_start + offset,
                    });
                }

                parts.push(Part::Expression(parse_expression(
                    template,
                    content_start,
                    closing,
                )?));
                cursor = closing + 1;
            }
        }
    }

    Ok(parts)
}

fn parse_expression(
    template: &str,
    start: usize,
    end: usize,
) -> Result<Expression, Error> {
    let content = &template[start..end];
    let mut operator = None;
    let mut spec_start = start;

    if let Some(first) = content.chars().next() {
        if matches!(first, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            let operator_pos = start;
            if !matches!(first, '+' | '#') {
                return Err(Error::UnsupportedOperator { pos: operator_pos });
            }
            operator = Some(first);
            spec_start += first.len_utf8();
        }
    }

    if spec_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let specifiers = &template[spec_start..end];
    let mut variables = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut offset = spec_start;

    for specifier in specifiers.split(',') {
        if specifier.is_empty() {
            return Err(Error::EmptyVariable { pos: offset });
        }

        let name = specifier;
        let invalid = first_invalid_name_byte(name);
        if let Some(invalid_offset) = invalid {
            return Err(Error::InvalidName {
                pos: offset + invalid_offset,
            });
        }

        if seen.iter().any(|previous| previous == name) {
            return Err(Error::DuplicateVariable { pos: offset });
        }

        seen.push(name.to_owned());
        variables.push(Variable {
            name: name.to_owned(),
        });
        offset += specifier.len() + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn first_invalid_name_byte(name: &str) -> Option<usize> {
    let bytes = name.as_bytes();
    if bytes.is_empty() {
        return Some(0);
    }

    let is_start = |byte: u8| byte.is_ascii_alphabetic() || byte == b'_';
    let is_continue = |byte: u8| is_start(byte) || byte.is_ascii_digit();

    if !is_start(bytes[0]) {
        return Some(0);
    }

    bytes
        .iter()
        .enumerate()
        .skip(1)
        .find_map(|(index, byte)| (!is_continue(*byte)).then_some(index))
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }

        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
