#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, String)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(relative) = bytes[open + 1..close].iter().position(|&byte| byte == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + relative,
        });
    }

    let mut content_start = open + 1;
    let operator = match bytes[content_start] {
        b'+' => {
            content_start += 1;
            Operator::Reserved
        }
        b'#' => {
            content_start += 1;
            Operator::Fragment
        }
        b'.' | b'/' | b';' | b'?' | b'&' => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => Operator::Simple,
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |relative| spec_start + relative);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &bytes[spec_start..spec_end];
        if !is_name_start(spec[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(relative) = spec.iter().skip(1).position(|&byte| !is_name_byte(byte)) {
            return Err(Error::InvalidName {
                pos: spec_start + 1 + relative,
            });
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|prior| prior == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push((spec_start, name));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        names,
    })
}

fn encode_value(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            let first = bytes[index + 1];
            let second = bytes[index + 2];
            if first.is_ascii_hexdigit() && second.is_ascii_hexdigit() {
                output.push('%');
                output.push((first as char).to_ascii_uppercase());
                output.push((second as char).to_ascii_uppercase());
                index += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$'
                    | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let Some(relative_close) = bytes[index + 1..].iter().position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: index });
                };
                let close = index + 1 + relative_close;
                expressions.push(parse_expression(template, index, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let reserved = expression.operator != Operator::Simple;
        let mut fields = Vec::new();

        for (_, name) in &expression.names {
            let mut value = None;
            for &(binding_name, binding_value) in vars {
                if binding_name == name {
                    value = Some(binding_value);
                }
            }
            if let Some(value) = value {
                fields.push(encode_value(value, reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Operator::Fragment {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
