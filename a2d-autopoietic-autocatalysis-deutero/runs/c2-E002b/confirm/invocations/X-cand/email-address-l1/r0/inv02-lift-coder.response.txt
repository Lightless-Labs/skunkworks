#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    let local = &bytes[..separator];
    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];

    if local.is_empty() {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    const LOCAL_SPECIALS: &[u8] = b"!#$%&'*+-/=?^_`{|}~";
    for (index, &byte) in local.iter().enumerate() {
        if byte == b'.' {
            if index == 0 || index + 1 == local.len() || local[index - 1] == b'.' {
                return Err(EmailError::LocalDot { at: index });
            }
        } else if !byte.is_ascii_alphanumeric() && !LOCAL_SPECIALS.contains(&byte) {
            return Err(EmailError::LocalChar { at: index });
        }
    }

    let root_dot = raw_domain.last().copied() == Some(b'.');
    let domain = if root_dot {
        &raw_domain[..raw_domain.len() - 1]
    } else {
        raw_domain
    };
    let root_dot_at = input.len() - 1;

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0;
    for (index, &byte) in domain.iter().enumerate() {
        if byte == b'.' {
            labels.push((label_start, &domain[label_start..index]));
            label_start = index + 1;
        }
    }
    labels.push((label_start, &domain[label_start..]));

    for (index, &(relative_start, label)) in labels.iter().enumerate() {
        let absolute_start = domain_start + relative_start;
        if label.is_empty() {
            let at = if index + 1 == labels.len() && root_dot {
                root_dot_at
            } else {
                absolute_start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: absolute_start + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at: absolute_start });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: absolute_start + label.len() - 1,
            });
        }

        for (offset, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar {
                    at: absolute_start + offset,
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII local part");
    let domain_string = domain
        .iter()
        .map(|byte| byte.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}
