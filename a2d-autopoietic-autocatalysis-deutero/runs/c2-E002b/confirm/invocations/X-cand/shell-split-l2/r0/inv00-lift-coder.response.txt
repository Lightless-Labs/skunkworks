#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn next_char(input: &str, pos: usize) -> Option<(char, usize)> {
    input[pos..].chars().next().map(|c| (c, pos + c.len_utf8()))
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn find_brace_end(input: &str, mut pos: usize, dollar: usize) -> Result<usize, SplitError> {
    while let Some((c, next)) = next_char(input, pos) {
        if c == '\\' {
            if let Some((_, after)) = next_char(input, next) {
                pos = after;
            } else {
                pos = next;
            }
        } else if c == '}' {
            return Ok(pos);
        } else {
            pos = next;
        }
    }
    Err(SplitError::UnterminatedExpansion { pos: dollar })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut pos = 0;
    while let Some((c, next)) = next_char(text, pos) {
        if c == '\\' {
            if let Some((escaped, after)) = next_char(text, next) {
                out.push(escaped);
                pos = after;
                continue;
            }
        }
        out.push(c);
        pos = next;
    }
    out
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar + 2;
    let close = find_brace_end(input, content_start, dollar)?;
    let contents = &input[content_start..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, c)| (!is_name_continue(c)).then_some(i))
        .unwrap_or(contents.len());

    if name_end == 0 || !contents[..name_end].chars().all(is_name_continue) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    if !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let rest = &contents[name_end..];
    let value = if rest.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

fn expand_unbraced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Option<(String, usize)> {
    let (first, mut pos) = next_char(input, dollar + 1)?;
    if !is_name_start(first) {
        return None;
    }

    while let Some((c, next)) = next_char(input, pos) {
        if !is_name_continue(c) {
            break;
        }
        pos = next;
    }

    let name = &input[dollar + 1..pos];
    Some((lookup(env, name).unwrap_or("").to_string(), pos))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut pos = 0;

    while let Some((c, next)) = next_char(input, pos) {
        match quote {
            Quote::Single(open) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                pos = next;
                let _ = open;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next;
                } else if c == '\\' {
                    let Some((escaped, after)) = next_char(input, next) else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if matches!(escaped, '"' | '\\' | '$') {
                        word.push(escaped);
                        active = true;
                        pos = after;
                    } else if escaped == '\n' {
                        active = true;
                        pos = after;
                    } else {
                        word.push('\\');
                        active = true;
                        pos = next;
                    }
                } else if c == '$' {
                    if let Some((next_c, _)) = next_char(input, next) {
                        if next_c == '{' {
                            let (value, after) = expand_braced(input, pos, env)?;
                            word.push_str(&value);
                            active = true;
                            pos = after;
                        } else if let Some((value, after)) = expand_unbraced(input, pos, env) {
                            word.push_str(&value);
                            active = true;
                            pos = after;
                        } else {
                            word.push('$');
                            pos = next;
                        }
                    } else {
                        word.push('$');
                        pos = next;
                    }
                } else {
                    word.push(c);
                    active = true;
                    pos = next;
                }
                let _ = open;
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    pos = next;
                } else if c == '\'' {
                    quote = Quote::Single(pos);
                    active = true;
                    pos = next;
                } else if c == '"' {
                    quote = Quote::Double(pos);
                    active = true;
                    pos = next;
                } else if c == '\\' {
                    let Some((escaped, after)) = next_char(input, next) else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if escaped == '\n' {
                        pos = after;
                    } else {
                        word.push(escaped);
                        active = true;
                        pos = after;
                    }
                } else if c == '$' {
                    if let Some((next_c, _)) = next_char(input, next) {
                        if next_c == '{' {
                            let (value, after) = expand_braced(input, pos, env)?;
                            word.push_str(&value);
                            active = true;
                            pos = after;
                        } else if let Some((value, after)) = expand_unbraced(input, pos, env) {
                            word.push_str(&value);
                            active = true;
                            pos = after;
                        } else {
                            word.push('$');
                            active = true;
                            pos = next;
                        }
                    } else {
                        word.push('$');
                        active = true;
                        pos = next;
                    }
                } else {
                    word.push(c);
                    active = true;
                    pos = next;
                }
            }
        }
    }

    match quote {
        Quote::Single(open) => return Err(SplitError::UnterminatedSingle { pos: open }),
        Quote::Double(open) => return Err(SplitError::UnterminatedDouble { pos: open }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
