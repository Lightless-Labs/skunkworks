#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }

    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(open) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    let _ = open;
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(open) => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let next = input[next_i..].chars().next().unwrap();
                        let after_next = next_i + next.len_utf8();
                        if matches!(next, '"' | '\\' | '$') {
                            word.push(next);
                            i = after_next;
                        } else if next == '\n' {
                            i = after_next;
                        } else {
                            word.push('\\');
                            i = next_i;
                        }
                    }
                    '$' => {
                        active = true;
                        if let Some((value, end)) = parse_expansion(input, i, env)? {
                            word.push_str(&value);
                            i = end;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    }
                    _ => {
                        word.push(c);
                        i = next_i;
                    }
                }
                let _ = open;
            }
            Quote::None => {
                match c {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(std::mem::take(&mut word));
                            active = false;
                        }
                        i = next_i;
                    }
                    '\'' => {
                        active = true;
                        quote = Quote::Single(i);
                        i = next_i;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double(i);
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let next = input[next_i..].chars().next().unwrap();
                        let after_next = next_i + next.len_utf8();
                        if next == '\n' {
                            i = after_next;
                        } else {
                            active = true;
                            word.push(next);
                            i = after_next;
                        }
                    }
                    '$' => {
                        if let Some((value, end)) = parse_expansion(input, i, env)? {
                            active = true;
                            word.push_str(&value);
                            i = end;
                        } else {
                            active = true;
                            word.push('$');
                            i = next_i;
                        }
                    }
                    _ => {
                        active = true;
                        word.push(c);
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<Option<(String, usize)>, SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok(None);
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            let end = cursor + c.len_utf8();
            if c == '\\' {
                if end < input.len() {
                    let skipped = input[end..].chars().next().unwrap();
                    cursor = end + skipped.len_utf8();
                } else {
                    cursor = end;
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor = end;
            }
        }

        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };

        let contents = &input[content_start..close_pos];
        let (name, default) = if let Some(operator_pos) = contents.find(":-") {
            (&contents[..operator_pos], Some(&contents[operator_pos + 2..]))
        } else {
            (contents, None)
        };

        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
            || !name.chars().all(is_name_continue)
        {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        if let Some(default_text) = default {
            let value = lookup(name, env).unwrap_or("");
            if value.is_empty() {
                Ok(Some((decode_default(default_text), close_pos + 1)))
            } else {
                Ok(Some((value.to_string(), close_pos + 1)))
            }
        } else {
            Ok(Some((lookup(name, env).unwrap_or("").to_string(), close_pos + 1)))
        }
    } else if is_name_start(first) {
        let mut end = after_dollar + first.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if !is_name_continue(c) {
                break;
            }
            end += c.len_utf8();
        }
        let name = &input[after_dollar..end];
        Ok(Some((lookup(name, env).unwrap_or("").to_string(), end)))
    } else {
        Ok(None)
    }
}
