#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|item| item == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|item| item == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !matches!(state, State::Quoted { .. }) && b == b'\r' {
            if bytes.get(i + 1) == Some(&b'\n') {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record_index += 1;
                record_started = false;
                state = State::Start;
                i += 2;
                continue;
            }
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Start => {
                if b == delimiter {
                    fields.push(String::new());
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    i += 1;
                } else if b == quote {
                    state = State::Quoted { opening: i };
                    record_started = true;
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    state = State::Unquoted;
                    record_started = true;
                    i += ch.len_utf8();
                }
            }
            State::Unquoted => {
                if b == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::Start;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    state = State::Start;
                    i += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    let ch = input[i..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    record_started = true;
                    i += ch.len_utf8();
                }
            }
            State::Quoted { opening } => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    record_started = true;
                    i += ch.len_utf8();
                    if i == bytes.len() {
                        let _ = opening;
                    }
                }
            }
            State::AfterQuote => {
                if b == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::Start;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    state = State::Start;
                    i += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::AfterQuote | State::Unquoted => {
            fields.push(field);
            records.push(fields);
        }
        State::Start => {
            if record_started || !fields.is_empty() {
                fields.push(field);
                records.push(fields);
            }
        }
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut records = records.into_iter();
    let header = if config.headers {
        records.next()
    } else {
        None
    };

    if let Some(ref header_fields) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..header_fields.len() {
                if let Some(first) = header_fields[..second]
                    .iter()
                    .position(|name| name == &header_fields[second])
                {
                    return Err(CsvError::DuplicateHeader {
                        name: header_fields[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }
    }

    let mut rows = Vec::new();
    for (offset, row) in records.enumerate() {
        let record = if config.headers { offset + 1 } else { offset };
        if let Some(ref header_fields) = header {
            if row.len() != header_fields.len() {
                return Err(CsvError::HeaderWidth {
                    record,
                    expected: header_fields.len(),
                    actual: row.len(),
                });
            }
        }
        rows.push(row);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn is_integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !is_integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn is_decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let body = if bytes.first() == Some(&b'-') {
        &bytes[1..]
    } else {
        bytes
    };
    let mut parts = body.split(|&b| b == b'.');
    let integral = match parts.next() {
        Some(part) => part,
        None => return false,
    };
    let fractional = match parts.next() {
        Some(part) => part,
        None => return false,
    };
    if parts.next().is_some() || fractional.len() != 2 || !fractional.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral == b"0" {
        return bytes.first() != Some(&b'-') || fractional != b"00";
    }
    !integral.is_empty()
        && integral[0].is_ascii_digit()
        && integral[0] != b'0'
        && integral.iter().all(u8::is_ascii_digit)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !is_decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let dot = unsigned.find('.').expect("validated decimal");
    let whole = unsigned[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let fraction = unsigned[dot + 1..]
        .parse::<i128>()
        .expect("validated fractional digits");
    let magnitude = whole
        .checked_mul(100)
        .and_then(|v| v.checked_add(fraction))
        .ok_or(CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    table: &Table,
    row: &[String],
    spec: &ColumnSpec,
    row_index: usize,
    spec_index: usize,
) -> Result<Value, CellError> {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => table.column_index(name),
    };

    let column = match column {
        Some(column) => column,
        None => {
            return Err(CellError {
                row: row_index,
                spec: spec_index,
                column: None,
                kind: CellErrorKind::UnknownColumn,
            });
        }
    };

    let cell = match row.get(column) {
        Some(cell) => cell,
        None => {
            return Err(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind: CellErrorKind::MissingCell,
            });
        }
    };

    if spec.nullable && cell.is_empty() {
        return Ok(Value::Null);
    }

    if cell.is_empty() && spec.kind != ColumnType::Text {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: Some(column),
            kind: CellErrorKind::Empty,
        });
    }

    let value = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };

    value.map_err(|kind| CellError {
        row: row_index,
        spec: spec_index,
        column: Some(column),
        kind,
    })
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                output.push(None);
                continue;
            }
            match evaluate_cell(table, row, spec, row_index, spec_index) {
                Ok(value) => output.push(Some(value)),
                Err(error) => {
                    output.push(None);
                    result.errors.push(error);
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => failed = true,
                        ErrorPolicy::Stop => {
                            result.rows.push(output);
                            return result;
                        }
                    }
                }
            }
        }
        result.rows.push(output);
    }

    result
}
