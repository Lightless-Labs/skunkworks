#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn numeric(
    bytes: &[u8],
    field: Field,
    offset: usize,
) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax_error(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(n)
}

fn endpoint(
    bytes: &[u8],
    field: Field,
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        let n = numeric(bytes, field, offset)?;
        let (min, max) = domain(field);
        if n < min || n > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok((n, true));
    }

    if let Some(list) = names(field) {
        if bytes.len() == 3 && bytes.iter().all(u8::is_ascii_alphabetic) {
            let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
            if let Some(i) = list.iter().position(|name| name.as_bytes() == upper) {
                let n = if field == Field::Month {
                    (i + 1) as u32
                } else {
                    i as u32
                };
                return Ok((n, false));
            }
        }
    }
    Err(syntax_error(field, offset))
}

fn parse_field(
    text: &str,
    field: Field,
    selected: &mut [bool],
) -> Result<bool, ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let unrestricted = matches!(field, Field::DayOfMonth | Field::DayOfWeek) && text == "*";
    let mut item_start = 0usize;

    for item in text.split(',') {
        let item_bytes = item.as_bytes();
        if item_bytes.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_count = item_bytes.iter().filter(|&&b| b == b'/').count();
        if slash_count > 1 {
            return Err(syntax_error(field, item_start));
        }
        let slash = item_bytes.iter().position(|&b| b == b'/');
        let base_end = slash.unwrap_or(item_bytes.len());
        let base = &item_bytes[..base_end];

        let (range_start, range_end, start_offset, end_offset, is_star) =
            if base == b"*" {
                (None, None, item_start, item_start, true)
            } else {
                let hyphens: Vec<usize> = base
                    .iter()
                    .enumerate()
                    .filter_map(|(i, &b)| (b == b'-').then_some(i))
                    .collect();
                if hyphens.len() > 1 {
                    return Err(syntax_error(field, item_start));
                }
                if let Some(&h) = hyphens.first() {
                    if h == 0 || h + 1 == base.len() {
                        return Err(syntax_error(field, item_start));
                    }
                    (
                        Some(&base[..h]),
                        Some(&base[h + 1..]),
                        item_start,
                        item_start + h + 1,
                        false,
                    )
                } else {
                    (Some(base), None, item_start, item_start, false)
                }
            };

        let step_bytes = if let Some(s) = slash {
            let part = &item_bytes[s + 1..];
            if part.is_empty() || !part.iter().all(u8::is_ascii_digit) {
                return Err(syntax_error(field, item_start));
            }
            Some((part, item_start + s + 1))
        } else {
            None
        };

        let mut start = None;
        let mut end = None;
        let mut start_numeric = false;
        let mut end_numeric = false;

        if !is_star {
            let (v, numeric_flag) =
                endpoint(range_start.unwrap(), field, start_offset)?;
            start = Some(v);
            start_numeric = numeric_flag;
            if let Some(end_bytes) = range_end {
                let (v, numeric_flag) = endpoint(end_bytes, field, end_offset)?;
                end = Some(v);
                end_numeric = numeric_flag;
            }
            if let (Some(a), Some(b)) = (start, end) {
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
            }
        }

        if let Some((step_part, step_offset)) = step_bytes {
            let step = numeric(step_part, field, step_offset)?;
            let width = domain(field).1 - domain(field).0 + 1;
            if step == 0 || step > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            let min = domain(field).0;
            let max = domain(field).1;
            let from = start.unwrap_or(min);
            let through = if is_star || range_end.is_some() {
                end.unwrap_or(max)
            } else {
                max
            };
            let mut n = from;
            while n <= through {
                selected[(n - min) as usize] = true;
                match n.checked_add(step) {
                    Some(next) => n = next,
                    None => break,
                }
            }
        } else if is_star {
            selected.fill(true);
        } else if let (Some(a), Some(b)) = (start, end) {
            for n in a..=b {
                selected[(n - domain(field).0) as usize] = true;
            }
        } else if let Some(a) = start {
            selected[(a - domain(field).0) as usize] = true;
        }

        let _ = (start_numeric, end_numeric);
        item_start += item.len() + 1;
    }
    Ok(unrestricted)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let source = if input.starts_with('@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let parts: Vec<&str> = source.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 31],
        vec![false; 12],
        vec![false; 7],
    ];
    let mut unrestricted_days = [false; 2];

    for (i, field) in FIELDS.iter().enumerate() {
        let unrestricted = parse_field(parts[i], *field, &mut sets[i])?;
        if i == 2 {
            unrestricted_days[0] = unrestricted;
        } else if i == 4 {
            unrestricted_days[1] = unrestricted;
        }
    }

    Ok(Cron {
        sets,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = domain(field);
        self.sets[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &v)| v.then_some((i as u32 + min) as u8))
            .collect()
    }

    fn selected(&self, field: Field, value: u8) -> bool {
        let (min, _) = domain(field);
        let index = value as u32 - min;
        self.sets[field_index(field)][index as usize]
    }

    fn day_matches(&self, date: CivilDateTime) -> bool {
        let dom = self.selected(Field::DayOfMonth, date.day);
        let dow = self.selected(Field::DayOfWeek, weekday(date.year, date.month, date.day));
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                let last = date.day == month_length(date.year, date.month);
                if last {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.selected(Field::Minute, at.minute)
            && self.selected(Field::Hour, at.hour)
            && self.selected(Field::Month, at.month)
            && self.day_matches(at)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            current = increment_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut current = before;
        loop {
            if current == CivilDateTime {
                year: 2000,
                month: 1,
                day: 1,
                hour: 0,
                minute: 0,
            } {
                return Ok(None);
            }
            current = decrement_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut current = start;
        while current < end {
            if self.matches_valid(current) {
                if fires.len() == cap {
                    return Ok(FireWindow {
                        fires,
                        truncated: true,
                    });
                }
                fires.push(current);
            }
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                break;
            }
            current = increment_minute(current);
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::new();
        for field in FIELDS {
            let (min, max) = domain(field);
            let idx = field_index(field);
            let full = self.sets[idx].iter().all(|&v| v);
            let rendered = if full {
                if field == Field::DayOfMonth && !self.unrestricted_days[0] {
                    "*/1".to_string()
                } else if field == Field::DayOfWeek && !self.unrestricted_days[1] {
                    "*/1".to_string()
                } else {
                    "*".to_string()
                }
            } else if (field == Field::DayOfMonth && self.unrestricted_days[0])
                || (field == Field::DayOfWeek && self.unrestricted_days[1])
            {
                "*".to_string()
            } else {
                self.allowed(field)
                    .into_iter()
                    .map(|v| v.to_string())
                    .collect::<Vec<_>>()
                    .join(",")
            };
            let _ = (min, max);
            fields.push(rendered);
        }
        fields.join(" ")
    }
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < month_length(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

fn decrement_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute > 0 {
        at.minute -= 1;
        return at;
    }
    at.minute = 59;
    if at.hour > 0 {
        at.hour -= 1;
        return at;
    }
    at.hour = 23;
    if at.day > 1 {
        at.day -= 1;
        return at;
    }
    if at.month > 1 {
        at.month -= 1;
    } else {
        at.month = 12;
        at.year -= 1;
    }
    at.day = month_length(at.year, at.month);
    at
}
