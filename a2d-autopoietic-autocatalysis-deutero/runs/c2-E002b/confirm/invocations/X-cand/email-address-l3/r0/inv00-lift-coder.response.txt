#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut first_retained = true;
    let mut quote_protected = false;
    let mut quote_closed = false;

    while i < input.len() {
        if quote_protected && !quote_closed {
            let b = input[i];
            out.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    i += 1;
                    out.push(input[i]);
                    positions.push(i);
                }
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            let b = input[i];
            if b == b'\\' {
                let slash = i;
                if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: slash });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !is_printable(b) && !is_hws(b) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let b = input[i];
        if b == b'\r' || b == b'\n' {
            if let Some(end) = is_fold(input, i) {
                let start = i;
                i = end;
                runs.push((start, i));
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }
        if is_hws(b) {
            let start = i;
            i += 1;
            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }
            runs.push((start, i));
            continue;
        }
        if b == b'(' {
            outer_open = i;
            depth = 1;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if first_retained && b == b'"' {
            quote_protected = true;
        }
        first_retained = false;
        out.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }

    let mut at_offsets = Vec::new();
    for (idx, &b) in out.iter().enumerate() {
        if b == b'@' {
            at_offsets.push(idx);
        }
    }

    for &(start, end) in &runs {
        let run_start_idx = positions.partition_point(|&p| p < end);
        let left_at = run_start_idx > 0 && out[run_start_idx - 1] == b'@';
        let right_at = run_start_idx < out.len() && out[run_start_idx] == b'@';
        if run_start_idx != 0 && run_start_idx != out.len() && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    let _ = at_offsets;
    Ok(Cleaned { bytes: out, positions, runs })
}

fn original_pos(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let clean = cleanup(raw)?;
    let bytes = &clean.bytes;
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    if bytes[0] == b'"' {
        let mut i = 1usize;
        let mut closing = None;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                closing = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        let Some(end) = closing else {
            return Err(EmailError::UnclosedQuote {
                at: original_pos(&clean, 0, raw.len()),
            });
        };
        protected_end = end + 1;
    }

    let separators: Vec<usize> = (0..bytes.len())
        .filter(|&i| bytes[i] == b'@' && i >= protected_end)
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: original_pos(&clean, separators[1], raw.len()),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: original_pos(&clean, sep, raw.len()),
        });
    }

    let local_bytes = &bytes[..sep];
    let mut local = String::new();
    let quoted;

    if local_bytes[0] == b'"' {
        quoted = true;
        let closing = protected_end - 1;
        if closing + 1 != local_bytes.len() {
            return Err(EmailError::TrailingLocal {
                at: original_pos(&clean, closing + 1, raw.len()),
            });
        }

        let mut origins = Vec::new();
        let mut i = 1usize;
        while i < closing {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= closing
                    || !matches!(local_bytes[i + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: original_pos(&clean, i, raw.len()),
                    });
                }
                local.push(local_bytes[i + 1] as char);
                origins.push(original_pos(&clean, i, raw.len()));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: original_pos(&clean, i, raw.len()),
                    });
                }
                local.push(b as char);
                origins.push(original_pos(&clean, i, raw.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: original_pos(&clean, closing, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: original_pos(&clean, 64, raw.len()),
            });
        }
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: original_pos(&clean, i, raw.len()),
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: original_pos(&clean, i, raw.len()),
                });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
    }

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let mut domain_end = bytes.len();
    let mut root_pos = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        root_pos = Some(domain_end - 1);
        domain_end -= 1;
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: original_pos(&clean, domain_start, raw.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: original_pos(&clean, domain_start + 253, raw.len()),
        });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0usize;
    for i in 0..=domain_bytes.len() {
        if i == domain_bytes.len() || domain_bytes[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }

    for (label_start, label_end) in &labels {
        if label_start == label_end {
            let pos = if *label_start == domain_bytes.len() {
                root_pos.map(|p| original_pos(&clean, p, raw.len()))
                    .unwrap_or_else(|| original_pos(&clean, domain_end, raw.len()))
            } else {
                original_pos(&clean, domain_start + *label_start, raw.len())
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        if label_end - label_start > 63 {
            return Err(EmailError::LabelTooLong {
                at: original_pos(&clean, domain_start + label_start + 63, raw.len()),
            });
        }
        if domain_bytes[*label_start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&clean, domain_start + *label_start, raw.len()),
            });
        }
        if domain_bytes[*label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&clean, domain_start + *label_end - 1, raw.len()),
            });
        }
        for i in *label_start..*label_end {
            let b = domain_bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: original_pos(&clean, domain_start + i, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: original_pos(&clean, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if domain_bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: original_pos(&clean, domain_start + last_start, raw.len()),
        });
    }

    let domain = String::from_utf8(domain_bytes.to_vec())
        .unwrap()
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
