#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Byte {
    value: u8,
    pos: usize,
}

struct Cleaned {
    bytes: Vec<Byte>,
    runs: Vec<(usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_protected = false;

    while i < input.len() {
        if quote_protected {
            let b = input[i];
            bytes.push(Byte { value: b, pos: i });
            i += 1;
            if b == b'\\' {
                if i < input.len() {
                    bytes.push(Byte { value: input[i], pos: i });
                    i += 1;
                }
            } else if b == b'"' {
                quote_protected = false;
            }
            continue;
        }

        let b = input[i];
        if b == b'"' && bytes.is_empty() {
            quote_protected = true;
            bytes.push(Byte { value: b, pos: i });
            i += 1;
            continue;
        }

        let is_cfws_start = is_hws(b)
            || b == b'\r'
            || b == b'\n'
            || b == b'(';

        if !is_cfws_start && b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if is_cfws_start {
            let start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let b = input[i];
                if is_hws(b) {
                    i += 1;
                    continue;
                }
                if b == b'\r' || b == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if b == b'(' {
                    let outer = i;
                    let mut depth = 0usize;
                    while i < input.len() {
                        let c = input[i];
                        if c == b'\\' {
                            let slash = i;
                            if i + 1 >= input.len()
                                || !(0x20..=0x7e).contains(&input[i + 1])
                            {
                                return Err(EmailError::CommentEscape { at: slash });
                            }
                            i += 2;
                            continue;
                        }
                        if c == b'\r' || c == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                                continue;
                            }
                            return Err(EmailError::BadFold { at: i });
                        }
                        if c == b'(' {
                            depth += 1;
                            if depth > 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            i += 1;
                            continue;
                        }
                        if c == b')' {
                            depth -= 1;
                            i += 1;
                            if depth == 0 {
                                break;
                            }
                            continue;
                        }
                        if c == b'\t' || (0x20..=0x7e).contains(&c) {
                            i += 1;
                            continue;
                        }
                        return Err(EmailError::CommentChar { at: i });
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    continue;
                }
                break;
            }
            runs.push((start, i));
            continue;
        }

        bytes.push(Byte { value: b, pos: i });
        i += 1;
    }

    Ok(Cleaned { bytes, runs })
}

fn pos_at(bytes: &[Byte], idx: usize, input_len: usize) -> usize {
    bytes.get(idx).map_or(input_len, |b| b.pos)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;

    for &(start, end) in &cleaned.runs {
        let left_is_at = start > 0 && bytes.get(start - 1).is_some_and(|b| b.value == b'@');
        let right_is_at = end < bytes.len() && bytes.get(end).is_some_and(|b| b.value == b'@');
        if start != 0 && end != raw.len() && !left_is_at && !right_is_at {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0].value == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            match bytes[i].value {
                b'\\' => i += 2,
                b'"' => {
                    quote_end = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: bytes[0].pos });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i].value == b'@' && !(quote_end.is_some() && i <= protected_end) {
            separators.push(i);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: bytes[separators[1]].pos,
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: bytes[sep].pos,
        });
    }

    let local_bytes = &bytes[..sep];
    let (local, quoted) = if local_bytes[0].value == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: bytes[end + 1].pos,
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < end {
            let b = bytes[i].value;
            if b == b'\\' {
                let next = bytes.get(i + 1).map(|x| x.value);
                match next {
                    Some(b'"' | b'\\' | b'@') => {
                        decoded.push(next.unwrap());
                        origins.push(bytes[i].pos);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: bytes[i].pos }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: bytes[i].pos });
                }
                decoded.push(b);
                origins.push(bytes[i].pos);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: bytes[end].pos,
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: bytes[64].pos,
            });
        }
        for i in 0..local_bytes.len() {
            let b = local_bytes[i].value;
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes[i - 1].value == b'.'
                {
                    return Err(EmailError::LocalDot {
                        at: local_bytes[i].pos,
                    });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar {
                    at: local_bytes[i].pos,
                });
            }
        }
        (
            String::from_utf8(local_bytes.iter().map(|b| b.value).collect()).unwrap(),
            false,
        )
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let mut removed_root = None;
    if domain_end > domain_start && bytes[domain_end - 1].value == b'.' {
        domain_end -= 1;
        removed_root = Some(bytes[domain_end].pos);
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos_at(bytes, domain_start, raw.len()),
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: bytes[domain_start + 253].pos,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if bytes[i].value == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    labels.push((start, domain_end));

    for (label_start, label_end) in &labels {
        if label_start == label_end {
            let at = if *label_start == domain_start {
                pos_at(bytes, *label_start, raw.len())
            } else if *label_start == domain_end {
                removed_root.unwrap_or_else(|| pos_at(bytes, *label_start, raw.len()))
            } else {
                bytes[*label_start - 1].pos
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if label_end - label_start > 63 {
            return Err(EmailError::LabelTooLong {
                at: bytes[label_start + 63].pos,
            });
        }
        if bytes[*label_start].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: bytes[*label_start].pos,
            });
        }
        if bytes[label_end - 1].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: bytes[label_end - 1].pos,
            });
        }
        for i in *label_start..*label_end {
            let b = bytes[i].value;
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: bytes[i].pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: pos_at(bytes, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if bytes[last_start..last_end]
        .iter()
        .all(|b| b.value.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: bytes[last_start].pos,
        });
    }

    let domain = String::from_utf8(
        bytes[domain_start..domain_end]
            .iter()
            .map(|b| b.value.to_ascii_lowercase())
            .collect(),
    )
    .unwrap();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
