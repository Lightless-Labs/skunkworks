fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value: u32 = 0;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    let (min, max) = domain(field);
    if is_step {
        let width = max - min + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else if value < min || value > max {
        return Err(ParseError::OutOfRange {
            field,
            offset: start,
        });
    }

    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    let item = &bytes[item_start..item_end];

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    let hyphen_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    if slash_positions.len() > 1 || hyphen_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let slash = slash_positions.first().copied();
    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];

    let is_star = base == b"*";
    if is_star {
        if !hyphen_positions.is_empty() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    } else if base.is_empty() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    } else if let Some(&hyphen) = hyphen_positions.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        if base[..hyphen].iter().any(|b| !b.is_ascii_digit())
            || base[hyphen + 1..].iter().any(|b| !b.is_ascii_digit())
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    } else if base.iter().any(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let step = if let Some(slash_index) = slash {
        let step_start = item_start + slash_index + 1;
        let step_end = item_end;
        if step_start == step_end
            || item[slash_index + 1..]
                .iter()
                .any(|b| !b.is_ascii_digit())
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        Some(parse_number(
            bytes,
            step_start,
            step_end,
            field,
            true,
        )?)
    } else {
        None
    };

    let (min, max) = domain(field);
    let (start, end, is_range) = if is_star {
        (min, max, true)
    } else if let Some(&hyphen) = hyphen_positions.first() {
        let start_pos = item_start;
        let end_pos = item_start + hyphen + 1;
        let start = parse_number(bytes, start_pos, start_pos + hyphen, field, false)?;
        let end = parse_number(bytes, end_pos, item_end, field, false)?;
        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: end_pos,
            });
        }
        (start, end, true)
    } else {
        let value = parse_number(bytes, item_start, item_start + base_end, field, false)?;
        (value, if step.is_some() { max } else { value }, false)
    };

    let effective_end = if is_star || is_range {
        end
    } else {
        end
    };
    let stride = step.unwrap_or(1);
    let mut value = start;
    while value <= effective_end {
        selected[value as usize] = true;
        match value.checked_add(stride) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(input: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = [false; 60];
    let mut item_start = 0;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            if i == item_start {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: item_start,
                });
            }
            parse_item(bytes, item_start, i, field, &mut selected)?;
            item_start = i + 1;
        }
    }

    if item_start == bytes.len() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }
    parse_item(bytes, item_start, bytes.len(), field, &mut selected)?;

    let (min, max) = domain(field);
    Ok((min..=max)
        .filter(|&value| selected[value as usize])
        .map(|value| value as u8)
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];

    for i in 0..5 {
        selected[i] = parse_field(parts[i], fields[i])?;
    }

    Ok(Cron {
        selected,
        day_wildcards: [parts[2] == "*", parts[4] == "*"],
    })
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    let error = |kind: fn(Field, usize) -> ParseError| kind(field, start);

    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(|field, offset| ParseError::LeadingZero { field, offset }));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| error(|field, offset| ParseError::NumberOverflow { field, offset }))?;
    }

    if is_step {
        let (_, max) = field_bounds(field);
        if value == 0 || value > max + 1 {
            return Err(error(|field, offset| ParseError::BadStep { field, offset }));
        }
    } else {
        let (min, max) = field_bounds(field);
        if value < min || value > max {
            return Err(error(|field, offset| ParseError::OutOfRange { field, offset }));
        }
    }

    Ok(value)
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (domain_min, domain_max) = field_bounds(field);
    let mut selected = vec![false; (domain_max + 1) as usize];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_part) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some(slash + 1)),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let base = &item[..base_end];
        let hyphen_positions: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        enum Base {
            Star,
            Single(usize, usize),
            Range(usize, usize, usize, usize),
        }

        let parsed_base = if base == b"*" {
            Base::Star
        } else {
            match hyphen_positions.as_slice() {
                [] if !base.is_empty() => Base::Single(0, base.len()),
                [hyphen] if *hyphen > 0 && *hyphen + 1 < base.len() => {
                    Base::Range(0, *hyphen, *hyphen + 1, base.len())
                }
                _ => {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
            }
        };

        if step_part.is_some() && matches!(parsed_base, Base::Star) == false && base.is_empty() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let mut step = None;
        if let Some(step_start) = step_part {
            if step_start == item.len() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            if item[step_start..].contains(&b'-') || item[step_start..].contains(&b'/') {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            let value = parse_number(
                item,
                step_start,
                item.len(),
                field,
                true,
            )?;
            step = Some(value);
        }

        let (start, end) = match parsed_base {
            Base::Star => (domain_min, domain_max),
            Base::Single(number_start, number_end) => {
                let value = parse_number(item, number_start, number_end, field, false)?;
                let end = if step.is_some() { domain_max } else { value };
                (value, end)
            }
            Base::Range(start_start, start_end, end_start, end_end) => {
                let start = parse_number(item, start_start, start_end, field, false)?;
                let end = parse_number(item, end_start, end_end, field, false)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + end_start,
                    });
                }
                (start, end)
            }
        };

        let increment = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {

    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed = Vec::with_capacity(5);
    for (field, text) in field_kinds.into_iter().zip(fields.iter().copied()) {
        parsed.push(parse_field(field, text)?);

        selected: parsed.try_into().expect("exactly five fields"),
        day_of_month_exact_star: fields[2] == "*",
        day_of_week_exact_star: fields[4] == "*",
    })
}

    field: Field,
    offset: usize,

    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {

        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value: u32 = 0;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))

                offset: start,

            field,
            offset: start,
        });
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let min = field_min(field);
    let max = field_max(field);
    let mut selected = vec![false; max - min + 1];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();

        let (base_end, step_start) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some(*slash + 1)),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        enum Base {
            Star,
            Single(usize, usize),
            Range(usize, usize, usize, usize),
        }

        let parsed_base = if base == b"*" {
            Base::Star
        } else if hyphens.is_empty() && !base.is_empty() {
            Base::Single(0, base.len())
        } else if let [hyphen] = hyphens.as_slice() {
            if *hyphen == 0 || *hyphen + 1 == base.len() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Base::Range(0, *hyphen, *hyphen + 1, base.len())
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        let step = if let Some(start) = step_start {
            if start == item.len() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Some(parse_number(
                item,
                start,
                item.len(),
                field,
                true,
            )? as usize)
        } else {
            None
        };

        let (start_value, end_value) = match parsed_base {
            Base::Star => (min, max),
            Base::Single(start, end) => {
                let value = parse_number(
                    base,
                    start,
                    end,
                    field,
                    false,
                )? as usize;
                (value, if step.is_some() { max } else { value })
            }
            Base::Range(start_start, start_end, end_start, end_end) => {
                let start_value = parse_number(
                    base,
                    start_start,
                    start_end,
                    field,
                    false,
                )? as usize;
                let end_value = parse_number(
                    base,
                    end_start,
                    end_end,
                    field,
                    false,
                )? as usize;
                if start_value > end_value {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + end_start,
                    });
                }
                (start_value, end_value)
            }
        };

        let stride = step.unwrap_or(1);
        let mut value = start_value;
        while value <= end_value {
            selected[value - min] = true;
            match value.checked_add(stride) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        parsed[i] = parse_field(pieces[i], fields[i])?;
    }

    Ok(Cron {
        selected: parsed,
        day_wildcards: [pieces[2] == "*", pieces[4] == "*"],
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<u8>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_info(index: usize) -> (Field, u32, u32, u32) {
    match index {
        0 => (Field::Minute, 0, 59, 60),
        1 => (Field::Hour, 0, 23, 24),
        2 => (Field::DayOfMonth, 1, 31, 31),
        3 => (Field::Month, 1, 12, 12),
        _ => (Field::DayOfWeek, 0, 6, 7),
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let mut sets: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_stars = [false; 2];

    for (index, text) in fields.iter().enumerate() {
        let (field, min, max, width) = field_info(index);
        let bytes = text.as_bytes();

        for (offset, &byte) in bytes.iter().enumerate() {
            if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if index == 2 {
            day_stars[0] = *text == "*";
        } else if index == 4 {
            day_stars[1] = *text == "*";
        }

        let mut selected = vec![false; (max - min + 1) as usize];
        let mut item_start = 0;

        for item in text.split(',') {
            let item_bytes = item.as_bytes();
            if item_bytes.is_empty() {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: item_start,
                });
            }

            parse_item(
                item_bytes,
                item_start,
                field,
                min,
                max,
                width,
                &mut selected,
            )?;

            item_start += item_bytes.len() + 1;
        }

        sets[index] = selected
            .iter()
            .enumerate()
            .filter_map(|(i, &is_selected)| {
                if is_selected {
                    Some((min as usize + i) as u8)
                } else {
                    None
                }
            })
            .collect();
    }

    Ok(Cron { sets, day_stars })
}

fn parse_item(
    item: &[u8],
    offset: usize,
    field: Field,
    min: u32,
    max: u32,
    width: u32,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();

    let (base, step_part) = match slash_positions.as_slice() {
        [] => (item, None),
        [slash] => (&item[..*slash], Some(&item[*slash + 1..])),
        _ => {
            return Err(ParseError::InvalidSyntax {
                field,
                offset,
            });
        }
    };

    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
        .collect();

    enum Base {
        Star,
        Single((u32, usize)),
        Range((u32, usize), (u32, usize)),
    }

    let parsed_base = if is_star {
        Base::Star
    } else {
        match hyphens.as_slice() {
            [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                Base::Single((0, 0))
            }
            [hyphen]
                if *hyphen > 0
                    && *hyphen + 1 < base.len()
                    && base[..*hyphen].iter().all(u8::is_ascii_digit)
                    && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
            {
                Base::Range((0, 0), (0, 0))
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset,
                });
            }
        }
    };

    let parsed_base = match parsed_base {
        Base::Star => Base::Star,
        Base::Single(_) => Base::Single((parse_number(base, offset, field)?, offset)),
        Base::Range(_, _) => {
            let hyphen = hyphens[0];
            let start = parse_number(&base[..hyphen], offset, field)?;
            let end_offset = offset + hyphen + 1;
            let end = parse_number(&base[hyphen + 1..], end_offset, field)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            Base::Range((start, offset), (end, end_offset))
        }
    };

    let step = match step_part {
        None => None,
        Some(part) => {
            if part.is_empty() || !part.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax { field, offset });
            }
            let step_offset = offset + base.len() + 1;
            let value = parse_number(part, step_offset, field)?;
            if value < 1 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            Some(value)
        }
    };

    let (start, end) = match parsed_base {
        Base::Star => (min, max),
        Base::Single((value, _)) => {
            if value < min || value > max {
                return Err(ParseError::OutOfRange { field, offset });
            }
            match step {
                Some(_) => (value, max),
                None => (value, value),
            }
        }
        Base::Range((start, start_offset), (end, end_offset)) => {
            if start < min || start > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start_offset,
                });
            }
            if end < min || end > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_offset,
                });
            }
            (start, end)
        }
    };

    let increment = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(increment) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_number(bytes: &[u8], offset: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    Ok(value)
}
