```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let path = path.split(['?', '#']).next().unwrap_or("");
        let parsed = parse_path(path)?;
        let mut best: Option<(Vec<u8>, usize, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(route, &parsed.segments) {
                match &best {
                    None => best = Some((route.rank.clone(), route.id, params)),
                    Some((rank, id, _)) => {
                        if route.rank > *rank || (route.rank == *rank && route.id < *id) {
                            best = Some((route.rank.clone(), route.id, params));
                        }
                    }
                }
            }
        }

        best.map(|(_, route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &input[start..end];
        let first = bytes[start];

        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if first == b'*' {
                if end < bytes.len() && end + 1 < bytes.len() {
                    return Err(Error::InvalidPattern { at: start });
                }
                result.push(Segment::Wildcard(name.to_owned()));
            } else {
                result.push(Segment::Param(name.to_owned()));
            }
        } else {
            result.push(Segment::Static(decode_segment(input, start, end, true)?));
        }

        if end == bytes.len() {
            break;
        }

        if end + 1 == bytes.len() {
            return Err(Error::InvalidPattern { at: end });
        }

        start = end + 1;
    }

    Ok(result)
}

fn parse_path(input: &str) -> Result<ParsedPath, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(ParsedPath { segments: Vec::new() });
    }

    let bytes = input.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(input, start, end, false)?);

        if end == bytes.len() {
            break;
        }
        if end + 1 == bytes.len() {
            return Err(Error::InvalidPath { at: end });
        }
        start = end + 1;
    }

    Ok(ParsedPath { segments })
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|&b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn decode_segment(input: &str, start: usize, end: usize, pattern: bool) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut output = Vec::new();
    let mut i = start;

    while i < end {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= end {
                return invalid(pattern, i);
            }
            let hi = hex(bytes[i + 1]);
            let lo = hex(bytes[i + 2]);
            let Some((hi, lo)) = hi.zip(lo) else {
                return invalid(pattern, i);
            };
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return invalid(pattern, i);
            }
            output.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            output.push(b);
            i += 1;
        } else {
            return invalid(pattern, i);
        }
    }

    Ok(String::from_utf8(output).expect("decoded segments are ASCII"))
}

fn invalid<T>(pattern: bool, at: usize) -> Result<T, Error> {
    if pattern {
        Err(Error::InvalidPattern { at })
    } else {
        Err(Error::InvalidPath { at })
    }
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
```
