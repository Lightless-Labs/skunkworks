#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    text: String,
    pieces: usize,
    gap: usize,
    marker_ended: bool,
    consumed: usize,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_count: usize,
    lines: Vec<Line>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn line_vector_cmp(a: &[Line], b: &[Line]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b) {
        match y.consumed.cmp(&x.consumed) {
            std::cmp::Ordering::Equal => {}
            other => return other,
        }
    }
    match a.len().cmp(&b.len()) {
        std::cmp::Ordering::Equal => {}
        other => return other,
    }
    for (x, y) in a.iter().zip(b) {
        match x.text.cmp(&y.text) {
            std::cmp::Ordering::Equal => {}
            other => return other,
        }
    }
    std::cmp::Ordering::Equal
}

fn better(a: &Layout, b: &Layout) -> bool {
    a.cost < b.cost
        || (a.cost == b.cost
            && (a.marker_count < b.marker_count
                || (a.marker_count == b.marker_count
                    && (a.lines.len() < b.lines.len()
                        || (a.lines.len() == b.lines.len()
                            && line_vector_cmp(&a.lines, &b.lines)
                                == std::cmp::Ordering::Less)))))
}

fn solve(
    words: &[Word],
    wi: usize,
    off: usize,
    first: bool,
    opts: &WrapOptions,
    memo: &mut Vec<Vec<[Option<Layout>; 2]>>,
) -> Layout {
    if wi == words.len() {
        return Layout {
            cost: 0,
            marker_count: 0,
            lines: Vec::new(),
        };
    }
    let fi = if first { 1 } else { 0 };
    if let Some(v) = &memo[wi][off][fi] {
        return v.clone();
    }

    let indent = if first {
        opts.first_indent
    } else {
        opts.rest_indent
    };
    let cap = opts.width - indent;
    let mut best: Option<Layout> = None;
    let mut rendered = String::new();
    let mut used = 0usize;
    let mut consumed = 0usize;
    let mut pieces = 0usize;

    // Enumerate lines that complete one or more consecutive words.
    let mut j = wi;
    let mut start = off;
    loop {
        let word = &words[j];
        let remain = word.chars.len() - start;
        if used > 0 {
            if used + 1 > cap {
                break;
            }
            rendered.push(' ');
            used += 1;
        }
        let take = remain.min(cap - used);
        if take == 0 {
            break;
        }
        let complete = take == remain;
        if complete {
            for c in &word.chars[start..] {
                rendered.push(*c);
            }
            used += remain;
            consumed += remain;
            pieces += 1;

            let next_wi = j + 1;
            let tail = solve(words, next_wi, 0, false, opts, memo);
            let gap = opts.width - indent - used;
            let mut candidate = tail;
            let is_final = next_wi == words.len();
            candidate.cost += if is_final {
                (gap * gap / 4) as u64
            } else {
                (gap * gap) as u64 + if pieces == 1 { 3 } else { 0 }
            };
            let line = Line {
                indent,
                text: format!("{}{}", " ".repeat(indent), rendered),
                pieces,
                gap,
                marker_ended: false,
                consumed,
            };
            candidate.lines.insert(0, line);
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
            if next_wi == words.len() || used >= cap {
                break;
            }
            j += 1;
            start = 0;
            continue;
        }
        break;
    }

    // Marker-ended lines within the current word.
    let word = &words[wi];
    for &m in &word.markers {
        if m <= off || m >= word.chars.len() {
            continue;
        }
        let n = m - off;
        if n + 1 > cap {
            continue;
        }
        let prefix: String = word.chars[off..m].iter().collect();
        let line_content = format!("{prefix}-");
        let tail = solve(words, wi, m, false, opts, memo);
        let gap = cap - (n + 1);
        let mut candidate = tail;
        candidate.cost += (gap * gap) as u64 + opts.hyphen_cost as u64;
        candidate.marker_count += 1;
        let mut text = " ".repeat(indent);
        text.push_str(&line_content);
        candidate.lines.insert(
            0,
            Line {
                indent,
                text,
                pieces: 1,
                gap,
                marker_ended: true,
                consumed: n,
            },
        );
        if best.as_ref().map_or(true, |b| better(&candidate, b)) {
            best = Some(candidate);
        }
    }

    // Raw split is allowed only on an otherwise empty line and only when no
    // remaining marker prefix plus hyphen fits.
    let remain = word.chars.len() - off;
    if remain > cap {
        let marker_fits = word
            .markers
            .iter()
            .any(|&m| m > off && m < word.chars.len() && m - off + 1 <= cap);
        if !marker_fits {
            let prefix: String = word.chars[off..off + cap].iter().collect();
            let tail = solve(words, wi, off + cap, false, opts, memo);
            let gap = 0;
            let mut candidate = tail;
            candidate.cost += (2 * opts.width + 1) as u64;
            let mut text = " ".repeat(indent);
            text.push_str(&prefix);
            candidate.lines.insert(
                0,
                Line {
                    indent,
                    text,
                    pieces: 1,
                    gap,
                    marker_ended: false,
                    consumed: cap,
                },
            );
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }
    }

    let result = best.unwrap_or_else(|| Layout {
        cost: u64::MAX / 4,
        marker_count: usize::MAX / 4,
        lines: Vec::new(),
    });
    memo[wi][off][fi] = Some(result.clone());
    result
}

fn layout_paragraph(words: &[Word], opts: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut memo: Vec<Vec<[Option<Layout>; 2]>> = words
        .iter()
        .map(|w| vec![[None, None]; w.chars.len() + 1])
        .collect();
    let layout = solve(words, 0, 0, true, opts, &mut memo);
    let mut output = Vec::new();
    for (i, line) in layout.lines.iter().enumerate() {
        let final_line = i + 1 == layout.lines.len();
        if opts.justify && !final_line && line.pieces >= 2 {
            let mut parts = line.text[line.indent..].split(' ');
            let mut words = Vec::new();
            if let Some(first) = parts.next() {
                words.push(first.to_string());
                words.extend(parts.map(str::to_string));
            }
            let gaps = words.len().saturating_sub(1);
            let extra = opts.width.saturating_sub(line.text.chars().count());
            if gaps > 0 {
                let each = extra / gaps;
                let rem = extra % gaps;
                let mut s = " ".repeat(line.indent);
                for (g, w) in words.iter().enumerate() {
                    if g > 0 {
                        let start = (i % gaps);
                        let cyclic = (g - 1 + gaps - start) % gaps;
                        s.push_str(&" ".repeat(1 + each + usize::from(cyclic < rem)));
                    }
                    s.push_str(w);
                }
                output.push(s);
                continue;
            }
        }
        output.push(line.text.clone());
    }
    output
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let invalid = i == 0
                || i + 1 == chars.len()
                || chars[i - 1].1 == '~'
                || chars[i + 1].1 == '~'
                || is_separator(chars[i - 1].1)
                || is_separator(chars[i + 1].1);
            if invalid {
                return Err(WrapError::InvalidMarker {
                    index: chars[i].0,
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i].1 == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(chars[i].1);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let mut q = p;
            while q < normalized.len() && normalized[q] == '\n' {
                q += 1;
            }
            if q - p >= 2 {
                let words = parse_words(&segment);
                output.extend(layout_paragraph(&words, options));
                output.extend(std::iter::repeat(String::new()).take(q - p - 1));
                segment.clear();
                p = q;
                continue;
            }
        }
        segment.push(normalized[p]);
        p += 1;
    }
    let words = parse_words(&segment);
    output.extend(layout_paragraph(&words, options));
    Ok(output)
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Vec::new();
    let mut markers = Vec::new();
    for &c in chars {
        if is_separator(c) {
            if !current.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(current.len());
        } else {
            current.push(c);
        }
    }
    if !current.is_empty() {
        words.push(Word {
            chars: current,
            markers,
        });
    }
    words
}
