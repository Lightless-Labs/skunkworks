#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    displayed: String,
    consumed: usize,
    marker_ended: bool,
    raw: bool,
    cost: u64,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_count: usize,
    lines: Vec<Line>,
    consumed_vector: Vec<usize>,
    strings: Vec<String>,
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_count != b.marker_count {
        return a.marker_count < b.marker_count;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed_vector != b.consumed_vector {
        return a.consumed_vector > b.consumed_vector;
    }
    a.strings < b.strings
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    use std::collections::HashMap;

    fn rec(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Solution>,
    ) -> Solution {
        if wi == words.len() {
            return Solution {
                cost: 0,
                marker_count: 0,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                strings: Vec::new(),
            };
        }
        if let Some(s) = memo.get(&(wi, off, first)) {
            return s.clone();
        }

        let indent = if first { options.first_indent } else { options.rest_indent };
        let cap = options.width - indent;
        let mut best: Option<Solution> = None;

        let mut consider = |end_wi: usize,
                            end_off: usize,
                            pieces: Vec<String>,
                            displayed: String,
                            consumed: usize,
                            marker_ended: bool,
                            raw: bool| {
            let tail = rec(end_wi, end_off, false, words, options, memo);
            let scalar_width = indent + displayed.chars().count();
            let gap = options.width - scalar_width;
            let mut line_cost = if tail.lines.is_empty() {
                (gap as u64 * gap as u64) / 4
            } else {
                gap as u64 * gap as u64
            };
            if marker_ended {
                line_cost += options.hyphen_cost as u64;
            }
            if raw {
                line_cost += (2 * options.width + 1) as u64;
            }
            if !tail.lines.is_empty() && pieces.len() == 1 {
                line_cost += 3;
            }

            let mut line_text = " ".repeat(indent);
            line_text.push_str(&displayed);
            let line = Line {
                indent,
                pieces,
                displayed,
                consumed,
                marker_ended,
                raw,
                cost: line_cost,
            };
            let mut candidate = Solution {
                cost: line_cost + tail.cost,
                marker_count: usize::from(marker_ended) + tail.marker_count,
                lines: Vec::with_capacity(1 + tail.lines.len()),
                consumed_vector: Vec::with_capacity(1 + tail.consumed_vector.len()),
                strings: Vec::with_capacity(1 + tail.strings.len()),
            };
            candidate.lines.push(line);
            candidate.lines.extend(tail.lines);
            candidate.consumed_vector.push(consumed);
            candidate.consumed_vector.extend(tail.consumed_vector);
            candidate.strings.push(line_text);
            candidate.strings.extend(tail.strings);

            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                best = Some(candidate);
            }
        };

        let mut display = String::new();
        let mut pieces = Vec::new();
        let mut used = 0usize;
        let mut cur_wi = wi;
        let mut cur_off = off;

        loop {
            let word = &words[cur_wi];
            let remaining = word.chars.len() - cur_off;
            let separator = usize::from(!pieces.is_empty());
            if used + separator >= cap {
                break;
            }
            let available = cap - used - separator;

            let mut prefix = String::new();
            for (j, &ch) in word.chars[cur_off..].iter().enumerate().take(available) {
                prefix.push(ch);
                let end_off = cur_off + j + 1;
                let is_word_end = end_off == word.chars.len();
                let marker_here = word.markers.binary_search(&end_off).is_ok();

                if marker_here && !is_word_end && prefix.chars().count() + 1 <= available {
                    let mut shown = display.clone();
                    if !pieces.is_empty() {
                        shown.push(' ');
                    }
                    shown.push_str(&prefix);
                    shown.push('-');
                    let mut p = pieces.clone();
                    p.push(prefix.clone());
                    consider(
                        cur_wi,
                        end_off,
                        p,
                        shown,
                        used + separator + prefix.chars().count(),
                        true,
                        false,
                    );
                }

                if is_word_end {
                    let mut shown = display.clone();
                    if !pieces.is_empty() {
                        shown.push(' ');
                    }
                    shown.push_str(&prefix);
                    let mut p = pieces.clone();
                    p.push(prefix.clone());
                    let next_wi = cur_wi + 1;
                    consider(
                        next_wi,
                        0,
                        p,
                        shown,
                        used + separator + prefix.chars().count(),
                        false,
                        false,
                    );
                }
            }

            if remaining > available {
                let marker_fits = word.markers.iter().any(|&m| {
                    m > cur_off && m < word.chars.len() && m - cur_off + 1 <= available
                });
                if pieces.is_empty() && !marker_fits {
                    let raw_chars: String = word.chars[cur_off..cur_off + available].iter().collect();
                    consider(
                        cur_wi,
                        cur_off + available,
                        vec![raw_chars.clone()],
                        raw_chars,
                        available,
                        false,
                        true,
                    );
                }
                break;
            }

            used += separator + remaining;
            if !pieces.is_empty() {
                display.push(' ');
            }
            display.push_str(&word.chars[cur_off..].iter().collect::<String>());
            pieces.push(word.chars[cur_off..].iter().collect());
            cur_wi += 1;
            cur_off = 0;
            if cur_wi == words.len() || used >= cap {
                break;
            }
        }

        let result = best.expect("raw splitting guarantees a continuation");
        memo.insert((wi, off, first), result.clone());
        result
    }

    let mut memo = HashMap::new();
    rec(0, 0, true, words, options, &mut memo).lines
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if ch == '~' {
            markers.push(chars.len());
        } else if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else {
            chars.push(ch);
        }
    }
    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00a0}'
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let count = input.chars().count();
    if count > 512 {
        return Err(WrapError::InputTooLong { chars: count });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_side = |j: isize| -> bool {
                if j < 0 || j as usize >= chars.len() {
                    return false;
                }
                let ch = chars[j as usize].1;
                ch != '~' && !is_separator(ch)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let ch = chars[i].1;
        if ch == '\r' && i + 1 < chars.len() && chars[i + 1].1 == '\n' {
            normalized.push('\n');
            i += 2;
        } else {
            normalized.push(if ch == '\r' { '\n' } else { ch });
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut idx = 0usize;
    while idx < normalized.len() {
        if normalized[idx] == '\n' {
            let mut end = idx;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - idx >= 2 {
                let segment = &normalized[start..idx];
                let words = parse_words(segment);
                if !words.is_empty() {
                    for line in solve_paragraph(&words, options) {
                        output.push(render_line(line, options, output.len()));
                    }
                }
                for _ in 1..(end - idx) {
                    output.push(String::new());
                }
                start = end;
                idx = end;
                continue;
            }
        }
        idx += 1;
    }

    let segment = &normalized[start..];
    let words = parse_words(segment);
    if !words.is_empty() {
        let lines = solve_paragraph(&words, options);
        for (line_idx, line) in lines.into_iter().enumerate() {
            output.push(render_line(line, options, line_idx));
        }
    }

    Ok(output)
}

fn render_line(line: Line, options: &WrapOptions, paragraph_line_index: usize) -> String {
    let mut out = " ".repeat(line.indent);
    let gaps = line.pieces.len().saturating_sub(1);
    if gaps == 0 {
        out.push_str(&line.displayed);
        return out;
    }

    if options.justify && line.cost >= 0 && paragraph_line_index + 1 > 0 {
        // Finality is determined by the selected paragraph sequence by the caller's line cost context.
    }

    out.push_str(&line.pieces[0]);
    for i in 0..gaps {
        let mut spaces = 1usize;
        if options.justify {
            let extra = options.width.saturating_sub(line.indent + line.displayed.chars().count());
            let each = extra / gaps;
            let rem = extra % gaps;
            spaces += each + usize::from((i + gaps - (paragraph_line_index % gaps)) % gaps < rem);
        }
        out.push_str(&" ".repeat(spaces));
        out.push_str(&line.pieces[i + 1]);
    }
    if line.marker_ended {
        out.push('-');
    }
    out
}
