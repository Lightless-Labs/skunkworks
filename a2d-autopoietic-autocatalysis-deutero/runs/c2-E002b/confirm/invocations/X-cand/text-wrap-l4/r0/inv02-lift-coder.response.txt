#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
enum EndKind {
    CompleteWord,
    Marker,
    Raw,
}

#[derive(Clone)]
struct Piece {
    text: String,
    words: usize,
}

#[derive(Clone)]
struct Line {
    pieces: Vec<Piece>,
    end: EndKind,
    consumed: usize,
    cost_gap: usize,
    indent: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    rendered: Vec<String>,
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.rendered < b.rendered
}

fn scalar_string(chars: &[char]) -> String {
    chars.iter().collect()
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let width = options.width;
    let mut memo: Vec<Vec<Vec<Option<Solution>>>> = words
        .iter()
        .map(|w| {
            (0..=w.chars.len())
                .map(|_| vec![None, None])
                .collect()
        })
        .collect();

    fn recurse(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Vec<Vec<Option<Solution>>>],
    ) -> Solution {
        if wi == words.len() {
            return Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                rendered: Vec::new(),
            };
        }
        let fi = usize::from(first);
        if let Some(v) = &memo[wi][off][fi] {
            return v.clone();
        }

        let width = options.width;
        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = width - indent;
        let mut candidates = Vec::new();

        // Enumerate lines beginning with the current word suffix, then optionally
        // extending through completed words.
        let mut pieces = Vec::<Piece>::new();
        let mut used = 0usize;
        let mut cur_wi = wi;
        let mut cur_off = off;
        let mut first_piece = true;

        loop {
            let word = &words[cur_wi];
            let remaining = word.chars.len() - cur_off;
            let between = if first_piece { 0 } else { 1 };
            if used + between >= cap {
                break;
            }
            let base = used + between;
            let avail = cap - base;

            // Completed word choice, including a full suffix.
            if remaining <= avail {
                let mut text = String::new();
                if !first_piece {
                    text.push(' ');
                }
                text.push_str(&scalar_string(&word.chars[cur_off..]));
                let mut line_pieces = pieces.clone();
                line_pieces.push(Piece { text, words: 1 });
                let consumed_here = remaining
                    + line_pieces[..line_pieces.len() - 1]
                        .iter()
                        .map(|p| p.text.chars().count())
                        .sum::<usize>();
                let next = recurse(cur_wi + 1, 0, false, words, options, memo);
                candidates.push(make_candidate(
                    line_pieces,
                    EndKind::CompleteWord,
                    consumed_here,
                    indent,
                    cap,
                    next,
                    options,
                ));
            }

            // Marker-ended prefix choices.
            for &m in &word.markers {
                if m <= cur_off || m >= word.chars.len() {
                    continue;
                }
                let prefix_len = m - cur_off;
                if prefix_len + 1 > avail {
                    continue;
                }
                let mut text = String::new();
                if !first_piece {
                    text.push(' ');
                }
                text.push_str(&scalar_string(&word.chars[cur_off..m]));
                text.push('-');
                let mut line_pieces = pieces.clone();
                line_pieces.push(Piece { text, words: 1 });
                let consumed_here = prefix_len
                    + line_pieces[..line_pieces.len() - 1]
                        .iter()
                        .map(|p| p.text.chars().count())
                        .sum::<usize>();
                let next = recurse(cur_wi, m, false, words, options, memo);
                candidates.push(make_candidate(
                    line_pieces,
                    EndKind::Marker,
                    consumed_here,
                    indent,
                    cap,
                    next,
                    options,
                ));
            }

            // A raw split is available only when the line contains no earlier piece
            // and no marker prefix in the suffix fits.
            if first_piece && remaining > cap {
                let marker_fits = word.markers.iter().any(|&m| {
                    m > cur_off && m < word.chars.len() && (m - cur_off) + 1 <= cap
                });
                if !marker_fits {
                    let text = scalar_string(&word.chars[cur_off..cur_off + cap]);
                    let next = recurse(cur_wi, cur_off + cap, false, words, options, memo);
                    candidates.push(make_candidate(
                        vec![Piece { text, words: 1 }],
                        EndKind::Raw,
                        cap,
                        indent,
                        cap,
                        next,
                        options,
                    ));
                }
            }

            if remaining > avail {
                break;
            }
            used = base + remaining;
            pieces.push(Piece {
                text: scalar_string(&word.chars[cur_off..]),
                words: 1,
            });
            cur_wi += 1;
            if cur_wi >= words.len() {
                break;
            }
            cur_off = 0;
            first_piece = false;
        }

        let best = candidates
            .into_iter()
            .reduce(|a, b| if better(&a, &b) { a } else { b })
            .expect("raw split ensures a legal continuation");

        memo[wi][off][fi] = Some(best.clone());
        best
    }

    let solution = recurse(0, 0, true, words, options, &mut memo);
    let mut output = Vec::new();

    for (line_index, line) in solution.lines.iter().enumerate() {
        let mut rendered = String::new();
        rendered.push_str(&" ".repeat(line.indent));

        let gap_count = line.pieces.len().saturating_sub(1);
        let pre_width = line.indent + line.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
            + gap_count;
        let is_final = line_index + 1 == solution.lines.len();
        let justify = options.justify && !is_final && gap_count > 0;

        for (i, piece) in line.pieces.iter().enumerate() {
            if i > 0 {
                let mut spaces = 1;
                if justify {
                    let extra = width - pre_width;
                    spaces += extra / gap_count;
                    let remainder = extra % gap_count;
                    let start = line_index % gap_count;
                    let distance = (i - 1 + gap_count - start) % gap_count;
                    if distance < remainder {
                        spaces += 1;
                    }
                }
                rendered.push_str(&" ".repeat(spaces));
            }
            rendered.push_str(&piece.text);
        }
        output.push(rendered);
    }
    output
}

fn make_candidate(
    pieces: Vec<Piece>,
    end: EndKind,
    consumed: usize,
    indent: usize,
    cap: usize,
    next: Solution,
    options: &WrapOptions,
) -> Solution {
    let gaps = pieces.len().saturating_sub(1);
    let displayed = pieces.iter().map(|p| p.text.chars().count()).sum::<usize>() + gaps;
    let gap = cap - displayed;
    let final_line = next.lines.is_empty();
    let gap_cost = (gap as u64) * (gap as u64);
    let mut cost = if final_line { gap_cost / 4 } else { gap_cost };

    let marker = matches!(end, EndKind::Marker);
    if marker {
        cost += options.hyphen_cost as u64;
    } else if matches!(end, EndKind::Raw) {
        cost += (2 * options.width + 1) as u64;
    }
    if !final_line && pieces.len() == 1 {
        cost += 3;
    }

    let mut line = Line {
        pieces,
        end,
        consumed,
        cost_gap: gap,
        indent,
    };
    let mut rendered = String::new();
    rendered.push_str(&" ".repeat(indent));
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            rendered.push(' ');
        }
        rendered.push_str(&piece.text);
    }

    let mut lines = vec![line.clone()];
    lines.extend(next.lines);
    let mut consumed_vec = vec![consumed];
    consumed_vec.extend(next.consumed);
    let mut rendered_vec = vec![rendered];
    rendered_vec.extend(next.rendered);

    line.end = end;
    Solution {
        cost: cost + next.cost,
        marker_lines: usize::from(marker) + next.marker_lines,
        lines,
        consumed: consumed_vec,
        rendered: rendered_vec,
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_side = |j: isize| -> bool {
                if j < 0 || j as usize >= chars.len() {
                    return false;
                }
                let c = chars[j as usize].1;
                c != '~' && !is_separator(c)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::<char>::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            if i - start >= 2 {
                let words = parse_words(&segment);
                output.extend(solve_paragraph(&words, options));
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(i - start - 1));
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    let words = parse_words(&segment);
    output.extend(solve_paragraph(&words, options));
    Ok(output)
}

fn parse_words(input: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();
    for &c in input {
        if is_separator(c) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}
