#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    displayed: String,
    consumed: usize,
    marker_ended: bool,
    raw: bool,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed_vector: Vec<usize>,
    strings: Vec<String>,
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed_vector != b.consumed_vector {
        return a.consumed_vector > b.consumed_vector;
    }
    a.strings < b.strings
}

fn line_text(line: &Line) -> String {
    let mut s = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(piece);
    }
    s
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    let mut memo: Vec<Vec<[Option<Solution>; 2]>> = words
        .iter()
        .map(|w| {
            (0..=w.chars.len())
                .map(|_| [None, None])
                .collect()
        })
        .collect();

    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Vec<[Option<Solution>; 2]>],
    ) -> Solution {
        if wi == words.len() {
            return Solution {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                strings: Vec::new(),
            };
        }
        if let Some(sol) = &memo[wi][off][usize::from(!first)] {
            return sol.clone();
        }

        let word = &words[wi];
        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut candidates: Vec<(usize, usize, bool, bool, Vec<String>, String)> = Vec::new();

        let mut pos = off;
        let mut rendered = String::new();
        let mut used = 0usize;
        let mut pieces = Vec::new();

        loop {
            let remain = word.chars.len() - pos;
            let room = cap.saturating_sub(used + usize::from(!pieces.is_empty()));
            let available = room.min(remain);

            // Candidate marker endings within this word.
            for &m in word.markers.iter().filter(|&&m| m > pos && m - pos <= room.saturating_sub(1)) {
                let prefix: String = word.chars[pos..m].iter().collect();
                let mut ps = pieces.clone();
                ps.push(format!("{prefix}-"));
                let mut text = rendered.clone();
                if !text.is_empty() {
                    text.push(' ');
                }
                text.push_str(&format!("{prefix}-"));
                candidates.push((wi, m, true, false, ps, text));
            }

            if remain <= room {
                let whole: String = word.chars[pos..].iter().collect();
                let mut ps = pieces.clone();
                ps.push(whole.clone());
                let mut text = rendered.clone();
                if !text.is_empty() {
                    text.push(' ');
                }
                text.push_str(&whole);
                if wi + 1 == words.len() {
                    candidates.push((words.len(), 0, false, false, ps, text));
                } else {
                    candidates.push((wi + 1, 0, false, false, ps, text));
                }
            } else if pieces.is_empty() {
                let marker_fits = word
                    .markers
                    .iter()
                    .any(|&m| m > pos && m - pos <= cap.saturating_sub(1));
                if !marker_fits && cap > 0 {
                    let end = pos + cap;
                    let raw: String = word.chars[pos..end].iter().collect();
                    candidates.push((wi, end, false, true, vec![raw.clone()], raw));
                }
            }

            if remain > room || wi + 1 >= words.len() {
                break;
            }

            let whole: String = word.chars[pos..].iter().collect();
            if !pieces.is_empty() {
                rendered.push(' ');
            }
            rendered.push_str(&whole);
            pieces.push(whole);
            used += remain + usize::from(pieces.len() > 1);
            pos = word.chars.len();

            let next_word = &words[wi + 1];
            let next_room = cap.saturating_sub(used + 1);
            if next_room == 0 {
                break;
            }
            let take = next_room.min(next_word.chars.len());
            if take == 0 {
                break;
            }
            // Continue the line with the next word; subsequent iterations use its suffix.
            let next_text: String = next_word.chars[..take].iter().collect();
            if take < next_word.chars.len() {
                break;
            }
            rendered.push(' ');
            rendered.push_str(&next_text);
            pieces.push(next_text);
            used += take + 1;
            pos = 0;
            // The implementation below handles only completed words in this loop;
            // candidate generation for later words is evaluated by recursive states.
            break;
        }

        let mut best: Option<Solution> = None;
        for (next_wi, next_off, marker_ended, raw, pieces, text) in candidates {
            let consumed = pieces
                .iter()
                .map(|p| p.chars().count())
                .sum::<usize>();
            let displayed = indent + text.chars().count();
            let gap = options.width - displayed;
            let final_line = next_wi == words.len();
            let mut cost = if final_line {
                (gap * gap / 4) as u64
            } else {
                (gap * gap) as u64
            };
            if marker_ended {
                cost += options.hyphen_cost as u64;
            }
            if raw {
                cost += (2 * options.width + 1) as u64;
            }
            if !final_line && pieces.len() == 1 {
                cost += 3;
            }

            let line = Line {
                indent,
                pieces,
                displayed: text,
                consumed,
                marker_ended,
                raw,
            };
            let mut tail = if final_line {
                Solution {
                    cost: 0,
                    marker_lines: 0,
                    lines: Vec::new(),
                    consumed_vector: Vec::new(),
                    strings: Vec::new(),
                }
            } else {
                solve(next_wi, next_off, false, words, options, memo)
            };
            let mut candidate = Solution {
                cost: cost + tail.cost,
                marker_lines: usize::from(marker_ended) + tail.marker_lines,
                lines: {
                    let mut v = vec![line.clone()];
                    v.append(&mut tail.lines);
                    v
                },
                consumed_vector: {
                    let mut v = vec![consumed];
                    v.append(&mut tail.consumed_vector);
                    v
                },
                strings: {
                    let mut v = vec![line_text(&line)];
                    v.append(&mut tail.strings);
                    v
                },
            };
            if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        let result = best.unwrap_or_else(|| {
            // Capacity is always positive; raw splitting guarantees continuation.
            let end = (off + cap).min(word.chars.len());
            let raw: String = word.chars[off..end].iter().collect();
            let line = Line {
                indent,
                pieces: vec![raw.clone()],
                displayed: raw,
                consumed: end - off,
                marker_ended: false,
                raw: true,
            };
            let tail = solve(wi, end, false, words, options, memo);
            Solution {
                cost: (2 * options.width + 1) as u64 + tail.cost,
                marker_lines: tail.marker_lines,
                lines: std::iter::once(line.clone()).chain(tail.lines).collect(),
                consumed_vector: std::iter::once(end - off)
                    .chain(tail.consumed_vector)
                    .collect(),
                strings: std::iter::once(line_text(&line))
                    .chain(tail.strings)
                    .collect(),
            }
        });

        memo[wi][off][usize::from(!first)] = Some(result.clone());
        result
    }

    if words.is_empty() {
        Vec::new()
    } else {
        solve(0, 0, true, words, options, &mut memo).lines
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut marker_error = None;
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let left_ok = i > 0 && chars[i - 1].1 != '~' && !separators(chars[i - 1].1);
            let right_ok = i + 1 < chars.len()
                && chars[i + 1].1 != '~'
                && !separators(chars[i + 1].1);
            if (!left_ok || !right_ok) && marker_error.is_none() {
                marker_error = Some(chars[i].0);
            }
        }
    }
    if let Some(index) = marker_error {
        return Err(WrapError::InvalidMarker { index });
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();

    fn process_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
        let mut words = Vec::new();
        let mut current_chars = Vec::new();
        let mut markers = Vec::new();
        for &c in segment {
            if separators(c) {
                if !current_chars.is_empty() {
                    words.push(Word {
                        chars: std::mem::take(&mut current_chars),
                        markers: std::mem::take(&mut markers),
                    });
                }
            } else if c == '~' {
                markers.push(current_chars.len());
            } else {
                current_chars.push(c);
            }
        }
        if !current_chars.is_empty() {
            words.push(Word {
                chars: current_chars,
                markers,
            });
        }

        let mut lines = solve_paragraph(&words, options);
        let count = lines.len();
        for (idx, line) in lines.drain(..).enumerate() {
            let mut text = " ".repeat(line.indent);
            for (piece_idx, piece) in line.pieces.iter().enumerate() {
                if piece_idx > 0 {
                    text.push(' ');
                }
                text.push_str(piece);
            }
            let gap = options.width.saturating_sub(text.chars().count());
            if options.justify && idx + 1 < count && line.pieces.len() >= 2 {
                let gaps = line.pieces.len() - 1;
                let each = gap / gaps;
                let extra = gap % gaps;
                let mut rebuilt = " ".repeat(line.indent);
                for (piece_idx, piece) in line.pieces.iter().enumerate() {
                    if piece_idx > 0 {
                        let g = piece_idx - 1;
                        rebuilt.push_str(&" ".repeat(1 + each + usize::from((g + idx % gaps) % gaps < extra)));
                    }
                    rebuilt.push_str(piece);
                }
                text = rebuilt;
            }
            output.push(text);
        }
    }

    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let start = p;
            while p < normalized.len() && normalized[p] == '\n' {
                p += 1;
            }
            if p - start >= 2 {
                process_segment(&segment, options, &mut output);
                segment.clear();
                for _ in 1..(p - start) {
                    output.push(String::new());
                }
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[p]);
            p += 1;
        }
    }
    process_segment(&segment, options, &mut output);
    Ok(output)
}
