#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
}

pub struct IndexedHeap {
    items: Vec<HeapItem>,
}

struct HeapItem {
    item: Item,
    age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self { items: Vec::new() }
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.items.is_empty()
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize(key)?;
        if self.find_index(&key).is_some() {
            return Err(Error::Duplicate { key });
        }

        let age = self.items.iter().map(|entry| entry.age).max().unwrap_or(0);
        let age = if self.items.is_empty() { 0 } else { age + 1 };
        self.items.push(HeapItem {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age,
        });
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.items
            .iter()
            .min_by(|a, b| compare(a, b))
            .map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let index = self
            .items
            .iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| compare(a, b))
            .map(|(index, _)| index)?;
        Some(self.items.remove(index).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self.find_index(&key).map(|index| self.items[index].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let index = self.find_index(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.items[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.items[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let index = self.find_index(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.items[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.items[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize(key)?;
        let index = self.find_index(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(std::mem::replace(
            &mut self.items[index].item.value,
            value.to_owned(),
        ))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize(key)?;
        let index = self.find_index(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(self.items.remove(index).item)
    }

    fn find_index(&self, key: &str) -> Option<usize> {
        self.items.iter().position(|entry| entry.item.key == key)
    }
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    let trimmed = &bytes[start..end];
    for (offset, &byte) in trimmed.iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(trimmed
        .iter()
        .map(|byte| byte.to_ascii_lowercase() as char)
        .collect())
}

fn compare(a: &HeapItem, b: &HeapItem) -> std::cmp::Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}
