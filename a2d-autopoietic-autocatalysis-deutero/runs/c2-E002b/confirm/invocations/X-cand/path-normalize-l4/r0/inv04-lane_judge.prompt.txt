use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn render_absolute(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn render_normalized(stack: &[String], absolute: bool, directory: bool) -> String {
    if absolute {
        if stack.is_empty() {
            return "/".to_owned();
        }
        let mut result = render_absolute(stack);
        if directory {
            result.push('/');
        }
        result
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        let mut result = stack.join("/");
        if directory {
            result.push('/');
        }
        result
    }
}

fn normalize_components(
    raw: &str,
    initial: &mut Vec<String>,
    absolute: bool,
    offset_base: usize,
) -> Result<(), PathError> {
    let mut cursor = 0;
    for component in raw.split('/') {
        let start = cursor;
        cursor += component.len() + 1;
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if initial.last().is_some_and(|last| last != "..") {
                initial.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: offset_base + start,
                });
            } else {
                initial.push("..".to_owned());
            }
        } else {
            initial.push(component.to_owned());
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let mut stack = Vec::new();
    normalize_components(path, &mut stack, absolute, 0)?;
    Ok(render_normalized(
        &stack,
        absolute,
        directory_intent(path),
    ))
}

fn validated_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(components(&normalized))
}

fn requested_remainder(path: &str, home: Option<&str>) -> Result<(bool, Vec<String>, usize), PathError> {
    if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let home = home.ok_or(PathError::MissingHome)?;
            let offset = if path == "~" { path.len() } else { 2 };
            let rest = if path == "~" { "" } else { &path[2..] };
            return Ok((false, components(home), offset + rest.len() - rest.len()));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if path.starts_with('/') {
        Ok((true, Vec::new(), 0))
    } else {
        Ok((false, Vec::new(), 0))
    }
}

fn resolve_core(

    cwd: &str,
    home: Option<&str>,
    links: Option<(&[Link], usize)>,
) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    let link_map = if let Some((table, _)) = links {
        let mut map: HashMap<&str, usize> = HashMap::new();
        for (index, link) in table.iter().enumerate() {
            let valid_path = link.path.starts_with('/')
                && !link.path.ends_with('/')
                && link.path != "/"
                && normalize(&link.path).is_ok_and(|value| value == link.path);
            let valid_target = validate_raw(&link.target).is_ok();
            if !valid_path || !valid_target {
                return Err(PathError::InvalidLink { index });
            }
            if let Some(&first) = map.get(link.path.as_str()) {
                return Err(PathError::DuplicateLink {
                    first,
                    second: index,
                });
            }
            map.insert(link.path.as_str(), index);
        }
        Some(map)
    } else {
        None
    };

    validate_raw(path)?;

    let mut absolute;
    let mut stack;
    let mut request_start;
    if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let home = home_components.ok_or(PathError::MissingHome)?;
            absolute = true;
            stack = home;
            request_start = if path == "~" { path.len() } else { 2 };
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        absolute = true;
        stack = Vec::new();
        request_start = 0;
    } else {
        absolute = true;
        stack = cwd_components;
        request_start = 0;
    }

    let directory = directory_intent(path);
    let mut expansion_count = 0usize;
    let mut active = Vec::<usize>::new();

    if let Some((table, max_expansions)) = links {
        let mut work = Vec::<(String, usize, usize, bool)>::new();
        if path.starts_with('~') {
            if path != "~" {
                work.push((path[2..].to_owned(), 2, 0, false));
            }
        } else {
            work.push((path[request_start..].to_owned(), request_start, 0, false));

                // Marker used to pop the active link after its target completes.
                if raw.is_empty() && index == usize::MAX {
                    if let Some(done) = active.pop() {
                        let _ = done;
                    }
                    continue;
                }

            let parts: Vec<(String, usize)> = raw
                .split('/')
                .map(|part| {
                    let at = cursor;
                    cursor += part.len() + 1;
                    (part.to_owned(), at)
                })
                .collect();

                work.push((component, offset_base + local_at, 1, is_target));
            }

                if component_flag == 1 {
                    if component.is_empty() || component == "." {
                        continue;
                    }
                    if component == ".." {

                            stack.pop();
                        } else {
                            return Err(PathError::Escape { at });
                        }
                        continue;

                        if active.contains(&link_index) {
                            return Err(PathError::Loop { index: link_index });
                        }
                        if expansion_count >= max_expansions {
                            return Err(PathError::Limit { index: link_index });
                        }
                        expansion_count += 1;
                        active.push(link_index);
                        stack.pop();

                        work.push((String::new(), usize::MAX, usize::MAX, true));

                        let target_abs = target.starts_with('/');

                        let target_parts: Vec<(String, usize)> = target

                            .enumerate()

                                    .split('/')
                                    .take(i)
                                    .map(str::len)
                                    .sum::<usize>()
                                    + i;

                            })
                            .collect();

                        }

                    let _ = (at, target_flag);
                }
            }

            if path == "~" {
                ""
            } else {
                &path[2..]
            }
        } else {
            path
        };

    }

    let mut result = render_absolute(&stack);
    if directory && result != "/" {
        result.push('/');
    }
    Ok(result)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_core(path, cwd, home, None)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home, Some((links, max_expansions)))
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(left, right)| left == right)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = components(&resolved_base);
    let target_components = components(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_components, &target_components)];
    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        if let Ok(probe) = probe {
            let alias_components = components(&probe);
            if target_components.starts_with(&alias_components) {
                let mut candidate = route(&base_components, &components(&link.path));
                let suffix = &target_components[alias_components.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let mut valid = Vec::<String>::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }
        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if components(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid
        .into_iter()
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())))
        .ok_or(PathError::Escape { at: 0 })
}

    let bytes = path.as_bytes();
    let mut components = Vec::new();
    let mut i = start;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let begin = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        components.push((path[begin..i].to_owned(), begin));
    }
    components
}

fn render_absolute(components: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&components.join("/"));
    if components.is_empty() {
        return result;
    }
    if intent {
        result.push('/');
    }
    result
}

fn normalize_impl(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    for (component, at) in raw_components(path, 0) {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|c| c != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(component);
                }
            }
            _ => stack.push(component),
        }
    }

    if absolute {
        Ok(render_absolute(&stack, intent))
    } else if stack.is_empty() {
        Ok(".".to_owned())
    } else {
        let mut result = stack.join("/");
        if intent {
            result.push('/');
        }
        Ok(result)
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    normalize_impl(path)
}

fn absolute_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    let normalized = normalize_impl(path).map_err(|_| error)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect())
}

fn validate_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = absolute_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(absolute_context(value, PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

fn tilde_start(path: &str, home: Option<&[String]>) -> Result<(usize, bool, Vec<String>), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((consumed, false, home.to_vec()));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    Ok((0, path.starts_with('/'), Vec::new()))
}

fn resolve_impl(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (start, absolute, mut stack) =
        tilde_start(path, home_parts.as_deref())?;

    if absolute {
        stack.clear();
    } else if start == 0 {
        stack = cwd_parts;
    }

    for (component, at) in raw_components(path, start) {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(component),
        }
    }
    Ok(render_absolute(&stack, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_impl(path, cwd, home)
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        let canonical = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize_impl(&link.path)
                .is_ok_and(|value| value == link.path);
        if !canonical || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        for (first, earlier) in links[..index].iter().enumerate() {
            if earlier.path == link.path {
                return Err(PathError::DuplicateLink {
                    first,
                    second: index,
                });
            }
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Task {
    text: String,
    pos: usize,
    kind: TaskKind,
}

#[derive(Clone, Copy)]
enum TaskKind {
    Context,
    Request,
    Target(usize),
}

struct Resolver<'a> {
    links: &'a [Link],
    max_expansions: usize,
    expansions: usize,
    active: Vec<usize>,
    stack: Vec<String>,
}

impl<'a> Resolver<'a> {
    fn new(links: &'a [Link], max_expansions: usize, stack: Vec<String>) -> Self {
        Self {
            links,
            max_expansions,
            expansions: 0,
            active: Vec::new(),
            stack,
        }
    }

    fn key(&self) -> String {
        let mut key = String::from("/");
        key.push_str(&self.stack.join("/"));
        key
    }

    fn find_link(&self, key: &str) -> Option<usize> {
        self.links.iter().position(|link| link.path == key)
    }

    fn run(&mut self, initial: Vec<Task>) -> Result<(), PathError> {
        let mut tasks = initial;
        while let Some(mut task) = tasks.pop() {
            let bytes = task.text.as_bytes();
            while task.pos < bytes.len() && bytes[task.pos] == b'/' {
                task.pos += 1;
            }
            if task.pos >= bytes.len() {
                if let TaskKind::Target(index) = task.kind {
                    if self.active.last() == Some(&index) {
                        self.active.pop();
                    } else if let Some(position) = self.active.iter().rposition(|&v| v == index) {
                        self.active.remove(position);
                    }
                }
                continue;
            }

            let begin = task.pos;
            while task.pos < bytes.len() && bytes[task.pos] != b'/' {
                task.pos += 1;
            }
            let component = &task.text[begin..task.pos];
            let offset = match task.kind {
                TaskKind::Context => 0,
                TaskKind::Request => begin,
                TaskKind::Target(_) => begin,
            };

            if task.pos < bytes.len() {
                tasks.push(task.clone());
            }
            match component {
                "." => continue,
                ".." => {
                    if self.stack.pop().is_none() {
                        return Err(PathError::Escape { at: offset });
                    }
                }
                _ => {
                    self.stack.push(component.to_owned());
                    let key = self.key();
                    if let Some(index) = self.find_link(&key) {
                        if self.active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if self.expansions >= self.max_expansions {
                            return Err(PathError::Limit { index });
                        }
                        self.expansions += 1;
                        self.active.push(index);

                        let target = &self.links[index].target;
                        let mut target_start = 0;
                        if target.starts_with('/') {
                            self.stack.clear();
                        } else {
                            self.stack.pop();
                        }
                        tasks.push(Task {
                            text: String::new(),
                            pos: 0,
                            kind: TaskKind::Target(index),
                        });
                        tasks.push(Task {
                            text: target.clone(),
                            pos: target_start,
                            kind: TaskKind::Target(index),
                        });
                        target_start = 0;
                        let _ = target_start;
                    }
                }
            }
        }
        Ok(())
    }
}

fn resolve_links_impl(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (start, absolute, mut stack) = tilde_start(path, home_parts.as_deref())?;
    if absolute {
        stack.clear();
    } else if start == 0 {
        stack = cwd_parts;
    }

    let request = Task {
        text: path.to_owned(),
        pos: start,
        kind: TaskKind::Request,
    };
    let mut resolver = Resolver::new(links, max_expansions, stack);
    resolver.run(vec![request])?;
    Ok(render_absolute(&resolver.stack, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_impl(path, cwd, home, links, max_expansions)
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

fn add_suffix(candidate: String, suffix: &[String]) -> String {
    if suffix.is_empty() {
        return candidate;
    }
    let mut result = if candidate == "." {
        String::new()
    } else {
        candidate
    };
    if !result.is_empty() {
        result.push('/');
    }
    result.push_str(&suffix.join("/"));
    result
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;

    let base_parts = components(&resolved_base);
    let target_parts = components(&resolved_target);
    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let alias_parts = components(&probe);
        if alias_parts.len() > target_parts.len()
            || alias_parts
                .iter()
                .zip(&target_parts)
                .any(|(a, b)| a != b)
        {
            continue;
        }
        let mut candidate = route(&base_parts, &components(&link.path));
        candidate = add_suffix(candidate, &target_parts[alias_parts.len()..]);
        candidates.push(candidate);
    }

    let target_has_directory_intent = resolved_target.ends_with('/');
    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_has_directory_intent && candidate != "." {
            candidate.push('/');
        }
        let Ok(result) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) else {
            continue;
        };
        if components(&result) == target_parts {
            valid.push(candidate);
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}

    Backslash { at: usize },

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

#[derive(Clone)]
struct Component {
    text: String,
    at: usize,
}

fn components(path: &str, offset: usize) -> Vec<Component> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push(Component {
            text: path[start..i].to_owned(),
            at: offset + start,
        });
    }
    result
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    components(path, 0)
        .last()
        .map(|c| c.text == "." || c.text == "..")
        .unwrap_or(false)
}

fn format_absolute(parts: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if intent && result != "/" {
        result.push('/');
    }
    result
}

fn format_normalized(absolute: bool, parts: &[String], intent: bool) -> String {
    if absolute {
        return format_absolute(parts, intent);
    }
    let mut result = if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };
    if intent && result != "." && !result.is_empty() {
        result.push('/');
    }
    result
}

fn reduce_relative_or_absolute(
    path: &str,
    offset: usize,
    absolute: bool,
    initial: Vec<String>,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    for component in components(path, offset) {
        match component.text.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|s| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component.at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(component.text),
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let parts = reduce_relative_or_absolute(path, 0, absolute, Vec::new())?;
    Ok(format_normalized(absolute, &parts, intent))
}

fn normalized_context(path: &str) -> Option<Vec<String>> {
    if !path.starts_with('/') {
        return None;
    }
    let normalized = normalize(path).ok()?;
    Some(components(&normalized, 0).into_iter().map(|c| c.text).collect())
}

fn expand_request<'a>(path: &'a str, home: Option<&'a str>) -> Result<(&'a str, usize, Vec<String>, bool), PathError> {
    if path.starts_with('~') {
        if path.len() > 1 && !path.as_bytes()[1..].starts_with(b"/") {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home.ok_or(PathError::MissingHome)?;
        let parts = components(home, 0).into_iter().map(|c| c.text).collect();
        let rest = if path == "~" { "" } else { &path[2..] };
        let offset = if path == "~" { path.len() } else { 2 };
        return Ok((rest, offset, parts, true));
    }
    if path.starts_with('/') {
        Ok((path, 0, Vec::new(), true))
    } else {
        Ok((path, 0, Vec::new(), false))
    }
}

fn prepare_contexts(cwd: &str, home: Option<&str>) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = normalized_context(cwd).ok_or(PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(normalized_context(value).ok_or(PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = prepare_contexts(cwd, home)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (rest, offset, expanded_parts, absolute) = expand_request(path, home)?;
    let initial = if path.starts_with('~') {
        expanded_parts
    } else if absolute {
        Vec::new()
    } else {
        cwd_parts
    };
    let parts = reduce_relative_or_absolute(rest, offset, true, initial)?;
    Ok(format_absolute(&parts, intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut keys = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .map(|v| v == link.path)
                .unwrap_or(false);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = keys.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        keys.insert(link.path.clone(), index);
    }
    Ok(keys)
}

#[derive(Clone)]
struct WorkComponent {
    text: String,
    at: usize,
}

fn absolute_key(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn push_component(
    stack: &mut Vec<String>,
    component: &WorkComponent,
    absolute: bool,
) -> Result<(), PathError> {
    match component.text.as_str() {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|s| s != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at: component.at });
            } else {
                stack.push("..".to_owned());
            }
        }
        _ => stack.push(component.text.clone()),
    }
    Ok(())
}

fn traverse_linked(
    stack: &mut Vec<String>,
    items: &[WorkComponent],
    links: &[Link],
    keys: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    for (pos, item) in items.iter().enumerate() {
        match item.text.as_str() {
            "." => continue,
            ".." => {
                push_component(stack, item, true)?;
                continue;
            }
            _ => stack.push(item.text.clone()),
        }

        let key = absolute_key(stack);
        let Some(&index) = keys.get(&key) else {
            continue;
        };
        if active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if *expansions >= max_expansions {
            return Err(PathError::Limit { index });
        }
        *expansions += 1;
        active.push(index);

        let link_parent_len = stack.len() - 1;
        if links[index].target.starts_with('/') {
            stack.clear();
        } else {
            stack.truncate(link_parent_len);
        }
        let target_items: Vec<WorkComponent> = components(&links[index].target, 0)
            .into_iter()
            .map(|c| WorkComponent {
                text: c.text,
                at: c.at,
            })
            .collect();
        traverse_linked(
            stack,
            &target_items,
            links,
            keys,
            active,
            expansions,
            max_expansions,
        )?;
        active.pop();

        // The recursive target traversal completes before the remainder is resumed.
        let _ = pos;
    }
    Ok(())
}

fn linked_request(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
    links: &[Link],
    keys: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);

    let (rest, offset, initial, absolute) = if path.starts_with('~') {
        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home_parts.ok_or(PathError::MissingHome)?;
        if path == "~" {
            ("", path.len(), home.to_vec(), true)
        } else {
            (&path[2..], 2, home.to_vec(), true)
        }
    } else if path.starts_with('/') {
        (path, 0, Vec::new(), true)
    } else {
        (path, 0, cwd_parts.to_vec(), true)
    };

    let mut stack = Vec::<String>::new();
    let mut active = Vec::new();
    let mut expansions = 0usize;

    // Context components are traversed first, with local offset zero.
    for name in initial {
        stack.push(name);
        if let Some(&index) = keys.get(&absolute_key(&stack)) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);
            let parent_len = stack.len() - 1;
            if links[index].target.starts_with('/') {
                stack.clear();
            } else {
                stack.truncate(parent_len);
            }
            let target: Vec<WorkComponent> = components(&links[index].target, 0)
                .into_iter()
                .map(|c| WorkComponent { text: c.text, at: c.at })
                .collect();
            traverse_linked(
                &mut stack,
                &target,
                links,
                keys,
                &mut active,
                &mut expansions,
                max_expansions,
            )?;
            active.pop();
        }
    }

    let request_items: Vec<WorkComponent> = components(rest, offset)
        .into_iter()
        .map(|c| WorkComponent { text: c.text, at: c.at })
        .collect();
    traverse_linked(
        &mut stack,
        &request_items,
        links,
        keys,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

    let _ = absolute;
    Ok(format_absolute(&stack, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = prepare_contexts(cwd, home)?;
    let keys = validate_links(links)?;
    linked_request(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        &keys,
        max_expansions,
    )
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend_from_slice(&target[common..]);
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

fn candidate_result_spelling(candidate: &str, target: &str) -> String {
    let mut result = candidate.to_owned();
    if result.starts_with('~') {
        result.insert_str(0, "./");
    }
    let target_directory = target.ends_with('/');
    if result != "." && target_directory && !result.ends_with('/') {
        result.push('/');
    }
    result
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let base_resolved = resolve_links(base, cwd, home, links, max_expansions)?;
    let target_resolved = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = result_parts(&base_resolved);
    let target_parts = result_parts(&target_resolved);

    let keys = validate_links(links)?;
    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = match resolve_links(&link.path, "/", None, links, max_expansions) {
            Ok(value) => value,
            Err(_) => continue,
        };
        let anchor = result_parts(&probe);
        if anchor.len() <= target_parts.len()
            && anchor
                .iter()
                .zip(target_parts.iter())
                .all(|(a, b)| a == b)
        {
            let key_parts = result_parts(&link.path);
            let mut candidate = route(&base_parts, &key_parts);
            let suffix = &target_parts[anchor.len()..];
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::<String>::new();
    for candidate in candidates {
        let candidate = candidate_result_spelling(&candidate, &target_resolved);
        let resolved = match resolve_links(
            &candidate,
            &base_resolved,
            home,
            links,
            max_expansions,
        ) {
            Ok(value) => value,
            Err(_) => continue,
        };
        if result_parts(&resolved) == target_parts && !valid.contains(&candidate) {
            valid.push(candidate);
        }
    }

    let _ = keys;
    valid
        .into_iter()
        .min_by(|a, b| {
            a.as_bytes()
                .len()
                .cmp(&b.as_bytes().len())
                .then_with(|| a.as_bytes().cmp(b.as_bytes()))
        })
        .ok_or(PathError::Escape { at: 0 })
}

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn append_directory_slash(mut result: String, intent: bool) -> String {
    if intent && result != "/" && result != "." && !result.ends_with('/') {
        result.push('/');
    }
    result
}

fn render_absolute(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn render_normalized(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        render_absolute(parts)
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };
    result = append_directory_slash(result, intent);
    result
}

fn split_components(path: &str, offset: usize) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_owned(), offset + start));
    }
    result
}

fn reduce_plain(
    path: &str,
    absolute: bool,
    starting: &[String],
    offset: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = starting.to_vec();
    for (component, at) in split_components(path, offset) {
        if component == "." {
            continue;
        }
        if component == ".." {
            if stack.pop().is_none() {
                if absolute {
                    return Err(PathError::Escape { at });
                }
                stack.push(component);
            }
        } else {
            stack.push(component);
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let parts = reduce_plain(path, absolute, &[], 0)?;
    Ok(render_normalized(&parts, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    let parts = normalized
        .split('/')
        .filter(|s| !s.is_empty())
        .map(str::to_owned)
        .collect();
    Ok(parts)
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => validate_context(path)
            .map(Some)
            .map_err(|_| PathError::InvalidHome),
    }
}

fn expand_request<'a>(
    path: &'a str,
    home: &'a Option<Vec<String>>,
) -> Result<(&'a str, usize, Vec<String>, bool), PathError> {
    let bytes = path.as_bytes();
    if bytes.first() == Some(&b'~') {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let home_parts = home.clone().ok_or(PathError::MissingHome)?;
            let skip = if bytes.len() >= 2 { 2 } else { 1 };
            Ok((&path[skip..], skip, home_parts, true))
        } else {
            Err(PathError::UnsupportedTilde { at: 1 })
        }
    } else if bytes.first() == Some(&b'/') {
        Ok((path, 0, Vec::new(), true))
    } else {
        Ok((path, 0, Vec::new(), false))
    }
}

fn resolve_core(
    path: &str,
    cwd: &[String],
    home: &Option<Vec<String>>,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (remainder, offset, tilde_start, absolute) = expand_request(path, home)?;
    let starting = if path.starts_with('~') {
        tilde_start
    } else if absolute {
        Vec::new()
    } else {
        cwd.to_vec()
    };
    let parts = reduce_plain(remainder, true, &starting, offset)?;
    Ok(append_directory_slash(render_absolute(&parts), intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = validate_home(home)?;
    resolve_core(path, &cwd_parts, &home_parts)
}

#[derive(Clone)]
struct Work {
    raw: String,
    offset: usize,
    index: usize,
    active_link: Option<usize>,
}

fn lookup_link(path: &[String], links: &[Link]) -> Option<usize> {
    let key = render_absolute(path);
    links.iter().position(|link| link.path == key)
}

fn resolve_links_core(
    path: &str,
    cwd: &[String],
    home: &Option<Vec<String>>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (remainder, offset, tilde_start, absolute) = expand_request(path, home)?;
    let mut stack = if path.starts_with('~') {
        tilde_start
    } else if absolute {
        Vec::new()
    } else {
        cwd.to_vec()
    };

    let mut work = vec![Work {
        raw: remainder.to_owned(),
        offset,
        index: 0,
        active_link: None,
    }];
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(frame) = work.last_mut() {
        let bytes = frame.raw.as_bytes();
        while frame.index < bytes.len() && bytes[frame.index] == b'/' {
            frame.index += 1;
        }
        if frame.index >= bytes.len() {
            if let Some(done) = work.pop() {
                if let Some(index) = done.active_link {
                    if active.last() == Some(&index) {
                        active.pop();
                    } else if let Some(pos) = active.iter().rposition(|&x| x == index) {
                        active.remove(pos);
                    }
                }
            }
            continue;
        }

        let start = frame.index;
        while frame.index < bytes.len() && bytes[frame.index] != b'/' {
            frame.index += 1;
        }
        let component = frame.raw[start..frame.index].to_owned();
        let at = frame.offset + start;

        if component == "." {
            continue;
        }
        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
            continue;
        }

        stack.push(component);
        if let Some(index) = lookup_link(&stack, links) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;

            let target = links[index].target.clone();
            let absolute_target = target.starts_with('/');
            if !absolute_target {
                stack.pop();
            } else {
                stack.clear();
            }
            active.push(index);
            work.push(Work {
                raw: target,
                offset: 0,
                index: 0,
                active_link: Some(index),
            });
        }
    }

    Ok(append_directory_slash(render_absolute(&stack), intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut seen = HashMap::<String, usize>::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = seen.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        seen.insert(link.path.clone(), index);
    }
    Ok(seen)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = validate_home(home)?;
    validate_links(links)?;
    resolve_links_core(path, &cwd_parts, &home_parts, links, max_expansions)
}

fn components_of_absolute(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_owned()));
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = components_of_absolute(&resolved_base);
    let target_parts = components_of_absolute(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probed) = probe else {
            continue;
        };
        let alias_parts = components_of_absolute(&link.path);
        let actual_parts = components_of_absolute(&probed);
        if actual_parts.len() > target_parts.len()
            || actual_parts
                .iter()
                .zip(&target_parts)
                .any(|(a, b)| a != b)
        {
            continue;
        }

        let mut candidate = route(&base_parts, &alias_parts);
        let suffix = &target_parts[actual_parts.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::<String>::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let resolved = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        );
        let Ok(resolved) = resolved else {
            continue;
        };
        if components_of_absolute(&resolved) == target_parts {
            valid.push(candidate);
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.dedup();
    valid.into_iter().next().ok_or(PathError::Empty)
}
