    mut cursor: usize,
    end: usize,
) -> Result<(u64, usize), Error> {

    let (body_length, body_start) = read_varint(input, start, input.len())?;
    let remaining = input.len() - body_start;
    if body_length > remaining as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: start,
            length: body_length,
        });
    }

    let body_end = body_start + body_length as usize;
    let mut cursor = body_start;
    let mut fields = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    while cursor < body_end {
        let key_start = cursor;
        let (key, after_key) = read_varint(input, cursor, body_end)?;
        cursor = after_key;

        let tag_u64 = key >> 3;
        let wire = (key & 7) as u8;

        if !(1..=255).contains(&tag_u64) {
            return Err(Error::InvalidTag {
                offset: key_start,
                tag: tag_u64,
            });
        }
        let tag = tag_u64 as u32;

        if !matches!(wire, 0 | 1 | 2 | 5) {
            return Err(Error::InvalidWire {
                offset: key_start,
                wire,
            });
        }

        if let Some((prev_tag, prev_wire)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder {
                    offset: key_start,
                    previous: prev_tag,
                    tag,
                });
            }
            if tag == prev_tag && !(prev_wire == 2 && wire == 2) {
                return Err(Error::Duplicate {
                    offset: key_start,
                    tag,
                });
            }
        }

        match wire {
            0 => {
                let payload_start = cursor;
                let (value, after_value) = read_varint(input, cursor, body_end)?;
                cursor = after_value;
                if tag <= 15 {
                    fields.push(Field::UInt { tag, value });
                }
                let _ = payload_start;
            }
            1 | 5 => {
                let size = if wire == 1 { 8usize } else { 4usize };
                if body_end - cursor < size {
                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: size as u64,
                    });
                }
                cursor += size;
            }
            2 => {
                let length_start = cursor;
                let (length, after_length) = read_varint(input, cursor, body_end)?;
                cursor = after_length;
                let available = body_end - cursor;
                if length > available as u64 {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length,
                    });
                }
                let data_end = cursor + length as usize;
                if tag <= 15 {
                    if let Some(Field::Bytes {
                        tag: existing_tag,
                        data,
                    }) = fields.last_mut()
                    {
                        if *existing_tag == tag {
                            data.extend_from_slice(&input[cursor..data_end]);
                        } else {
                            fields.push(Field::Bytes {
                                tag,
                                data: input[cursor..data_end].to_vec(),
                            });
                        }
                    } else {
                        fields.push(Field::Bytes {
                            tag,
                            data: input[cursor..data_end].to_vec(),
                        });
                    }
                }
                cursor = data_end;
            }
            _ => unreachable!(),
        }

        previous = Some((tag, wire));
    }

    Ok((Message { fields }, body_end))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Message {
    pub fields: Vec<Field>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Field {
    UInt { tag: u32, value: u64 },
    Bytes { tag: u32, data: Vec<u8> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
}

fn encode_varint(mut value: u64, out: &mut Vec<u8>) {
    while value >= 0x80 {
        out.push((value as u8 & 0x7f) | 0x80);
        value >>= 7;
    }
    out.push(value as u8);
}

fn decode_varint(
    input: &[u8],
    cursor: &mut usize,
    end: usize,
    start: usize,
) -> Result<u64, Error> {
    let mut value = 0u64;

    for index in 0..10 {
        if *cursor >= end {
            return Err(Error::Truncated {
                offset: end,
                start,
            });
        }

        let offset = *cursor;
        let byte = input[offset];
        *cursor += 1;
        let payload = byte & 0x7f;

        if index == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset, start });
        }

        value |= (payload as u64) << (index * 7);

        if byte & 0x80 == 0 {
            return Ok(value);
        }
    }

    unreachable!()
}

pub fn encode_message(message: &Message) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous: Option<(u32, bool)> = None;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, is_bytes) = match field {
            Field::UInt { tag, .. } => (*tag, false),
            Field::Bytes { tag, .. } => (*tag, true),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag {
                offset: index,
                tag: tag as u64,
            });
        }

        if let Some((previous_tag, previous_is_bytes)) = previous {
            if tag < previous_tag {
                return Err(Error::OutOfOrder {
                    offset: index,
                    previous: previous_tag,
                    tag,
                });
            }
            if tag == previous_tag && !(previous_is_bytes && is_bytes) {
                return Err(Error::Duplicate {
                    offset: index,
                    tag,
                });
            }
        }

        let wire = if is_bytes { 2 } else { 0 };
        encode_varint(((tag as u64) << 3) | wire, &mut body);

        match field {
            Field::UInt { value, .. } => encode_varint(*value, &mut body),
            Field::Bytes { data, .. } => {
                encode_varint(data.len() as u64, &mut body);
                body.extend_from_slice(data);
            }
        }

        previous = Some((tag, is_bytes));
    }

    let mut frame = Vec::new();
    encode_varint(body.len() as u64, &mut frame);
    frame.extend_from_slice(&body);
    Ok(frame)
}

pub fn decode_message(input: &[u8], start: usize) -> Result<(Message, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut cursor = start;
    let header_start = cursor;
    let declared_length = decode_varint(input, &mut cursor, input.len(), header_start)?;

    let remaining = input.len() - cursor;
    if declared_length > remaining as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: header_start,
            length: declared_length,
        });
    }

    let body_length = declared_length as usize;
    let frame_end = cursor + body_length;
    let mut fields = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    while cursor < frame_end {
        let key_start = cursor;
        let key = decode_varint(input, &mut cursor, frame_end, key_start)?;
        let tag_value = key >> 3;
        let wire = (key & 7) as u8;

        if !(1..=255).contains(&tag_value) {
            return Err(Error::InvalidTag {
                offset: key_start,
                tag: tag_value,
            });
        }
        let tag = tag_value as u32;

        if !matches!(wire, 0 | 1 | 2 | 5) {
            return Err(Error::InvalidWire {
                offset: key_start,
                wire,
            });
        }

        if let Some((previous_tag, previous_wire)) = previous {
            if tag < previous_tag {
                return Err(Error::OutOfOrder {
                    offset: key_start,
                    previous: previous_tag,
                    tag,
                });
            }
            if tag == previous_tag && !(previous_wire == 2 && wire == 2) {
                return Err(Error::Duplicate {
                    offset: key_start,
                    tag,
                });
            }
        }

        match wire {
            0 => {
                let payload_start = cursor;
                let value = decode_varint(input, &mut cursor, frame_end, payload_start)?;
                if tag <= 15 {
                    fields.push(Field::UInt { tag, value });
                }
            }
            1 | 5 => {
                let length = if wire == 1 { 8usize } else { 4usize };
                let available = frame_end - cursor;
                if available < length {
                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: length as u64,
                    });
                }
                cursor += length;
            }
            2 => {
                let length_start = cursor;
                let declared_data_length =
                    decode_varint(input, &mut cursor, frame_end, length_start)?;
                let available = frame_end - cursor;
                if declared_data_length > available as u64 {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length: declared_data_length,
                    });
                }
                let data_length = declared_data_length as usize;
                if tag <= 15 {
                    let data = input[cursor..cursor + data_length].to_vec();

                        tag: previous_tag,

                    }
                }
                cursor += data_length;
            }
            _ => unreachable!(),
        }

        previous = Some((tag, wire));
    }

    Ok((Message { fields }, frame_end))
}
