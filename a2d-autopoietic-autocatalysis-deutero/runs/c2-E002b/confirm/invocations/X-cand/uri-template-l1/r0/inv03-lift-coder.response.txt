#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, String)>,
}

fn validate_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let content_start = open + 1;

    if let Some(relative) = template[content_start..close].find('{') {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let content = &template[content_start..close];
    let (specifiers_start, operator) = if let Some(first) = content.as_bytes().first().copied() {
        if matches!(first, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
            if first != b'+' {
                return Err(Error::UnsupportedOperator {
                    pos: content_start,
                });
            }
            (content_start + 1, true)
        } else {
            (content_start, false)
        }
    } else {
        (content_start, false)
    };

    let _ = operator;

    if specifiers_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut specifier_start = specifiers_start;

    loop {
        let specifier_end = template[specifier_start..close]
            .find(',')
            .map(|relative| specifier_start + relative)
            .unwrap_or(close);

        if specifier_start == specifier_end {
            return Err(Error::EmptyVariable {
                pos: specifier_start,
            });
        }

        let name = &template[specifier_start..specifier_end];
        let bytes = name.as_bytes();

        if !matches!(bytes[0], b'A'..=b'Z' | b'a'..=b'z' | b'_') {
            return Err(Error::InvalidName {
                pos: specifier_start,
            });
        }

        for (index, byte) in bytes.iter().copied().enumerate().skip(1) {
            if !matches!(byte, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'_') {
                return Err(Error::InvalidName {
                    pos: specifier_start + index,
                });
            }
        }

        if seen.iter().any(|previous| previous == name) {
            return Err(Error::DuplicateVariable {
                pos: specifier_start,
            });
        }

        let owned_name = name.to_owned();
        seen.push(owned_name.clone());
        names.push((specifier_start, owned_name));

        if specifier_end == close {
            break;
        }
        specifier_start = specifier_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        names,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression>, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        match template.as_bytes()[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match template[open + 1..].find('}') {
                    Some(relative) => open + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open }),
                };
                expressions.push(validate_expression(template, open, close)?);
                cursor = close + 1;
            }
            _ => {
                let character = template[cursor..].chars().next().unwrap();
                cursor += character.len_utf8();
            }
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    matches!(
        byte,
        b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~'
    )
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn uppercase_hex(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn encode_value(value: &str, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut cursor = 0;

    while cursor < bytes.len() {
        let byte = bytes[cursor];

        if byte == b'%'
            && cursor + 2 < bytes.len()
            && is_hex(bytes[cursor + 1])
            && is_hex(bytes[cursor + 2])
        {
            output.push('%');
            output.push(uppercase_hex(bytes[cursor + 1]) as char);
            output.push(uppercase_hex(bytes[cursor + 2]) as char);
            cursor += 3;
        } else if is_unreserved(byte) {
            output.push(byte as char);
            cursor += 1;
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
            cursor += 1;
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let mut first = true;
        for (_, name) in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
                if !first {
                    output.push(',');
                }
                encode_value(value, &mut output);
                first = false;
            }
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
