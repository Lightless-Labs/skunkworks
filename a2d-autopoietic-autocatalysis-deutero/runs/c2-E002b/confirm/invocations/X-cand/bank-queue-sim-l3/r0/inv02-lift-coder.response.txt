```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    id: u32,
    arrival: u64,
    service: u64,
    priority: u8,
    started: bool,
    remaining: u64,
    pause_time: u64,
}

struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn better_waiter(a: &Job, b: &Job) -> bool {
    if a.priority != b.priority {
        return a.priority < b.priority;
    }
    if a.started != b.started {
        return a.started;
    }
    if a.started {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        a.id < b.id
    } else if a.arrival != b.arrival {
        a.arrival < b.arrival
    } else {
        a.id > b.id
    }
}

fn take_best(waiting: &mut Vec<Job>) -> Option<Job> {
    let index = waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| {
            if better_waiter(a, b) {
                std::cmp::Ordering::Less
            } else if better_waiter(b, a) {
                std::cmp::Ordering::Greater
            } else {
                std::cmp::Ordering::Equal
            }
        })
        .map(|(i, _)| i)?;
    Some(waiting.remove(index))
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Job> = customers
        .iter()
        .map(|c| Job {
            id: c.id,
            arrival: c.arrival,
            service: c.service,
            priority: c.priority,
            started: false,
            remaining: c.service,
            pause_time: 0,
        })
        .collect();

    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then(a.priority.cmp(&b.priority))
            .then(b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;

    loop {
        if next_arrival == arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut tick: Option<u64> = None;
        if next_arrival < arrivals.len() {
            tick = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = &running {
            for t in [Some(active.finish), (active.work_at > 0).then_some(active.work_at)]
                .into_iter()
                .flatten()
            {
                if tick.is_none_or(|current| t < current) {
                    tick = Some(t);
                }
            }
        }

        let now = tick.expect("pending work must have a next event");

        if running.as_ref().is_some_and(|active| active.finish == now) {
            let active = running.take().unwrap();
            trace.push(Event {
                time: now,
                customer: active.job.id,
                kind: Kind::Finish,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let job = arrivals[next_arrival].clone();
            trace.push(Event {
                time: now,
                customer: job.id,
                kind: Kind::Arrive,
            });
            waiting.push(job);
            next_arrival += 1;
        }

        if let Some(active) = &running {
            if now >= active.work_at {
                let best_priority = waiting.iter().map(|j| j.priority).min();
                if best_priority.is_some_and(|p| p < active.job.priority) {
                    let by = waiting
                        .iter()
                        .filter(|j| Some(j.priority) == best_priority)
                        .fold(None, |best: Option<&Job>, candidate| {
                            match best {
                                None => Some(candidate),
                                Some(current) if better_waiter(candidate, current) => {
                                    Some(candidate)
                                }
                                _ => Some(current),
                            }
                        })
                        .unwrap()
                        .id;
                    let mut active = running.take().unwrap();
                    let remaining = active.finish - now;
                    trace.push(Event {
                        time: now,
                        customer: active.job.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    active.job.remaining = remaining;
                    active.job.pause_time = now;
                    waiting.push(active.job);
                }
            }
        }

        if running.is_none() {
            while let Some(mut job) = take_best(&mut waiting) {
                let cost = if last_dispatched == Some(job.id) {
                    0
                } else if last_dispatched.is_none() {
                    0
                } else {
                    switch_cost
                };
                let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: job.id,
                    time: now,
                })?;
                let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                    customer: job.id,
                    time: now,
                })?;

                let kind = if job.started {
                    Kind::Resume {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                };
                trace.push(Event {
                    time: now,
                    customer: job.id,
                    kind,
                });

                job.started = true;
                last_dispatched = Some(job.id);
                if finish == now {
                    trace.push(Event {
                        time: now,
                        customer: job.id,
                        kind: Kind::Finish,
                    });
                } else {
                    running = Some(Running {
                        job,
                        work_at,
                        finish,
                    });
                    break;
                }
            }
        }
    }

    Ok(trace)
}
```
