#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128).map_err(|_| RationalError::Overflow)
    }

    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128
            || n > i64::MAX as i128
            || d < 1
            || d > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128
            + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d).map_err(|_| RationalError::Overflow)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128
            - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d).map_err(|_| RationalError::Overflow)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d).map_err(|_| RationalError::Overflow)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        Self::from_i128(n, d).map_err(|_| RationalError::Overflow)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        Self::from_i128(n, self.den as i128).map_err(|_| RationalError::Overflow)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn err(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() && bytes[i] == b'/' {
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
        }
        if i < bytes.len() {
            return Err(err(i, ParseKind::InvalidByte));
        }

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && &text[..num_end] == "-0" {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let num_digits = &text[num_start..num_end];
        let nmag = parse_bounded(num_digits, if negative { 1u128 << 63 } else { i64::MAX as u128 });
        let nmag = match nmag {
            Some(v) => v,
            None => return Err(err(0, ParseKind::OutOfRange)),
        };
        let num = if negative {
            if nmag == (1u128 << 63) {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        let (den, slash) = if let (Some(ds), Some(de)) = (den_start, den_end) {
            let value = match parse_bounded(&text[ds..de], i64::MAX as u128) {
                Some(v) => v as i64,
                None => return Err(err(ds, ParseKind::OutOfRange)),
            };
            (Some(value), ds - 1)
        } else {
            (None, 0)
        };

        if den == Some(0) {
            return Err(err(den_start.unwrap(), ParseKind::ZeroDenominator));
        }
        if den == Some(1) {
            return Err(err(slash, ParseKind::NonCanonical));
        }
        if den.is_some() && num == 0 {
            return Err(err(slash, ParseKind::NonCanonical));
        }
        if let Some(d) = den {
            if gcd_u128(num.unsigned_abs() as u128, d as u128) > 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
        }

        Ok(Self {
            num,
            den: den.unwrap_or(1),
        })
    }
}

fn parse_bounded(s: &str, limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for b in s.bytes() {
        let digit = (b - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl std::fmt::Display for Rational {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows:self.rows, cols:rhs.cols, data })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut out = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for col in 0..candidate_cols {
            if p == out.rows { break; }
            let mut best: Option<usize> = None;
            for r in p..out.rows {
                let v = out.data[r * out.cols + col];
                if v.num == 0 { continue; }
                best = match best {
                    None => Some(r),
                    Some(old) => {
                        let a = (v.den, v.num.unsigned_abs(), r);
                        let bval = out.data[old * out.cols + col];
                        let b = (bval.den, bval.num.unsigned_abs(), old);
                        if a < b { Some(r) } else { Some(old) }
                    }
                };
            }
            let Some(q) = best else { continue };
            if q != p {
                for c in 0..out.cols {
                    out.data.swap(p * out.cols + c, q * out.cols + c);
                }
                swaps.push((p,q));
            }
            let pivot = out.data[p * out.cols + col];
            for c in 0..out.cols {
                let current = out.data[p * out.cols + c];
                out.data[p * out.cols + c] = current.div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row:p,col:c,step:p,source })?;
            }
            for r in 0..out.rows {
                if r == p { continue; }
                let factor = out.data[r * out.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..out.cols {
                    let product = factor.mul(&out.data[p * out.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:p,source })?;
                    out.data[r * out.cols + c] = out.data[r * out.cols + c].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:p,source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix: out, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> { self.eliminate(self.cols) }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows:self.rows, cols:self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n { data.push(self.data[r*n+c]); }
            for c in 0..n { data.push(Rational { num:(c==r) as i64, den:1 }); }
        }
        let aug = Self { rows:n, cols:width, data };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank:red.pivots.len() });
        }
        let mut right = Vec::with_capacity(n*n);
        for r in 0..n {
            right.extend_from_slice(&red.matrix.data[r*width+n..r*width+width]);
        }
        Ok(Self { rows:n, cols:n, data:right })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        let (m,n,k) = (self.rows,self.cols,rhs.cols);
        if rhs.rows != m {
            return Err(SolveError::RowMismatch { coefficients:m, rhs:rhs.rows });
        }
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..m {
            data.extend_from_slice(&self.data[r*n..r*n+n]);
            data.extend_from_slice(&rhs.data[r*k..r*k+k]);
        }
        let aug = Self { rows:m, cols:width, data };
        let red = aug.eliminate(n).map_err(SolveError::Matrix)?;
        for r in 0..m {
            let coeff_zero = (0..n).all(|c| red.matrix.data[r*width+c].num == 0);
            if coeff_zero {
                if let Some(c) = (0..k).find(|&c| red.matrix.data[r*width+n+c].num != 0) {
                    return Ok(Solution::None {
                        row:r, rhs_col:c,
                        witness:red.matrix.data[r*width..r*width+width].to_vec(),
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let mut free = Vec::new();
        for c in 0..n {
            if !red.pivots.contains(&c) { free.push(c); }
        }
        let plen = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut pdata = vec![Rational { num:0, den:1 }; plen];
        for (i, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                pdata[pivot_col*k+c] = red.matrix.data[i*width+n+c];
            }
        }
        let particular = Self { rows:n, cols:k, data:pdata };
        if rank == n {
            return Ok(Solution::Unique { values:particular, pivots:red.pivots });
        }
        let blen = n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut bdata = vec![Rational { num:0, den:1 }; blen];
        for (j, &free_col) in free.iter().enumerate() {
            bdata[free_col*free.len()+j] = Rational { num:1, den:1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let value = red.matrix.data[i*width+free_col].neg().map_err(|source|
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row:pivot_col, col:j, step:rank, source
                    }))?;
                bdata[pivot_col*free.len()+j] = value;
            }
        }
        Ok(Solution::Infinite {
            particular, free, basis:Self { rows:n, cols:free.len(), data:bdata }, pivots:red.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}
