use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(err(i, ParseKind::InvalidByte));
        }
        let num_end = i;

        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(err(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let neg = bytes[0] == b'-';
        let digits_start = usize::from(neg);
        if num_end - digits_start >= 2 && bytes[digits_start] == b'0' {
            return Err(err(digits_start + 1, ParseKind::NonCanonical));
        }
        if neg && num_end - digits_start == 1 && bytes[digits_start] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let parse_magnitude = |start: usize, end: usize, limit: u128| -> Option<u128> {
            let mut value = 0u128;
            for &b in &bytes[start..end] {
                value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
                if value > limit {
                    return None;
                }
            }
            Some(value)
        };

        let num_limit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        let mag = parse_magnitude(digits_start, num_end, num_limit)
            .ok_or_else(|| err(0, ParseKind::OutOfRange))?;
        let numerator = if neg {
            if mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(mag as i64)
            }
        } else {
            mag as i64
        };

        let slash = num_end;
        let denominator = if let (Some(ds), Some(de)) = (den_start, den_end) {
            Some(
                parse_magnitude(ds, de, i64::MAX as u128)
                    .ok_or_else(|| err(ds, ParseKind::OutOfRange))? as i64,
            )
        } else {
            None
        };

        if let Some(d) = denominator {
            if d == 0 {
                return Err(err(den_start.unwrap(), ParseKind::ZeroDenominator));
            }
            if d == 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if numerator == 0 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator.unsigned_abs() as u128, d as u128) > 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            Ok(Self {
                num: numerator,
                den: d,
            })
        } else {
            Ok(Self {
                num: numerator,
                den: 1,
            })
        }
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

fn arithmetic(row: usize, col: usize, step: usize, source: RationalError) -> MatrixError {
    MatrixError::Arithmetic {
        row,
        col,
        step,
        source,
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut sum = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|e| arithmetic(r, c, k, e))?;
                    sum = sum.add(&term).map_err(|e| arithmetic(r, c, k, e))?;
                }
                data.push(sum);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        mut matrix: Self,
        pivot_limit: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_limit {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), row);
                if chosen
                    .map(|old| {
                        let v = matrix.data[old * matrix.cols + col];
                        key < (v.den, v.num.unsigned_abs(), old)
                    })
                    .unwrap_or(true)
                {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else {
                continue;
            };

            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..matrix.cols {
                let index = p * matrix.cols + c;
                matrix.data[index] = matrix.data[index]
                    .div(&pivot)
                    .map_err(|e| arithmetic(p, c, p, e))?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let product = factor
                        .mul(&matrix.data[p * matrix.cols + c])
                        .map_err(|e| arithmetic(row, c, p, e))?;
                    let index = row * matrix.cols + c;
                    matrix.data[index] = matrix.data[index]
                        .sub(&product)
                        .map_err(|e| arithmetic(row, c, p, e))?;
                }
            }
            pivots.push(col);
            p += 1;
        }

        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        Self::eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let size = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(size);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(Rational {
                    num: if r == c { 1 } else { 0 },
                    den: 1,
                });
            }
        }
        let augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let reduction = Self::eliminate(augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }
        let mut inverse = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                inverse.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let size = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(size);
        for r in 0..self.rows {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                data.push(rhs.data[r * k + c]);
            }
        }
        let augmented = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let reduction = Self::eliminate(augmented, n).map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let coefficients_zero = (0..n)
                .all(|c| reduction.matrix.data[row * width + c].num == 0);
            if coefficients_zero {
                if let Some(rhs_col) =
                    (0..k).find(|&c| reduction.matrix.data[row * width + n + c].num != 0)
                {
                    let witness = reduction.matrix.data[row * width..row * width + width].to_vec();
                    return Ok(Solution::None {
                        row,
                        rhs_col,
                        witness,
                    });
                }
            }
        }

        let pivots = reduction.pivots.clone();
        let free: Vec<usize> = (0..n).filter(|c| !pivots.contains(c)).collect();
        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(
            MatrixError::ShapeOverflow,
        ))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] =
                    reduction.matrix.data[pivot_row * width + n + c];
            }
        }
        let particular = Self {
            rows: n,
            cols: k,
            data: particular,
        };

        if pivots.len() == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let basis_len = n.checked_mul(free.len()).ok_or(SolveError::Matrix(
            MatrixError::ShapeOverflow,
        ))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_len];
        for (basis_col, &free_col) in free.iter().enumerate() {
            basis[free_col * free.len() + basis_col] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
                let coefficient =
                    reduction.matrix.data[pivot_row * width + free_col];
                basis[pivot_col * free.len() + basis_col] =
                    coefficient.neg().map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: basis_col,
                            step: pivots.len(),
                            source,
                        })
                    })?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self {
                rows: n,
                cols: basis_len.checked_div(n.max(1)).unwrap_or(0),
                data: basis,
            },
            pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
