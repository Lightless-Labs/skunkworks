use std::str::FromStr;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(
    field: Field,
    offset: usize,
    kind: fn(Field, usize) -> ParseError,
) -> ParseError {
    kind(field, offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(parse_error(field, start, |field, offset| {
            ParseError::LeadingZero { field, offset }
        }));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| {
                parse_error(field, start, |field, offset| {
                    ParseError::NumberOverflow { field, offset }
                })
            })?;
    }

    let (minimum, maximum) = domain(field);
    if step {
        let width = maximum - minimum + 1;
        if value == 0 || value > width {
            return Err(parse_error(field, start, |field, offset| {
                ParseError::BadStep { field, offset }
            }));
        }
    } else if value < minimum || value > maximum {
        return Err(parse_error(field, start, |field, offset| {
            ParseError::OutOfRange { field, offset }
        }));
    }

    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let (base_end, step_pos) = match slash_positions.as_slice() {
        [] => (item_end, None),
        [pos] => (*pos + item_start, Some(*pos + item_start + 1)),
        _ => {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            })
        }
    };

    let base = &bytes[item_start..base_end];
    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let (start_range, end_range, range_end_pos) = if is_star {
        (None, None, None)
    } else {
        match hyphens.as_slice() {
            [] if !base.is_empty() => {
                (Some((item_start, base_end)), None, None)
            }
            [pos] if *pos > 0 && *pos + 1 < base.len() => {
                let pos = item_start + *pos;
                (
                    Some((item_start, pos)),
                    Some((pos + 1, base_end)),
                    Some(pos + 1),
                )
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                })
            }
        }
    };

    if step_pos.is_some() && base.is_empty() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let step = if let Some(step_start) = step_pos {
        if step_start >= item_end {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        Some(parse_number(
            bytes,
            step_start,
            item_end,
            field,
            true,
        )?)
    } else {
        None
    };

    let (minimum, maximum) = domain(field);
    let (start, end) = if is_star {
        (minimum, maximum)
    } else {
        let start_span = start_range.expect("numeric base");
        let start = parse_number(bytes, start_span.0, start_span.1, field, false)?;
        if let Some(end_span) = end_range {
            let end = parse_number(bytes, end_span.0, end_span.1, field, false)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: range_end_pos.unwrap(),
                });
            }
            (start, end)
        } else if step.is_some() {
            (start, maximum)
        } else {
            (start, start)
        }
    };

    let stride = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(stride) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, maximum) = domain(field);
    let mut selected = vec![false; maximum as usize + 1];
    let mut item_start = 0;

    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(bytes, item_start, i, field, &mut selected)?;
            item_start = i + 1;
        }
    }
    parse_item(bytes, item_start, bytes.len(), field, &mut selected)?;

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected = Vec::with_capacity(5);
    for (text, field) in pieces.iter().zip(FIELDS) {
        selected.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_is_star: pieces[2].as_bytes() == b"*",
        day_of_week_is_star: pieces[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.selected[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.contains(Field::DayOfMonth, at.day)
            && self.contains(Field::DayOfWeek, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

impl PartialEq for Cron {
    fn eq(&self, other: &Self) -> bool {
        self.selected == other.selected
            && self.day_of_month_is_star == other.day_of_month_is_star
            && self.day_of_week_is_star == other.day_of_week_is_star
    }
}

impl Eq for Cron {}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        true
    } else {
        false
    }
}
