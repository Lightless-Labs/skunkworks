#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_stars: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if step {
        let (_, max) = bounds(field);
        if value < 1 || value > max - bounds(field).0 + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field, selected: &mut [bool]) -> Result<bool, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = bounds(field);
    let mut item_start = 0;
    let mut exact_star = false;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash = item.iter().position(|&b| b == b'/');
        if let Some(pos) = slash {
            if item[pos + 1..].contains(&b'/') {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }
        let base_end = slash.unwrap_or(item.len());
        let base = &item[..base_end];

        let star = base == b"*";
        let hyphen = base.iter().position(|&b| b == b'-');
        let valid_base = if star {
            true
        } else if let Some(pos) = hyphen {
            pos > 0
                && pos + 1 < base.len()
                && !base[pos + 1..].contains(&b'-')
                && !base[..pos].contains(&b'*')
                && !base[pos + 1..].contains(&b'*')
        } else {
            !base.is_empty() && !base.contains(&b'*')
        };

        if !valid_base {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        if let Some(pos) = slash {
            if pos + 1 == item.len() || item[pos + 1..].contains(&b'-') || item[pos + 1..].contains(&b'*') {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            if !item[pos + 1..].iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let (start_value, end_value, start_offset, end_offset) = if star {
            (min, max, item_start, item_start)
        } else if let Some(pos) = hyphen {
            let left = parse_number(&bytes, item_start, item_start + pos, field, false)?;
            let right_start = item_start + pos + 1;
            let right = parse_number(&bytes, right_start, item_end - (slash.map_or(0, |_| item.len() - base_end)), field, false)?;
            if left > right {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: right_start,
                });
            }
            (left, right, item_start, right_start)
        } else {
            let value = parse_number(&bytes, item_start, item_start + base_end, field, false)?;
            (value, if slash.is_some() { max } else { value }, item_start, item_start)
        };

        let step_value = if let Some(pos) = slash {
            let step_start = item_start + pos + 1;
            Some(parse_number(&bytes, step_start, item_end, field, true)?)
        } else {
            None
        };

        let step = step_value.unwrap_or(1);
        let mut value = start_value;
        while value <= end_value {
            selected[(value - min) as usize] = true;
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    exact_star = text == "*";
    Ok(exact_star)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (min, max) = bounds(FIELDS[i]);
        vec![false; (max - min + 1) as usize]
    });
    let mut day_stars = [false; 2];

    for i in 0..5 {
        let exact_star = parse_field(pieces[i], FIELDS[i], &mut selected[i])?;
        if i == 2 {
            day_stars[0] = exact_star;
        } else if i == 4 {
            day_stars[1] = exact_star;
        }
    }

    Ok(Cron {
        selected,
        day_stars,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min + i as u32).map(|v| v as u8))
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let (min, max) = bounds(field);
        let value = value as u32;
        value >= min
            && value <= max
            && self.selected[field_index(field)][(value - min) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::DayOfMonth, at.day)
            && self.contains(Field::Month, at.month)
            && self.contains(Field::DayOfWeek, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if current == after {
                advance_minute(&mut current);
            }
            if current.year > 2399 {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
            advance_minute(&mut current);
        }
    }
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) {
    if at.minute < 59 {
        at.minute += 1;
        return;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return;
    }
    at.hour = 0;
    if at.day < month_length(at.year, at.month) {
        at.day += 1;
        return;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return;
    }
    at.month = 1;
    at.year += 1;
}
