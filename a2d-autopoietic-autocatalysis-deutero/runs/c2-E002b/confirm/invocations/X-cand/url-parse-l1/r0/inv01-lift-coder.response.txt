#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }

    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(input.len(), |offset| authority_start + offset);
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(offset, byte)| (byte == b'@').then_some(offset))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_port_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_port_start..];
    let host_port_bytes = host_port.as_bytes();
    let host_port_colon = host_port_bytes.iter().position(|&byte| byte == b':');
    let host_end = host_port_colon.unwrap_or(host_port.len());
    let host_start_at = authority_start + host_port_start;

    if host_end == 0 {
        return Err(ParseError::InvalidHost { at: host_start_at });
    }

    for (offset, &byte) in host_port_bytes[..host_end].iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_start_at + offset,
            });
        }
    }

    let host = host_port[..host_end].to_ascii_lowercase();
    let port = if let Some(colon_offset) = host_port_colon {
        let port_start = colon_offset + 1;
        let port_text = &host_port[port_start..];
        let port_at = host_start_at + port_start;

        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_at });
        }

        if let Some(offset) = port_text.bytes().position(|byte| !byte.is_ascii_digit()) {
            return Err(ParseError::InvalidPort { at: port_at + offset });
        }

        if port_text.len() >= 2 && port_text.starts_with('0') {
            return Err(ParseError::InvalidPort { at: port_at });
        }

        let mut value: u32 = 0;
        for byte in port_text.bytes() {
            value = match value
                .checked_mul(10)
                .and_then(|n| n.checked_add(u32::from(byte - b'0')))
            {
                Some(n) => n,
                None => return Err(ParseError::InvalidPort { at: port_at }),
            };
        }

        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_at });
        }
        Some(value as u16)
    } else {
        None
    };

    let fragment_pos = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| authority_end + offset);

    let before_fragment_end = fragment_pos.unwrap_or(input.len());
    let query_pos = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path_text = &input[authority_end..path_end];
    let path = if path_text.is_empty() {
        "/".to_owned()
    } else {
        path_text.to_owned()
    };

    let query = query_pos.map(|at| input[at + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|at| input[at + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
    })
}
