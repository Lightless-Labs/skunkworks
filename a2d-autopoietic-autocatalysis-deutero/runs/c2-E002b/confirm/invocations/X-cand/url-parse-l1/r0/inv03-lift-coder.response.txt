#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map_or(input.len(), |relative| authority_start + relative);

    let authority = &input[authority_start..authority_end];
    let authority_bytes = &bytes[authority_start..authority_end];

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_relative) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_start_relative..];
    let host_port_bytes = &authority_bytes[host_start_relative..];
    let host_start = authority_start + host_start_relative;

    let host_port_colon = host_port_bytes.iter().position(|&b| b == b':');
    let host_end_relative = host_port_colon.unwrap_or(host_port.len());

    if host_end_relative == 0 {
        return Err(ParseError::InvalidHost { at: host_start });
    }

    for (relative, &byte) in host_port_bytes[..host_end_relative].iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_start + relative,
            });
        }
    }

    let host = host_port[..host_end_relative].to_ascii_lowercase();
    let port = if let Some(colon_relative) = host_port_colon {
        let port_start = host_start + colon_relative + 1;
        let port_text = &host_port[colon_relative + 1..];
        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        if let Some(relative) = port_text.bytes().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidPort {
                at: port_start + relative,
            });
        }

        let port_bytes = port_text.as_bytes();
        if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        let mut value: u32 = 0;
        for &digit in port_bytes {
            value = match value
                .checked_mul(10)
                .and_then(|v| v.checked_add((digit - b'0') as u32))
            {
                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
        }
        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        Some(value as u16)
    } else {
        None
    };

    let remainder = &input[authority_end..];
    let fragment_relative = remainder.find('#');
    let before_fragment = fragment_relative.map_or(remainder, |i| &remainder[..i]);
    let fragment = fragment_relative.map(|i| remainder[i + 1..].to_owned());

    let query_relative = before_fragment.find('?');
    let path = query_relative
        .map_or(before_fragment, |i| &before_fragment[..i]);
    let query = query_relative.map(|i| before_fragment[i + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path: if path.is_empty() {
            "/".to_owned()
        } else {
            path.to_owned()
        },
        query,
        fragment,
    })
}
