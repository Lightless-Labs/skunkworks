#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 2 && index < input.len() {
                self.tail[self.tail_len] = input[index];
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 2 && index < input.len() {
                let a = self.tail[0];
                let b = self.tail[1];
                let c = input[index];
                append_triplet(&mut output, a, b, c);
                self.tail_len = 0;
                index += 1;
            } else if self.tail_len < 2 || index == input.len() {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            append_triplet(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        let remaining = input.len() - index;
        if remaining > 0 {
            self.tail[0] = input[index];
            self.tail_len = 1;
            if remaining == 2 {
                self.tail[1] = input[index + 1];
                self.tail_len = 2;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        if self.tail_len == 1 {
            let a = self.tail[0];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push_str("==");
        } else if self.tail_len == 2 {
            let a = self.tail[0];
            let b = self.tail[1];
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn append_triplet(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub struct Decoder {
    group: [u8; 4],
    group_len: usize,
    group_start: usize,
    offset: usize,
    terminal: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            group_len: 0,
            group_start: 0,
            offset: 0,
            terminal: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current = self.offset;
            self.offset += 1;

            let scan_result = if self.terminal {
                Err(Error::BadPadding { offset: current })
            } else if self.pending_second_padding.is_some() && byte != b'=' {
                Err(Error::BadPadding { offset: current })
            } else if self.group_len < 2 && byte == b'=' {
                Err(Error::BadPadding { offset: current })
            } else if byte == b'=' {
                self.consume_padding(byte, current, &mut output)
            } else if let Some(value) = digit(byte) {
                self.consume_digit(value, byte, current, &mut output)
            } else {
                Err(Error::InvalidByte {
                    offset: current,
                    byte,
                })
            };

            if let Err(error) = scan_result {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(output)
    }

    fn consume_padding(
        &mut self,
        _byte: u8,
        current: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        match self.group_len {
            2 => {
                self.group[2] = b'=';
                self.group_len = 3;
                self.pending_second_padding = Some(current);
                Ok(())
            }
            3 if self.pending_second_padding.is_some() => {
                self.group[3] = b'=';
                self.group_len = 4;
                output.push((self.group[0] << 2) | (self.group[1] >> 4));
                self.reset_group();
                self.terminal = true;
                Ok(())
            }
            3 => {
                self.group[3] = b'=';
                self.group_len = 4;
                output.push((self.group[0] << 2) | (self.group[1] >> 4));
                output.push((self.group[1] << 4) | (self.group[2] >> 2));
                self.reset_group();
                self.terminal = true;
                Ok(())
            }
            _ => Err(Error::BadPadding { offset: current }),
        }
    }

    fn consume_digit(
        &mut self,
        value: u8,
        _byte: u8,
        current: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.group_len == 0 {
            self.group_start = current;
        }
        self.group[self.group_len] = value;
        self.group_len += 1;

        if self.group_len == 4 {
            output.push((self.group[0] << 2) | (self.group[1] >> 4));
            output.push((self.group[1] << 4) | (self.group[2] >> 2));
            output.push((self.group[2] << 6) | self.group[3]);
            self.reset_group();
        }
        Ok(())
    }

    fn reset_group(&mut self) {
        self.group_len = 0;
        self.pending_second_padding = None;
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }
        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.group_start,
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
