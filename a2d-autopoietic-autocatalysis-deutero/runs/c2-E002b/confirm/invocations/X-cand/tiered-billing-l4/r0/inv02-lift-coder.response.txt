#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct DayRange {
    pub start: u32,
    pub end: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Rounding {

    Up,
    HalfEven,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Proration {
    pub active_days: u32,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Meter {
    pub name: String,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
    Month,
    Period,
    Credit { index: usize },
    CreditOrder { index: usize },
    InvoiceDiscount,
    InvalidName { index: usize },
    DuplicateName { index: usize },
    Discount,
    CombinedOverflow,
    Meter { index: usize, source: Box<Error> },
    InvoiceOverflow,
}

fn band_amount(units: u64, unit: u64, start: u64) -> Option<u64> {
    units.checked_mul(unit)?.checked_add(start)
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().is_some_and(|tier| tier.up_to.is_some()) {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn quote_tiers(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut amount = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                previous = bound;
                remaining.min(width)
            }
            None => remaining,
        };

        if units == 0 {
            continue;
        }

        let charge = band_amount(units, tier.unit, tier.start)
            .ok_or(Error::Overflow { index })?;
        amount = amount
            .checked_add(charge)
            .ok_or(Error::Overflow { index })?;
        bands.push(Band {
            index,
            units,
            amount: charge,
        });
        remaining -= units;
    }

    Ok(Charge { amount, bands })
}

fn volume_tiers(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let (index, tier) = tiers
        .iter()
        .enumerate()
        .find(|(_, tier)| tier.up_to.is_none_or(|bound| bound >= quantity))
        .expect("validated tiers always have a final unbounded tier");

    let amount = band_amount(quantity, tier.unit, tier.start)
        .ok_or(Error::Overflow { index })?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index,
            units: quantity,
            amount,
        }],
    })
}

fn charge(quantity: u64, price: &Price, mode: Mode) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                })
            } else {
                let amount =
                    band_amount(quantity, *unit, *start).ok_or(Error::Overflow { index: 0 })?;
                Ok(Charge {
                    amount,
                    bands: vec![Band {
                        index: 0,
                        units: quantity,
                        amount,
                    }],
                })
            }
        }
        Price::Tiers(tiers) => match mode {
            Mode::Graduated => quote_tiers(quantity, tiers),
            Mode::Volume => volume_tiers(quantity, tiers),
        },
    }
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    charge(quantity, price, Mode::Graduated)
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = charge(quantity, price, mode)?;
    let minimum_adjustment = if quantity == 0 {
        0
    } else {
        minimum.saturating_sub(charge.amount)
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .expect("minimum adjustment cannot make total exceed minimum or charge");
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if !matches!(days, 28 | 29 | 30 | 31) {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut previous_end = start;
    let mut credited_days = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        credited_days += credit.end - credit.start;
        previous_end = credit.end;
    }

    let active_days = end - start - credited_days;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice_remainder = remainder * 2;
            if twice_remainder > denominator
                || (twice_remainder == denominator && quotient % 2 == 1)
            {
                quotient + 1
            } else {
                quotient
            }
        }
    };

    Ok(Proration {
        active_days,
        amount: rounded as u64,
    })
}

fn valid_name(name: &str) -> bool {
    let len = name.len();
    (1..=32).contains(&len)
        && !name
            .bytes()
            .any(|b| b <= 0x1f || b == 0x7f || b == b'|' || b == b':')
}

fn discount_amount(base: u64, bps: u32) -> u64 {
    ((base as u128) * (bps as u128) / 10_000) as u64
}

pub fn invoice(
    meters: &[Meter],
    invoice_discount_bps: u32,
    credit: u64,
) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 {
        return Err(Error::InvoiceDiscount);
    }

    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) {
            return Err(Error::InvalidName { index });
        }
        if meters[..index]
            .iter()
            .any(|previous| previous.name == meter.name)
        {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;

    for (index, meter) in meters.iter().enumerate() {
        if meter.discount_bps > 10_000 {
            return Err(Error::Meter {
                index,
                source: Box::new(Error::Discount),
            });
        }

        let bill = bill(
            meter.quantity,
            &meter.price,
            meter.mode.clone(),
            meter.minimum,
        )
        .map_err(|source| Error::Meter {
            index,
            source: Box::new(source),
        })?;

        let monthly = prorate(
            meter.monthly_fee,
            meter.days,
            meter.start,
            meter.end,
            &meter.credits,
            meter.rounding.clone(),
        )
        .map_err(|source| Error::Meter {
            index,
            source: Box::new(source),
        })?;

        let base = bill
            .total
            .checked_add(monthly.amount)
            .ok_or(Error::Meter {
                index,
                source: Box::new(Error::CombinedOverflow),
            })?;
        let discount = discount_amount(base, meter.discount_bps);
        let total = base - discount;

        subtotal = subtotal
            .checked_add(total)
            .ok_or(Error::InvoiceOverflow)?;

        lines.push(InvoiceLine {
            name: meter.name.clone(),
            usage: bill.charge.amount,
            minimum_adjustment: bill.minimum_adjustment,
            monthly: monthly.amount,
            discount,
            total,
        });
    }

    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));

    let discount = discount_amount(subtotal, invoice_discount_bps);
    let credit = credit.min(subtotal - discount);
    let total = subtotal - discount - credit;

    let mut rendered = String::new();
    for line in &lines {
        use std::fmt::Write;
        writeln!(
            rendered,
            "{}:{}|{}|{}|{}|{}|{}",
            line.name.len(),
            line.name,
            line.usage,
            line.minimum_adjustment,
            line.monthly,
            line.discount,
            line.total
        )
        .expect("writing to String cannot fail");
    }
    use std::fmt::Write;
    writeln!(rendered, "TOTAL|{}|{}|{}|{}", subtotal, discount, credit, total)
        .expect("writing to String cannot fail");

    Ok(Invoice {
        lines,
        subtotal,
        discount,
        credit,
        total,
        rendered,
    })
}
