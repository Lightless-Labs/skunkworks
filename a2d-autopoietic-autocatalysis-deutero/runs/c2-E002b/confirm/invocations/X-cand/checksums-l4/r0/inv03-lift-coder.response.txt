pub enum VerifyError {

    Length,

const ADLER_MOD: u32 = 65521;

fn crc32_byte(mut crc: u32, byte: u8) -> u32 {
    crc ^= byte as u32;
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb88320;
        }
    }
    crc
}

fn adler_byte(a: &mut u32, b: &mut u32, byte: u8) {
    *a = (*a + byte as u32) % ADLER_MOD;
    *b = (*b + *a) % ADLER_MOD;
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    for &byte in data {
        crc = crc32_byte(crc, byte);
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    for &byte in data {
        adler_byte(&mut a, &mut b, byte);
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn for_each_selected<F>(data: &[u8], spans: &[Span], mut f: F)
where
    F: FnMut(u8),
{
    for span in spans {
        for &byte in &data[span.start..span.end] {
            f(byte);
        }
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for_each_selected(data, spans, |byte| {
        crc = crc32_byte(crc, byte);
        adler_byte(&mut a, &mut b, byte);
    });
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc_raw = crc32_byte(self.crc_raw, byte);
            adler_byte(&mut self.adler_a, &mut self.adler_b, byte);
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for_each_selected(data, spans, |byte| {
            self.crc_raw = crc32_byte(self.crc_raw, byte);
            adler_byte(&mut self.adler_a, &mut self.adler_b, byte);
        });
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let crc = parse_hex(bytes, 1, 9)?;
        let adler = parse_hex(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8, offset: usize) -> Result<u32, SnapshotError> {
    match byte {
        b'0'..=b'9' => Ok((byte - b'0') as u32),
        b'a'..=b'f' => Ok((byte - b'a' + 10) as u32),
        _ => Err(SnapshotError::Hex { offset }),
    }
}

fn parse_hex(bytes: &[u8], start: usize, end: usize) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        value = (value << 4) | hex_value(bytes[offset], offset)?;
    }
    Ok(value)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut i = 0;
    while i + 1 < data.len() {
        let word = ((data[i] as u32) << 8) | data[i + 1] as u32;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
        i += 2;
    }
    if i < data.len() {
        let word = (data[i] as u32) << 8;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending = None;
    for_each_selected(data, spans, |byte| {
        if let Some(high) = pending.take() {
            let word = ((high as u32) << 8) | byte as u32;
            a = (a + word) % 65535;
            b = (b + a) % 65535;
        } else {
            pending = Some(byte);
        }
    });
    if let Some(high) = pending {
        let word = (high as u32) << 8;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }
    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
        }
    }
    if variant == Crc16::X25 {
        crc ^ 0xffff
    } else {
        crc
    }
}

fn crc16_byte(mut crc: u16, byte: u8, variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            crc ^= (byte as u16) << 8;
            for _ in 0..8 {
                let top = crc & 0x8000;
                crc = crc.wrapping_shl(1);
                if top != 0 {
                    crc ^= 0x1021;
                }
            }
            crc
        }
        Crc16::X25 => {
            crc ^= byte as u16;
            for _ in 0..8 {
                let low = crc & 1;
                crc >>= 1;
                if low != 0 {
                    crc ^= 0x8408;
                }
            }
            crc
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut adler_a = 1u32;
    let mut adler_b = 0u32;
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_pending = None;
    let mut crc16_reg = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };

    for_each_selected(data, spans, |byte| {
        crc = crc32_byte(crc, byte);
        adler_byte(&mut adler_a, &mut adler_b, byte);

        f16_a = (f16_a + byte as u32) % 255;
        f16_b = (f16_b + f16_a) % 255;

        if let Some(high) = f32_pending.take() {
            let word = ((high as u32) << 8) | byte as u32;
            f32_a = (f32_a + word) % 65535;
            f32_b = (f32_b + f32_a) % 65535;
        } else {
            f32_pending = Some(byte);
        }

        crc16_reg = crc16_byte(crc16_reg, byte, variant);
    });

    if let Some(high) = f32_pending {
        let word = (high as u32) << 8;
        f32_a = (f32_a + word) % 65535;
        f32_b = (f32_b + f32_a) % 65535;
    }

    if variant == Crc16::X25 {
        crc16_reg ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_reg,
    })
}

fn crc32_zero_byte(mut crc: u32) -> u32 {
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb88320;
        }
    }
    crc
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for n in 0..32 {
        square[n] = gf2_matrix_times(matrix, matrix[n]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];

    odd[0] = 0xedb88320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }

    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }

    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn selected_len(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |total, span| {
        total + (span.end - span.start) as u64
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let hashes = checksums_spans(data, spans)?;
    let len = selected_len(spans);
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> Result<u64, usize> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => (bytes[offset] - b'0') as u64,
            b'a'..=b'f' => (bytes[offset] - b'a' + 10) as u64,
            _ => return Err(offset),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> Result<u32, usize> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => (bytes[offset] - b'0') as u32,
            b'a'..=b'f' => (bytes[offset] - b'a' + 10) as u32,
            _ => return Err(offset),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let length = parse_hex_u64(bytes, 4, 20).map_err(|offset| VerifyError::Hex { offset })?;
    let expected_crc =
        parse_hex_u32(bytes, 25, 33).map_err(|offset| VerifyError::Hex { offset })?;
    let expected_adler =
        parse_hex_u32(bytes, 40, 48).map_err(|offset| VerifyError::Hex { offset })?;

    if length != selected_len(spans) {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
