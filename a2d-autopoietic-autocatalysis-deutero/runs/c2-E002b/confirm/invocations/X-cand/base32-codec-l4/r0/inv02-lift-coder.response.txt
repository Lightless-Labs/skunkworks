#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut data = self.pending.clone();
        data.extend_from_slice(input);

        let block_len = data.len() / 5 * 5;
        let output = encode_symbols(&data[..block_len], self.config.alphabet);
        self.pending = data[block_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = encode_symbols(&self.pending, self.config.alphabet);
        let symbols = (self.pending.len() * 8 + 4) / 5;
        if self.config.padding == Padding::Required {
            let pad_count = (8 - symbols % 8) % 8;
            output.extend(std::iter::repeat_n('=', pad_count));
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_symbols(input: &[u8], alphabet: Alphabet) -> String {
    if input.is_empty() {
        return String::new();
    }

    let table = alphabet_bytes(alphabet);
    let mut output = String::with_capacity((input.len() * 8 + 4) / 5);

    let mut acc: u32 = 0;
    let mut bits = 0usize;
    for &byte in input {
        acc = (acc << 8) | byte as u32;
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            output.push(table[((acc >> bits) & 0x1f) as usize] as char);
        }
    }
    if bits > 0 {
        let value = (acc << (5 - bits)) & 0x1f;
        output.push(table[value as usize] as char);
    }

    output
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    pad_count: usize,
    first_pad_offset: Option<usize>,
    expected_pads: Option<usize>,
    last_value: u8,
    bits: u32,
    bit_count: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            pad_count: 0,
            first_pad_offset: None,
            expected_pads: None,
            last_value: 0,
            bits: 0,
            bit_count: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                let residue = next.data_count % 8;
                let expected = match residue {
                    0 => return Err(StreamError::Padding { offset }),
                    2 => 6,
                    4 => 4,
                    5 => 3,
                    7 => 1,
                    _ => return Err(StreamError::Padding { offset }),
                };

                if let Some(required) = next.expected_pads {
                    if next.pad_count >= required {
                        return Err(StreamError::Padding { offset });
                    }
                } else {
                    next.expected_pads = Some(expected);
                    next.first_pad_offset = Some(offset);
                }

                next.pad_count += 1;
                next.position += 1;
                continue;
            }

            let value = match symbol_value(byte, next.config.alphabet) {
                Some(value) => value,
                None => return Err(StreamError::InvalidByte { offset, byte }),
            };

            if byte.is_ascii_lowercase() && next.config.canonical {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: byte.to_ascii_uppercase(),
                });
            }

            if next.first_pad_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count += 1;
            next.last_value = value;
            next.bits = (next.bits << 5) | value as u32;
            next.bit_count += 5;
            while next.bit_count >= 8 {
                next.bit_count -= 8;
                output.push(((next.bits >> next.bit_count) & 0xff) as u8);
            }
            next.position += 1;
        }

        self.copy_state_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected_for_residue = match residue {
            0 => Some(0),
            2 => Some(6),
            4 => Some(4),
            5 => Some(3),
            7 => Some(1),
            _ => None,
        };

        if expected_for_residue.is_none() {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = expected_for_residue.unwrap();
            if self.pad_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_pad_offset.unwrap_or(self.position),
                });
            }
        }

        let unused = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => unreachable!(),
        };
        if unused > 0 && (self.last_value & ((1u8 << unused) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.position.saturating_sub(self.pad_count + 1),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            pad_count: self.pad_count,
            first_pad_offset: self.first_pad_offset,
            expected_pads: self.expected_pads,
            last_value: self.last_value,
            bits: self.bits,
            bit_count: self.bit_count,
            closed: self.closed,
        }
    }

    fn copy_state_from(&mut self, other: Self) {
        self.position = other.position;
        self.data_count = other.data_count;
        self.pad_count = other.pad_count;
        self.first_pad_offset = other.first_pad_offset;
        self.expected_pads = other.expected_pads;
        self.last_value = other.last_value;
        self.bits = other.bits;
        self.bit_count = other.bit_count;
        self.closed = other.closed;
    }
}

fn symbol_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let upper = byte.to_ascii_uppercase();
    alphabet_bytes(alphabet)
        .iter()
        .position(|&candidate| candidate == upper)
        .map(|value| value as u8)
}
