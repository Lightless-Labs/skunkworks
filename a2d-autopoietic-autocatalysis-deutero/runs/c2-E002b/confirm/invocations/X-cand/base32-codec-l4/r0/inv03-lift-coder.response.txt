#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut bytes = Vec::with_capacity(self.pending.len() + input.len());
        bytes.extend_from_slice(&self.pending);
        bytes.extend_from_slice(input);

        let complete_len = (bytes.len() / 5) * 5;
        let mut output = String::new();
        encode_bytes(
            &bytes[..complete_len],
            self.config.alphabet,
            &mut output,
        );

        self.pending.clear();
        self.pending.extend_from_slice(&bytes[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        let symbols = encode_bytes(&self.pending, self.config.alphabet, &mut output);

        if self.config.padding == Padding::Required && symbols != 0 {
            let padding = (8 - symbols % 8) % 8;
            output.extend(std::iter::repeat_n('=', padding));
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(input: &[u8], alphabet: Alphabet, output: &mut String) -> usize {
    let table = alphabet_bytes(alphabet);
    let mut accumulator: u32 = 0;
    let mut bits = 0usize;
    let mut symbols = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            output.push(table[((accumulator >> bits) & 31) as usize] as char);
            symbols += 1;
        }
    }

    if bits > 0 {
        output.push(table[((accumulator << (5 - bits)) & 31) as usize] as char);
        symbols += 1;
    }

    symbols
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    accumulator: u32,
    bits: usize,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            accumulator: 0,
            bits: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let (is_padding, value, canonical_byte) =
                match classify(byte, next.config.alphabet) {
                    Some((is_padding, value, canonical_byte)) => {
                        (is_padding, value, canonical_byte)
                    }
                    None => return Err(StreamError::InvalidByte { offset, byte }),
                };

            if !is_padding && byte != canonical_byte && next.config.canonical {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: canonical_byte,
                });
            }

            if is_padding {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                if next.first_padding_offset.is_none() {
                    let residue = next.data_count % 8;
                    let expected = match residue {
                        2 => 6,
                        4 => 4,
                        5 => 3,
                        7 => 1,
                        _ => return Err(StreamError::Padding { offset }),
                    };
                    next.first_padding_offset = Some(offset);
                    next.expected_padding = Some(expected);
                }

                let expected = next.expected_padding.unwrap_or(0);
                if next.padding_count >= expected {
                    return Err(StreamError::Padding { offset });
                }
                next.padding_count += 1;
            } else {
                if next.first_padding_offset.is_some() {
                    return Err(StreamError::Padding { offset });
                }

                next.data_count = next.data_count.checked_add(1).ok_or(
                    StreamError::InvalidLength { offset },
                )?;
                next.last_data_offset = Some(offset);
                next.accumulator = (next.accumulator << 5) | u32::from(value);
                next.bits += 5;

                while next.bits >= 8 {
                    next.bits -= 8;
                    output.push(((next.accumulator >> next.bits) & 0xff) as u8);
                }
            }

            next.position = next.position.checked_add(1).ok_or(
                StreamError::InvalidLength { offset },
            )?;
        }

        self.copy_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => unreachable!(),
        };

        if unused_bits > 0 && (self.accumulator & ((1u32 << unused_bits) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            accumulator: self.accumulator,
            bits: self.bits,
            last_data_offset: self.last_data_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }

    fn copy_from(&mut self, other: Self) {
        self.position = other.position;
        self.data_count = other.data_count;
        self.accumulator = other.accumulator;
        self.bits = other.bits;
        self.last_data_offset = other.last_data_offset;
        self.first_padding_offset = other.first_padding_offset;
        self.padding_count = other.padding_count;
        self.expected_padding = other.expected_padding;
        self.closed = other.closed;
    }
}

fn classify(byte: u8, alphabet: Alphabet) -> Option<(bool, u8, u8)> {
    if byte == b'=' {
        return Some((true, 0, b'='));
    }

    let upper = byte.to_ascii_uppercase();
    let table = alphabet_bytes(alphabet);
    table
        .iter()
        .position(|&candidate| candidate == upper)
        .map(|value| (false, value as u8, upper))
}
