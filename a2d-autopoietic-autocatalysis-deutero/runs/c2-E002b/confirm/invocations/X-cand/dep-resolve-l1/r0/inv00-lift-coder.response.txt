#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    let (prefix, version) = match constraint {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{prefix}{}", version_text(version))
}

fn canonical_constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    let mut encoded = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            encoded.push('%');
            encoded.push_str(&format!("{byte:02X}"));
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            canonical_constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            canonical_constraints_text(requirements)
        ),
    }
}

fn validate_input(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::HashSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::HashSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    Ok(())
}

fn search(
    catalog: &[Package],
    active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    for requirements in active.values_mut() {
        canonicalize(requirements);
    }

    let missing = active
        .keys()
        .filter(|name| !catalog.iter().any(|package| package.name == ***name))
        .next()
        .cloned();
    if let Some(package) = missing {
        return Err(ResolveError::Missing { package });
    }

    for (package, version) in selected.iter() {
        let requirements = active.get(package).map(Vec::as_slice).unwrap_or(&[]);
        if requirements
            .iter()
            .any(|constraint| !matches(*version, *constraint))
        {
            return Err(ResolveError::Conflict {
                package: package.clone(),
                selected: *version,
                requirements: requirements.to_vec(),
            });
        }
    }

    for (package, requirements) in active.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let catalog_package = catalog
            .iter()
            .find(|candidate| candidate.name == *package)
            .expect("missing packages handled above");
        if !catalog_package.releases.iter().any(|release| {
            requirements
                .iter()
                .all(|constraint| matches(release.version, *constraint))
        }) {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements: requirements.clone(),
            });
        }
    }

    let next = active
        .keys()
        .find(|package| !selected.contains_key(*package))
        .cloned();
    let Some(package) = next else {
        return Ok(());
    };

    let requirements = active.get(&package).cloned().unwrap_or_default();
    let catalog_package = catalog
        .iter()
        .find(|candidate| candidate.name == package)
        .expect("missing packages handled above");

    let version = catalog_package
        .releases
        .iter()
        .filter(|release| {
            requirements
                .iter()
                .all(|constraint| matches(release.version, *constraint))
        })
        .map(|release| release.version)
        .max()
        .expect("zero candidates handled above");

    selected.insert(package.clone(), version);

    let release = catalog_package
        .releases
        .iter()
        .find(|release| release.version == version)
        .expect("selected version came from this package");
    for dependency in &release.deps {
        active
            .entry(dependency.package.clone())
            .or_default()
            .push(dependency.constraint);
    }

    search(catalog, active, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_input(catalog, roots)?;

    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected = std::collections::BTreeMap::<String, Version>::new();
    search(catalog, &mut active, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}
