#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, &byte) in input.as_bytes().iter().enumerate() {
        if !(32..=126).contains(&byte) && byte != b'\t' && byte != b'\n' && byte != b'\r' {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| c == ' ' || c == '\t' || c == '\n' || c == '\r') {
        if word.is_empty() {
            continue;
        }

        let mut remaining = word;

        while !remaining.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if remaining.len() <= width {
                let needed = remaining.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(remaining);
                    break;
                }

                if !line.is_empty() {
                    lines.push(std::mem::take(&mut line));
                    continue;
                }
            }

            if line.is_empty() {
                let (part, rest) = remaining.split_at(width.min(remaining.len()));
                line.push_str(part);
                remaining = rest;
                if !remaining.is_empty() {
                    lines.push(std::mem::take(&mut line));
                }
                continue;
            }

            let available = width - line.len() - 1;
            if available >= 2 {
                line.push(' ');
                let (part, rest) = remaining.split_at(available);
                line.push_str(part);
                remaining = rest;
                lines.push(std::mem::take(&mut line));
            } else {
                lines.push(std::mem::take(&mut line));
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
