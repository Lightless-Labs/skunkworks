use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct RuntimeJob {
    job: Job,
    deps: Vec<usize>,
    state: State,
    remaining: Option<u64>,
    started_at: u64,
}

pub struct Pool {
    jobs: Vec<RuntimeJob>,
    by_id: HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    started: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if tag.is_empty()
                    || tag.len() > 8
                    || !tag
                        .bytes()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
                {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut deps = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(dep) {
                    Some(&idx) => deps[i].push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        fn visit(i: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[i] == 1 {
                return true;
            }
            if marks[i] == 2 {
                return false;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if visit(d, deps, marks) {
                    return true;
                }
            }
            marks[i] = 2;
            false
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if visit(i, &deps, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let runtime = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| RuntimeJob {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                deps: deps[i].clone(),
                remaining: None,
                started_at: 0,
            })
            .collect();

        Ok(Self {
            jobs: runtime,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            started: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.jobs[i].state.clone())
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut jobs = self.jobs.clone();
        let mut lines = Vec::new();
        let mut now = if self.started { self.horizon } else { 0 };
        let mut first_phase = !self.started;

        loop {
            if first_phase {
                first_phase = false;
            } else if now == self.horizon && self.started {
                // The previous horizon was fully processed; do not replay it.
                let next = Self::next_event(&jobs, now);
                match next {
                    Some(t) if t <= until => now = t,
                    _ => break,
                }
            }

            let mut completed: Vec<usize> = jobs
                .iter()
                .enumerate()
                .filter_map(|(i, r)| match r.state {
                    State::Running { finish_at, .. } if finish_at == now => Some(i),
                    _ => None,
                })
                .collect();
            completed.sort_by(|&a, &b| jobs[a].job.id.cmp(&jobs[b].job.id));

            for i in completed {
                let attempt = match jobs[i].state {
                    State::Running { attempt, .. } => attempt,
                    _ => unreachable!(),
                };
                if attempt > jobs[i].job.failures {
                    jobs[i].state = State::Succeeded;
                    lines.push(format!("{} finish {} {} ok", now, jobs[i].job.id, attempt));
                } else {
                    lines.push(format!("{} finish {} {} fail", now, jobs[i].job.id, attempt));
                    if attempt == jobs[i].job.max_attempts {
                        jobs[i].state = State::Failed;
                    } else {
                        let shift = u32::from(attempt.saturating_sub(1));
                        let delay = self
                            .retry_base
                            .checked_shl(shift)
                            .unwrap_or(u64::MAX)
                            .min(self.retry_cap);
                        let ready = now.saturating_add(delay);
                        jobs[i].state = State::Waiting {
                            ready_at: ready,
                            attempt,
                        };
                        jobs[i].remaining = None;
                        lines.push(format!("{} retry {} {} {}", now, jobs[i].job.id, attempt, ready));
                    }
                }
            }

            loop {
                let mut newly_blocked = Vec::new();
                for i in 0..jobs.len() {
                    if matches!(jobs[i].state, State::Waiting { .. })
                        && jobs[i].deps.iter().any(|&d| {
                            matches!(jobs[d].state, State::Failed | State::Blocked)
                        })
                    {
                        newly_blocked.push(i);
                    }
                }
                if newly_blocked.is_empty() {
                    break;
                }
                for &i in &newly_blocked {
                    jobs[i].state = State::Blocked;
                }
                newly_blocked.sort_by(|&a, &b| jobs[a].job.id.cmp(&jobs[b].job.id));
                for i in newly_blocked {
                    let dep = jobs[i]
                        .deps
                        .iter()
                        .find(|&&d| matches!(jobs[d].state, State::Failed | State::Blocked))
                        .map(|&d| jobs[d].job.id.clone())
                        .unwrap_or_default();
                    lines.push(format!("{} skip {} {}", now, jobs[i].job.id, dep));
                }
            }

            loop {
                let mut candidates: Vec<usize> = jobs
                    .iter()
                    .enumerate()
                    .filter_map(|(i, r)| match r.state {
                        State::Waiting { ready_at, .. }
                            if ready_at <= now
                                && r.deps
                                    .iter()
                                    .all(|&d| matches!(jobs[d].state, State::Succeeded)) =>
                        {
                            Some(i)
                        }
                        _ => None,
                    })
                    .collect();
                candidates.sort_by(|&a, &b| {
                    let priority = |i: usize| {
                        if self.aging_ms == 0 {
                            jobs[i].job.priority
                        } else {
                            let ready_at = match jobs[i].state {
                                State::Waiting { ready_at, .. } => ready_at,
                                _ => now,
                            };
                            (u64::from(jobs[i].job.priority)
                                + (now.saturating_sub(ready_at) / self.aging_ms))
                                .min(9) as u8
                        }
                    };
                    priority(b)
                        .cmp(&priority(a))
                        .then_with(|| a.cmp(&b))
                });

                let mut selected = None;
                for candidate in candidates {
                    let running: Vec<usize> = jobs
                        .iter()
                        .enumerate()
                        .filter_map(|(i, r)| matches!(r.state, State::Running { .. }).then_some(i))
                        .collect();
                    let tag = jobs[candidate].job.tag.as_deref();
                    let tag_busy = tag.map_or(false, |t| {
                        running.iter().any(|&i| jobs[i].job.tag.as_deref() == Some(t))
                    });
                    if running.len() < self.capacity && !tag_busy {
                        selected = Some((candidate, None));
                        break;
                    }

                    let mut victims: Vec<usize> = running
                        .iter()
                        .copied()
                        .filter(|&v| {
                            jobs[candidate].job.priority >= jobs[v].job.priority.saturating_add(2)
                                && (tag.is_none()
                                    || jobs[v].job.tag.as_deref() == tag
                                    || !tag_busy)
                        })
                        .collect();
                    victims.sort_by(|&a, &b| {
                        jobs[a]
                            .job
                            .priority
                            .cmp(&jobs[b].job.priority)
                            .then_with(|| jobs[b].started_at.cmp(&jobs[a].started_at))
                            .then_with(|| jobs[b].job.id.cmp(&jobs[a].job.id))
                    });
                    for victim in victims {
                        let mut remaining_slots = running.len() - 1 < self.capacity;
                        let tag_freed = tag.map_or(true, |t| {
                            !running.iter().any(|&i| {
                                i != victim && jobs[i].job.tag.as_deref() == Some(t)
                            })
                        });
                        if remaining_slots && tag_freed {
                            selected = Some((candidate, Some(victim)));
                            break;
                        }
                    }
                    if selected.is_some() {
                        break;
                    }
                }

                let Some((candidate, victim)) = selected else {
                    break;
                };

                if let Some(v) = victim {
                    let (attempt, finish_at) = match jobs[v].state {
                        State::Running { attempt, finish_at } => (attempt, finish_at),
                        _ => unreachable!(),
                    };
                    let remaining = finish_at - now;
                    jobs[v].state = State::Waiting {
                        ready_at: now,
                        attempt,
                    };
                    jobs[v].remaining = Some(remaining);
                    lines.push(format!(
                        "{} preempt {} by {}",
                        now, jobs[v].job.id, jobs[candidate].job.id
                    ));
                }

                let (attempt, _) = match jobs[candidate].state {
                    State::Waiting { attempt, .. } => (attempt, ()),
                    _ => unreachable!(),
                };
                let (attempt, duration) = if let Some(remaining) = jobs[candidate].remaining.take() {
                    (attempt, remaining)
                } else {
                    let next_attempt = attempt.saturating_add(1);
                    (next_attempt, jobs[candidate].job.duration)
                };
                let finish_at = now.saturating_add(duration);
                jobs[candidate].state = State::Running { attempt, finish_at };
                jobs[candidate].started_at = now;
                lines.push(format!(
                    "{} start {} {} {}",
                    now, jobs[candidate].job.id, attempt, duration
                ));
            }

            if now == until {
                break;
            }
            match Self::next_event(&jobs, now) {
                Some(t) if t <= until => now = t,
                _ => break,
            }
        }

        self.jobs = jobs;
        self.horizon = until;
        self.started = true;
        Ok(lines)
    }

    fn next_event(jobs: &[RuntimeJob], now: u64) -> Option<u64> {
        jobs.iter()
            .filter_map(|r| match r.state {
                State::Running { finish_at, .. } if finish_at > now => Some(finish_at),
                State::Waiting { ready_at, .. } if ready_at > now => Some(ready_at),
                _ => None,
            })
            .min()
    }

    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct RuntimeJob {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    remaining: Option<u64>,
    last_ready: u64,
    segment_order: u64,
}

pub struct Pool {
    jobs: Vec<RuntimeJob>,
    by_id: std::collections::HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    first_run: bool,
    next_segment_order: u64,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job
                .tag
                .as_ref()
                .is_some_and(|tag| !valid_token(tag, 1, 8))
            {
                return Err(Error::BadTag(i));
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(dep) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(resolved);
        }

        if contains_cycle(&dependencies) {
            return Err(Error::Cycle);
        }

        let runtime = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| RuntimeJob {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                last_ready: job.release_at,
                job,
                dependencies: dependencies[i].clone(),
                remaining: None,
                segment_order: 0,
            })
            .collect();

        Ok(Self {
            jobs: runtime,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
            next_segment_order: 0,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if !self.first_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if !self.first_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut now = if self.first_run { 0 } else { self.horizon };
        let mut process_current = self.first_run;

        loop {
            if process_current {
                self.process_timestamp(now, &mut lines);
            }

            let Some(next) = self.next_event_time(now) else {
                break;
            };
            if next > until {
                break;
            }
            now = next;
            process_current = true;
        }

        self.horizon = until;
        self.first_run = false;
        Ok(lines)
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let index = *self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.jobs[index].state.clone())
    }

    fn process_timestamp(&mut self, now: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .jobs
            .iter()
            .enumerate()
            .filter_map(|(i, job)| match job.state {
                State::Running { finish_at, .. } if finish_at == now => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.jobs[a].job.id.cmp(&self.jobs[b].job.id));

        for index in completions {
            let (attempt, finish_at) = match self.jobs[index].state {
                State::Running { attempt, finish_at } => (attempt, finish_at),
                _ => continue,
            };
            let _ = finish_at;
            if attempt > self.jobs[index].job.failures {
                self.jobs[index].state = State::Succeeded;
                lines.push(format!(
                    "{} finish {} {} ok",
                    now, self.jobs[index].job.id, attempt
                ));
            } else {
                lines.push(format!(
                    "{} finish {} {} fail",
                    now, self.jobs[index].job.id, attempt
                ));
                if attempt >= self.jobs[index].job.max_attempts {
                    self.jobs[index].state = State::Failed;
                } else {
                    let delay = retry_delay(self.retry_base, self.retry_cap, attempt);
                    let ready_at = now.saturating_add(delay);
                    self.jobs[index].state = State::Waiting { ready_at, attempt };
                    self.jobs[index].last_ready = ready_at;
                    self.jobs[index].remaining = None;
                    lines.push(format!(
                        "{} retry {} {} {}",
                        now, self.jobs[index].job.id, attempt, ready_at
                    ));
                }
            }
        }

        let newly_blocked = self.propagate_failures();
        let mut newly_blocked = newly_blocked;
        newly_blocked.sort_by(|&a, &b| self.jobs[a].job.id.cmp(&self.jobs[b].job.id));
        for index in newly_blocked {
            let dependency = self.jobs[index]
                .dependencies
                .iter()
                .copied()
                .find(|&dep| {
                    matches!(self.jobs[dep].state, State::Failed | State::Blocked)
                })
                .map(|dep| self.jobs[dep].job.id.clone())
                .unwrap_or_default();
            lines.push(format!(
                "{} skip {} {}",
                now, self.jobs[index].job.id, dependency
            ));
        }

        self.schedule(now, lines);
    }

    fn propagate_failures(&mut self) -> Vec<usize> {
        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if !matches!(self.jobs[i].state, State::Waiting { .. }) {
                    continue;
                }
                if self.jobs[i]
                    .dependencies
                    .iter()
                    .any(|&dep| matches!(self.jobs[dep].state, State::Failed | State::Blocked))
                {
                    self.jobs[i].state = State::Blocked;
                    newly_blocked.push(i);
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked
    }

    fn schedule(&mut self, now: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self
                .jobs
                .iter()
                .enumerate()
                .filter_map(|(i, job)| match job.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= now
                            && job.dependencies.iter().all(|&dep| {
                                matches!(self.jobs[dep].state, State::Succeeded)
                            }) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, now);
                let pb = self.effective_priority(b, now);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut started = false;
            for candidate in candidates {
                let running: Vec<usize> = self
                    .jobs
                    .iter()
                    .enumerate()
                    .filter_map(|(i, job)| {
                        if matches!(job.state, State::Running { .. }) {
                            Some(i)
                        } else {
                            None
                        }
                    })
                    .collect();
                let candidate_tag = self.jobs[candidate].job.tag.as_deref();
                let slot_free = running.len() < self.capacity;
                let tag_free = candidate_tag.is_none()
                    || !running.iter().any(|&i| {
                        self.jobs[i].job.tag.as_deref() == candidate_tag
                    });

                if slot_free && tag_free {
                    self.start_job(candidate, now, lines, None);
                    started = true;
                    break;
                }

                let mut victims: Vec<usize> = running
                    .into_iter()
                    .filter(|&victim| {
                        self.jobs[candidate].job.priority
                            >= self.jobs[victim].job.priority.saturating_add(2)
                    })
                    .filter(|&victim| {
                        let remaining_running: Vec<usize> = self
                            .jobs
                            .iter()
                            .enumerate()
                            .filter_map(|(i, job)| {
                                if i != victim && matches!(job.state, State::Running { .. }) {
                                    Some(i)
                                } else {
                                    None
                                }
                            })
                            .collect();
                        remaining_running.len() < self.capacity
                            && (candidate_tag.is_none()
                                || !remaining_running.iter().any(|&i| {
                                    self.jobs[i].job.tag.as_deref() == candidate_tag
                                }))
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .job
                        .priority
                        .cmp(&self.jobs[b].job.priority)
                        .then_with(|| {
                            self.jobs[b]
                                .segment_order
                                .cmp(&self.jobs[a].segment_order)
                        })
                        .then_with(|| self.jobs[b].job.id.cmp(&self.jobs[a].job.id))
                });

                if let Some(victim) = victims.first().copied() {
                    let finish_at = match self.jobs[victim].state {
                        State::Running { finish_at, .. } => finish_at,
                        _ => continue,
                    };
                    let attempt = match self.jobs[victim].state {
                        State::Running { attempt, .. } => attempt,
                        _ => continue,
                    };
                    let remaining = finish_at.saturating_sub(now);
                    lines.push(format!(
                        "{} preempt {} by {}",
                        now, self.jobs[victim].job.id, self.jobs[candidate].job.id
                    ));
                    self.jobs[victim].state = State::Waiting {
                        ready_at: now,
                        attempt,
                    };
                    self.jobs[victim].remaining = Some(remaining);
                    self.jobs[victim].last_ready = now;
                    self.start_job(candidate, now, lines, None);
                    started = true;
                    break;
                }
            }

            if !started {
                break;
            }
        }
    }

    fn start_job(
        &mut self,
        index: usize,
        now: u64,
        lines: &mut Vec<String>,
        _unused: Option<()>,
    ) {
        let (attempt, remaining) = match self.jobs[index].state {
            State::Waiting { attempt, .. } => {
                if let Some(saved) = self.jobs[index].remaining.take() {
                    (attempt, saved)
                } else {
                    let next_attempt = attempt.saturating_add(1);
                    (next_attempt, self.jobs[index].job.duration)
                }
            }
            _ => return,
        };
        let finish_at = now.saturating_add(remaining);
        self.next_segment_order = self.next_segment_order.saturating_add(1);
        self.jobs[index].segment_order = self.next_segment_order;
        self.jobs[index].state = State::Running { attempt, finish_at };
        lines.push(format!(
            "{} start {} {} {}",
            now, self.jobs[index].job.id, attempt, remaining
        ));
    }

    fn effective_priority(&self, index: usize, now: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[index].job.priority;
        }
        let waited = now.saturating_sub(self.jobs[index].last_ready);
        let boost = waited / self.aging_ms;
        (self.jobs[index].job.priority as u64)
            .saturating_add(boost)
            .min(9) as u8
    }

    fn next_event_time(&self, now: u64) -> Option<u64> {
        self.jobs
            .iter()
            .filter_map(|job| match job.state {
                State::Running { finish_at, .. } if finish_at > now => Some(finish_at),
                State::Waiting { ready_at, .. } if ready_at > now => Some(ready_at),
                _ => None,
            })
            .min()
    }
}

fn valid_token(value: &str, min: usize, max: usize) -> bool {
    let bytes = value.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn contains_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
        if marks[node] == 1 {
            return true;
        }
        if marks[node] == 2 {
            return false;
        }
        marks[node] = 1;
        for &dep in &deps[node] {
            if visit(dep, deps, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    for node in 0..dependencies.len() {
        if visit(node, dependencies, &mut marks) {
            return true;
        }
    }
    false
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        if delay >= cap {
            return cap;
        }
        delay = delay.saturating_mul(2).min(cap);
    }
    delay.min(cap)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

struct Runtime {
    state: State,
    age_from: u64,
    remaining: Option<u64>,
    segment_started: u64,
}

pub struct Pool {
    jobs: Vec<Job>,
    states: Vec<Runtime>,
    dependencies: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    started: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_token(tag, 1, 8) {
                    return Err(Error::BadTag(i));
                }
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, name) in job.dependencies.iter().enumerate() {
                if !seen.insert(name) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&dep) = ids.get(name) else {
                    return Err(Error::MissingDependency(i, j));
                };
                dependencies[i].push(dep);
            }
        }

        let mut marks = vec![0u8; jobs.len()];
        fn visit(i: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[i] == 1 {
                return false;
            }
            if marks[i] == 2 {
                return true;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[i] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|j| Runtime {
                state: State::Waiting {
                    ready_at: j.release_at,
                    attempt: 0,
                },
                age_from: j.release_at,
                remaining: None,
                segment_started: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            dependencies,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            started: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.jobs
            .iter()
            .position(|j| j.id == id)
            .map(|i| self.states[i].state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.started {
            self.horizon
        } else {
            self.started = true;
            0
        };

        loop {
            let mut completed: Vec<usize> = self
                .states
                .iter()
                .enumerate()
                .filter_map(|(i, r)| match r.state {
                    State::Running { finish_at, .. } if finish_at == time => Some(i),
                    _ => None,
                })
                .collect();
            completed.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));

            for i in completed {
                let attempt = match self.states[i].state {
                    State::Running { attempt, .. } => attempt,
                    _ => unreachable!(),
                };
                if attempt > self.jobs[i].failures {
                    self.states[i].state = State::Succeeded;
                    lines.push(format!("{} finish {} {} ok", time, self.jobs[i].id, attempt));
                } else {
                    lines.push(format!("{} finish {} {} fail", time, self.jobs[i].id, attempt));
                    if attempt >= self.jobs[i].max_attempts {
                        self.states[i].state = State::Failed;
                    } else {
                        let shift = u32::from(attempt - 1);
                        let delay = self
                            .retry_base
                            .checked_shl(shift)
                            .unwrap_or(u64::MAX)
                            .min(self.retry_cap);
                        let ready = time.saturating_add(delay);
                        self.states[i].state = State::Waiting {
                            ready_at: ready,
                            attempt,
                        };
                        self.states[i].age_from = ready;
                        lines.push(format!(
                            "{} retry {} {} {}",
                            time, self.jobs[i].id, attempt, ready
                        ));
                    }
                }
            }

            loop {
                let mut changed = Vec::new();
                for i in 0..self.jobs.len() {
                    if matches!(self.states[i].state, State::Waiting { .. }) {
                        if let Some(&failed_dep) = self.dependencies[i].iter().find(|&&d| {
                            matches!(self.states[d].state, State::Failed | State::Blocked)
                        }) {
                            self.states[i].state = State::Blocked;
                            changed.push((i, failed_dep));
                        }
                    }
                }
                if changed.is_empty() {
                    break;
                }
                changed.sort_by(|(a, _), (b, _)| self.jobs[*a].id.cmp(&self.jobs[*b].id));
                for (i, dep) in changed {
                    lines.push(format!(
                        "{} skip {} {}",
                        time, self.jobs[i].id, self.jobs[dep].id
                    ));
                }
            }

            self.schedule(time, &mut lines);

            if time == until {
                break;
            }

            let next = self
                .states
                .iter()
                .filter_map(|r| match r.state {
                    State::Waiting { ready_at, .. } if ready_at > time => Some(ready_at),
                    State::Running { finish_at, .. } if finish_at > time => Some(finish_at),
                    _ => None,
                })
                .min();

            let Some(next_time) = next.filter(|&t| t <= until) else {
                break;
            };
            time = next_time;
        }

        self.horizon = until;
        Ok(lines)
    }

    fn schedule(&mut self, time: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = (0..self.jobs.len())
                .filter(|&i| {
                    matches!(self.states[i].state, State::Waiting { ready_at, .. } if ready_at <= time)
                        && self.dependencies[i]
                            .iter()
                            .all(|&d| matches!(self.states[d].state, State::Succeeded))
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut selected = None;
            for candidate in candidates {
                let tag = self.jobs[candidate].tag.as_deref();
                let running: Vec<usize> = (0..self.jobs.len())
                    .filter(|&i| matches!(self.states[i].state, State::Running { .. }))
                    .collect();
                let slot_free = running.len() < self.capacity;
                let tag_free = tag.is_none()
                    || !running
                        .iter()
                        .any(|&i| self.jobs[i].tag.as_deref() == tag);

                if slot_free && tag_free {
                    selected = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = running
                    .into_iter()
                    .filter(|&v| {
                        self.jobs[candidate].priority >= self.jobs[v].priority.saturating_add(2)
                            && (tag.is_none()
                                || self.jobs[v].tag.as_deref() == tag
                                || !self
                                    .states
                                    .iter()
                                    .enumerate()
                                    .any(|(i, r)| {
                                        i != v
                                            && matches!(r.state, State::Running { .. })
                                            && self.jobs[i].tag.as_deref() == tag
                                    }))
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .priority
                        .cmp(&self.jobs[b].priority)
                        .then_with(|| {
                            let ta = self.states[a].segment_started;
                            let tb = self.states[b].segment_started;
                            tb.cmp(&ta)
                        })
                        .then_with(|| self.jobs[b].id.cmp(&self.jobs[a].id))
                });

                if let Some(victim) = victims.first().copied() {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(v) = victim {
                let finish = match self.states[v].state {
                    State::Running { finish_at, .. } => finish_at,
                    _ => unreachable!(),
                };
                let attempt = match self.states[v].state {
                    State::Running { attempt, .. } => attempt,
                    _ => unreachable!(),
                };
                let remaining = finish.saturating_sub(time);
                lines.push(format!(
                    "{} preempt {} by {}",
                    time, self.jobs[v].id, self.jobs[candidate].id
                ));
                self.states[v].state = State::Waiting {
                    ready_at: time,
                    attempt,
                };
                self.states[v].age_from = time;
                self.states[v].remaining = Some(remaining);
            }

            let (attempt, remaining) = match self.states[candidate].state {
                State::Waiting { attempt, .. } => {
                    if let Some(rem) = self.states[candidate].remaining.take() {
                        (attempt, rem)
                    } else {
                        let next_attempt = attempt.saturating_add(1);
                        (next_attempt, self.jobs[candidate].duration)
                    }
                }
                _ => unreachable!(),
            };
            let finish_at = time.saturating_add(remaining);
            self.states[candidate].state = State::Running { attempt, finish_at };
            self.states[candidate].segment_started = time;
            lines.push(format!(
                "{} start {} {} {}",
                time, self.jobs[candidate].id, attempt, remaining
            ));
        }
    }

    fn effective_priority(&self, i: usize, time: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[i].priority;
        }
        let age = time.saturating_sub(self.states[i].age_from) / self.aging_ms;
        (u64::from(self.jobs[i].priority) + age).min(9) as u8
    }
}

fn valid_token(s: &str, min_len: usize, max_len: usize) -> bool {
    let bytes = s.as_bytes();
    (min_len..=max_len).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Runtime {
    state: State,
    remaining: u64,
    segment_started: u64,
    ready_at: u64,
}

pub struct Pool {
    jobs: Vec<Job>,
    dependencies: Vec<Vec<usize>>,
    states: Vec<Runtime>,
    by_id: HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = HashMap::new();

        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_token(tag, 1, 8) {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(dep) {
                    Some(&index) => dependencies[i].push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        let mut marks = vec![0u8; jobs.len()];
        fn visit(i: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[i] == 1 {
                return false;
            }
            if marks[i] == 2 {
                return true;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[i] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|job| Runtime {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                remaining: job.duration,
                segment_started: 0,
                ready_at: job.release_at,
            })
            .collect();

        Ok(Self {
            jobs,
            dependencies,
            states,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.states[i].state.clone())
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut t = if self.has_run {
            self.horizon
        } else {
            0
        };
        let mut first_phase = !self.has_run;

        loop {
            if first_phase || t > self.horizon || !self.has_run {
                self.process_timestamp(t, &mut lines);
                first_phase = false;
            }

            let next = self.next_event_after(t);
            match next {
                Some(n) if n <= until => t = n,
                _ => break,
            }
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn process_timestamp(&mut self, t: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .states
            .iter()
            .enumerate()
            .filter_map(|(i, r)| match r.state {
                State::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));

        for i in completions {
            let attempt = match self.states[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            if attempt > self.jobs[i].failures {
                self.states[i].state = State::Succeeded;
                lines.push(format!("{} finish {} {} ok", t, self.jobs[i].id, attempt));
            } else {
                lines.push(format!("{} finish {} {} fail", t, self.jobs[i].id, attempt));
                if attempt >= self.jobs[i].max_attempts {
                    self.states[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self
                        .retry_base
                        .checked_shl(shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready = t.saturating_add(delay);
                    self.states[i].state = State::Waiting {
                        ready_at: ready,
                        attempt,
                    };
                    self.states[i].ready_at = ready;
                    self.states[i].remaining = self.jobs[i].duration;
                    lines.push(format!("{} retry {} {} {}", t, self.jobs[i].id, attempt, ready));
                }
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if !matches!(self.states[i].state, State::Waiting { .. }) {
                    continue;
                }
                if self.dependencies[i].iter().any(|&d| {
                    matches!(self.states[d].state, State::Failed | State::Blocked)
                }) {
                    self.states[i].state = State::Blocked;
                    newly_blocked.push(i);
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));
        for &i in &newly_blocked {
            let dependency = self.dependencies[i]
                .iter()
                .filter(|&&d| matches!(self.states[d].state, State::Failed | State::Blocked))
                .map(|&d| self.jobs[d].id.as_str())
                .min()
                .unwrap_or("");
            lines.push(format!("{} skip {} {}", t, self.jobs[i].id, dependency));
        }

        self.schedule(t, lines);
    }

    fn schedule(&mut self, t: u64, lines: &mut Vec<String>) {
        loop {
            let running: Vec<usize> = self
                .states
                .iter()
                .enumerate()
                .filter_map(|(i, r)| matches!(r.state, State::Running { .. }).then_some(i))
                .collect();

            let mut candidates: Vec<usize> = self
                .states
                .iter()
                .enumerate()
                .filter_map(|(i, r)| match r.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= t
                            && self.dependencies[i].iter().all(|&d| {
                                matches!(self.states[d].state, State::Succeeded)
                            }) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut selected = None;
            for candidate in candidates {
                let tag = self.jobs[candidate].tag.as_deref();
                let slot_free = running.len() < self.capacity;
                let tag_free = tag.is_none()
                    || !running
                        .iter()
                        .any(|&r| self.jobs[r].tag.as_deref() == tag);

                if slot_free && tag_free {
                    selected = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&v| {
                        self.jobs[candidate].priority
                            >= self.jobs[v].priority.saturating_add(2)
                    })
                    .filter(|&v| {
                        let remaining_after = running.iter().filter(|&&x| x != v).count();
                        remaining_after < self.capacity
                    })
                    .filter(|&v| {
                        tag.is_none()
                            || !running.iter().any(|&r| {
                                r != v && self.jobs[r].tag.as_deref() == tag
                            })
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .priority
                        .cmp(&self.jobs[b].priority)
                        .then_with(|| {
                            let sa = self.states[a].segment_started;
                            let sb = self.states[b].segment_started;
                            sb.cmp(&sa)
                        })
                        .then_with(|| self.jobs[b].id.as_bytes().cmp(self.jobs[a].id.as_bytes()))
                });

                if let Some(victim) = victims.first().copied() {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(victim) = victim {
                let finish_at = match self.states[victim].state {
                    State::Running { finish_at, .. } => finish_at,
                    _ => t,
                };
                let attempt = match self.states[victim].state {
                    State::Running { attempt, .. } => attempt,
                    _ => 0,
                };
                let remaining = finish_at.saturating_sub(t);
                self.states[victim].state = State::Waiting {
                    ready_at: t,
                    attempt,
                };
                self.states[victim].ready_at = t;
                self.states[victim].remaining = remaining;
                lines.push(format!(
                    "{} preempt {} by {}",
                    t, self.jobs[victim].id, self.jobs[candidate].id
                ));
            }

            let (attempt, remaining) = match self.states[candidate].state {
                State::Waiting { attempt, .. } => {
                    if self.states[candidate].remaining == self.jobs[candidate].duration {
                        (attempt.saturating_add(1), self.jobs[candidate].duration)
                    } else {
                        (attempt, self.states[candidate].remaining)
                    }
                }
                _ => continue,
            };
            let finish_at = t.saturating_add(remaining);
            self.states[candidate].state = State::Running { attempt, finish_at };
            self.states[candidate].segment_started = t;
            lines.push(format!(
                "{} start {} {} {}",
                t, self.jobs[candidate].id, attempt, remaining
            ));
        }
    }

    fn effective_priority(&self, i: usize, t: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[i].priority;
        }
        let age_from = match self.states[i].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => self.states[i].ready_at,
        };
        let gain = t.saturating_sub(age_from) / self.aging_ms;
        (u64::from(self.jobs[i].priority) + gain).min(9) as u8
    }

    fn next_event_after(&self, t: u64) -> Option<u64> {
        let mut next: Option<u64> = None;
        for runtime in &self.states {
            let candidate = match runtime.state {
                State::Running { finish_at, .. } if finish_at > t => Some(finish_at),
                State::Waiting { ready_at, .. } if ready_at > t => Some(ready_at),
                _ => None,
            };
            if let Some(candidate) = candidate {
                next = Some(next.map_or(candidate, |n| n.min(candidate)));
            }
        }
        next
    }
}

fn valid_token(s: &str, min: usize, max: usize) -> bool {
    let bytes = s.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes.iter().all(|b| {
            b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-'
        })
}
