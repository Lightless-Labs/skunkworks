use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(b: u8) -> bool {
    b == b'.' || b == b'-' || b == b'+'
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    if start == bytes.len() || is_core_missing_byte(bytes[start]) {
        return Err(VersionError::MissingCore { at: start });
    }
    if !bytes[start].is_ascii_digit() {
        return Err(VersionError::InvalidChar { at: start });
    }

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in digits {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(u64::from(digit - b'0')))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let identifier = &s[id_start..i];
        if prerelease
            && identifier.bytes().all(|b| b.is_ascii_digit())
            && identifier.len() > 1
            && identifier.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        result.push(identifier.to_owned());
        if i == end {
            break;
        }
        id_start = i + 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;

    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || is_core_missing_byte(s.as_bytes()[pos]) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;

    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || is_core_missing_byte(s.as_bytes()[pos]) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;

    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pre_start + offset);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let left_numeric = left.bytes().all(|c| c.is_ascii_digit());
                let right_numeric = right.bytes().all(|c| c.is_ascii_digit());
                let ordering = match (left_numeric, right_numeric) {
                    (true, true) => compare_numeric_identifiers(left, right),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    (false, false) => left.as_bytes().cmp(right.as_bytes()),
                };
                if ordering != Ordering::Equal {
                    return ordering;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CompareOp,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum CompareOp {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = vec![Vec::new()];
    let mut pos = 0;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (op, op_len) = if token.starts_with(">=") {
            (CompareOp::GreaterEqual, 2)
        } else if token.starts_with("<=") {
            (CompareOp::LessEqual, 2)
        } else if token.starts_with('>') {
            (CompareOp::Greater, 1)
        } else if token.starts_with('<') {
            (CompareOp::Less, 1)
        } else if token.starts_with('=') {
            (CompareOp::Equal, 1)
        } else {
            (CompareOp::Equal, 0)
        };

        let operand_start = token_start + op_len;
        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[op_len..token.len()];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        arms[0].push(Comparator { op, version });
    }

    if arms[0].is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted_prerelease = v.pre.is_empty()
            || arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && (v.major, v.minor, v.patch)
                        == (
                            comparator.version.major,
                            comparator.version.minor,
                            comparator.version.patch,
                        )
            });

        admitted_prerelease
            && arm.iter().all(|comparator| {
                let ordering = compare_versions(v, &comparator.version);
                match comparator.op {
                    CompareOp::Equal => ordering == Ordering::Equal,
                    CompareOp::Less => ordering == Ordering::Less,
                    CompareOp::LessEqual => ordering != Ordering::Greater,
                    CompareOp::Greater => ordering == Ordering::Greater,
                    CompareOp::GreaterEqual => ordering != Ordering::Less,
                }
            })
    })
}
