use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        let b = bytes.get(*pos).copied();
        return Err(if is_core_terminator(b) {
            VersionError::MissingCore { at: *pos }
        } else {
            VersionError::InvalidChar { at: *pos }
        });
    }

    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &digit in digits {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            if !(bytes[i].is_ascii_alphanumeric() || bytes[i] == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let id = &s[id_start..i];
        if is_pre
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
            && id.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_owned());

        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        if s.as_bytes().get(pos) != Some(&b'.') {
            return Err(if is_core_terminator(s.as_bytes().get(pos).copied()) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        if _ == 0 {
            // The loop index is used only to distinguish the first core separator.
        }
    }

    // Reparse with explicit separators to preserve left-to-right error precedence.
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return Err(if is_core_terminator(s.as_bytes().get(pos).copied()) {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return Err(if is_core_terminator(s.as_bytes().get(pos).copied()) {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    let tail = s.as_bytes().get(pos).copied();
    if !matches!(tail, None | Some(b'-' | b'+')) {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if tail == Some(b'-') {
        let pre_start = pos + 1;
        let build_mark = s.as_bytes()[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|n| pre_start + n);
        let pre_end = build_mark.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        if let Some(mark) = build_mark {
            build = parse_identifiers(s, mark + 1, s.len(), false)?;
        }
    } else if tail == Some(b'+') {
        build = parse_identifiers(s, pos + 1, s.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(id: &str) -> bool {
    !id.is_empty() && id.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_strings(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = is_numeric_identifier(x);
        let yn = is_numeric_identifier(y);
        let cmp = match (xn, yn) {
            (true, true) => compare_numeric_strings(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if cmp != Ordering::Equal {
            return cmp;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut i = 0;
    let mut comparators = Vec::new();

    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let token_start = i;
        while i < bytes.len() && !matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        let token_end = i;
        let token = &s[token_start..token_end];

        let (op, operand_offset) = if token.starts_with(">=") {
            (Op::Ge, 2)
        } else if token.starts_with("<=") {
            (Op::Le, 2)
        } else if token.starts_with('>') {
            (Op::Gt, 1)
        } else if token.starts_with('<') {
            (Op::Lt, 1)
        } else if token.starts_with('=') {
            (Op::Eq, 1)
        } else {
            (Op::Eq, 0)
        };

        if operand_offset == token.len()
            || !token.as_bytes()[operand_offset].is_ascii_digit()
        {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand_start = token_start + operand_offset;
        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range {
        arms: vec![comparators],
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted = arm.iter().any(|c| {
            !c.version.pre.is_empty()
                && (c.version.major, c.version.minor, c.version.patch)
                    == (v.major, v.minor, v.patch)
        });
        if !v.pre.is_empty() && !admitted {
            return false;
        }

        arm.iter().all(|c| {
            let ordering = compare_versions(v, &c.version);
            match c.op {
                Op::Eq => ordering == Ordering::Equal,
                Op::Lt => ordering == Ordering::Less,
                Op::Le => ordering != Ordering::Greater,
                Op::Gt => ordering == Ordering::Greater,
                Op::Ge => ordering != Ordering::Less,
            }
        })
    })
}
