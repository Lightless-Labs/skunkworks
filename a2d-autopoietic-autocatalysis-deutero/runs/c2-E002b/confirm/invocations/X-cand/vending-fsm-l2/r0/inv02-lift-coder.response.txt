#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::<String>::new();

        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }

            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let Some(index) = DENOMINATIONS.iter().position(|&denom| denom == coin) else {
            return Err(Error::BadCoin { coin });
        };
        let _ = index;

        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(product) = self.products.iter().find(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };

        if self.selected.as_deref() == Some(product.id.as_str()) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(product.id.clone());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let Some(product) = self.products.iter().find(|product| &product.id == selected_id) else {
            return Err(Error::NoSelection);
        };

        let price = product.price;
        let id = product.id.clone();

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let mut change_counts = [0u16; 4];
        let mut remaining = amount;

        for index in (0..4).rev() {
            let coin = DENOMINATIONS[index];
            let count = self.tubes[index].min(remaining / coin);
            change_counts[index] = count;
            remaining -= count * coin;
        }

        if remaining != 0 {
            if let Some(solution) = find_best_change(amount, &self.tubes) {
                change_counts = solution;
            } else {
                return Err(Error::NoChange { amount });
            }
        }

        let mut proposed_tubes = self.tubes;
        for index in 0..4 {
            proposed_tubes[index] -= change_counts[index];
        }

        for &coin in &self.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|&denom| denom == coin)
                .expect("escrow contains only accepted coins");
            proposed_tubes[index] += 1;
        }

        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        let mut change = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..change_counts[index] {
                change.push(DENOMINATIONS[index]);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let mut coins = self.escrow.clone();
        coins.reverse();

        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(coins)
    }
}

fn find_best_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<[u16; 4]> = None;
    let mut best_total = u32::MAX;

    for count_10 in 0..=tubes[3].min(amount / 10) {
        for count_4 in 0..=tubes[2].min(amount / 4) {
            for count_3 in 0..=tubes[1].min(amount / 3) {
                for count_1 in 0..=tubes[0].min(amount) {
                    let value = count_10 * 10 + count_4 * 4 + count_3 * 3 + count_1;
                    if value != amount {
                        continue;
                    }

                    let candidate = [count_1, count_3, count_4, count_10];
                    let total = u32::from(count_1)
                        + u32::from(count_3)
                        + u32::from(count_4)
                        + u32::from(count_10);

                    let replace = match best {
                        None => true,
                        Some(current) => {
                            total < best_total
                                || (total == best_total
                                    && [count_10, count_4, count_3, count_1]
                                        > [current[3], current[2], current[1], current[0]])
                        }
                    };

                    if replace {
                        best = Some(candidate);
                        best_total = total;
                    }
                }
            }
        }
    }

    best
}
