#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(char, usize)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].0 == '~' {
            let valid_neighbor = |j: usize| {
                j < chars.len() && chars[j].0 != '~' && !is_separator(chars[j].0)
            };
            if i == 0 || i + 1 >= chars.len() || !valid_neighbor(i - 1) || !valid_neighbor(i + 1)
            {
                return Err(WrapError::InvalidMarker { index: chars[i].1 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let ch = chars[i].0;
        if ch == '\r' {
            if i + 1 < chars.len() && chars[i + 1].0 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(ch);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut pos = 0;
    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let run_start = pos;
            while pos < normalized.len() && normalized[pos] == '\n' {
                pos += 1;
            }
            let run_len = pos - run_start;
            if run_len >= 2 {
                layout_segment(&normalized[segment_start..run_start], options, &mut output);
                for _ in 1..run_len {
                    output.push(String::new());
                }
                segment_start = pos;
            }
        } else {
            pos += 1;
        }
    }
    layout_segment(&normalized[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

#[derive(Clone)]
struct Piece {
    text: String,
    gaps_before: usize,
}

fn layout_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut current = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !current.is_empty() {
                words.push(std::mem::take(&mut current));
            }
        } else {
            current.push(ch);
        }
    }
    if !current.is_empty() {
        words.push(current);
    }
    if words.is_empty() {
        return;
    }

    let mut lines: Vec<Vec<Piece>> = Vec::new();
    let mut line: Vec<Piece> = Vec::new();
    let mut content_len = 0usize;
    let mut line_index = 0usize;

    for word in words {
        let mut offset = 0usize;
        while offset < word.len() {
            let indent = if line_index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            let gap = if line.is_empty() { 0 } else { 1 };
            let room = capacity.saturating_sub(content_len + gap);
            let suffix_len = word.len() - offset;

            if suffix_len <= room {
                if gap > 0 {
                    content_len += 1;
                }
                let text: String = word[offset..].iter().filter(|&&c| c != '~').collect();
                content_len += text.chars().count();
                line.push(Piece {
                    text,
                    gaps_before: gap,
                });
                offset = word.len();
                continue;
            }

            let mut best: Option<(usize, usize)> = None;
            for marker in offset + 1..word.len() {
                if word[marker] == '~' {
                    let prefix_len = marker - offset;
                    if prefix_len + 1 <= room {
                        best = Some((marker, prefix_len));
                    }
                }
            }

            if let Some((marker, prefix_len)) = best {
                let unused = room - (prefix_len + 1);
                let accept = line.is_empty()
                    || (unused * unused + options.hyphen_cost
                        < (capacity.saturating_sub(content_len)).pow(2));
                if accept {
                    if gap > 0 {
                        content_len += 1;
                    }
                    let mut text: String = word[offset..marker]
                        .iter()
                        .filter(|&&c| c != '~')
                        .collect();
                    text.push('-');
                    content_len += prefix_len + 1;
                    line.push(Piece {
                        text,
                        gaps_before: gap,
                    });
                    lines.push(std::mem::take(&mut line));
                    line_index += 1;
                    content_len = 0;
                    offset = marker + 1;
                    continue;
                }
            }

            if !line.is_empty() {
                lines.push(std::mem::take(&mut line));
                line_index += 1;
                content_len = 0;
                continue;
            }

            if suffix_len > capacity {
                let mut consumed = 0usize;
                let mut raw = String::new();
                let mut next_offset = offset;
                while next_offset < word.len() && consumed < capacity {
                    if word[next_offset] != '~' {
                        raw.push(word[next_offset]);
                        consumed += 1;
                    }
                    next_offset += 1;
                }
                line.push(Piece {
                    text: raw,
                    gaps_before: 0,
                });
                content_len = consumed;
                lines.push(std::mem::take(&mut line));
                line_index += 1;
                content_len = 0;
                offset = next_offset;
            } else {
                let text: String = word[offset..].iter().filter(|&&c| c != '~').collect();
                content_len = text.chars().count();
                line.push(Piece {
                    text,
                    gaps_before: 0,
                });
                offset = word.len();
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    let final_line = lines.len().saturating_sub(1);
    for (idx, pieces) in lines.into_iter().enumerate() {
        let indent = if idx == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let gaps = pieces.iter().filter(|p| p.gaps_before > 0).count();
        let text_width: usize = pieces.iter().map(|p| p.text.chars().count()).sum();
        let pre_width = indent + text_width + gaps;
        let justify = options.justify && idx != final_line && pieces.len() >= 2 && gaps > 0;
        let extra = if justify {
            options.width - pre_width
        } else {
            0
        };
        let each = if gaps > 0 { extra / gaps } else { 0 };
        let rem = if gaps > 0 { extra % gaps } else { 0 };

        let mut rendered = " ".repeat(indent);
        let mut gap_index = 0usize;
        for piece in pieces {
            if piece.gaps_before > 0 {
                let add = if justify {
                    each + usize::from((gap_index + idx % gaps) % gaps < rem)
                } else {
                    1
                };
                rendered.push_str(&" ".repeat(add));
                gap_index += 1;
            }
            rendered.push_str(&piece.text);
        }
        output.push(rendered);
    }
}
