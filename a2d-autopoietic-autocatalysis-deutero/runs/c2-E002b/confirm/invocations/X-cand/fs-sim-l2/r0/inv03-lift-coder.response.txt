```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
            cwd: Vec::new(),
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.bytes().position(|b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_owned()),
            }
        }
        Ok(components)
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn lookup<'a>(&'a self, components: &[String]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut prefix = Vec::new();

        for component in components {
            let dir = match current {
                Entry::Dir(children) => children,
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(&prefix))),
            };

            prefix.push(component.clone());
            current = dir
                .get(component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
        }

        Ok(current)
    }

    fn lookup_parent<'a>(
        &'a self,
        components: &[String],
    ) -> Result<(&'a BTreeMap<String, Entry>, String), Error> {
        let parent_components = &components[..components.len() - 1];
        let parent = self.lookup(parent_components)?;
        let children = match parent {
            Entry::Dir(children) => children,
            Entry::File(_) => return Err(Error::NotDir(Self::canonical(parent_components))),
        };
        Ok((children, components.last().unwrap().clone()))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let (children, name) = self.lookup_parent(&components)?;
        if children.contains_key(&name) {
            return Err(Error::Exists(Self::canonical(&components)));
        }

        let parent = self.lookup_mut(&components[..components.len() - 1])?;
        if let Entry::Dir(children) = parent {
            children.insert(name, Entry::Dir(BTreeMap::new()));
        }
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let (exists, is_dir) = {
            let (children, name) = self.lookup_parent(&components)?;
            match children.get(&name) {
                Some(Entry::Dir(_)) => (true, true),
                Some(Entry::File(_)) => (true, false),
                None => (false, false),
            }
        };

        if is_dir {
            return Err(Error::IsDir(Self::canonical(&components)));
        }

        let parent = self.lookup_mut(&components[..components.len() - 1])?;
        if let Entry::Dir(children) = parent {
            let name = components.last().unwrap().clone();
            if exists {
                if let Some(Entry::File(contents)) = children.get_mut(&name) {
                    contents.clear();
                    contents.extend_from_slice(data);
                }
            } else {
                children.insert(name, Entry::File(data.to_vec()));
            }
        }
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        match self.lookup(&components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical(&components))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        match self.lookup(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
        }
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        match self.lookup(&components)? {
            Entry::Dir(_) => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
        }
    }

    fn lookup_mut(&mut self, components: &[String]) -> Result<&mut Entry, Error> {
        let mut current = &mut self.root;
        let mut prefix = Vec::new();

        for component in components {
            let dir = match current {
                Entry::Dir(children) => children,
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(&prefix))),
            };

            prefix.push(component.clone());
            current = dir
                .get_mut(component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
        }

        Ok(current)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
