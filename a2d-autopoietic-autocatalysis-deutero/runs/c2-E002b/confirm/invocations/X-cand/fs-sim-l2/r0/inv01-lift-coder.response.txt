```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: std::collections::HashMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::HashMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_string()),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|component| !component.is_empty())
            .map(str::to_string)
            .collect()
    }

    fn parent_and_name(path: &str) -> Option<(String, String)> {
        let mut components = Self::components(path);
        let name = components.pop()?;
        let parent = if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        };
        Some((parent, name))
    }

    fn check_traversal(&self, path: &str) -> Result<(), Error> {
        let components = Self::components(path);
        let mut prefix = String::new();

        for (index, component) in components.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(component);

            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::File(_)) if index + 1 < components.len() => {
                    return Err(Error::NotDir(prefix));
                }
                Some(_) => {}
            }
        }

        Ok(())
    }

    fn check_parent(&self, parent: &str) -> Result<(), Error> {
        self.check_traversal(parent)?;
        match self.entries.get(parent) {
            Some(Entry::Dir) => Ok(()),
            Some(Entry::File(_)) => Err(Error::NotDir(parent.to_string())),
            None => Err(Error::Missing(parent.to_string())),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let (parent, _) = match Self::parent_and_name(&canonical) {
            Some(parts) => parts,
            None => return Err(Error::Exists("/".to_string())),
        };

        self.check_parent(&parent)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;

        if canonical == "/" {
            return Err(Error::IsDir(canonical));
        }

        let (parent, _) = Self::parent_and_name(&canonical).unwrap();
        self.check_parent(&parent)?;

        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        self.check_traversal(&canonical)?;

        match self.entries.get(&canonical) {
            Some(Entry::File(contents)) => Ok(contents.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            None => Err(Error::Missing(canonical)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        self.check_traversal(&canonical)?;

        match self.entries.get(&canonical) {
            Some(Entry::File(_)) => return Err(Error::NotDir(canonical)),
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(canonical)),
        }

        let prefix = if canonical == "/" {
            "/".to_string()
        } else {
            format!("{}/", canonical)
        };

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if let Some(remainder) = entry_path.strip_prefix(&prefix) {
                if !remainder.is_empty() && !remainder.contains('/') {
                    let kind = match entry {
                        Entry::Dir => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    children.push((remainder.to_string(), kind));
                }
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        self.check_traversal(&canonical)?;

        match self.entries.get(&canonical) {
            Some(Entry::Dir) => {
                self.cwd = canonical;
                Ok(())
            }
            Some(Entry::File(_)) => Err(Error::NotDir(canonical)),
            None => Err(Error::Missing(canonical)),
        }
    }
}
```
