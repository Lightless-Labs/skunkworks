fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_changes(base: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = base.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut changes = Vec::new();
    let (mut i, mut j) = (0usize, 0usize);
    let mut active_start = 0usize;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active_start: &mut usize,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if *deletes > 0 || !inserts.is_empty() {
            changes.push(Change {
                start: *active_start,
                end: *active_start + *deletes,
                replacement: std::mem::take(inserts),
                side,
            });
            *active_start += *deletes;
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            flush(
                &mut changes,
                &mut active_start,
                &mut deletes,
                &mut inserts,
            );
            i += 1;
            j += 1;
            active_start += 1;
        } else if i == n {
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            deletes += 1;
            i += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            };

            if delete {
                deletes += 1;
                i += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(
        &mut changes,
        &mut active_start,
        &mut deletes,
        &mut inserts,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;

    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[Change],
    component_changes: &[usize],
    side: usize,
) -> Vec<String> {
    let mut relevant: Vec<&Change> = component_changes
        .iter()
        .map(|&index| &changes[index])
        .filter(|change| change.side == side)
        .collect();
    relevant.sort_by_key(|change| (change.start, change.end));

    let mut output = Vec::new();
    let mut cursor = start;

    for change in relevant {
        while cursor < change.start {
            output.push(base[cursor].clone());
            cursor += 1;
        }
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    while cursor < end {
        output.push(base[cursor].clone());
        cursor += 1;
    }
    output
}

fn push_line(output: &mut Vec<String>, line: impl Into<String>) {
    output.push(line.into());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let inputs = [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ];

    for (text, side) in &inputs {
        if let Some(byte) = text.as_bytes().iter().position(|&byte| byte == b'\r') {
            return Err(MergeError::CarriageReturn {
                side: side.clone(),
                byte,
            });
        }
    }

    for (text, side) in &inputs {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side: side.clone() });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = edit_changes(&base_lines, &left_lines, 0);
    changes.extend(edit_changes(&base_lines, &right_lines, 1));

    let mut visited = vec![false; changes.len()];
    let mut components = Vec::new();

    for seed in 0..changes.len() {
        if visited[seed] {
            continue;
        }
        visited[seed] = true;
        let mut stack = vec![seed];
        let mut members = Vec::new();

        while let Some(current) = stack.pop() {
            members.push(current);
            for candidate in 0..changes.len() {
                if !visited[candidate] && interacts(&changes[current], &changes[candidate]) {
                    visited[candidate] = true;
                    stack.push(candidate);
                }
            }
        }

        let start = members
            .iter()
            .map(|&index| changes[index].start)
            .min()
            .unwrap();
        let end = members
            .iter()
            .map(|&index| changes[index].end)
            .max()
            .unwrap();
        components.push(Component {
            start,
            end,
            changes: members,
        });
    }

    components.sort_by_key(|component| {
        (
            component.start,
            usize::from(component.end > component.start),
            component.end,
        )
    });

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        while base_cursor < component.start {
            push_line(&mut output, base_lines[base_cursor].clone());
            base_cursor += 1;
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &changes,
            &component.changes,
            0,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &changes,
            &component.changes,
            1,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    push_line(&mut output, format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());
                    let base_position = if component.end > component.start {
                        component.start + 1
                    } else {
                        component.start
                    };
                    push_line(
                        &mut output,
                        format!(
                            "||||||| BASE {base_position},{}",
                            component.end - component.start
                        ),
                    );
                    output.extend(base_projection.iter().cloned());
                    push_line(&mut output, "=======");
                    output.extend(right_projection.iter().cloned());
                    push_line(&mut output, format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: output.len() - output_start,
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    while base_cursor < base_lines.len() {
        push_line(&mut output, base_lines[base_cursor].clone());
        base_cursor += 1;
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn build_components(changes: &[Change]) -> Vec<Component> {
    let mut visited = vec![false; changes.len()];
    let mut components = Vec::new();

    for root in 0..changes.len() {
        if visited[root] {
            continue;
        }

        visited[root] = true;
        let mut stack = vec![root];
        let mut members = Vec::new();

        while let Some(current) = stack.pop() {
            members.push(current);
            for other in 0..changes.len() {
                if !visited[other] && interacts(&changes[current], &changes[other]) {
                    visited[other] = true;
                    stack.push(other);
                }
            }
        }

        let start = members
            .iter()
            .map(|&idx| changes[idx].start)
            .min()
            .unwrap();
        let end = members
            .iter()
            .map(|&idx| changes[idx].end)
            .max()
            .unwrap();

        components.push(Component {
            start,
            end,
            changes: members,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then_with(|| a.end.cmp(&b.end))
    });
    components
}

    let mut cursor = start;

    for change in selected {
        if change.start > cursor {
            output.extend(base[cursor..change.start].iter().cloned());
        }
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < end {
        output.extend(base[cursor..end].iter().cloned());
    }
    output
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    validate_final_lf(base, MergeSide::Base)?;
    validate_final_lf(left, MergeSide::Left)?;
    validate_final_lf(right, MergeSide::Right)?;

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = canonical_changes(&base_lines, &left_lines, 0);
    changes.extend(canonical_changes(&base_lines, &right_lines, 1));

    let components = build_components(&changes);
    let mut output_lines = Vec::new();
    let mut regions = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        if base_cursor < component.start {
            output_lines.extend(base_lines[base_cursor..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &changes,
            &component.changes,
            0,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &changes,
            &component.changes,
            1,
        );

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = regions.len() + 1;
                    let output_start = output_lines.len();
                    let base_len = component.end - component.start;
                    let base_label_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output_lines.push(format!("<<<<<<< LEFT {ordinal}"));
                    output_lines.extend(left_projection.iter().cloned());
                    output_lines.push(format!(
                        "||||||| BASE {base_label_start},{base_len}"
                    ));
                    output_lines.extend(base_projection.iter().cloned());
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection.iter().cloned());
                    output_lines.push(format!(">>>>>>> RIGHT {ordinal}"));

                    regions.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        output_lines.extend(base_lines[base_cursor..].iter().cloned());
    }

    let text = if output_lines.is_empty() {
        String::new()
    } else {
        let mut text = output_lines.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}

fn lines(input: &str) -> Vec<String> {

fn canonical_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut edits = Vec::new();
    while i < n && j < m {
        if old[i] == new[j] {
            edits.push(Edit::Keep);
            i += 1;
            j += 1;
        } else {
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            if delete_len > insert_len || (delete_len == insert_len && (i + j) % 2 == 0) {
                edits.push(Edit::Delete);
                i += 1;
            } else {
                edits.push(Edit::Insert(new[j].clone()));
                j += 1;
            }
        }
    }
    while i < n {
        edits.push(Edit::Delete);
        i += 1;
    }
    while j < m {
        edits.push(Edit::Insert(new[j].clone()));
        j += 1;
    }

    let mut changes = Vec::new();
    let mut base_pos = 0;
    let mut k = 0;
    while k < edits.len() {
        if matches!(edits[k], Edit::Keep) {
            base_pos += 1;
            k += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0;
        let mut replacement = Vec::new();
        while k < edits.len() && !matches!(edits[k], Edit::Keep) {
            match &edits[k] {
                Edit::Delete => {
                    deleted += 1;
                    base_pos += 1;
                }
                Edit::Insert(line) => replacement.push(line.clone()),
                Edit::Keep => unreachable!(),
            }
            k += 1;
        }
        changes.push(Change {
            start,
            end: start + deleted,
            replacement,
        });
    }
    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_positive = a.start < a.end;
    let b_positive = b.start < b.end;

    match (a_positive, b_positive) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn projections(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[Change],

        if change.start > cursor {

        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let inputs = [
        (MergeSide::Base, base),
        (MergeSide::Left, left),
        (MergeSide::Right, right),
    ];

    for (side, input) in &inputs {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn {
                side: side.clone(),
                byte,
            });
        }
    }

    for (side, input) in &inputs {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side: side.clone() });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = canonical_changes(&base_lines, &left_lines);
    let right_changes = canonical_changes(&base_lines, &right_lines);

    let mut nodes: Vec<(bool, usize)> = Vec::new();
    for i in 0..left_changes.len() {
        nodes.push((true, i));
    }
    for i in 0..right_changes.len() {
        nodes.push((false, i));
    }

    let mut parent: Vec<usize> = (0..nodes.len()).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        x
    }

    for a in 0..nodes.len() {
        for b in a + 1..nodes.len() {
            let ca = if nodes[a].0 {
                &left_changes[nodes[a].1]
            } else {
                &right_changes[nodes[a].1]
            };
            let cb = if nodes[b].0 {
                &left_changes[nodes[b].1]
            } else {
                &right_changes[nodes[b].1]
            };
            if interact(ca, cb) {
                let ra = root(&mut parent, a);
                let rb = root(&mut parent, b);
                if ra != rb {
                    parent[rb] = ra;
                }
            }
        }
    }

    let mut groups: Vec<(usize, Vec<Change>, Vec<Change>)> = Vec::new();
    for (node, &(is_left, index)) in nodes.iter().enumerate() {
        let r = root(&mut parent, node);
        let change = if is_left {
            left_changes[index].clone()
        } else {
            right_changes[index].clone()
        };
        if let Some(group) = groups.iter_mut().find(|g| g.0 == r) {
            if is_left {
                group.1.push(change);
            } else {
                group.2.push(change);
            }
        } else {
            let mut l = Vec::new();
            let mut rr = Vec::new();
            if is_left {
                l.push(change);
            } else {
                rr.push(change);
            }
            groups.push((r, l, rr));
        }
    }

    let mut components = Vec::new();
    for (_, mut lchanges, mut rchanges) in groups {
        lchanges.sort_by_key(|c| (c.start, c.end));
        rchanges.sort_by_key(|c| (c.start, c.end));
        let all = lchanges.iter().chain(rchanges.iter());
        let start = all.clone().map(|c| c.start).min().unwrap();
        let end = all.map(|c| c.end).max().unwrap();
        components.push(Component {
            start,
            end,
            left: lchanges,
            right: rchanges,
        });
    }
    components.sort_by_key(|c| (c.start, usize::from(c.start < c.end)));

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0;

    for component in components {
        if base_cursor < component.start {
            append_lines(&mut output, &base_lines[base_cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projections(
            &base_lines,
            component.start,
            component.end,
            &component.left,
        );
        let right_projection = projections(
            &base_lines,
            component.start,
            component.end,
            &component.right,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_start = component.start;
                    let base_len = component.end - component.start;
                    let displayed_start = if base_len > 0 {
                        base_start + 1
                    } else {
                        base_start
                    };

                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    output.push(format!("||||||| BASE {displayed_start},{base_len}"));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start,
                        base_len,
                        output_start,
                        output_len: output.len() - output_start,
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[base_cursor..]);
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn changes(base: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = base.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut edits: Vec<(bool, String)> = Vec::new();
    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            edits.push((true, base[i].clone()));
            i += 1;
            j += 1;
        } else if i == n {
            edits.push((false, new[j].clone()));
            j += 1;
        } else if j == m {
            edits.push((false, String::new()));
            i += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            };
            if delete {
                edits.push((false, String::new()));
                i += 1;
            } else {
                edits.push((false, new[j].clone()));
                j += 1;
            }
        }
    }

    let mut result = Vec::new();
    let mut base_pos = 0;
    let mut run_start = None;
    let mut deleted = 0usize;
    let mut replacement = Vec::new();

    let flush = |result: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deleted: &mut usize,
                 replacement: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            result.push(Change {
                start,
                end: start + *deleted,
                replacement: std::mem::take(replacement),
                side,
            });
            *deleted = 0;
        }
    };

    for (keep, value) in edits {
        if keep {
            flush(&mut result, &mut run_start, &mut deleted, &mut replacement);
            base_pos += 1;
        } else {
            if run_start.is_none() {
                run_start = Some(base_pos);
            }
            if value.is_empty() {
                // Empty inserted lines are valid. A deletion is identified by
                // whether the old cursor consumed a base line, handled below.
            }
            // Reconstruct operation type from the base traversal: delete
            // operations consumed a base line and have an empty sentinel.
            // An inserted empty string is distinguished by the new cursor path
            // through an explicit operation representation in the generation
            // above; this loop is replaced by the operation-aware pass below.
            replacement.push(value);
        }
    }
    flush(&mut result, &mut run_start, &mut deleted, &mut replacement);

    // The operation stream is regenerated with explicit Keep/Delete/Insert
    // tags so empty lines remain unambiguous.
    let mut ops: Vec<(u8, String)> = Vec::new();
    let (mut i, mut j) = (0, 0);
    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            ops.push((0, base[i].clone()));
            i += 1;
            j += 1;
        } else if i == n {
            ops.push((2, new[j].clone()));
            j += 1;
        } else if j == m {
            ops.push((1, String::new()));
            i += 1;
        } else {
            let d = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            let delete = if d > ins {
                true
            } else if ins > d {
                false
            } else {
                (i + j) % 2 == 0
            };
            if delete {
                ops.push((1, String::new()));
                i += 1;
            } else {
                ops.push((2, new[j].clone()));
                j += 1;
            }
        }
    }

    let mut result = Vec::new();
    let mut base_pos = 0;
    let mut run_start: Option<usize> = None;
    let mut deleted = 0usize;
    let mut replacement = Vec::new();
    for (kind, value) in ops {
        match kind {
            0 => {
                flush(&mut result, &mut run_start, &mut deleted, &mut replacement);
                base_pos += 1;
            }
            1 => {
                if run_start.is_none() {
                    run_start = Some(base_pos);
                }
                deleted += 1;
                base_pos += 1;
            }
            _ => {
                if run_start.is_none() {
                    run_start = Some(base_pos);
                }
                replacement.push(value);
            }
        }
    }
    flush(&mut result, &mut run_start, &mut deleted, &mut replacement);
    result
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn project(
    base: &[String],
    component: &Component,
    all: &[Change],
    side: usize,
) -> Vec<String> {
    let mut selected: Vec<&Change> = component
        .changes
        .iter()
        .map(|&idx| &all[idx])
        .filter(|change| change.side == side)
        .collect();
    selected.sort_by_key(|change| (change.start, change.end));

    let mut output = Vec::new();
    let mut cursor = component.start;
    for change in selected {
        output.extend(base[cursor..change.start].iter().cloned());
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    output.extend(base[cursor..component.end].iter().cloned());
    output
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);
    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = changes(&base_lines, &left_lines, 0);
    let right_changes = changes(&base_lines, &right_lines, 1);
    let mut all_changes = left_changes;
    all_changes.extend(right_changes);

    let count = all_changes.len();
    let mut adjacency = vec![Vec::new(); count];
    for i in 0..count {
        for j in i + 1..count {
            if interact(&all_changes[i], &all_changes[j]) {
                adjacency[i].push(j);
                adjacency[j].push(i);
            }
        }
    }

    let mut seen = vec![false; count];
    let mut components = Vec::new();
    for i in 0..count {
        if seen[i] {
            continue;
        }
        let mut stack = vec![i];
        seen[i] = true;
        let mut indexes = Vec::new();
        while let Some(node) = stack.pop() {
            indexes.push(node);
            for &next in &adjacency[node] {
                if !seen[next] {
                    seen[next] = true;
                    stack.push(next);
                }
            }
        }
        let start = indexes
            .iter()
            .map(|&idx| all_changes[idx].start)
            .min()
            .unwrap();
        let end = indexes
            .iter()
            .map(|&idx| all_changes[idx].end)
            .max()
            .unwrap();
        components.push(Component {
            start,
            end,
            changes: indexes,
        });
    }
    components.sort_by_key(|component| (component.start, component.end != component.start, component.end));

    let mut output = Vec::new();
    let mut output_lines = 0usize;
    let mut base_cursor = 0usize;
    let mut conflicts = Vec::new();

    for component in components {
        output.extend(base_lines[base_cursor..component.start].iter().cloned());
        output_lines += component.start - base_cursor;

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(&base_lines, &component, &all_changes, 0);
        let right_projection = project(&base_lines, &component, &all_changes, 1);

        if left_projection == right_projection {
            output.extend(left_projection.iter().cloned());
            output_lines += left_projection.len();
        } else if left_projection == base_projection {
            output.extend(right_projection.iter().cloned());
            output_lines += right_projection.len();
        } else if right_projection == base_projection {
            output.extend(left_projection.iter().cloned());
            output_lines += left_projection.len();
        } else {
            match resolution {
                Resolution::PreferLeft => {
                    output.extend(left_projection.iter().cloned());
                    output_lines += left_projection.len();
                }
                Resolution::PreferRight => {
                    output.extend(right_projection.iter().cloned());
                    output_lines += right_projection.len();
                }
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines;
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());
                    output.push(format!(
                        "||||||| BASE {},{}",
                        if component.end > component.start {
                            component.start + 1
                        } else {
                            component.start
                        },
                        component.end - component.start
                    ));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_string());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len =
                        4 + left_projection.len() + base_projection.len() + right_projection.len();
                    output_lines += output_len;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len,
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    output.extend(base_lines[base_cursor..].iter().cloned());
    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        conflicts: conflicts.len(),
        regions: conflicts,
        text,
    })
}
