}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut quoted = false;
    let mut after_quote = false;
    let mut opening_quote = 0usize;
    let mut field_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        if quoted {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_index += 1;
                after_quote = false;
                field_started = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record: record_index, byte: i });
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            field_started = false;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
            quoted = true;
            opening_quote = i;
            i += 1;
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            field_started = true;
            i += ch.len_utf8();
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: opening_quote });
    }
    if !row.is_empty() || !field.is_empty() || field_started || after_quote {
        row.push(field);
        records.push(row);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;
    let mut records = parse_dialect(input, config.delimiter, config.quote)?;
    let mut header = None;
    if config.headers && !records.is_empty() {
        header = Some(records.remove(0));
    }

    if let Some(ref names) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in names.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }
        for (data_index, row) in records.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: data_index + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows: records, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = build_table(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let body = if b.first() == Some(&b'-') { &b[1..] } else { b };
    let Some(dot) = body.iter().position(|&c| c == b'.') else { return false };
    if body[dot + 1..].len() != 2 || !body[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let whole = &body[..dot];
    if whole.is_empty() || !whole.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if whole.len() > 1 && whole[0] == b'0' {
        return false;
    }
    if b.first() == Some(&b'-') && whole == b"0" && body[dot + 1..] == *b"00" {
        return false;
    }
    true
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let (whole, frac) = body.split_once('.').unwrap();
    let value = whole.parse::<i128>().unwrap() * 100
        + frac.parse::<i128>().unwrap();
    let signed = if negative { -value } else { value };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate(cell: Option<&String>, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    let Some(cell) = cell else { return Err(CellErrorKind::MissingCell) };
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut out = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut values = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let Some(column_index) = resolved else {
                out.errors.push(CellError {
                    row: row_index, spec: spec_index, column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                if policy == ErrorPolicy::Stop {
                    out.rows.push(values);
                    break 'rows;
                }
                if policy == ErrorPolicy::FirstPerRow {
                    break;
                }
                continue;
            };
            match evaluate(row.get(column_index), spec) {
                Ok(value) => values[spec_index] = Some(value),
                Err(kind) => {
                    out.errors.push(CellError {
                        row: row_index, spec: spec_index, column: Some(column_index), kind,
                    });
                    if policy == ErrorPolicy::Stop {
                        out.rows.push(values);
                        break 'rows;
                    }
                    if policy == ErrorPolicy::FirstPerRow {
                        break;
                    }
                }
            }
        }
        out.rows.push(values);
    }
    out
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut all_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        all_records.push(header);
    }
    all_records.extend(table.rows.iter());

    for (record_index, record) in all_records.iter().enumerate() {
        for (column, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record: record_index, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (row_index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in all_records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns { record: record_index });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                let mut seen: Vec<(&str, usize)> = Vec::new();
                for (second, name) in header.iter().enumerate() {
                    if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
                        return Err(WriteError::DuplicateHeader {
                            name: name.clone(),
                            first: *first,
                            second,
                        });
                    }
                    seen.push((name, second));
                }
            }
        }
    }

    let mut header_out: Option<Vec<String>> = None;
    let mut rows_out: Vec<Vec<String>> = Vec::new();

    if config.mode == WriteMode::Preserve {
        header_out = table.header.clone();
        rows_out = table.rows.clone();
    } else if let Some(header) = &table.header {
        let mut selected: Vec<(String, usize)> = Vec::new();
        for (i, name) in header.iter().enumerate() {
            match table.duplicates {
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                    if !selected.iter().any(|(n, _)| n == name) {
                        selected.push((name.clone(), i));
                    }
                }
                DuplicatePolicy::KeepLast => {
                    if let Some(pos) = selected.iter().position(|(n, _)| n == name) {
                        selected[pos].1 = i;
                    } else {
                        selected.push((name.clone(), i));
                    }
                }
            }
        }
        selected.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
        header_out = Some(selected.iter().map(|(n, _)| n.clone()).collect());
        for row in &table.rows {
            rows_out.push(selected.iter().map(|(_, i)| row[*i].clone()).collect());
        }
        rows_out.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
    } else {
        rows_out = table.rows.clone();
        rows_out.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    if let Some(header) = header_out {
        records.push(header);
    }
    records.extend(rows_out);

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let terminator: &str = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        for (column, field) in record.iter().enumerate() {
            if column > 0 {
                output.push(config.delimiter as char);
            }
            let bytes = field.as_bytes();
            let needs_quote = field.contains(config.delimiter as char)
                || field.contains(config.quote as char)
                || field.contains('\r')
                || field.contains('\n')
                || bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
                || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t')
                || (record_index == 0 && column == 0 && field.starts_with('\u{feff}'))
                || (record.len() == 1 && field.is_empty());

            if needs_quote {
                output.push(config.quote as char);
                for ch in field.chars() {
                    if ch as u32 == config.quote as u32 {
                        output.push(config.quote as char);
                    }
                    output.push(ch);
                }
                output.push(config.quote as char);
            } else {
                output.push_str(field);
            }
        }
    }

    if canonical || config.final_terminator {
        output.push_str(terminator);
    }
    Ok(output)
}
