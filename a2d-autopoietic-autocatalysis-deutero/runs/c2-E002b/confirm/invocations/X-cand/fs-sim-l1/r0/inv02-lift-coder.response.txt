```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut parts = Vec::new();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                let part = &path[start..i];
                if part.is_empty() || part == "." || part == ".." {
                    let offset = if part.is_empty() {
                        if i == bytes.len() { bytes.len() - 1 } else { i }
                    } else {
                        start
                    };
                    return Err(Error::InvalidPath(offset));
                }
                parts.push(part);
                start = i + 1;
            }
        }

        Ok(parts)
    }

    fn canonical_prefix(parts: &[&str]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            let mut result = String::new();
            for part in parts {
                result.push('/');
                result.push_str(part);
            }
            result
        }
    }

    fn find<'a>(&'a self, parts: &[&str]) -> Result<&'a Entry, Error> {
        let mut entry = &self.root;
        let mut prefix = Vec::new();

        for (index, part) in parts.iter().enumerate() {
            match entry {
                Entry::Dir(children) => {
                    prefix.push(*part);
                    entry = children
                        .get(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical_prefix(&prefix)))?;
                    if index + 1 < parts.len() && !matches!(entry, Entry::Dir(_)) {
                        return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                    }
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }
            }
        }

        Ok(entry)
    }

    fn parent_mut<'a>(
        &'a mut self,
        parts: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let parent_parts = &parts[..parts.len() - 1];
        let mut entry = &mut self.root;
        let mut prefix = Vec::new();

        for part in parent_parts {
            match entry {
                Entry::Dir(children) => {
                    prefix.push(*part);
                    entry = children
                        .get_mut(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical_prefix(&prefix)))?;
                    if !matches!(entry, Entry::Dir(_)) {
                        return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                    }
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }
            }
        }

        match entry {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(parent_parts))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = Self::components(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent = self.parent_mut(&parts)?;
        let name = parts[parts.len() - 1];
        let canonical = Self::canonical_prefix(&parts);
        if parent.contains_key(name) {
            return Err(Error::Exists(canonical));
        }

        parent.insert(name.to_owned(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = Self::components(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let canonical = Self::canonical_prefix(&parts);
        let parent = self.parent_mut(&parts)?;
        let name = parts[parts.len() - 1];

        match parent.get_mut(name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name.to_owned(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = Self::components(path)?;
        match self.find(&parts)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical_prefix(&parts))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = Self::components(path)?;
        match self.find(&parts)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(&parts))),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
