#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn overhead(mode: &Mode, n: usize) -> usize {
    match mode {
        Mode::Compact => 2 * n.saturating_sub(1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    }
}

fn minimum_column_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn normalize_and_tokenize(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let line = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for ch in line.chars() {
                if markdown && ch == '|' {
                    tokens.push(Token {
                        text: "\\|".to_owned(),
                        width: 2,
                    });
                } else if markdown && ch == '\\' {
                    tokens.push(Token {
                        text: "\\\\".to_owned(),
                        width: 2,
                    });
                } else {
                    tokens.push(Token {
                        text: ch.to_string(),
                        width: 1,
                    });
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.to_vec()];
            }
            let available = width.saturating_sub(1);
            let mut used = 0;
            let mut kept = Vec::new();
            for token in tokens {
                if used + token.width > available {
                    break;
                }
                used += token.width;
                kept.push(token.clone());
            }
            kept.push(Token {
                text: "…".to_owned(),
                width: 1,
            });
            vec![kept]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut index = 0;
            while index < tokens.len() {
                let mut used = 0;
                let start = index;
                while index < tokens.len() && used + tokens[index].width <= width {
                    used += tokens[index].width;
                    index += 1;
                }
                result.push(tokens[start..index].to_vec());
            }
            result
        }
    }
}

fn join_tokens(tokens: &[Token]) -> String {
    tokens.iter().map(|t| t.text.as_str()).collect()
}

fn aligned(content: &str, content_width: usize, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), content, " ".repeat(right))
}

fn data_line(
    cells: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    match mode {
        Mode::Grid | Mode::Markdown => {
            let mut line = String::from("|");
            for (i, content) in cells.iter().enumerate() {
                let count = content.chars().count();
                line.push(' ');
                line.push_str(&aligned(content, count, widths[i], &alignments[i]));
                line.push(' ');
                line.push('|');
            }
            line.push('\n');
            line
        }
        Mode::Compact => {
            let mut line = String::new();
            for i in 0..cells.len() {
                if i > 0 {
                    line.push_str("  ");
                }
                let count = cells[i].chars().count();
                let padded = aligned(&cells[i], count, widths[i], &alignments[i]);
                if i + 1 == cells.len() {
                    let right_padding = match alignments[i] {
                        Alignment::Left => widths[i].saturating_sub(count),
                        Alignment::Center => widths[i].saturating_sub(count)
                            - (widths[i].saturating_sub(count) / 2),
                        Alignment::Right => 0,
                    };
                    let keep = padded.chars().count().saturating_sub(right_padding);
                    line.extend(padded.chars().take(keep));
                } else {
                    line.push_str(&padded);
                }
            }
            line.push('\n');
            line
        }
    }
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

fn markdown_separator(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut line = String::from("|");
    for (width, alignment) in widths.iter().zip(alignments) {
        let marker = match alignment {
            Alignment::Left => format!(":{}", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        line.push(' ');
        line.push_str(&marker);
        line.push(' ');
        line.push('|');
    }
    line.push('\n');
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_col = minimum_column_width(&options.mode);
    let fixed = overhead(&options.mode, n);
    let minimum = fixed + n * min_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut byte = 0;
            let chars: Vec<char> = cell.chars().collect();
            let mut i = 0;
            while i < chars.len() {
                let ch = chars[i];
                if ch == '\r' {
                    if i + 1 < chars.len() && chars[i + 1] == '\n' {
                        byte += ch.len_utf8() + chars[i + 1].len_utf8();
                        i += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                byte += ch.len_utf8();
                i += 1;
            }
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let transformed: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| normalize_and_tokenize(cell, options.mode == Mode::Markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![min_col; n];
    for row in &transformed {
        for (column, logical_lines) in row.iter().enumerate() {
            for line in logical_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + fixed > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_col)
                .max_by_key(|&i| (widths[i], i));
            if let Some(column) = candidate {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in transformed.iter().enumerate() {
        let per_cell: Vec<Vec<Vec<Token>>> = row
            .iter()
            .enumerate()
            .map(|(column, logical_lines)| {
                let mut all = Vec::new();
                for logical_line in logical_lines {
                    all.extend(fragments(logical_line, widths[column], &overflows[column]));
                }
                all
            })
            .collect();

        let height = per_cell.iter().map(Vec::len).max().unwrap_or(1);
        for fragment_index in 0..height {
            let contents: Vec<String> = (0..n)
                .map(|column| {
                    per_cell[column]
                        .get(fragment_index)
                        .map(|tokens| join_tokens(tokens))
                        .unwrap_or_default()
                })
                .collect();
            output.push_str(&data_line(
                &contents,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if options.mode == Mode::Markdown && row_index == 0 {
            output.push_str(&markdown_separator(&widths, &alignments));
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
