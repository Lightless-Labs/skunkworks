struct Token {
    text: String,
    width: usize,

fn overhead(mode: &Mode, n: usize) -> usize {
    match mode {
        Mode::Compact => 2 * n.saturating_sub(1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    }
}

fn minimum_column_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn normalize_and_tokenize(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let line = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for ch in line.chars() {
                if markdown && ch == '|' {
                    tokens.push(Token {
                        text: "\\|".to_owned(),
                        width: 2,
                    });
                } else if markdown && ch == '\\' {
                    tokens.push(Token {
                        text: "\\\\".to_owned(),
                        width: 2,
                    });
                } else {
                    tokens.push(Token {
                        text: ch.to_string(),
                        width: 1,
                    });
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.to_vec()];
            }
            let available = width.saturating_sub(1);
            let mut used = 0;
            let mut kept = Vec::new();
            for token in tokens {
                if used + token.width > available {
                    break;
                }
                used += token.width;
                kept.push(token.clone());
            }
            kept.push(Token {
                text: "…".to_owned(),
                width: 1,
            });
            vec![kept]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut index = 0;
            while index < tokens.len() {
                let mut used = 0;
                let start = index;
                while index < tokens.len() && used + tokens[index].width <= width {
                    used += tokens[index].width;
                    index += 1;
                }
                result.push(tokens[start..index].to_vec());
            }
            result
        }
    }
}

fn join_tokens(tokens: &[Token]) -> String {
    tokens.iter().map(|t| t.text.as_str()).collect()
}

fn aligned(content: &str, content_width: usize, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), content, " ".repeat(right))
}

fn data_line(
    cells: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    match mode {
        Mode::Grid | Mode::Markdown => {
            let mut line = String::from("|");
            for (i, content) in cells.iter().enumerate() {
                let count = content.chars().count();
                line.push(' ');
                line.push_str(&aligned(content, count, widths[i], &alignments[i]));
                line.push(' ');
                line.push('|');
            }
            line.push('\n');
            line
        }
        Mode::Compact => {
            let mut line = String::new();
            for i in 0..cells.len() {
                if i > 0 {
                    line.push_str("  ");
                }
                let count = cells[i].chars().count();
                let padded = aligned(&cells[i], count, widths[i], &alignments[i]);
                if i + 1 == cells.len() {
                    let right_padding = match alignments[i] {
                        Alignment::Left => widths[i].saturating_sub(count),
                        Alignment::Center => widths[i].saturating_sub(count)
                            - (widths[i].saturating_sub(count) / 2),
                        Alignment::Right => 0,
                    };
                    let keep = padded.chars().count().saturating_sub(right_padding);
                    line.extend(padded.chars().take(keep));
                } else {
                    line.push_str(&padded);
                }
            }
            line.push('\n');
            line
        }
    }
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

fn markdown_separator(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut line = String::from("|");
    for (width, alignment) in widths.iter().zip(alignments) {
        let marker = match alignment {
            Alignment::Left => format!(":{}", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        line.push(' ');
        line.push_str(&marker);
        line.push(' ');
        line.push('|');
    }
    line.push('\n');
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_col = minimum_column_width(&options.mode);
    let fixed = overhead(&options.mode, n);
    let minimum = fixed + n * min_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut byte = 0;
            let chars: Vec<char> = cell.chars().collect();
            let mut i = 0;
            while i < chars.len() {
                let ch = chars[i];
                if ch == '\r' {
                    if i + 1 < chars.len() && chars[i + 1] == '\n' {
                        byte += ch.len_utf8() + chars[i + 1].len_utf8();
                        i += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                byte += ch.len_utf8();
                i += 1;
            }
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let transformed: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| normalize_and_tokenize(cell, options.mode == Mode::Markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![min_col; n];
    for row in &transformed {
        for (column, logical_lines) in row.iter().enumerate() {
            for line in logical_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + fixed > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_col)
                .max_by_key(|&i| (widths[i], i));
            if let Some(column) = candidate {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in transformed.iter().enumerate() {
        let per_cell: Vec<Vec<Vec<Token>>> = row
            .iter()
            .enumerate()
            .map(|(column, logical_lines)| {
                let mut all = Vec::new();
                for logical_line in logical_lines {
                    all.extend(fragments(logical_line, widths[column], &overflows[column]));
                }
                all
            })
            .collect();

        let height = per_cell.iter().map(Vec::len).max().unwrap_or(1);
        for fragment_index in 0..height {
            let contents: Vec<String> = (0..n)
                .map(|column| {
                    per_cell[column]
                        .get(fragment_index)
                        .map(|tokens| join_tokens(tokens))
                        .unwrap_or_default()
                })
                .collect();
            output.push_str(&data_line(
                &contents,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if options.mode == Mode::Markdown && row_index == 0 {
            output.push_str(&markdown_separator(&widths, &alignments));
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }
    Ok(output)
}

fn validate(rows: &[Vec<String>]) -> Result<(), TableError> {
    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }

    Ok(())

fn invalid_char(ch: char) -> bool {
    (ch == '\r') || (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}'
}

fn validate_cells(rows: &[Vec<String>]) -> Result<(), TableError> {
    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut iter = cell.char_indices().peekable();
            while let Some((byte, ch)) = iter.next() {
                if ch == '\r' {
                    if matches!(iter.peek(), Some((_, '\n'))) {
                        iter.next();
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                if invalid_char(ch) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }
    Ok(())
}

fn split_fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut out = String::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut index = 0;
            while index < tokens.len() {
                let mut used = 0;
                let start = index;
                while index < tokens.len() && used + tokens[index].width <= width {
                    used += tokens[index].width;
                    index += 1;
                }
                if index == start {
                    // The mode-specific minimum guarantees this is unreachable.
                    index += 1;
                }
                result.push(tokens[start..index].iter().map(|t| t.text.as_str()).collect());
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align_content(content: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let extra = width.saturating_sub(scalar_len(content));
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let right = if omit_right { 0 } else { right };
    format!("{}{}{}", " ".repeat(left), content, " ".repeat(right))
}

fn append_data_line(
    output: &mut String,
    contents: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Compact => {
            for column in 0..contents.len() {
                if column > 0 {
                    output.push_str("  ");
                }
                output.push_str(&align_content(
                    &contents[column],
                    widths[column],
                    &alignments[column],
                    column + 1 == contents.len(),
                ));
            }
            output.push('\n');
        }
        Mode::Grid | Mode::Markdown => {
            output.push('|');
            for column in 0..contents.len() {
                output.push(' ');
                output.push_str(&align_content(
                    &contents[column],
                    widths[column],
                    &alignments[column],
                    false,
                ));
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
    }
}

fn append_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for width in widths {
        output.push_str(&"-".repeat(width + 2));
        output.push('+');
    }
    output.push('\n');
}

fn append_markdown_separator(
    output: &mut String,
    widths: &[usize],
    alignments: &[Alignment],
) {
    output.push('|');
    for (width, alignment) in widths.iter().zip(alignments) {
        output.push(' ');
        match alignment {
            Alignment::Left => {
                output.push(':');
                output.push_str(&"-".repeat(width - 1));
            }
            Alignment::Right => {
                output.push_str(&"-".repeat(width - 1));
                output.push(':');
            }
            Alignment::Center => {
                output.push(':');
                output.push_str(&"-".repeat(width - 2));
                output.push(':');
            }
        }
        output.push(' ');
        output.push('|');
    }
    output.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    validate(rows)?;
    let n = rows[0].len();

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column = match options.mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_column;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    validate_cells(rows)?;

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let logical: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| normalized_lines(cell, markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_column; n];
    for row in &logical {
        for (column, cell_lines) in row.iter().enumerate() {
            for line in cell_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let mut candidate: Option<usize> = None;
            for column in 0..n {
                if widths[column] > minimum_column
                    && candidate.map_or(true, |current| widths[column] >= widths[current])
                {
                    candidate = Some(column);
                }
            }
            if let Some(column) = candidate {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        append_border(&mut output, &widths);
    }

    for (row_index, row) in logical.iter().enumerate() {
        let mut fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let cell_fragments: Vec<String> = row[column]
                .iter()
                .flat_map(|line| split_fragments(line, widths[column], &overflows[column]))
                .collect();
            height = height.max(cell_fragments.len());
            fragments.push(cell_fragments);
        }

        for fragment_index in 0..height {
            let contents: Vec<String> = fragments
                .iter()
                .map(|column| column.get(fragment_index).cloned().unwrap_or_default())
                .collect();
            append_data_line(&mut output, &contents, &widths, &alignments, &options.mode);
        }

        if markdown && options.header && row_index == 0 {
            append_markdown_separator(&mut output, &widths, &alignments);
        }
        if matches!(options.mode, Mode::Grid)
            && options.header
            && row_index == 0
            && rows.len() > 1
        {
            append_border(&mut output, &widths);
        }
    }

    if matches!(options.mode, Mode::Grid) {
        append_border(&mut output, &widths);
    }

    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type LogicalLines = Vec<Vec<Token>>;

fn invalid_scalar(c: char, next: Option<char>) -> bool {
    if c == '\r' {
        return next != Some('\n');
    }
    c.is_control() && c != '\n' || c == '\u{2028}' || c == '\u{2029}'
}

fn tokenize_line(line: &str, mode: &Mode) -> Vec<Token> {
    let trimmed = line.trim_end_matches(' ');
    let mut tokens = Vec::new();

    for c in trimmed.chars() {
        if matches!(mode, Mode::Markdown) && c == '|' {
            tokens.push(Token {
                text: "\\|".to_owned(),
                width: 2,
            });
        } else if matches!(mode, Mode::Markdown) && c == '\\' {
            tokens.push(Token {
                text: "\\\\".to_owned(),
                width: 2,
            });
        } else {
            tokens.push(Token {
                text: c.to_string(),
                width: 1,
            });
        }
    }

    tokens
}

fn tokenize_cell(cell: &str, mode: &Mode) -> LogicalLines {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| tokenize_line(line, mode))
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }

            let mut out = String::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                out.push_str(&token.text);
                used += token.width;
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align(content: &str, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(scalar_len(content));
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut result = String::new();
    result.extend(std::iter::repeat_n(' ', left));
    result.push_str(content);
    result.extend(std::iter::repeat_n(' ', right));
    result
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn data_line(cells: &[String], widths: &[usize], alignments: &[Alignment], mode: &Mode) -> String {
    match mode {
        Mode::Compact => {
            let mut out = String::new();
            for i in 0..cells.len() {
                if i > 0 {
                    out.push_str("  ");
                }
                let padded = align(&cells[i], widths[i], &alignments[i]);
                if i + 1 == cells.len() {
                    let right_padding = match alignments[i] {
                        Alignment::Left => widths[i].saturating_sub(scalar_len(&cells[i])),
                        Alignment::Center => {
                            let extra = widths[i].saturating_sub(scalar_len(&cells[i]));
                            extra - extra / 2
                        }
                        Alignment::Right => 0,
                    };
                    let keep = padded.chars().count().saturating_sub(right_padding);
                    out.extend(padded.chars().take(keep));
                } else {
                    out.push_str(&padded);
                }
            }
            out.push('\n');
            out
        }
        Mode::Grid | Mode::Markdown => {
            let mut out = String::from("|");
            for i in 0..cells.len() {
                out.push(' ');
                out.push_str(&align(&cells[i], widths[i], &alignments[i]));
                out.push_str(" |");
            }
            out.push('\n');
            out
        }
    }
}

fn markdown_separator(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::from("|");
    for (i, &width) in widths.iter().enumerate() {
        out.push(' ');
        match alignments[i] {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push_str(" |");
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column = if matches!(options.mode, Mode::Markdown) {
        3
    } else {
        1
    };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_column;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            let chars: Vec<char> = cell.chars().collect();
            let mut byte = 0;
            for (i, &c) in chars.iter().enumerate() {
                if invalid_scalar(c, chars.get(i + 1).copied()) {
                    return Err(TableError::InvalidChar {
                        row,
                        column,
                        byte,
                    });
                }
                byte += c.len_utf8();
            }
        }
    }

    let default_alignment = Alignment::Left;
    let default_overflow = Overflow::Wrap;
    let alignments: Vec<Alignment> = (0..n)
        .map(|i| options.alignments.get(i).unwrap_or(&default_alignment).clone())
        .collect();
    let overflows: Vec<Overflow> = (0..n)
        .map(|i| options.overflow.get(i).unwrap_or(&default_overflow).clone())
        .collect();

    let logical: Vec<Vec<LogicalLines>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| tokenize_cell(cell, &options.mode))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_column; n];
    for row in &logical {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > minimum_column)
                .max_by_key(|&i| (widths[i], i));
            match candidate {
                Some(i) => {
                    widths[i] -= 1;
                    total -= 1;
                }
                None => break,
            }
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in logical.iter().enumerate() {
        let cell_fragments: Vec<Vec<String>> = row
            .iter()
            .enumerate()
            .map(|(column, lines)| {
                lines
                    .iter()
                    .flat_map(|line| fragments(line, widths[column], &overflows[column]))
                    .collect()
            })
            .collect();

        let height = cell_fragments.iter().map(Vec::len).max().unwrap_or(1);
        for fragment_index in 0..height {
            let cells: Vec<String> = cell_fragments
                .iter()
                .map(|frags| frags.get(fragment_index).cloned().unwrap_or_default())
                .collect();
            output.push_str(&data_line(&cells, &widths, &alignments, &options.mode));
        }

        if matches!(options.mode, Mode::Grid) {
            if options.header && row_index == 0 && rows.len() > 1 {
                output.push_str(&border(&widths));
            } else if row_index + 1 == rows.len() {
                output.push_str(&border(&widths));
            }
        } else if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            output.push_str(&markdown_separator(&widths, &alignments));
        }
    }

    Ok(output)
}
