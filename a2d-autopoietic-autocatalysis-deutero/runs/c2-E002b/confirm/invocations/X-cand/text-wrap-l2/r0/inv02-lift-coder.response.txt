#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut normalized = String::with_capacity(input.len());
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        if ch == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut segment = String::new();
    let mut consecutive_lfs = 0usize;

    for ch in normalized.chars() {
        if ch == '\n' {
            consecutive_lfs += 1;
        } else {
            if consecutive_lfs >= 2 {
                wrap_segment(&segment, options, &mut output);
                segment.clear();
                output.extend((1..consecutive_lfs).map(|_| String::new()));
            } else if consecutive_lfs == 1 {
                segment.push('\n');
            }
            consecutive_lfs = 0;
            segment.push(ch);
        }
    }

    if consecutive_lfs >= 2 {
        wrap_segment(&segment, options, &mut output);
        output.extend((1..consecutive_lfs).map(|_| String::new()));
    } else {
        if consecutive_lfs == 1 {
            segment.push('\n');
        }
        wrap_segment(&segment, options, &mut output);
    }

    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(segment: &str, options: &WrapOptions, output: &mut Vec<String>) {
    let words: Vec<Vec<char>> = segment
        .split(is_separator)
        .filter(|word| !word.is_empty())
        .map(|word| word.chars().collect())
        .collect();

    let mut first_line = true;
    let mut line = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;

            if content_len == 0 {
                let remaining = word.len() - offset;
                if remaining <= capacity {
                    line = " ".repeat(indent);
                    line.extend(word[offset..].iter());
                    content_len = remaining;
                    offset = word.len();
                } else {
                    line = " ".repeat(indent);
                    line.extend(word[offset..offset + capacity].iter());
                    output.push(std::mem::take(&mut line));
                    offset += capacity;
                    first_line = false;
                }
            } else {
                let remaining = word.len() - offset;
                let available = capacity.saturating_sub(content_len + 1);

                if remaining <= available {
                    line.push(' ');
                    line.extend(word[offset..].iter());
                    content_len += remaining + 1;
                    offset = word.len();
                } else if remaining <= capacity {
                    output.push(std::mem::take(&mut line));
                    content_len = 0;
                    first_line = false;
                } else if available >= 2 {
                    line.push(' ');
                    line.extend(word[offset..offset + available].iter());
                    output.push(std::mem::take(&mut line));
                    content_len = 0;
                    offset += available;
                    first_line = false;
                } else {
                    output.push(std::mem::take(&mut line));
                    content_len = 0;
                    first_line = false;
                }
            }
        }

        // A word may leave a short suffix on a line; the next word will use
        // the normal fit and splitting rules, including its separator.
    }

    if content_len > 0 {
        output.push(line);
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn public_unicode_words() {
        assert_eq!(
            wrap(
                "猫犬 鳥魚",
                &WrapOptions {
                    width: 5,
                    first_indent: 0,
                    rest_indent: 0
                }
            ),
            Ok(vec!["猫犬 鳥魚".to_string()])
        );
    }

    #[test]
    fn public_hanging() {
        assert_eq!(
            wrap(
                "a bb ccc dd",
                &WrapOptions {
                    width: 7,
                    first_indent: 1,
                    rest_indent: 2
                }
            ),
            Ok(vec![
                " a bb".to_string(),
                "  ccc".to_string(),
                "  dd".to_string()
            ])
        );
    }

    #[test]
    fn public_paragraph() {
        assert_eq!(
            wrap(
                "a\n\nb",
                &WrapOptions {
                    width: 4,
                    first_indent: 0,
                    rest_indent: 1
                }
            ),
            Ok(vec!["a".to_string(), "".to_string(), "b".to_string()])
        );
    }

    #[test]
    fn public_bad_indent() {
        assert_eq!(
            wrap(
                "x",
                &WrapOptions {
                    width: 3,
                    first_indent: 3,
                    rest_indent: 0
                }
            ),
            Err(WrapError::FirstIndentTooWide { indent: 3 })
        );
    }
}
