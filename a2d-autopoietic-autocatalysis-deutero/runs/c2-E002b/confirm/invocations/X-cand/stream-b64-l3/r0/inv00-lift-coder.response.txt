#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_byte(alphabet: Alphabet, value: u8) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn decode_digit(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let mut out = String::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(alphabet_byte(alphabet, a >> 2) as char);
    out.push(alphabet_byte(alphabet, ((a & 0x03) << 4) | (b >> 4)) as char);

    if bytes.len() > 1 {
        out.push(alphabet_byte(alphabet, ((b & 0x0f) << 2) | (c >> 6)) as char);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(alphabet_byte(alphabet, c & 0x3f) as char);
    } else if padding != Padding::Forbidden {
        out.push('=');
    }

    out
}

fn format_wrapped(raw: String, whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return raw;
    };

    if raw.is_empty() {
        return raw;
    }

    let separator = match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };
    let mut out = String::with_capacity(raw.len() + raw.len() / width * separator.len());
    for (i, chunk) in raw.as_bytes().chunks(width).enumerate() {
        if i != 0 {
            out.push_str(separator);
        }
        out.push_str(std::str::from_utf8(chunk).unwrap());
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;

    let mut raw = String::with_capacity(input.len().div_ceil(3) * 4);
    for chunk in input.chunks(3) {
        raw.push_str(&encode_group(chunk, options.alphabet, options.padding));
    }
    Ok(format_wrapped(raw, options.whitespace))
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            pending: Vec::with_capacity(2),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.pending.len() + input.len());
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let mut raw = String::with_capacity(complete_len / 3 * 4);
        for chunk in combined[..complete_len].chunks(3) {
            raw.push_str(&encode_group(chunk, self.options.alphabet, self.options.padding));
        }

        self.pending.clear();
        self.pending.extend_from_slice(&combined[complete_len..]);
        Ok(format_wrapped(raw, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let raw = if self.pending.is_empty() {
            String::new()
        } else {
            encode_group(&self.pending, self.options.alphabet, self.options.padding)
        };
        self.pending.clear();
        Ok(format_wrapped(raw, self.options.whitespace))
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

struct DecoderState {
    options: Options,
    group: Vec<Symbol>,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    offset: usize,
    row_symbols: usize,
    wrap_state: WrapState,
    finished: bool,
    sticky_error: Option<Error>,
}

#[derive(Clone, Copy)]
enum WrapState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

impl DecoderState {
    fn new(options: Options) -> Self {
        Self {
            options,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            pending_second_padding: None,
            offset: 0,
            row_symbols: 0,
            wrap_state: WrapState::Normal,
            finished: false,
            sticky_error: None,
        }
    }

    fn ascii_ignored(byte: u8) -> bool {
        matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn scan_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if matches!(self.options.whitespace, Whitespace::IgnoreAscii)
            && Self::ascii_ignored(byte)
        {
            return Ok(());
        }

        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            match self.wrap_state {
                WrapState::NeedLf { cr_offset } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.wrap_state = WrapState::NeedSymbol {
                        separator_offset: cr_offset,
                    };
                    return Ok(());
                }
                WrapState::NeedSymbol { .. } => {
                    if matches!(byte, b'\r' | b'\n' | 9 | 11 | 12 | 32) {
                        return Err(Error::BadWrap { offset });
                    }
                    self.wrap_state = WrapState::Normal;
                    self.row_symbols = 0;
                }
                WrapState::Normal => {}
            }

            if self.row_symbols == width {
                match ending {
                    LineEnding::Lf if byte == b'\n' => {
                        self.wrap_state = WrapState::NeedSymbol {
                            separator_offset: offset,
                        };
                        return Ok(());
                    }
                    LineEnding::CrLf if byte == b'\r' => {
                        self.wrap_state = WrapState::NeedLf { cr_offset: offset };
                        return Ok(());
                    }
                    _ => return Err(Error::BadWrap { offset }),
                }
            } else if matches!(byte, b'\r' | b'\n' | 9 | 11 | 12 | 32) {
                return Err(Error::BadWrap { offset });
            }
        }

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.pending_second_padding.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match self.group.len() {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_second_padding = Some(offset);
                    self.group.push(Symbol { byte, offset });
                    if let Whitespace::Wrapped { .. } = self.options.whitespace {
                        self.row_symbols += 1;
                    }
                    return Ok(());
                }
                3 => {
                    self.group.push(Symbol { byte, offset });
                    self.emit_group(output)?;
                    self.terminal_padding = true;
                    if let Whitespace::Wrapped { .. } = self.options.whitespace {
                        self.row_symbols += 1;
                    }
                    return Ok(());
                }
                _ => return Err(Error::BadPadding { offset }),
            }
        }

        let Some(_) = decode_digit(self.options.alphabet, byte) else {
            return Err(Error::InvalidByte { offset, byte });
        };

        self.group.push(Symbol { byte, offset });
        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            self.row_symbols += 1;
        }

        if self.group.len() == 4 {
            self.emit_group(output)?;
        }
        Ok(())
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.group.len() != 4 {
            return Ok(());
        }

        let q = &self.group;
        let a = decode_digit(self.options.alphabet, q[0].byte)
            .ok_or(Error::InvalidByte { offset: q[0].offset, byte: q[0].byte })?;
        let b = decode_digit(self.options.alphabet, q[1].byte)
            .ok_or(Error::InvalidByte { offset: q[1].offset, byte: q[1].byte })?;

        if q[2].byte == b'=' {
            if q[3].byte != b'=' {
                return Err(Error::BadPadding { offset: q[3].offset });
            }
            output.push((a << 2) | (b >> 4));
        } else {
            let c = decode_digit(self.options.alphabet, q[2].byte)
                .ok_or(Error::InvalidByte { offset: q[2].offset, byte: q[2].byte })?;
            output.push((a << 2) | (b >> 4));

            if q[3].byte == b'=' {
                output.push((b << 4) | (c >> 2));
            } else {
                let d = decode_digit(self.options.alphabet, q[3].byte)
                    .ok_or(Error::InvalidByte { offset: q[3].offset, byte: q[3].byte })?;
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
            }
        }

        self.group.clear();
        self.pending_second_padding = None;
        Ok(())
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        if let WrapState::NeedLf { cr_offset } = self.wrap_state {
            return Err(Error::BadWrap { offset: cr_offset });
        }
        if let WrapState::NeedSymbol { separator_offset } = self.wrap_state {
            return Err(Error::BadWrap { offset: separator_offset });
        }

        if let Some(offset) = self.pending_second_padding {
            return Err(Error::BadPadding { offset });
        }

        let mut output = Vec::new();
        let len = self.group.len();
        match len {
            0 => {}
            1 => return Err(Error::Truncated { offset: self.group[0].offset }),
            2 | 3 => match self.options.padding {
                Padding::Required => {
                    return Err(Error::Truncated { offset: self.group[0].offset });
                }
                Padding::Forbidden | Padding::Optional => {
                    let a = decode_digit(self.options.alphabet, self.group[0].byte).unwrap();
                    let b = decode_digit(self.options.alphabet, self.group[1].byte).unwrap();
                    output.push((a << 2) | (b >> 4));
                    if len == 3 {
                        let c = decode_digit(self.options.alphabet, self.group[2].byte).unwrap();
                        output.push((b << 4) | (c >> 2));
                    }
                }
            },
            _ => {}
        }
        self.group.clear();
        Ok(output)
    }

    fn push_inner(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.scan_byte(byte, offset, &mut output) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }
}

pub struct Decoder {
    state: DecoderState,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            state: DecoderState::new(options),
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        self.state.push_inner(input)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky_error {
            return Err(error.clone());
        }
        match self.state.finish_inner() {
            Ok(output) => Ok(output),
            Err(error) => {
                self.state.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate_options(options)?;
    let mut decoder = DecoderState::new(options);
    decoder.push_inner(input)?;
    decoder.finish_inner()
}
