use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]

    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(error) => {
            let at = origins.get(error.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

fn component_safe(component: Component, byte: u8) -> bool {
    byte.is_ascii_alphanumeric()
        || matches!(byte, b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if component_safe(component, byte) {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

fn global_byte_scan(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (offset, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }

    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
        {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    Ok(value as u16)
}

fn codec_to_parse(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_byte_scan(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| authority_start + n)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let open_at = hostport_start;
        let close_rel = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open_at }),
        };
        let interior = &hostport[1..close_rel];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }
        let suffix = &hostport[close_rel + 1..];
        let suffix_start = hostport_start + close_rel + 1;
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], suffix_start + 1)?)
        } else {
            return Err(ParseError::InvalidHost { at: suffix_start });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let port_colon = hostport.bytes().position(|b| b == b':');
        let (raw_host, parsed_port) = if let Some(index) = port_colon {
            (
                &hostport[..index],
                Some(parse_port(&hostport[index + 1..], hostport_start + index + 1)?),
            )
        } else {
            (hostport, None)
        };

        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (offset, byte) in raw_host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + offset,
                });
            }
        }
        (raw_host.to_ascii_lowercase(), parsed_port)
    };

    let remainder = &input[authority_end..];
    let fragment_pos = remainder.find('#');
    let before_fragment = fragment_pos.map(|i| &remainder[..i]).unwrap_or(remainder);
    let fragment = fragment_pos.map(|i| remainder[i + 1..].to_owned());

    let query_pos = before_fragment.find('?');
    let path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| before_fragment[i + 1..].to_owned());

    let path = if path.is_empty() {
        "/".to_owned()
    } else {
        path.to_owned()
    };

    if let Some(ref raw_userinfo) = userinfo {
        let userinfo_offset = authority_start;
        decode_component(Component::UserInfo, raw_userinfo)
            .map_err(|e| codec_to_parse(e, userinfo_offset))?;
    }

    if !path.is_empty() && path != "/" {
        let path_offset = authority_end;
        decode_component(Component::Path, &path)
            .map_err(|e| codec_to_parse(e, path_offset))?;
    }

    if let Some(ref raw_query) = query {
        let query_offset = authority_end
            + query_pos.unwrap_or(0)
            + 1;
        decode_component(Component::Query, raw_query)
            .map_err(|e| codec_to_parse(e, query_offset))?;
    }

    if let Some(ref raw_fragment) = fragment {
        let fragment_offset = authority_end + fragment_pos.unwrap_or(0) + 1;
        decode_component(Component::Fragment, raw_fragment)
            .map_err(|e| codec_to_parse(e, fragment_offset))?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn codec_to_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_to_path)?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut final_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(codec_to_path)?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if is_dot {
            if index + 1 == segments.len() {
                final_navigation = true;
            }
        } else if is_dotdot {
            if !stack.is_empty() {
                stack.pop();
            }
            if index + 1 == segments.len() {
                final_navigation = true;
            }
        } else {
            stack.push((*segment).to_owned());
        }
    }

    if final_navigation {
        stack.push(String::new());
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn has_invalid_byte(input: &str) -> Option<usize> {
    input
        .bytes()
        .enumerate()
        .find_map(|(at, byte)| (!(0x21..=0x7e).contains(&byte)).then_some(at))
}

fn split_components(rest: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_pos = rest.find('#');
    let before_fragment = fragment_pos.map(|i| &rest[..i]).unwrap_or(rest);
    let fragment = fragment_pos.map(|i| &rest[i + 1..]);
    let query_pos = before_fragment.find('?');
    let path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| &before_fragment[i + 1..]);
    (path, query, fragment)
}

fn make_absolute_url(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut raw = String::new();
    raw.push_str(&url.scheme);
    raw.push_str("://");
    if let Some(userinfo) = &url.userinfo {
        raw.push_str(userinfo);
        raw.push('@');
    }
    if url.ipv6 {
        raw.push('[');
        raw.push_str(&url.host);
        raw.push(']');
    } else {
        raw.push_str(&url.host);
    }
    if let Some(port) = url.port {
        raw.push(':');
        raw.push_str(&port.to_string());
    }
    raw.push_str(path);
    if let Some(query) = query {
        raw.push('?');
        raw.push_str(query);
    }
    if let Some(fragment) = fragment {
        raw.push('#');
        raw.push_str(fragment);
    }
    raw
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Some(at) = has_invalid_byte(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    if let Some(colon) = reference.find(':') {
        let first_delim = reference
            .find(|c| matches!(c, '/' | '?' | '#'))
            .unwrap_or(reference.len());
        if colon < first_delim {
            let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
            url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
            return Ok(url);
        }
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_components(reference);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_path.clone(),
            if reference_query.is_some() {
                reference_query
            } else {
                base_url.query.as_deref()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let last_slash = base_path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_path[..=last_slash], reference_path)
        };
        (merged, reference_query)
    };

    let raw = make_absolute_url(
        &base_url,
        &selected_path,
        selected_query,
        reference_fragment,
    );
    let mut url = parse_url(&raw).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn query_codec_error(error: CodecError, offset: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

fn form_plus_to_space(text: &str) -> String {
    text.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();

    for (field_start, field) in input
        .split('&')
        .scan(0usize, |offset, field| {
            let start = *offset;
            *offset += field.len() + 1;
            Some((start, field))
        })
    {
        if field.is_empty() {
            continue;
        }

        let (raw_key, raw_value) = match field.find('=') {
            Some(index) => (&field[..index], &field[index + 1..]),
            None => (field, ""),
        };

        let key_input = form_plus_to_space(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|e| query_codec_error(e, field_start))?;

        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }

        let value_input = form_plus_to_space(raw_value);
        let value_offset = field_start + raw_key.len() + usize::from(field.contains('='));
        let value = decode_component(Component::Query, &value_input)
            .map_err(|e| query_codec_error(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| existing == &key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn encode_form(text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &byte in text.as_bytes() {
        if byte == b' ' {
            output.push('+');
        } else if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }

    output
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();

    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing_values)) = merged.iter_mut().find(|(existing, _)| existing == key) {
            existing_values.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = encode_form(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, encode_form(&value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }

    output
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query = match &url.query {
        Some(raw) => Some(parse_query(raw).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = String::new();
    output.push_str(&url.scheme);
    output.push_str("://");

    if let Some(userinfo) = &url.userinfo {
        let decoded =
            decode_component(Component::UserInfo, userinfo).map_err(|e| {
                CanonicalError::Parse(codec_to_parse(e, 0))
            })?;
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }

    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host)
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        output.push('[');
        output.push_str(&address.to_string());
        output.push(']');
    } else {
        output.push_str(&url.host);
    }

    let omit_port = matches!((url.scheme.as_str(), url.port), ("http", Some(80)) | ("https", Some(443)));
    if let Some(port) = url.port.filter(|_| !omit_port) {
        output.push(':');
        output.push_str(&port.to_string());
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(codec_to_path(e)))?;
        canonical_segments.push(encode_unreserved(&decoded));
    }
    output.push_str(&canonical_segments.join("/"));

    if let Some(map) = query {
        output.push('?');
        output.push_str(&serialise_query(&map));
    }

    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(codec_to_parse(e, 0)))?;
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}
