fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = match hex_value(bytes[i + 1]) {
                Some(v) => v,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            let lo = match hex_value(bytes[i + 2]) {
                Some(v) => v,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            decoded.push((hi << 4) | lo);
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = offsets.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

fn component_safe(component: Component, byte: u8) -> bool {
    matches!(byte, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if component_safe(component, byte) {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

fn invalid_global_byte(input: &str) -> Option<usize> {
    input
        .as_bytes()
        .iter()
        .position(|&b| !(0x21..=0x7e).contains(&b))
}

fn ascii_lower(s: &str) -> String {
    s.bytes().map(|b| b.to_ascii_lowercase() as char).collect()
}

fn codec_to_parse(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(text: &str, base: usize, component: Component) -> Result<(), ParseError> {
    decode_component(component, text)
        .map(|_| ())
        .map_err(|e| codec_to_parse(e, base))
}

fn validate_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    if let Some(index) = port.as_bytes().iter().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort {
            at: port_start + index,
        });
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let mut value: u32 = 0;
    for &b in port.as_bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some(at) = invalid_global_byte(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .as_bytes()
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }
    let (userinfo, hostport, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let suffix_start = close + 1;
        let suffix = &hostport[suffix_start..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(validate_port(
                &suffix[1..],
                hostport_start + suffix_start + 1,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + suffix_start,
            });
        };
        (ascii_lower(interior), port, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, port) = if let Some(i) = split {
            (
                &hostport[..i],
                Some(validate_port(&hostport[i + 1..], hostport_start + i + 1)?),
            )
        } else {
            (hostport, None)
        };

        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        if let Some(i) = raw_host
            .as_bytes()
            .iter()
            .position(|&b| !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')))
        {
            return Err(ParseError::InvalidHost {
                at: hostport_start + i,
            });
        }
        (ascii_lower(raw_host), port, false)
    };

    let fragment_start = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|i| authority_end + i);
    let pre_fragment_end = fragment_start.unwrap_or(bytes.len());
    let query_start = bytes[authority_end..pre_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| authority_end + i);

    let path_end = query_start.unwrap_or(pre_fragment_end);
    let raw_path = &input[authority_end..path_end];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };
    let query = query_start.map(|i| input[i + 1..pre_fragment_end].to_owned());
    let fragment = fragment_start.map(|i| input[i + 1..].to_owned());

    if let Some(ref value) = userinfo {
        validate_component(value, authority_start, Component::UserInfo)?;
    }
    if !raw_path.is_empty() {
        validate_component(raw_path, authority_end, Component::Path)?;
    }
    if let Some(i) = query_start {
        validate_component(&input[i + 1..pre_fragment_end], i + 1, Component::Query)?;
    }
    if let Some(i) = fragment_start {
        validate_component(&input[i + 1..], i + 1, Component::Fragment)?;
    }

    Ok(Url {
        scheme: ascii_lower(&input[..colon]),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn codec_to_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_to_path)?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(codec_to_path)?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if index + 1 == raw_segments.len() {
            terminal_navigation = is_dot || is_dotdot;
        }
        if is_dot {
            continue;
        } else if is_dotdot {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        return Ok("/".to_owned());
    }
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    Ok(result)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Some(at) = invalid_global_byte(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let first_structural = reference
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference.as_bytes()[..first_structural].contains(&b':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let bytes = reference.as_bytes();
    let fragment_start = bytes.iter().position(|&b| b == b'#');
    let before_fragment = fragment_start.unwrap_or(bytes.len());
    let query_start = bytes[..before_fragment].iter().position(|&b| b == b'?');
    let path_end = query_start.unwrap_or(before_fragment);
    let reference_path = &reference[..path_end];
    let reference_query = query_start.map(|i| reference[i + 1..before_fragment].to_owned());
    let reference_fragment = fragment_start.map(|i| reference[i + 1..].to_owned());

    let selected_query = if reference_path.is_empty() && query_start.is_none() {
        base_url.query.clone()
    } else {
        reference_query
    };

    let selected_path = if reference_path.is_empty() {
        base_path
    } else if reference_path.starts_with('/') {
        reference_path.to_owned()
    } else {
        let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
        format!("{}{}", &base_path[..slash], reference_path)
    };

    let mut absolute = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        absolute.push_str(userinfo);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&selected_path);
    if let Some(query) = &selected_query {
        absolute.push('?');
        absolute.push_str(query);
    }
    if let Some(fragment) = &reference_fragment {
        absolute.push('#');
        absolute.push_str(fragment);
    }

    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn replace_plus(text: &str) -> String {
    text.replace('+', " ")
}

fn codec_to_query(error: CodecError, base: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut offset = 0;

    for field in input.split('&') {
        let field_start = offset;
        offset += field.len() + 1;
        if field.is_empty() {
            continue;
        }

        let equal = field.as_bytes().iter().position(|&b| b == b'=');
        let (raw_key, raw_value, value_offset) = match equal {
            Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
            None => (field, "", field_start + field.len()),
        };

        let key_input = replace_plus(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|e| codec_to_query(e, field_start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }

        let value_input = replace_plus(raw_value);
        let value = decode_component(Component::Query, &value_input)
            .map_err(|e| codec_to_query(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| existing == &key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing_values)) = merged.iter_mut().find(|(existing, _)| existing == key) {
            existing_values.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

fn canonical_component(text: &str, component: Component) -> Result<String, CodecError> {
    decode_component(component, text).map(|decoded| encode_component(component, &decoded))
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut result = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let encoded = canonical_component(userinfo, Component::UserInfo)
            .map_err(|e| CanonicalError::Parse(codec_to_parse(e, 0)))?;
        result.push_str(&encoded);
        result.push('@');
    }

    if url.ipv6 {
        let address = url
            .host
            .parse::<std::net::Ipv6Addr>()
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        result.push('[');
        result.push_str(&address.to_string());
        result.push(']');
    } else {
        result.push_str(&url.host);
    }

    let omit_port = matches!((url.scheme.as_str(), url.port), ("http", Some(80)) | ("https", Some(443)));
    if let Some(port) = url.port {
        if !omit_port {
            result.push(':');
            result.push_str(&port.to_string());
        }
    }

    let segments: Result<Vec<String>, CodecError> = path
        .split('/')
        .map(|segment| {
            decode_component(Component::Path, segment).map(|decoded| {
                let mut encoded = String::with_capacity(decoded.len());
                const HEX: &[u8; 16] = b"0123456789ABCDEF";
                for &b in decoded.as_bytes() {
                    if matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~') {
                        encoded.push(b as char);
                    } else {
                        encoded.push('%');
                        encoded.push(HEX[(b >> 4) as usize] as char);
                        encoded.push(HEX[(b & 0x0f) as usize] as char);
                    }
                }
                encoded
            })
        })
        .collect();
    result.push_str(&segments.map_err(|e| CanonicalError::Parse(codec_to_parse(e, 0)))?.join("/"));

    if let Some(map) = query_map {
        result.push('?');
        result.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        result.push('#');
        result.push_str(
            &canonical_component(fragment, Component::Fragment)
                .map_err(|e| CanonicalError::Parse(codec_to_parse(e, 0)))?,
        );
    }
    Ok(result)
}
