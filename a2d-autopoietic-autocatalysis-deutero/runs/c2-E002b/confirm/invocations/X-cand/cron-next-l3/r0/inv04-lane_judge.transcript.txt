fn parse_atom(
    field: Field,
    bytes: &[u8],
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    if is_step {
        if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
    } else if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    if !is_step {
        if let Some(value) = name_value(field, bytes) {
            return Ok(value as u32);
        }
    }

    if !bytes.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_item(
    field: Field,
    item: &[u8],
    item_offset: usize,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, 0)
    };

    if base.is_empty() || step_bytes.is_some_and(|s| s.is_empty()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (minimum, maximum) = bounds(field);
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if hyphens.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let wildcard = base == b"*";
    let (start_bytes, end_bytes) = if wildcard {
        (None, None)
    } else if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        (Some(&base[..hyphen]), Some(&base[hyphen + 1..]))
    } else {
        (Some(base), None)
    };

    let mut start = minimum as u32;
    let mut end = maximum as u32;
    let mut end_offset = item_offset;

    if let Some(start_part) = start_bytes {
        start = parse_atom(field, start_part, item_offset, false)?;
        if start < minimum as u32 || start > maximum as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_offset,
            });
        }
        if let Some(end_part) = end_bytes {
            end_offset = item_offset + (end_part.as_ptr() as usize - item.as_ptr() as usize);
            end = parse_atom(field, end_part, end_offset, false)?;
            if end < minimum as u32 || end > maximum as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_offset,
                });
            }
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
        } else if step_bytes.is_none() {
            end = start;
        }
    }

    let step = if let Some(bytes) = step_bytes {
        let value = parse_atom(field, bytes, step_offset, true)?;
        let width = (maximum - minimum + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];

    for (index, field) in FIELDS.iter().copied().enumerate() {
        let bytes = parts[index].as_bytes();
        for (offset, &byte) in bytes.iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut selected = [false; 60];
        let mut offset = 0usize;
        for item in bytes.split(|&b| b == b',') {
            parse_item(field, item, offset, &mut selected)?;
            offset += item.len() + 1;
        }

        let (minimum, maximum) = bounds(field);
        sets[index] = (minimum..=maximum)
            .filter(|&value| selected[value as usize])
            .collect();
    }

    Ok(Cron {
        sets,
        day_of_month_star: parts[2] == "*",
        day_of_week_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if let Some(next) = increment_minute(current) {
                current = next;
            } else {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.sets[0].contains(&at.minute)
            || !self.sets[1].contains(&at.hour)
            || !self.sets[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.sets[2].contains(&at.day);
        let dow = self.sets[4].contains(&weekday(at.year, at.month, at.day));
        match (self.day_of_month_star, self.day_of_week_star) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        at.minute += 1;
        return Some(at);
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return Some(at);
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return Some(at);
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return Some(at);
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return Some(at);
    }
    None
}

    let upper: Vec<u8> = token.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => {
            const NAMES: [&[u8]; 12] = [
                b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN", b"JUL", b"AUG", b"SEP", b"OCT",
                b"NOV", b"DEC",
            ];
            NAMES
                .iter()
                .position(|name| *name == upper.as_slice())
                .map(|i| i as u32 + 1)
        }
        Field::DayOfWeek => {
            const NAMES: [&[u8]; 7] =
                [b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT"];
            NAMES
                .iter()
                .position(|name| *name == upper.as_slice())
                .map(|i| i as u32)
        }
        _ => None,
    }
}

fn parse_number(
    field: Field,
    bytes: &[u8],
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }

    if step {
        if value == 0 || value > field_domain(field).width() {
            return Err(parse_error(field, offset, 5));
        }
    } else {
        let domain = field_domain(field);
        if value < domain.min || value > domain.max {
            return Err(parse_error(field, offset, 3));
        }
    }
    Ok(value)
}

fn parse_atom(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        Ok((parse_number(field, bytes, offset, false)?, true))
    } else if let Some(value) = name_value(field, bytes) {
        Ok((value, false))
    } else {
        Err(parse_error(field, offset, 0))
    }
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        let legal = byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b',');
        if !legal {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let domain = field_domain(field);
    let mut selected = vec![false; (domain.max + 1) as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(parse_error(field, item_start, 0));
        }

        let (base, step_part) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some((slash + 1, &item[slash + 1..])))
        } else {
            (item, None)
        };

        if base.is_empty() || step_part.is_some_and(|(_, part)| part.is_empty()) {
            return Err(parse_error(field, item_start, 0));
        }

        let hyphen_positions: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_is_star = base == b"*";
        let range_parts = if hyphen_positions.len() == 1 {
            let hyphen = hyphen_positions[0];
            if base[..hyphen].is_empty() || base[hyphen + 1..].is_empty() {
                return Err(parse_error(field, item_start, 0));
            }
            Some((&base[..hyphen], &base[hyphen + 1..], hyphen))
        } else if hyphen_positions.len() > 1 || (base_is_star && !hyphen_positions.is_empty()) {
            return Err(parse_error(field, item_start, 0));
        } else {
            None
        };

        if !base_is_star && range_parts.is_none() {
            let valid_atom = base.iter().all(u8::is_ascii_digit)
                || name_value(field, base).is_some();
            if !valid_atom {
                return Err(parse_error(field, item_start, 0));
            }
        }

        if let Some((left, right, _)) = range_parts {
            if left == b"*" || right == b"*" {
                return Err(parse_error(field, item_start, 0));
            }
            let left_valid =
                left.iter().all(u8::is_ascii_digit) || name_value(field, left).is_some();
            let right_valid =
                right.iter().all(u8::is_ascii_digit) || name_value(field, right).is_some();
            if !left_valid || !right_valid {
                return Err(parse_error(field, item_start, 0));
            }
        }

        if let Some((_, step_bytes)) = step_part {
            if !step_bytes.iter().all(u8::is_ascii_digit) {
                return Err(parse_error(field, item_start, 0));
            }
        }

        let mut start = domain.min;
        let mut end = domain.max;
        let mut end_offset = item_start;

        if !base_is_star {
            if let Some((left, right, hyphen)) = range_parts {
                let left_offset = item_start;
                let right_offset = item_start + hyphen + 1;

                let left_value = if left.iter().all(u8::is_ascii_digit) {
                    parse_number(field, left, left_offset, false)?
                } else {
                    name_value(field, left).unwrap()
                };
                let right_value = if right.iter().all(u8::is_ascii_digit) {
                    parse_number(field, right, right_offset, false)?
                } else {
                    name_value(field, right).unwrap()
                };

                start = left_value;
                end = right_value;
                end_offset = right_offset;
                if start > end {
                    return Err(parse_error(field, end_offset, 4));
                }
            } else {
                let (value, numeric) = parse_atom(field, base, item_start)?;
                start = value;
                if step_part.is_none() {
                    end = value;
                } else {
                    end = domain.max;
                }
                if numeric && step_part.is_some() {
                    end = domain.max;
                }
            }
        }

        let step = if let Some((step_offset, step_bytes)) = step_part {
            parse_number(field, step_bytes, item_start + step_offset, true)?
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());

    for (i, field) in field_order.into_iter().enumerate() {
        sets[i] = parse_field(field, fields[i])?;
    }

    Ok(Cron {
        selected: sets,
        day_of_month_unrestricted: fields[2] == "*",
        day_of_week_unrestricted: fields[4] == "*",
    })
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if is_leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += (at.day - 1) as u32;
    ((6 + days) % 7) as u8
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| {
                (is_selected && value <= u8::MAX as usize).then_some(value as u8)
            })
            .collect()
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday(at) as usize];

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if at.day == days_in_month(at.year, at.month) => dom && dow,
            (false, false) => dom || dow,
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut candidate = after;
        loop {
            if candidate.year == 2399
                && candidate.month == 12
                && candidate.day == 31
                && candidate.hour == 23
                && candidate.minute == 59
            {
                return Ok(None);
            }

            if candidate.minute < 59 {
                candidate.minute += 1;
            } else {
                candidate.minute = 0;
                if candidate.hour < 23 {
                    candidate.hour += 1;
                } else {
                    candidate.hour = 0;
                    if candidate.day < days_in_month(candidate.year, candidate.month) {
                        candidate.day += 1;
                    } else {
                        candidate.day = 1;
                        if candidate.month < 12 {
                            candidate.month += 1;
                        } else {
                            candidate.month = 1;
                            if candidate.year == 2399 {
                                return Ok(None);
                            }
                            candidate.year += 1;
                        }
                    }
                }
            }

            if self.matches_valid(candidate) {
                return Ok(Some(candidate));
            }
        }
    }
}

#[derive(Clone, Copy)]
struct Spec {
    field: Field,
    min: u32,
    max: u32,
}

const SPECS: [Spec; 5] = [
    Spec { field: Field::Minute, min: 0, max: 59 },
    Spec { field: Field::Hour, min: 0, max: 23 },
    Spec { field: Field::DayOfMonth, min: 1, max: 31 },
    Spec { field: Field::Month, min: 1, max: 12 },

fn parse_err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn name_value(field: Field, token: &str) -> Option<u32> {
    if field == Field::Month {
        const NAMES: [&str; 12] = [
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN",
            "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
        ];
        NAMES.iter().position(|n| token.eq_ignore_ascii_case(n)).map(|i| i as u32 + 1)
    } else if field == Field::DayOfWeek {
        const NAMES: [&str; 7] = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];
        NAMES.iter().position(|n| token.eq_ignore_ascii_case(n)).map(|i| i as u32)
    } else {
        None
    }
}

fn is_digits(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn atom_value(field: Field, token: &str, offset: usize, spec: Spec) -> Result<u32, ParseError> {
    if is_digits(token) {
        if token.len() > 1 && token.as_bytes()[0] == b'0' {
            return Err(parse_err(field, offset, 1));
        }
        let mut value = 0u32;
        for b in token.bytes() {
            value = value.checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or_else(|| parse_err(field, offset, 2))?;
        }
        if value < spec.min || value > spec.max {
            return Err(parse_err(field, offset, 3));
        }
        Ok(value)
    } else {
        name_value(field, token).ok_or_else(|| parse_err(field, offset, 0))
    }
}

fn parse_field(text: &str, spec: Spec) -> Result<Vec<u8>, ParseError> {
    for (offset, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field: spec.field, offset });
        }
    }

    let width = spec.max - spec.min + 1;
    let mut selected = vec![false; width as usize];

    let mut start = 0usize;
    loop {
        let end = text[start..].find(',').map(|n| start + n).unwrap_or(text.len());
        let item = &text[start..end];
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field: spec.field, offset: start });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(parse_err(spec.field, start, 0));
        }

        let (base, step_text, step_offset) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some(&item[pos + 1..]), start + pos + 1)
        } else {
            (item, None, 0)
        };

        if base.is_empty() || step_text == Some("") {
            return Err(parse_err(spec.field, start, 0));
        }

        let hyphens = base.bytes().filter(|&b| b == b'-').count();
        let is_star = base == "*";
        let (a_text, b_text) = if is_star {
            (None, None)
        } else if hyphens == 0 {
            (Some(base), None)
        } else if hyphens == 1 {
            let pos = base.find('-').unwrap();
            let left = &base[..pos];
            let right = &base[pos + 1..];
            if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                return Err(parse_err(spec.field, start, 0));
            }
            (Some(left), Some(right))
        } else {
            return Err(parse_err(spec.field, start, 0));
        };

        let valid_atom = |s: &str| is_digits(s) || name_value(spec.field, s).is_some();
        if !is_star
            && (!valid_atom(a_text.unwrap())
                || b_text.is_some_and(|s| !valid_atom(s)))
        {
            return Err(parse_err(spec.field, start, 0));
        }

        let a = if let Some(s) = a_text {
            Some(atom_value(spec.field, s, start, spec)?)
        } else {
            None
        };

        let b_offset = if let Some(s) = b_text {
            start + base.find('-').unwrap() + 1
        } else {
            0
        };
        let b = if let Some(s) = b_text {
            Some(atom_value(spec.field, s, b_offset, spec)?)
        } else {
            None
        };

        if let (Some(av), Some(bv)) = (a, b) {
            if av > bv {
                return Err(parse_err(spec.field, b_offset, 4));
            }
        }

        let step = if let Some(st) = step_text {
            if !is_digits(st) {
                return Err(parse_err(spec.field, start, 0));
            }
            if st.len() > 1 && st.as_bytes()[0] == b'0' {
                return Err(parse_err(spec.field, step_offset, 1));
            }
            let mut n = 0u32;
            for byte in st.bytes() {
                n = n.checked_mul(10)
                    .and_then(|v| v.checked_add((byte - b'0') as u32))
                    .ok_or_else(|| parse_err(spec.field, step_offset, 2))?;
            }
            if n == 0 || n > width {
                return Err(parse_err(spec.field, step_offset, 5));
            }
            Some(n)
        } else {
            None
        };

        let (lo, hi) = if is_star {
            (spec.min, spec.max)
        } else if let Some(av) = a {
            (av, b.unwrap_or(if step.is_some() { spec.max } else { av }))
        } else {
            (spec.min, spec.max)
        };

        let increment = step.unwrap_or(1);
        let mut value = lo;
        while value <= hi {
            selected[(value - spec.min) as usize] = true;
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        if end == text.len() {
            break;
        }
        start = end + 1;
    }

    Ok(selected.iter().enumerate()
        .filter_map(|(i, yes)| yes.then_some((i as u32 + spec.min) as u8))
        .collect())
}

                    } else {
                        current.day += 1;
                    }
                } else {
                    current.hour += 1;
                }
            } else {
                current.minute += 1;
            }

            if self.matches_valid(current) {
                return Ok(Some(current));
            }
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23 && current.minute == 59
            {
                return Ok(None);
            }
        }
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}
