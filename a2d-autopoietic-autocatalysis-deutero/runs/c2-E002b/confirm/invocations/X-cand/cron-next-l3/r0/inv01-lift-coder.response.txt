#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<u8>; 5],
    day_of_month_star: bool,
    day_of_week_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u8> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    names
        .iter()
        .position(|name| name.as_bytes() == upper.as_slice())
        .map(|i| {
            if field == Field::Month {
                (i + 1) as u8
            } else {
                i as u8
            }
        })
}

fn parse_atom(
    field: Field,
    bytes: &[u8],
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    if is_step {
        if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
    } else if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    if !is_step {
        if let Some(value) = name_value(field, bytes) {
            return Ok(value as u32);
        }
    }

    if !bytes.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_item(
    field: Field,
    item: &[u8],
    item_offset: usize,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, 0)
    };

    if base.is_empty() || step_bytes.is_some_and(|s| s.is_empty()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (minimum, maximum) = bounds(field);
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if hyphens.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let wildcard = base == b"*";
    let (start_bytes, end_bytes) = if wildcard {
        (None, None)
    } else if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        (Some(&base[..hyphen]), Some(&base[hyphen + 1..]))
    } else {
        (Some(base), None)
    };

    let mut start = minimum as u32;
    let mut end = maximum as u32;
    let mut end_offset = item_offset;

    if let Some(start_part) = start_bytes {
        start = parse_atom(field, start_part, item_offset, false)?;
        if start < minimum as u32 || start > maximum as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_offset,
            });
        }
        if let Some(end_part) = end_bytes {
            end_offset = item_offset + (end_part.as_ptr() as usize - item.as_ptr() as usize);
            end = parse_atom(field, end_part, end_offset, false)?;
            if end < minimum as u32 || end > maximum as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_offset,
                });
            }
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
        } else if step_bytes.is_none() {
            end = start;
        }
    }

    let step = if let Some(bytes) = step_bytes {
        let value = parse_atom(field, bytes, step_offset, true)?;
        let width = (maximum - minimum + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];

    for (index, field) in FIELDS.iter().copied().enumerate() {
        let bytes = parts[index].as_bytes();
        for (offset, &byte) in bytes.iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut selected = [false; 60];
        let mut offset = 0usize;
        for item in bytes.split(|&b| b == b',') {
            parse_item(field, item, offset, &mut selected)?;
            offset += item.len() + 1;
        }

        let (minimum, maximum) = bounds(field);
        sets[index] = (minimum..=maximum)
            .filter(|&value| selected[value as usize])
            .collect();
    }

    Ok(Cron {
        sets,
        day_of_month_star: parts[2] == "*",
        day_of_week_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if let Some(next) = increment_minute(current) {
                current = next;
            } else {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.sets[0].contains(&at.minute)
            || !self.sets[1].contains(&at.hour)
            || !self.sets[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.sets[2].contains(&at.day);
        let dow = self.sets[4].contains(&weekday(at.year, at.month, at.day));
        match (self.day_of_month_star, self.day_of_week_star) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        at.minute += 1;
        return Some(at);
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return Some(at);
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return Some(at);
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return Some(at);
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return Some(at);
    }
    None
}
