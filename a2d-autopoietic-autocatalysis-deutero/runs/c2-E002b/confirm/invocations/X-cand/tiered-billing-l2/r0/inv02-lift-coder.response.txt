```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().and_then(|tier| tier.up_to).is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn checked_band(index: usize, units: u64, unit: u64, start: u64) -> Result<Band, Error> {
    let amount = units
        .checked_mul(unit)
        .and_then(|value| value.checked_add(start))
        .ok_or(Error::Overflow { index })?;
    Ok(Band { index, units, amount })
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        let units = match tier.up_to {
            Some(bound) => remaining.min(bound - previous),
            None => remaining,
        };

        if units > 0 {
            let band = checked_band(index, units, tier.unit, tier.start)?;
            total = total
                .checked_add(band.amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(band);
            remaining -= units;
        }

        if let Some(bound) = tier.up_to {
            previous = bound;
        }
        if remaining == 0 {
            break;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}

fn volume(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let (index, tier) = tiers
        .iter()
        .enumerate()
        .find(|(_, tier)| tier.up_to.map_or(true, |bound| quantity <= bound))
        .expect("validated tier schedule has a final unbounded tier");

    let band = checked_band(index, quantity, tier.unit, tier.start)?;
    Ok(Charge {
        amount: band.amount,
        bands: vec![band],
    })
}

fn flat(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let band = checked_band(0, quantity, unit, start)?;
    Ok(Charge {
        amount: band.amount,
        bands: vec![band],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => flat(quantity, *unit, *start),
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match price {
        Price::Flat { unit, start } => flat(quantity, *unit, *start)?,
        Price::Tiers(tiers) => match mode {
            Mode::Graduated => graduated(quantity, tiers)?,
            Mode::Volume => volume(quantity, tiers)?,
        },
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .expect("minimum adjustment cannot make total exceed minimum");

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```
