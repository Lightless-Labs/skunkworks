#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

#[derive(Clone, Debug)]
struct JobState {
    burst_index: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: 0,
            index,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut states: Vec<JobState> = jobs
        .iter()
        .map(|job| JobState {
            burst_index: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut event_pos = 0usize;
    let mut completed = 0usize;
    let mut last_dispatch_id: Option<u32> = None;

    fn admit(
        through: u64,
        events: &[Event],
        event_pos: &mut usize,
        jobs: &[Job],
        states: &mut [JobState],
        queues: &mut [std::collections::VecDeque<usize>],
    ) {
        while *event_pos < events.len() && events[*event_pos].time <= through {
            let event = events[*event_pos];
            let level = if event.kind == 0 {
                jobs[event.index].level
            } else {
                states[event.index].level
            };
            queues[level].push_back(event.index);
            *event_pos += 1;
        }
    }

    while completed < jobs.len() {
        admit(time, &events, &mut event_pos, jobs, &mut states, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_pos >= events.len() {
                break;
            }
            let next_time = events[event_pos].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = jobs[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != id) {
            time += switch_cost as u64;
        }
        admit(time, &events, &mut event_pos, jobs, &mut states, &mut queues);

        let start = time;
        let consumed = states[index]
            .remaining
            .min(quanta[level]);
        time += consumed as u64;
        states[index].remaining -= consumed;

        let burst_finished = states[index].remaining == 0;
        let mut finished = false;
        if burst_finished {
            if states[index].burst_index + 1 == jobs[index].bursts.len() {
                finished = true;
                states[index].completion = Some(time);
                completed += 1;
            }
        }

        runs.push(Run {
            id,
            start,
            end: time,
            finished,
            level,
        });

        if burst_finished && !finished {
            let io_position = states[index].burst_index + 1;
            let wake_time = time + jobs[index].bursts[io_position] as u64;
            states[index].burst_index += 2;
            states[index].remaining = jobs[index].bursts[states[index].burst_index];
            states[index].level = level.saturating_sub(1);
            events.push(Event {
                time: wake_time,
                kind: 1,
                index,
            });
            events[event_pos..].sort_by_key(|event| (event.time, event.kind, event.index));
        }

        admit(time, &events, &mut event_pos, jobs, &mut states, &mut queues);

        if !burst_finished {
            states[index].level = (level + 1).min(quanta.len() - 1);
            queues[states[index].level].push_back(index);
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap();
        let turnaround = completion - job.arrival as u64;
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;
    let end = stats.iter().map(|stat| stat.completion).max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}
