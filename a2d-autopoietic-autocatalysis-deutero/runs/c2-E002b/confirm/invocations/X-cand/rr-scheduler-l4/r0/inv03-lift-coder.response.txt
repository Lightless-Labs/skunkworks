#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone)]
struct Proc {
    id: u32,
    initial_arrival: u64,
    bursts: Vec<u32>,
    burst_index: usize,
    remaining: u32,
    level: usize,
    completion: u64,
    cpu_total: u64,
    io_total: u64,
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    kind: u8, // 0 = external arrival, 1 = I/O wake
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut procs: Vec<Proc> = jobs
        .iter()
        .map(|job| Proc {
            id: job.id,
            initial_arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            burst_index: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: 0,
            cpu_total: job
                .bursts
                .iter()
                .step_by(2)
                .map(|&value| u64::from(value))
                .sum(),
            io_total: job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&value| u64::from(value))
                .sum(),
        })
        .collect();

    let mut events = Vec::new();
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut event_pos = 0;
    let mut now = 0_u64;
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0;

    let admit_through = |limit: u64,
                         events: &[Event],
                         event_pos: &mut usize,
                         queues: &mut [std::collections::VecDeque<usize>]| {
        while *event_pos < events.len() && events[*event_pos].time <= limit {
            let event = events[*event_pos];
            queues[event.level].push_back(event.index);
            *event_pos += 1;
        }
    };

    while completed_count < jobs.len() {
        admit_through(now, &events, &mut event_pos, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_pos >= events.len() {
                break;
            }
            let next_time = events[event_pos].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().expect("nonempty queue");
        let id = procs[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != id) {
            now += u64::from(switch_cost);
        }
        admit_through(now, &events, &mut event_pos, &mut queues);

        let quantum = quanta[level];
        let consumed = procs[index].remaining.min(quantum);
        let start = now;
        now += u64::from(consumed);
        let end = now;

        procs[index].remaining -= consumed;
        let burst_finished = procs[index].remaining == 0;
        let finished = burst_finished && procs[index].burst_index + 1 == procs[index].bursts.len();

        runs.push(Run {
            id,
            start,
            end,
            finished,
            level,
        });

        // Admit all events already pending through the run endpoint.
        admit_through(end, &events, &mut event_pos, &mut queues);

        if finished {
            procs[index].completion = end;
            completed_count += 1;
        } else if burst_finished {
            let io_position = procs[index].burst_index + 1;
            let delay = u64::from(procs[index].bursts[io_position]);
            procs[index].burst_index += 2;
            procs[index].remaining = procs[index].bursts[procs[index].burst_index];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: end + delay,
                kind: 1,
                index,
                level: wake_level,
            });
            // Newly generated events participate in timestamp/kind/index ordering.
            events[event_pos..].sort_by_key(|event| (event.time, event.kind, event.index));
            admit_through(end, &events, &mut event_pos, &mut queues);
        } else {
            procs[index].level = (level + 1).min(quanta.len() - 1);
            queues[procs[index].level].push_back(index);
        }

        last_dispatch_id = Some(id);
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let proc = &procs[index];
            let turnaround = proc.completion - proc.initial_arrival;
            Stat {
                id: job.id,
                completion: proc.completion,
                turnaround,
                waiting: turnaround - proc.cpu_total - proc.io_total,
            }
        })
        .collect();

    let count = jobs.len() as u64;
    let waiting_total: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let turnaround_total: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: (waiting_total * 100 + count / 2) / count,
        mean_turnaround_x100: (turnaround_total * 100 + count / 2) / count,
        end: now,
    })
}
