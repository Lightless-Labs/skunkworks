#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(&str, u128); 6] = [
    ("thousand", 1_000),
    ("million", 1_000_000),
    ("billion", 1_000_000_000),
    ("trillion", 1_000_000_000_000),
    ("quadrillion", 1_000_000_000_000_000),
    ("quintillion", 1_000_000_000_000_000_000),
];

fn spell_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_owned();
    }

    if n < 100 {
        let tens = TENS[(n / 10 - 2) as usize];
        let units = n % 10;
        if units == 0 {
            tens.to_owned()
        } else {
            format!("{tens}-{}", SMALL[units as usize])
        }
    } else {
        let hundreds = SMALL[(n / 100) as usize];
        let remainder = n % 100;
        if remainder == 0 {
            format!("{hundreds} hundred")
        } else {
            format!("{hundreds} hundred and {}", spell_under_thousand(remainder))
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    if magnitude == 0 {
        return Ok("zero".to_owned());
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    for (scale, scale_value) in SCALES.iter().rev() {
        let group = remaining / scale_value;
        remaining %= scale_value;
        if group != 0 {
            groups.push((group as u16, Some(*scale)));
        }
    }
    if remaining != 0 {
        groups.push((remaining as u16, None));
    }

    let units_is_small = groups
        .last()
        .is_some_and(|(group, scale)| scale.is_none() && *group < 100);

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    for (index, (group, scale)) in groups.iter().enumerate() {
        if index != 0 {
            if units_is_small && index + 1 == groups.len() {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }

        result.push_str(&spell_under_thousand(*group));
        if let Some(scale) = scale {
            result.push(' ');
            result.push_str(scale);
        }
    }

    Ok(result)
}

fn word_value(word: &str) -> Option<u128> {
    if let Some(index) = SMALL.iter().position(|candidate| *candidate == word) {
        return Some(index as u128);
    }
    if let Some(index) = TENS.iter().position(|candidate| *candidate == word) {
        return Some(((index + 2) * 10) as u128);
    }
    match word {
        "hundred" | "thousand" | "million" | "billion" | "trillion"
        | "quadrillion" | "quintillion" | "and" | "minus" => Some(0),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (index, (token, position)) in tokens.iter().enumerate() {
        if *token == "minus" {
            if index != 0 {
                return Err(WordsError::InvalidWord { index: *position });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let negative = tokens.first().is_some_and(|(token, _)| *token == "minus");
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total = 0u128;
    let mut group = 0u128;

    for (token, _) in arithmetic_tokens {
        match *token {
            "and" | "zero" => {}
            "hundred" => group = group.saturating_mul(100),
            "thousand" => {
                total = total.saturating_add(group.saturating_mul(1_000));
                group = 0;
            }
            "million" => {
                total = total.saturating_add(group.saturating_mul(1_000_000));
                group = 0;
            }
            "billion" => {
                total = total.saturating_add(group.saturating_mul(1_000_000_000));
                group = 0;
            }
            "trillion" => {
                total = total.saturating_add(group.saturating_mul(1_000_000_000_000));
                group = 0;
            }
            "quadrillion" => {
                total = total.saturating_add(group.saturating_mul(1_000_000_000_000_000));
                group = 0;
            }
            "quintillion" => {
                total = total.saturating_add(group.saturating_mul(1_000_000_000_000_000_000));
                group = 0;
            }
            _ => group = group.saturating_add(word_value(token).unwrap_or(0)),
        }

        if total.saturating_add(group) > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let differing = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(value)
}
