#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn encode_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }
    if n < 100 {
        let tens = TENS[(n / 10 - 2) as usize];
        let units = n % 10;
        return if units == 0 {
            tens.to_string()
        } else {
            format!("{}-{}", tens, SMALL[units as usize])
        };
    }

    let hundreds = format!("{} hundred", SMALL[(n / 100) as usize]);
    let remainder = n % 100;
    if remainder == 0 {
        hundreds
    } else {
        format!("{} and {}", hundreds, encode_under_thousand(remainder))
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut parts = Vec::new();
    for index in (0..7).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        let mut part = encode_under_thousand(group);
        if index > 0 {
            part.push(' ');
            part.push_str(SCALES[index - 1]);
        }
        parts.push((index, group, part));
    }

    let last_index = parts.len().saturating_sub(1);
    let mut output = String::new();
    for (position, (index, group, part)) in parts.iter().enumerate() {
        if position != 0 {
            let use_and = position == last_index
                && *index == 0
                && *group < 100;
            output.push_str(if use_and { " and " } else { " " });
        }
        output.push_str(part);
    }

    if value < 0 {
        Ok(format!("minus {}", output))
    } else {
        Ok(output)
    }
}

fn word_value(word: &str) -> Option<(u128, WordKind)> {
    for (value, candidate) in SMALL.iter().enumerate() {
        if word == *candidate {
            return Some((value as u128, WordKind::Number));
        }
    }
    for (index, candidate) in TENS.iter().enumerate() {
        if word == *candidate {
            return Some((((index + 2) * 10) as u128, WordKind::Number));
        }
    }
    match word {
        "hundred" => Some((100, WordKind::Hundred)),
        "thousand" => Some((1_000, WordKind::Scale)),
        "million" => Some((1_000_000, WordKind::Scale)),
        "billion" => Some((1_000_000_000, WordKind::Scale)),
        "trillion" => Some((1_000_000_000_000, WordKind::Scale)),
        "quadrillion" => Some((1_000_000_000_000_000, WordKind::Scale)),
        "quintillion" => Some((1_000_000_000_000_000_000, WordKind::Scale)),
        "and" => Some((0, WordKind::And)),
        "minus" => Some((0, WordKind::Minus)),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((start, &input[start..index]));
            start = index + 1;
        }
    }
    tokens.push((start, &input[start..]));

    let mut kinds = Vec::with_capacity(tokens.len());
    for &(index, token) in &tokens {
        let Some((value, kind)) = word_value(token) else {
            return Err(WordsError::InvalidWord { index });
        };
        if matches!(kind, WordKind::Minus) && index != 0 {
            return Err(WordsError::InvalidWord { index });
        }
        kinds.push((value, kind));
    }

    let negative = matches!(kinds.first(), Some((_, WordKind::Minus)));
    let first = usize::from(negative);
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(value, kind) in &kinds[first..] {
        match kind {
            WordKind::Number => group += value,
            WordKind::Hundred => group *= 100,
            WordKind::Scale => {
                total += group * value;
                group = 0;
            }
            WordKind::And | WordKind::Minus => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical != input {
        let differing = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }
    Ok(signed)
}
