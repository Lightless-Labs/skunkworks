#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const ONES: [&str; 20] = [
    "", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty",
    "ninety",
];

const SCALES: [&str; 7] = [
    "", "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(ONES[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_under_hundred(rem, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

fn append_under_hundred(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(ONES[n as usize]);
    } else {
        out.push_str(TENS[(n / 10) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(ONES[(n % 10) as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let highest = (0..7).rev().find(|&i| groups[i] != 0).unwrap();
    for i in (0..=highest).rev() {
        let n = groups[i];
        if n == 0 {
            continue;
        }
        if !out.is_empty() && !(i == 0 && n < 100) {
            out.push(' ');
        }
        if i == 0 && n < 100 && highest > 0 {
            out.push_str("and ");
        }
        append_group(n, &mut out);
        if i > 0 {
            out.push(' ');
            out.push_str(SCALES[i]);
        }
    }

    Ok(out)
}

fn token_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }
    for (i, &w) in ONES.iter().enumerate().skip(1) {
        if word == w {
            return Some(i as u128);
        }
    }
    for (i, &w) in TENS.iter().enumerate().skip(2) {
        if word == w {
            return Some((i * 10) as u128);
        }
    }
    match word {
        "hundred" => Some(100),
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        "and" => Some(0),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_lowercase() || b == b' ' || b == b'-') {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b' ' || b == b'-' {
            tokens.push((&input[start..i], start));
            start = i + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, &(word, position)) in tokens.iter().enumerate() {
        if word == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: position });
            }
        } else if token_value(word).is_none() {
            return Err(WordsError::InvalidWord { index: position });
        }
    }

    let negative = tokens.first().is_some_and(|(word, _)| *word == "minus");
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for &(word, _) in arithmetic {
        match word {
            "minus" => unreachable!(),
            "and" => {}
            "hundred" => group = group.checked_mul(100).ok_or(WordsError::Range)?,
            "thousand" | "million" | "billion" | "trillion" | "quadrillion"
            | "quintillion" => {
                let scale = token_value(word).unwrap();
                total = total
                    .checked_add(group.checked_mul(scale).ok_or(WordsError::Range)?)
                    .ok_or(WordsError::Range)?;
                group = 0;
            }
            _ => group = group.checked_add(token_value(word).unwrap()).ok_or(WordsError::Range)?,
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(value)?;
    if canonical != input {
        let mismatch = canonical
            .bytes()
            .zip(bytes.iter().copied())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }
    Ok(value)
}
