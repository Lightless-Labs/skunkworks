#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&base) {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag {
                operand,
                index: 0,
            });
        }
        payload_start = hash + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Big-endian base-256 integer helpers.
fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for byte in value.iter_mut().rev() {
        let product = (*byte as u16) * (multiplier as u16) + carry;
        *byte = product as u8;
        carry = product >> 8;
    }
    while carry != 0 {
        value.insert(0, carry as u8);
        carry >>= 8;
    }
}

fn from_digits(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    value
}

fn add_values(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry != 0 {
        let a = if i > 0 {
            i -= 1;
            left[i] as u16
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            right[j] as u16
        } else {
            0
        };
        let sum = a + b + carry;
        result.push(sum as u8);
        carry = sum >> 8;
    }

    if result.is_empty() {
        result.push(0);
    }
    result.reverse();
    result
}

fn divide_small(value: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for byte in value.iter_mut() {
        let current = (remainder << 8) | *byte as u16;
        *byte = (current / divisor as u16) as u8;
        remainder = current % divisor as u16;
    }

    let first_nonzero = value.iter().position(|&byte| byte != 0);
    if let Some(index) = first_nonzero {
        if index > 0 {
            value.drain(..index);
        }
    } else {
        value.clear();
        value.push(0);
    }

    remainder as u8
}

fn render(value: &[u8], radix: u8, style: BaseStyle) -> String {
    let mut number = value.to_vec();
    if number.is_empty() {
        number.push(0);
    }

    let mut digits = Vec::new();
    if number.iter().all(|&byte| byte == 0) {
        digits.push(0);
    } else {
        while number.iter().any(|&byte| byte != 0) {
            digits.push(divide_small(&mut number, radix));
        }
        digits.reverse();
    }

    while digits.len() < style.min_width {
        digits.insert(0, 0);
    }

    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };
    let mut output = String::with_capacity(digits.len() + digits.len() / style.group.max(1));
    for (index, digit) in digits.iter().enumerate() {
        if style.group != 0 && index != 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        output.push(alphabet[*digit as usize] as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let value = from_digits(&digits, from);
    Ok(render(&value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = from_digits(&left_digits, base);
    let right_value = from_digits(&right_digits, base);
    let sum = add_values(&left_value, &right_value);
    Ok(render(&sum, base, style))
}
