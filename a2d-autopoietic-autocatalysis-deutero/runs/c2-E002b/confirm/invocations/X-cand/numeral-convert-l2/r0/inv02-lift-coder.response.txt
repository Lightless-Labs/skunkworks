#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;
    if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = tag_end + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.is_empty() {
        let offset = payload.iter().position(|&b| b == b'_').unwrap_or(0);
        return Err(BaseError::Separator {
            operand,
            index: payload_start + offset,
        });
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn to_decimal_limbs(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut decimal = vec![0u8];
    for &digit in digits {
        let mut carry = digit as u32;
        for limb in &mut decimal {
            let value = *limb as u32 * radix as u32 + carry;
            *limb = (value % 10) as u8;
            carry = value / 10;
        }
        while carry != 0 {
            decimal.push((carry % 10) as u8);
            carry /= 10;
        }
    }
    decimal
}

fn from_decimal_limbs(decimal: &[u8], radix: u8) -> Vec<u8> {
    let mut work = decimal.to_vec();
    let mut output = Vec::new();

    while work.iter().any(|&digit| digit != 0) {
        let mut remainder = 0u32;
        for digit in work.iter_mut().rev() {
            let value = remainder * 10 + *digit as u32;
            *digit = (value / radix as u32) as u8;
            remainder = value % radix as u32;
        }
        output.push(remainder as u8);
        while work.len() > 1 && work[0] == 0 {
            work.remove(0);
        }
    }

    if output.is_empty() {
        output.push(0);
    }
    output.reverse();
    output
}

fn add_decimal_limbs(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut sum = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u8;
    for index in 0..left.len().max(right.len()) {
        let value = left.get(index).copied().unwrap_or(0)
            + right.get(index).copied().unwrap_or(0)
            + carry;
        sum.push(value % 10);
        carry = value / 10;
    }
    if carry != 0 {
        sum.push(carry);
    }
    sum
}

fn render(digits: &[u8], style: BaseStyle) -> String {
    let padding = style.min_width.saturating_sub(digits.len());
    let total = padding + digits.len();
    let mut output = String::with_capacity(total + total / style.group.max(1));
    for index in 0..total {
        if style.group != 0 && index != 0 && (total - index) % style.group == 0 {
            output.push('_');
        }
        let value = if index < padding {
            0
        } else {
            digits[index - padding]
        };
        let byte = match value {
            0..=9 => b'0' + value,
            _ if style.upper => b'A' + (value - 10),
            _ => b'a' + (value - 10),
        };
        output.push(byte as char);
    }
    output
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    let decimal = to_decimal_limbs(&digits, from);
    let result = from_decimal_limbs(&decimal, to);
    Ok(render(&result, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_decimal = to_decimal_limbs(&left_digits, base);
    let right_decimal = to_decimal_limbs(&right_digits, base);
    let sum = add_decimal_limbs(&left_decimal, &right_decimal);
    let result = from_decimal_limbs(&sum, base);
    Ok(render(&result, style))
}
