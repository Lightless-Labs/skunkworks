```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Entry {
    value: String,
    deadline: Option<u64>,
}

#[derive(Clone)]
enum Stored {
    Value(Entry),
    Tombstone,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    stored: Stored,
}

#[derive(Clone)]
struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Stored>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn validate_key(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn selected_stored(&self, key: &str, tx: Option<u64>) -> Option<Stored> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(stored) = transaction.writes.get(key) {
                return Some(stored.clone());
            }
            self.data
                .get(key)?
                .iter()
                .rev()
                .find(|version| version.revision <= transaction.begin_revision)
                .map(|version| version.stored.clone())
        } else {
            self.data
                .get(key)?
                .last()
                .map(|version| version.stored.clone())
        }
    }

    fn live_entry(stored: Option<Stored>, now: u64) -> Option<Entry> {
        match stored {
            Some(Stored::Value(entry))
                if entry.deadline.map_or(true, |deadline| deadline > now) =>
            {
                Some(entry)
            }
            _ => None,
        }
    }

    fn visible_entries(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, Entry> {
        let mut keys: BTreeSet<String> = self.data.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                keys.extend(transaction.writes.keys().cloned());
            }
        }

        let mut result = BTreeMap::new();
        for key in keys {
            if let Some(entry) = Self::live_entry(self.selected_stored(&key, tx), now) {
                result.insert(key, entry);
            }
        }
        result
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|end| end <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |end| key < end)
            && prefix.map_or(true, |prefix| key.starts_with(prefix))
    }

    fn append(&mut self, key: String, stored: Stored, revision: u64) {
        self.data
            .entry(key)
            .or_default()
            .push(Version { revision, stored });
    }

    fn valid_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self
            .transactions
            .get(&id)
            .cloned()
            .ok_or(Error::UnknownTransaction(id))?;

        if !transaction.writes.is_empty() {
            for key in transaction.writes.keys() {
                if self
                    .data
                    .get(key)
                    .and_then(|versions| versions.last())
                    .is_some_and(|version| version.revision > transaction.begin_revision)
                {
                    return Err(Error::Conflict {
                        tx: id,
                        key: key.clone(),
                    });
                }
            }

            self.revision += 1;
            let revision = self.revision;
            for (key, stored) in transaction.writes {
                self.append(key, stored, revision);
            }
            self.transactions.remove(&id);
            Ok(revision)
        } else {
            self.transactions.remove(&id);
            Ok(self.revision)
        }
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_key(key)?;

        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let stored = match ttl {
            Some(0) => Stored::Tombstone,
            Some(duration) => Stored::Value(Entry {
                value: value.to_string(),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            }),
            None => Stored::Value(Entry {
                value: value.to_string(),
                deadline: None,
            }),
        };

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), stored);
        } else {
            self.revision += 1;
            self.append(key.to_string(), stored, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_key(key)?;

        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let existed = Self::live_entry(self.selected_stored(key, tx), now).is_some();
        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), Stored::Tombstone);
        } else {
            self.revision += 1;
            self.append(key.to_string(), Stored::Tombstone, self.revision);
        }
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_key(key)?;
        Ok(Self::live_entry(self.selected_stored(key, tx), now).map(|entry| entry.value))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self
            .visible_entries(tx, now)
            .values()
            .filter(|entry| entry.value == value)
            .count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;

        if let Some(cursor) = cursor {
            if cursor.start != start
                || cursor.end.as_deref() != end
                || cursor.prefix.as_deref() != prefix
                || cursor.offset != offset
                || cursor.last.is_empty()
                || !Self::matches(&cursor.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let entries = self.visible_entries(tx, now);
        let mut matching = entries.into_iter().filter(|(key, _)| {
            Self::matches(key, start, end, prefix)
                && cursor.map_or(true, |cursor| key.as_str() > cursor.last.as_str())
        });

        let mut skip = if cursor.is_some() { 0 } else { offset };
        while skip > 0 {
            if matching.next().is_none() {
                return Ok(Page {
                    items: Vec::new(),
                    next: None,
                });
            }
            skip -= 1;
        }

        let mut items = Vec::new();
        for _ in 0..limit {
            match matching.next() {
                Some((key, entry)) => items.push((key, entry.value)),
                None => break,
            }
        }

        let next = if !items.is_empty() && matching.next().is_some() {
            let last = items.last().unwrap().0.clone();
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last,
            })
        } else {
            None
        };

        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;

        let mut result = Aggregate { count: 0, sum: 0 };
        for (key, entry) in self.visible_entries(tx, now) {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            let parsed = Self::parse_number(&key, &entry.value)?;
            result.sum = result
                .sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            result.count += 1;
        }
        Ok(result)
    }

    fn parse_number(key: &str, value: &str) -> Result<i64, Error> {
        let bytes = value.as_bytes();
        let valid_syntax = if bytes == b"0" {
            true
        } else if bytes.first() == Some(&b'-') {
            bytes.len() >= 2
                && bytes[1].is_ascii_digit()
                && bytes[1] != b'0'
                && bytes[2..].iter().all(u8::is_ascii_digit)
        } else {
            !bytes.is_empty()
                && bytes[0].is_ascii_digit()
                && bytes[0] != b'0'
                && bytes[1..].iter().all(u8::is_ascii_digit)
        };

        if !valid_syntax {
            return Err(Error::InvalidNumber {
                key: key.to_string(),
                value: value.to_string(),
            });
        }
        value.parse::<i64>().map_err(|_| Error::InvalidNumber {
            key: key.to_string(),
            value: value.to_string(),
        })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_name(name)?;

        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }

        let copy: BTreeMap<String, Entry> = self
            .visible_entries(None, now)
            .into_iter()
            .collect();
        self.snapshots.insert(name.to_string(), copy);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }

        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;

        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let saved_live: BTreeMap<String, Entry> = saved
            .into_iter()
            .filter(|(_, entry)| entry.deadline.map_or(true, |deadline| deadline > now))
            .collect();
        let current_live = self.visible_entries(None, now);

        let keys: BTreeSet<String> = current_live
            .keys()
            .chain(saved_live.keys())
            .cloned()
            .collect();

        if keys.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in keys {
            let stored = saved_live
                .get(&key)
                .cloned()
                .map(Stored::Value)
                .unwrap_or(Stored::Tombstone);
            self.append(key, stored, revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;

        for versions in self.data.values_mut() {
            if versions.len() <= 1 {
                continue;
            }

            let mut keep = BTreeSet::new();
            keep.insert(versions.len() - 1);

            for transaction in self.transactions.values() {
                if let Some(index) = versions
                    .iter()
                    .enumerate()
                    .rev()
                    .find(|(_, version)| version.revision <= transaction.begin_revision)
                    .map(|(index, _)| index)
                {
                    keep.insert(index);
                }
            }

            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }

        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;

        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self
            .data
            .values()
            .flat_map(|versions| versions.iter())
            .filter(|version| matches!(version.stored, Stored::Tombstone))
            .count();
        let live_keys = self.visible_entries(None, now).len();
        let read_only_transactions = self
            .transactions
            .values()
            .filter(|transaction| transaction.read_only)
            .count();
        let staged_writes = self
            .transactions
            .values()
            .map(|transaction| transaction.writes.len())
            .sum();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}
```
