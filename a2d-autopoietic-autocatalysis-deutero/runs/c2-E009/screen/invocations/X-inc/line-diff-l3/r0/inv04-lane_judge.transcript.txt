fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.starts_with('0') {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let inner = line.strip_prefix("@@ -")?.strip_suffix(" @@")?;
    let (old, new) = inner.split_once(" +")?;
    if new.contains(' ') || old.contains(' ') {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    Some((
        parse_decimal(a)?,
        parse_decimal(b)?,
        parse_decimal(c)?,
        parse_decimal(d)?,
    ))
}

        return Vec::new();

    }

            return Err(ApplyError::BadHunkHeader { line: header_line });

                return Err(ApplyError::Misordered { line: header_line });

            new_count: d,

            lead += 1;

        let mut best_starts = Vec::new();

                        old_payloads.push(row.payload.as_str());
                    }

                    continue;

                    if s + hunk.old_count > working.len() {
                        continue;

                    if distance > options.max_offset as u128 {
                        continue;

                        .iter()
                        .enumerate()
                        .all(|(i, p)| working[s + l + i] == *p);

                        continue;

                            best_starts.clear();
                            best_starts.push(s);

                            best_starts.clear();
                            best_starts.push(s);
                            selected = Some((s, l, r));
                        }

                            if !best_starts.contains(&s) {
                                best_starts.push(s);
                            }
                            if let Some((old_s, old_l, _)) = selected {
                                if s == old_s && l < old_l {
                                    selected = Some((s, l, r));
                                }
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }
        best_starts.sort_unstable();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let (s, l, r) = selected.expect("matching choice exists");
        let body_end = hunk.rows.len() - r;
        let mut replacement = Vec::new();
        for row in &hunk.rows[l..body_end] {
            if matches!(row.kind, RowKind::Context | RowKind::Add) {
                replacement.push(row.payload.clone());
            }
        }

        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

                kind,

            });

    }

    let hunks = parse_patch(patch)?;

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut working: Vec<String> = physical_lines(source)
        .into_iter()
        .map(str::to_owned)
        .collect();
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut starts = Vec::<(usize, usize, usize)>::new();

        for left in 0..=options.fuzz.min(lead) {
            for right in 0..=options.fuzz.min(trail) {
                if left + right > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                let mut plus_payloads = Vec::new();
                let mut context_index = 0usize;
                for row in &hunk.rows {
                    if row.kind == b' ' {
                        let dropped = (context_index < left)
                            || (context_index >= lead + (trail.saturating_sub(right)));
                        context_index += 1;
                        if !dropped {
                            old_payloads.push(row.payload.as_str());
                        }
                    } else if row.kind == b'-' {
                        old_payloads.push(row.payload.as_str());
                    }
                    if row.kind == b' ' || row.kind == b'+' {
                        plus_payloads.push(row.payload.as_str());
                    }
                }

                let b = hunk.old_count;
                if b > working.len() {
                    continue;
                }
                for s in 0..=working.len() - b {
                    let dist = (s as i128 - expected).unsigned_abs();
                    if dist > options.max_offset as u128 {
                        continue;
                    }
                    let start = s + left;
                    let end = s + b - right;
                    if end < start || end - start != old_payloads.len() {
                        continue;
                    }
                    if !working[start..end]
                        .iter()
                        .map(String::as_str)
                        .eq(old_payloads.iter().copied())
                    {
                        continue;
                    }
                    let rank = (left + right, dist);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            starts.clear();
                            starts.push((s, left, right));
                        }
                        Some(old_rank) if rank < old_rank => {
                            best_rank = Some(rank);
                            starts.clear();
                            starts.push((s, left, right));
                        }
                        Some(old_rank) if rank == old_rank => {
                            starts.push((s, left, right));
                        }
                        _ => {}
                    }
                }
            }
        }

        if starts.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }
        let mut distinct: Vec<usize> = starts.iter().map(|x| x.0).collect();
        distinct.sort_unstable();
        distinct.dedup();
        if distinct.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }
        let s = distinct[0];
        let (_, left, right) = starts
            .iter()
            .filter(|x| x.0 == s)
            .min_by_key(|x| x.1)
            .copied()
            .unwrap();

        let mut replacement = Vec::new();
        let mut context_index = 0usize;
        let lead_kept_from = left;
        let trail_kept_until = lead + trail.saturating_sub(right);
        for row in &hunk.rows {
            match row.kind {
                b' ' => {
                    let dropped = context_index < lead_kept_from || context_index >= trail_kept_until;
                    context_index += 1;
                    if !dropped {
                        replacement.push(row.payload.clone());
                    }
                }
                b'+' => replacement.push(row.payload.clone()),
                b'-' => {}
                _ => unreachable!(),
            }
        }

        let replace_start = s + left;
        let replace_end = s + hunk.old_count - right;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        drift = s as i128 - (hunk.old_start as i128 + net);
        net += hunk.new_count as i128 - hunk.old_count as i128;
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

fn parse_number(s: &str) -> Option<u16> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.starts_with('0') {
        return None;
    }
    let n = s.parse::<u16>().ok()?;
    if n > 128 { None } else { Some(n) }
}

fn parse_hunk_header(s: &str) -> Option<(i64, usize, i64, usize)> {
    let rest = s.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, ending) = rest.split_once(" @@")?;
    if ending != "" {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    let a = parse_number(a)? as i64;
    let b = parse_number(b)? as usize;
    let c = parse_number(c)? as i64;
    let d = parse_number(d)? as usize;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    let old_start = if b > 0 { a - 1 } else { a };
    let new_start = if d > 0 { c - 1 } else { c };
    Some((old_start, b, new_start, d))
}

fn patch_lines(patch: &str) -> Vec<&str> {
    patch.split_terminator('\n').collect()

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let mut working: Vec<String> = Vec::new();
    if !source.is_empty() {
        if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(ApplyError::SourceFormat { byte });
        }
        if !source.ends_with('\n') {
            return Err(ApplyError::SourceMissingLf);
        }
        working = source[..source.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect();
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
    }

    if patch.is_empty() {
        return Ok(source.to_owned());
    }

    if !patch.ends_with('\n') {
        let line = patch.bytes().filter(|&b| b == b'\n').count() + 1;
        return Err(ApplyError::BadHeader { line });
    }

    let lines = patch_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut idx = 2;
    let mut prev_old_end = 0i64;
    let mut net = 0i64;

    while idx < lines.len() {
        let header_line = idx + 1;
        let Some((old_start, old_count, new_start, new_count)) =
            parse_hunk_header(lines[idx])
        else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        idx += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while idx < lines.len() && !lines[idx].starts_with("@@") {
            let line = lines[idx];
            let row_line = idx + 1;
            let bytes = line.as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            if line.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let prefix = bytes[0];
            let payload = &line[1..];
            if prefix == b' ' || prefix == b'-' {
                old_seen += 1;
            }
            if prefix == b' ' || prefix == b'+' {
                new_seen += 1;
            }
            rows.push(Row {
                prefix,
                payload: payload.to_owned(),
                line: row_line,
            });
            idx += 1;
        }

        if old_seen != old_count || new_seen != new_count {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }
        if old_start < prev_old_end
            || new_start != old_start + net
            || new_start < 0
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        prev_old_end = old_start + old_count as i64;
        net += new_count as i64 - old_count as i64;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count,
            new_start,
            new_count,
            rows,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    let mut net_applied = 0i64;
    let mut drift = 0i64;
    for (hunk_idx, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.prefix == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.prefix == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start + net_applied + drift;

        let mut best_rank: Option<(usize, i64)> = None;
        let mut best_by_start: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut expected_payloads = Vec::new();
                let mut old_index = 0usize;
                for row in &hunk.rows {
                    if row.prefix == b' ' || row.prefix == b'-' {
                        let dropped = (old_index < l)
                            || (old_index >= hunk.old_count - r);
                        if !dropped {
                            expected_payloads.push(row.payload.as_str());
                        }
                        old_index += 1;
                    }
                }

                let start_min = 0usize;
                let start_max = working.len().saturating_sub(hunk.old_count);
                if hunk.old_count > working.len() {
                    continue;
                }
                for s in start_min..=start_max {
                    let distance = (s as i64 - expected).abs();
                    if distance > options.max_offset as i64 {
                        continue;
                    }
                    if s + l + expected_payloads.len() > working.len() {
                        continue;
                    }
                    if expected_payloads.iter().enumerate().all(|(i, p)| {
                        working[s + l + i].as_str() == *p
                    }) {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best_by_start.clear();
                                best_by_start.push((s, l, r));
                            }
                            Some(old_rank) if rank < old_rank => {
                                best_rank = Some(rank);
                                best_by_start.clear();
                                best_by_start.push((s, l, r));
                            }
                            Some(old_rank) if rank == old_rank => {
                                best_by_start.push((s, l, r));
                            }
                            _ => {}
                        }
                    }
                }
            }
        }

        if best_by_start.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_idx + 1 });
        }

        best_by_start.sort_by_key(|&(s, l, _)| (s, l));
        let mut starts: Vec<usize> = best_by_start.iter().map(|x| x.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_idx + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        let s = starts[0];
        let (_, l, r) = best_by_start
            .iter()
            .filter(|&&(start, _, _)| start == s)
            .min_by_key(|&&(_, l, _)| l)
            .copied()
            .unwrap();

        let mut replacement = Vec::new();
        let mut old_index = 0usize;
        for row in &hunk.rows {
            match row.prefix {
                b' ' | b'-' => {
                    let dropped = old_index < l || old_index >= hunk.old_count - r;
                    old_index += 1;
                    if dropped {
                        continue;
                    }
                    if row.prefix == b' ' || row.prefix == b'+' {
                        replacement.push(row.payload.clone());
                    }
                }
                b'+' => replacement.push(row.payload.clone()),
                _ => {}
            }
        }

        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net_applied;
        net_applied += hunk.new_count as i64 - hunk.old_count as i64;
        drift = s as i64 - (hunk.old_start + previous_net);
    }

    let mut out = String::new();
    for line in working {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    // Validate patch termination before inspecting its syntax.
    if !patch.is_empty() && !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let patch_lines = physical_lines(patch);
    let mut hunks = Vec::new();

    if !patch.is_empty() {
        for (idx, expected) in ["--- before", "+++ after"].iter().enumerate() {
            if patch_lines.get(idx).copied() != Some(*expected) {
                return Err(ApplyError::BadHeader { line: idx + 1 });
            }
        }

        if patch_lines.len() <= 2 {
            return Err(ApplyError::BadHunkHeader { line: 3 });
        }

        let mut i = 2;
        let mut previous_old_end: i128 = 0;
        let mut cumulative_delta: i128 = 0;

        while i < patch_lines.len() {
            let header_line = i + 1;
            let Some((a, b, c, d)) = parse_hunk_header(patch_lines[i]) else {
                return Err(ApplyError::BadHunkHeader { line: header_line });
            };
            let old_start = old_full_start(a, b);
            let new_start = new_full_start(c, d);

            i += 1;
            let mut rows = Vec::new();
            let mut old_seen = 0usize;
            let mut new_seen = 0usize;
            while i < patch_lines.len() && !patch_lines[i].starts_with("@@") {
                let line = patch_lines[i];
                let row_line = i + 1;
                if line.is_empty() {
                    return Err(ApplyError::BadBody { line: row_line });
                }
                let payload = &line[1..];
                if payload.as_bytes().contains(&b'\r') {
                    return Err(ApplyError::BadBody { line: row_line });
                }
                match line.as_bytes()[0] {
                    b' ' => {
                        rows.push(Row::Context(payload.to_string()));
                        old_seen += 1;
                        new_seen += 1;
                    }
                    b'-' => {
                        rows.push(Row::Remove(payload.to_string()));
                        old_seen += 1;
                    }
                    b'+' => {
                        rows.push(Row::Add(payload.to_string()));
                        new_seen += 1;
                    }
                    _ => return Err(ApplyError::BadBody { line: row_line }),
                }
                i += 1;
            }

            if old_seen != b || new_seen != d {
                return Err(ApplyError::CountMismatch { line: header_line });
            }
            if !rows.iter().any(|r| matches!(r, Row::Add(_) | Row::Remove(_))) {
                return Err(ApplyError::BadBody { line: header_line });
            }

            if old_start < previous_old_end || new_start != old_start + cumulative_delta {
                return Err(ApplyError::Misordered { line: header_line });
            }
            previous_old_end = old_start + b as i128;
            cumulative_delta += d as i128 - b as i128;

            hunks.push(Hunk {
                line: header_line,
                old_start: old_start as usize,
                old_count: b,
                new_start: new_start as usize,
                new_count: d,
                rows,
            });

            if hunks.len() == 17 {
                return Err(ApplyError::TooLarge);
            }
        }
    }

    // Source validation follows complete patch parsing.
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }

    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(str::to_string)
            .collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_idx, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| matches!(r, Row::Context(_))).count();
        let trail = hunk
            .rows
            .iter()
            .rev()
            .take_while(|r| matches!(r, Row::Context(_)))
            .count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut best: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }

                let mut expected_payloads = Vec::new();
                let mut context_position = 0usize;
                for row in &hunk.rows {
                    match row {
                        Row::Context(payload) => {
                            let is_dropped = (context_position < l)
                                || (context_position >= lead + hunk
                                    .rows
                                    .iter()
                                    .rev()
                                    .take_while(|x| matches!(x, Row::Context(_)))
                                    .count()
                                    && context_position >= hunk.old_count - r);
                            context_position += 1;
                            if !is_dropped {
                                expected_payloads.push(payload.as_str());
                            }
                        }
                        Row::Remove(payload) => expected_payloads.push(payload.as_str()),
                        Row::Add(_) => {}
                    }
                }

                let match_len = hunk.old_count - l - r;
                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    if abs_distance(s as i128, expected) > options.max_offset as u128 {
                        continue;
                    }
                    if expected_payloads.len() != match_len {
                        continue;
                    }
                    let actual = &working[s + l..s + hunk.old_count - r];
                    if actual.iter().map(String::as_str).eq(expected_payloads.iter().copied()) {
                        let rank = (l + r, abs_distance(s as i128, expected));
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(current) if rank < current => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(current) if rank == current => best.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
            }
        }

        if best.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_idx + 1 });
        }

        best.sort_by_key(|&(s, l, _)| (s, l));
        let mut starts: Vec<usize> = best.iter().map(|x| x.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_idx + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        let (s, l, r) = *best.iter().min_by_key(|&&(_, l, _)| l).unwrap();

        let mut replacement = Vec::new();
        let mut context_position = 0usize;
        for row in &hunk.rows {
            match row {
                Row::Context(payload) => {
                    let is_dropped = context_position < l
                        || context_position >= hunk.old_count - r;
                    context_position += 1;
                    if !is_dropped {
                        replacement.push(payload.clone());
                    } else if matches!(row, Row::Context(_)) {
                        // Dropped context is retained from the source slice below.
                    }
                }
                Row::Add(payload) => replacement.push(payload.clone()),
                Row::Remove(_) => {}
            }
        }

        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        let mut full_replacement = working[replace_start..replace_end].to_vec();
        let mut output_cursor = 0usize;
        for row in &hunk.rows {
            match row {
                Row::Context(payload) => {
                    let position = output_cursor;
                    output_cursor += 1;
                    if position < l || position >= hunk.old_count - r {
                        // Keep corresponding source context.
                    } else {
                        full_replacement.push(payload.clone());
                    }
                }
                Row::Add(payload) => full_replacement.push(payload.clone()),
                Row::Remove(_) => {}
            }
        }

        // Build the trimmed replacement directly in body order.
        let mut trimmed_replacement = Vec::new();
        let mut leading_context = 0usize;
        let trailing_start = hunk.old_count - r;
        let mut old_position = 0usize;
        for row in &hunk.rows {
            match row {
                Row::Context(payload) => {
                    let keep = old_position >= l && old_position < trailing_start;
                    old_position += 1;
                    if keep {
                        trimmed_replacement.push(payload.clone());
                    } else if old_position <= l {
                        leading_context += 1;
                    }
                }
                Row::Remove(_) => old_position += 1,
                Row::Add(payload) => trimmed_replacement.push(payload.clone()),
            }
        }
        let _ = (replacement, full_replacement, leading_context);
        working.splice(replace_start..replace_end, trimmed_replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut result = String::new();
    for line in working {
        result.push_str(&line);
        result.push('\n');
    }
    Ok(result)
}
