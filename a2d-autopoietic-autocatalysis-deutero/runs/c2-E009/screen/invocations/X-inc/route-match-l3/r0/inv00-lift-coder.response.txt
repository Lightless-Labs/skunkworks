#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn name_valid(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(src: &str, base: usize) -> Result<String, usize> {
    let bytes = src.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if b.is_ascii() && unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }

    let mut pieces: Vec<(usize, &str)> = Vec::new();
    let mut start = 1;
    let mut trailing = false;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                trailing = true;
                break;
            }
            pieces.push((start, &pattern[start..i]));
            start = i + 1;
        }
    }
    if !trailing {
        pieces.push((start, &pattern[start..]));
    }

    let mut segments = Vec::new();
    let mut rank = Vec::new();
    let mut names = Vec::<String>::new();

    for (idx, (at, piece)) in pieces.iter().enumerate() {
        if piece.is_empty() {
            return Err(Error::InvalidPattern { at: *at });
        }

        let first = piece.as_bytes()[0];
        if first == b':' || first == b'*' {
            let name = &piece[1..];
            if !name_valid(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: *at });
            }
            if first == b'*' && idx + 1 != pieces.len() {
                return Err(Error::InvalidPattern { at: *at });
            }
            names.push(name.to_owned());
            if first == b':' {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            } else {
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            }
        } else {
            let decoded = decode_segment(piece, *at)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }
    Ok((segments, trailing, rank))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let mut pieces: Vec<(usize, &str)> = Vec::new();
    let mut start = 1;
    let mut trailing = false;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                trailing = true;
                break;
            }
            pieces.push((start, &path[start..i]));
            start = i + 1;
        }
    }
    if !trailing {
        pieces.push((start, &path[start..]));
    }

    let mut decoded = Vec::new();
    for (at, piece) in pieces {
        if piece.is_empty() {
            return Err(Error::InvalidPath { at });
        }
        decoded.push(decode_segment(piece, at).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((decoded, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing, rank) = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head)?;

        let mut candidates = Vec::new();
        for (id, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }

            let mut params = Vec::new();
            let mut pos = 0;
            let mut matched = true;

            for segment in &route.segments {
                match segment {
                    Segment::Static(value) => {
                        if request_segments.get(pos) != Some(value) {
                            matched = false;
                            break;
                        }
                        pos += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = request_segments.get(pos) {
                            params.push((name.clone(), value.clone()));
                            pos += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = request_segments[pos..].join("/");
                        params.push((name.clone(), value));
                        pos = request_segments.len();
                        break;
                    }
                }
            }

            if matched && pos == request_segments.len() {
                candidates.push((id, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_has_route = method == "HEAD"
            && candidates
                .iter()
                .any(|(id, _)| self.routes[*id].method == "HEAD");
        let effective_method = if method == "HEAD" && !head_has_route {
            "GET"
        } else {
            method
        };

        let eligible: Vec<(usize, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(id, _)| self.routes[*id].method == effective_method)
            .cloned()
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(id, _)| self.routes[*id].method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut best_id = eligible[0].0;
        for (id, _) in eligible.iter().skip(1) {
            let left = &self.routes[*id].rank;
            let right = &self.routes[best_id].rank;
            if left > right {
                best_id = *id;
            }
        }
        let params = eligible
            .into_iter()
            .find(|(id, _)| *id == best_id)
            .expect("winner is eligible")
            .1;

        let route = &self.routes[best_id];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: best_id,
            params,
            redirect,
        })
    }
}
