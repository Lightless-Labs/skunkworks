use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, Debug, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone, Debug)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(s: &str) -> bool {
    let mut b = s.bytes();
    matches!(b.next(), Some(b'a'..=b'z'))
        && b.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == b'_' || c == b'-')
}

fn valid_capture_name(s: &str) -> bool {
    let mut b = s.bytes();
    matches!(b.next(), Some(b'a'..=b'z'))
        && b.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn parse_static(raw: &[u8], start: usize, pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let mut out = Vec::with_capacity(raw.len());
    let mut i = 0;
    while i < raw.len() {
        let b = raw[i];
        if b == b'%' {
            if i + 2 >= raw.len() {
                return Err(invalid(start + i));
            }
            let (Some(h), Some(l)) = (hex_upper(raw[i + 1]), hex_upper(raw[i + 2])) else {
                return Err(invalid(start + i));
            };
            let decoded = h * 16 + l;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(invalid(start + i));
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(invalid(start + i));
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let bad = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(bad(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1;

    while start < end {
        let rel_end = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);
        if rel_end == start {
            return Err(bad(start));
        }
        let raw = &bytes[start..rel_end];

        if pattern && (raw[0] == b':' || raw[0] == b'*') {
            let wildcard = raw[0] == b'*';
            let name = std::str::from_utf8(&raw[1..]).unwrap_or("");
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(bad(start));
            }
            if wildcard && rel_end != end {
                return Err(bad(start));
            }
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = parse_static(raw, start, pattern)?;
            segments.push(Segment::Static(decoded));
        }

        if rel_end == end {
            break;
        }
        start = rel_end + 1;
        if start == end {
            break;
        }
    }

    if segments.is_empty() {
        return Err(bad(1));
    }
    Ok((segments, trailing))
}

fn shapes_conflict(a: &Route, b_segments: &[Segment], trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != b_segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != trailing {
        return false;
    }
    a.segments.iter().zip(b_segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn route_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for s in segments {
        match s {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, routes: Vec::new() }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }
        if let Some(route) = self.routes.iter().find(|r| {
            r.method == method && shapes_conflict(r, &segments, trailing, self.policy)
        }) {
            return Err(Error::Conflict { with: route.id });
        }
        let id = self.routes.len();
        let rank = route_rank(&segments);
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request, trailing) = parse_path(head, false)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &request) {
                candidates.push((route, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_exists = method == "HEAD"
            && candidates.iter().any(|(r, _)| r.method == "HEAD");
        let wanted = if method == "HEAD" && !head_exists { "GET" } else { method };
        let eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(r, _)| r.method == wanted)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = self
                .routes
                .iter()
                .filter(|route| {
                    route_path_matches(route, &request)
                        && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
                })
                .map(|route| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (route, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id)))
            .unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect
            && !route.segments.is_empty()
            && route.trailing != trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };
        Ok(Match { route: route.id, params, redirect })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = HashSet::new();
        for (key, _) in params {
            if !supplied.insert(*key) {
                return Err(Error::DuplicateParam { name: (*key).to_owned() });
            }
        }

        let required: Vec<&str> = route.segments.iter().filter_map(|s| match s {
            Segment::Param(n) | Segment::Wildcard(n) => Some(n.as_str()),
            Segment::Static(_) => None,
        }).collect();

        for key in &required {
            if !params.iter().any(|(k, _)| k == key) {
                return Err(Error::MissingParam { name: (*key).to_owned() });
            }
        }

        let mut extras: Vec<&str> = params.iter()
            .map(|(k, _)| *k)
            .filter(|k| !required.contains(k))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam { name: (*key).to_owned() });
        }

        for key in &required {
            let value = params.iter().find(|(k, _)| k == key).unwrap().1;
            if !valid_build_value(value, matches!(
                route.segments.iter().find(|s| match s {
                    Segment::Param(n) | Segment::Wildcard(n) => n == key,
                    Segment::Static(_) => false,
                }),
                Some(Segment::Wildcard(_))
            )) {
                return Err(Error::InvalidValue { name: (*key).to_owned() });
            }
        }

        let mut output = String::new();
        for segment in &route.segments {
            output.push('/');
            match segment {
                Segment::Static(s) => output.push_str(&encode_component(s)),
                Segment::Param(n) => {
                    let v = params.iter().find(|(k, _)| *k == n).unwrap().1;
                    output.push_str(&encode_component(v));
                }
                Segment::Wildcard(n) => {
                    let v = params.iter().find(|(k, _)| *k == n).unwrap().1;
                    if !v.is_empty() {
                        for (i, part) in v.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part));
                        }
                    }
                }
            }
        }
        if route.segments.is_empty() {
            output.push('/');
        } else if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn match_segments(pattern: &[Segment], request: &[Segment]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;
    for segment in pattern {
        match segment {
            Segment::Wildcard(name) => {
                let rest = request[i..].iter().map(|s| match s {
                    Segment::Static(v) => v.as_str(),
                    _ => "",
                }).collect::<Vec<_>>().join("/");
                params.push((name.clone(), rest));
                return Some(params);
            }
            Segment::Static(expected) => match request.get(i) {
                Some(Segment::Static(actual)) if expected == actual => {}
                _ => return None,
            },
            Segment::Param(name) => match request.get(i) {
                Some(Segment::Static(value)) => params.push((name.clone(), value.clone())),
                _ => return None,
            },
        }
        i += 1;
    }
    if i == request.len() { Some(params) } else { None }
}

fn route_path_matches(route: &Route, request: &[Segment]) -> bool {
    match_segments(&route.segments, request).is_some()
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
        return false;
    }
    if wildcard {
        value.is_empty() || value.split('/').all(|part| !part.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn encode_component(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}
