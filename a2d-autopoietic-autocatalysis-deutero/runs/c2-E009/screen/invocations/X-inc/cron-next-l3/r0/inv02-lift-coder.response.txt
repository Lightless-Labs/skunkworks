#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_width(field: Field) -> usize {
    let (min, max) = domain(field);
    (max - min + 1) as usize
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(field: Field, bytes: &[u8], start: usize, end: usize) -> Result<u32, ParseError> {
    let part = &bytes[start..end];
    if part.len() > 1 && part[0] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in part {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn named_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[&[u8]] = match field {
        Field::Month => &[
            b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN", b"JUL", b"AUG", b"SEP", b"OCT",
            b"NOV", b"DEC",
        ],
        Field::DayOfWeek => &[b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT"],
        _ => return None,
    };

    if bytes.len() != 3 {
        return None;
    }
    let upper = [
        bytes[0].to_ascii_uppercase(),
        bytes[1].to_ascii_uppercase(),
        bytes[2].to_ascii_uppercase(),
    ];
    names
        .iter()
        .position(|name| **name == upper)
        .map(|index| {
            if field == Field::Month {
                index as u32 + 1
            } else {
                index as u32
            }
        })
}

fn parse_atom(
    field: Field,
    bytes: &[u8],
    start: usize,
    end: usize,
) -> Result<(u32, bool), ParseError> {
    let atom = &bytes[start..end];
    if atom.iter().all(u8::is_ascii_digit) {
        return Ok((parse_number(field, bytes, start, end)?, true));
    }
    if let Some(value) = named_value(field, atom) {
        return Ok((value, false));
    }
    Err(syntax_error(field, start))
}

fn parse_item(
    field: Field,
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let item = &bytes[item_start..item_end];
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(syntax_error(field, item_start));
    }

    let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
        if slash == 0 || slash + 1 == item.len() {
            return Err(syntax_error(field, item_start));
        }
        (slash, Some(item_start + slash + 1))
    } else {
        (item.len(), None)
    };

    let base = &item[..base_end];
    let (is_star, range_dash) = if base == b"*" {
        (true, None)
    } else {
        let dashes: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if dashes.len() > 1 {
            return Err(syntax_error(field, item_start));
        }
        if let Some(&dash) = dashes.first() {
            if dash == 0 || dash + 1 == base.len() {
                return Err(syntax_error(field, item_start));
            }
            (false, Some(dash))
        } else {
            (false, None)
        }
    };

    let mut start_value = 0u32;
    let mut end_value = 0u32;
    let mut numeric_start = false;
    let mut numeric_end = false;

    if is_star {
        let (min, max) = domain(field);
        start_value = min;
        end_value = max;
    } else if let Some(dash) = range_dash {
        let (s, s_numeric) = parse_atom(field, base, 0, dash)?;
        let (e, e_numeric) = parse_atom(field, base, dash + 1, base.len())?;
        start_value = s;
        end_value = e;
        numeric_start = s_numeric;
        numeric_end = e_numeric;

        let (min, max) = domain(field);
        if numeric_start && !(min..=max).contains(&start_value) {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_start,
            });
        }
        if numeric_end && !(min..=max).contains(&end_value) {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_start + dash + 1,
            });
        }
        if start_value > end_value {
            return Err(ParseError::ReversedRange {
                field,
                offset: item_start + dash + 1,
            });
        }
    } else {
        let (value, numeric) = parse_atom(field, base, 0, base.len())?;
        start_value = value;
        numeric_start = numeric;
        let (min, max) = domain(field);
        if numeric_start && !(min..=max).contains(&start_value) {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_start,
            });
        }
        if step_start.is_none() {
            end_value = start_value;
        } else {
            end_value = max;
        }
    }

    let step = if let Some(step_start) = step_start {
        let step_end = item.len();
        if !item[step_start - item_start..]
            .iter()
            .all(u8::is_ascii_digit)
        {
            return Err(syntax_error(field, item_start));
        }
        let value = parse_number(field, bytes, step_start, item_start + step_end)?;
        let width = (domain(field).1 - domain(field).0 + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_start,
            });
        }
        value
    } else {
        1
    };

    let (min, _) = domain(field);
    let mut value = start_value;
    while value <= end_value {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; field_width(field)];
    let mut start = 0usize;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(field, bytes, start, i, &mut selected)?;
            start = i + 1;
        }
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (index, field) in FIELDS.iter().copied().enumerate() {
        sets[index] = parse_field(field, parts[index])?;
    }

    Ok(Cron {
        selected: sets,
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = domain(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &allowed)| allowed.then_some(min as u8 + i as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            current = advance_minute(current);
            if current.year > 2399 {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let minute = self.selected[0][at.minute as usize];
        let hour = self.selected[1][at.hour as usize];
        let day_of_month = self.selected[2][(at.day - 1) as usize];
        let month = self.selected[3][(at.month - 1) as usize];
        let day_of_week = self.selected[4][weekday(at.year, at.month, at.day) as usize];

        let day_matches = match self.unrestricted_days {
            [true, true] => true,
            [false, true] => day_of_month,
            [true, false] => day_of_week,
            [false, false] => {
                let last_day = days_in_month(at.year, at.month);
                if at.day == last_day {
                    day_of_month && day_of_week
                } else {
                    day_of_month || day_of_week
                }
            }
        };

        minute && hour && month && day_matches
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
