#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Clone, Copy)]
struct Domain {
    min: u32,
    max: u32,
}

impl Domain {
    fn width(self) -> u32 {
        self.max - self.min + 1
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: fn(Field, usize) -> ParseError) -> ParseError {
    kind(field, offset)
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_decimal(bytes: &[u8], start: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }
    let mut value = 0u32;
    for &b in bytes {
        let digit = (b - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn month_name(bytes: &[u8]) -> Option<u32> {
    const NAMES: [&[u8]; 12] = [
        b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN",
        b"JUL", b"AUG", b"SEP", b"OCT", b"NOV", b"DEC",
    ];
    NAMES.iter().position(|n| bytes.eq_ignore_ascii_case(n)).map(|i| (i + 1) as u32)
}

fn weekday_name(bytes: &[u8]) -> Option<u32> {
    const NAMES: [&[u8]; 7] = [
        b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT",
    ];
    NAMES.iter().position(|n| bytes.eq_ignore_ascii_case(n)).map(|i| i as u32)
}

fn parse_atom(bytes: &[u8], offset: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(invalid_syntax(field, offset));
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        let value = parse_decimal(bytes, offset, field)?;
        let d = domain(field);
        if value < d.min || value > d.max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok(value);
    }
    let named = match field {
        Field::Month => month_name(bytes),
        Field::DayOfWeek => weekday_name(bytes),
        _ => None,
    };
    named.ok_or_else(|| invalid_syntax(field, offset))
}

fn parse_step(bytes: &[u8], offset: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(invalid_syntax(field, offset));
    }
    let value = parse_decimal(bytes, offset, field)?;
    if value == 0 || value > domain(field).width() {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_item(
    item: &[u8],
    item_start: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset: item_start });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, item_start));
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), Some(item_start + slash + 1))
    } else {
        (item, None, None)
    };

    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if hyphens.len() > 1 {
        return Err(invalid_syntax(field, item_start));
    }

    let is_star = base == b"*";
    let (start_bytes, end_bytes, range_pos) = if is_star {
        (None, None, None)
    } else if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(invalid_syntax(field, item_start));
        }
        if base[..hyphen].contains(&b'*') || base[hyphen + 1..].contains(&b'*') {
            return Err(invalid_syntax(field, item_start));
        }
        (
            Some(&base[..hyphen]),
            Some(&base[hyphen + 1..]),
            Some(item_start + hyphen + 1),
        )
    } else {
        if base.contains(&b'*') && !is_star {
            return Err(invalid_syntax(field, item_start));
        }
        (Some(base), None, None)
    };

    // Validate base shapes before interpreting any numbers.
    let valid_atom_shape = |bytes: &[u8]| {
        !bytes.is_empty()
            && (bytes.iter().all(u8::is_ascii_digit)
                || bytes.iter().all(u8::is_ascii_alphabetic))
    };
    if !is_star {
        if let Some(end) = end_bytes {
            if !valid_atom_shape(start_bytes.unwrap())
                || !valid_atom_shape(end)
            {
                return Err(invalid_syntax(field, item_start));
            }
        } else if !valid_atom_shape(start_bytes.unwrap()) {
            return Err(invalid_syntax(field, item_start));
        }
    }
    if let Some(step) = step_bytes {
        if step.is_empty() || !step.iter().all(u8::is_ascii_digit) {
            return Err(invalid_syntax(field, item_start));
        }
    }

    let step = if let (Some(bytes), Some(offset)) = (step_bytes, step_offset) {
        Some(parse_step(bytes, offset, field)?)
    } else {
        None
    };

    let d = domain(field);
    let (start, end) = if is_star {
        (d.min, d.max)
    } else {
        let start = parse_atom(start_bytes.unwrap(), item_start, field)?;
        if let Some(end_bytes) = end_bytes {
            let end = parse_atom(end_bytes, range_pos.unwrap(), field)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: range_pos.unwrap(),
                });
            }
            (start, end)
        } else if step.is_some() {
            (start, d.max)
        } else {
            (start, start)
        }
    };

    let increment = step.unwrap_or(1);
    let mut value = start;
    loop {
        selected[(value - d.min) as usize] = true;
        match value.checked_add(increment) {
            Some(next) if next <= end => value = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let d = domain(field);
    let mut selected = vec![false; d.width() as usize];
    let mut start = 0usize;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b',' {
            parse_item(&bytes[start..i], start, field, &mut selected)?;
            start = i + 1;
        }
    }
    parse_item(&bytes[start..], start, field, &mut selected)?;
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed = Vec::with_capacity(5);
    for (part, field) in parts.iter().zip(fields) {
        parsed.push(parse_field(part, field)?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().unwrap(),
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let d = domain(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some((i as u32 + d.min) as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn has(&self, field: Field, value: u8) -> bool {
        let d = domain(field);
        self.selected[field_index(field)][(value as u32 - d.min) as usize]
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.has(Field::Minute, at.minute)
            || !self.has(Field::Hour, at.hour)
            || !self.has(Field::Month, at.month)
        {
            return false;
        }

        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return true;
    }
    false
}
