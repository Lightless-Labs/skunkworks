#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Parameter(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let route = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let segments = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if route.segments.len() != segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_segment, request_segment) in route.segments.iter().zip(&segments) {
                match pattern_segment {
                    Segment::Static(value) if value == request_segment => {}
                    Segment::Static(_) => {
                        matches = false;
                        break;
                    }
                    Segment::Parameter(name) => {
                        params.push((name.clone(), request_segment.clone()));
                    }
                }
            }

            if !matches {
                continue;
            }

            let replace = best
                .as_ref()
                .is_none_or(|(best_id, _)| route.rank > self.routes[*best_id].rank);
            if replace {
                best = Some((route_id, params));
            }
        }

        best.map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(pattern: &str) -> Result<Route, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();

    if pattern != "/" {
        let bytes = pattern.as_bytes();
        let mut start = 1;

        for end in (1..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b'/') {
            if end == start {
                if end == bytes.len() {
                    return Err(Error::InvalidPattern { at: end - 1 });
                }
                return Err(Error::InvalidPattern { at: start });
            }

            let raw = &pattern[start..end];
            if raw.starts_with(':') {
                let name = &raw[1..];
                if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                    return Err(Error::InvalidPattern { at: start });
                }
                segments.push(Segment::Parameter(name.to_owned()));
            } else if raw.starts_with('*') {
                return Err(Error::InvalidPattern { at: start });
            } else {
                let decoded = decode_segment(raw, start)
                    .map_err(|at| Error::InvalidPattern { at })?;
                segments.push(Segment::Static(decoded));
            }

            if end < bytes.len() {
                start = end + 1;
                if start == bytes.len() {
                    return Err(Error::InvalidPattern { at: bytes.len() - 1 });
                }
            }
        }
    }

    let rank = segments
        .iter()
        .map(|segment| match segment {
            Segment::Static(_) => 2,
            Segment::Parameter(_) => 1,
        })
        .collect();

    Ok(Route { segments, rank })
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;

    for end in (1..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b'/') {
        if end == start {
            if end == bytes.len() {
                return Err(Error::InvalidPath { at: end - 1 });
            }
            return Err(Error::InvalidPath { at: start });
        }

        let raw = &path[start..end];
        let decoded =
            decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);

        if end < bytes.len() {
            start = end + 1;
            if start == bytes.len() {
                return Err(Error::InvalidPath { at: bytes.len() - 1 });
            }
        }
    }

    Ok(segments)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }

            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segments are ASCII"))
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}
