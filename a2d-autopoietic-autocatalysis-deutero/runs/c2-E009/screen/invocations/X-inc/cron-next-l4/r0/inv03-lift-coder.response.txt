use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [BTreeSet<u8>; 5],
    dom_unrestricted: bool,
    dow_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn err(field: Field, offset: usize, make: fn(Field, usize) -> ParseError) -> ParseError {
    make(field, offset)
}

fn syntax(field: Field, offset: usize) -> ParseError {
    err(field, offset, |field, offset| ParseError::InvalidSyntax { field, offset })
}

fn number(
    s: &str,
    field: Field,
    offset: usize,
    step: bool,
) -> Result<u8, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut n: u32 = 0;
    for b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if step {
        let (_, max) = domain(field);
        let width = max as u32 + 1 - domain(field).0 as u32;
        if n == 0 || n > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = domain(field);
        if n < min as u32 || n > max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(n as u8)
}

fn named_value(s: &str, field: Field) -> Option<u8> {
    let upper = s.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn atom(
    s: &str,
    field: Field,
    offset: usize,
) -> Result<(u8, bool), ParseError> {
    if s.as_bytes().iter().all(u8::is_ascii_digit) && !s.is_empty() {
        return Ok((number(s, field, offset, false)?, true));
    }
    if s.as_bytes().iter().all(u8::is_ascii_alphabetic) {
        if let Some(v) = named_value(s, field) {
            return Ok((v, false));
        }
    }
    Err(syntax(field, offset))
}

fn parse_field(text: &str, field: Field) -> Result<BTreeSet<u8>, ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = domain(field);
    let mut selected = BTreeSet::new();
    let mut start = 0usize;

    for item in text.split(',') {
        let item_offset = start;
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_offset,
            });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(syntax(field, item_offset));
        }
        let (base, step_text) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((&item[pos + 1..], item_offset + pos + 1)))
        } else {
            (item, None)
        };

        let (is_star, range_parts) = if base == "*" {
            (true, None)
        } else {
            let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
            if hyphen_count > 1 {
                return Err(syntax(field, item_offset));
            }
            if hyphen_count == 1 {
                let p = base.find('-').unwrap();
                let left = &base[..p];
                let right = &base[p + 1..];
                if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                    return Err(syntax(field, item_offset));
                }
                (false, Some((left, item_offset, right, item_offset + p + 1)))
            } else {
                (false, None)
            }
        };

        let step = if let Some((s, off)) = step_text {
            if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
                return Err(syntax(field, item_offset));
            }
            Some(number(s, field, off, true)?)
        } else {
            None
        };

        if is_star {
            let step = step.unwrap_or(max - min + 1);
            let mut n = min;
            while n <= max {
                selected.insert(n);
                match n.checked_add(step) {
                    Some(next) => n = next,
                    None => break,
                }
            }
        } else if let Some((left, left_off, right, right_off)) = range_parts {
            let (a, a_numeric) = atom(left, field, left_off)?;
            let (b, b_numeric) = atom(right, field, right_off)?;
            if a_numeric && b_numeric {
                // Numeric range checks are performed in endpoint order by atom().
            }
            if a > b {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: right_off,
                });
            }
            let step = step.unwrap_or(1);
            let mut n = a;
            while n <= b {
                selected.insert(n);
                match n.checked_add(step) {
                    Some(next) => n = next,
                    None => break,
                }
            }
        } else {
            let (a, _) = atom(base, field, item_offset)?;
            let end = if step.is_some() { max } else { a };
            let step = step.unwrap_or(1);
            let mut n = a;
            while n <= end {
                selected.insert(n);
                match n.checked_add(step) {
                    Some(next) => n = next,
                    None => break,
                }
            }
        }

        start += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.starts_with('@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let pieces: Vec<&str> = expanded.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut values: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    for i in 0..5 {
        values[i] = parse_field(pieces[i], FIELDS[i])?;
    }

    Ok(Cron {
        values,
        dom_unrestricted: pieces[2] == "*",
        dow_unrestricted: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.values[field_index(field)].iter().copied().collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut d = after;
        loop {
            if d > after && self.matches_valid(d) {
                return Ok(Some(d));
            }
            match next_minute(d) {
                Some(n) => d = n,
                None => return Ok(None),
            }
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut d = before;
        loop {
            if d < before && self.matches_valid(d) {
                return Ok(Some(d));
            }
            match previous_minute(d) {
                Some(n) => d = n,
                None => return Ok(None),
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while cursor < end {
            if self.matches_valid(cursor) {
                if fires.len() == cap {
                    return Ok(FireWindow { fires, truncated: true });
                }
                fires.push(cursor);
            }
            match next_minute(cursor) {
                Some(n) => cursor = n,
                None => break,
            }
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut parts = Vec::with_capacity(5);
        for (i, field) in FIELDS.iter().enumerate() {
            let (min, max) = domain(*field);
            let set = &self.values[i];
            let unrestricted = match field {
                Field::DayOfMonth => self.dom_unrestricted,
                Field::DayOfWeek => self.dow_unrestricted,
                _ => false,
            };
            if unrestricted {
                parts.push("*".to_string());
            } else if set.len() == (max - min + 1) as usize {
                match field {
                    Field::Minute | Field::Hour | Field::Month => parts.push("*".to_string()),
                    Field::DayOfMonth | Field::DayOfWeek => parts.push("*/1".to_string()),
                }
            } else {
                parts.push(
                    set.iter()
                        .map(u8::to_string)
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        parts.join(" ")
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.values[0].contains(&at.minute)
            || !self.values[1].contains(&at.hour)
            || !self.values[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.values[2].contains(&at.day);
        let dow = self.values[4].contains(&weekday(at.year, at.month, at.day));
        match (self.dom_unrestricted, self.dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_length(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn next_minute(mut d: CivilDateTime) -> Option<CivilDateTime> {
    if d.minute < 59 {
        d.minute += 1;
        return Some(d);
    }
    d.minute = 0;
    if d.hour < 23 {
        d.hour += 1;
        return Some(d);
    }
    d.hour = 0;
    if d.day < month_length(d.year, d.month) {
        d.day += 1;
        return Some(d);
    }
    d.day = 1;
    if d.month < 12 {
        d.month += 1;
        return Some(d);
    }
    if d.year < 2399 {
        d.year += 1;
        d.month = 1;
        return Some(d);
    }
    None
}

fn previous_minute(mut d: CivilDateTime) -> Option<CivilDateTime> {
    if d.minute > 0 {
        d.minute -= 1;
        return Some(d);
    }
    d.minute = 59;
    if d.hour > 0 {
        d.hour -= 1;
        return Some(d);
    }
    d.hour = 23;
    if d.day > 1 {
        d.day -= 1;
        return Some(d);
    }
    if d.month > 1 {
        d.month -= 1;
    } else {
        if d.year == 2000 {
            return None;
        }
        d.year -= 1;
        d.month = 12;
    }
    d.day = month_length(d.year, d.month);
    Some(d)
}
