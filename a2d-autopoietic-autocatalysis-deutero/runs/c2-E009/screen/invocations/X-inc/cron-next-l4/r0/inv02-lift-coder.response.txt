#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_width(field: Field) -> u32 {
    let (lo, hi) = field_domain(field);
    hi - lo + 1
}

fn parse_number(text: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn name_value(text: &str, field: Field) -> Option<u32> {
    if field == Field::Month {
        let names = [
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT",
            "NOV", "DEC",
        ];
        names
            .iter()
            .position(|name| text.eq_ignore_ascii_case(name))
            .map(|i| i as u32 + 1)
    } else if field == Field::DayOfWeek {
        let names = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];
        names
            .iter()
            .position(|name| text.eq_ignore_ascii_case(name))
            .map(|i| i as u32)
    } else {
        None
    }
}

fn parse_base_value(
    text: &str,
    field: Field,
    offset: usize,
) -> Result<u32, ParseError> {
    if text.as_bytes().iter().all(u8::is_ascii_digit) {
        let value = parse_number(text, field, offset)?;
        let (lo, hi) = field_domain(field);
        if value < lo || value > hi {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok(value)
    } else if let Some(value) = name_value(text, field) {
        Ok(value)
    } else {
        Err(ParseError::InvalidSyntax { field, offset })
    }
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (domain_min, domain_max) = field_domain(field);
    let mut selected = vec![false; (domain_max + 1) as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = text[item_start..]
            .find(',')
            .map(|i| item_start + i)
            .unwrap_or(text.len());
        let item = &text[item_start..item_end];

        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        let slash_pos = item.find('/');
        let (base, step_text) = match slash_pos {
            Some(pos) => (&item[..pos], Some(&item[pos + 1..])),
            None => (item, None),
        };

        if slash_count > 1 || base.is_empty() || step_text.is_some_and(str::is_empty) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let is_star = base == "*";
        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        let (start_text, end_text) = if is_star {
            if hyphen_count != 0 {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (None, None)
        } else if hyphen_count == 0 {
            (Some(base), None)
        } else if hyphen_count == 1 {
            let pos = base.find('-').unwrap();
            let left = &base[..pos];
            let right = &base[pos + 1..];
            if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (Some(left), Some(right))
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        // Validate base and endpoint forms before reading numeric values.
        if let Some(value) = start_text {
            let valid = value.as_bytes().iter().all(u8::is_ascii_digit)
                || name_value(value, field).is_some();
            if !valid {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }
        if let Some(value) = end_text {
            let valid = value.as_bytes().iter().all(u8::is_ascii_digit)
                || name_value(value, field).is_some();
            if !valid {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let (start, end) = if is_star {
            (domain_min, domain_max)
        } else {
            let left = start_text.unwrap();
            let left_offset = item_start;
            let start = parse_base_value(left, field, left_offset)?;
            if let Some(right) = end_text {
                let right_offset = item_start + base.find('-').unwrap() + 1;
                let end = parse_base_value(right, field, right_offset)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: right_offset,
                    });
                }
                (start, end)
            } else if step_text.is_some() {
                (start, domain_max)
            } else {
                (start, start)
            }
        };

        let step = if let Some(step_text) = step_text {
            if !step_text.as_bytes().iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            let step_offset = item_start + slash_pos.unwrap() + 1;
            let value = parse_number(step_text, field, step_offset)?;
            if value == 0 || value > field_width(field) {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            value
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == text.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let input = if input.starts_with('@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        parsed[i] = parse_field(parts[i], fields[i])?;
    }

    Ok(Cron {
        selected: parsed,
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let idx = field_index(field);
        let (lo, hi) = field_domain(field);
        (lo..=hi)
            .filter(|&value| self.selected[idx][value as usize])
            .map(|value| value as u8)
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.selected[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn date_matches(&self, year: u16, month: u8, day: u8) -> bool {
        let dom = self.contains(Field::DayOfMonth, day);
        let dow = self.contains(Field::DayOfWeek, weekday(year, month, day));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if day == days_in_month(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(&at)?;
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.date_matches(at.year, at.month, at.day))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(&after)?;
        let mut date = after.clone();
        loop {
            if self.contains(Field::Month, date.month)
                && self.date_matches(date.year, date.month, date.day)
            {
                for hour in self.allowed(Field::Hour) {
                    for minute in self.allowed(Field::Minute) {
                        let candidate = CivilDateTime {
                            year: date.year,
                            month: date.month,
                            day: date.day,
                            hour,
                            minute,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if !advance_day(&mut date) {
                return Ok(None);
            }
            date.hour = 0;
            date.minute = 0;
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(&before)?;
        let mut date = before.clone();
        loop {
            if self.contains(Field::Month, date.month)
                && self.date_matches(date.year, date.month, date.day)
            {
                let hours = self.allowed(Field::Hour);
                let minutes = self.allowed(Field::Minute);
                for &hour in hours.iter().rev() {
                    for &minute in minutes.iter().rev() {
                        let candidate = CivilDateTime {
                            year: date.year,
                            month: date.month,
                            day: date.day,
                            hour,
                            minute,
                        };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if !retreat_day(&mut date) {
                return Ok(None);
            }
            date.hour = 23;
            date.minute = 59;
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(&start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(&end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start.clone();
        loop {
            let Some(fire) = self
                .next_fire(cursor.clone())
                .expect("validated datetime")
            else {
                break;
            };
            if fire >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(fire.clone());
            cursor = fire;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let fields = [
            Field::Minute,
            Field::Hour,
            Field::DayOfMonth,
            Field::Month,
            Field::DayOfWeek,
        ];
        let mut out = Vec::new();
        for field in fields {
            let values = self.allowed(field);
            let full = values.len() as u32 == field_width(field);
            let rendered = match field {
                Field::DayOfMonth if self.unrestricted_dom => "*".to_string(),
                Field::DayOfWeek if self.unrestricted_dow => "*".to_string(),
                Field::DayOfMonth | Field::DayOfWeek if full => "*/1".to_string(),
                _ if full => "*".to_string(),
                _ => values
                    .iter()
                    .map(u8::to_string)
                    .collect::<Vec<_>>()
                    .join(","),
            };
            out.push(rendered);
        }
        out.join(" ")
    }
}

fn validate_datetime(at: &CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if !(1..=days_in_month(at.year, at.month)).contains(&at.day) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_day(date: &mut CivilDateTime) -> bool {
    if date.day < days_in_month(date.year, date.month) {
        date.day += 1;
        true
    } else if date.month < 12 {
        date.month += 1;
        date.day = 1;
        true
    } else if date.year < 2399 {
        date.year += 1;
        date.month = 1;
        date.day = 1;
        true
    } else {
        false
    }
}

fn retreat_day(date: &mut CivilDateTime) -> bool {
    if date.day > 1 {
        date.day -= 1;
        true
    } else if date.month > 1 {
        date.month -= 1;
        date.day = days_in_month(date.year, date.month);
        true
    } else if date.year > 2000 {
        date.year -= 1;
        date.month = 12;
        date.day = 31;
        true
    } else {
        false
    }
}
