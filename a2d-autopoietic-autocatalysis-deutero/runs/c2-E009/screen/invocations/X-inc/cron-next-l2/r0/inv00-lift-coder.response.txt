#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(bytes: &[u8], start: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset: start });
    }

    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_part, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, 0)
    };

    let star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let base_kind = if star && hyphens.is_empty() {
        0
    } else if !star && hyphens.is_empty() && !base.is_empty() {
        1
    } else if !star && hyphens.len() == 1 {
        let h = hyphens[0];
        if h == 0 || h + 1 == base.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        2
    } else {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    };

    if let Some(step) = step_part {
        if step.is_empty() || !step.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    }

    let range_end_position = if base_kind == 2 {
        Some(hyphens[0])
    } else {
        None
    };

    let (min, max) = domain(field);

    let (start, end) = match base_kind {
        0 => (min, max),
        1 => {
            let value = parse_number(base, item_offset, field)?;
            if value < min || value > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_offset,
                });
            }
            (value, value)
        }
        _ => {
            let split = range_end_position.unwrap();
            let start_value = parse_number(&base[..split], item_offset, field)?;
            let end_offset = item_offset + split + 1;
            let end_value = parse_number(&base[split + 1..], end_offset, field)?;
            if start_value < min || start_value > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_offset,
                });
            }
            if end_value < min || end_value > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_offset,
                });
            }
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (start_value, end_value)
        }
    };

    let step = if let Some(step_bytes) = step_part {
        let value = parse_number(step_bytes, step_offset, field)?;
        let width = max - min + 1;
        if value < 1 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    for value in (start..=end).step_by(step as usize) {
        selected[(value - min) as usize] = true;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut start = 0usize;

    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(&bytes[start..i], start, field, &mut selected)?;
            start = i + 1;
        }
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut parsed = Vec::with_capacity(5);
    for (i, field) in FIELDS.iter().copied().enumerate() {
        parsed.push(parse_field(pieces[i], field)?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().unwrap(),
        day_of_month_is_star: pieces[2].as_bytes() == b"*",
        day_of_week_is_star: pieces[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = domain(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some((min as usize + i) as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize - 1]
            && self.selected[3][at.month as usize - 1]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut count = 0u32;
    for y in 2000..year {
        count += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        count += days_in_month(year, m) as u32;
    }
    count += day as u32 - 1;
    ((6 + count) % 7) as u8
}

fn increment_minute(value: &mut CivilDateTime) -> bool {
    if value.minute < 59 {
        value.minute += 1;
        return true;
    }
    value.minute = 0;

    if value.hour < 23 {
        value.hour += 1;
        return true;
    }
    value.hour = 0;

    if value.day < days_in_month(value.year, value.month) {
        value.day += 1;
        return true;
    }
    value.day = 1;

    if value.month < 12 {
        value.month += 1;
        return true;
    }
    value.month = 1;

    if value.year < 2399 {
        value.year += 1;
        return true;
    }
    false
}
