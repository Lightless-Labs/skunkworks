#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();

    if let Some(at) = bytes.iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = bytes.iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = bytes
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = bytes[0] == b'/';
    let directory_intent = bytes.last() == Some(&b'/');

    let mut components: Vec<(String, usize)> = Vec::new();
    let mut last_nonempty: Option<(&str, usize)> = None;
    let mut start = 0;

    while start < bytes.len() {
        if bytes[start] == b'/' {
            start += 1;
            continue;
        }

        let mut end = start;
        while end < bytes.len() && bytes[end] != b'/' {
            end += 1;
        }

        let component = &path[start..end];
        last_nonempty = Some((component, start));

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|(name, _)| name != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push((component.to_owned(), start));
                }
            }
            _ => components.push((component.to_owned(), start)),
        }

        start = end;
    }

    let directory_intent = directory_intent
        || last_nonempty.is_some_and(|(component, _)| component == "." || component == "..");

    let joined = components
        .into_iter()
        .map(|(component, _)| component)
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
