fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let width = m + 1;
    let mut lcs = vec![0usize; (n + 1) * (m + 1)];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            let value = if old[i] == new[j] {
                1 + lcs[(i + 1) * width + j + 1]
            } else {
                lcs[(i + 1) * width + j].max(lcs[i * width + j + 1])
            };
            lcs[i * width + j] = value;
        }
    }

    #[derive(Clone)]
    enum Op {
        Keep,
        Delete,
        Insert(String),
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut ops = Vec::new();

    while i < n && j < m {
        if old[i] == new[j] {
            ops.push(Op::Keep);
            i += 1;
            j += 1;
        } else {
            let delete_len = lcs[(i + 1) * width + j];
            let insert_len = lcs[i * width + j + 1];
            let delete = if delete_len > insert_len {
                true
            } else if insert_len > delete_len {
                false
            } else {
                (i + j) % 2 == 0
            };

            if delete {
                ops.push(Op::Delete);
                i += 1;
            } else {
                ops.push(Op::Insert(new[j].clone()));
                j += 1;
            }
        }
    }

    while i < n {
        ops.push(Op::Delete);
        i += 1;
    }
    while j < m {
        ops.push(Op::Insert(new[j].clone()));
        j += 1;
    }

    let mut changes = Vec::new();
    let mut base_pos = 0usize;
    let mut k = 0usize;

    while k < ops.len() {
        if matches!(ops[k], Op::Keep) {
            base_pos += 1;
            k += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0usize;
        let mut replacement = Vec::new();

        while k < ops.len() && !matches!(ops[k], Op::Keep) {
            match &ops[k] {
                Op::Delete => {
                    deleted += 1;
                    base_pos += 1;
                }
                Op::Insert(line) => replacement.push(line.clone()),
                Op::Keep => unreachable!(),
            }
            k += 1;
        }

        changes.push(Change {
            start,
            end: start + deleted,
            replacement,
        });
    }

    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;

    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn apply_changes(
    base: &[String],
    start: usize,
    end: usize,
    changes: impl Iterator<Item = Change>,
) -> Vec<String> {
    let mut output = Vec::new();
    let mut cursor = start;

    for change in changes {
        if change.start > cursor {
            output.extend(base[cursor..change.start].iter().cloned());
        }
        output.extend(change.replacement);
        cursor = change.end;
    }

    if cursor < end {
        output.extend(base[cursor..end].iter().cloned());
    }
    output
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let mut nodes = Vec::new();
    for (side, changes) in [(false, &left_changes), (true, &right_changes)] {
        for (index, change) in changes.iter().enumerate() {
            nodes.push((side, index, change.clone()));
        }
    }

    let count = nodes.len();
    let mut seen = vec![false; count];
    let mut components = Vec::new();

    for root in 0..count {
        if seen[root] {
            continue;
        }

        let mut stack = vec![root];
        seen[root] = true;
        let mut members = Vec::new();

        while let Some(current) = stack.pop() {
            members.push(current);
            for other in 0..count {
                if !seen[other] && interacts(&nodes[current].2, &nodes[other].2) {
                    seen[other] = true;
                    stack.push(other);
                }
            }
        }

        let start = members
            .iter()
            .map(|&idx| nodes[idx].2.start)
            .min()
            .unwrap();
        let end = members
            .iter()
            .map(|&idx| nodes[idx].2.end)
            .max()
            .unwrap();

        components.push(Component {
            changes: members
                .into_iter()
                .map(|idx| (nodes[idx].0, nodes[idx].1))
                .collect(),
            start,
            end,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output_lines = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;
    let mut marked_ordinal = 0usize;

    for component in components {
        if base_cursor < component.start {
            output_lines.extend(base_lines[base_cursor..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();

        let mut left_indices: Vec<usize> = component
            .changes
            .iter()
            .filter_map(|&(is_right, index)| (!is_right).then_some(index))
            .collect();
        left_indices.sort_by_key(|&i| left_changes[i].start);

        let mut right_indices: Vec<usize> = component
            .changes
            .iter()
            .filter_map(|&(is_right, index)| is_right.then_some(index))
            .collect();
        right_indices.sort_by_key(|&i| right_changes[i].start);

        let left_projection = apply_changes(
            &base_lines,
            component.start,
            component.end,
            left_indices.into_iter().map(|i| left_changes[i].clone()),
        );
        let right_projection = apply_changes(
            &base_lines,
            component.start,
            component.end,
            right_indices.into_iter().map(|i| right_changes[i].clone()),
        );

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    marked_ordinal += 1;
                    let output_start = output_lines.len();
                    let base_len = component.end - component.start;
                    let base_marker_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output_lines.push(format!("<<<<<<< LEFT {marked_ordinal}"));
                    output_lines.extend(left_projection);
                    output_lines.push(format!(
                        "||||||| BASE {base_marker_start},{base_len}"
                    ));
                    output_lines.extend(base_projection);
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection);
                    output_lines.push(format!(">>>>>>> RIGHT {marked_ordinal}"));

                    conflicts.push(Conflict {
                        ordinal: marked_ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: output_lines.len() - output_start,
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        output_lines.extend(base_lines[base_cursor..].iter().cloned());
    }

    let mut text = String::new();
    for line in output_lines {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn diff(old: &[String], new: &[String], side: usize, changes: &mut Vec<Change>) {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0, 0, 0);
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>,
                 base_pos: usize| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
                side,
            });
            *deletes = 0;
        }
        let _ = base_pos;
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(
                changes,
                &mut run_start,
                &mut deletes,
                &mut inserts,
                base_pos,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            run_start.get_or_insert(base_pos);
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            run_start.get_or_insert(base_pos);
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            let choose_delete = if del > ins {
                true
            } else if ins > del {
                false
            } else {
                (i + j) % 2 == 0

            if choose_delete {
                deletes += 1;
                i += 1;
                base_pos += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(
        changes,
        &mut run_start,
        &mut deletes,
        &mut inserts,
        base_pos,
    );
}

        components.push(Component {
            start,
            end,
            changes: members,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });
    components
}

fn project(
    base: &[String],
    component: &Component,
    changes: &[Change],
    side: usize,
) -> Vec<String> {
    let mut selected: Vec<&Change> = component
        .changes
        .iter()
        .map(|&idx| &changes[idx])
        .filter(|change| change.side == side)
        .collect();
    selected.sort_by_key(|change| (change.start, change.end));

    let mut out = Vec::new();
    let mut cursor = component.start;

    for change in selected {
        if change.start > cursor {
            out.extend(base[cursor..change.start].iter().cloned());
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < component.end {
        out.extend(base[cursor..component.end].iter().cloned());
    }
    out
}

fn emit(lines: &[String], output: &mut Vec<String>) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = Vec::new();
    diff(&base_lines, &left_lines, 0, &mut changes);
    diff(&base_lines, &right_lines, 1, &mut changes);
    let components = build_components(&changes);

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0usize;

    for component in &components {
        if cursor < component.start {
            emit(&base_lines[cursor..component.start], &mut output);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(&base_lines, component, &changes, 0);
        let right_projection = project(&base_lines, component, &changes, 1);

        if left_projection == right_projection {
            emit(&left_projection, &mut output);
        } else if left_projection == base_projection {
            emit(&right_projection, &mut output);
        } else if right_projection == base_projection {
            emit(&left_projection, &mut output);
        } else {
            match resolution {
                Resolution::PreferLeft => emit(&left_projection, &mut output),
                Resolution::PreferRight => emit(&right_projection, &mut output),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    emit(&left_projection, &mut output);
                    let base_label_start = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_label_start},{}",
                        component.end - component.start
                    ));
                    emit(&base_projection, &mut output);
                    output.push("=======".to_owned());
                    emit(&right_projection, &mut output);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: output.len() - output_start,
                    });
                }
            }
        }
        cursor = component.end;
    }

    if cursor < base_lines.len() {
        emit(&base_lines[cursor..], &mut output);
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn edit_changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0usize, 0usize, 0usize);
    let mut changes = Vec::new();
    let mut active_start = None;
    let mut deleted = 0usize;
    let mut replacement = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active_start: &mut Option<usize>,
                 deleted: &mut usize,
                 replacement: &mut Vec<String>,
                 base_pos: usize| {
        if let Some(start) = active_start.take() {
            changes.push(Change {
                start,
                end: start + *deleted,
                replacement: std::mem::take(replacement),
                side,
            });
            *deleted = 0;
        } else {
            let _ = base_pos;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(
                &mut changes,
                &mut active_start,
                &mut deleted,
                &mut replacement,
                base_pos,
            );
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        let take_delete = if j == m {
            true
        } else if i == n {
            false
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            }
        };

        if active_start.is_none() {
            active_start = Some(base_pos);
        }

        if take_delete {
            i += 1;
            base_pos += 1;
            deleted += 1;
        } else {
            replacement.push(new[j].clone());
            j += 1;
        }
    }

    flush(
        &mut changes,
        &mut active_start,
        &mut deleted,
        &mut replacement,
        base_pos,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

    base: &[String],
    start: usize,
    end: usize,
    changes: &[Change],
    indices: &[usize],

    }

    }
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = edit_changes(&base_lines, &left_lines, 0);
    changes.extend(edit_changes(&base_lines, &right_lines, 1));

    let count = changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        x
    }

    for i in 0..count {
        for j in i + 1..count {
            if interacts(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut grouped: Vec<Component> = Vec::new();
    for i in 0..count {
        let r = root(&mut parent, i);
        if let Some(component) = grouped.iter_mut().find(|c| c.changes.first().is_some_and(|&x| {
            root(&mut parent, x) == r
        })) {
            component.start = component.start.min(changes[i].start);
            component.end = component.end.max(changes[i].end);
            component.changes.push(i);
        } else {
            grouped.push(Component {
                start: changes[i].start,
                end: changes[i].end,
                changes: vec![i],
            });
        }
    }

    grouped.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in grouped {
        if base_cursor < component.start {
            append_lines(&mut output, &base_lines[base_cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_indices: Vec<usize> = component
            .changes
            .iter()
            .copied()
            .filter(|&i| changes[i].side == 0)
            .collect();
        let right_indices: Vec<usize> = component
            .changes
            .iter()
            .copied()
            .filter(|&i| changes[i].side == 1)
            .collect();

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &changes,
            &left_indices,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &changes,
            &right_indices,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let region_len = component.end - component.start;
                    let marker_start = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("||||||| BASE {marker_start},{region_len}"));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[base_cursor..]);
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn script(old: &[String], new: &[String]) -> Vec<(char, Option<String>)> {
    let (n, m) = (old.len(), new.len());
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut ops = Vec::new();
    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            ops.push(('K', None));
            i += 1;
            j += 1;
        } else if i == n {
            ops.push(('I', Some(new[j].clone())));
            j += 1;
        } else if j == m {
            ops.push(('D', None));
            i += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins || (del == ins && (i + j) % 2 == 0) {
                ops.push(('D', None));
                i += 1;
            } else {
                ops.push(('I', Some(new[j].clone())));
                j += 1;
            }
        }
    }
    ops
}

fn changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let ops = script(old, new);
    let mut result = Vec::new();
    let mut base_pos = 0;
    let mut idx = 0;

    while idx < ops.len() {
        if ops[idx].0 == 'K' {
            base_pos += 1;
            idx += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0;
        let mut replacement = Vec::new();
        while idx < ops.len() && ops[idx].0 != 'K' {
            match ops[idx].0 {
                'D' => {
                    deleted += 1;
                    base_pos += 1;
                }
                'I' => replacement.push(ops[idx].1.as_ref().unwrap().clone()),
                _ => unreachable!(),
            }
            idx += 1;
        }
        result.push(Change {
            start,
            end: start + deleted,
            replacement,
            side,
        });
    }
    result
}

fn interacts(a: &Change, b: &Change) -> bool {
    match (a.start == a.end, b.start == b.end) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    component: &Component,
    all: &[Change],
    side: usize,
) -> Vec<String> {
    let mut indices: Vec<usize> = component
        .changes
        .iter()
        .copied()
        .filter(|&i| all[i].side == side)
        .collect();
    indices.sort_by_key(|&i| (all[i].start, all[i].end));

    let mut out = Vec::new();
    let mut cursor = component.start;
    for i in indices {
        let change = &all[i];
        out.extend(base[cursor..change.start].iter().cloned());
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    out.extend(base[cursor..component.end].iter().cloned());
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);
    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    all.extend(changes(&base_lines, &right_lines, 2));

    let mut components = Vec::new();

        if seen[seed] {
            continue;
        }
        seen[seed] = true;
        let mut stack = vec![seed];
        let mut members = Vec::new();
        while let Some(i) = stack.pop() {
            members.push(i);
            for j in 0..all.len() {

                    seen[j] = true;
                    stack.push(j);
                }

        let end = members.iter().map(|&i| all[i].end).max().unwrap();

            changes: members,
            start,
            end,
        });

    let mut conflicts = Vec::new();

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(&base_lines, &component, &all, 1);
        let right_projection = projection(&base_lines, &component, &all, 2);

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);

                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_label_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());
                    output.push(format!(
                        "||||||| BASE {base_label_start},{base_len}"
                    ));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    output.extend(base_lines[base_cursor..].iter().cloned());
    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
