#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
}

struct Component {
    changes: Vec<usize>,
    start: usize,
    end: usize,
}

fn lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn validate_cr(input: &str, side: MergeSide) -> Result<(), MergeError> {
    if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(MergeError::CarriageReturn { side, byte });
    }
    Ok(())
}

fn script(old: &[String], new: &[String]) -> Vec<(char, Option<String>)> {
    let (n, m) = (old.len(), new.len());
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut ops = Vec::new();
    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            ops.push(('K', None));
            i += 1;
            j += 1;
        } else if i == n {
            ops.push(('I', Some(new[j].clone())));
            j += 1;
        } else if j == m {
            ops.push(('D', None));
            i += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins || (del == ins && (i + j) % 2 == 0) {
                ops.push(('D', None));
                i += 1;
            } else {
                ops.push(('I', Some(new[j].clone())));
                j += 1;
            }
        }
    }
    ops
}

fn changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let ops = script(old, new);
    let mut result = Vec::new();
    let mut base_pos = 0;
    let mut idx = 0;

    while idx < ops.len() {
        if ops[idx].0 == 'K' {
            base_pos += 1;
            idx += 1;
            continue;
        }

        let start = base_pos;
        let mut deleted = 0;
        let mut replacement = Vec::new();
        while idx < ops.len() && ops[idx].0 != 'K' {
            match ops[idx].0 {
                'D' => {
                    deleted += 1;
                    base_pos += 1;
                }
                'I' => replacement.push(ops[idx].1.as_ref().unwrap().clone()),
                _ => unreachable!(),
            }
            idx += 1;
        }
        result.push(Change {
            start,
            end: start + deleted,
            replacement,
            side,
        });
    }
    result
}

fn interacts(a: &Change, b: &Change) -> bool {
    match (a.start == a.end, b.start == b.end) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    component: &Component,
    all: &[Change],
    side: usize,
) -> Vec<String> {
    let mut indices: Vec<usize> = component
        .changes
        .iter()
        .copied()
        .filter(|&i| all[i].side == side)
        .collect();
    indices.sort_by_key(|&i| (all[i].start, all[i].end));

    let mut out = Vec::new();
    let mut cursor = component.start;
    for i in indices {
        let change = &all[i];
        out.extend(base[cursor..change.start].iter().cloned());
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    out.extend(base[cursor..component.end].iter().cloned());
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);
    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut all = changes(&base_lines, &left_lines, 1);
    all.extend(changes(&base_lines, &right_lines, 2));

    let mut seen = vec![false; all.len()];
    let mut components = Vec::new();
    for seed in 0..all.len() {
        if seen[seed] {
            continue;
        }
        seen[seed] = true;
        let mut stack = vec![seed];
        let mut members = Vec::new();
        while let Some(i) = stack.pop() {
            members.push(i);
            for j in 0..all.len() {
                if !seen[j] && interacts(&all[i], &all[j]) {
                    seen[j] = true;
                    stack.push(j);
                }
            }
        }
        let start = members.iter().map(|&i| all[i].start).min().unwrap();
        let end = members.iter().map(|&i| all[i].end).max().unwrap();
        components.push(Component {
            changes: members,
            start,
            end,
        });
    }

    components.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0;

    for component in components {
        output.extend(base_lines[base_cursor..component.start].iter().cloned());

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(&base_lines, &component, &all, 1);
        let right_projection = projection(&base_lines, &component, &all, 2);

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_label_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());
                    output.push(format!(
                        "||||||| BASE {base_label_start},{base_len}"
                    ));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    output.extend(base_lines[base_cursor..].iter().cloned());
    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
