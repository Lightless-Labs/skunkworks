#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
    ResourceShape { task: usize, expected: usize, actual: usize },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation must precede all other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = Vec::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut available = Vec::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    // Kahn order with the smallest currently available ID first.
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone)]
    struct Running {
        task: usize,
        finish: u64,
        worker: usize,
    }

    let worker_count = cooldowns.len();
    let mut worker_available = vec![0u64; worker_count];
    let mut running: Vec<Running> = Vec::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut used: Vec<u64> = vec![0; capacity.len()];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut trace = format!("workers={};resources={}\n", worker_count, capacity.len());
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    fn append_usage(trace: &mut String, used: &[u64]) {
        if used.is_empty() {
            trace.push('-');
        } else {
            for (i, value) in used.iter().enumerate() {
                if i != 0 {
                    trace.push(',');
                }
                trace.push_str(&value.to_string());
            }
        }
    }

    while completed_count < n {
        // Complete all tasks finishing now, in task-ID order.
        let mut finished: Vec<Running> = running
            .iter()
            .filter(|entry| entry.finish == now)
            .cloned()
            .collect();
        finished.sort_by_key(|entry| entry.task);

        if !finished.is_empty() {
            running.retain(|entry| entry.finish != now);
            for entry in finished {
                let id = entry.task;
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for (resource, &need) in tasks[id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!("{now}|end|{id}|{}|", entry.worker));
                append_usage(&mut trace, &used);
                trace.push('\n');
            }
        }

        // Keep admitting tasks at this time, reevaluating after every start.
        loop {
            let mut eligible: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && tasks[id].deps.iter().all(|&dep| completed[dep])
                        && tasks[id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r])
                })
                .collect();

            eligible.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(id) = eligible.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now)
                .expect("admission requires an available worker");

            let finish = now
                .checked_add(tasks[id].duration)
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Finish,
                })?;
            let worker_free = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Turnaround,
                })?;

            started[id] = true;
            worker_available[worker] = worker_free;
            for (resource, &need) in tasks[id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            trace.push_str(&format!("{now}|start|{id}|{worker}|"));
            append_usage(&mut trace, &used);
            trace.push('\n');

            if finish == now {
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for (resource, &need) in tasks[id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!("{now}|end|{id}|{worker}|"));
                append_usage(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running {
                    task: id,
                    finish,
                    worker,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next = u64::MAX;
        for entry in &running {
            if entry.finish > now {
                next = next.min(entry.finish);
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = next.min(tasks[id].release);
            }
        }
        for &when in &worker_available {
            if when > now {
                next = next.min(when);
            }
        }

        // Every unfinished task has either a future release, a running task,
        // or an occupied worker event that can advance the simulation.
        if next == u64::MAX {
            // A release at u64::MAX is itself a valid future event.
            if (0..n).any(|id| !started[id] && tasks[id].release == u64::MAX && now < u64::MAX) {
                next = u64::MAX;
            } else {
                unreachable!("unfinished acyclic tasks must have a future event");
            }
        }
        now = next;
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}
