```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    capacity: Option<usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::build(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::build(Some(capacity))
    }

    fn build(capacity: Option<usize>) -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            capacity,
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn canonical_key(key: &str) -> Result<String, Error> {
        let bytes = key.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }

        if start == end {
            return Err(Error::EmptyKey);
        }

        for i in start..end {
            let b = bytes[i];
            let valid = if i == start {
                b.is_ascii_alphabetic()
            } else {
                b.is_ascii_alphanumeric() || b == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: i });
            }
        }

        Ok(key[start..end].to_ascii_lowercase())
    }

    fn compare(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (i, entry) in self.entries.iter().enumerate() {
            self.indices.insert(entry.item.key.clone(), i);
        }
    }

    fn sort_best_first(&mut self) {
        self.entries.sort_by(Self::compare);
        self.rebuild_indices();
    }

    fn allocate_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut order: Vec<usize> = (0..self.entries.len()).collect();
            order.sort_by_key(|&i| self.entries[i].age);
            for (rank, index) in order.into_iter().enumerate() {
                self.entries[index].age = rank as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    fn remove_at(&mut self, index: usize) -> Entry {
        let entry = self.entries.remove(index);
        self.indices.remove(&entry.item.key);
        self.rebuild_indices();
        entry
    }

    fn trim_to_capacity(&mut self) -> Vec<Item> {
        let mut evicted = Vec::new();
        if let Some(capacity) = self.capacity {
            while self.entries.len() > capacity {
                self.sort_best_first();
                let entry = self.entries.pop().expect("nonempty while over capacity");
                self.indices.remove(&entry.item.key);
                evicted.push(entry.item);
            }
        }
        evicted
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<Admission, Error> {
        let key = Self::canonical_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Entry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed.item));
        }

        if let Some(capacity) = self.capacity {
            if self.entries.len() >= capacity {
                let worst_index = self.entries.iter().enumerate().max_by(|(_, a), (_, b)| Self::compare(a, b)).map(|(i, _)| i);
                if let Some(index) = worst_index {
                    if Self::compare(&proposed, &self.entries[index]).is_lt() {
                        let evicted = self.remove_at(index).item;
                        let age = self.allocate_age();
                        self.entries.push(Entry { age, ..proposed });
                        self.sort_best_first();
                        return Ok(Admission::Evicted(evicted));
                    }
                    return Ok(Admission::Discarded(proposed.item));
                }
            }
        }

        let age = self.allocate_age();
        self.entries.push(Entry { age, ..proposed });
        self.sort_best_first();
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.iter().min_by(|a, b| Self::compare(a, b)).map(|e| e.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let index = self.entries.iter().enumerate().min_by(|(_, a), (_, b)| Self::compare(a, b)).map(|(i, _)| i)?;
        Some(self.remove_at(index).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonical_key(key)?;
        Ok(self.indices.get(&key).map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection { current, requested: priority });
        }
        self.entries[index].item.priority = priority;
        self.sort_best_first();
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection { current, requested: priority });
        }
        self.entries[index].item.priority = priority;
        self.sort_best_first();
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let old = std::mem::replace(&mut self.entries[index].item.value, value.to_owned());
        self.sort_best_first();
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.remove_at(index).item)
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = self.trim_to_capacity();
        evicted.reverse();
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        let mut entries = self.entries.clone();
        entries.sort_by(Self::compare);
        entries.into_iter().map(|entry| entry.item).collect::<Vec<_>>().into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source = other.entries.clone();
        source.sort_by(Self::compare);

        if policy == CollisionPolicy::Reject {
            for entry in &source {
                if self.indices.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        other.entries.clear();
        other.indices.clear();

        for entry in source {
            if let Some(&index) = self.indices.get(&entry.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = entry.item.priority;
                        self.entries[index].item.value = entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.allocate_age();
                self.entries.push(Entry {
                    item: entry.item,
                    age,
                });
                report.moved += 1;
            }
        }

        self.sort_best_first();
        report.evicted = self.trim_to_capacity();
        Ok(report)
    }
}
```
