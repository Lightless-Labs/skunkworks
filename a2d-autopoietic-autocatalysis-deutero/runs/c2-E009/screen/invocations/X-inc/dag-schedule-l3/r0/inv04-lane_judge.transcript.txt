                    task: task_id,
                    position,
                    value,
                });

                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }

        return Err(Error::Workers { count: workers });

            successors[dep].push(task_id);
        }

        if indegree[id] == 0 {
            available.insert(id);
        }

    while let Some(id) = available.pop_first() {
        order.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }

            let mut v = vec![false; n];
            for &id in &order {
                v[id] = true;
            }
            v

        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });

            .unwrap_or(0);

                stage: Stage::Tail,

                    continue;

                    Some(prev) => {

                };

                break;

                stage: Stage::Finish,

                task,
                stage: Stage::Turnaround,

                start: now,
                finish,
                worker,

                time: now,
                task,
                worker,
                kind: EventKind::Start,

                completed[task] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &successors[task] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change eligibility.
        let mut next = None::<u64>;
        for &(finish, _, _) in &running {
            if finish > now {
                next = Some(next.map_or(finish, |v| v.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |v| v.min(tasks[id].release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next = Some(next.map_or(available_at, |v| v.min(available_at)));
            }
        }

        // With an acyclic graph, unfinished tasks always have a future release,
        // running completion, or worker expiry capable of making progress.
        if let Some(next_time) = next {
            now = next_time;
        } else {
            break;
        }
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}

    workers: usize,
    turnaround: u64,
) -> Result<Schedule, Error> {

    // Validate every dependency entry before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest currently available ID.
    let mut available = std::collections::BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        topo.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if topo.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &id in &topo {
                flags[id] = true;
            }
            flags
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute tails in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]

            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;

    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed = vec![false; n];
    let mut started = vec![false; n];

    let mut worker_available: Vec<Option<u64>> = vec![None; workers];

    // Running tasks are stored as (finish, task, worker).
    let mut running: Vec<(u64, usize, usize)> = Vec::new();

    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

        let mut finishing: Vec<(u64, usize, usize)> = running

                task,

                kind: EventKind::Finish,

        for available_at in &mut worker_available {

            }

        loop {

                break;

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > time {
                    continue;

                }
            }

            let Some(task_id) = best else {
                break;
            };

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(turnaround)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            events.push(Event {
                time,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(time);
                events.push(Event {
                    time,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task_id, worker));
            }

            if available_at > time {
                worker_available[worker] = Some(available_at);
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest future event that may change eligibility.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > time {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > time {
                next_time = Some(next_time.map_or(tasks[id].release, |current| {
                    current.min(tasks[id].release)
                }));
            }
        }
        for &available_at in &worker_available {
            if let Some(t) = available_at {
                if t > time {
                    next_time = Some(next_time.map_or(t, |current| current.min(t)));
                }
            }
        }

        // If there are no future events, no unfinished task can become eligible.
        // This is unreachable for an acyclic graph with a valid worker count.
        time = next_time.expect("unfinished acyclic tasks must have a future event");
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task must start exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    let mut ready = BinaryHeap::<std::cmp::Reverse<usize>>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            ready.push(std::cmp::Reverse(id));
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = ready.pop() {
        topo.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.push(std::cmp::Reverse(child));
            }
        }
    }

    if topo.len() != n {
        let mut removed = vec![false; n];
        for &id in &topo {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut remaining_deps: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut eligible = BinaryHeap::<AvailableTask>::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut worker_free = vec![0u64; workers];
    let mut running = Vec::<Running>::new();
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    for id in 0..n {
        if remaining_deps[id] == 0 {
            eligible.push(AvailableTask {
                tail: tails[id],
                priority: tasks[id].priority,
                id,
            });
        }
    }

    while completed_count < n {
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == time)
            .collect();
        finishing.sort_by_key(|entry| entry.task);

        if !finishing.is_empty() {
            running.retain(|entry| entry.finish != time);
            for entry in finishing {
                completed[entry.task] = true;
                completed_count += 1;
                makespan = makespan.max(time);
                events.push(Event {
                    time,
                    task: entry.task,
                    worker: entry.worker,
                    kind: EventKind::Finish,
                });
                for &child in &children[entry.task] {
                    remaining_deps[child] -= 1;
                    if remaining_deps[child] == 0 {
                        eligible.push(AvailableTask {
                            tail: tails[child],
                            priority: tasks[child].priority,
                            id: child,
                        });
                    }
                }
            }
        }

        loop {
            for id in 0..n {
                if !started[id]
                    && remaining_deps[id] == 0
                    && tasks[id].release <= time
                    && !eligible.iter().any(|entry| entry.id == id)
                {
                    eligible.push(AvailableTask {
                        tail: tails[id],
                        priority: tasks[id].priority,
                        id,
                    });
                }
            }

            let worker = (0..workers).find(|&w| worker_free[w] <= time);
            let Some(worker) = worker else {
                break;
            };

            let mut deferred = Vec::new();
            let chosen = loop {
                match eligible.pop() {
                    Some(entry) if !started[entry.id] && tasks[entry.id].release <= time => {
                        break Some(entry);
                    }
                    Some(entry) if !started[entry.id] => deferred.push(entry),
                    Some(_) => {}
                    None => break None,
                }
            };
            eligible.extend(deferred);
            let Some(entry) = chosen else {
                break;
            };

            let id = entry.id;
            let finish = time.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let free_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            worker_free[worker] = free_at;
            runs[id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            events.push(Event {
                time,
                task: id,
                worker,
                kind: EventKind::Start,
            });

            if finish == time {
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(time);
                events.push(Event {
                    time,
                    task: id,
                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &children[id] {
                    remaining_deps[child] -= 1;
                    if remaining_deps[child] == 0 {
                        eligible.push(AvailableTask {
                            tail: tails[child],
                            priority: tasks[child].priority,
                            id: child,
                        });
                    }
                }
            } else {
                running.push(Running {
                    task: id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for entry in &running {
            if entry.finish > time {
                next_time = Some(next_time.map_or(entry.finish, |old| old.min(entry.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && remaining_deps[id] == 0 && tasks[id].release > time {
                next_time = Some(next_time.map_or(tasks[id].release, |old| old.min(tasks[id].release)));
            }
        }
        for &free_at in &worker_free {
            if free_at > time {
                next_time = Some(next_time.map_or(free_at, |old| old.min(free_at)));
            }
        }

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before checking any other condition.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Edges run from predecessor to dependent task.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Sorted available set gives Kahn order with the smallest current ID first.
    let mut available = Vec::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let id = available.remove(0);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let pos = available.binary_search(&child).unwrap_or_else(|p| p);
                available.insert(pos, child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut set = vec![false; n];
            for &id in &order {
                set[id] = true;
            }
            set
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // (finish, task, worker)
    let mut remaining = n;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while remaining > 0 {
        // Complete all positive-duration tasks finishing now, in task-ID order.
        let mut finishing: Vec<(usize, usize)> = running
            .iter()
            .filter(|&&(finish, _, _)| finish == now)
            .map(|&(_, task, worker)| (task, worker))
            .collect();
        finishing.sort_unstable_by_key(|&(task, _)| task);

        for (task, worker) in finishing {
            running.retain(|&(_, t, _)| t != task);
            completed[task] = true;
            remaining -= 1;
            events.push(Event {
                time: now,
                task,
                worker,
                kind: EventKind::Finish,
            });
        }

        loop {
            let worker = worker_available
                .iter()
                .position(|&available_at| available_at <= now);
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if !tasks[id].deps.iter().all(|&dep| completed[dep]) {
                    continue;
                }
                let replace = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if replace {
                    best = Some(id);
                }
            }

            let Some(task) = best else {
                break;
            };

            let finish = now
                .checked_add(tasks[task].duration)
                .ok_or(Error::Overflow {
                    task,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task,
                stage: Stage::Turnaround,
            })?;

            started[task] = true;
            worker_available[worker] = available_at;
            runs[task] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task,
                worker,
                kind: EventKind::Start,
            });
            makespan = makespan.max(finish);

            if tasks[task].duration == 0 {
                completed[task] = true;
                remaining -= 1;
                events.push(Event {
                    time: now,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task, worker));
            }
        }

        if remaining == 0 {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        // Every unfinished task must eventually have a relevant event. This
        // fallback also avoids an infinite loop if that invariant is violated.
        now = next_time.expect("unfinished tasks must have a future event");
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task must have started"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
