#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000_000, "quintillion"),
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
];

fn append_under_hundred(n: u128, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        let ten = n / 10;
        let unit = n % 10;
        out.push_str(TENS[(ten - 2) as usize]);
        if unit != 0 {
            out.push('-');
            out.push_str(SMALL[unit as usize]);
        }
    }
}

fn append_group(n: u128, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_under_hundred(rem, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut remaining = magnitude;
    let mut emitted_higher = false;
    for &(scale, name) in &SCALES {
        let group = remaining / scale;
        remaining %= scale;
        if group != 0 {
            if emitted_higher {
                out.push(' ');
            }
            append_group(group, &mut out);
            out.push(' ');
            out.push_str(name);
            emitted_higher = true;
        }
    }

    if remaining != 0 {
        if emitted_higher {
            if remaining < 100 {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        append_group(remaining, &mut out);
    }

    Ok(out)
}

fn dictionary_value(word: &str) -> Option<Word> {
    if word == "zero" {
        return Some(Word::Number(0));
    }
    for (i, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(Word::Number(i as u128));
        }
    }
    for (i, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Number((i as u128 + 2) * 10));
        }
    }
    match word {
        "hundred" => Some(Word::Hundred),
        "thousand" => Some(Word::Scale(1_000)),
        "million" => Some(Word::Scale(1_000_000)),
        "billion" => Some(Word::Scale(1_000_000_000)),
        "trillion" => Some(Word::Scale(1_000_000_000_000)),
        "quadrillion" => Some(Word::Scale(1_000_000_000_000_000)),
        "quintillion" => Some(Word::Scale(1_000_000_000_000_000_000)),
        "and" => Some(Word::And),
        "minus" => Some(Word::Minus),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum Word {
    Number(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..i], start));
            start = i + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut words = Vec::with_capacity(tokens.len());
    for &(token, index) in &tokens {
        match dictionary_value(token) {
            Some(Word::Minus) if index != tokens[0].1 => {
                return Err(WordsError::InvalidWord { index });
            }
            Some(word) => words.push(word),
            None => return Err(WordsError::InvalidWord { index }),
        }
    }

    let negative = matches!(words.first(), Some(Word::Minus));
    let arithmetic = if negative { &words[1..] } else { &words[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for word in arithmetic {
        match *word {
            Word::Number(n) => group = group.checked_add(n).ok_or(WordsError::Range)?,
            Word::Hundred => group = group.checked_mul(100).ok_or(WordsError::Range)?,
            Word::Scale(scale) => {
                let scaled = group.checked_mul(scale).ok_or(WordsError::Range)?;
                total = total.checked_add(scaled).ok_or(WordsError::Range)?;
                group = 0;
            }
            Word::And => {}
            Word::Minus => unreachable!(),
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != input.as_bytes() {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(input.as_bytes())
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(value)
}
