#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const ONES: [&str; 20] = [
    "", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten",
    "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "quintillion",
    "quadrillion",
    "trillion",
    "billion",
    "million",
    "thousand",
    "",
];

fn append_group(value: u16, output: &mut String) {
    if value >= 100 {
        output.push_str(ONES[(value / 100) as usize]);
        output.push_str(" hundred");
        let remainder = value % 100;
        if remainder != 0 {
            output.push_str(" and ");
            append_group(remainder, output);
        }
    } else if value >= 20 {
        output.push_str(TENS[(value / 10) as usize]);
        if value % 10 != 0 {
            output.push('-');
            output.push_str(ONES[(value % 10) as usize]);
        }
    } else if value != 0 {
        output.push_str(ONES[value as usize]);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in groups.iter_mut().rev() {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let nonzero: Vec<usize> = groups
        .iter()
        .enumerate()
        .filter_map(|(index, &group)| (group != 0).then_some(index))
        .collect();

    for (position, &index) in nonzero.iter().enumerate() {
        if position != 0 {
            let group = groups[index];
            let use_and = position + 1 == nonzero.len()
                && index == 6
                && group < 100;
            result.push_str(if use_and { " and " } else { " " });
        }

        append_group(groups[index], &mut result);
        if !SCALES[index].is_empty() {
            result.push(' ');
            result.push_str(SCALES[index]);
        }
    }

    Ok(result)
}

#[derive(Clone, Copy)]
enum Word {
    Number(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(usize, &str, Word)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            let word = &input[start..index];
            let value = dictionary_value(word).ok_or(WordsError::InvalidWord { index: start })?;
            tokens.push((start, word, value));
            start = index + 1;
        }
    }
    let word = &input[start..];
    let value = dictionary_value(word).ok_or(WordsError::InvalidWord { index: start })?;
    tokens.push((start, word, value));

    let negative = matches!(tokens.first().map(|token| token.2), Some(Word::Minus));
    for (index, (position, _, word)) in tokens.iter().enumerate() {
        if matches!(word, Word::Minus) && index != 0 {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let arithmetic_start = usize::from(negative);
    let mut total = 0u128;
    let mut group = 0u128;

    for (_, _, word) in tokens.iter().skip(arithmetic_start) {
        match word {
            Word::Number(value) => group += value,
            Word::Hundred => group *= 100,
            Word::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
            Word::And | Word::Minus => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != input.as_bytes() {
        let differing = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(signed)
}
