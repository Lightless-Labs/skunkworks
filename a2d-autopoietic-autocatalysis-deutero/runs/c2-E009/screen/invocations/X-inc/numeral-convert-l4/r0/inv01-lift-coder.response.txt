#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten",
    "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(&str, u128); 6] = [
    ("thousand", 1_000),
    ("million", 1_000_000),
    ("billion", 1_000_000_000),
    ("trillion", 1_000_000_000_000),
    ("quadrillion", 1_000_000_000_000_000),
    ("quintillion", 1_000_000_000_000_000_000),
];

fn spell_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_owned();
    }
    if n < 100 {
        let tens = TENS[(n / 10 - 2) as usize];
        let units = n % 10;
        return if units == 0 {
            tens.to_owned()
        } else {
            format!("{tens}-{}", SMALL[units as usize])
        };
    }

    let hundreds = format!("{} hundred", SMALL[(n / 100) as usize]);
    let remainder = n % 100;
    if remainder == 0 {
        hundreds
    } else {
        format!("{hundreds} and {}", spell_under_thousand(remainder))
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_owned());
    }

    let mut remaining = magnitude;
    let mut groups: Vec<(u16, Option<&'static str>)> = Vec::new();

    for (scale_word, scale_value) in SCALES.iter().rev() {
        let group = remaining / *scale_value;
        if group != 0 {
            groups.push((group as u16, Some(*scale_word)));
            remaining %= *scale_value;
        }
    }
    if remaining != 0 {
        groups.push((remaining as u16, None));
    }

    let has_cross_group_and = groups.len() > 1
        && groups
            .last()
            .is_some_and(|(group, scale)| scale.is_none() && *group < 100);

    let mut parts = Vec::with_capacity(groups.len());
    for (group, scale) in groups {
        let mut part = spell_under_thousand(group);
        if let Some(scale) = scale {
            part.push(' ');
            part.push_str(scale);
        }
        parts.push(part);
    }

    let mut result = if has_cross_group_and {
        let last = parts.pop().expect("units group exists");
        format!("{} and {}", parts.join(" "), last)
    } else {
        parts.join(" ")
    };

    if value < 0 {
        result.insert_str(0, "minus ");
    }
    Ok(result)
}

fn dictionary_value(word: &str) -> Option<Word> {
    for (i, spelling) in SMALL.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Number(i as u128));
        }
    }
    for (i, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Number((i as u128 + 2) * 10));
        }
    }
    if word == "hundred" {
        return Some(Word::Hundred);
    }
    if word == "and" {
        return Some(Word::And);
    }
    for (spelling, value) in SCALES {
        if word == spelling {
            return Some(Word::Scale(value));
        }
    }
    if word == "minus" {
        return Some(Word::Minus);
    }
    None
}

#[derive(Clone, Copy)]
enum Word {
    Number(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut words = Vec::with_capacity(tokens.len());
    for &(token, index) in &tokens {
        let word = dictionary_value(token).ok_or(WordsError::InvalidWord { index })?;
        words.push((word, index));
    }

    let mut negative = false;
    let mut begin = 0;
    for (i, (word, index)) in words.iter().enumerate() {
        if matches!(word, Word::Minus) {
            if i == 0 {
                negative = true;
                begin = 1;
            } else {
                return Err(WordsError::InvalidWord { index: *index });
            }
        }
    }

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (word, _) in words.iter().skip(begin) {
        match word {
            Word::Number(value) => group += *value,
            Word::Hundred => group *= 100,
            Word::Scale(scale) => {
                total += group * *scale;
                group = 0;
            }
            Word::And => {}
            Word::Minus => unreachable!(),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let common = canonical
            .bytes()
            .zip(input.bytes())
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(value)
}
