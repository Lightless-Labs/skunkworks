```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn abs(&self) -> Self {
        let mut out = self.clone();
        out.negative = false;
        out
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len()).then_with(|| {
            self.limbs.iter().rev().cmp(rhs.limbs.iter().rev())
        })
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let x = *a.limbs.get(i).unwrap_or(&0) as u64;
            let y = *b.limbs.get(i).unwrap_or(&0) as u64;
            let sum = x + y + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self { negative: false, limbs: out }
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let x = a.limbs[i] as i64 - borrow;
            let y = *b.limbs.get(i).unwrap_or(&0) as i64;
            if x < y {
                out.push((x + Self::BASE as i64 - y) as u32);
                borrow = 1;
            } else {
                out.push((x - y) as u32);
                borrow = 0;
            }
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn mul_small_assign(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            *self = Self::zero();
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let v = *limb as u64 * factor as u64 + carry;
            *limb = (v % Self::BASE) as u32;
            carry = v / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_assign(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let v = self.limbs[i] as u64 + carry;
            self.limbs[i] = (v % Self::BASE) as u32;
            carry = v / Self::BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut out = vec![0; self.limbs.len()];
        let mut rem = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let cur = rem * Self::BASE + self.limbs[i] as u64;
            out[i] = (cur / divisor as u64) as u32;
            rem = cur % divisor as u64;
        }
        let mut q = Self { negative: false, limbs: out };
        q.normalize();
        (q, rem as u32)
    }

    fn mul_abs(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u32; a.limbs.len() + b.limbs.len()];
        for (i, &x) in a.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.limbs.iter().enumerate() {
                let k = i + j;
                let v = out[k] as u64 + x as u64 * y as u64 + carry;
                out[k] = (v % Self::BASE) as u32;
                carry = v / Self::BASE;
            }
            let mut k = i + b.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let v = out[k] as u64 + carry;
                out[k] = (v % Self::BASE) as u32;
                carry = v / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.cmp_abs(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }

        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut rem = Self::zero();
        for &limb in a.limbs.iter().rev() {
            if !rem.is_zero() {
                rem.limbs.insert(0, limb);
            } else if limb != 0 {
                rem.limbs.push(limb);
            }
            rem.normalize();

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = Self::mul_abs(b, &Self { negative: false, limbs: if mid == 0 { Vec::new() } else { vec![mid as u32] } });
                match product.cmp_abs(&rem) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }

            if best != 0 {
                let product = Self::mul_abs(b, &Self { negative: false, limbs: vec![best as u32] });
                rem = Self::sub_abs(&rem, &product);
            }
            quotient_be.push(best as u32);
        }
        quotient_be.reverse();
        let mut q = Self { negative: false, limbs: quotient_be };
        q.normalize();
        (q, rem)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let mut first_invalid = None;
        let mut digit_positions = Vec::new();
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            match Self::digit_value(bytes[i]) {
                Some(d) if d < radix => digit_positions.push((i, d)),
                _ => {
                    if first_invalid.is_none() {
                        first_invalid = Some(i);
                    }
                }
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                return Err(Error::Separator { index: i });
            }
        }

        if digit_positions.len() > 1 && digit_positions[0].1 == 0 {
            return Err(Error::LeadingZero { index: digit_positions[0].0 });
        }
        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits].0,
            });
        }

        let mut value = Self::zero();
        for &(_, digit) in &digit_positions {
            value.mul_small_assign(radix);
            value.add_small_assign(digit);
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0);
        } else {
            while !n.is_zero() {
                let (q, rem) = n.div_rem_small_abs(radix);
                digits.push(rem as u8);
                n = q;
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, &d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(if d < 10 { (b'0' + d) as char } else { (b'a' + d - 10) as char });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self::add_abs(self, rhs);
            out.negative = self.negative && !out.is_zero();
            out
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self::sub_abs(self, rhs);
                    out.negative = self.negative;
                    out
                }
                Ordering::Less => {
                    let mut out = Self::sub_abs(rhs, self);
                    out.negative = rhs.negative;
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut out = Self::mul_abs(self, rhs);
        out.negative = !out.is_zero() && (self.negative != rhs.negative);
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = Self::div_rem_abs(&self.abs(), &rhs.abs());
        q.negative = !q.is_zero() && (self.negative != rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut result = Self::one();
        let mut base = self.clone();
        let mut exp = exponent;
        while exp != 0 {
            if exp & 1 == 1 {
                result = result.mul(&base);
            }
            exp >>= 1;
            if exp != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, r) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = r.abs();
        }
        a
    }
}
```
