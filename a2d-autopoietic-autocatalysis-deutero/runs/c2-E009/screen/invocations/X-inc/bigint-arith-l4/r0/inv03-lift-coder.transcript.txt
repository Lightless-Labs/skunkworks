#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            digits: vec![1],
        }
    }

    fn from_digits(mut digits: Vec<u8>, negative: bool) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add_digits(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u16;
        for i in 0..a.len().max(b.len()) {
            let sum = carry
                + u16::from(*a.get(i).unwrap_or(&0))
                + u16::from(*b.get(i).unwrap_or(&0));
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        if carry != 0 {
            out.push(carry as u8);
        }
        out
    }

    // Requires a >= b.
    fn abs_sub_digits(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i16;
        for i in 0..a.len() {
            let mut value = i16::from(a[i]) - borrow - i16::from(*b.get(i).unwrap_or(&0));
            if value < 0 {
                value += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn abs_add(&self, rhs: &Self) -> Self {
        Self::from_digits(Self::abs_add_digits(&self.digits, &rhs.digits), false)
    }

    fn abs_sub(&self, rhs: &Self) -> Self {
        Self::from_digits(Self::abs_sub_digits(&self.digits, &rhs.digits), false)
    }

    fn mul_small_add(&mut self, multiplier: u32, addend: u32) {
        let mut carry = addend;
        for digit in &mut self.digits {
            let value = u32::from(*digit) * multiplier + carry;
            *digit = (value % 10) as u8;
            carry = value / 10;

            self.digits.push((carry % 10) as u8);
            carry /= 10;
        }
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
    }

    fn div_rem_abs(dividend: &Self, divisor: &Self) -> (Self, Self) {
        let mut quotient_be = Vec::with_capacity(dividend.digits.len());
        let mut remainder = Self::zero();

            Self::from_digits(remainder.digits, false),

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_values = Vec::new();
        let mut digit_offsets = Vec::new();
        let mut invalid = None;

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            let value = match byte {
                b'0'..=b'9' => u32::from(byte - b'0'),
                b'a'..=b'z' => u32::from(byte - b'a') + 10,
                b'A'..=b'Z' => u32::from(byte - b'A') + 10,
                _ => {
                    invalid.get_or_insert(index);
                    continue;
                }
            };
            if value >= radix {
                invalid.get_or_insert(index);
                continue;
            }
            digit_values.push(value as u8);
            digit_offsets.push(index);
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                let previous_valid = index > start
                    && bytes[index - 1] != b'_'
                    && digit_values_at_byte(bytes[index - 1], radix);
                let next_valid = index + 1 < bytes.len()
                    && bytes[index + 1] != b'_'
                    && digit_values_at_byte(bytes[index + 1], radix);
                if !previous_valid || !next_valid {
                    return Err(Error::Separator { index });
                }
            }
        }

        if digit_values.len() > 1 && digit_values[0] == 0 {
            return Err(Error::LeadingZero {
                index: digit_offsets[0],
            });
        }
        if digit_values.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_offsets[max_digits],
            });
        }

        let mut value = Self::zero();
        for digit in digit_values {
            value.mul_small_add(radix, u32::from(digit));
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.digits.clone();
        let mut output_digits = Vec::new();
        if value.len() == 1 && value[0] == 0 {
            output_digits.push(0u8);
        } else {
            while !(value.len() == 1 && value[0] == 0) {
                let mut remainder = 0u32;
                for digit in value.iter_mut().rev() {
                    let current = remainder * 10 + u32::from(*digit);
                    *digit = (current / radix) as u8;
                    remainder = current % radix;
                }
                output_digits.push(remainder as u8);
                while value.len() > 1 && value.last() == Some(&0) {
                    value.pop();
                }
            }
        }

        while output_digits.len() < min_digits.max(1) {
            output_digits.push(0);
        }

        let mut chars = Vec::with_capacity(output_digits.len());
        for digit in output_digits.iter().rev() {
            chars.push(if *digit < 10 {
                (b'0' + *digit) as char
            } else {
                (b'a' + (*digit - 10)) as char
            });
        }

        let mut result = String::new();
        if self.negative && !self.is_zero() {
            result.push('-');
        }
        for (i, ch) in chars.iter().enumerate() {
            if group > 0 && i > 0 && (chars.len() - i) % group == 0 {
                result.push('_');
            }
            result.push(*ch);
        }
        Ok(result)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = self.abs_add(rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Greater => {
                    let mut result = self.abs_sub(rhs);
                    result.negative = self.negative && !result.is_zero();
                    result
                }
                Ordering::Less => {
                    let mut result = rhs.abs_sub(self);
                    result.negative = rhs.negative && !result.is_zero();
                    result
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let index = i + j;
                let value = u16::from(out[index]) + u16::from(a) * u16::from(b) + carry;
                out[index] = (value % 10) as u8;
                carry = value / 10;
            }
            let mut index = i + rhs.digits.len();

                let value = u16::from(out[index]) + carry;
                out[index] = (value % 10) as u8;
                carry = value / 10;
                index += 1;
                if index == out.len() && carry != 0 {
                    out.push(0);
                }
            }
        }
        Self::from_digits(out, self.negative ^ rhs.negative)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::div_rem_abs(self, rhs);
        quotient.negative = (self.negative ^ rhs.negative) && !quotient.is_zero();
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power != 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self::from_digits(self.digits.clone(), false);
        let mut b = Self::from_digits(rhs.digits.clone(), false);
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = Self::from_digits(remainder.digits, false);
        }
        a
    }
