#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn format_absolute(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return "/".to_string();
    }
    let mut result = format!("/{}", parts.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn format_normalized(parts: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return format_absolute(parts, intent);
    }
    if parts.is_empty() {
        return ".".to_string();
    }
    let mut result = parts.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|part| part != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        }
        _ => stack.push(component.to_string()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        let start = offset;
        offset += component.len() + 1;
        if component.is_empty() {
            continue;
        }
        reduce_component(&mut stack, component, absolute, start)?;
    }

    Ok(format_normalized(&stack, absolute, intent))
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_string).collect()),
        Err(_) => Err(error),
    }
}

fn tilde_remainder(path: &str, home: Option<&str>) -> Result<Option<&str>, PathError> {
    if !path.starts_with('~') {
        return Ok(None);
    }
    let rest = &path[1..];
    if !rest.is_empty() && !rest.starts_with('/') {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if home.is_none() {
        return Err(PathError::MissingHome);
    }
    Ok(Some(rest.strip_prefix('/').unwrap_or(rest)))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);
    let tilde_rest = tilde_remainder(path, home)?;

    let (mut stack, request, absolute) = if let Some(rest) = tilde_rest {
        (home_parts.unwrap(), rest, true)
    } else if path.starts_with('/') {
        (Vec::new(), path, true)
    } else {
        (cwd_parts, path, true)
    };

    let mut offset = if tilde_rest.is_some() { path.len() - request.len() } else { 0 };
    for component in request.split('/') {
        let at = offset;
        offset += component.len() + 1;
        if component.is_empty() {
            continue;
        }
        reduce_component(&mut stack, component, absolute, at)?;
    }

    Ok(format_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !path_valid || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(first) = links[..index].iter().position(|prior| prior.path == link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct WorkComponent {
    text: String,
    at: usize,
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let tilde_rest = tilde_remainder(path, home)?;

    let (mut stack, request, request_base, is_absolute) = if let Some(rest) = tilde_rest {
        (home_parts.unwrap(), rest, path.len() - rest.len(), false)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_parts, path, 0, false)
    };

    let mut queue: Vec<WorkComponent> = Vec::new();
    if !is_absolute {
        // Context components have offset zero and are traversed before request components.
        // The stack already represents the context; lookups are unnecessary because validated
        // contexts are lexical directories, not requests to expand.
    }

    let mut offset = request_base;
    for component in request.split('/') {
        let at = offset;
        offset += component.len() + 1;
        if !component.is_empty() {
            queue.push(WorkComponent {
                text: component.to_string(),
                at,
            });
        }
    }

    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;
    let mut frames: Vec<(Vec<WorkComponent>, usize)> = Vec::new();

    while !queue.is_empty() || !frames.is_empty() {
        if queue.is_empty() {
            let (remaining, link_index) = frames.pop().unwrap();
            active.pop();
            queue = remaining;
            let _ = link_index;
            continue;
        }

        let item = queue.remove(0);
        match item.text.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: item.at });
                }
            }
            _ => {
                stack.push(item.text);
                let current = format!("/{}", stack.join("/"));
                if let Some((index, link)) = links
                    .iter()
                    .enumerate()
                    .find(|(_, link)| link.path == current)
                {
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    expansions += 1;

                    stack.pop();
                    active.push(index);

                    let mut target_components = Vec::new();
                    let target_absolute = link.target.starts_with('/');
                    let mut target_offset = 0;
                    for component in link.target.split('/') {
                        let at = target_offset;
                        target_offset += component.len() + 1;
                        if !component.is_empty() {
                            target_components.push(WorkComponent {
                                text: component.to_string(),
                                at,
                            });
                        }
                    }

                    let remaining = std::mem::take(&mut queue);
                    frames.push((remaining, index));
                    if target_absolute {
                        stack.clear();
                    }
                    queue = target_components;
                }
            }
        }
    }

    Ok(format_absolute(&stack, intent))
}
