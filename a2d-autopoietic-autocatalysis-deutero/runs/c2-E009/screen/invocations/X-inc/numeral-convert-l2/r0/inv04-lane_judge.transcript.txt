    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },

    LeadingZero { operand: usize, index: usize },

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(hash) = input.as_bytes().iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input.as_bytes()[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
        if payload_start == input.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let bytes = input.as_bytes();
    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        let value = match byte {
            b'0'..=b'9' => byte - b'0',
            b'A'..=b'Z' => byte - b'A' + 10,
            b'a'..=b'z' => byte - b'a' + 10,
            _ => u8::MAX,
        };
        if value >= radix {
            return Err(BaseError::InvalidDigit {
                operand,
                index: payload_start + offset,
            });
        }
        digits.push((payload_start + offset, value));
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let left_is_digit = offset > 0 && payload[offset - 1] != b'_';
            let right_is_digit = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !left_is_digit || !right_is_digit {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0].1 == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits.into_iter().map(|(_, value)| value).collect())
}

fn multiply_add(number: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for digit in number.iter_mut() {
        let value = *digit as u16 * multiplier as u16 + carry;
        *digit = (value & 0xff) as u8;
        carry = value >> 8;
    }
    while carry != 0 {
        number.push((carry & 0xff) as u8);
        carry >>= 8;
    }
}

fn add_numbers(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for i in 0..left.len().max(right.len()) {
        let sum = left.get(i).copied().unwrap_or(0) as u16
            + right.get(i).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum & 0xff) as u8);
        carry = sum >> 8;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn to_radix_digits(mut value: Vec<u8>, radix: u8) -> Vec<u8> {
    if value.iter().all(|&byte| byte == 0) {
        return vec![0];
    }

    let mut digits = Vec::new();
    while value.iter().any(|&byte| byte != 0) {
        let mut remainder = 0u16;
        for byte in value.iter_mut().rev() {
            let current = (remainder << 8) | *byte as u16;
            *byte = (current / radix as u16) as u8;
            remainder = current % radix as u16;
        }
        digits.push(remainder as u8);
    }
    digits.reverse();
    digits
}

fn format_digits(mut digits: Vec<u8>, style: BaseStyle) -> String {
    if digits.len() < style.min_width {
        let mut padded = vec![0; style.min_width - digits.len()];
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::new();
    for (index, digit) in digits.iter().copied().enumerate() {
        if style.group != 0 && index > 0 && (digits.len() - index) % style.group == 0 {

            _ if style.upper => b'A' + (digit - 10),

}

}

fn to_base(mut number: Vec<u8>, radix: u8) -> Vec<u8> {
    while number.first() == Some(&0) && number.len() > 1 {
        number.remove(0);
    }
    if number.is_empty() {
        return vec![0];
    }

    let mut output = Vec::new();
    while number.iter().any(|&digit| digit != 0) {
        let mut quotient = Vec::with_capacity(number.len());
        let mut remainder = 0u16;
        for &digit in &number {
            let current = remainder * 256 + digit as u16;
            let q = current / radix as u16;
            remainder = current % radix as u16;
            if !quotient.is_empty() || q != 0 {
                quotient.push(q as u8);
            }
        }
        output.push(remainder as u8);
        number = if quotient.is_empty() { vec![0] } else { quotient };
    }
    output.reverse();
    output
}

fn render(digits: Vec<u8>, style: BaseStyle) -> String {
    let mut chars: Vec<u8> = digits
        .into_iter()
        .map(|digit| match digit {
            0..=9 => b'0' + digit,
            10..=35 => {
                if style.upper {
                    b'A' + (digit - 10)
                } else {
                    b'a' + (digit - 10)
                }
            }
            _ => unreachable!(),
        })
        .collect();

    if chars.len() < style.min_width {
        let padding = style.min_width - chars.len();
        let mut padded = vec![b'0'; padding];
        padded.extend(chars);
        chars = padded;
    }

    if style.group == 0 {
        return String::from_utf8(chars).expect("ASCII output");
    }

    let mut output = Vec::with_capacity(chars.len() + chars.len() / style.group);
    let first_group = chars.len() % style.group;
    for (index, byte) in chars.into_iter().enumerate() {
        if index != 0 && (index - first_group) % style.group == 0 {
            output.push(b'_');
        }
        output.push(byte);
    }
    String::from_utf8(output).expect("ASCII output")
}

fn convert_to_bytes(digits: &[u8], from: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        multiply_add(&mut value, from, digit);
    }
    value
}

    right: &str,
    base: u8,
    style: BaseStyle,

    }
    digits.reverse();

    let mut out = String::with_capacity(digits.len() + digits.len() / style.group.max(1));
    for (i, digit) in digits.iter().enumerate() {
        if style.group != 0 && i != 0 && (digits.len() - i) % style.group == 0 {
            out.push('_');
        }
        let ch = match *digit {
            0..=9 => (b'0' + *digit) as char,
            n if style.upper => (b'A' + n - 10) as char,
            n => (b'a' + n - 10) as char,
        };
        out.push(ch);
    }
    out
}

fn parse_to_integer(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    value
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input[..hash] != expected {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn add_small(number: &mut Vec<u8>, value: u8, radix: u8) {
    let mut carry = value as u16;
    let base = radix as u16;
    let mut index = number.len();

    while carry > 0 && index > 0 {
        index -= 1;
        let sum = number[index] as u16 + carry;
        number[index] = (sum % base) as u8;
        carry = sum / base;
    }

    while carry > 0 {
        number.insert(0, (carry % base) as u8);
        carry /= base;
    }
}

fn multiply_small(number: &mut Vec<u8>, factor: u8, radix: u8) {
    let base = radix as u16;
    let mut carry = 0u16;

    for digit in number.iter_mut().rev() {
        let product = *digit as u16 * factor as u16 + carry;
        *digit = (product % base) as u8;
        carry = product / base;
    }

    while carry > 0 {
        number.insert(0, (carry % base) as u8);
        carry /= base;
    }
}

fn add_numbers(left: &[u8], right: &[u8], radix: u8) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u16;
    let base = radix as u16;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry > 0 {
        let mut sum = carry;
        if i > 0 {
            i -= 1;
            sum += left[i] as u16;
        }
        if j > 0 {
            j -= 1;
            sum += right[j] as u16;
        }
        result.push((sum % base) as u8);
        carry = sum / base;
    }

    if result.is_empty() {
        result.push(0);
    }
    result.reverse();
    result
}

fn render(mut digits: Vec<u8>, style: BaseStyle) -> String {
    while digits.len() > 1 && digits[0] == 0 {
        digits.remove(0);
    }

    if digits.len() < style.min_width {
        let padding = style.min_width - digits.len();
        let mut padded = Vec::with_capacity(style.min_width);
        padded.resize(padding, 0);
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::with_capacity(digits.len() + digits.len() / 2);
    for (index, digit) in digits.iter().enumerate() {
        if style.group != 0 && index > 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        let byte = match *digit {
            0..=9 => b'0' + *digit,
            value if style.upper => b'A' + (value - 10),
            value => b'a' + (value - 10),
        };
        output.push(byte as char);
    }
    output
}

fn convert_digits(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_small(&mut value, from, to);
        add_small(&mut value, digit, to);
    }
    value
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let converted = convert_digits(&digits, from, to);
    Ok(render(converted, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let sum = add_numbers(&left_digits, &right_digits, base);
    Ok(render(sum, style))
}
