#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input[..hash] != expected {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn add_small(number: &mut Vec<u8>, value: u8, radix: u8) {
    let mut carry = value as u16;
    let base = radix as u16;
    let mut index = number.len();

    while carry > 0 && index > 0 {
        index -= 1;
        let sum = number[index] as u16 + carry;
        number[index] = (sum % base) as u8;
        carry = sum / base;
    }

    while carry > 0 {
        number.insert(0, (carry % base) as u8);
        carry /= base;
    }
}

fn multiply_small(number: &mut Vec<u8>, factor: u8, radix: u8) {
    let base = radix as u16;
    let mut carry = 0u16;

    for digit in number.iter_mut().rev() {
        let product = *digit as u16 * factor as u16 + carry;
        *digit = (product % base) as u8;
        carry = product / base;
    }

    while carry > 0 {
        number.insert(0, (carry % base) as u8);
        carry /= base;
    }
}

fn add_numbers(left: &[u8], right: &[u8], radix: u8) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u16;
    let base = radix as u16;
    let mut i = left.len();
    let mut j = right.len();

    while i > 0 || j > 0 || carry > 0 {
        let mut sum = carry;
        if i > 0 {
            i -= 1;
            sum += left[i] as u16;
        }
        if j > 0 {
            j -= 1;
            sum += right[j] as u16;
        }
        result.push((sum % base) as u8);
        carry = sum / base;
    }

    if result.is_empty() {
        result.push(0);
    }
    result.reverse();
    result
}

fn render(mut digits: Vec<u8>, style: BaseStyle) -> String {
    while digits.len() > 1 && digits[0] == 0 {
        digits.remove(0);
    }

    if digits.len() < style.min_width {
        let padding = style.min_width - digits.len();
        let mut padded = Vec::with_capacity(style.min_width);
        padded.resize(padding, 0);
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::with_capacity(digits.len() + digits.len() / 2);
    for (index, digit) in digits.iter().enumerate() {
        if style.group != 0 && index > 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        let byte = match *digit {
            0..=9 => b'0' + *digit,
            value if style.upper => b'A' + (value - 10),
            value => b'a' + (value - 10),
        };
        output.push(byte as char);
    }
    output
}

fn convert_digits(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value = vec![0];
    for &digit in digits {
        multiply_small(&mut value, from, to);
        add_small(&mut value, digit, to);
    }
    value
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let converted = convert_digits(&digits, from, to);
    Ok(render(converted, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let sum = add_numbers(&left_digits, &right_digits, base);
    Ok(render(sum, style))
}
