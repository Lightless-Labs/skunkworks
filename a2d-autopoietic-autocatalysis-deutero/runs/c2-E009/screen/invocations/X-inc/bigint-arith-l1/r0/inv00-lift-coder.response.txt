```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian decimal digits; zero is represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan the complete magnitude for invalid bytes before checking separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let magnitude = &bytes[start..];

        // A separator must have an immediate digit on both sides.
        for (offset, &byte) in magnitude.iter().enumerate() {
            if byte == b'_'
                && (offset == 0
                    || offset + 1 == magnitude.len()
                    || !magnitude[offset - 1].is_ascii_digit()
                    || !magnitude[offset + 1].is_ascii_digit())
            {
                return Err(Error::Separator {
                    index: start + offset,
                });
            }
        }

        let first_digit = magnitude.iter().position(|&byte| byte != b'_').unwrap();
        let first_digit_index = start + first_digit;
        let digit_count = magnitude.iter().filter(|&&byte| byte != b'_').count();

        if digit_count > 1 && magnitude[first_digit] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let excess_offset = magnitude
                .iter()
                .enumerate()
                .filter(|(_, &byte)| byte != b'_')
                .nth(excess_number)
                .map(|(offset, _)| offset)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: start + excess_offset,
            });
        }

        let digits = magnitude
            .iter()
            .filter(|&&byte| byte != b'_')
            .map(|&byte| byte - b'0')
            .rev()
            .collect();

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = magnitude_len.max(min_digits);
        let padding = width - magnitude_len;
        let mut output = String::with_capacity(width + width / group.max(1));

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                output.push('_');
            }

            if position < padding {
                output.push('0');
            } else {
                let digit_index = width - 1 - position - padding;
                output.push((b'0' + self.digits[digit_index]) as char);
            }
        }

        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => self.digits.iter().rev().cmp(rhs.digits.iter().rev()),
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let max_len = self.digits.len().max(rhs.digits.len());
        let mut result = Vec::with_capacity(max_len + 1);
        let mut carry = 0u8;

        for index in 0..max_len {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            result.push(carry);
        }

        Self { digits: result }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;

        for index in 0..self.digits.len() {
            let left = self.digits[index] as i16 - borrow;
            let right = rhs.digits.get(index).copied().unwrap_or(0) as i16;
            if left < right {
                result.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                result.push((left - right) as u8);
                borrow = 0;
            }
        }

        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }

        Ok(Self { digits: result })
    }
}
```
