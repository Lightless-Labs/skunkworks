pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let task_count = tasks.len();

    // Validate all dependency entries before constructing or traversing the graph.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    // Edges are stored from predecessor to child. Sorting each child list is
    // unnecessary because the available set determines the Kahn tie-break.
    let mut children = vec![Vec::new(); task_count];
    let mut indegree = vec![0usize; task_count];

    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Keep the currently available IDs ordered so each removal selects the
    // smallest ID, including newly unblocked IDs.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(task_count);
    let mut remaining_indegree = indegree;

    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != task_count {
        let blocked = remaining_indegree
            .iter()
            .enumerate()
            .filter_map(|(task_id, &degree)| (degree != 0).then_some(task_id))
            .collect();
        return Err(Error::Cycle { blocked });
    }

    let mut times = vec![Timing { start: 0, finish: 0 }; task_count];
    let mut makespan = 0;

    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&predecessor| times[predecessor].finish)
            .max()
            .unwrap_or(0);

        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;

        times[task_id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}

                    position,

                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }

            children[dependency].push(task_id);
        }

    // available task is always removed first.

        if degree == 0 {
            available.insert(task_id);
        }

    let mut remaining_indegree = indegree;
    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);

        for &child in &children[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != task_count {
        let removed: Vec<bool> = {
            let mut marks = vec![false; task_count];
            for &task_id in &order {
                marks[task_id] = true;
            }
            marks
        };
        let blocked = (0..task_count).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut starts = vec![0u64; task_count];
    let mut times = vec![Timing { start: 0, finish: 0 }; task_count];
    let mut makespan = 0u64;

    for &task_id in &order {
        let start = starts[task_id];

            .ok_or(Error::Overflow { task: task_id })?;

            starts[child] = starts[child].max(finish);
        }
    }

        order,
        times,
        makespan,
    })

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,

                    task: task_id,
                    position,

                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    // Build successor lists and indegrees. A dependency names a predecessor.
    let mut children = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Keep available IDs sorted so each step removes the smallest one.
    let mut available = Vec::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut next_indegree = indegree;
    while !available.is_empty() {
        let id = available.remove(0);
        order.push(id);

        for &child in &children[id] {
            next_indegree[child] -= 1;
            if next_indegree[child] == 0 {
                let insert_at = available.partition_point(|&available_id| available_id < child);
                available.insert(insert_at, child);
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut starts = vec![0u64; n];
    let mut times = vec![Timing { start: 0, finish: 0 }; n];
    let mut makespan = 0u64;

    for &id in &order {
        let start = starts[id];

        order,

    // Validate every dependency before constructing or traversing the graph.
    for (task_id, task) in tasks.iter().enumerate() {

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });

                    task: task_id,
                    position,
                });

            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    // Build predecessor-to-child adjacency and indegrees.
    let mut children = vec![Vec::new(); count];

            children[predecessor].push(task_id);

    // first, including when a newly unblocked ID is smaller than others.
    let mut available = std::collections::BinaryHeap::new();

            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }

        let mut removed = vec![false; count];
        for &task_id in &order {
            removed[task_id] = true;
        }
        let blocked = (0..count).filter(|&id| !removed[id]).collect();

        times,
