#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn value(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|i| i as u8)
}

fn encode_group(input: &[u8], output: &mut String) {
    let a = input[0];
    let b = if input.len() > 1 { input[1] } else { 0 };
    let c = if input.len() > 2 { input[2] } else { 0 };

    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if input.len() > 1 {
        output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        output.push('=');
    }

    if input.len() > 2 {
        output.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        output.push('=');
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let mut output = String::with_capacity(complete_len / 3 * 4);
        for group in combined[..complete_len].chunks_exact(3) {
            encode_group(group, &mut output);
        }

        self.tail.clear();
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        if !self.tail.is_empty() {
            encode_group(&self.tail, &mut output);
            self.tail.clear();
        }
        Ok(output)
    }
}

#[derive(Clone, Copy)]
enum DecodeState {
    Collecting,
    NeedSecondPadding,
    TerminalPadding,
}

pub struct Decoder {
    group: [u8; 4],
    group_len: usize,
    group_start: usize,
    offset: usize,
    state: DecodeState,
    sticky_error: Option<Error>,
    finished: bool,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            group_len: 0,
            group_start: 0,
            offset: 0,
            state: DecodeState::Collecting,
            sticky_error: None,
            finished: false,
        }
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    fn decode_complete_group(&mut self, output: &mut Vec<u8>) {
        let a = value(self.group[0]).unwrap();
        let b = value(self.group[1]).unwrap();

        output.push((a << 2) | (b >> 4));

        if self.group[2] != b'=' {
            let c = value(self.group[2]).unwrap();
            output.push((b << 4) | (c >> 2));

            if self.group[3] != b'=' {
                let d = value(self.group[3]).unwrap();
                output.push((c << 6) | d);
            }
        }

        if self.group[2] == b'=' || self.group[3] == b'=' {
            self.state = DecodeState::TerminalPadding;
        }
        self.group_len = 0;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            match self.state {
                DecodeState::TerminalPadding => {
                    return self.fail(Error::BadPadding {
                        offset: current_offset,
                    });
                }
                DecodeState::NeedSecondPadding => {
                    if byte != b'=' {
                        return self.fail(Error::BadPadding {
                            offset: current_offset,
                        });
                    }
                    self.group[3] = byte;
                    self.state = DecodeState::TerminalPadding;
                    self.decode_complete_group(&mut output);
                }
                DecodeState::Collecting => {
                    let position = self.group_len;

                    if byte == b'=' {
                        match position {
                            0 | 1 => {
                                return self.fail(Error::BadPadding {
                                    offset: current_offset,
                                });
                            }
                            2 => {
                                self.group[2] = byte;
                                self.group[3] = 0;
                                self.group_len = 3;
                                self.state = DecodeState::NeedSecondPadding;
                            }
                            3 => {
                                self.group[3] = byte;
                                self.decode_complete_group(&mut output);
                            }
                            _ => unreachable!(),
                        }
                    } else if let Some(_) = value(byte) {
                        if position == 3 && self.group[2] == b'=' {
                            return self.fail(Error::BadPadding {
                                offset: current_offset,
                            });
                        }

                        self.group[position] = byte;
                        self.group_len += 1;
                        if self.group_len == 4 {
                            self.decode_complete_group(&mut output);
                        }
                    } else {
                        return self.fail(Error::InvalidByte {
                            offset: current_offset,
                            byte,
                        });
                    }
                }
            }

            if self.group_len == 0 && !matches!(self.state, DecodeState::NeedSecondPadding) {
                self.group_start = self.offset;
            } else if self.group_len == 1 {
                self.group_start = current_offset;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.state {
            DecodeState::NeedSecondPadding => {
                return self.fail(Error::BadPadding {
                    offset: self.group_start + 2,
                });
            }
            DecodeState::Collecting if self.group_len != 0 => {
                return self.fail(Error::Truncated {
                    offset: self.group_start,
                });
            }
            _ => {}
        }

        self.finished = true;
        Ok(Vec::new())
    }
}
