const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut pos = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && pos < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[pos];
                }
                self.tail_len += 1;
                pos += 1;
            }

            if self.tail_len == 3 {
                let a = self.tail[0];
                let b = self.tail[1];
                let c = input[pos - 1];
                encode_triplet(a, b, c, &mut output);
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while input.len() - pos >= 3 {
            encode_triplet(input[pos], input[pos + 1], input[pos + 2], &mut output);
            pos += 3;
        }

        let remaining = input.len() - pos;
        self.tail_len = remaining;
        self.tail[..remaining].copy_from_slice(&input[pos..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            0 => {}
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => unreachable!(),
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn encode_triplet(a: u8, b: u8, c: u8, output: &mut String) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

fn decode_digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    pending_padding: Option<usize>,
    terminal: bool,
    next_offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            pending_padding: None,
            terminal: false,
            next_offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            let result = self.process_byte(byte, offset, &mut output);
            if let Err(error) = result {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn process_byte(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal {
            return Err(Error::BadPadding { offset });
        }

        if let Some(padding_offset) = self.pending_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_padding = None;
            self.group[3] = b'=';
            self.offsets[3] = offset;
            self.group_len = 4;
            self.terminal = true;
            let _ = padding_offset;
            output.push((self.group[0] << 2) | (self.group[1] >> 4));
            return Ok(());
        }

        let position = self.group_len;
        if byte == b'=' {
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_padding = Some(offset);
                    return Ok(());
                }
                3 => {
                    self.group[3] = b'=';
                    self.offsets[3] = offset;
                    self.group_len = 4;
                    self.terminal = true;
                    output.push((self.group[0] << 2) | (self.group[1] >> 4));
                    output.push((self.group[1] << 4) | (self.group[2] >> 2));
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let digit = decode_digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
        self.group[position] = digit;
        self.offsets[position] = offset;
        self.group_len += 1;

        if self.group_len == 4 {
            output.push((self.group[0] << 2) | (self.group[1] >> 4));
            output.push((self.group[1] << 4) | (self.group[2] >> 2));
            output.push((self.group[2] << 6) | self.group[3]);
            self.group_len = 0;
        }
        Ok(())
    }
}
