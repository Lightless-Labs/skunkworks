#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8]) -> String {
    let mut out = String::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        out.push_str(&encode_group(chunk));
    }

    let tail = chunks.remainder();
    if !tail.is_empty() {
        out.push_str(&encode_group(tail));
    }
    out
}

pub struct Encoder {
    pending: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            pending: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.pending.len() + input.len());
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let mut out = String::with_capacity(complete_len / 3 * 4);
        for chunk in combined[..complete_len].chunks_exact(3) {
            out.push_str(&encode_group(chunk));
        }

        self.pending.clear();
        self.pending.extend_from_slice(&combined[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let out = if self.pending.is_empty() {
            String::new()
        } else {
            encode_group(&self.pending)
        };
        self.pending.clear();
        self.finished = true;
        Ok(out)
    }
}

#[derive(Clone, Copy)]
enum DecodeState {
    Open,
    PendingThirdPadding { group_start: usize, padding_offset: usize },
    Terminal,
}

pub struct Decoder {
    group: [u8; 4],
    group_offsets: [usize; 4],
    len: usize,
    next_offset: usize,
    state: DecodeState,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            group_offsets: [0; 4],
            len: 0,
            next_offset: 0,
            state: DecodeState::Open,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            match self.state {
                DecodeState::Terminal => {
                    return Err(self.fail(Error::BadPadding { offset }));
                }
                DecodeState::PendingThirdPadding {
                    group_start: _,
                    padding_offset,
                } => {
                    if byte != b'=' {
                        return Err(self.fail(Error::BadPadding { offset }));
                    }
                    self.group[3] = b'=';
                    self.group_offsets[3] = offset;
                    out.push((self.group[0] << 2) | (self.group[1] >> 4));
                    self.len = 0;
                    self.state = DecodeState::Terminal;
                    let _ = padding_offset;
                }
                DecodeState::Open => {
                    let position = self.len;
                    if position < 2 && byte == b'=' {
                        return Err(self.fail(Error::BadPadding { offset }));
                    }
                    if position == 2 && byte == b'=' {
                        self.group[2] = b'=';
                        self.group_offsets[2] = offset;
                        self.len = 3;
                        self.state = DecodeState::PendingThirdPadding {
                            group_start: self.group_offsets[0],
                            padding_offset: offset,
                        };
                        continue;
                    }

                    if position == 3 && byte == b'=' {
                        if self.group[2] == b'=' {
                            return Err(self.fail(Error::BadPadding { offset }));
                        }
                        self.group[3] = b'=';
                        self.group_offsets[3] = offset;
                        let a = digit(self.group[0]).unwrap();
                        let b = digit(self.group[1]).unwrap();
                        let c = digit(self.group[2]).unwrap();
                        out.push((a << 2) | (b >> 4));
                        out.push((b << 4) | (c >> 2));
                        self.len = 0;
                        self.state = DecodeState::Terminal;
                        continue;
                    }

                    let Some(value) = digit(byte) else {
                        return Err(self.fail(Error::InvalidByte { offset, byte }));
                    };
                    self.group[position] = value;
                    self.group_offsets[position] = offset;
                    self.len += 1;

                    if self.len == 4 {
                        let a = self.group[0];
                        let b = self.group[1];
                        let c = self.group[2];
                        let d = self.group[3];
                        out.push((a << 2) | (b >> 4));
                        out.push((b << 4) | (c >> 2));
                        out.push((c << 6) | d);
                        self.len = 0;
                    }
                }
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.state {
            DecodeState::PendingThirdPadding { padding_offset, .. } => {
                let error = Error::BadPadding {
                    offset: padding_offset,
                };
                Err(self.fail(error))
            }
            _ if self.len != 0 => {
                let error = Error::Truncated {
                    offset: self.group_offsets[0],
                };
                Err(self.fail(error))
            }
            _ => {
                self.finished = true;
                Ok(Vec::new())
            }
        }
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
