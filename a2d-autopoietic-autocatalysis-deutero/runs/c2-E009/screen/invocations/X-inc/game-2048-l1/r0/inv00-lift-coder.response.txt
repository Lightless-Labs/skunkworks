```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && !(2..=2_147_483_648).contains(&value).then(|| value.is_power_of_two()).unwrap_or(false) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Box<dyn Iterator<Item = usize>> = match side {
        Side::Left => Box::new(0..row.len()),
        Side::Right => Box::new((0..row.len()).rev()),
    };

    let mut tiles: Vec<(usize, u32)> = indices
        .filter_map(|index| (row[index] != 0).then_some((index, row[index])))
        .collect();

    let mut cells = vec![0; row.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut output_position = 0usize;
    let mut i = 0usize;

    while i < tiles.len() {
        if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
            let (first, value) = tiles[i];
            let (second, _) = tiles[i + 1];
            let Some(merged_value) = value.checked_mul(2) else {
                return Err(Error::MergeOverflow { first, second });
            };

            let to = match side {
                Side::Left => output_position,
                Side::Right => row.len() - 1 - output_position,
            };
            cells[to] = merged_value;
            merges.push(Merge {
                from: [first, second],
                to,
                value: merged_value,
            });
            score += u64::from(merged_value);
            i += 2;
        } else {
            let (_, value) = tiles[i];
            let to = match side {
                Side::Left => output_position,
                Side::Right => row.len() - 1 - output_position,
            };
            cells[to] = value;
            i += 1;
        }
        output_position += 1;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
```
