#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_wildcards: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn error(field: Field, offset: usize, kind: fn(Field, usize) -> ParseError) -> ParseError {
    kind(field, offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax { field, offset: start });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn parse_field(input: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_domain(field);
    let width = maximum - minimum + 1;
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let has_step = !slash_positions.is_empty();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let base_end = slash_positions.first().copied().unwrap_or(item.len());
        let step_start = if has_step { base_end + 1 } else { item.len() };

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphen_positions: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if is_star && hyphen_positions.is_empty() {
            0 // star
        } else if hyphen_positions.is_empty() && !base.is_empty()
            && base.iter().all(u8::is_ascii_digit)
        {
            1 // number
        } else if hyphen_positions.len() == 1 {
            let hyphen = hyphen_positions[0];
            if hyphen > 0
                && hyphen + 1 < base.len()
                && base[..hyphen].iter().all(u8::is_ascii_digit)
                && base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                2 // range
            } else {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        if has_step && (step_start == item.len() || !item[step_start..].iter().all(u8::is_ascii_digit)) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (range_start, range_end, end_offset) = match base_kind {
            0 => (minimum, maximum, item_start),
            1 => {
                let value = parse_number(base, 0, base.len(), field)?;
                if value < minimum || value > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start,
                    });
                }
                if has_step {
                    (value, maximum, item_start)
                } else {
                    (value, value, item_start)
                }
            }
            _ => {
                let hyphen = hyphen_positions[0];
                let start = parse_number(base, 0, hyphen, field)?;
                let end = parse_number(base, hyphen + 1, base.len(), field)?;
                let start_offset = item_start;
                let end_number_offset = item_start + hyphen + 1;

                if start < minimum || start > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: start_offset,
                    });
                }
                if end < minimum || end > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: end_number_offset,
                    });
                }
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_number_offset,
                    });
                }
                (start, end, end_number_offset)
            }
        };

        let step = if has_step {
            let step_value = parse_number(item, step_start, item.len(), field)
                .map_err(|e| match e {
                    ParseError::LeadingZero { .. } => ParseError::LeadingZero {
                        field,
                        offset: item_start + step_start,
                    },
                    ParseError::NumberOverflow { .. } => ParseError::NumberOverflow {
                        field,
                        offset: item_start + step_start,
                    },
                    _ => e,
                })?;
            if step_value == 0 || step_value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + step_start,
                });
            }
            step_value
        } else {
            1
        };

        let mut value = range_start;
        while value <= range_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        let _ = end_offset;

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_wildcards = [false; 2];

    for (index, (&part, &field)) in parts.iter().zip(fields.iter()).enumerate() {
        selected[index] = parse_field(part, field)?;
        if field == Field::DayOfMonth || field == Field::DayOfWeek {
            day_wildcards[if field == Field::DayOfMonth { 0 } else { 1 }] = part == "*";
        }
    }

    Ok(Cron {
        selected,
        day_wildcards,
    })
}
