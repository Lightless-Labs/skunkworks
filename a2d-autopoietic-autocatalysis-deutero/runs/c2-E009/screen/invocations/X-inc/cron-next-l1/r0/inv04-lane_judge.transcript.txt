    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
    width: u32,
    min: u32,
    max: u32,
) -> Result<u32, ParseError> {

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))

            return Err(ParseError::BadStep { field, offset });
        }
    } else if value < min || value > max {

    }

    Ok(value)
}

        }
    } else {
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(error(3));
        }
    }
    Ok(value)
}

        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    enum Base {
        Star,
        Single(u32, usize),
        Range(u32, usize, u32, usize),
    }

    let parsed_base = if is_star {
        Base::Star
    } else if hyphens.is_empty() {
        if base.is_empty() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let n = parse_number(bytes, item_start, item_start + base.len(), field)?;
        let (lo, hi) = domain(field);
        if n < lo as u32 || n > hi as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_start,
            });
        }
        Base::Single(n, item_start)
    } else if hyphens.len() == 1 {
        let h = hyphens[0];
        if h == 0 || h + 1 == base.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let left_start = item_start;
        let left_end = item_start + h;
        let right_start = left_end + 1;
        let right_end = item_start + base.len();
        let start = parse_number(bytes, left_start, left_end, field)?;
        let end = parse_number(bytes, right_start, right_end, field)?;
        let (lo, hi) = domain(field);
        if start < lo as u32 || start > hi as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: left_start,
            });
        }
        if end < lo as u32 || end > hi as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: right_start,
            });
        }
        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: right_start,
            });
        }
        Base::Range(start, left_start, end, right_start)
    } else {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    };

    let step = if let Some(start) = step_start {
        let end = item_end;
        let n = parse_number(bytes, start, end, field)?;
        if n == 0 || n > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
        Some(n)
    } else {
        None
    };

    match parsed_base {
        Base::Star => {
            let step = step.unwrap_or(1);
            let mut value = min as u32;
            while value <= max as u32 {
                selected[value as usize] = true;
                value += step;
            }
        }
        Base::Single(n, _) => {
            if let Some(step) = step {
                let mut value = n;
                while value <= max as u32 {
                    selected[value as usize] = true;
                    value += step;
                }
            } else {
                selected[n as usize] = true;
            }
        }
        Base::Range(start, _, end, _) => {
            let step = step.unwrap_or(1);
            let mut value = start;
            while value <= end {
                selected[value as usize] = true;
                value += step;
            }
        }
    }

    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, max) = domain(field);
    let mut selected = vec![false; max as usize + 1];

    let mut item_start = 0usize;
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(bytes, item_start, offset, field, &mut selected)?;
            item_start = offset + 1;
        }
    }
    parse_item(bytes, item_start, bytes.len(), field, &mut selected)?;

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());

    for i in 0..5 {
        selected[i] = parse_field(fields[i], kinds[i])?;
    }

    Ok(Cron {
        selected,
        day_stars: [fields[2].as_bytes() == b"*", fields[4].as_bytes() == b"*"],
    })
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax { field, offset: start });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn parse_field(input: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_domain(field);
    let width = maximum - minimum + 1;
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let has_step = !slash_positions.is_empty();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let base_end = slash_positions.first().copied().unwrap_or(item.len());
        let step_start = if has_step { base_end + 1 } else { item.len() };

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphen_positions: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if is_star && hyphen_positions.is_empty() {
            0 // star
        } else if hyphen_positions.is_empty() && !base.is_empty()
            && base.iter().all(u8::is_ascii_digit)
        {
            1 // number
        } else if hyphen_positions.len() == 1 {
            let hyphen = hyphen_positions[0];
            if hyphen > 0
                && hyphen + 1 < base.len()
                && base[..hyphen].iter().all(u8::is_ascii_digit)
                && base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                2 // range
            } else {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        if has_step && (step_start == item.len() || !item[step_start..].iter().all(u8::is_ascii_digit)) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (range_start, range_end, end_offset) = match base_kind {
            0 => (minimum, maximum, item_start),
            1 => {
                let value = parse_number(base, 0, base.len(), field)?;
                if value < minimum || value > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start,
                    });
                }
                if has_step {
                    (value, maximum, item_start)
                } else {
                    (value, value, item_start)
                }
            }
            _ => {
                let hyphen = hyphen_positions[0];
                let start = parse_number(base, 0, hyphen, field)?;
                let end = parse_number(base, hyphen + 1, base.len(), field)?;
                let start_offset = item_start;
                let end_number_offset = item_start + hyphen + 1;

                if start < minimum || start > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: start_offset,
                    });
                }
                if end < minimum || end > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: end_number_offset,
                    });
                }
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_number_offset,
                    });
                }
                (start, end, end_number_offset)
            }
        };

        let step = if has_step {
            let step_value = parse_number(item, step_start, item.len(), field)
                .map_err(|e| match e {
                    ParseError::LeadingZero { .. } => ParseError::LeadingZero {
                        field,
                        offset: item_start + step_start,
                    },
                    ParseError::NumberOverflow { .. } => ParseError::NumberOverflow {
                        field,
                        offset: item_start + step_start,
                    },
                    _ => e,
                })?;
            if step_value == 0 || step_value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + step_start,
                });
            }
            step_value
        } else {
            1
        };

        let mut value = range_start;
        while value <= range_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        let _ = end_offset;

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_wildcards = [false; 2];

    for (index, (&part, &field)) in parts.iter().zip(fields.iter()).enumerate() {
        selected[index] = parse_field(part, field)?;
        if field == Field::DayOfMonth || field == Field::DayOfWeek {
            day_wildcards[if field == Field::DayOfMonth { 0 } else { 1 }] = part == "*";
        }
    }

    Ok(Cron {
        selected,
        day_wildcards,
    })
}
