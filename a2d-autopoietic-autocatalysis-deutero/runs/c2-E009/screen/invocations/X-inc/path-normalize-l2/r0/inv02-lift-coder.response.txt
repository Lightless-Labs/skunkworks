#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|component| *component == "." || *component == "..")
}

fn normalized_absolute_components(path: &str) -> Result<Vec<String>, PathError> {
    let mut stack = Vec::new();

    for component in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape {
                        at: path.find("..").unwrap_or(0),
                    });
                }
            }
            ordinary => stack.push(ordinary.to_owned()),
        }
    }

    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let wants_directory = directory_intent(path);
    let mut stack: Vec<(String, usize)> = Vec::new();

    for (at, component) in path.split('/').scan(0usize, |offset, component| {
        let current = *offset;
        *offset += component.len() + 1;
        Some((current, component))
    }) {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if stack.last().is_some_and(|(name, _)| name != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push(("..".to_owned(), at));
            }
        } else {
            stack.push((component.to_owned(), at));
        }
    }

    let mut result = if absolute {
        format!(
            "/{}",
            stack
                .iter()
                .map(|(name, _)| name.as_str())
                .collect::<Vec<_>>()
                .join("/")
        )
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack
            .iter()
            .map(|(name, _)| name.as_str())
            .collect::<Vec<_>>()
            .join("/")
    };

    if wants_directory && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }

    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_owned).collect()),
        Err(_) => Err(error),
    }
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let expanded = if tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        Some(home_components.as_ref().ok_or(PathError::MissingHome)?)
    } else {
        None
    };

    let absolute = !tilde && bytes.first() == Some(&b'/');
    let mut stack = if absolute {
        Vec::new()
    } else if let Some(home) = expanded {
        home.clone()
    } else {
        cwd_components
    };

    let remainder = if tilde {
        if bytes.len() == 1 {
            ""
        } else {
            &path[2..]
        }
    } else {
        path
    };

    for (at, component) in remainder.split('/').scan(
        if tilde { 2usize } else { 0usize },
        |offset, component| {
            let current = *offset;
            *offset += component.len() + 1;
            Some((current, component))
        },
    ) {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            stack.push(component.to_owned());
        }
    }

    let mut result = format!("/{}", stack.join("/"));
    if directory_intent(path) && result != "/" {
        result.push('/');
    }
    Ok(result)
}
