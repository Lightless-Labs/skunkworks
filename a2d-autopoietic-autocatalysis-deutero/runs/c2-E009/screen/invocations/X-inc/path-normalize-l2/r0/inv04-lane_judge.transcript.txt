pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a>(
    components: impl IntoIterator<Item = (&'a str, usize)>,
    absolute: bool,
) -> Result<Vec<&'a str>, PathError> {
    let mut result = Vec::new();

    for (component, offset) in components {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if result.last().is_some_and(|last| *last != "..") {
                result.pop();
            } else if absolute {
                return Err(PathError::Escape { at: offset });
            } else {
                result.push("..");
            }
        } else {
            result.push(component);
        }
    }

    Ok(result)
}

fn assemble(components: &[&str], absolute: bool, dir_intent: bool) -> String {
    let mut output = String::new();

    if absolute {
        output.push('/');
    }
    output.push_str(&components.join("/"));

    if output.is_empty() {
        output.push('.');
    }

    if dir_intent && output != "/" && output != "." {
        output.push('/');
    }

    output
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let components = reduce_components(
        path.split('/')
            .scan(0usize, |offset, component| {
                let start = *offset;
                *offset += component.len() + 1;
                Some((component, start))
            }),
        absolute,
    )?;

    Ok(assemble(&components, absolute, intent))
}

}

        return Err(PathError::Empty);
    }

    Ok(())

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()

}

fn normalized_absolute_components(path: &str) -> Result<Vec<String>, PathError> {
    let mut stack = Vec::new();

    for component in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape {
                        at: path.find("..").unwrap_or(0),
                    });
                }
            }
            ordinary => stack.push(ordinary.to_owned()),
        }
    }

    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let wants_directory = directory_intent(path);
    let mut stack: Vec<(String, usize)> = Vec::new();

    for (at, component) in path.split('/').scan(0usize, |offset, component| {
        let current = *offset;
        *offset += component.len() + 1;
        Some((current, component))
    }) {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if stack.last().is_some_and(|(name, _)| name != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push(("..".to_owned(), at));
            }
        } else {
            stack.push((component.to_owned(), at));
        }
    }

    let mut result = if absolute {
        format!(
            "/{}",
            stack
                .iter()
                .map(|(name, _)| name.as_str())
                .collect::<Vec<_>>()
                .join("/")
        )
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack
            .iter()
            .map(|(name, _)| name.as_str())
            .collect::<Vec<_>>()
            .join("/")
    };

    if wants_directory && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }

    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_owned).collect()),
        Err(_) => Err(error),
    }
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let expanded = if tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        Some(home_components.as_ref().ok_or(PathError::MissingHome)?)
    } else {
        None
    };

    let absolute = !tilde && bytes.first() == Some(&b'/');
    let mut stack = if absolute {
        Vec::new()
    } else if let Some(home) = expanded {
        home.clone()
    } else {
        cwd_components
    };

    let remainder = if tilde {
        if bytes.len() == 1 {
            ""
        } else {
            &path[2..]
        }
    } else {
        path
    };

    for (at, component) in remainder.split('/').scan(
        if tilde { 2usize } else { 0usize },
        |offset, component| {
            let current = *offset;
            *offset += component.len() + 1;
            Some((current, component))
        },
    ) {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            stack.push(component.to_owned());
        }
    }

    let mut result = format!("/{}", stack.join("/"));
    if directory_intent(path) && result != "/" {
        result.push('/');
    }
    Ok(result)
}

pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.match_indices('/')
        .map(|(i, _)| i)
        .chain(std::iter::once(path.len()))
        .scan(0usize, |start, end| {
            let current = *start;
            *start = end + 1;
            Some((current, end))
        })
        .filter_map(|(start, end)| {
            (start < end).then_some((start, &path[start..end]))
        })
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, component)| *component == "." || *component == "..")
}

fn reduce(
    stack: &mut Vec<String>,
    path: &str,
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, component) in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|part| part != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset + at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            name => stack.push(name.to_owned()),
        }
    }
    Ok(())
}

    let mut result = if absolute {
        if stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", stack.join("/"))

    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if want_directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

    } else {
        (cwd_components, path, 0)
    };

    reduce(&mut stack, remainder, offset, true)?;
    Ok(assemble(&stack, true, want_directory))
}

    let cwd_stack = validated_absolute_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(home) => Some(
            validated_absolute_context(home).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let (absolute, mut stack, remainder, offset_base) = if path.starts_with('/') {
        (true, Vec::new(), path, 0)
    } else if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let stack = home_stack.ok_or(PathError::MissingHome)?;
        let remainder = if bytes.len() > 1 { &path[2..] } else { "" };
        (true, stack, remainder, if bytes.len() > 1 { 2 } else { 1 })
    } else {
        (true, cwd_stack, path, 0)
    };

    reduce(&mut stack, remainder, absolute, offset_base)?;

    Ok(render(&stack, true, directory_intent(path)))
}
