#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

fn render_hunk(script: &[Edit], start: usize, end: usize, output: &mut String) {
    let mut old_before = 0usize;
    let mut new_before = 0usize;

    for edit in &script[..start] {
        match edit {
            Edit::Keep { .. } => {
                old_before += 1;
                new_before += 1;
            }
            Edit::Delete { .. } => old_before += 1,
            Edit::Insert { .. } => new_before += 1,
        }
    }

    let mut old_count = 0usize;
    let mut new_count = 0usize;
    for edit in &script[start..end] {
        match edit {
            Edit::Keep { .. } => {
                old_count += 1;
                new_count += 1;
            }
            Edit::Delete { .. } => old_count += 1,
            Edit::Insert { .. } => new_count += 1,
        }
    }

    let old_start = if old_count > 0 {
        old_before + 1
    } else {
        old_before
    };
    let new_start = if new_count > 0 {
        new_before + 1
    } else {
        new_before
    };

    output.push_str(&format!(
        "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
    ));

    for edit in &script[start..end] {
        match edit {
            Edit::Keep { line, .. } => {
                output.push(' ');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Delete { line, .. } => {
                output.push('-');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Insert { line, .. } => {
                output.push('+');
                output.push_str(line);
                output.push('\n');
            }
        }
    }
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }

    if let Some(byte) = old.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::Old,
            byte,
        });
    }
    if let Some(byte) = new.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::New,
            byte,
        });
    }

    if !old.is_empty() && !old.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);
    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let m = old_lines.len();
    let n = new_lines.len();
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];
    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old_lines[i] == new_lines[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut script = Vec::new();
    let (mut i, mut j) = (0usize, 0usize);
    while i < m && j < n {
        if old_lines[i] == new_lines[j] {
            script.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old_lines[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else if lcs[i + 1][j] > lcs[i][j + 1] {
            script.push(Edit::Delete {
                old_index: i,
                line: old_lines[i].to_owned(),
            });
            i += 1;
        } else if lcs[i + 1][j] < lcs[i][j + 1] {
            script.push(Edit::Insert {
                new_index: j,
                line: new_lines[j].to_owned(),
            });
            j += 1;
        } else if (i + j) % 2 == 0 {
            script.push(Edit::Delete {
                old_index: i,
                line: old_lines[i].to_owned(),
            });
            i += 1;
        } else {
            script.push(Edit::Insert {
                new_index: j,
                line: new_lines[j].to_owned(),
            });
            j += 1;
        }
    }

    while i < m {
        script.push(Edit::Delete {
            old_index: i,
            line: old_lines[i].to_owned(),
        });
        i += 1;
    }
    while j < n {
        script.push(Edit::Insert {
            new_index: j,
            line: new_lines[j].to_owned(),
        });
        j += 1;
    }

    let mut groups = Vec::<(usize, usize)>::new();
    let mut k = 0usize;
    while k < script.len() {
        if matches!(script[k], Edit::Keep { .. }) {
            k += 1;
            continue;
        }
        let start = k;
        while k < script.len() && !matches!(script[k], Edit::Keep { .. }) {
            k += 1;
        }
        groups.push((start, k));
    }

    if groups.is_empty() {
        return Ok(String::new());
    }

    let merge_distance = 2 * context;
    let mut merged = Vec::<(usize, usize)>::new();
    for (start, end) in groups {
        if let Some(last) = merged.last_mut() {
            let keeps_between = script[last.1..start]
                .iter()
                .filter(|edit| matches!(edit, Edit::Keep { .. }))
                .count();
            if keeps_between <= merge_distance {
                last.1 = end;
                continue;
            }
        }
        merged.push((start, end));
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (start, end) in merged {
        let mut hunk_start = start;
        let mut preceding = context;
        while hunk_start > 0 && preceding > 0 {
            hunk_start -= 1;
            if matches!(script[hunk_start], Edit::Keep { .. }) {
                preceding -= 1;
            }
        }

        let mut hunk_end = end;
        let mut following = context;
        while hunk_end < script.len() && following > 0 {
            if matches!(script[hunk_end], Edit::Keep { .. }) {
                following -= 1;
            }
            hunk_end += 1;
        }

        render_hunk(&script, hunk_start, hunk_end, &mut output);
    }

    Ok(output)
}
