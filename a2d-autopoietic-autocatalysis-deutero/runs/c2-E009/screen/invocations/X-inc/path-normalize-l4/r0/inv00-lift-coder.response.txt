#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn raw_components(path: &str, base_offset: usize) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_string(), base_offset + start));
    }
    result
}

fn append_directory_slash(mut result: String, intent: bool) -> String {
    if intent && result != "/" && result != "." && !result.is_empty() {
        if !result.ends_with('/') {
            result.push('/');
        }
    }
    result
}

fn normalize_components(
    stack: &mut Vec<String>,
    components: Vec<(String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (component, at) in components {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(component);
                }
            }
            _ => stack.push(component),
        }
    }
    Ok(())
}

fn compose(stack: &[String], absolute: bool) -> String {
    if absolute {
        if stack.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    normalize_components(&mut stack, raw_components(path, 0), absolute)?;
    Ok(append_directory_slash(compose(&stack, absolute), intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(raw_components(&normalized, 0)
        .into_iter()
        .map(|(component, _)| component)
        .collect())
}

fn validate_home(home: &str) -> Result<Vec<String>, PathError> {
    if !home.starts_with('/') {
        return Err(PathError::InvalidHome);
    }
    let normalized = normalize(home).map_err(|_| PathError::InvalidHome)?;
    Ok(raw_components(&normalized, 0)
        .into_iter()
        .map(|(component, _)| component)
        .collect())
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, Vec<(String, usize)>, bool, Option<Vec<String>>), PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let home_components = match home {
                Some(home) => Some(validate_home(home)?),
                None => return Err(PathError::MissingHome),
            };
            let rest_start = if path == "~" { path.len() } else { 2 };
            let components = if rest_start < path.len() {
                raw_components(&path[rest_start..], rest_start)
            } else {
                Vec::new()
            };
            Ok((true, components, intent, home_components))
        } else {
            Err(PathError::UnsupportedTilde { at: 1 })
        }
    } else {
        Ok((
            path.starts_with('/'),
            raw_components(path, 0),
            intent,
            None,
        ))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_home(home)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);

    let (absolute, components, start) = if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let start = home_components.ok_or(PathError::MissingHome)?;
            let offset = if path == "~" { path.len() } else { 2 };
            let components = if offset < path.len() {
                raw_components(&path[offset..], offset)
            } else {
                Vec::new()
            };
            (false, components, start)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (true, raw_components(path, 0), Vec::new())
    } else {
        (false, raw_components(path, 0), cwd_components)
    };

    let mut stack = start;
    normalize_components(&mut stack, components, true)?;
    Ok(append_directory_slash(compose(&stack, true), intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<&str> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(first) = seen.iter().position(|path| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        seen.push(&link.path);
    }
    Ok(())
}

fn stack_matches_link(stack: &[String], links: &[Link]) -> Option<usize> {
    let current = compose(stack, true);
    links.iter().position(|link| link.path == current)
}

#[derive(Clone)]
struct WorkItem {
    component: String,
    at: usize,
    source: String,
}

fn link_work(target: &str) -> Vec<WorkItem> {
    raw_components(target, 0)
        .into_iter()
        .map(|(component, at)| WorkItem {
            component,
            at,
            source: target.to_string(),
        })
        .collect()
}

fn request_work(path: &str, components: Vec<(String, usize)>) -> Vec<WorkItem> {
    components
        .into_iter()
        .map(|(component, at)| WorkItem {
            component,
            at,
            source: path.to_string(),
        })
        .collect()
}

fn expand_links(
    mut stack: Vec<String>,
    mut work: Vec<WorkItem>,
    links: &[Link],
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while !work.is_empty() {
        let item = work.remove(0);
        match item.component.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: item.at });
                }
                continue;
            }
            _ => stack.push(item.component),
        }

        if let Some(index) = stack_matches_link(&stack, links) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;

            let link = &links[index];
            stack.pop();
            let mut target_work = link_work(&link.target);
            if link.target.starts_with('/') {
                stack.clear();
            }
            active.push(index);

            // Process the target before the pending work; a marker ends this link's
            // active lifetime once all recursively inserted target work is complete.
            target_work.push(WorkItem {
                component: String::new(),
                at: usize::MAX,
                source: String::new(),
            });
            target_work.extend(work);
            work = target_work;

            // The marker is handled by the loop below through a sentinel encoding.
            // Its link index is carried separately by the source field.
            if let Some(marker) = work.iter_mut().find(|entry| entry.at == usize::MAX) {
                marker.source = index.to_string();
            }
        }

        if work.first().is_some_and(|entry| entry.at == usize::MAX) {
            // Marker handling is performed at the top of the loop.
        }
    }

    Ok(stack)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_home(home)?),
        None => None,
    };
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (absolute, components, tilde_start) = if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let start = home_components.ok_or(PathError::MissingHome)?;
            let offset = if path == "~" { path.len() } else { 2 };
            (
                false,
                if offset < path.len() {
                    raw_components(&path[offset..], offset)
                } else {
                    Vec::new()
                },
                Some(start),
            )
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (true, raw_components(path, 0), None)
    } else {
        (false, raw_components(path, 0), None)
    };

    let mut stack = if absolute {
        Vec::new()
    } else {
        tilde_start.unwrap_or(cwd_components)
    };

    // Context components are already canonical and are traversed before the request.
    let mut context_work = Vec::new();
    if !absolute && tilde_start.is_none() {
        context_work = stack
            .drain(..)
            .map(|component| WorkItem {
                component,
                at: 0,
                source: String::new(),
            })
            .collect();
    } else if tilde_start.is_some() {
        context_work = stack
            .drain(..)
            .map(|component| WorkItem {
                component,
                at: 0,
                source: String::new(),
            })
            .collect();
    }

    let mut work = context_work;
    work.extend(request_work(path, components));
    let stack = expand_links(stack, work, links, max_expansions)?;
    Ok(append_directory_slash(compose(&stack, true), intent))
}

fn components_of_absolute(path: &str) -> Vec<String> {
    raw_components(path, 0)
        .into_iter()
        .map(|(component, _)| component)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(left, right)| left == right)
        .count();
    let mut components = vec!["..".to_string(); base.len() - common];
    components.extend(target[common..].iter().cloned());
    if components.is_empty() {
        ".".to_string()
    } else {
        components.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = components_of_absolute(&resolved_base);
    let target_components = components_of_absolute(&resolved_target);

    let mut candidates = vec![route(&base_components, &target_components)];
    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let probe_components = components_of_absolute(&probe);
        if probe_components.len() > target_components.len()
            || probe_components != target_components[..probe_components.len()]
        {
            continue;
        }

        let mut candidate = route(&base_components, &components_of_absolute(&link.path));
        let suffix = &target_components[probe_components.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let target_directory = resolved_target.ends_with('/');
    let mut valid = Vec::<String>::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) else {
            continue;
        };
        if components_of_absolute(&resolved) == target_components {
            valid.push(candidate);
        }
    }

    valid.sort_by(|left, right| {
        left.len()
            .cmp(&right.len())
            .then_with(|| left.as_bytes().cmp(right.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
