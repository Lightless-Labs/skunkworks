use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if absolute && parts.is_empty() {
        return "/".to_owned();
    }
    if !absolute && parts.is_empty() {
        return ".".to_owned();
    }
    if intent {
        result.push('/');
    }
    result
}

fn reduce_plain(
    initial: &mut Vec<String>,
    raw: &str,
    offset_base: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let mut cursor = 0usize;
    for component in raw.split('/') {
        let start = cursor;
        cursor += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if initial.last().is_some_and(|part| part != "..") {
                initial.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: offset_base + start,
                });
            } else {
                initial.push("..".to_owned());
            }
        } else {
            initial.push(component.to_owned());
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut parts = Vec::new();
    reduce_plain(&mut parts, path, 0, absolute)?;
    Ok(render(&parts, absolute, intent))
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized))
}

fn validated_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = normalized_context(cwd)?;
    let home_parts = match home {
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let normalized = normalize(path).map_err(|_| PathError::InvalidHome)?;
            Some(components(&normalized))
        }
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

fn tilde_request<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(Option<&'a [String]>, &'a str, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((None, path, 0));
    }
    let bytes = path.as_bytes();
    if bytes.len() == 1 || bytes[1] == b'/' {
        let home = home.ok_or(PathError::MissingHome)?;
        let skip = if bytes.len() > 1 { 2 } else { 1 };
        return Ok((Some(home), &path[skip..], skip));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

fn resolve_core(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
) -> Result<String, PathError> {
    validate_raw(path)?;

    let (tilde_home, remainder, offset) = tilde_request(path, home_parts)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);

    let mut parts = if let Some(home) = tilde_home {
        home.to_vec()
    } else if absolute {
        Vec::new()
    } else {
        cwd_parts.to_vec()
    };

    reduce_plain(&mut parts, remainder, offset, true)?;
    Ok(render(&parts, true, intent))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validated_contexts(cwd, home)?;
    resolve_core(path, &cwd_parts, home_parts.as_deref())
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut table = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        table.insert(link.path.clone(), index);
    }
    Ok(table)
}

#[derive(Clone)]
struct WorkComponent {
    text: String,
    at: usize,
}

#[derive(Clone)]
struct Frame {
    components: Vec<WorkComponent>,
    next: usize,
    origin: Option<usize>,
}

fn work_components(raw: &str, offset: usize) -> Vec<WorkComponent> {
    let mut result = Vec::new();
    let mut cursor = 0usize;
    for component in raw.split('/') {
        result.push(WorkComponent {
            text: component.to_owned(),
            at: offset + cursor,
        });
        cursor += component.len() + 1;
    }
    result
}

fn absolute_key(parts: &[String]) -> String {
    format!("/{}", parts.join("/"))
}

fn resolve_links_core(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
    links: &[Link],
    table: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let absolute = path.starts_with('/');
    let (tilde_home, remainder, request_offset) = tilde_request(path, home_parts)?;

    let mut stack = if absolute {
        Vec::new()
    } else if let Some(home) = tilde_home {
        home.to_vec()
    } else {
        cwd_parts.to_vec()
    };

    let mut frames = Vec::new();
    if !absolute {
        let context = if let Some(home) = tilde_home {
            home
        } else {
            cwd_parts
        };
        for component in context {
            frames.push(Frame {
                components: vec![WorkComponent {
                    text: component.clone(),
                    at: 0,
                }],
                next: 0,
                origin: None,
            });
        }
    }
    frames.push(Frame {
        components: work_components(remainder, request_offset),
        next: 0,
        origin: None,
    });

    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(frame) = frames.last_mut() {
        if frame.next >= frame.components.len() {
            let finished = frames.pop().unwrap();
            if let Some(index) = finished.origin {
                if active.last() == Some(&index) {
                    active.pop();
                } else if let Some(position) = active.iter().rposition(|&entry| entry == index) {
                    active.remove(position);
                }
            }
            continue;
        }

        let component = frame.components[frame.next].clone();
        frame.next += 1;

        if component.text.is_empty() || component.text == "." {
            continue;
        }
        if component.text == ".." {
            if stack.last().is_some() {
                stack.pop();
            } else {
                return Err(PathError::Escape { at: component.at });
            }
            continue;
        }

        stack.push(component.text);
        let key = absolute_key(&stack);
        if let Some(&index) = table.get(&key) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);

            stack.pop();
            let target = &links[index].target;
            let target_absolute = target.starts_with('/');
            if target_absolute {
                stack.clear();
            }
            frames.push(Frame {
                components: work_components(target, 0),
                next: 0,
                origin: Some(index),
            });
        }
    }

    Ok(render(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validated_contexts(cwd, home)?;
    let table = validate_links(links)?;
    resolve_links_core(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        &table,
        max_expansions,
    )
}

fn path_components(path: &str) -> Vec<String> {
    components(path)
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();

    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_owned()));
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = path_components(&resolved_base);
    let target_parts = path_components(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = resolve_links(
            &link.path,
            "/",
            None,
            links,
            max_expansions,
        );
        let Ok(resolved_key) = probe else {
            continue;
        };

        let alias_parts = path_components(&resolved_key);
        if alias_parts.len() > target_parts.len()
            || !alias_parts
                .iter()
                .zip(target_parts.iter())
                .all(|(a, b)| a == b)
        {
            continue;
        }

        let suffix = &target_parts[alias_parts.len()..];
        let mut candidate = route(&base_parts, &path_components(&link.path));
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }

        if target_directory && candidate != "." {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if path_components(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    valid
        .into_iter()
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())))
        .ok_or(PathError::Empty)
}
