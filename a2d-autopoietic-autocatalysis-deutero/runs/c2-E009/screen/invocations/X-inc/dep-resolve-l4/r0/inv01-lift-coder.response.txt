fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(v: &[Constraint]) -> Vec<Constraint> {
    let mut out = v.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(v: &[Evidence]) -> Vec<Evidence> {
    let mut out = v.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn evidence_suffix(v: &[Evidence]) -> String {
    let mut out = String::new();
    for e in evidence_sorted(v) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_suffix(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(","),
            evidence_suffix(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(","),
            evidence_suffix(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    constraints: BTreeMap<String, Vec<Constraint>>,
    features: BTreeMap<String, BTreeSet<String>>,
    origins: BTreeMap<String, Vec<Origin>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn feature_token(fs: &[String]) -> String {
    let mut f = fs.to_vec();
    f.sort();
    f.dedup();
    f.join(",")
}

fn root_token(r: &Requirement) -> String {
    format!(
        "root:{}@{}+{{{}}}",
        r.package,
        constraint_text(r.constraint),
        feature_token(&r.features)
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    d: &Dependency,
) -> String {
    let gate = d.gate.as_deref().unwrap_or("*");
    let platform = match d.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        d.package,
        constraint_text(d.constraint),
        feature_token(&d.features),
        gate,
        platform
    )
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut a = Active::default();
    let mut queue: Vec<(String, String, Constraint, Vec<String>, Vec<String>)> = Vec::new();

    for r in roots {
        let token = root_token(r);
        queue.push((
            r.package.clone(),
            format!("root:{}", token),
            r.constraint,
            vec![token],
            r.features.clone(),
        ));
    }

    let mut processed_origins = BTreeSet::new();
    let mut processed_features = BTreeSet::<(String, String)>::new();

    loop {
        while let Some((pkg, origin_id, constraint, path, requested)) = queue.pop() {
            if processed_origins.insert(origin_id.clone()) {
                a.constraints
                    .entry(pkg.clone())
                    .or_default()
                    .push(constraint);
                a.origins.entry(pkg.clone()).or_default().push(Origin {
                    constraint,
                    path: path.clone(),
                });
                let replace = match a.paths.get(&pkg) {
                    None => true,
                    Some(old) => path.len() < old.len() || (path.len() == old.len() && path < *old),
                };
                if replace {
                    a.paths.insert(pkg.clone(), path);
                }
            }
            let set = a.features.entry(pkg.clone()).or_default();
            for f in requested {
                set.insert(f);
            }
        }

        let mut changed = false;
        let names: Vec<String> = a.features.keys().cloned().collect();
        for pkg in names {
            let Some(&version) = selected.get(&pkg) else {
                continue;
            };
            let Some(package) = catalog.get(&pkg) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let activated: Vec<String> = a.features[&pkg].iter().cloned().collect();
            for feature_name in activated {
                if !processed_features.insert((pkg.clone(), feature_name.clone())) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        let inserted = a
                            .features
                            .entry(pkg.clone())
                            .or_default()
                            .insert(enabled.clone());
                        changed |= inserted;
                    }
                }
            }

            let Some(source_path) = a.paths.get(&pkg).cloned() else {
                continue;
            };
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !a.features.get(&pkg).is_some_and(|s| s.contains(g)))
                {
                    continue;
                }
                let token = dependency_token(&pkg, version, j, dep);
                let mut path = source_path.clone();
                path.push(token.clone());
                let origin_id = format!("dep:{}:{}:{}:{}", pkg, version_text(version), j, dep.package);
                if processed_origins.contains(&origin_id) {
                    continue;
                }
                queue.push((
                    dep.package.clone(),
                    origin_id,
                    dep.constraint,
                    path,
                    dep.features.clone(),
                ));
                changed = true;
            }
        }

        if queue.is_empty() && !changed {
            break;
        }
        if queue.is_empty() {
            // Newly activated local features may need another pass.
            let pending = a.features.iter().any(|(pkg, fs)| {
                let Some(version) = selected.get(pkg) else {
                    return false;
                };
                let Some(package) = catalog.get(pkg) else {
                    return false;
                };
                let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
                    return false;
                };
                fs.iter()
                    .any(|f| !processed_features.contains(&(pkg.clone(), f.clone())) && release.features.iter().any(|d| d.name == *f))
            });
            if !pending {
                break;
            }
        }
    }

    for constraints in a.constraints.values_mut() {
        *constraints = canonical_constraints(constraints);
    }
    for origins in a.origins.values_mut() {
        origins.sort_by(|x, y| {
            constraint_key(x.constraint)
                .cmp(&constraint_key(y.constraint))
                .then_with(|| x.path.cmp(&y.path))
        });
    }
    a
}

fn evidence_for(a: &Active, package: &str) -> Vec<Evidence> {
    a.origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect()
}

fn requirements_for(a: &Active, package: &str) -> Vec<Constraint> {
    a.constraints.get(package).cloned().unwrap_or_default()
}

fn search_node(
    catalog: &BTreeMap<String, &Package>,
    all_names: &[String],
    roots: &[Requirement],
    locks: &BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: &mut usize,
    selected: &mut BTreeMap<String, Version>,
    best: &mut Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>)>,
    first_error: &mut Option<ResolveError>,
) -> Result<(), ResolveError> {
    let active = compute_active(catalog, roots, selected, target);
    let active_names: BTreeSet<String> = active.constraints.keys().cloned().collect();

    for package in active_names.iter() {
        if !catalog.contains_key(package) {
            let err = ResolveError::Missing {
                package: package.clone(),
                evidence: evidence_for(&active, package),
            };
            if first_error.is_none() {
                *first_error = Some(err);
            }
            return Ok(());
        }
    }

    for package in selected.keys() {
        if let Some(reqs) = active.constraints.get(package) {
            let version = selected[package];
            if !reqs.iter().all(|c| matches(*c, version)) {
                let err = ResolveError::Conflict {
                    package: package.clone(),
                    selected: version,
                    requirements: reqs.clone(),
                    evidence: evidence_for(&active, package),
                };
                if first_error.is_none() {
                    *first_error = Some(err);
                }
                return Ok(());
            }
        }
    }

    for package in active_names.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let reqs = requirements_for(&active, package);
        let has_match = catalog[package]
            .releases
            .iter()
            .any(|r| reqs.iter().all(|c| matches(*c, r.version)));
        if !has_match {
            let err = ResolveError::NoMatch {
                package: package.clone(),
                requirements: reqs,
                evidence: evidence_for(&active, package),
            };
            if first_error.is_none() {
                *first_error = Some(err);
            }
            return Ok(());
        }
    }

    for package in selected.keys() {
        let version = selected[package];
        let Some(pkg) = catalog.get(package) else {
            continue;
        };
        let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        if let Some(features) = active.features.get(package) {
            if let Some(feature) = features
                .iter()
                .find(|f| !release.features.iter().any(|decl| decl.name == **f))
            {
                let err = ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: version,
                    feature: feature.clone(),
                    path: active.paths.get(package).cloned().unwrap_or_default(),
                };
                if first_error.is_none() {
                    *first_error = Some(err);
                }
                return Ok(());
            }
        }
    }

    if let Some(package) = active_names.iter().find(|p| !selected.contains_key(*p)) {
        let reqs = requirements_for(&active, package);
        let mut candidates: Vec<Version> = catalog[*package]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| reqs.iter().all(|c| matches(*c, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock_version) = locks.get(package) {
            if let Some(pos) = candidates.iter().position(|v| v == lock_version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if *attempts == max_attempts {
                return Err(ResolveError::Limit {
                    attempts: max_attempts,
                });
            }
            *attempts += 1;
            selected.insert(package.clone(), candidate);
            search_node(
                catalog,
                all_names,
                roots,
                locks,
                target,
                max_attempts,
                attempts,
                selected,
                best,
                first_error,
            )?;
            selected.remove(package);
        }
        return Ok(());
    }

    let cost = locks
        .iter()
        .filter(|(p, v)| selected.get(*p) != Some(*v))
        .count();
    let vector: Vec<Option<Version>> = all_names.iter().map(|p| selected.get(p).copied()).collect();
    let improve = match best {
        None => true,
        Some((old_cost, old_vector, _)) => {
            cost < *old_cost || (cost == *old_cost && vector > *old_vector)
        }
    };
    if improve {
        *best = Some((cost, vector, selected.clone()));
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let mut lock_map = BTreeMap::new();
    for lock in locks {
        lock_map.insert(lock.package.clone(), lock.version);
    }
    let mut names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    names.sort();

    let mut attempts = 0;
    let mut selected = BTreeMap::new();
    let mut best = None;
    let mut first_error = None;
    search_node(
        &catalog_map,
        &names,
        roots,
        &lock_map,
        target,
        max_attempts,
        &mut attempts,
        &mut selected,
        &mut best,
        &mut first_error,
    )?;

    let Some((_, _, solution)) = best else {
        return Err(first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_active = compute_active(&catalog_map, roots, &solution, target);
    let selected_out: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_change_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_change_names.extend(solution.keys().cloned());
    let mut changes = Vec::new();
    for package in all_change_names {
        let before = lock_map.get(&package).copied();
        let after = solution.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                if !catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old))
                {
                    ChangeReason::Unavailable
                } else if !final_active
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| cs.iter().all(|c| matches(*c, old)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
