fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || *b.last().unwrap() == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then(a.path.cmp(&b.path)));
    out
}

fn evidence_suffix(evidence: &[Evidence]) -> String {
    let mut out = String::new();
    for e in sorted_evidence(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_suffix(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "no-match|package={}|constraints={}{}",
                encode(package),
                cs,
                evidence_suffix(evidence)
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={}{}",
                encode(package),
                version_text(*selected),
                cs,
                evidence_suffix(evidence)
            )
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}{}",
            encode(package),
            version_text(*selected),
            encode(feature),
            format!("\n  F:{} via {}", encode(feature), path.join(" -> "))
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.enable[{}]", base, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut names = BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.as_str()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Origin {

    path: Vec<String>,

#[derive(Clone, Default)]
struct State {
    selected: BTreeMap<String, Version>,

#[derive(Clone)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    active_deps: BTreeMap<String, BTreeSet<usize>>,
}

fn platform_name(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    match p {
        Platform::Any => true,
        Platform::Linux => t == Target::Linux,
        Platform::Mac => t == Target::Mac,
        Platform::Windows => t == Target::Windows,
    }
}

fn root_token(i: usize, root: &Requirement) -> String {
    let mut fs = root.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        root.package,
        constraint_text(root.constraint),
        fs.join(",")
    )
}

fn dep_token(source: &str, version: Version, i: usize, dep: &Dependency) -> String {
    let mut fs = dep.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        i,
        dep.package,
        constraint_text(dep.constraint),
        fs.join(","),
        dep.gate.as_deref().unwrap_or("*"),
        platform_name(dep.platform)
    )
}

fn canonical_path(paths: &[Vec<String>]) -> Option<Vec<String>> {
    paths.iter().min_by(|a, b| a.len().cmp(&b.len()).then(a.cmp(b))).cloned()
}

fn compute_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    state: &State,
    target: Target,
) -> Closure {
    let mut closure = Closure {
        constraints: BTreeMap::new(),
        origins: BTreeMap::new(),
        requests: BTreeMap::new(),
        active_features: BTreeMap::new(),
        paths: BTreeMap::new(),
        active_deps: BTreeMap::new(),
    };

    for (i, root) in roots.iter().enumerate() {
        closure.constraints.entry(root.package.clone()).or_default().push(root.constraint);
        closure.origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: vec![root_token(i, root)],
        });
        closure.paths.entry(root.package.clone()).or_default().push(vec![root_token(i, root)]);
        closure.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
    }

    let mut changed = true;
    while changed {
        changed = false;

        let packages: Vec<String> = state.selected.keys().cloned().collect();
        for package_name in packages {
            let Some(version) = state.selected.get(&package_name).copied() else { continue };
            let Some(package) = catalog.get(&package_name) else { continue };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else { continue };

            let requested = closure.requests.get(&package_name).cloned().unwrap_or_default();
            let active = closure.active_features.entry(package_name.clone()).or_default();
            let old_len = active.len();
            active.extend(requested);
            if active.len() != old_len {
                changed = true;
            }

            let current_features = closure.active_features.get(&package_name).cloned().unwrap_or_default();
            for feature_name in current_features {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    let reqs = closure.requests.entry(package_name.clone()).or_default();
                    for enabled in &feature.enables {
                        if reqs.insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            for (i, dep) in release.deps.iter().enumerate() {
                let enabled = platform_matches(dep.platform, target)
                    && dep.gate.as_ref().map_or(true, |g| {
                        closure.active_features.get(&package_name).is_some_and(|fs| fs.contains(g))
                    });
                if !enabled {
                    continue;
                }

                if closure.active_deps.entry(package_name.clone()).or_default().insert(i) {
                    changed = true;
                }

                let token = dep_token(&package_name, version, i, dep);
                let source_path = closure.paths.get(&package_name)
                    .and_then(|ps| canonical_path(ps))
                    .unwrap_or_default();
                let mut incoming_path = source_path;
                incoming_path.push(token);

                let origins = closure.origins.entry(dep.package.clone()).or_default();
                let origin = Origin { constraint: dep.constraint, path: incoming_path.clone() };
                if !origins.iter().any(|o| o.constraint == origin.constraint && o.path == origin.path) {
                    origins.push(origin);
                    changed = true;
                }
                let constraints = closure.constraints.entry(dep.package.clone()).or_default();
                if !constraints.contains(&dep.constraint) {
                    constraints.push(dep.constraint);
                    changed = true;
                }

                let paths = closure.paths.entry(dep.package.clone()).or_default();
                if !paths.contains(&incoming_path) {
                    paths.push(incoming_path);
                    changed = true;
                }

                let requests = closure.requests.entry(dep.package.clone()).or_default();
                for feature in &dep.features {
                    if requests.insert(feature.clone()) {
                        changed = true;
                    }
                }
            }
        }
    }

    for cs in closure.constraints.values_mut() {
        *cs = canonical_constraints(cs);
    }
    closure
}

fn evidence_for(c: &Closure, package: &str) -> Vec<Evidence> {
    let mut out = c.origins.get(package).cloned().unwrap_or_default()
        .into_iter()
        .map(|o| Evidence { constraint: o.constraint, path: o.path })
        .collect::<Vec<_>>();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then(a.path.cmp(&b.path)));
    out
}

enum NodeResult {
    Complete,
    Error(ResolveError),
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    target: Target,
    locks: &'a BTreeMap<String, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, state: &mut State) -> NodeResult {
        let closure = compute_closure(self.catalog, self.roots, state, self.target);

        for package in closure.constraints.keys() {
            if !self.catalog.contains_key(package) {
                return NodeResult::Error(ResolveError::Missing {
                    package: package.clone(),
                    evidence: evidence_for(&closure, package),
                });
            }
        }

        for (package, selected) in &state.selected {
            let cs = closure.constraints.get(package).cloned().unwrap_or_default();
            if cs.iter().any(|c| !matches(*c, *selected)) {
                return NodeResult::Error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *selected,
                    requirements: cs,
                    evidence: evidence_for(&closure, package),
                });
            }
        }

        for (package, version) in &state.selected {
            if let Some(pkg) = self.catalog.get(package) {
                if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                    let active = closure.active_features.get(package).cloned().unwrap_or_default();
                    for feature in active {
                        if !release.features.iter().any(|f| f.name == feature) {
                            let path = closure.paths.get(package)
                                .and_then(|ps| canonical_path(ps))
                                .unwrap_or_default();
                            return NodeResult::Error(ResolveError::MissingFeature {
                                package: package.clone(),
                                selected: *version,
                                feature,
                                path,
                            });
                        }
                    }
                }
            }
        }

        for package in closure.constraints.keys() {
            if state.selected.contains_key(package) {
                continue;
            }
            let cs = closure.constraints.get(package).cloned().unwrap_or_default();
            let has_match = self.catalog.get(package).is_some_and(|pkg| {
                pkg.releases.iter().any(|r| cs.iter().all(|c| matches(*c, r.version)))
            });
            if !has_match {
                return NodeResult::Error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: cs,
                    evidence: evidence_for(&closure, package),
                });
            }
        }

        let next = closure.constraints.keys().find(|p| !state.selected.contains_key(*p)).cloned();
        let Some(package_name) = next else {
            let lock_cost = self.locks.iter().filter(|(p, v)| state.selected.get(*p) != Some(*v)).count();
            let names: Vec<String> = self.catalog.keys().cloned().collect();
            let vector = names.iter().map(|p| state.selected.get(p).copied()).collect::<Vec<_>>();
            let should_replace = match &self.best {
                None => true,
                Some((cost, best_vector, _)) => {
                    lock_cost < *cost || (lock_cost == *cost && vector > *best_vector)
                }
            };
            if should_replace {
                self.best = Some((lock_cost, vector, state.selected.clone()));
            }
            return NodeResult::Complete;
        };

        let cs = closure.constraints.get(&package_name).cloned().unwrap_or_default();
        let pkg = self.catalog[&package_name];
        let mut candidates = pkg.releases.iter()
            .filter(|r| cs.iter().all(|c| matches(*c, r.version)))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = self.locks.get(&package_name) {
            if let Some(i) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        let mut first = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return NodeResult::Error(ResolveError::Limit { attempts: self.max_attempts });
            }
            self.attempts += 1;
            state.selected.insert(package_name.clone(), version);
            match self.inspect(state) {
                NodeResult::Complete => {}
                NodeResult::Error(ResolveError::Limit { attempts }) => {
                    state.selected.remove(&package_name);
                    return NodeResult::Error(ResolveError::Limit { attempts });
                }
                NodeResult::Error(error) => {
                    if first.is_none() {
                        first = Some(error);
                    }
                }
            }
            state.selected.remove(&package_name);
        }

        if let Some(error) = first {
            NodeResult::Error(error)
        } else {
            NodeResult::Error(ResolveError::NoMatch {
                package: package_name.clone(),
                requirements: cs,
                evidence: evidence_for(&closure, &package_name),
            })
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> =
        locks.iter().map(|l| (l.package.clone(), l.version)).collect();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        target,
        locks: &lock_map,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut state = State::default();
    match search.inspect(&mut state) {
        NodeResult::Error(ResolveError::Limit { attempts }) => {
            return Err(ResolveError::Limit { attempts });
        }
        NodeResult::Error(error) => {
            if search.best.is_none() {
                return Err(search.first_error.unwrap_or(error));
            }
        }
        NodeResult::Complete => {}
    }

    let Some((_, _, selected_map)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: vec![],
            evidence: vec![],
        }));
    };

    let selected = selected_map.iter().map(|(package, version)| Selection {
        package: package.clone(),
        version: *version,
    }).collect::<Vec<_>>();

    let final_state = State { selected: selected_map.clone() };
    let final_closure = compute_closure(&catalog_map, roots, &final_state, target);

    let mut changes = Vec::new();
    let mut all_names = BTreeSet::new();
    all_names.extend(lock_map.keys().cloned());
    all_names.extend(selected_map.keys().cloned());
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map.get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let cs = final_closure.constraints.get(&package).cloned().unwrap_or_default();
                    if cs.iter().any(|c| !matches(*c, old)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change { package, before, after, reason });
    }

    Ok(Resolution { selected, changes })
}
