    pub route: usize,

}

    Static(String),

    Wildcard(String),

    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {

        let mut rank = Vec::new();
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head, false)?;

        let mut best: Option<(usize, Match)> = None;
        for (index, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(&route.segments, &request_segments) {
                let replace = match &best {
                    None => true,
                    Some((best_index, _)) => {
                        route.rank > self.routes[*best_index].rank
                    }
                };
                if replace {
                    best = Some((index, Match { route: route.id, params }));
                }
            }
        }

        best.map(|(_, matched)| matched).ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    let raw_segments = parse_raw_segments(pattern, true)?;
    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();

    for (raw, start) in raw_segments {
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if segments.iter().any(|segment| matches!(segment, Segment::Wildcard(_))) {
                return Err(Error::InvalidPattern { at: start });
            }
            // A wildcard must be the final segment. This is checked before
            // validating any following segment.
            let is_last = raw_segments_is_last(pattern, start, raw);
            if !is_last {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            let decoded = decode_segment(raw, start, true)?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok(segments)
}

fn raw_segments_is_last(pattern: &str, start: usize, raw: &str) -> bool {
    let end = start + raw.len();
    end == pattern.len() || (end + 1 == pattern.len() && pattern.as_bytes().get(end) == Some(&b'/'))
}

fn parse_path(path: &str, pattern: bool) -> Result<Vec<String>, Error> {
    parse_raw_segments(path, pattern)?
        .into_iter()
        .map(|(raw, start)| decode_segment(raw, start, pattern))
        .collect()
}

fn parse_raw_segments<'a>(
    input: &'a str,
    is_pattern: bool,
) -> Result<Vec<(&'a str, usize)>, Error> {
    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1usize;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(invalid(i));
            }
            if i == start {
                return Err(invalid(start));
            }
            parts.push((&input[start..i], start));
            start = i + 1;
        }
    }

    if start >= bytes.len() {
        return Err(invalid(bytes.len() - 1));
    }
    parts.push((&input[start..], start));

    if is_pattern {
        // A non-final wildcard is diagnosed at its own segment start before
        // any later segment is inspected.
        for (idx, (part, part_start)) in parts.iter().enumerate() {
            if part.starts_with('*') && idx + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *part_start });
            }
        }
    }

    Ok(parts)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, start: usize, is_pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let bytes = raw.as_bytes();
    let mut decoded = Vec::new();
    let mut i = 0usize;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(start + i));
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return Err(invalid(start + i));
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return Err(invalid(start + i));
            };
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(start + i));
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(byte) {
                return Err(invalid(start + i));
            }
            decoded.push(byte);
            i += 1;
        }
    }

    // All accepted bytes are ASCII, including percent-decoded bytes.
    Ok(String::from_utf8(decoded).unwrap_or_default())
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0usize;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = path.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index != path.len() {
        return None;
    }
    Some(params)
}

        let mut best: Option<(usize, Match)> = None;

        for route in &self.routes {
            let mut params = Vec::new();
            let mut request_index = 0;
            let mut matched = true;

            for segment in &route.segments {
                match segment {
                    Segment::Static(expected) => {
                        if request_segments.get(request_index) != Some(expected) {
                            matched = false;
                            break;
                        }
                        request_index += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = request_segments.get(request_index) {
                            params.push((name.clone(), value.clone()));
                            request_index += 1;
                        } else {
                            matched = false;
                            break;
                        }

                    Segment::Wildcard(name) => {
                        let value = request_segments[request_index..].join("/");
                        params.push((name.clone(), value));
                        request_index = request_segments.len();
                        break;
                    }
                }
            }

            if !matched || request_index != request_segments.len() {
                continue;
            }

            let candidate = Match {
                route: route.id,
                params,
            };
            match &best {
                None => best = Some((route.id, candidate)),
                Some((best_id, _)) => {
                    let current = &self.routes[route.id];
                    let previous = &self.routes[*best_id];

                        best = Some((route.id, candidate));
                    }

            }
        }

    }

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|byte| *byte == b'/')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let segment_bytes = &bytes[start..end];
        if segment_bytes[0] == b':' || segment_bytes[0] == b'*' {
            let wildcard = segment_bytes[0] == b'*';
            let name_bytes = &segment_bytes[1..];
            if !valid_name(name_bytes) {
                return Err(Error::InvalidPattern { at: start });
            }
            let name = String::from_utf8(name_bytes.to_vec()).unwrap();
            if names.iter().any(|existing| existing == &name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.clone());

            if wildcard && end != bytes.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if wildcard {
                Segment::Wildcard(name)
            } else {
                Segment::Param(name)
            });
        } else {
            let decoded = decode_segment(segment_bytes, start, true)?;
            segments.push(Segment::Static(decoded));
        }

        if end == bytes.len() {
            break;
        }
        if end + 1 == bytes.len() {
            return Err(Error::InvalidPattern { at: end });
        }
        start = end + 1;
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut segments = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|byte| *byte == b'/')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }

        segments.push(decode_segment(&bytes[start..end], start, false)?);

        if end == bytes.len() {
            break;
        }
        if end + 1 == bytes.len() {
            return Err(Error::InvalidPath { at: end });
        }
        start = end + 1;
    }

    Ok(segments)
}

fn valid_name(name: &[u8]) -> bool {
    let Some((&first, rest)) = name.split_first() else {
        return false;
    };
    if !first.is_ascii_lowercase() {
        return false;
    }
    rest.iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn decode_segment(bytes: &[u8], base: usize, pattern: bool) -> Result<String, Error> {
    let mut decoded = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' {
            let valid_escape = index + 2 < bytes.len()
                && is_upper_hex(bytes[index + 1])
                && is_upper_hex(bytes[index + 2]);
            if !valid_escape {
                return invalid_segment(pattern, base + index);
            }
            let value = (hex_value(bytes[index + 1]) << 4) | hex_value(bytes[index + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return invalid_segment(pattern, base + index);
            }
            decoded.push(value);
            index += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            index += 1;
        } else {
            return invalid_segment(pattern, base + index);
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn invalid_segment<T>(pattern: bool, at: usize) -> Result<T, Error> {
    if pattern {
        Err(Error::InvalidPattern { at })
    } else {
        Err(Error::InvalidPath { at })
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

    match byte {
        b'0'..=b'9' => byte - b'0',

        _ => 0,
    }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(route, &request) {
                let replace = match &best {
                    None => true,
                    Some((old, _)) => route.rank > old.rank,
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    parse_segments(input, true)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    let segments = parse_segments(input, false)?;
    let mut decoded = Vec::new();

    for segment in segments {
        match segment {
            Segment::Static(value) => decoded.push(value),
            _ => unreachable!(),
        }
    }
    Ok(decoded)
}

fn parse_segments(input: &str, pattern: bool) -> Result<Vec<Segment>, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut spans: Vec<(usize, usize)> = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == start {
                return Err(invalid(start));
            }
            spans.push((start, i));
            start = i + 1;
        }
    }

    let trailing_slash = bytes.last() == Some(&b'/');
    if trailing_slash {
        if start == bytes.len() {
            // The preceding segments are still validated before reporting this.
        } else {
            spans.push((start, bytes.len()));
        }
    } else {
        spans.push((start, bytes.len()));
    }

    let mut result = Vec::new();
    let mut names: Vec<String> = Vec::new();
    let mut wildcard_seen = false;

    for (index, &(begin, end)) in spans.iter().enumerate() {
        let raw = &input[begin..end];

        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            if wildcard_seen {
                return Err(invalid(begin));
            }

            let is_wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(invalid(begin));
            }
            if is_wildcard && index + 1 != spans.len() {
                return Err(invalid(begin));
            }

            names.push(name.to_string());
            if is_wildcard {
                wildcard_seen = true;
                result.push(Segment::Wildcard(name.to_string()));
            } else {
                result.push(Segment::Param(name.to_string()));
            }
        } else {
            let value = decode_static(raw, begin).map_err(invalid)?;
            result.push(Segment::Static(value));
        }
    }

    if trailing_slash {
        return Err(invalid(bytes.len() - 1));
    }

    Ok(result)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_static(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value as char);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b as char);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(decoded)
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(expected) => {
                if request.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = request.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let mut value = String::new();
                for (i, part) in request[path_index..].iter().enumerate() {
                    if i != 0 {
                        value.push('/');
                    }
                    value.push_str(part);
                }
                params.push((name.clone(), value));
                return Some(params);
            }
        }
    }

    if path_index == request.len() {
        Some(params)
    } else {
        None
    }
}

fn parse_pattern(pattern: &str) -> Result<Route, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    if pattern == "/" {
        return Ok(Route {
            segments: Vec::new(),
            trailing_slash: false,
            rank: vec![3],
        });
    }

    let bytes = pattern.as_bytes();
    let trailing_slash = bytes.last() == Some(&b'/');
    let content_end = if trailing_slash { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let mut rank = Vec::new();

    let mut start = 1;
    while start < content_end {
        let end = bytes[start..content_end]
            .iter()
            .position(|b| *b == b'/')
            .map(|offset| start + offset)
            .unwrap_or(content_end);

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..end];
        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if marker == b'*' && end != content_end {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            if marker == b':' {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            } else {
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            }
        } else {
            let decoded = decode_segment(raw, start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }

        if end == content_end {
            break;
        }
        start = end + 1;
    }

    if trailing_slash {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }

    Ok(Route {
        segments,
        trailing_slash,
        rank,
    })
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    let head_end = path
        .char_indices()
        .find(|(_, ch)| *ch == '?' || *ch == '#')
        .map(|(index, _)| index)
        .unwrap_or(path.len());
    let head = &path[..head_end];
    if !head.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    if head == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = head.as_bytes();
    let trailing_slash = bytes.last() == Some(&b'/');
    let content_end = if trailing_slash { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < content_end {
        let end = bytes[start..content_end]
            .iter()
            .position(|b| *b == b'/')
            .map(|offset| start + offset)
            .unwrap_or(content_end);

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }

        let decoded = decode_segment(&head[start..end], start)
            .map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);

        if end == content_end {
            break;
        }
        start = end + 1;
    }

    if trailing_slash {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }

    Ok((segments, trailing_slash))
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let route = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let (path_segments, trailing_slash) = parse_path(path)?;
        let mut best: Option<(usize, Match)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if route.trailing_slash != trailing_slash {
                continue;
            }

            let has_wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
            let fixed_len = if has_wildcard {
                route.segments.len() - 1
            } else {
                route.segments.len()
            };

            if (!has_wildcard && fixed_len != path_segments.len())
                || (has_wildcard && path_segments.len() < fixed_len)
            {
                continue;
            }

            let mut params = Vec::new();
            let mut matched = true;
            for i in 0..fixed_len {
                match &route.segments[i] {
                    Segment::Static(value) if value == &path_segments[i] => {}
                    Segment::Static(_) => {
                        matched = false;
                        break;
                    }
                    Segment::Param(name) => {
                        params.push((name.clone(), path_segments[i].clone()));
                    }
                    Segment::Wildcard(_) => {
                        matched = false;
                        break;
                    }
                }
            }

            if !matched {
                continue;
            }

            if let Some(Segment::Wildcard(name)) = route.segments.last() {
                let mut value = String::new();
                for (index, segment) in path_segments[fixed_len..].iter().enumerate() {
                    if index > 0 {
                        value.push('/');
                    }
                    value.push_str(segment);
                }
                params.push((name.clone(), value));
            }

            let candidate = Match {
                route: route_id,
                params,
            };
            let replace = match &best {
                None => true,
                Some((best_id, _)) => {
                    route.rank > self.routes[*best_id].rank
                        || (route.rank == self.routes[*best_id].rank && route_id < *best_id)
                }
            };
            if replace {
                best = Some((route_id, candidate));
            }
        }

        best.map(|(_, result)| result).ok_or(Error::NotFound)
    }
}
