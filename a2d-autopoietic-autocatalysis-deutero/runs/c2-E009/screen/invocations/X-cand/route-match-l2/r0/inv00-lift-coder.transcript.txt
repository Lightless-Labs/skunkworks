#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::new();
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head, false)?;

        let mut best: Option<(usize, Match)> = None;
        for (index, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(&route.segments, &request_segments) {
                let replace = match &best {
                    None => true,
                    Some((best_index, _)) => {
                        route.rank > self.routes[*best_index].rank
                    }
                };
                if replace {
                    best = Some((index, Match { route: route.id, params }));
                }
            }
        }

        best.map(|(_, matched)| matched).ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    let raw_segments = parse_raw_segments(pattern, true)?;
    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();

    for (raw, start) in raw_segments {
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if segments.iter().any(|segment| matches!(segment, Segment::Wildcard(_))) {
                return Err(Error::InvalidPattern { at: start });
            }
            // A wildcard must be the final segment. This is checked before
            // validating any following segment.
            let is_last = raw_segments_is_last(pattern, start, raw);
            if !is_last {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            let decoded = decode_segment(raw, start, true)?;
            segments.push(Segment::Static(decoded));
        }
    }

    Ok(segments)
}

fn raw_segments_is_last(pattern: &str, start: usize, raw: &str) -> bool {
    let end = start + raw.len();
    end == pattern.len() || (end + 1 == pattern.len() && pattern.as_bytes().get(end) == Some(&b'/'))
}

fn parse_path(path: &str, pattern: bool) -> Result<Vec<String>, Error> {
    parse_raw_segments(path, pattern)?
        .into_iter()
        .map(|(raw, start)| decode_segment(raw, start, pattern))
        .collect()
}

fn parse_raw_segments<'a>(
    input: &'a str,
    is_pattern: bool,
) -> Result<Vec<(&'a str, usize)>, Error> {
    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1usize;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(invalid(i));
            }
            if i == start {
                return Err(invalid(start));
            }
            parts.push((&input[start..i], start));
            start = i + 1;
        }
    }

    if start >= bytes.len() {
        return Err(invalid(bytes.len() - 1));
    }
    parts.push((&input[start..], start));

    if is_pattern {
        // A non-final wildcard is diagnosed at its own segment start before
        // any later segment is inspected.
        for (idx, (part, part_start)) in parts.iter().enumerate() {
            if part.starts_with('*') && idx + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *part_start });
            }
        }
    }

    Ok(parts)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, start: usize, is_pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let bytes = raw.as_bytes();
    let mut decoded = Vec::new();
    let mut i = 0usize;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(start + i));
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return Err(invalid(start + i));
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return Err(invalid(start + i));
            };
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(start + i));
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(byte) {
                return Err(invalid(start + i));
            }
            decoded.push(byte);
            i += 1;
        }
    }

    // All accepted bytes are ASCII, including percent-decoded bytes.
    Ok(String::from_utf8(decoded).unwrap_or_default())
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0usize;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = path.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index != path.len() {
        return None;
    }
    Some(params)
}
