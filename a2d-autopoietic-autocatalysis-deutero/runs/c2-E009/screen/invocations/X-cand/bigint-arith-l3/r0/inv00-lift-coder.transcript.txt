use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian decimal digits. Zero is represented by [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' || bytes[0] == b'-' {
            1
        } else {
            0
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = separator_error {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut excess_digit = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' {
                digit_count += 1;
                if first_digit.is_none() {
                    first_digit = Some((index, byte));
                }
                if digit_count == max_digits.saturating_add(1) {
                    excess_digit = Some(index);
                }
            }
        }

        if digit_count > 1 {
            if let Some((index, b'0')) = first_digit {
                return Err(Error::LeadingZero { index });
            }
        }

        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: excess_digit.unwrap_or_else(|| {
                    first_digit.map_or(start, |(index, _)| index)
                }),
            });
        }

        let mut digits = Vec::new();
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        let negative = bytes[0] == b'-' && !is_zero_digits(&digits);
        Ok(Self { negative, digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }

        for position_from_left in 0..width {
            let position_from_right = width - 1 - position_from_left;
            if group > 0
                && position_from_left > 0
                && position_from_right % group == 0
            {
                out.push('_');
            }

            let digit = if position_from_right < self.digits.len() {
                self.digits[position_from_right]
            } else {
                0
            };
            out.push(char::from(b'0' + digit));
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        let self_zero = self.is_zero();
        let rhs_zero = rhs.is_zero();

        match (self.negative && !self_zero, rhs.negative && !rhs_zero) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let self_negative = self.negative && !self.is_zero();
        let rhs_negative = rhs.negative && !rhs.is_zero();

        if self_negative == rhs_negative {
            Self::from_parts(
                self_negative,
                add_magnitude(&self.digits, &rhs.digits),
            )
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => Self::from_parts(
                    self_negative,
                    sub_magnitude(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    rhs_negative,
                    sub_magnitude(&rhs.digits, &self.digits),
                ),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let index = i + j;
                let value = result[index] as u16 + a as u16 * b as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
            }

            let mut index = i + rhs.digits.len();
            while carry > 0 {
                if index == result.len() {
                    result.push(0);
                }
                let value = result[index] as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
                index += 1;
            }
        }

        Self::from_parts(self.negative != rhs.negative, result)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        match cmp_magnitude(&self.digits, &rhs.digits) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                let quotient_negative = self.negative != rhs.negative;
                let quotient = Self::from_parts(quotient_negative, vec![1]);
                return Ok((quotient, Self::zero()));
            }
            Ordering::Greater => {}
        }

        let mut quotient_be = Vec::new();
        let mut remainder = vec![0u8];

        for &digit in self.digits.iter().rev() {
            remainder.insert(0, digit);
            trim_magnitude(&mut remainder);

            let mut quotient_digit = 0u8;
            while cmp_magnitude(&remainder, &rhs.digits) != Ordering::Less {
                remainder = sub_magnitude(&remainder, &rhs.digits);
                quotient_digit += 1;
            }
            quotient_be.push(quotient_digit);
        }

        while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
            quotient_be.remove(0);
        }
        quotient_be.reverse();

        let quotient = Self::from_parts(self.negative != rhs.negative, quotient_be);
        let remainder = Self::from_parts(self.negative, remainder);
        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        trim_magnitude(&mut digits);
        let is_zero = is_zero_digits(&digits);
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn is_zero(&self) -> bool {
        is_zero_digits(&self.digits)
    }
}

fn is_zero_digits(digits: &[u8]) -> bool {
    digits.len() == 1 && digits[0] == 0
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}

fn cmp_magnitude(lhs: &[u8], rhs: &[u8]) -> Ordering {
    match lhs.len().cmp(&rhs.len()) {
        Ordering::Equal => {
            for (&a, &b) in lhs.iter().rev().zip(rhs.iter().rev()) {
                match a.cmp(&b) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u8;
    let max_len = lhs.len().max(rhs.len());

    for index in 0..max_len {
        let a = lhs.get(index).copied().unwrap_or(0);
        let b = rhs.get(index).copied().unwrap_or(0);
        let value = a + b + carry;
        result.push(value % 10);
        carry = value / 10;
    }
    if carry > 0 {
        result.push(carry);
    }
    result
}

fn sub_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    // The caller ensures lhs >= rhs.
    let mut result = Vec::new();
    let mut borrow = 0i8;

    for index in 0..lhs.len() {
        let a = lhs[index] as i8 - borrow;
        let b = rhs.get(index).copied().unwrap_or(0) as i8;
        if a < b {
            result.push((a + 10 - b) as u8);
            borrow = 1;
        } else {
            result.push((a - b) as u8);
            borrow = 0;
        }
    }

    trim_magnitude(&mut result);
    result
}
