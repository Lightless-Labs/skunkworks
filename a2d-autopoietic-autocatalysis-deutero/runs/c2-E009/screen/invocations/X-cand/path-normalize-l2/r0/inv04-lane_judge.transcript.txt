pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str, offset: usize) -> Vec<(String, usize)> {
    let mut result = Vec::new();
    let mut start = None;

    for (at, byte) in path.bytes().enumerate() {
        if byte == b'/' {
            if let Some(begin) = start.take() {
                result.push((path[begin..at].to_owned(), offset + begin));
            }
        } else if start.is_none() {
            start = Some(at);
        }
    }

    if let Some(begin) = start {
        result.push((path[begin..].to_owned(), offset + begin));
    }
    result
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_absolute(
    initial: Vec<String>,
    parts: Vec<(String, usize)>,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    for (part, at) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(stack)
}

fn format_absolute(parts: Vec<String>, intent: bool) -> String {
    let mut output = String::from("/");
    output.push_str(&parts.join("/"));
    if intent && output.len() > 1 {
        output.push('/');
    }
    output
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let parts = components(path, 0);

    if absolute {
        let reduced = reduce_absolute(Vec::new(), parts)?;
        Ok(format_absolute(reduced, intent))
    } else {
        let mut stack: Vec<String> = Vec::new();
        for (part, _) in parts {
            match part.as_str() {
                "." => {}
                ".." => {
                    if stack.last().is_some_and(|last| last != "..") {
                        stack.pop();
                    } else {
                        stack.push(part);
                    }
                }
                _ => stack.push(part),
            }
        }

        if stack.is_empty() {
            return Ok(".".to_owned());
        }
        let mut output = stack.join("/");
        if intent {
            output.push('/');
        }
        Ok(output)
    }
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    validate_raw(path).map_err(|_| ())?;
    let parts = components(path, 0);
    reduce_absolute(Vec::new(), parts).map_err(|_| ())
}

    let cwd_parts = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }

    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a, I>(
    components: I,
    absolute: bool,
    mut offset_of: impl FnMut(&str) -> usize,
) -> Result<Vec<&'a str>, PathError>
where
    I: IntoIterator<Item = &'a str>,
{
    let mut result = Vec::new();

    for component in components {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if result.last().is_some_and(|last| *last != "..") {
                result.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: offset_of(component),
                });
            } else {
                result.push(component);
            }
        } else {
            result.push(component);
        }
    }

    Ok(result)
}

fn build_path<'a>(components: &[&'a str], absolute: bool, trailing: bool) -> String {
    let mut output = String::new();

    if absolute {
        output.push('/');
    }

    let mut first = true;
    for component in components {
        if !first {
            output.push('/');
        }
        output.push_str(component);
        first = false;
    }

    if components.is_empty() {
        if absolute {
            return output;
        }
        return ".".to_string();
    }

    if trailing {
        output.push('/');
    }

    output
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let trailing = directory_intent(path);
    let mut search_from = 0usize;

    let components = reduce_components(
        path.split('/'),
        absolute,
        |_| {
            let offset = path[search_from..]
                .find("..")
                .map(|relative| search_from + relative)
                .unwrap_or(search_from);
            search_from = offset.saturating_add(2);
            offset
        },
    )?;

    Ok(build_path(&components, absolute, trailing))
}

fn normalize_context(path: &str) -> Result<Vec<&str>, PathError> {
    validate_raw(path)?;
    if path.as_bytes().first() != Some(&b'/') {
        return Err(PathError::InvalidCwd);
    }

    let components = reduce_components(path.split('/'), true, |_| 0)
        .map_err(|_| PathError::InvalidCwd)?;
    Ok(components)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalize_context(cwd).map_err(|_| PathError::InvalidCwd)?;

    let home_components = match home {
        Some(home_path) => Some(normalize_context(home_path).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let (base, remainder, remainder_offset) = if bytes.first() == Some(&b'/') {
        (&[][..], path, 0usize)
    } else if bytes.first() == Some(&b'~') {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }

        let base = home_components
            .as_deref()
            .ok_or(PathError::MissingHome)?;
        if bytes.len() == 1 {
            (base, "", path.len())
        } else {
            (base, &path[2..], 2)
        }
    } else {
        (&cwd_components[..], path, 0usize)
    };

    let trailing = directory_intent(path);
    let mut components = base.to_vec();

    for (component, relative_offset) in remainder
        .split('/')
        .scan(0usize, |offset, component| {
            let current = *offset;
            *offset = offset.saturating_add(component.len()).saturating_add(1);
            Some((component, current))
        })
    {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if components.last().is_some_and(|last| *last != "..") {
                components.pop();
            } else {
                return Err(PathError::Escape {
                    at: remainder_offset.saturating_add(relative_offset),
                });
            }
        } else {
            components.push(component);
        }
    }

    let _ = remainder_offset;
    Ok(build_path(&components, true, trailing))
}

pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/').find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a>(
    components: impl Iterator<Item = (usize, &'a str)>,
    absolute: bool,
    initial: Vec<(String, usize)>,
) -> Result<Vec<(String, usize)>, PathError> {
    let mut result = initial;

    for (at, component) in components {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if result.last().is_some_and(|(name, _)| name != "..") {
                result.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                result.push(("..".to_owned(), at));
            }
        } else {
            result.push((component.to_owned(), at));
        }
    }

    Ok(result)
}

fn component_offsets(path: &str, start: usize) -> impl Iterator<Item = (usize, &str)> {
    path[start..]
        .split('/')
        .scan(start, |offset, component| {
            let current = *offset;
            *offset += component.len() + 1;
            Some((current, component))
        })
}

fn format_result(components: &[(String, usize)], absolute: bool, intent: bool) -> String {
    let mut output = String::new();
    if absolute {
        output.push('/');
    }

    for (index, (component, _)) in components.iter().enumerate() {
        if index > 0 {
            output.push('/');
        }
        output.push_str(component);
    }

    if output.is_empty() {
        output.push('.');
    }

    if intent && output != "/" && output != "." {
        output.push('/');
    }

    output
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let components = reduce_components(component_offsets(path, 0), absolute, Vec::new())?;
    let intent = directory_intent(path);

    Ok(format_result(&components, absolute, intent))
}

fn validated_context(path: &str) -> Result<Vec<(String, usize)>, ()> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return Err(());
    }
    reduce_components(component_offsets(path, 0), true, Vec::new()).map_err(|_| ())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(validated_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let (absolute, initial, remainder_start) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home_components = home_components.ok_or(PathError::MissingHome)?;
            let start = if bytes.len() > 1 { 2 } else { 1 };
            (true, home_components, start)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (true, Vec::new(), 0)
    } else {
        (true, cwd_components, 0)
    };

    let components = reduce_components(
        component_offsets(path, remainder_start),
        absolute,
        initial,
    )?;
    Ok(format_result(&components, true, directory_intent(path)))
}
