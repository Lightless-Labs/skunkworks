#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }

    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a, I>(
    components: I,
    absolute: bool,
    mut offset_of: impl FnMut(&str) -> usize,
) -> Result<Vec<&'a str>, PathError>
where
    I: IntoIterator<Item = &'a str>,
{
    let mut result = Vec::new();

    for component in components {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if result.last().is_some_and(|last| *last != "..") {
                result.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: offset_of(component),
                });
            } else {
                result.push(component);
            }
        } else {
            result.push(component);
        }
    }

    Ok(result)
}

fn build_path<'a>(components: &[&'a str], absolute: bool, trailing: bool) -> String {
    let mut output = String::new();

    if absolute {
        output.push('/');
    }

    let mut first = true;
    for component in components {
        if !first {
            output.push('/');
        }
        output.push_str(component);
        first = false;
    }

    if components.is_empty() {
        if absolute {
            return output;
        }
        return ".".to_string();
    }

    if trailing {
        output.push('/');
    }

    output
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let trailing = directory_intent(path);
    let mut search_from = 0usize;

    let components = reduce_components(
        path.split('/'),
        absolute,
        |_| {
            let offset = path[search_from..]
                .find("..")
                .map(|relative| search_from + relative)
                .unwrap_or(search_from);
            search_from = offset.saturating_add(2);
            offset
        },
    )?;

    Ok(build_path(&components, absolute, trailing))
}

fn normalize_context(path: &str) -> Result<Vec<&str>, PathError> {
    validate_raw(path)?;
    if path.as_bytes().first() != Some(&b'/') {
        return Err(PathError::InvalidCwd);
    }

    let components = reduce_components(path.split('/'), true, |_| 0)
        .map_err(|_| PathError::InvalidCwd)?;
    Ok(components)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalize_context(cwd).map_err(|_| PathError::InvalidCwd)?;

    let home_components = match home {
        Some(home_path) => Some(normalize_context(home_path).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let (base, remainder, remainder_offset) = if bytes.first() == Some(&b'/') {
        (&[][..], path, 0usize)
    } else if bytes.first() == Some(&b'~') {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }

        let base = home_components
            .as_deref()
            .ok_or(PathError::MissingHome)?;
        if bytes.len() == 1 {
            (base, "", path.len())
        } else {
            (base, &path[2..], 2)
        }
    } else {
        (&cwd_components[..], path, 0usize)
    };

    let trailing = directory_intent(path);
    let mut components = base.to_vec();

    for (component, relative_offset) in remainder
        .split('/')
        .scan(0usize, |offset, component| {
            let current = *offset;
            *offset = offset.saturating_add(component.len()).saturating_add(1);
            Some((component, current))
        })
    {
        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if components.last().is_some_and(|last| *last != "..") {
                components.pop();
            } else {
                return Err(PathError::Escape {
                    at: remainder_offset.saturating_add(relative_offset),
                });
            }
        } else {
            components.push(component);
        }
    }

    let _ = remainder_offset;
    Ok(build_path(&components, true, trailing))
}
