fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push('%');
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_rank(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

fn sorted_evidence(items: &[Evidence]) -> Vec<Evidence> {
    let mut out = items.to_vec();
    out.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut out = format!("missing|package={}", encode(package));
            for item in sorted_evidence(evidence) {
                out.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(item.constraint),
                    path_text(&item.path)
                ));
            }
            out
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            for item in sorted_evidence(evidence) {
                out.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(item.constraint),
                    path_text(&item.path)
                ));
            }
            out
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            for item in sorted_evidence(evidence) {
                out.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(item.constraint),
                    path_text(&item.path)
                ));
            }
            out
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.enable[{}]", base, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = Vec::<String>::new();
    for package in catalog {
        if package_names.iter().any(|name| name == &package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        package_names.push(package.name.clone());
    }

    for package in catalog {
        for (i, release) in package.releases.iter().enumerate() {
            if package.releases[..i]
                .iter()
                .any(|prior| prior.version == release.version)
            {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = Vec::<String>::new();
    for lock in locks {
        if lock_names.iter().any(|name| name == &lock.package) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
        lock_names.push(lock.package.clone());
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = Vec::<String>::new();
            for feature in &release.features {
                if seen.iter().any(|f| f == &feature.name) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
                seen.push(feature.name.clone());
            }
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    requests: Vec<String>,
}

struct Closure {
    active: std::collections::BTreeMap<String, Active>,
    activated: std::collections::BTreeMap<String, Vec<String>>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
    error: Option<ResolveError>,
}

fn platform_name(platform: Platform) -> &'static str {
    match platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn release_for<'a>(
    catalog: &'a [Package],
    selected: &std::collections::BTreeMap<String, Version>,
    name: &str,
) -> Option<&'a Release> {
    let version = selected.get(name)?;
    let package = catalog.iter().find(|p| p.name == name)?;
    package.releases.iter().find(|r| r.version == *version)
}

fn add_request(active: &mut Active, feature: &str) -> bool {
    if active.requests.iter().any(|f| f == feature) {
        false
    } else {
        active.requests.push(feature.to_string());
        true
    }
}

fn compute_closure(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    use std::collections::BTreeMap;

    let mut active = BTreeMap::<String, Active>::new();
    for (i, root) in roots.iter().enumerate() {
        let entry = active.entry(root.package.clone()).or_default();
        entry.constraints.push(root.constraint);
        entry.origins.push(Origin {
            constraint: root.constraint,
            path: vec![format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                root.package,
                constraint_text(root.constraint),
                {
                    let mut fs = root.features.clone();
                    fs.sort();
                    fs.dedup();
                    fs.join(",")
                }
            )],
        });
        for feature in &root.features {
            add_request(entry, feature);
        }
    }

    let mut changed = true;
    while changed {
        changed = false;

        let names: Vec<String> = active.keys().cloned().collect();
        for name in names {
            let Some(release) = release_for(catalog, selected, &name) else {
                continue;
            };
            let Some(entry) = active.get_mut(&name) else {
                continue;
            };

            let mut index = 0;
            while index < entry.requests.len() {
                let requested = entry.requests[index].clone();
                index += 1;
                if let Some(feature) = release.features.iter().find(|f| f.name == requested) {
                    for enabled in &feature.enables {
                        if add_request(entry, enabled) {
                            changed = true;
                        }
                    }
                }
            }

            let requests = entry.requests.clone();
            for (dep_index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !requests.iter().any(|f| f == g))
                {
                    continue;
                }
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    name,
                    version_text(release.version),
                    dep_index,
                    dep.package,
                    constraint_text(dep.constraint),
                    {
                        let mut fs = dep.features.clone();
                        fs.sort();
                        fs.dedup();
                        fs.join(",")
                    },
                    dep.gate.as_deref().unwrap_or("*"),
                    platform_name(dep.platform)
                );
                let entry = active.entry(dep.package.clone()).or_default();
                let origin_exists = entry.origins.iter().any(|origin| {
                    origin.constraint == dep.constraint
                        && origin.path.last().is_some_and(|last| last == &token)
                });
                if !origin_exists {
                    entry.constraints.push(dep.constraint);
                    entry.origins.push(Origin {
                        constraint: dep.constraint,
                        path: vec![token],
                    });
                    changed = true;
                }
                for feature in &dep.features {
                    if add_request(entry, feature) {
                        changed = true;
                    }
                }
            }
        }
    }

    let mut paths = BTreeMap::<String, Vec<String>>::new();
    for (i, root) in roots.iter().enumerate() {
        let path = vec![format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            root.package,
            constraint_text(root.constraint),
            {
                let mut fs = root.features.clone();
                fs.sort();
                fs.dedup();
                fs.join(",")
            }
        )];
        let current = paths.entry(root.package.clone()).or_insert_with(|| path.clone());
        if (path.len(), &path) < (current.len(), current) {
            *current = path;
        }
    }

    let mut progressed = true;
    while progressed {
        progressed = false;
        for (source, source_path) in paths.clone() {
            let Some(release) = release_for(catalog, selected, &source) else {
                continue;
            };
            let Some(entry) = active.get(&source) else {
                continue;
            };
            let requests = &entry.requests;
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !requests.iter().any(|f| f == g))
                {
                    continue;
                }
                let mut next = source_path.clone();
                next.push(format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    source,
                    version_text(release.version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    {
                        let mut fs = dep.features.clone();
                        fs.sort();
                        fs.dedup();
                        fs.join(",")
                    },
                    dep.gate.as_deref().unwrap_or("*"),
                    platform_name(dep.platform)
                ));
                match paths.get(&dep.package) {
                    None => {
                        paths.insert(dep.package.clone(), next);
                        progressed = true;
                    }
                    Some(old) if (next.len(), &next) < (old.len(), old) => {
                        paths.insert(dep.package.clone(), next);
                        progressed = true;
                    }
                    _ => {}
                }
            }
        }
    }

    for (name, entry) in active.iter_mut() {
        entry.constraints = canonical_constraints(std::mem::take(&mut entry.constraints));
        entry.requests.sort();
        entry.requests.dedup();
        entry.origins.sort_by(|a, b| {
            constraint_rank(a.constraint)
                .cmp(&constraint_rank(b.constraint))
                .then_with(|| a.path.cmp(&b.path))
        });
        for origin in &mut entry.origins {
            if let Some(canonical) = paths.get(name) {
                if origin.path.len() > 1 {
                    origin.path = canonical.clone();
                }
            }
        }
    }

    Closure {
        active,
        activated: BTreeMap::new(),
        paths,
        error: None,
    }
}

fn make_evidence(entry: &Active) -> Vec<Evidence> {
    let mut evidence = entry
        .origins
        .iter()
        .map(|origin| Evidence {
            constraint: origin.constraint,
            path: origin.path.clone(),
        })
        .collect::<Vec<_>>();
    evidence.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    evidence
}

fn inspect_node(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> Result<(Closure, Vec<String>), ResolveError> {
    use std::collections::BTreeSet;

    let mut closure = compute_closure(catalog, roots, selected, target);

    let missing: Vec<String> = closure
        .active
        .keys()
        .filter(|name| !catalog.iter().any(|p| &p.name == *name))
        .cloned()
        .collect();
    if let Some(package) = missing.first() {
        return Err(ResolveError::Missing {
            package: package.clone(),
            evidence: make_evidence(&closure.active[package]),
        });
    }

    for (package, version) in selected {
        let Some(entry) = closure.active.get(package) else {
            continue;
        };
        if entry
            .constraints
            .iter()
            .any(|c| !matches(*version, *c))
        {
            return Err(ResolveError::Conflict {
                package: package.clone(),
                selected: *version,
                requirements: entry.constraints.clone(),
                evidence: make_evidence(entry),
            });
        }
    }

    let mut unselected = Vec::new();
    for (package, entry) in &closure.active {
        if selected.contains_key(package) {
            continue;
        }
        let p = catalog.iter().find(|p| &p.name == package).unwrap();
        if !p.releases.iter().any(|r| {
            entry
                .constraints
                .iter()
                .all(|c| matches(r.version, *c))
        }) {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements: entry.constraints.clone(),
                evidence: make_evidence(entry),
            });
        }
        unselected.push(package.clone());
    }

    for (package, version) in selected {
        let Some(entry) = closure.active.get(package) else {
            continue;
        };
        let Some(release) = release_for(catalog, selected, package) else {
            continue;
        };
        let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
        for feature in &entry.requests {
            if !declared.contains(feature.as_str()) {
                return Err(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: closure.paths.get(package).cloned().unwrap_or_default(),
                });
            }
        }
    }

    Ok((closure, unselected))
}

#[derive(Clone)]
struct Best {
    cost: usize,
    selected: std::collections::BTreeMap<String, Version>,
}

fn lex_better(
    left: &std::collections::BTreeMap<String, Version>,
    right: &std::collections::BTreeMap<String, Version>,
    names: &[String],
) -> bool {
    for name in names {
        match (left.get(name), right.get(name)) {
            (None, None) => {}
            (None, Some(_)) => return false,
            (Some(_), None) => return true,
            (Some(a), Some(b)) if a != b => return a > b,
            _ => {}
        }
    }
    false
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
    attempts: &mut usize,
    selected: &mut std::collections::BTreeMap<String, Version>,
    first_error: &mut Option<ResolveError>,
    best: &mut Option<Best>,
    catalog_names: &[String],
) -> Result<(), ResolveError> {
    let (closure, unselected) = match inspect_node(catalog, roots, selected, target) {
        Ok(v) => v,
        Err(error) => {
            if first_error.is_none() {
                *first_error = Some(error);
            }
            return Ok(());
        }
    };

    let Some(package) = unselected.first() else {
        let cost = locks
            .iter()
            .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
            .count();
        let replace = match best {
            None => true,
            Some(current) if cost < current.cost => true,
            Some(current) if cost == current.cost => {
                lex_better(selected, &current.selected, catalog_names)
            }
            _ => false,
        };
        if replace {
            *best = Some(Best {
                cost,
                selected: selected.clone(),
            });
        }
        return Ok(());
    };

    let entry = &closure.active[package];
    let package_data = catalog.iter().find(|p| p.name == *package).unwrap();
    let mut candidates: Vec<Version> = package_data
        .releases
        .iter()
        .filter(|release| {
            entry
                .constraints
                .iter()
                .all(|c| matches(release.version, *c))
        })
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(lock) = locks.iter().find(|lock| lock.package == *package) {
        if let Some(position) = candidates.iter().position(|v| *v == lock.version) {
            let version = candidates.remove(position);
            candidates.insert(0, version);
        }
    }

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.insert(package.clone(), version);
        let result = search(
            catalog,
            roots,
            locks,
            target,
            max_attempts,
            attempts,
            selected,
            first_error,
            best,
            catalog_names,
        );
        selected.remove(package);
        result?;
    }
    Ok(())
}

fn final_constraints(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    compute_closure(catalog, roots, selected, target)
        .active
        .into_iter()
        .map(|(name, active)| (name, active.constraints))
        .collect()
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    use std::collections::BTreeMap;

    validate(catalog, roots, locks)?;

    let mut selected = BTreeMap::new();
    let mut attempts = 0;
    let mut first_error = None;
    let mut best = None;
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    search(
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        &mut attempts,
        &mut selected,
        &mut first_error,
        &mut best,
        &catalog_names,
    )?;

    let Some(best) = best else {
        return Err(first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let constraints = final_constraints(catalog, roots, &best.selected, target);
    let selected_out = best
        .selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut lock_map = BTreeMap::<String, Version>::new();
    for lock in locks {
        lock_map.insert(lock.package.clone(), lock.version);
    }
    let mut all_names: Vec<String> = lock_map
        .keys()
        .chain(best.selected.keys())
        .cloned()
        .collect();
    all_names.sort();
    all_names.dedup();

    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = best.selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !constraints.get(&package).is_some_and(|cs| {
                    cs.iter().all(|c| matches(old, *c))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
