#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<u8>; 5],
    day_unrestricted: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_width(field: Field) -> u8 {
    let (lo, hi) = bounds(field);
    hi - lo + 1
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_name(field: Field, text: &str) -> Option<u8> {
    let upper = text.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn is_digits(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn valid_atom_shape(field: Field, s: &str) -> bool {
    is_digits(s) || parse_name(field, s).is_some()
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for b in s.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(n)
}

fn atom_value(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if is_digits(s) {
        return parse_number(s, field, offset);
    }
    parse_name(field, s)
        .map(u32::from)
        .ok_or_else(|| syntax_error(field, offset))
}

fn parse_item(
    item: &str,
    field: Field,
    item_offset: usize,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    let (base, step_text) = if slash_count == 0 {
        (item, None)
    } else if slash_count == 1 {
        let i = item.find('/').unwrap();
        (&item[..i], Some(&item[i + 1..]))
    } else {
        return Err(syntax_error(field, item_offset));
    };

    let is_star = base == "*";
    let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
    let (start_text, end_text) = if is_star {
        if hyphen_count != 0 {
            return Err(syntax_error(field, item_offset));
        }
        (None, None)
    } else if hyphen_count == 0 {
        if !valid_atom_shape(field, base) {
            return Err(syntax_error(field, item_offset));
        }
        (Some(base), None)
    } else if hyphen_count == 1 {
        let i = base.find('-').unwrap();
        let left = &base[..i];
        let right = &base[i + 1..];
        if !valid_atom_shape(field, left) || !valid_atom_shape(field, right) {
            return Err(syntax_error(field, item_offset));
        }
        (Some(left), Some(right))
    } else {
        return Err(syntax_error(field, item_offset));
    };

    if step_text.is_some_and(|s| !is_digits(s)) {
        return Err(syntax_error(field, item_offset));
    }

    let (min, max) = bounds(field);
    let (start, end, start_offset, end_offset) = if is_star {
        (min as u32, max as u32, item_offset, item_offset)
    } else {
        let start_s = start_text.unwrap();
        let start_off = item_offset;
        let start = atom_value(start_s, field, start_off)?;
        if start > max as u32 || start < min as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: start_off,
            });
        }
        if let Some(end_s) = end_text {
            let end_off = item_offset + start_s.len() + 1;
            let end = atom_value(end_s, field, end_off)?;
            if end > max as u32 || end < min as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_off,
                });
            }
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_off,
                });
            }
            (start, end, start_off, end_off)
        } else if step_text.is_some() {
            (start, max as u32, start_off, start_off)
        } else {
            (start, start, start_off, start_off)
        }
    };

    let step = if let Some(step_s) = step_text {
        let off = item_offset + base.len() + 1;
        let n = parse_number(step_s, field, off)?;
        if n == 0 || n > field_width(field) as u32 {
            return Err(ParseError::BadStep { field, offset: off });
        }
        n
    } else {
        1
    };

    let mut v = start;
    while v <= end {
        selected[v as usize] = true;
        match v.checked_add(step) {
            Some(next) if next > v => v = next,
            _ => break,
        }
    }
    let _ = (start_offset, end_offset);
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    for (offset, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = [false; 60];
    let mut start = 0usize;
    for item in text.split(',') {
        parse_item(item, field, start, &mut selected)?;
        start += item.len() + 1;
    }
    let (lo, hi) = bounds(field);
    Ok((lo..=hi).filter(|&n| selected[n as usize]).collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    if input.as_bytes().first() == Some(&b'@') {
        let expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        return parse_cron(expanded);
    }

    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets = Vec::new();
    for (i, part) in parts.iter().enumerate() {
        sets.push(parse_field(part, FIELDS[i])?);
    }
    let day_unrestricted = [parts[2] == "*", parts[4] == "*"];
    Ok(Cron {
        sets: sets.try_into().ok().unwrap(),
        day_unrestricted,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = after;
        loop {
            let day_ok = self.day_matches(date);
            if day_ok && self.has_month(date.month) {
                let times = self.times_on_date(date);
                for (hour, minute) in times {
                    let candidate = CivilDateTime { hour, minute, ..date };
                    if candidate > after {
                        return Ok(Some(candidate));
                    }
                }
            }
            if !advance_day(&mut date) {
                return Ok(None);
            }
            date.hour = 0;
            date.minute = 0;
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = before;
        loop {
            if self.day_matches(date) && self.has_month(date.month) {
                let times = self.times_on_date(date);
                for (hour, minute) in times.into_iter().rev() {
                    let candidate = CivilDateTime { hour, minute, ..date };
                    if candidate < before {
                        return Ok(Some(candidate));
                    }
                }
            }
            if !retreat_day(&mut date) {
                return Ok(None);
            }
            date.hour = 23;
            date.minute = 59;
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(fire) = self.next_fire(cursor).unwrap() {
            if fire >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(fire);
            cursor = fire;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::new();
        for (i, field) in FIELDS.iter().enumerate() {
            let (lo, hi) = bounds(*field);
            let full = self.sets[i].len() == (hi - lo + 1) as usize;
            let rendered = if (i == 2 && self.day_unrestricted[0])
                || (i == 4 && self.day_unrestricted[1])
            {
                "*".to_string()
            } else if full {
                if i == 2 || i == 4 {
                    "*/1".to_string()
                } else {
                    "*".to_string()
                }
            } else {
                self.sets[i]
                    .iter()
                    .map(u8::to_string)
                    .collect::<Vec<_>>()
                    .join(",")
            };
            fields.push(rendered);
        }
        fields.join(" ")
    }

    fn has(&self, field: Field, value: u8) -> bool {
        self.sets[field_index(field)].binary_search(&value).is_ok()
    }

    fn has_month(&self, month: u8) -> bool {
        self.has(Field::Month, month)
    }

    fn day_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match self.day_unrestricted {
            [true, true] => true,
            [false, true] => dom,
            [true, false] => dow,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.has(Field::Minute, at.minute)
            && self.has(Field::Hour, at.hour)
            && self.has_month(at.month)
            && self.day_matches(at)
    }

    fn times_on_date(&self, date: CivilDateTime) -> Vec<(u8, u8)> {
        let mut out = Vec::new();
        for &hour in &self.sets[1] {
            for &minute in &self.sets[0] {
                out.push((hour, minute));
            }
        }
        if date == CivilDateTime {
            year: date.year,
            month: date.month,
            day: date.day,
            hour: 0,
            minute: 0,
        } {
            out.sort_unstable();
        }
        out
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn advance_day(at: &mut CivilDateTime) -> bool {
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return true;
    }
    if at.year == 2399 {
        return false;
    }
    at.month = 1;
    at.year += 1;
    true
}

fn retreat_day(at: &mut CivilDateTime) -> bool {
    if at.day > 1 {
        at.day -= 1;
        return true;
    }
    if at.month > 1 {
        at.month -= 1;
    } else {
        if at.year == 2000 {
            return false;
        }
        at.year -= 1;
        at.month = 12;
    }
    at.day = days_in_month(at.year, at.month);
    true
}
