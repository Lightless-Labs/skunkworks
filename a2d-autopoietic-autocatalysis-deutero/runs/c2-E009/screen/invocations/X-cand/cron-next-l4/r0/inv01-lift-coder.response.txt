use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    unrestricted_day_of_month: bool,
    unrestricted_day_of_week: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Clone, Copy)]
struct Domain {
    min: u8,
    max: u8,
}

impl Domain {
    fn width(self) -> u32 {
        u32::from(self.max - self.min) + 1
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u8> {
    if bytes.iter().any(|b| !b.is_ascii_alphabetic()) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => match upper.as_slice() {
            b"JAN" => Some(1),
            b"FEB" => Some(2),
            b"MAR" => Some(3),
            b"APR" => Some(4),
            b"MAY" => Some(5),
            b"JUN" => Some(6),
            b"JUL" => Some(7),
            b"AUG" => Some(8),
            b"SEP" => Some(9),
            b"OCT" => Some(10),
            b"NOV" => Some(11),
            b"DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_slice() {
            b"SUN" => Some(0),
            b"MON" => Some(1),
            b"TUE" => Some(2),
            b"WED" => Some(3),
            b"THU" => Some(4),
            b"FRI" => Some(5),
            b"SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum Atom {
    Number { start: usize, end: usize },
    Name { value: u8 },
}

fn classify_atom(field: Field, bytes: &[u8], start: usize, end: usize) -> Option<Atom> {
    let atom = &bytes[start..end];
    if atom.iter().all(u8::is_ascii_digit) {
        return Some(Atom::Number { start, end });
    }
    name_value(field, atom).map(|value| Atom::Name { value })
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    let digits = &bytes[start..end];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }
    let mut value = 0u32;
    for &byte in digits {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(u32::from(byte - b'0')))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn atom_numeric(
    atom: Atom,
    bytes: &[u8],
    field: Field,
) -> Result<u8, ParseError> {
    match atom {
        Atom::Name { value } => Ok(value),
        Atom::Number { start, end } => {
            let value = parse_number(bytes, start, end, field)?;
            let d = domain(field);
            if value < u32::from(d.min) || value > u32::from(d.max) {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start,
                });
            }
            Ok(value as u8)
        }
    }
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_count = item.iter().filter(|&&b| b == b'/').count();
    let slash = item.iter().position(|&b| b == b'/');
    let base_end = slash.unwrap_or(item_end);
    let step_start = slash.map(|p| item_start + p + 1);

    if slash_count > 1 || step_start == Some(item_end) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let base_start = item_start;
    let base_bytes = &bytes[base_start..base_end];
    let stars = base_bytes.iter().filter(|&&b| b == b'*').count();
    let hyphens = base_bytes.iter().filter(|&&b| b == b'-').count();

    let structure = if base_bytes == b"*" {
        Some((None, None))
    } else if stars != 0 || hyphens > 1 {
        None
    } else if hyphens == 0 {
        classify_atom(field, bytes, base_start, base_end).map(|a| (Some(a), None))
    } else {
        let hyphen = base_start + base_bytes.iter().position(|&b| b == b'-').unwrap();
        if hyphen == base_start || hyphen + 1 == base_end {
            None
        } else {
            match (
                classify_atom(field, bytes, base_start, hyphen),
                classify_atom(field, bytes, hyphen + 1, base_end),
            ) {
                (Some(left), Some(right)) => Some((Some(left), Some(right))),
                _ => None,
            }
        }
    };

    let (first_atom, second_atom) = structure.ok_or(ParseError::InvalidSyntax {
        field,
        offset: item_start,
    })?;

    let start_value;
    let end_value;
    if let Some(first) = first_atom {
        start_value = atom_numeric(first, bytes, field)?;
        if let Some(second) = second_atom {
            end_value = atom_numeric(second, bytes, field)?;
            if start_value > end_value {
                let offset = match second {
                    Atom::Number { start, .. } => start,
                    Atom::Name { .. } => {
                        let h = base_start
                            + base_bytes.iter().position(|&b| b == b'-').unwrap();
                        h + 1
                    }
                };
                return Err(ParseError::ReversedRange { field, offset });
            }
        } else {
            end_value = if step_start.is_some() {
                domain(field).max
            } else {
                start_value
            };
        }
    } else {
        start_value = domain(field).min;
        end_value = domain(field).max;
    }

    let step = if let Some(start) = step_start {
        let value = parse_number(bytes, start, item_end, field)?;
        if value == 0 || value > domain(field).width() {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
        value as u8
    } else {
        1
    };

    let mut value = start_value;
    loop {
        selected[usize::from(value)] = true;
        if value > end_value.saturating_sub(step) {
            break;
        }
        value += step;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let d = domain(field);
    let mut selected = vec![false; usize::from(d.max) + 1];
    let mut start = 0usize;
    for end in 0..=bytes.len() {
        if end == bytes.len() || bytes[end] == b',' {
            parse_item(bytes, start, end, field, &mut selected)?;
            start = end + 1;
        }
    }
    Ok((d.min..=d.max).filter(|&n| selected[usize::from(n)]).collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let input = if input.as_bytes().first() == Some(&b'@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, field) in fields.into_iter().enumerate() {
        parsed[i] = parse_field(parts[i], field)?;
    }
    Ok(Cron {
        selected: parsed,
        unrestricted_day_of_month: parts[2] == "*",
        unrestricted_day_of_week: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = after;
        loop {
            if self.date_matches(date) {
                let times = self.times_for_date(date, true);
                if let Some(t) = times.into_iter().find(|t| *t > (after.hour, after.minute)) {
                    return Ok(Some(CivilDateTime {
                        hour: t.0,
                        minute: t.1,
                        ..date
                    }));
                }
            }
            match next_date(date) {
                Some(next) => date = next,
                None => return Ok(None),
            }
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = before;
        loop {
            if self.date_matches(date) {
                let times = self.times_for_date(date, false);
                if let Some(t) = times
                    .into_iter()
                    .rev()
                    .find(|t| *t < (before.hour, before.minute))
                {
                    return Ok(Some(CivilDateTime {
                        hour: t.0,
                        minute: t.1,
                        ..date
                    }));
                }
            }
            match previous_date(date) {
                Some(prev) => date = prev,
                None => return Ok(None),
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while cursor < end {
            if self.matches_valid(cursor) {
                if fires.len() == cap {
                    return Ok(FireWindow {
                        fires,
                        truncated: true,
                    });
                }
                fires.push(cursor);
            }
            match next_minute(cursor) {
                Some(next) => cursor = next,
                None => break,
            }
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let fields = [
            canonical_field(&self.selected[0], 0, 59, false),
            canonical_field(&self.selected[1], 0, 23, false),
            canonical_field(
                &self.selected[2],
                1,
                31,
                self.unrestricted_day_of_month,
            ),
            canonical_field(&self.selected[3], 1, 12, false),
            canonical_field(
                &self.selected[4],
                0,
                6,
                self.unrestricted_day_of_week,
            ),
        ];
        fields.join(" ")
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.selected[0].contains(&at.minute)
            && self.selected[1].contains(&at.hour)
            && self.selected[3].contains(&at.month)
            && self.day_rule(at)
    }

    fn date_matches(&self, at: CivilDateTime) -> bool {
        self.selected[3].contains(&at.month) && self.day_rule(at)
    }

    fn day_rule(&self, at: CivilDateTime) -> bool {
        let dom = self.selected[2].contains(&at.day);
        let dow = self.selected[4].contains(&weekday(at));
        match (
            self.unrestricted_day_of_month,
            self.unrestricted_day_of_week,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn times_for_date(&self, date: CivilDateTime, forward: bool) -> Vec<(u8, u8)> {
        let mut result = Vec::new();
        for &hour in &self.selected[1] {
            for &minute in &self.selected[0] {
                result.push((hour, minute));
            }
        }
        if !forward {
            result.reverse();
        }
        let _ = date;
        result
    }
}

fn canonical_field(values: &[u8], min: u8, max: u8, unrestricted: bool) -> String {
    if unrestricted {
        return "*".to_string();
    }
    if values.len() == usize::from(max - min) + 1
        && values.first() == Some(&min)
        && values.last() == Some(&max)
    {
        if min == 1 && max == 31 || min == 0 && max == 6 {
            return "*/1".to_string();
        }
        return "*".to_string();
    }
    values
        .iter()
        .map(u8::to_string)
        .collect::<Vec<_>>()
        .join(",")
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if leap(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += u32::from(days_in_month(at.year, month));
    }
    days += u32::from(at.day - 1);
    ((days + 6) % 7) as u8
}

fn next_date(mut date: CivilDateTime) -> Option<CivilDateTime> {
    if date.day < days_in_month(date.year, date.month) {
        date.day += 1;
    } else if date.month < 12 {
        date.month += 1;
        date.day = 1;
    } else if date.year < 2399 {
        date.year += 1;
        date.month = 1;
        date.day = 1;
    } else {
        return None;
    }
    date.hour = 0;
    date.minute = 0;
    Some(date)
}

fn previous_date(mut date: CivilDateTime) -> Option<CivilDateTime> {
    if date.day > 1 {
        date.day -= 1;
    } else if date.month > 1 {
        date.month -= 1;
        date.day = days_in_month(date.year, date.month);
    } else if date.year > 2000 {
        date.year -= 1;
        date.month = 12;
        date.day = 31;
    } else {
        return None;
    }
    date.hour = 0;
    date.minute = 0;
    Some(date)
}

fn next_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        Some(CivilDateTime {
            minute: at.minute + 1,
            ..at
        })
    } else if at.hour < 23 {
        Some(CivilDateTime {
            hour: at.hour + 1,
            minute: 0,
            ..at
        })
    } else if at.day < days_in_month(at.year, at.month) {
        Some(CivilDateTime {
            day: at.day + 1,
            hour: 0,
            minute: 0,
            ..at
        })
    } else if at.month < 12 {
        Some(CivilDateTime {
            month: at.month + 1,
            day: 1,
            hour: 0,
            minute: 0,
            ..at
        })
    } else if at.year < 2399 {
        Some(CivilDateTime {
            year: at.year + 1,
            month: 1,
            day: 1,
            hour: 0,
            minute: 0,
            ..at
        })
    } else {
        None
    }
}
