#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_unrestricted: [bool; 2],
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u8> {
    let names: &[(&[u8], u8)] = match field {
        Field::Month => &[
            (b"JAN", 1), (b"FEB", 2), (b"MAR", 3), (b"APR", 4),
            (b"MAY", 5), (b"JUN", 6), (b"JUL", 7), (b"AUG", 8),
            (b"SEP", 9), (b"OCT", 10), (b"NOV", 11), (b"DEC", 12),
        ],
        Field::DayOfWeek => &[
            (b"SUN", 0), (b"MON", 1), (b"TUE", 2), (b"WED", 3),
            (b"THU", 4), (b"FRI", 5), (b"SAT", 6),
        ],
        _ => return None,
    };
    names.iter().find_map(|(name, value)| {
        if bytes.len() == name.len()
            && bytes.iter().zip(name.iter()).all(|(a, b)| a.to_ascii_uppercase() == *b)
        {
            Some(*value)
        } else {
            None
        }
    })
}

#[derive(Clone, Copy)]
enum Atom {
    Number(u8),
    Name(u8),
}

fn atom_syntax(field: Field, bytes: &[u8]) -> bool {
    !bytes.is_empty()
        && (bytes.iter().all(u8::is_ascii_digit)
            || name_value(field, bytes).is_some())
}

fn parse_number(field: Field, bytes: &[u8], offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_atom(field: Field, bytes: &[u8], offset: usize) -> Result<u8, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        let value = parse_number(field, bytes, offset)?;
        let (lo, hi) = bounds(field);
        if value < lo as u32 || value > hi as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok(value as u8)
    } else {
        name_value(field, bytes).ok_or_else(|| syntax_error(field, offset))
    }
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (lo, hi) = bounds(field);
    let mut selected = vec![false; hi as usize + 1];
    let mut item_start = 0usize;

    for item in text.split(',') {
        let item_offset = item_start;
        item_start += item.len() + 1;
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: item_offset });
        }
        let ib = item.as_bytes();
        let slashes: Vec<usize> = ib.iter().enumerate().filter_map(|(i, b)| (*b == b'/').then_some(i)).collect();
        if slashes.len() > 1 {
            return Err(syntax_error(field, item_offset));
        }
        let (base, step_part, step_offset) = if let Some(&slash) = slashes.first() {
            (&ib[..slash], Some(&ib[slash + 1..]), item_offset + slash + 1)
        } else {
            (ib, None, 0)
        };

        let hyphens: Vec<usize> = base.iter().enumerate().filter_map(|(i, b)| (*b == b'-').then_some(i)).collect();
        let base_is_star = base == b"*";
        let (first_part, second_part) = if base_is_star {
            if !hyphens.is_empty() {
                return Err(syntax_error(field, item_offset));
            }
            (None, None)
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            let left = &base[..h];
            let right = &base[h + 1..];
            if !atom_syntax(field, left) || !atom_syntax(field, right) {
                return Err(syntax_error(field, item_offset));
            }
            (Some((left, item_offset)), Some((right, item_offset + h + 1)))
        } else if hyphens.is_empty() && atom_syntax(field, base) {
            (Some((base, item_offset)), None)
        } else {
            return Err(syntax_error(field, item_offset));
        };

        if step_part.is_some_and(|s| s.is_empty() || !s.iter().all(u8::is_ascii_digit)) {
            return Err(syntax_error(field, item_offset));
        }

        let start = if let Some((part, offset)) = first_part {
            Some(parse_atom(field, part, offset)?)
        } else {
            None
        };
        let end = if let Some((part, offset)) = second_part {
            Some(parse_atom(field, part, offset)?)
        } else {
            None
        };

        if let (Some(s), Some(e)) = (start, end) {
            if s > e {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_offset + (base.iter().position(|b| *b == b'-').unwrap()) + 1,
                });
            }
        }

        let step = if let Some(step_bytes) = step_part {
            let value = parse_number(field, step_bytes, step_offset)?;
            let width = (hi - lo + 1) as u32;
            if value == 0 || value > width {
                return Err(ParseError::BadStep { field, offset: step_offset });
            }
            value as u8
        } else {
            1
        };

        let range_start = start.unwrap_or(lo);
        let range_end = if let Some(e) = end {
            e
        } else if start.is_some() && step_part.is_some() {
            hi
        } else if start.is_some() {
            range_start
        } else {
            hi
        };

        let mut value = range_start as u16;
        let end_value = range_end as u16;
        while value <= end_value {
            selected[value as usize] = true;
            value += step as u16;
        }
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let pieces: Vec<&str> = expanded.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut selected = Vec::new();
    for (i, piece) in pieces.iter().enumerate() {
        selected.push(parse_field(FIELDS[i], piece)?);
    }
    let selected: [Vec<bool>; 5] = selected.try_into().unwrap();
    let day_unrestricted = [
        pieces[2] == "*",
        pieces[4] == "*",
    ];
    Ok(Cron { selected, day_unrestricted })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, chosen)| (*chosen).then_some(i as u8))
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.selected[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn day_matches(&self, year: u16, month: u8, day: u8) -> bool {
        let dom = self.contains(Field::DayOfMonth, day);
        let dow = self.contains(Field::DayOfWeek, weekday(year, month, day));
        match self.day_unrestricted {
            [true, true] => true,
            [false, true] => dom,
            [true, false] => dow,
            [false, false] => {
                if day == days_in_month(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.day_matches(at.year, at.month, at.day)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23 && current.minute == 59
            {
                return Ok(None);
            }
            current = add_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut current = before;
        loop {
            if current.year == 2000 && current.month == 1 && current.day == 1
                && current.hour == 0 && current.minute == 0
            {
                return Ok(None);
            }
            current = sub_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut current = start;
        while current < end {
            if self.matches_valid(current) {
                if fires.len() == cap {
                    return Ok(FireWindow { fires, truncated: true });
                }
                fires.push(current);
            }
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23 && current.minute == 59
            {
                break;
            }
            current = add_minute(current);
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::new();
        for i in 0..5 {
            let field = FIELDS[i];
            let values = self.allowed(field);
            let (lo, hi) = bounds(field);
            let full = values.len() == (hi - lo + 1) as usize
                && values.first() == Some(&lo)
                && values.last() == Some(&hi);
            if i == 2 && self.day_unrestricted[0] {
                fields.push("*".to_string());
            } else if i == 4 && self.day_unrestricted[1] {
                fields.push("*".to_string());
            } else if full && matches!(field, Field::Minute | Field::Hour | Field::Month) {
                fields.push("*".to_string());
            } else if full && matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                fields.push("*/1".to_string());
            } else {
                fields.push(values.iter().map(u8::to_string).collect::<Vec<_>>().join(","));
            }
        }
        fields.join(" ")
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn add_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
        } else {
            at.hour = 0;
            if at.day < days_in_month(at.year, at.month) {
                at.day += 1;
            } else {
                at.day = 1;
                if at.month < 12 {
                    at.month += 1;
                } else {
                    at.month = 1;
                    at.year += 1;
                }
            }
        }
    }
    at
}

fn sub_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute > 0 {
        at.minute -= 1;
    } else {
        at.minute = 59;
        if at.hour > 0 {
            at.hour -= 1;
        } else {
            at.hour = 23;
            if at.day > 1 {
                at.day -= 1;
            } else {
                if at.month > 1 {
                    at.month -= 1;
                } else {
                    at.month = 12;
                    at.year -= 1;
                }
                at.day = days_in_month(at.year, at.month);
            }
        }
    }
    at
}
