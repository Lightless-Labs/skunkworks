pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Validate resource shapes and demands task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Kahn traversal with a min-heap (Reverse) for deterministic smallest-ID order.
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children: Vec<Vec<usize>> = (0..n).map(|_| Vec::new()).collect();
    for (task_id, task) in tasks.iter().enumerate() {
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut available = std::collections::BinaryHeap::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut order = Vec::new();
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for id in 0..n {
            if indegree[id] != 0 {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Bottom levels, computed in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let mut longest_successor = 0u64;
        for &child in &children[id] {
            longest_successor = longest_successor.max(tail[child]);
        }
        tail[id] = tasks[id]
            .duration
            .checked_add(longest_successor)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();

    let mut runs: Vec<Option<Run>> = (0..n).map(|_| None).collect();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut used = vec![0u64; resource_count];

    #[derive(Clone)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
        available_at: u64,
    }

    let mut running: Vec<Running> = Vec::new();
    let mut worker_available_at = vec![0u64; worker_count];
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut trace = format!("workers={worker_count};resources={resource_count}\n");

    fn append_resources(trace: &mut String, used: &[u64]) {
        if used.is_empty() {
            trace.push('-');
        } else {
            for (index, value) in used.iter().enumerate() {
                if index != 0 {
                    trace.push(',');
                }
                trace.push_str(&value.to_string());
            }
        }
    }

    loop {
        // Complete all tasks ending now, in ascending task ID order.
        running.sort_by_key(|entry| entry.task);
        let mut still_running = Vec::new();
        for entry in running.drain(..) {
            if entry.finish == now {
                completed[entry.task] = true;
                for (resource, &need) in tasks[entry.task].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                for &child in &children[entry.task] {
                    remaining_deps[child] -= 1;
                }

                trace.push_str(&format!(
                    "{now}|end|{}|{}|",
                    entry.task, entry.worker
                ));
                append_resources(&mut trace, &used);
                trace.push('\n');
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;

        // Start one task at a time, re-evaluating eligibility and resource fit each time.
        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || remaining_deps[id] != 0
                {
                    continue;
                }

                let fits = tasks[id]
                    .needs
                    .iter()
                    .zip(used.iter())
                    .zip(capacity.iter())
                    .all(|((&need, &in_use), &cap)| {
                        u64::from(need) <= u64::from(cap) - in_use
                    });
                if !fits {
                    continue;
                }

                let is_better = match best {
                    None => true,
                    Some(current) => {
                        tail[id] > tail[current]
                            || (tail[id] == tail[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if is_better {
                    best = Some(id);
                }
            }

            let Some(id) = best else {
                break;
            };

            let preferred = id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&w| worker_available_at[w] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[id].duration)
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Turnaround,
                })?;

            started[id] = true;
            worker_available_at[worker] = available_at;
            for (resource, &need) in tasks[id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            trace.push_str(&format!("{now}|start|{id}|{worker}|"));
            append_resources(&mut trace, &used);
            trace.push('\n');

            if finish == now {
                // Immediate completion occurs before selecting another start.
                completed[id] = true;
                for (resource, &need) in tasks[id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                for &child in &children[id] {
                    remaining_deps[child] -= 1;
                }

                trace.push_str(&format!("{now}|end|{id}|{worker}|"));
                append_resources(&mut trace, &used);
                trace.push('\n');
            } else {
                makespan = makespan.max(finish);
                running.push(Running {
                    task: id,
                    worker,
                    finish,
                    available_at,
                });
            }
        }

        if completed.iter().all(|&done| done) {
            break;
        }

        // Advance to the earliest event that can change eligibility or worker availability.
        let mut next: Option<u64> = None;
        for entry in &running {
            if entry.finish > now {
                next = Some(next.map_or(entry.finish, |value| value.min(entry.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |value| value.min(tasks[id].release)));
            }
        }
        for worker in 0..worker_count {
            if worker_available_at[worker] > now {
                next = Some(
                    next.map_or(worker_available_at[worker], |value| {
                        value.min(worker_available_at[worker])
                    }),
                );
            }
        }

        if let Some(next_time) = next {
            now = next_time;
        } else {
            // A DAG always has a ready task; reaching this branch would require an internal bug.
            break;
        }
    }

    // Makespan is based on actual finishes, including zero-duration tasks.
    for run in runs.iter().flatten() {
        makespan = makespan.max(run.finish);
    }
    trace.push_str(&format!("makespan={makespan}\n"));

    Ok(Schedule {
        runs: runs
            .into_iter()
            .map(|run| run.expect("acyclic task must have been scheduled"))
            .collect(),
        makespan,
        trace,
    })
}

                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut children: Vec<Vec<usize>> = (0..n).map(|_| Vec::new()).collect();
    let mut indegree = Vec::new();
    for task in tasks {
        indegree.push(task.deps.len());
    }
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = Vec::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(id);
        }
    }
    let mut order = Vec::new();
    while !available.is_empty() {
        let mut smallest_index = 0;
        for i in 1..available.len() {
            if available[i] < available[smallest_index] {
                smallest_index = i;
            }
        }
        let id = available.swap_remove(smallest_index);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for id in 0..n {
            if indegree[id] != 0 {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tails = Vec::new();
    tails.resize(n, 0_u64);
    for &id in order.iter().rev() {
        let mut longest_child = 0_u64;
        for &child in &children[id] {
            if tails[child] > longest_child {
                longest_child = tails[child];
            }
        }
        tails[id] = tasks[id]
            .duration
            .checked_add(longest_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        finish: u64,
        worker: usize,
    }

    let mut runs: Vec<Option<Run>> = (0..n).map(|_| None).collect();
    let mut started = Vec::new();
    started.resize(n, false);
    let mut completed = Vec::new();
    completed.resize(n, false);
    let mut worker_ready = Vec::new();
    worker_ready.resize(worker_count, 0_u64);
    let mut running: Vec<Running> = Vec::new();
    let mut used: Vec<u64> = capacity.iter().map(|_| 0_u64).collect();
    let mut trace = String::new();
    use std::fmt::Write;
    let _ = writeln!(
        trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    );

    let mut now = 0_u64;
    let mut makespan = 0_u64;
    let mut completed_count = 0_usize;

    while completed_count < n {
        // Complete every task finishing now, in ascending task ID order.
        let mut ending: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == now)
            .collect();
        ending.sort_by_key(|entry| entry.task);

        if !ending.is_empty() {
            running.retain(|entry| entry.finish != now);
            for entry in ending {
                let id = entry.task;
                completed[id] = true;
                completed_count += 1;
                for r in 0..capacity.len() {
                    used[r] -= u64::from(tasks[id].needs[r]);
                }
                let _ = write!(trace, "{}|end|{}|{}|", now, id, entry.worker);
                write_used(&mut trace, &used);
                trace.push('\n');
                if now > makespan {
                    makespan = now;
                }
            }
        }

        // Start tasks one at a time, choosing the best currently fitting task.
        let mut started_any = false;
        loop {
            let mut selected: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if !tasks[id].deps.iter().all(|&dep| completed[dep]) {
                    continue;
                }
                let fits = (0..capacity.len()).all(|r| {
                    u64::from(tasks[id].needs[r]) <= u64::from(capacity[r]) - used[r]
                });
                if !fits {
                    continue;
                }
                match selected {
                    None => selected = Some(id),
                    Some(best) => {
                        if tails[id] > tails[best]
                            || (tails[id] == tails[best]
                                && (tasks[id].priority > tasks[best].priority
                                    || (tasks[id].priority == tasks[best].priority && id < best)))
                        {
                            selected = Some(id);
                        }
                    }
                }
            }

            let Some(id) = selected else {
                break;
            };

            // Choose the first available worker in the task-specific cyclic order.
            let mut chosen_worker = None;
            for offset in 0..worker_count {
                let worker = (id % worker_count + offset) % worker_count;
                if worker_ready[worker] <= now {
                    chosen_worker = Some(worker);
                    break;
                }
            }
            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = now.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let ready = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Turnaround,
                })?;

            started[id] = true;
            worker_ready[worker] = ready;
            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            for r in 0..capacity.len() {
                used[r] += u64::from(tasks[id].needs[r]);
            }
            let _ = write!(trace, "{}|start|{}|{}|", now, id, worker);
            write_used(&mut trace, &used);
            trace.push('\n');

            if tasks[id].duration == 0 {
                completed[id] = true;
                completed_count += 1;
                for r in 0..capacity.len() {
                    used[r] -= u64::from(tasks[id].needs[r]);
                }
                let _ = write!(trace, "{}|end|{}|{}|", now, id, worker);
                write_used(&mut trace, &used);
                trace.push('\n');
                if now > makespan {
                    makespan = now;
                }
            } else {
                running.push(Running {
                    task: id,
                    finish,
                    worker,
                });
            }

            started_any = true;
        }

        if completed_count == n {
            break;
        }

        // Find the next actual event. A task completion, release, or cooldown
        // expiry can make progress possible.
        let mut next_event: Option<u64> = None;
        for entry in &running {
            if entry.finish > now {
                next_event = Some(match next_event {
                    None => entry.finish,
                    Some(value) => value.min(entry.finish),
                });
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next_event = Some(match next_event {
                    None => tasks[id].release,
                    Some(value) => value.min(tasks[id].release),
                });
            }
        }
        for &ready in &worker_ready {
            if ready > now {
                next_event = Some(match next_event {
                    None => ready,
                    Some(value) => value.min(ready),
                });
            }
        }

        if let Some(next) = next_event {
            now = next;
        } else if !started_any {
            // The graph is acyclic, so this is only a defensive escape path.
            break;
        }
    }

    let mut ordered_runs = Vec::new();
    for run in runs {
        if let Some(run) = run {
            ordered_runs.push(run);
        }
    }

    let _ = writeln!(trace, "makespan={}", makespan);
    Ok(Schedule {
        runs: ordered_runs,
        makespan,
        trace,
    })
}

fn write_used(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            use std::fmt::Write;
            let _ = write!(trace, "{}", value);
        }
    }
}

pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists in input dependency order.
    let mut indegree = Vec::new();
    let mut children: Vec<Vec<usize>> = Vec::new();
    for task in tasks {
        indegree.push(task.deps.len());
        children.push(Vec::new());
    }
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::new();
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for task_id in 0..n {
            if indegree[task_id] != 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let mut longest_successor = 0u64;
        for &child in &children[task_id] {
            longest_successor = longest_successor.max(tail[child]);
        }
        tail[task_id] = tasks[task_id]
            .duration
            .checked_add(longest_successor)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = (0..n).map(|_| None).collect();
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut worker_available = vec![0u64; worker_count];
    let mut running: Vec<Running> = Vec::new();
    let mut used: Vec<u64> = capacity.iter().map(|_| 0u64).collect();
    let mut trace = String::new();
    use std::fmt::Write as _;
    writeln!(
        &mut trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    )
    .unwrap();

    let mut now = 0u64;
    let mut done_count = 0usize;
    let mut makespan = 0u64;

    while done_count < n {
        // Complete all tasks finishing now in ascending task ID.
        running.sort_by_key(|r| r.task);
        let mut i = 0;
        while i < running.len() {
            if running[i].finish == now {
                let finished = running.remove(i);
                let task_id = finished.task;
                completed[task_id] = true;
                done_count += 1;
                makespan = makespan.max(now);
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                write!(
                    &mut trace,
                    "{}|end|{}|{}|",
                    now, task_id, finished.worker
                )
                .unwrap();
                write_resources(&mut trace, &used);
                trace.push('\n');
            } else {
                i += 1;
            }
        }

        loop {
            let mut chosen_task: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > now {
                    continue;
                }
                if !tasks[task_id].deps.iter().all(|&p| completed[p]) {
                    continue;
                }
                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r]);
                if !fits {
                    continue;
                }

                let better = match chosen_task {
                    None => true,
                    Some(current) => {
                        tail[task_id] > tail[current]
                            || (tail[task_id] == tail[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    chosen_task = Some(task_id);
                }
            }

            let Some(task_id) = chosen_task else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now)
                .expect("admission requires an available worker");

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            write!(&mut trace, "{}|start|{}|{}|", now, task_id, worker).unwrap();
            write_resources(&mut trace, &used);
            trace.push('\n');

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                done_count += 1;
                makespan = makespan.max(now);
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                write!(&mut trace, "{}|end|{}|{}|", now, task_id, worker).unwrap();
                write_resources(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if done_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for task in &running {
            if task.finish > now {
                next_time = Some(next_time.map_or(task.finish, |t| t.min(task.finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                next_time = Some(next_time.map_or(tasks[task_id].release, |t| {
                    t.min(tasks[task_id].release)
                }));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |t| t.min(available_at)));
            }
        }

        // If tasks remain, at least one event can eventually enable progress.
        now = next_time.expect("unfinished tasks must have a future event");
    }

    writeln!(&mut trace, "makespan={}", makespan).unwrap();

    Ok(Schedule {
        runs: runs
            .into_iter()
            .map(|run| run.expect("every task must have a run"))
            .collect(),
        makespan,
        trace,
    })
}

    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, count) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            use std::fmt::Write as _;
            write!(trace, "{}", count).unwrap();
        }
    }
}
