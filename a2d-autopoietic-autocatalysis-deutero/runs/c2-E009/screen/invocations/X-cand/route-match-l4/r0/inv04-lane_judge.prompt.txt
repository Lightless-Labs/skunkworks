fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || is_unreserved(decoded) {
                return Err(i);
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(i);
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = input.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut start = 1;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(start);
        }

        let raw = &input[start..stop];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_string()) {
                return Err(start);
            }
            if marker == b'*' && stop != end {
                return Err(start);
            }
            if marker == b':' {
                segments.push(Segment::Param(name.to_string()));
            } else {
                segments.push(Segment::Wildcard(name.to_string()));
            }
        } else {
            match decode_segment(raw) {
                Ok(text) => segments.push(Segment::Static(text)),
                Err(offset) => return Err(start + offset),
            }
        }

        if stop == end {
            break;
        }
        start = stop + 1;
        if start == end {
            break;
        }
    }

    Ok((segments, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(s), Segment::Static(t)) => s == t,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn rank(route: &Route) -> Vec<u8> {
    let mut out = Vec::new();
    for seg in &route.segments {
        match seg {
            Segment::Static(_) => out.push(2),
            Segment::Param(_) => out.push(1),
            Segment::Wildcard(_) => {
                out.push(0);
                return out;
            }
        }
    }
    out.push(3);
    out
}

fn request_segments(path: &str) -> Result<(Vec<String>, bool), usize> {
    let (segments, trailing) = parse_path(path, false)?;
    let decoded = segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(v) => v,
            _ => unreachable!(),
        })
        .collect();
    Ok((decoded, trailing))
}

fn route_matches(route: &Route, request: &[String], trailing: bool, policy: SlashPolicy) -> Option<Vec<(String, String)>> {
    if policy == SlashPolicy::Strict && route.trailing != trailing {
        return None;
    }
    let wildcard_index = route.segments.iter().position(|s| matches!(s, Segment::Wildcard(_)));
    let fixed = wildcard_index.unwrap_or(route.segments.len());
    if request.len() < fixed || (wildcard_index.is_none() && request.len() != fixed) {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed {
        match &route.segments[i] {
            Segment::Static(s) if s != &request[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), request[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }
    if let Some(index) = wildcard_index {
        let Segment::Wildcard(name) = &route.segments[index] else {
            unreachable!()
        };
        let value = request[index..].join("/");
        params.push((name.clone(), value));
    }
    Some(params)
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }
        let candidate = Route {
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
        };
        if let Some((id, _)) = self
            .routes
            .iter()
            .enumerate()
            .find(|(_, route)| shape_conflicts(route, &candidate, self.policy))
        {
            return Err(Error::Conflict { with: id });
        }
        let id = self.routes.len();
        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request, trailing) =
            request_segments(head).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates = Vec::new();
        for (id, route) in self.routes.iter().enumerate() {
            if let Some(params) = route_matches(route, &request, trailing, self.policy) {
                candidates.push((id, route, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<_> = if method == "HEAD" {
            if candidates.iter().any(|(_, r, _)| r.method == "HEAD") {
                candidates
                    .into_iter()
                    .filter(|(_, r, _)| r.method == "HEAD")
                    .collect()
            } else {
                candidates
                    .into_iter()
                    .filter(|(_, r, _)| r.method == "GET")
                    .collect()
            }
        } else {
            candidates
                .into_iter()
                .filter(|(_, r, _)| r.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = Vec::new();
            for (_, route, _) in &eligible {
                allowed.push(route.method.clone());
            }
            // Candidates were consumed above only when filtering; reconstruct allowed from matching paths.
            for route in &self.routes {
                if route_matches(route, &request, trailing, self.policy).is_some() {
                    if !allowed.contains(&route.method) {
                        allowed.push(route.method.clone());
                    }
                }
            }
            if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut best_index = 0;
        for i in 1..eligible.len() {
            if rank(eligible[i].1) > rank(eligible[best_index].1) {
                best_index = i;
            }
        }
        let (id, route, params) = eligible.into_iter().nth(best_index).unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect && route.trailing != trailing {
            let mut target = head.to_string();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') && target != "/" {
                target.pop();
            }
            if target == "/" {
                None
            } else {
                Some(target)
            }
        } else {
            None
        };

        Ok(Match {
            route: id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(route) = self.routes.iter().find(|route| route.name == name) else {
            return Err(Error::UnknownName);
        };

        let mut supplied = std::collections::HashMap::<&str, &str>::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extra: Option<&str> = None;
        for key in supplied.keys() {
            if !required.contains(key) && extra.map_or(true, |e| *key < e) {
                extra = Some(key);
            }
        }
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: key.to_string(),
            });
        }

        for segment in &route.segments {
            let name = match segment {
                Segment::Param(n) | Segment::Wildcard(n) => n,
                Segment::Static(_) => continue,
            };
            let value = supplied[name.as_str()];
            let valid_ascii = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
            let valid = match segment {
                Segment::Param(_) => valid_ascii && !value.is_empty() && !value.contains('/'),
                Segment::Wildcard(_) => {
                    valid_ascii
                        && (value.is_empty()
                            || value
                                .split('/')
                                .all(|part| !part.is_empty()))
                }
                Segment::Static(_) => true,
            };
            if !valid {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut out = String::from("/");
        let mut emitted = false;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted {
                        out.push('/');
                    }
                    out.push_str(&encode_segment(value));
                    emitted = true;
                }
                Segment::Param(name) => {
                    if emitted {
                        out.push('/');
                    }
                    out.push_str(&encode_segment(supplied[name.as_str()]));
                    emitted = true;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        if emitted {
                            out.push('/');
                        }
                        let mut first = true;
                        for part in value.split('/') {
                            if !first {
                                out.push('/');
                            }
                            out.push_str(&encode_segment(part));
                            first = false;
                        }
                        emitted = true;
                    }
                }
            }
        }
        if !emitted {
            out = "/".to_string();
        } else if route.trailing {
            out.push('/');
        }
        Ok(out)
    }

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(input: &str, base: usize, pattern: bool) -> Result<String, usize> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if b.is_ascii() && unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            let _ = pattern;
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn split_path<'a>(input: &'a str, pattern: bool) -> Result<(Vec<(&'a str, usize)>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            parts.push((&input[start..i], start));
            start = i + 1;
        }
        i += 1;
    }
    if pattern {
        Ok((parts, trailing))
    } else {
        Ok((parts, trailing))
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let (parts, trailing) = split_path(pattern, true)
        .map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();

    for (index, (part, at)) in parts.iter().enumerate() {
        if part.starts_with(':') || part.starts_with('*') {
            let wildcard = part.as_bytes()[0] == b'*';
            let name = &part[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: *at });
            }
            if names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: *at });
            }
            if wildcard && index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *at });
            }
            names.push(name.to_string());
            if wildcard {
                segments.push(Segment::Wildcard(name.to_string()));
            } else {
                segments.push(Segment::Param(name.to_string()));
            }
        } else {
            let decoded = decode_segment(part, *at, true)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, trailing))
}

fn parse_request(path: &str) -> Result<(Vec<String>, bool, String), Error> {
    let cut = path
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(path.len());
    let head = &path[..cut];
    let (parts, trailing) =
        split_path(head, false).map_err(|at| Error::InvalidPath { at })?;
    let mut decoded = Vec::new();
    for (part, at) in parts {
        decoded.push(
            decode_segment(part, at, false).map_err(|at| Error::InvalidPath { at })?,
        );
    }
    Ok((decoded, trailing, head.to_string()))
}

fn same_shape(a: &Route, b_segments: &[Segment], b_trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != b_segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b_trailing {
        return false;
    }
    a.segments.iter().zip(b_segments).all(|(x, y)| match (x, y) {
        (Segment::Static(ax), Segment::Static(by)) => ax == by,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn route_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    let mut has_wildcard = false;
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                has_wildcard = true;
                break;
            }
        }
    }
    if !has_wildcard {
        rank.push(3);
    }
    rank
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut ri = 0;
    for segment in &route.segments {
        match segment {
            Segment::Static(expected) => {
                if request.get(ri)? != expected {
                    return None;
                }
                ri += 1;
            }
            Segment::Param(name) => {
                let value = request.get(ri)?;
                params.push((name.clone(), value.clone()));
                ri += 1;
            }
            Segment::Wildcard(name) => {
                let value = request[ri..].join("/");
                params.push((name.clone(), value));
                ri = request.len();
                break;
            }
        }
    }
    if ri != request.len() {
        return None;
    }
    Some(params)
}

fn encode_component(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (request, request_trailing, original_head) = parse_request(path)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>, Vec<u8>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some(params) = match_route(route, &request) {
                candidates.push((route, params, route_rank(&route.segments)));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let has_head = method == "HEAD"
            && candidates.iter().any(|(r, _, _)| r.method == "HEAD");
        let selected_method = if method == "HEAD" && !has_head {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _, _)| route.method == selected_method)
            .collect();
        if eligible.is_empty() {
            let mut allowed = Vec::new();
            for (route, _, _) in &candidates {
                if !allowed.iter().any(|m: &String| m == &route.method) {
                    allowed.push(route.method.clone());
                }
                if route.method == "GET" && !allowed.iter().any(|m| m == "HEAD") {
                    allowed.push("HEAD".to_string());
                }
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|a, b| b.2.cmp(&a.2).then_with(|| a.0.id.cmp(&b.0.id)));
        let (route, params, _) = eligible[0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !route.segments.is_empty()
            && route.trailing != request_trailing
        {
            let mut target = original_head.clone();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|route| route.name == name).ok_or(Error::UnknownName)?;

        let mut supplied: Vec<(&str, &str)> = Vec::new();
        for &(key, value) in params {
            if supplied.iter().any(|(existing, _)| *existing == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        let mut required: Vec<&str> = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(key) | Segment::Wildcard(key) => required.push(key),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.iter().any(|(given, _)| given == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extra: Option<&str> = None;
        for (key, _) in &supplied {
            if !required.iter().any(|known| known == key)
                && extra.map_or(true, |current| key.as_bytes() < current.as_bytes())
            {
                extra = Some(key);
            }
        }
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: key.to_string(),
            });
        }

        for key in &required {
            let value = supplied.iter().find(|(given, _)| given == key).unwrap().1;
            let wildcard = route.segments.iter().any(
                |segment| matches!(segment, Segment::Wildcard(name) if name == key),
            );
            let bytes = value.as_bytes();
            let ascii_printable = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid = if wildcard {
                ascii_printable
                    && (value.is_empty()
                        || (!value.starts_with('/')
                            && !value.ends_with('/')
                            && !value.contains("//")))
            } else {
                ascii_printable && !value.is_empty() && !value.contains('/')
            };
            if !valid {
                return Err(Error::InvalidValue {
                    name: (*key).to_string(),
                });
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if let Segment::Wildcard(name) = segment {
                let value = supplied.iter().find(|(key, _)| key == name).unwrap().1;
                if !value.is_empty() {
                    for component in value.split('/') {
                        if !first {
                            output.push('/');
                        }
                        output.push_str(&encode_component(component));
                        first = false;
                    }
                }
            } else {
                if !first {
                    output.push('/');
                }
                match segment {
                    Segment::Static(value) => output.push_str(&encode_component(value)),
                    Segment::Param(name) => {
                        let value = supplied.iter().find(|(key, _)| key == name).unwrap().1;
                        output.push_str(&encode_component(value));
                    }
                    Segment::Wildcard(_) => unreachable!(),
                }
                first = false;
            }
        }

        if route.trailing && !output.ends_with('/') && output != "/" {
            output.push('/');
        }
        Ok(output)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if let Some(params) = match_segments(&route.segments, &request_segments) {
                if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                    continue;
                }
                candidates.push((route, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_exists = method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD");
        let effective = if method == "HEAD" && !head_exists {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(route, _)| route.method == effective)
            .collect();

        if eligible.is_empty() {
            let mut allowed = Vec::new();
            for (route, _) in &eligible {
                if !allowed.contains(&route.method) {
                    allowed.push(route.method.clone());
                }
            }
            // Rebuild allowed from all path candidates; eligible is empty here.
            for route in &self.routes {
                if match_segments(&route.segments, &request_segments).is_some()
                    && (self.policy != SlashPolicy::Strict
                        || route.trailing == request_trailing)
                    && !allowed.contains(&route.method)
                {
                    allowed.push(route.method.clone());
                }
            }
            if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank
                .cmp(&a.rank)
                .then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !request_segments.is_empty()

            let mut target = head.to_string();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied: Vec<(&str, &str)> = Vec::new();
        let mut seen = HashSet::new();
        for &(key, value) in params {
            if !seen.insert(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.iter().any(|(provided, _)| provided == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .iter()
            .map(|(key, _)| *key)
            .filter(|key| !required.contains(key))
            .collect();
        extras.sort_unstable();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = lookup_param(&supplied, name).unwrap_or("");
                    if !valid_param_value(value) {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = lookup_param(&supplied, name).unwrap_or("");
                    if !valid_wildcard_value(value) {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut out = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            let parts: Vec<String> = match segment {
                Segment::Static(text) => vec![encode_segment(text)],
                Segment::Param(name) => {
                    vec![encode_segment(lookup_param(&supplied, name).unwrap_or(""))]
                }
                Segment::Wildcard(name) => {
                    let value = lookup_param(&supplied, name).unwrap_or("");
                    if value.is_empty() {
                        Vec::new()
                    } else {
                        value.split('/').map(encode_segment).collect()
                    }
                }
            };
            for part in parts {
                if !first {
                    out.push('/');
                }
                out.push_str(&part);
                first = false;
            }
        }
        if route.trailing && !out.ends_with('/') && out != "/" {
            out.push('/');
        }
        Ok(out)
    }

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }
    let bytes = pattern.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1;
    while start < end {
        let next = pattern[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPattern { at: start });
        }
        let part = &pattern[start..next];
        if part.starts_with(':') || part.starts_with('*') {
            let name = &part[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if part.starts_with('*') && next != end {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if part.starts_with(':') {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            });
        } else {
            segments.push(Segment::Static(decode_segment(part, start, true)?));
        }
        if next == end {
            break;
        }
        start = next + 1;
    }
    Ok((segments, trailing),)
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }
    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;
    while start < end {
        let next = path[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(&path[start..next], start, false)?);
        if next == end {
            break;
        }
        start = next + 1;
    }
    Ok((segments, trailing))
}

fn decode_segment(segment: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                });
            }
            let decoded = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                });
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: base + i }
            } else {
                Error::InvalidPath { at: base + i }
            });
        }
    }
    Ok(String::from_utf8(out).unwrap_or_default())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(left, right)| match (left, right) {
            (Segment::Static(a), Segment::Static(b)) => a == b,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn match_segments(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard = matches!(pattern.last(), Some(Segment::Wildcard(_)));
    let fixed = if wildcard {
        pattern.len() - 1
    } else {
        pattern.len()
    };
    if (!wildcard && request.len() != fixed) || (wildcard && request.len() < fixed) {
        return None;
    }
    let mut params = Vec::new();
    for (segment, value) in pattern[..fixed].iter().zip(request.iter()) {
        match segment {
            Segment::Static(text) if text != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }
    if let Some(Segment::Wildcard(name)) = pattern.last() {
        params.push((name.clone(), request[fixed..].join("/")));
    }
    Some(params)
}

fn lookup_param<'a>(params: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    params
        .iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn valid_param_value(value: &str) -> bool {
    !value.is_empty()
        && value.bytes().all(|b| (0x20..=0x7e).contains(&b))
        && !value.contains('/')
}

fn valid_wildcard_value(value: &str) -> bool {
    value.is_empty()
        || (value.bytes().all(|b| (0x20..=0x7e).contains(&b))
            && value.split('/').all(|part| !part.is_empty()))
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing_slash) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let candidate = Route {
            id: self.routes.len(),
            name: name.to_string(),
            method: method.to_string(),
            rank: make_rank(&segments),
            segments,
            trailing_slash,
        };

        for route in &self.routes {
            if route.method == candidate.method
                && (self.policy != SlashPolicy::Strict
                    || route.trailing_slash == candidate.trailing_slash)
                && same_shape(route, &candidate)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = candidate.id;
        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, ch)| if ch == '?' || ch == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path(head, ParseKind::Path)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if let Some(params) = match_route(route, &parsed) {
                if self.policy == SlashPolicy::Strict
                    && route.trailing_slash != parsed.trailing_slash
                {
                    continue;
                }
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        let mut allowed: Vec<String> = Vec::new();

        for (route, params) in candidates {
            if !allowed.iter().any(|m| m == &route.method) {
                allowed.push(route.method.clone());
            }
            if route.method != eligible_method {
                continue;
            }
            match &best {
                None => best = Some((route, params)),
                Some((best_route, _)) => {
                    if rank_cmp(&route.rank, &best_route.rank).is_gt()
                        || (route.rank == best_route.rank && route.id < best_route.id)
                    {
                        best = Some((route, params));
                    }
                }
            }
        }

        if let Some((route, params)) = best {
            let redirect = if self.policy == SlashPolicy::Redirect
                && route.trailing_slash != parsed.trailing_slash
                && !parsed.segments.is_empty()
            {
                let mut target = head.to_string();
                if route.trailing_slash {
                    target.push('/');
                } else if target.ends_with('/') {
                    target.pop();
                }
                Some(target)
            } else {
                None
            };
            return Ok(Match {
                route: route.id,
                params,
                redirect,
            });
        }

        if method == "HEAD" && eligible_method == "GET" && !allowed.iter().any(|m| m == "HEAD") {
            allowed.push("HEAD".to_string());
        }
        allowed.sort();
        allowed.dedup();
        Err(Error::MethodNotAllowed { allowed })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = match self.routes.iter().find(|route| route.name == name) {
            Some(route) => route,
            None => return Err(Error::UnknownName),
        };

        let mut supplied: Vec<(&str, &str)> = Vec::new();
        for &(key, value) in params {
            if supplied.iter().any(|(seen, _)| *seen == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        for segment in &route.segments {
            if let Segment::Param(key) | Segment::Wildcard(key) = segment {
                if !supplied.iter().any(|(supplied_key, _)| *supplied_key == key) {
                    return Err(Error::MissingParam { name: key.clone() });
                }
            }
        }

        let mut extra: Option<&str> = None;
        for (key, _) in &supplied {
            let is_capture = route.segments.iter().any(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => name == key,
                Segment::Static(_) => false,
            });
            if !is_capture && extra.map_or(true, |current| key.as_bytes() < current.as_bytes()) {
                extra = Some(key);
            }
        }
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: key.to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Static(_) => {}
                Segment::Param(key) => {
                    let value = supplied_value(&supplied, key).unwrap_or("");
                    if !valid_build_param(value) {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
                Segment::Wildcard(key) => {
                    let value = supplied_value(&supplied, key).unwrap_or("");
                    if !valid_build_wildcard(value) {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if let Segment::Wildcard(key) = segment {
                let value = supplied_value(&supplied, key).unwrap_or("");
                if !value.is_empty() {
                    for component in value.split('/') {
                        if !first {
                            output.push('/');
                        }
                        append_encoded(&mut output, component.as_bytes());
                        first = false;
                    }
                }
                continue;
            }

            if !first {
                output.push('/');
            }
            match segment {
                Segment::Static(value) => append_encoded(&mut output, value.as_bytes()),
                Segment::Param(key) => {
                    let value = supplied_value(&supplied, key).unwrap_or("");
                    append_encoded(&mut output, value.as_bytes());
                }
                Segment::Wildcard(_) => {}
            }
            first = false;
        }

        if route.trailing_slash && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let parsed = parse_path(pattern, ParseKind::Pattern)?;
    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();

    for (idx, raw) in raw_segments(pattern, parsed.trailing_slash).into_iter().enumerate() {
        let start = raw.start;
        let text = raw.text;
        if let Some(name) = text.strip_prefix(':') {
            if !valid_capture_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = text.strip_prefix('*') {
            if idx + 1 != raw_segments(pattern, parsed.trailing_slash).len() {
                return Err(Error::InvalidPattern { at: start });
            }
            if !valid_capture_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            let decoded = decode_segment(text, start, ParseKind::Pattern)?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, parsed.trailing_slash))
}

#[derive(Clone, Copy)]
struct RawSegment<'a> {
    text: &'a str,
    start: usize,
}

fn raw_segments(path: &str, trailing_slash: bool) -> Vec<RawSegment<'_>> {
    if path == "/" {
        return Vec::new();
    }
    let end = if trailing_slash {
        path.len().saturating_sub(1)
    } else {
        path.len()
    };
    let mut result = Vec::new();
    let mut start = 1usize;
    let bytes = path.as_bytes();
    let mut i = 1usize;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            result.push(RawSegment {
                text: &path[start..i],
                start,
            });
            start = i + 1;
        }
        i += 1;
    }
    result
}

fn parse_path(path: &str, kind: ParseKind) -> Result<ParsedPath, Error> {
    let invalid = |at| match kind {
        ParseKind::Pattern => Error::InvalidPattern { at },
        ParseKind::Path => Error::InvalidPath { at },
    };

    if !path.starts_with('/') {
        return Err(invalid(0));
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let trailing_slash = path.ends_with('/');
    let raw = raw_segments(path, trailing_slash);
    let mut segments = Vec::new();

    for segment in raw {
        if segment.text.is_empty() {
            return Err(invalid(segment.start));
        }
        if kind == ParseKind::Path {
            segments.push(decode_segment(segment.text, segment.start, kind)?);
        } else if !segment.text.starts_with(':') && !segment.text.starts_with('*') {
            decode_segment(segment.text, segment.start, kind)?;
        }
    }

    Ok(ParsedPath {
        segments,
        trailing_slash,
    })
}

fn decode_segment(text: &str, offset: usize, kind: ParseKind) -> Result<String, Error> {
    let invalid = |at| match kind {
        ParseKind::Pattern => Error::InvalidPattern { at },
        ParseKind::Path => Error::InvalidPath { at },
    };

    let bytes = text.as_bytes();
    let mut decoded = Vec::new();
    let mut i = 0usize;
    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(offset + i));
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            let value = match (hi, lo) {
                (Some(h), Some(l)) => h * 16 + l,
                _ => return Err(invalid(offset + i)),
            };
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(offset + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid(offset + i));
        }
    }
    Ok(String::from_utf8(decoded).unwrap_or_default())
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    let mut has_wildcard = false;
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                has_wildcard = true;
                break;
            }
        }
    }
    if !has_wildcard {
        rank.push(3);
    }
    rank
}

fn rank_cmp(left: &[u8], right: &[u8]) -> std::cmp::Ordering {
    left.cmp(right)
}

fn same_shape(left: &Route, right: &Route) -> bool {
    left.segments.len() == right.segments.len()
        && left
            .segments
            .iter()
            .zip(&right.segments)
            .all(|(a, b)| match (a, b) {
                (Segment::Static(x), Segment::Static(y)) => x == y,
                (Segment::Param(_), Segment::Param(_)) => true,
                (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                _ => false,
            })
}

fn match_route(route: &Route, path: &ParsedPath) -> Option<Vec<(String, String)>> {
    let has_wildcard = route
        .segments
        .last()
        .map_or(false, |segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = if has_wildcard {
        route.segments.len().saturating_sub(1)
    } else {
        route.segments.len()
    };

    if (!has_wildcard && path.segments.len() != fixed_len)
        || (has_wildcard && path.segments.len() < fixed_len)
    {
        return None;
    }

    let mut params = Vec::new();
    for (index, segment) in route.segments.iter().take(fixed_len).enumerate() {
        let value = path.segments.get(index)?;
        match segment {
            Segment::Static(expected) if expected == value => {}
            Segment::Static(_) => return None,
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if has_wildcard {
        if let Some(Segment::Wildcard(name)) = route.segments.last() {
            let mut value = String::new();
            for (i, part) in path.segments.iter().skip(fixed_len).enumerate() {
                if i != 0 {
                    value.push('/');
                }
                value.push_str(part);
            }
            params.push((name.clone(), value));
        }
    }
    Some(params)
}

fn supplied_value<'a>(params: &[(&'a str, &'a str)], key: &str) -> Option<&'a str> {
    params
        .iter()
        .find_map(|(supplied_key, value)| if *supplied_key == key { Some(*value) } else { None })
}

fn valid_build_param(value: &str) -> bool {
    !value.is_empty()
        && value
            .bytes()
            .all(|byte| (0x20..=0x7e).contains(&byte) && byte != b'/')
}

fn valid_build_wildcard(value: &str) -> bool {
    value.is_empty()
        || (value
            .bytes()
            .all(|byte| (0x20..=0x7e).contains(&byte))
            && !value.starts_with('/')
            && !value.ends_with('/')
            && !value.contains("//"))
}

fn append_encoded(output: &mut String, bytes: &[u8]) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in bytes {
        if is_unreserved(byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
}
