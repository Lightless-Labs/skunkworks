#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct ParsedPath {
    segments: Vec<String>,
    trailing_slash: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum ParseKind {
    Pattern,
    Path,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing_slash) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let candidate = Route {
            id: self.routes.len(),
            name: name.to_string(),
            method: method.to_string(),
            rank: make_rank(&segments),
            segments,
            trailing_slash,
        };

        for route in &self.routes {
            if route.method == candidate.method
                && (self.policy != SlashPolicy::Strict
                    || route.trailing_slash == candidate.trailing_slash)
                && same_shape(route, &candidate)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = candidate.id;
        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, ch)| if ch == '?' || ch == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path(head, ParseKind::Path)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if let Some(params) = match_route(route, &parsed) {
                if self.policy == SlashPolicy::Strict
                    && route.trailing_slash != parsed.trailing_slash
                {
                    continue;
                }
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        let mut allowed: Vec<String> = Vec::new();

        for (route, params) in candidates {
            if !allowed.iter().any(|m| m == &route.method) {
                allowed.push(route.method.clone());
            }
            if route.method != eligible_method {
                continue;
            }
            match &best {
                None => best = Some((route, params)),
                Some((best_route, _)) => {
                    if rank_cmp(&route.rank, &best_route.rank).is_gt()
                        || (route.rank == best_route.rank && route.id < best_route.id)
                    {
                        best = Some((route, params));
                    }
                }
            }
        }

        if let Some((route, params)) = best {
            let redirect = if self.policy == SlashPolicy::Redirect
                && route.trailing_slash != parsed.trailing_slash
                && !parsed.segments.is_empty()
            {
                let mut target = head.to_string();
                if route.trailing_slash {
                    target.push('/');
                } else if target.ends_with('/') {
                    target.pop();
                }
                Some(target)
            } else {
                None
            };
            return Ok(Match {
                route: route.id,
                params,
                redirect,
            });
        }

        if method == "HEAD" && eligible_method == "GET" && !allowed.iter().any(|m| m == "HEAD") {
            allowed.push("HEAD".to_string());
        }
        allowed.sort();
        allowed.dedup();
        Err(Error::MethodNotAllowed { allowed })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = match self.routes.iter().find(|route| route.name == name) {
            Some(route) => route,
            None => return Err(Error::UnknownName),
        };

        let mut supplied: Vec<(&str, &str)> = Vec::new();
        for &(key, value) in params {
            if supplied.iter().any(|(seen, _)| *seen == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        for segment in &route.segments {
            if let Segment::Param(key) | Segment::Wildcard(key) = segment {
                if !supplied.iter().any(|(supplied_key, _)| *supplied_key == key) {
                    return Err(Error::MissingParam { name: key.clone() });
                }
            }
        }

        let mut extra: Option<&str> = None;
        for (key, _) in &supplied {
            let is_capture = route.segments.iter().any(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => name == key,
                Segment::Static(_) => false,
            });
            if !is_capture && extra.map_or(true, |current| key.as_bytes() < current.as_bytes()) {
                extra = Some(key);
            }
        }
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: key.to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Static(_) => {}
                Segment::Param(key) => {
                    let value = supplied_value(&supplied, key).unwrap_or("");
                    if !valid_build_param(value) {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
                Segment::Wildcard(key) => {
                    let value = supplied_value(&supplied, key).unwrap_or("");
                    if !valid_build_wildcard(value) {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if let Segment::Wildcard(key) = segment {
                let value = supplied_value(&supplied, key).unwrap_or("");
                if !value.is_empty() {
                    for component in value.split('/') {
                        if !first {
                            output.push('/');
                        }
                        append_encoded(&mut output, component.as_bytes());
                        first = false;
                    }
                }
                continue;
            }

            if !first {
                output.push('/');
            }
            match segment {
                Segment::Static(value) => append_encoded(&mut output, value.as_bytes()),
                Segment::Param(key) => {
                    let value = supplied_value(&supplied, key).unwrap_or("");
                    append_encoded(&mut output, value.as_bytes());
                }
                Segment::Wildcard(_) => {}
            }
            first = false;
        }

        if route.trailing_slash && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let parsed = parse_path(pattern, ParseKind::Pattern)?;
    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();

    for (idx, raw) in raw_segments(pattern, parsed.trailing_slash).into_iter().enumerate() {
        let start = raw.start;
        let text = raw.text;
        if let Some(name) = text.strip_prefix(':') {
            if !valid_capture_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = text.strip_prefix('*') {
            if idx + 1 != raw_segments(pattern, parsed.trailing_slash).len() {
                return Err(Error::InvalidPattern { at: start });
            }
            if !valid_capture_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            let decoded = decode_segment(text, start, ParseKind::Pattern)?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, parsed.trailing_slash))
}

#[derive(Clone, Copy)]
struct RawSegment<'a> {
    text: &'a str,
    start: usize,
}

fn raw_segments(path: &str, trailing_slash: bool) -> Vec<RawSegment<'_>> {
    if path == "/" {
        return Vec::new();
    }
    let end = if trailing_slash {
        path.len().saturating_sub(1)
    } else {
        path.len()
    };
    let mut result = Vec::new();
    let mut start = 1usize;
    let bytes = path.as_bytes();
    let mut i = 1usize;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            result.push(RawSegment {
                text: &path[start..i],
                start,
            });
            start = i + 1;
        }
        i += 1;
    }
    result
}

fn parse_path(path: &str, kind: ParseKind) -> Result<ParsedPath, Error> {
    let invalid = |at| match kind {
        ParseKind::Pattern => Error::InvalidPattern { at },
        ParseKind::Path => Error::InvalidPath { at },
    };

    if !path.starts_with('/') {
        return Err(invalid(0));
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let trailing_slash = path.ends_with('/');
    let raw = raw_segments(path, trailing_slash);
    let mut segments = Vec::new();

    for segment in raw {
        if segment.text.is_empty() {
            return Err(invalid(segment.start));
        }
        if kind == ParseKind::Path {
            segments.push(decode_segment(segment.text, segment.start, kind)?);
        } else if !segment.text.starts_with(':') && !segment.text.starts_with('*') {
            decode_segment(segment.text, segment.start, kind)?;
        }
    }

    Ok(ParsedPath {
        segments,
        trailing_slash,
    })
}

fn decode_segment(text: &str, offset: usize, kind: ParseKind) -> Result<String, Error> {
    let invalid = |at| match kind {
        ParseKind::Pattern => Error::InvalidPattern { at },
        ParseKind::Path => Error::InvalidPath { at },
    };

    let bytes = text.as_bytes();
    let mut decoded = Vec::new();
    let mut i = 0usize;
    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(offset + i));
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            let value = match (hi, lo) {
                (Some(h), Some(l)) => h * 16 + l,
                _ => return Err(invalid(offset + i)),
            };
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(offset + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(invalid(offset + i));
        }
    }
    Ok(String::from_utf8(decoded).unwrap_or_default())
}

fn upper_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    let mut has_wildcard = false;
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                has_wildcard = true;
                break;
            }
        }
    }
    if !has_wildcard {
        rank.push(3);
    }
    rank
}

fn rank_cmp(left: &[u8], right: &[u8]) -> std::cmp::Ordering {
    left.cmp(right)
}

fn same_shape(left: &Route, right: &Route) -> bool {
    left.segments.len() == right.segments.len()
        && left
            .segments
            .iter()
            .zip(&right.segments)
            .all(|(a, b)| match (a, b) {
                (Segment::Static(x), Segment::Static(y)) => x == y,
                (Segment::Param(_), Segment::Param(_)) => true,
                (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                _ => false,
            })
}

fn match_route(route: &Route, path: &ParsedPath) -> Option<Vec<(String, String)>> {
    let has_wildcard = route
        .segments
        .last()
        .map_or(false, |segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = if has_wildcard {
        route.segments.len().saturating_sub(1)
    } else {
        route.segments.len()
    };

    if (!has_wildcard && path.segments.len() != fixed_len)
        || (has_wildcard && path.segments.len() < fixed_len)
    {
        return None;
    }

    let mut params = Vec::new();
    for (index, segment) in route.segments.iter().take(fixed_len).enumerate() {
        let value = path.segments.get(index)?;
        match segment {
            Segment::Static(expected) if expected == value => {}
            Segment::Static(_) => return None,
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if has_wildcard {
        if let Some(Segment::Wildcard(name)) = route.segments.last() {
            let mut value = String::new();
            for (i, part) in path.segments.iter().skip(fixed_len).enumerate() {
                if i != 0 {
                    value.push('/');
                }
                value.push_str(part);
            }
            params.push((name.clone(), value));
        }
    }
    Some(params)
}

fn supplied_value<'a>(params: &[(&'a str, &'a str)], key: &str) -> Option<&'a str> {
    params
        .iter()
        .find_map(|(supplied_key, value)| if *supplied_key == key { Some(*value) } else { None })
}

fn valid_build_param(value: &str) -> bool {
    !value.is_empty()
        && value
            .bytes()
            .all(|byte| (0x20..=0x7e).contains(&byte) && byte != b'/')
}

fn valid_build_wildcard(value: &str) -> bool {
    value.is_empty()
        || (value
            .bytes()
            .all(|byte| (0x20..=0x7e).contains(&byte))
            && !value.starts_with('/')
            && !value.ends_with('/')
            && !value.contains("//"))
}

fn append_encoded(output: &mut String, bytes: &[u8]) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in bytes {
        if is_unreserved(byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
}
