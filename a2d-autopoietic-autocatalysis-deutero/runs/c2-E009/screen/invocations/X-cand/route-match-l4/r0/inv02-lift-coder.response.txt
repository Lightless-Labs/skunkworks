use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method == method
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
                && same_shape(&route.segments, &segments)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        let rank = make_rank(&segments);
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if let Some(params) = match_segments(&route.segments, &request_segments) {
                if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                    continue;
                }
                candidates.push((route, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_exists = method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD");
        let effective = if method == "HEAD" && !head_exists {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(route, _)| route.method == effective)
            .collect();

        if eligible.is_empty() {
            let mut allowed = Vec::new();
            for (route, _) in &eligible {
                if !allowed.contains(&route.method) {
                    allowed.push(route.method.clone());
                }
            }
            // Rebuild allowed from all path candidates; eligible is empty here.
            for route in &self.routes {
                if match_segments(&route.segments, &request_segments).is_some()
                    && (self.policy != SlashPolicy::Strict
                        || route.trailing == request_trailing)
                    && !allowed.contains(&route.method)
                {
                    allowed.push(route.method.clone());
                }
            }
            if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank
                .cmp(&a.rank)
                .then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !request_segments.is_empty()
        {
            let mut target = head.to_string();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied: Vec<(&str, &str)> = Vec::new();
        let mut seen = HashSet::new();
        for &(key, value) in params {
            if !seen.insert(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(name) | Segment::Wildcard(name) => required.push(name.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.iter().any(|(provided, _)| provided == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .iter()
            .map(|(key, _)| *key)
            .filter(|key| !required.contains(key))
            .collect();
        extras.sort_unstable();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = lookup_param(&supplied, name).unwrap_or("");
                    if !valid_param_value(value) {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = lookup_param(&supplied, name).unwrap_or("");
                    if !valid_wildcard_value(value) {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut out = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            let parts: Vec<String> = match segment {
                Segment::Static(text) => vec![encode_segment(text)],
                Segment::Param(name) => {
                    vec![encode_segment(lookup_param(&supplied, name).unwrap_or(""))]
                }
                Segment::Wildcard(name) => {
                    let value = lookup_param(&supplied, name).unwrap_or("");
                    if value.is_empty() {
                        Vec::new()
                    } else {
                        value.split('/').map(encode_segment).collect()
                    }
                }
            };
            for part in parts {
                if !first {
                    out.push('/');
                }
                out.push_str(&part);
                first = false;
            }
        }
        if route.trailing && !out.ends_with('/') && out != "/" {
            out.push('/');
        }
        Ok(out)
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }
    let bytes = pattern.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1;
    while start < end {
        let next = pattern[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPattern { at: start });
        }
        let part = &pattern[start..next];
        if part.starts_with(':') || part.starts_with('*') {
            let name = &part[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if part.starts_with('*') && next != end {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if part.starts_with(':') {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            });
        } else {
            segments.push(Segment::Static(decode_segment(part, start, true)?));
        }
        if next == end {
            break;
        }
        start = next + 1;
    }
    Ok((segments, trailing),)
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }
    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;
    while start < end {
        let next = path[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(&path[start..next], start, false)?);
        if next == end {
            break;
        }
        start = next + 1;
    }
    Ok((segments, trailing))
}

fn decode_segment(segment: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                });
            }
            let decoded = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                });
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: base + i }
            } else {
                Error::InvalidPath { at: base + i }
            });
        }
    }
    Ok(String::from_utf8(out).unwrap_or_default())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn make_rank(segments: &[Segment]) -> Vec<u8> {
    let mut rank = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => rank.push(2),
            Segment::Param(_) => rank.push(1),
            Segment::Wildcard(_) => {
                rank.push(0);
                return rank;
            }
        }
    }
    rank.push(3);
    rank
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(left, right)| match (left, right) {
            (Segment::Static(a), Segment::Static(b)) => a == b,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn match_segments(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard = matches!(pattern.last(), Some(Segment::Wildcard(_)));
    let fixed = if wildcard {
        pattern.len() - 1
    } else {
        pattern.len()
    };
    if (!wildcard && request.len() != fixed) || (wildcard && request.len() < fixed) {
        return None;
    }
    let mut params = Vec::new();
    for (segment, value) in pattern[..fixed].iter().zip(request.iter()) {
        match segment {
            Segment::Static(text) if text != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }
    if let Some(Segment::Wildcard(name)) = pattern.last() {
        params.push((name.clone(), request[fixed..].join("/")));
    }
    Some(params)
}

fn lookup_param<'a>(params: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    params
        .iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn valid_param_value(value: &str) -> bool {
    !value.is_empty()
        && value.bytes().all(|b| (0x20..=0x7e).contains(&b))
        && !value.contains('/')
}

fn valid_wildcard_value(value: &str) -> bool {
    value.is_empty()
        || (value.bytes().all(|b| (0x20..=0x7e).contains(&b))
            && value.split('/').all(|part| !part.is_empty()))
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}
