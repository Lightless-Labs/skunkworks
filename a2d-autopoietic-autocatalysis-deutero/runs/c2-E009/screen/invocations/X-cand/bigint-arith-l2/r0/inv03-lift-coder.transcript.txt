use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan every magnitude byte before considering malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        // With invalid bytes ruled out, check every underscore's immediate neighbors.
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let first_digit_index = (start..bytes.len())
            .find(|&index| bytes[index] != b'_')
            .expect("a nonempty magnitude contains a digit");

        let digit_count = bytes[start..].iter().filter(|&&byte| byte != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::new();
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }

        let is_zero = digits.iter().all(|&digit| digit == 0);
        Ok(Self {
            negative: negative && !is_zero,
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let output_len = magnitude_len.max(min_digits);
        let mut magnitude = Vec::new();

        for _ in magnitude_len..output_len {
            magnitude.push(b'0');
        }
        for &digit in self.digits.iter().rev() {
            magnitude.push(b'0' + digit);
        }

        let separators = if group > 0 {
            (magnitude.len() - 1) / group
        } else {
            0
        };
        let mut output = String::new();
        if self.negative {
            output.push('-');
        }

        for (index, &byte) in magnitude.iter().enumerate() {
            if group > 0 && index > 0 && (magnitude.len() - index) % group == 0 {
                output.push('_');
            }
            output.push(byte as char);
        }

        // Keep the calculation explicit without using caller-provided sizes to allocate.
        let _ = separators;
        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => compare_magnitude(&self.digits, &rhs.digits),
            (true, true) => compare_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = add_magnitudes(&self.digits, &rhs.digits);
            Self::from_parts(self.negative, digits)
        } else {
            match compare_magnitude(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let digits = subtract_magnitudes(&self.digits, &rhs.digits);
                    Self::from_parts(self.negative, digits)
                }
                Ordering::Less => {
                    let digits = subtract_magnitudes(&rhs.digits, &self.digits);
                    Self::from_parts(rhs.negative, digits)
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !is_zero(&result.digits) {
            result.negative = !result.negative;
        } else {
            result.negative = false;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if is_zero(&self.digits) {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if is_zero(&self.digits) || is_zero(&rhs.digits) {
            return Self::zero();
        }

        let mut result = Vec::new();
        for &left_digit in &self.digits {
            let mut row = Vec::new();
            for _ in 0..result.len() {
                row.push(0);
            }

            let mut carry = 0u16;
            for &right_digit in &rhs.digits {
                let product = left_digit as u16 * right_digit as u16 + carry;
                row.push((product % 10) as u8);
                carry = product / 10;
            }
            while carry > 0 {
                row.push((carry % 10) as u8);
                carry /= 10;
            }

            result = add_magnitudes(&result, &row);
        }

        Self::from_parts(self.negative != rhs.negative, result)
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let mut result = Self { negative, digits };
        normalize(&mut result.digits);
        if is_zero(&result.digits) {
            result.negative = false;
        }
        result
    }
}

fn is_zero(digits: &[u8]) -> bool {
    digits.len() == 1 && digits[0] == 0
}

fn normalize(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}

fn compare_magnitude(left: &[u8], right: &[u8]) -> Ordering {
    match left.len().cmp(&right.len()) {
        Ordering::Equal => {
            for (&left_digit, &right_digit) in left.iter().rev().zip(right.iter().rev()) {
                match left_digit.cmp(&right_digit) {
                    Ordering::Equal => {}
                    ordering => return ordering,
                }
            }
            Ordering::Equal
        }
        ordering => ordering,
    }
}

fn add_magnitudes(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u8;
    let length = left.len().max(right.len());

    for index in 0..length {
        let left_digit = left.get(index).copied().unwrap_or(0);
        let right_digit = right.get(index).copied().unwrap_or(0);
        let sum = left_digit + right_digit + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry > 0 {
        result.push(carry);
    }
    normalize(&mut result);
    result
}

// Requires left >= right.
fn subtract_magnitudes(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut borrow = 0i8;

    for (index, &left_digit) in left.iter().enumerate() {
        let right_digit = right.get(index).copied().unwrap_or(0) as i8;
        let mut difference = left_digit as i8 - right_digit - borrow;
        if difference < 0 {
            difference += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(difference as u8);
    }

    normalize(&mut result);
    result
}
