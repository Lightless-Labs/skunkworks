#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Debug, Clone)]
struct Row {
    prefix: u8,
    payload: String,
}

#[derive(Debug, Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn canonical_number(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.starts_with('0')) {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let mut n = 0usize;
    for b in s.bytes() {
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, tail) = rest.split_once(" @@")?;
    if tail != "" {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    let (a, b, c, d) = (
        canonical_number(a)?,
        canonical_number(b)?,
        canonical_number(c)?,
        canonical_number(d)?,
    );
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn nominal_start(start: usize, count: usize) -> i128 {
    if count > 0 {
        start as i128 - 1
    } else {
        start as i128
    }
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    if !patch.is_empty() && !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let mut hunks = Vec::new();
    if !patch.is_empty() {
        let lines: Vec<&str> = patch.split_terminator('\n').collect();
        for i in 0..2 {
            let expected = if i == 0 { "--- before" } else { "+++ after" };
            if lines.get(i).copied() != Some(expected) {
                return Err(ApplyError::BadHeader { line: i + 1 });
            }
        }
        if lines.len() < 3 || !lines[2].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: 3 });
        }

        let mut i = 2usize;
        let mut previous_old_end: Option<i128> = None;
        let mut net: i128 = 0;

        while i < lines.len() {
            let header_line = i + 1;
            if !lines[i].starts_with("@@") {
                return Err(ApplyError::BadHunkHeader { line: header_line });
            }
            let (a, b, c, d) = parse_hunk_header(lines[i])
                .ok_or(ApplyError::BadHunkHeader { line: header_line })?;
            let old_start = nominal_start(a, b);
            let new_start = nominal_start(c, d);

            i += 1;
            let mut rows = Vec::new();
            let mut old_seen = 0usize;
            let mut new_seen = 0usize;
            while i < lines.len() && !lines[i].starts_with("@@") {
                let row = lines[i];
                let bytes = row.as_bytes();
                if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                    return Err(ApplyError::BadBody { line: i + 1 });
                }
                if row.as_bytes().contains(&b'\r') {
                    return Err(ApplyError::BadBody { line: i + 1 });
                }
                let payload = &row[1..];
                let prefix = bytes[0];
                if prefix != b'+' {
                    old_seen += 1;
                }
                if prefix != b'-' {
                    new_seen += 1;
                }
                rows.push(Row {
                    prefix,
                    payload: payload.to_owned(),
                });
                i += 1;
            }
            if old_seen != b || new_seen != d {
                return Err(ApplyError::CountMismatch { line: header_line });
            }
            if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
                return Err(ApplyError::BadBody { line: header_line });
            }
            if let Some(end) = previous_old_end {
                if old_start < end || new_start != old_start + net {
                    return Err(ApplyError::Misordered { line: header_line });
                }
            } else if new_start != old_start {
                return Err(ApplyError::Misordered { line: header_line });
            }
            if hunks.len() == 16 {
                return Err(ApplyError::TooLarge);
            }
            previous_old_end = Some(old_start + b as i128);
            net += d as i128 - b as i128;
            hunks.push(Hunk {
                line: header_line,
                old_start: old_start as usize,
                old_count: b,
                new_start: c,
                new_count: d,
                rows,
            });
        }
        if hunks.is_empty() {
            return Err(ApplyError::BadHunkHeader { line: 3 });
        }
    }

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;
    for (idx, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.prefix == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.prefix == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let nominal = hunk.old_start as i128;
        let expected = nominal + net + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut context = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    if row.prefix != b'+' {
                        context.push(row.payload.as_str());
                    }
                }
                let max_start = working.len().saturating_sub(hunk.old_count);
                for s in 0..=max_start {
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let begin = s + l;
                    let end = s + hunk.old_count - r;
                    if end > working.len() || context.len() != end - begin {
                        continue;
                    }
                    if context
                        .iter()
                        .zip(&working[begin..end])
                        .all(|(a, b)| *a == b)
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(old) if rank < old => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(old) if rank == old => best.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
            }
        }

        let Some(rank) = best_rank else {
            return Err(ApplyError::NoMatch { hunk: idx + 1 });
        };
        let mut starts: Vec<usize> = best
            .iter()
            .filter(|(s, _, _)| {
                let dist = (*s as i128 - expected).abs();
                (best_rank == Some((rank.0, dist)))
            })
            .map(|(s, _, _)| *s)
            .collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: idx + 1,
                first: starts[0],
                second: starts[1],
            });
        }
        let s = starts[0];
        let (_, l, r) = best
            .iter()
            .filter(|(candidate, _, _)| *candidate == s)
            .min_by_key(|(_, l, _)| *l)
            .copied()
            .unwrap();

        let begin = s + l;
        let end = s + hunk.old_count - r;
        let replacement: Vec<String> = hunk.rows[l..hunk.rows.len() - r]
            .iter()
            .filter(|row| row.prefix != b'-')
            .map(|row| row.payload.clone())
            .collect();
        working.splice(begin..end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (nominal + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
