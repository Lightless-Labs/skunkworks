pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let n = tasks.len();

    // Validate all dependency entries before doing any scheduling work.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut successors: Vec<Vec<usize>> = (0..n).map(|_| Vec::new()).collect();
    for (task_id, task) in tasks.iter().enumerate() {
        for &dependency in &task.deps {
            successors[dependency].push(task_id);
        }
    }

    // Sorted available IDs; insert each newly unblocked child in order.
    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::new();
    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);

        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for (task_id, &degree) in indegree.iter().enumerate() {
            if degree != 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut earliest: Vec<Option<Timing>> = (0..n).map(|_| None).collect();
    let mut makespan = 0;

    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&dependency| earliest[dependency].as_ref().unwrap().finish)
            .max()
            .unwrap_or(0);

        let finish = match start.checked_add(tasks[task_id].duration) {
            Some(finish) => finish,
            None => return Err(Error::Overflow { task: task_id }),
        };

        earliest[task_id] = Some(Timing { start, finish });
        makespan = makespan.max(finish);
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest: Vec<Option<Timing>> = (0..n).map(|_| None).collect();
    for &task_id in order.iter().rev() {
        let finish = if successors[task_id].is_empty() {
            target
        } else {
            successors[task_id]
                .iter()
                .map(|&child| latest[child].as_ref().unwrap().start)
                .min()
                .unwrap()
        };
        let start = finish - tasks[task_id].duration;
        latest[task_id] = Some(Timing { start, finish });
    }

    let earliest: Vec<Timing> = earliest.into_iter().map(Option::unwrap).collect();
    let latest: Vec<Timing> = latest.into_iter().map(Option::unwrap).collect();
    let slack: Vec<u64> = (0..n)
        .map(|task_id| latest[task_id].start - earliest[task_id].start)
        .collect();

    let baseline = target - makespan;
    let critical: Vec<usize> = (0..n)
        .filter(|&task_id| slack[task_id] == baseline)
        .collect();

    let mut path = Vec::new();
    if n != 0 {
        let mut current = (0..n)
            .filter(|&task_id| successors[task_id].is_empty())
            .filter(|&task_id| earliest[task_id].finish == makespan)
            .max()
            .unwrap();

        loop {
            path.push(current);
            if tasks[current].deps.is_empty() {
                break;

            current = *tasks[current]
                .deps
                .iter()
                .max_by_key(|&&dependency| (earliest[dependency].finish, dependency))
                .unwrap();
        }
        path.reverse();
    }

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}

            }
        }

        let mut current = sink.expect("nonempty DAG has a makespan sink");
        let mut reversed_path = Vec::new();
        loop {
            reversed_path.push(current);
            if tasks[current].deps.is_empty() {
                break;
            }

            let mut best: Option<usize> = None;
            for &predecessor in &tasks[current].deps {
                best = Some(match best {
                    None => predecessor,
                    Some(previous) => {
                        let previous_finish = earliest_result[previous].finish;
                        let candidate_finish = earliest_result[predecessor].finish;
                        if candidate_finish > previous_finish
                            || (candidate_finish == previous_finish
                                && predecessor > previous)
                        {
                            predecessor
                        } else {
                            previous
                        }
                    }
                });
            }
            current = best.expect("non-root task has a predecessor");
        }
        reversed_path.reverse();
        reversed_path
    };

    Ok(Analysis {
        order,
        earliest: earliest_result,
        latest: latest_result,
        slack,
        critical,
        path,
        makespan,
    })
}

struct BinaryMinHeap {
    data: Vec<usize>,
}

impl BinaryMinHeap {
    fn new() -> Self {
        Self { data: Vec::new() }
    }

    fn push(&mut self, value: usize) {
        self.data.push(value);
        let mut index = self.data.len() - 1;
        while index > 0 {
            let parent = (index - 1) / 2;
            if self.data[parent] <= self.data[index] {
                break;
            }
            self.data.swap(parent, index);
            index = parent;
        }
    }

    fn pop(&mut self) -> Option<usize> {
        let result = self.data.first().copied()?;
        let last = self.data.pop().expect("heap is nonempty");
        if !self.data.is_empty() {
            self.data[0] = last;
            let mut index = 0;
            loop {
                let left = index * 2 + 1;
                if left >= self.data.len() {
                    break;
                }
                let right = left + 1;
                let smaller_child = if right < self.data.len()
                    && self.data[right] < self.data[left]
                {
                    right
                } else {
                    left
                };
                if self.data[index] <= self.data[smaller_child] {
                    break;
                }
                self.data.swap(index, smaller_child);
                index = smaller_child;
            }
        }
        Some(result)
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Analysis {
    pub order: Vec<usize>,
    pub earliest: Vec<Timing>,
    pub latest: Vec<Timing>,
    pub slack: Vec<u64>,
    pub critical: Vec<usize>,
    pub path: Vec<usize>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
    TargetTooSmall {
        minimum: u64,
    },
}

pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let n = tasks.len();

    // Validate every dependency before building or traversing the graph.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let mut successors: Vec<Vec<usize>> = (0..n).map(|_| Vec::new()).collect();
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // A min-heap represented as Reverse ordering, so each step removes the
    // smallest currently available task ID.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::new();
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for task_id in 0..n {
            if indegree[task_id] != 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut earliest: Vec<Option<Timing>> = (0..n).map(|_| None).collect();
    let mut makespan = 0u64;

    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&predecessor| earliest[predecessor].as_ref().unwrap().finish)
            .max()
            .unwrap_or(0);

        let finish = match start.checked_add(tasks[task_id].duration) {
            Some(finish) => finish,
            None => return Err(Error::Overflow { task: task_id }),
        };

        earliest[task_id] = Some(Timing { start, finish });
        makespan = makespan.max(finish);
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest: Vec<Option<Timing>> = (0..n).map(|_| None).collect();
    for &task_id in order.iter().rev() {
        let finish = if successors[task_id].is_empty() {
            target
        } else {
            successors[task_id]
                .iter()
                .map(|&child| latest[child].as_ref().unwrap().start)
                .min()
                .unwrap()
        };
        let start = finish - tasks[task_id].duration;
        latest[task_id] = Some(Timing { start, finish });
    }

    let mut slack = Vec::new();
    for task_id in 0..n {
        slack.push(
            latest[task_id].as_ref().unwrap().start
                - earliest[task_id].as_ref().unwrap().start,
        );
    }

    let baseline = target - makespan;
    let critical = (0..n)
        .filter(|&task_id| slack[task_id] == baseline)
        .collect();

    let mut path = Vec::new();
    if n != 0 {
        let sink = (0..n)
            .filter(|&task_id| successors[task_id].is_empty())
            .filter(|&task_id| earliest[task_id].as_ref().unwrap().finish == makespan)
            .max()
            .unwrap();

        let mut current = sink;
        path.push(current);
        while !tasks[current].deps.is_empty() {
            let predecessor = tasks[current]
                .deps
                .iter()
                .copied()
                .max_by_key(|&id| (earliest[id].as_ref().unwrap().finish, id))
                .unwrap();
            current = predecessor;
            path.push(current);
        }
        path.reverse();
    }

    Ok(Analysis {
        order,
        earliest: earliest.into_iter().map(Option::unwrap).collect(),
        latest: latest.into_iter().map(Option::unwrap).collect(),
        slack,
        critical,
        path,
        makespan,
    })
}
