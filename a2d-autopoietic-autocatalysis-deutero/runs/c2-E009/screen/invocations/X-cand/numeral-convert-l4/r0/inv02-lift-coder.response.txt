#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const MAX_VALUE: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > MAX_VALUE {
        return Err(WordsError::Range);
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    if magnitude == 0 {
        result.push_str("zero");
        return Ok(result);
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut higher_present = false;
    for index in (1..groups.len()).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }
        if !result.is_empty() && result.as_bytes().last() != Some(&b' ') {
            result.push(' ');
        }
        append_group(&mut result, group);
        result.push(' ');
        result.push_str(SCALES[index - 1]);
        higher_present = true;
    }

    let units = groups[0];
    if units != 0 {
        if higher_present {
            if units < 100 {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        } else if !result.is_empty() && result.as_bytes().last() != Some(&b' ') {
            result.push(' ');
        }
        append_group(&mut result, units);
    }

    Ok(result)
}

fn append_group(output: &mut String, value: u16) {
    if value >= 100 {
        output.push_str(SMALL[(value / 100) as usize]);
        output.push_str(" hundred");
        let remainder = value % 100;
        if remainder != 0 {
            output.push_str(" and ");
            append_under_hundred(output, remainder);
        }
    } else {
        append_under_hundred(output, value);
    }
}

fn append_under_hundred(output: &mut String, value: u16) {
    if value < 20 {
        output.push_str(SMALL[value as usize]);
    } else {
        output.push_str(TENS[(value / 10 - 2) as usize]);
        if value % 10 != 0 {
            output.push('-');
            output.push_str(SMALL[(value % 10) as usize]);
        }
    }
}

fn dictionary_value(word: &str) -> Option<(WordKind, u128)> {
    match word {
        "zero" => Some((WordKind::Number, 0)),
        "one" => Some((WordKind::Number, 1)),
        "two" => Some((WordKind::Number, 2)),
        "three" => Some((WordKind::Number, 3)),

        "seven" => Some((WordKind::Number, 7)),
        "eight" => Some((WordKind::Number, 8)),
        "nine" => Some((WordKind::Number, 9)),
        "ten" => Some((WordKind::Number, 10)),
        "eleven" => Some((WordKind::Number, 11)),
        "twelve" => Some((WordKind::Number, 12)),
        "thirteen" => Some((WordKind::Number, 13)),
        "fourteen" => Some((WordKind::Number, 14)),
        "fifteen" => Some((WordKind::Number, 15)),
        "sixteen" => Some((WordKind::Number, 16)),
        "seventeen" => Some((WordKind::Number, 17)),
        "eighteen" => Some((WordKind::Number, 18)),
        "nineteen" => Some((WordKind::Number, 19)),
        "twenty" => Some((WordKind::Number, 20)),
        "thirty" => Some((WordKind::Number, 30)),
        "forty" => Some((WordKind::Number, 40)),
        "fifty" => Some((WordKind::Number, 50)),
        "sixty" => Some((WordKind::Number, 60)),
        "seventy" => Some((WordKind::Number, 70)),
        "eighty" => Some((WordKind::Number, 80)),
        "ninety" => Some((WordKind::Number, 90)),
        "hundred" => Some((WordKind::Hundred, 100)),

        "quintillion" => Some((WordKind::Scale, 1_000_000_000_000_000_000)),
        "and" => Some((WordKind::And, 0)),
        "minus" => Some((WordKind::Minus, 0)),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
    Minus,
}

#[derive(Clone, Copy)]
struct Token<'a> {
    word: &'a str,
    index: usize,
    kind: WordKind,
    value: u128,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0usize;
    for index in 0..=bytes.len() {
        if index == bytes.len() || bytes[index] == b' ' || bytes[index] == b'-' {
            let word = &input[start..index];
            let Some((kind, value)) = dictionary_value(word) else {
                return Err(WordsError::InvalidWord { index: start });
            };
            tokens.push(Token {
                word,
                index: start,
                kind,
                value,
            });
            start = index + 1;
        }
    }

    let mut negative = false;
    let mut first_arithmetic_token = 0usize;
    if let Some(first) = tokens.first() {
        if matches!(first.kind, WordKind::Minus) {
            negative = true;
            first_arithmetic_token = 1;
        }
    }

    for token in tokens.iter().skip(first_arithmetic_token) {
        if matches!(token.kind, WordKind::Minus) {
            return Err(WordsError::InvalidWord { index: token.index });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;

    for token in tokens.iter().skip(first_arithmetic_token) {
        match token.kind {
            WordKind::Number => group = group.saturating_add(token.value),
            WordKind::Hundred => group = group.saturating_mul(100),
            WordKind::Scale => {
                let scaled = group.saturating_mul(token.value);
                total = total.saturating_add(scaled);
                group = 0;
            }
            WordKind::And | WordKind::Minus => {}
        }

        if total.saturating_add(group) > MAX_VALUE {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != bytes {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(signed)
}
