#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];
const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];
const SCALES: [&str; 7] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion", "",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in groups.iter_mut().rev() {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let highest = groups.iter().rposition(|&group| group != 0);
    if let Some(highest) = highest {
        let units = groups[6];
        for (index, &group) in groups.iter().enumerate().take(7) {
            if group == 0 {
                continue;
            }
            if !result.is_empty() && result != "minus " {
                result.push(' ');
            }
            if index == 6 && units < 100 && highest < 6 {
                result.push_str("and ");
            }
            append_group(&mut result, group);
            let scale_index = 6 - index;
            if scale_index != 6 {
                result.push(' ');
                result.push_str(SCALES[scale_index]);
            }
        }
    }
    Ok(result)
}

fn append_group(output: &mut String, value: u16) {
    let value = value as usize;
    if value >= 100 {
        output.push_str(SMALL[value / 100]);
        output.push_str(" hundred");
        let remainder = value % 100;
        if remainder != 0 {
            output.push_str(" and ");
            append_subhundred(output, remainder);
        }
    } else {
        append_subhundred(output, value);
    }
}

fn append_subhundred(output: &mut String, value: usize) {
    if value < 20 {
        output.push_str(SMALL[value]);
    } else {
        output.push_str(TENS[value / 10 - 2]);
        if value % 10 != 0 {
            output.push('-');
            output.push_str(SMALL[value % 10]);
        }
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (index, (word, position)) in tokens.iter().enumerate() {
        if *word == "minus" {
            if index != 0 {
                return Err(WordsError::InvalidWord { index: *position });
            }
        } else if !is_dictionary_word(word) {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic_start = if negative { 1 } else { 0 };
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (word, _) in tokens.iter().skip(arithmetic_start) {
        match *word {
            "and" => {}
            "zero" => {}
            "hundred" => group *= 100,
            "thousand" => {
                total += group * 1_000;
                group = 0;
            }
            "million" => {
                total += group * 1_000_000;
                group = 0;
            }
            "billion" => {
                total += group * 1_000_000_000;
                group = 0;
            }
            "trillion" => {
                total += group * 1_000_000_000_000;
                group = 0;
            }
            "quadrillion" => {
                total += group * 1_000_000_000_000_000;
                group = 0;
            }
            "quintillion" => {
                total += group * 1_000_000_000_000_000_000;
                group = 0;
            }
            _ => group += word_value(word).unwrap_or(0),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    let canonical_bytes = canonical.as_bytes();
    let common_len = bytes.len().min(canonical_bytes.len());
    let first_difference = (0..common_len).find(|&index| bytes[index] != canonical_bytes[index]);
    if let Some(index) = first_difference {
        return Err(WordsError::NonCanonical { index });
    }
    if bytes.len() != canonical_bytes.len() {
        return Err(WordsError::NonCanonical { index: common_len });
    }
    Ok(value)
}

fn is_dictionary_word(word: &str) -> bool {
    word == "and"
        || word == "hundred"
        || word == "thousand"
        || word == "million"
        || word == "billion"
        || word == "trillion"
        || word == "quadrillion"
        || word == "quintillion"
        || SMALL[1..].contains(&word)
        || TENS.contains(&word)
}

fn word_value(word: &str) -> Option<u128> {
    if let Some(index) = SMALL.iter().position(|&candidate| candidate == word) {
        return Some(index as u128);
    }
    TENS
        .iter()
        .position(|&candidate| candidate == word)
        .map(|index| ((index + 2) * 10) as u128)
}
