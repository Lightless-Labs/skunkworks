#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const MAX_MAGNITUDE: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(&str, u128); 6] = [
    ("thousand", 1_000),
    ("million", 1_000_000),
    ("billion", 1_000_000_000),
    ("trillion", 1_000_000_000_000),
    ("quadrillion", 1_000_000_000_000_000),
    ("quintillion", 1_000_000_000_000_000_000),
];

fn append_under_hundred(n: u128, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        let tens = (n / 10) as usize - 2;
        out.push_str(TENS[tens]);
        let units = n % 10;
        if units != 0 {
            out.push('-');
            out.push_str(SMALL[units as usize]);
        }
    }
}

fn append_group(n: u128, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_under_hundred(remainder, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > MAX_MAGNITUDE {
        return Err(WordsError::Range);
    }

    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut groups = [0u16; 7];
    let mut rest = magnitude;
    for group in groups.iter_mut() {
        *group = (rest % 1000) as u16;
        rest /= 1000;
    }

    let mut first = true;
    for i in (0..groups.len()).rev() {
        let group = groups[i] as u128;
        if group == 0 {
            continue;
        }

        let units_group = i == 0;
        let use_cross_and = units_group
            && group < 100
            && groups[1..].iter().any(|&higher| higher != 0);

        if !first {
            if use_cross_and {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }

        append_group(group, &mut out);
        if i > 0 {
            out.push(' ');
            out.push_str(SCALES[i - 1].0);
        }
        first = false;
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }

    for (i, candidate) in SMALL.iter().enumerate().skip(1) {
        if word == *candidate {
            return Some(i as u128);
        }
    }

    for (i, candidate) in TENS.iter().enumerate() {
        if word == *candidate {
            return Some(((i + 2) * 10) as u128);
        }
    }

    match word {
        "hundred" => Some(100),
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        "and" => Some(0),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let before_is_letter = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let after_is_letter = index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !before_is_letter || !after_is_letter {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0usize;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, &(token, index)) in tokens.iter().enumerate() {
        if token == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic_start = usize::from(negative);
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(token, _) in tokens.iter().skip(arithmetic_start) {
        if token == "and" {
            // No arithmetic effect.
        } else if token == "hundred" {
            group = group.checked_mul(100).ok_or(WordsError::Range)?;
        } else if let Some((_, scale)) = SCALES.iter().find(|(name, _)| *name == token) {
            let scaled = group.checked_mul(*scale).ok_or(WordsError::Range)?;
            total = total.checked_add(scaled).ok_or(WordsError::Range)?;
            group = 0;
        } else {
            group = group
                .checked_add(word_value(token).ok_or(WordsError::Range)?)
                .ok_or(WordsError::Range)?;
        }

        let current = total.checked_add(group).ok_or(WordsError::Range)?;
        if current > MAX_MAGNITUDE {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    if magnitude > MAX_MAGNITUDE {
        return Err(WordsError::Range);
    }

    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let mismatch = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(value)
}
