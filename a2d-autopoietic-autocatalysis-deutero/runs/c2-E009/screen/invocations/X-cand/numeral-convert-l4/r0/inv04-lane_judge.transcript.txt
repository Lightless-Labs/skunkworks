const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut higher_present = false;
    for index in (0..7).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        if !result.is_empty() && result != "minus " {
            result.push(' ');
        }

        let is_units = index == 0;
        if is_units && higher_present && group < 100 {
            result.push_str("and ");
        }

        if index > 0 {
            result.push(' ');
            result.push_str(SCALES[index - 1]);
            higher_present = true;
        }
    }

    Ok(result)
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let before_is_letter = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let after_is_letter =
                index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !before_is_letter || !after_is_letter {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize, TokenKind)> = Vec::new();
    let mut start = 0;
    for index in 0..=bytes.len() {
        if index == bytes.len() || bytes[index] == b' ' || bytes[index] == b'-' {
            let word = &input[start..index];
            let kind = classify(word).ok_or(WordsError::InvalidWord { index: start })?;
            tokens.push((word, start, kind));
            start = index + 1;
        }
    }

    let negative = matches!(tokens.first(), Some((_, _, TokenKind::Minus)));
    for (index, (_, position, kind)) in tokens.iter().enumerate() {
        if matches!(kind, TokenKind::Minus) && index != 0 {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for (_, _, kind) in tokens.iter().skip(usize::from(negative)) {
        match *kind {
            TokenKind::Zero | TokenKind::And => {}
            TokenKind::Add(value) => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
            TokenKind::Minus => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let differing_index = canonical
            .bytes()
            .zip(bytes.iter().copied())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical {
            index: differing_index,
        });
    }
    Ok(value)
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
    Minus,
}

#[derive(Clone, Copy)]
struct Token<'a> {
    word: &'a str,
    index: usize,
    kind: WordKind,
    value: u128,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0usize;
    for index in 0..=bytes.len() {
        if index == bytes.len() || bytes[index] == b' ' || bytes[index] == b'-' {
            let word = &input[start..index];
            let Some((kind, value)) = dictionary_value(word) else {
                return Err(WordsError::InvalidWord { index: start });
            };
            tokens.push(Token {
                word,
                index: start,
                kind,
                value,
            });
            start = index + 1;
        }
    }

    let mut negative = false;
    let mut first_arithmetic_token = 0usize;
    if let Some(first) = tokens.first() {
        if matches!(first.kind, WordKind::Minus) {
            negative = true;
            first_arithmetic_token = 1;
        }
    }

    for token in tokens.iter().skip(first_arithmetic_token) {
        if matches!(token.kind, WordKind::Minus) {
            return Err(WordsError::InvalidWord { index: token.index });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;

    for token in tokens.iter().skip(first_arithmetic_token) {
        match token.kind {
            WordKind::Number => group = group.saturating_add(token.value),
            WordKind::Hundred => group = group.saturating_mul(100),
            WordKind::Scale => {
                let scaled = group.saturating_mul(token.value);
                total = total.saturating_add(scaled);
                group = 0;
            }
            WordKind::And | WordKind::Minus => {}
        }

        if total.saturating_add(group) > MAX_VALUE {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != bytes {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(signed)
}
