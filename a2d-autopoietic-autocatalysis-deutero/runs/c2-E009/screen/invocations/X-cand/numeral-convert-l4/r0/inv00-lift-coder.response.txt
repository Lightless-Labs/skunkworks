#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut higher_present = false;
    for index in (0..7).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        if !result.is_empty() && result != "minus " {
            result.push(' ');
        }

        let is_units = index == 0;
        if is_units && higher_present && group < 100 {
            result.push_str("and ");
        }

        append_group(&mut result, group);

        if index > 0 {
            result.push(' ');
            result.push_str(SCALES[index - 1]);
            higher_present = true;
        }
    }

    Ok(result)
}

fn append_group(output: &mut String, value: u16) {
    if value >= 100 {
        output.push_str(SMALL[(value / 100) as usize]);
        output.push_str(" hundred");
        let remainder = value % 100;
        if remainder != 0 {
            output.push_str(" and ");
            append_under_hundred(output, remainder);
        }
    } else {
        append_under_hundred(output, value);
    }
}

fn append_under_hundred(output: &mut String, value: u16) {
    if value < 20 {
        output.push_str(SMALL[value as usize]);
    } else {
        output.push_str(TENS[(value / 10 - 2) as usize]);
        if value % 10 != 0 {
            output.push('-');
            output.push_str(SMALL[(value % 10) as usize]);
        }
    }
}

#[derive(Clone, Copy)]
enum TokenKind {
    Zero,
    Add(u128),
    Hundred,
    Scale(u128),
    And,
    Minus,
}

fn classify(word: &str) -> Option<TokenKind> {
    if word == "zero" {
        return Some(TokenKind::Zero);
    }
    if word == "and" {
        return Some(TokenKind::And);
    }
    if word == "minus" {
        return Some(TokenKind::Minus);
    }
    if word == "hundred" {
        return Some(TokenKind::Hundred);
    }

    for (value, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(TokenKind::Add(value as u128));
        }
    }
    for (index, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(TokenKind::Add(((index + 2) * 10) as u128));
        }
    }

    match word {
        "thousand" => Some(TokenKind::Scale(1_000)),
        "million" => Some(TokenKind::Scale(1_000_000)),
        "billion" => Some(TokenKind::Scale(1_000_000_000)),
        "trillion" => Some(TokenKind::Scale(1_000_000_000_000)),
        "quadrillion" => Some(TokenKind::Scale(1_000_000_000_000_000)),
        "quintillion" => Some(TokenKind::Scale(1_000_000_000_000_000_000)),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let before_is_letter = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let after_is_letter =
                index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !before_is_letter || !after_is_letter {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize, TokenKind)> = Vec::new();
    let mut start = 0;
    for index in 0..=bytes.len() {
        if index == bytes.len() || bytes[index] == b' ' || bytes[index] == b'-' {
            let word = &input[start..index];
            let kind = classify(word).ok_or(WordsError::InvalidWord { index: start })?;
            tokens.push((word, start, kind));
            start = index + 1;
        }
    }

    let negative = matches!(tokens.first(), Some((_, _, TokenKind::Minus)));
    for (index, (_, position, kind)) in tokens.iter().enumerate() {
        if matches!(kind, TokenKind::Minus) && index != 0 {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for (_, _, kind) in tokens.iter().skip(usize::from(negative)) {
        match *kind {
            TokenKind::Zero | TokenKind::And => {}
            TokenKind::Add(value) => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
            TokenKind::Minus => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let differing_index = canonical
            .bytes()
            .zip(bytes.iter().copied())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical {
            index: differing_index,
        });
    }
    Ok(value)
}
