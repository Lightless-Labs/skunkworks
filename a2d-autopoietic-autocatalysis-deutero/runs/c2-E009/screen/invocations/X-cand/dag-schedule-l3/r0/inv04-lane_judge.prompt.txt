                    task: task_id,
                    position,
                    value,
                });

                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });

            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });

        return Err(Error::Workers { count: workers });

                running.push((finish, task_id, worker));
            }

            break;

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > time {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }

                next_time = Some(next_time.map_or(release, |current| current.min(release)));
            }

            if available_at > time {
                next_time = Some(next_time.map_or(available_at, |current| current.min(available_at)));
            }
        }

            Some(next) => time = next,
            None => {
                // With a validated acyclic graph, unfinished work always has a
                // future release, finish, or worker availability event.
                break;
            }
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task is started in an acyclic graph"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                let release = tasks[task_id].release;
                next_time = Some(next_time.map_or(release, |t| t.min(release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |t| t.min(available_at)));
            }
        }

        // If nothing can make progress, the remaining work is blocked only if
        // the graph was cyclic; that case was already handled above.
        let Some(next) = next_time else {
            break;
        };
        now = next;
    }

    let final_runs = runs.into_iter().map(Option::unwrap).collect();
    Ok(Schedule {
        runs: final_runs,
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before worker validation or computation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Kahn order uses a min-heap via Reverse.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::new();
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for (task_id, &degree) in indegree.iter().enumerate() {
            if degree != 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_successor_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();

    let mut worker_available = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // finish, task, worker
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all running tasks at this time in ascending task order.
        let mut finished_now = Vec::new();
        let mut i = 0;
        while i < running.len() {
            if running[i].0 == time {
                finished_now.push(running.swap_remove(i));
            } else {
                i += 1;
            }
        }
        finished_now.sort_by_key(|&(_, task_id, _)| task_id);
        for (finish, task_id, worker_id) in finished_now {
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }
        }

        loop {
            let worker_id = match worker_available
                .iter()
                .position(|&available_at| available_at <= time)
            {
                Some(id) => id,
                None => break,
            };

            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > time
                    || remaining_deps[task_id] != 0
                {
                    continue;
                }
                match best {
                    None => best = Some(task_id),
                    Some(current) => {
                        if tails[task_id] > tails[current]
                            || (tails[task_id] == tails[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                        {
                            best = Some(task_id);
                        }
                    }
                }
            }

            let task_id = match best {
                Some(id) => id,
                None => break,
            };

            let finish = time.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker_id] = available_at;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if finish == time {
                // Immediate completion of a zero-duration task.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > time && next_time.map_or(true, |current| finish < current) {
                next_time = Some(finish);
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > time {
                let release = tasks[task_id].release;
                if next_time.map_or(true, |current| release < current) {
                    next_time = Some(release);
                }
            }
        }
        for &available_at in &worker_available {
            if available_at > time && next_time.map_or(true, |current| available_at < current) {
                next_time = Some(available_at);
            }
        }

        // A task blocked only on a predecessor may become eligible at a finish;
        // such a finish is included among the running events above.
        time = next_time.expect("unfinished tasks always have a future event");
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task starts exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate all dependencies before any other validation or computation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Edges point from a predecessor to its children.
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children: Vec<Vec<usize>> = (0..n).map(|_| Vec::new()).collect();
    for (task_id, task) in tasks.iter().enumerate() {
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest available task ID.
    let mut available = std::collections::BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut order = Vec::new();
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for (id, &degree) in indegree.iter().enumerate() {
            if degree != 0 {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute each task's bottom level in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let mut successor_tail = 0;
        for &child in &children[id] {
            successor_tail = successor_tail.max(tails[child]);
        }
        tails[id] = tasks[id]
            .duration
            .checked_add(successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    if n == 0 {
        return Ok(Schedule {
            runs: Vec::new(),
            makespan: 0,
            events: Vec::new(),
        });
    }

    let mut runs: Vec<Option<Run>> = (0..n).map(|_| None).collect();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();

    // Per-worker next available time. Initially all workers are available at zero.
    let mut worker_available = vec![0u64; workers];

    // Running positive-duration tasks, ordered by (finish time, task ID).
    let mut running = std::collections::BTreeSet::new();

    let mut events = Vec::new();
    let mut completed_count = 0usize;
    let mut makespan = 0u64;
    let mut now = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, sorted by task ID.
        let mut finishing = Vec::new();
        for &(finish, task_id, worker_id) in &running {
            if finish == now {
                finishing.push((task_id, worker_id));
            } else if finish > now {
                break;
            }
        }
        for (task_id, worker_id) in finishing {
            running.remove(&(now, task_id, worker_id));
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(now);
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }
        }

        // Start tasks one at a time, re-evaluating eligibility and rank after each.
        loop {
            let worker_id = match worker_available.iter().position(|&t| t <= now) {
                Some(id) => id,
                None => break,
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now || remaining_deps[id] != 0 {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let task_id = match best {
                Some(id) => id,
                None => break,
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker_id] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                // Immediate completion is emitted before selecting another task.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.insert((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        // Find the earliest strictly later event that can change eligibility or
        // worker availability.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
                break;
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next_time = Some(next_time.map_or(tasks[id].release, |t| t.min(tasks[id].release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |t| t.min(available_at)));
            }
        }

        now = match next_time {
            Some(t) => t,
            None => break,
        };
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("all tasks must have started"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
