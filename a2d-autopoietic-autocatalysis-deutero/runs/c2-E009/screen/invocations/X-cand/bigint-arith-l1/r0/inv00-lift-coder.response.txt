```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits. Zero is represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let magnitude_start = usize::from(bytes[0] == b'+');
        if magnitude_start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan every magnitude byte before checking separators.
        for (index, &byte) in bytes.iter().enumerate().skip(magnitude_start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let magnitude = &bytes[magnitude_start..];
        let mut first_bad_separator = None;
        for (offset, &byte) in magnitude.iter().enumerate() {
            if byte == b'_'
                && (offset == 0
                    || offset + 1 == magnitude.len()
                    || !magnitude[offset - 1].is_ascii_digit()
                    || !magnitude[offset + 1].is_ascii_digit())
            {
                first_bad_separator = Some(magnitude_start + offset);
                break;
            }
        }
        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess_digit = None;
        for (offset, &byte) in magnitude.iter().enumerate() {
            if byte.is_ascii_digit() {
                digit_count += 1;
                if first_digit.is_none() {
                    first_digit = Some((byte, magnitude_start + offset));
                }
                if digit_count == max_digits.saturating_add(1) {
                    first_excess_digit = Some(magnitude_start + offset);
                }
            }
        }

        if digit_count > 1 {
            if let Some((b'0', index)) = first_digit {
                return Err(Error::LeadingZero { index });
            }
        }
        if digit_count > max_digits {
            // The first excess digit is present whenever digit_count > max_digits.
            let index = first_excess_digit.unwrap_or_else(|| {
                magnitude
                    .iter()
                    .enumerate()
                    .filter(|(_, byte)| byte.is_ascii_digit())
                    .nth(max_digits)
                    .map(|(offset, _)| magnitude_start + offset)
                    .unwrap_or(magnitude_start)
            });
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::new();
        for &byte in magnitude.iter().rev() {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }
        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let significant_len = self.digits.len().max(1);
        let width = significant_len.max(min_digits);
        let padding = width - significant_len;

        let mut output = String::new();
        for position in (0..width).rev() {
            if group > 0 && position + 1 < width && (width - 1 - position) % group == 0 {
                output.push('_');
            }
            let digit = if position < padding {
                0
            } else {
                self.digits[position - padding]
            };
            output.push(char::from(b'0' + digit));
        }
        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            std::cmp::Ordering::Equal => {
                for (left, right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(right) {
                        std::cmp::Ordering::Equal => {}
                        ordering => return ordering,
                    }
                }
                std::cmp::Ordering::Equal
            }
            ordering => ordering,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut digits = Vec::new();
        let mut carry = 0u8;
        let max_len = self.digits.len().max(rhs.digits.len());

        for index in 0..max_len {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }
        if carry != 0 {
            digits.push(carry);
        }
        if digits.is_empty() {
            digits.push(0);
        }
        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == std::cmp::Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::new();
        let mut borrow = 0i8;
        for index in 0..self.digits.len() {
            let left = self.digits[index] as i8 - borrow;
            let right = rhs.digits.get(index).copied().unwrap_or(0) as i8;
            if left < right {
                digits.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                digits.push((left - right) as u8);
                borrow = 0;
            }
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        Ok(Self { digits })
    }
}
```
