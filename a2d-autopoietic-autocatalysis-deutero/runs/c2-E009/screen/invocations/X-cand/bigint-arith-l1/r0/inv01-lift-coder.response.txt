```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian decimal digits; the most significant digit is nonzero
    // unless the value is zero, represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let magnitude_start = if bytes[0] == b'+' { 1 } else { 0 };
        if magnitude_start == bytes.len() && magnitude_start == 1 {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(magnitude_start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut digit_count = 0usize;
        let mut first_digit_index = None;
        let mut first_digit = 0u8;
        let mut previous_was_digit = false;
        let mut first_bad_separator = None;

        for (index, &byte) in bytes.iter().enumerate().skip(magnitude_start) {
            if byte.is_ascii_digit() {
                if first_digit_index.is_none() {
                    first_digit_index = Some(index);
                    first_digit = byte - b'0';
                }
                digit_count += 1;
                previous_was_digit = true;
            } else {
                let next_is_digit = bytes
                    .get(index + 1)
                    .is_some_and(u8::is_ascii_digit);
                if (!previous_was_digit || !next_is_digit) && first_bad_separator.is_none() {
                    first_bad_separator = Some(index);
                }
                previous_was_digit = false;
            }
        }

        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        if digit_count > 1 && first_digit == 0 {
            return Err(Error::LeadingZero {
                index: first_digit_index.expect("a nonempty magnitude has a digit"),
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(magnitude_start) {
                if byte.is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::new();
        for &byte in bytes[magnitude_start..].iter().rev() {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let mut output = String::new();

        for position in (0..width).rev() {
            if group > 0 && position + 1 < width && (width - 1 - position) % group == 0 {
                output.push('_');
            }

            let digit = if position < self.digits.len() {
                self.digits[position]
            } else {
                0
            };
            output.push(char::from(b'0' + digit));
        }

        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (left, right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(right) {
                        Ordering::Equal => {}
                        ordering => return ordering,
                    }
                }
                Ordering::Equal
            }
            ordering => ordering,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut digits = Vec::new();
        let mut carry = 0u8;
        let max_len = self.digits.len().max(rhs.digits.len());

        for index in 0..max_len {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            digits.push(carry);
        }

        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::new();
        let mut borrow = 0i8;

        for index in 0..self.digits.len() {
            let left = self.digits[index] as i8 - borrow;
            let right = rhs.digits.get(index).copied().unwrap_or(0) as i8;
            if left < right {
                digits.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                digits.push((left - right) as u8);
                borrow = 0;
            }
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }
}
```
