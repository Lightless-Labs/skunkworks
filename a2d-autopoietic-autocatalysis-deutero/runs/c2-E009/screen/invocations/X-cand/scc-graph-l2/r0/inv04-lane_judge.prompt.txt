    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }

    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut reverse_adjacency = vec![Vec::new(); n];

    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse_adjacency[to].push(from);
    }

    let mut visited = vec![false; n];
    let mut finish_order = Vec::new();

        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];

        while let Some((vertex, next_edge)) = stack.last_mut() {
            if *next_edge < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_edge];
                *next_edge += 1;

                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                finish_order.push(finished);
            }
        }
    }

            continue;

        while let Some(vertex) = stack.pop() {
            component.push(vertex);

            for &neighbor in &reverse_adjacency[vertex] {
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }

        component.sort_unstable();
        raw_components.push(component);
    }

    raw_components.sort_by(|a, b| {
        b.len()
            .cmp(&a.len())
            .then_with(|| a[0].cmp(&b[0]))

        for &vertex in component {
            component_of[vertex] = component_id;
        }
    }

        cyclic.push(component.len() >= 2);

            from,
            to,
            multiplicity,
            first_edge,

    Ok(Graph {
        components: raw_components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })

    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

struct DfsFrame {
    vertex: usize,
    next_edge: usize,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }

    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
    }

    // Iterative Tarjan SCC traversal avoids recursion depth limits on long paths.
    let mut discovery = vec![usize::MAX; n];
    let mut lowlink = vec![0; n];
    let mut on_stack = vec![false; n];
    let mut tarjan_stack = Vec::new();
    let mut raw_component_of = vec![usize::MAX; n];
    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    let mut next_discovery = 0usize;
    let mut frames = Vec::new();

    for root in 0..n {
        if discovery[root] != usize::MAX {
            continue;
        }

        discovery[root] = next_discovery;
        lowlink[root] = next_discovery;
        next_discovery += 1;
        tarjan_stack.push(root);
        on_stack[root] = true;
        frames.push(DfsFrame {
            vertex: root,
            next_edge: 0,
        });

        while !frames.is_empty() {
            let frame_index = frames.len() - 1;
            let vertex = frames[frame_index].vertex;

            if frames[frame_index].next_edge < adjacency[vertex].len() {
                let edge_index = frames[frame_index].next_edge;
                frames[frame_index].next_edge += 1;
                let neighbor = adjacency[vertex][edge_index];

                if discovery[neighbor] == usize::MAX {
                    discovery[neighbor] = next_discovery;
                    lowlink[neighbor] = next_discovery;
                    next_discovery += 1;
                    tarjan_stack.push(neighbor);
                    on_stack[neighbor] = true;
                    frames.push(DfsFrame {
                        vertex: neighbor,
                        next_edge: 0,
                    });
                } else if on_stack[neighbor] {
                    lowlink[vertex] = lowlink[vertex].min(discovery[neighbor]);
                }
            } else {
                frames.pop();

                if let Some(parent_frame) = frames.last() {
                    let parent = parent_frame.vertex;
                    lowlink[parent] = lowlink[parent].min(lowlink[vertex]);
                }

                if lowlink[vertex] == discovery[vertex] {
                    let component_id = raw_components.len();
                    let mut component = Vec::new();
                    loop {
                        let member = tarjan_stack.pop().unwrap();
                        on_stack[member] = false;
                        raw_component_of[member] = component_id;
                        component.push(member);
                        if member == vertex {
                            break;
                        }
                    }
                    raw_components.push(component);
                }
            }
        }
    }

    for component in &mut raw_components {
        component.sort_unstable();
    }
    raw_components.sort_by(|left, right| {
        right
            .len()
            .cmp(&left.len())
            .then_with(|| left[0].cmp(&right[0]))
    });

    let mut component_of = vec![0; n];
    for (component_id, component) in raw_components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
    }

    let mut cyclic = vec![false; component_count];
    let mut internal_edges = vec![0; component_count];
    for (component_id, component) in raw_components.iter().enumerate() {
        cyclic[component_id] = component.len() >= 2;
    }

    // Accumulate at most one entry per component pair using a bounded dense table.
    let mut arc_indices = vec![usize::MAX; component_count * component_count];
    let mut arcs = Vec::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let source_component = component_of[from];
        let target_component = component_of[to];

        if source_component == target_component {
            internal_edges[source_component] += 1;

                    first_edge: edge_index,

    });

    Ok(Graph {
        components: raw_components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })

        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,

                vertex: from,

            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }

        reverse[to].push(from);

                }

                finish_order.push(vertex);
            }

            members.push(vertex);
            for &next in &reverse[vertex] {
                if raw_component_of[next] == usize::MAX {
                    raw_component_of[next] = raw_id;
                    stack.push(next);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);

            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))

                slot @ None => *slot = Some((1, edge_index)),

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })

                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                finish_order.push(finished);
            }
        }
    }

    // Second pass assigns temporary component IDs.
    let mut temporary_id = vec![usize::MAX; n];
    let mut temporary_components: Vec<Vec<usize>> = Vec::new();
    for &start in finish_order.iter().rev() {
        if temporary_id[start] != usize::MAX {
            continue;
        }
        let id = temporary_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        temporary_id[start] = id;
        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &neighbor in &reverse_adjacency[vertex] {
                if temporary_id[neighbor] == usize::MAX {
                    temporary_id[neighbor] = id;
                    stack.push(neighbor);
                }
            }
        }
        members.sort_unstable();
        temporary_components.push(members);
    }

    // Canonicalize component ordering by size descending, then minimum vertex ascending.
    let mut order: Vec<usize> = (0..temporary_components.len()).collect();
    order.sort_unstable_by(|&a, &b| {
        temporary_components[b]
            .len()
            .cmp(&temporary_components[a].len())
            .then_with(|| temporary_components[a][0].cmp(&temporary_components[b][0]))
    });

    let mut old_to_new = vec![0usize; temporary_components.len()];
    let mut components = Vec::new();
    for (new_id, &old_id) in order.iter().enumerate() {
        old_to_new[old_id] = new_id;
        components.push(temporary_components[old_id].clone());
    }

    let mut component_of = vec![0usize; n];
    for (vertex, &old_id) in temporary_id.iter().enumerate() {
        component_of[vertex] = old_to_new[old_id];
    }

    let component_count = components.len();
    let mut cyclic = vec![false; component_count];
    for (id, members) in components.iter().enumerate() {
        cyclic[id] = members.len() >= 2;
    }

    let mut internal_edges = vec![0usize; component_count];
    let mut arc_data: Vec<(usize, usize, usize, usize)> = Vec::new();
    // At most MAX_EDGES entries are collected before sorting and combining.
    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let source_component = component_of[from];
        let target_component = component_of[to];
        if source_component == target_component {
            internal_edges[source_component] += 1;
            if from == to {
                cyclic[source_component] = true;
            }
        } else {
            arc_data.push((
                source_component,
                target_component,
                edge_index,
                1usize,
            ));
        }
    }

    arc_data.sort_unstable_by(|a, b| {
        a.0.cmp(&b.0)
            .then_with(|| a.1.cmp(&b.1))
            .then_with(|| a.2.cmp(&b.2))
    });

    let mut arcs = Vec::new();
    let mut index = 0;
    while index < arc_data.len() {
        let from = arc_data[index].0;
        let to = arc_data[index].1;
        let first_edge = arc_data[index].2;
        let mut multiplicity = 0usize;
        while index < arc_data.len() && arc_data[index].0 == from && arc_data[index].1 == to {
            multiplicity += arc_data[index].3;
            index += 1;
        }
        arcs.push(Arc {
            from,
            to,
            multiplicity,
            first_edge,
        });
    }

    // Arc sort contract uses target component descending within each source.
    arcs.sort_unstable_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
