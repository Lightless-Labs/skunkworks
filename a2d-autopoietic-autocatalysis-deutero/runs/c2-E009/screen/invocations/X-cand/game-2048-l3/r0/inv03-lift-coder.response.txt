#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let horizontal = matches!(direction, Direction::Left | Direction::Right);
    let line_count = if horizontal { board.height } else { board.width };
    let line_len = if horizontal { board.width } else { board.height };
    let reverse = matches!(direction, Direction::Right | Direction::Down);

    for line in 0..line_count {
        let mut inputs: Vec<(usize, u32)> = Vec::new();

        for offset in 0..line_len {
            let position = if reverse {
                line_len - 1 - offset
            } else {
                offset
            };
            let index = if horizontal {
                line * board.width + position
            } else {
                position * board.width + line
            };
            let value = board.cells[index];
            if value != 0 {
                inputs.push((index, value));
            }
        }

        let mut output_offset = 0usize;
        let mut i = 0usize;
        while i < inputs.len() {
            let (first_index, first_value) = inputs[i];
            let (value, consumed) = if i + 1 < inputs.len()
                && inputs[i + 1].1 == first_value
            {
                let second_index = inputs[i + 1].0;
                if first_value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: second_index,
                    });
                }
                let merged_value = first_value * 2;
                let destination_position = if reverse {
                    line_len - 1 - output_offset
                } else {
                    output_offset
                };
                let to = if horizontal {
                    line * board.width + destination_position
                } else {
                    destination_position * board.width + line
                };
                merges.push(Merge {
                    from: [first_index, second_index],
                    to,
                    value: merged_value,
                });
                score += u64::from(merged_value);
                (merged_value, 2)
            } else {
                (first_value, 1)
            };

            let destination_position = if reverse {
                line_len - 1 - output_offset
            } else {
                output_offset
            };
            let destination_index = if horizontal {
                line * board.width + destination_position
            } else {
                destination_position * board.width + line
            };
            cells[destination_index] = value;
            output_offset += 1;
            i += consumed;
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let mut directions = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(result) if result.changed => directions.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(directions)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Spawn {
    pub index: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {
    pub board: Board,
    pub rng: u64,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
    pub spawn: Option<Spawn>,
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;

    let mut empties = Vec::new();
    for (index, &value) in board.cells.iter().enumerate() {
        if value == 0 {
            empties.push(index);
        }
    }

    if empties.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let a = next_rng(rng);
    let index = empties[(a % empties.len() as u64) as usize];
    let b = next_rng(a);
    let value = if b % 10 == 0 { 4 } else { 2 };

    let mut spawned_board = board.clone();
    spawned_board.cells[index] = value;

    Ok(SpawnResult {
        board: spawned_board,
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;

    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: moved.changed,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width)
        || !(1..=8).contains(&board.height)
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

fn next_rng(state: u64) -> u64 {
    state
        .wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}
