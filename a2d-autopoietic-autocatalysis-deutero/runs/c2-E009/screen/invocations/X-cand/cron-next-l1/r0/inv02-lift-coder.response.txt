#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_wildcards: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_info(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(u32::from(byte - b'0')))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (_, maximum) = field_info(field);
        let width = maximum - field_info(field).0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (minimum, maximum) = field_info(field);
        if value < minimum || value > maximum {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_info(field);
    let mut selected = [false; 60];
    let mut item_start = 0usize;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_range) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some((*slash + 1, item.len()))),
            _ => return Err(syntax_error(field, item_start)),
        };

        let base = &item[..base_end];
        let (kind, first_range, second_range) = if base == b"*" {
            (0u8, None, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();

            match hyphens.as_slice() {
                [] if !base.is_empty() => (1, Some((0, base.len())), None),
                [hyphen]
                    if *hyphen > 0 && *hyphen + 1 < base.len() =>
                {
                    (
                        2,
                        Some((0, *hyphen)),
                        Some((*hyphen + 1, base.len())),
                    )
                }
                _ => return Err(syntax_error(field, item_start)),
            }
        };

        if step_range.is_some() && base.is_empty() {
            return Err(syntax_error(field, item_start));
        }

        let (first, last_base) = match kind {
            0 => (None, None),
            1 => {
                let (start, end) = first_range.unwrap();
                let value = parse_number(item, start, end, field, false)?;
                (Some(value), Some(value))
            }
            _ => {
                let (start_start, start_end) = first_range.unwrap();
                let (end_start, end_end) = second_range.unwrap();
                let start = parse_number(item, start_start, start_end, field, false)?;
                let end = parse_number(item, end_start, end_end, field, false)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_start,
                    });
                }
                (Some(start), Some(end))
            }
        };

        let step = if let Some((start, end)) = step_range {
            if start == end {
                return Err(syntax_error(field, item_start));
            }
            Some(parse_number(item, start, end, field, true)?)
        } else {
            None
        };

        let (start, end) = match kind {
            0 => (minimum, maximum),
            1 => {
                let value = first.unwrap();
                if step.is_some() {
                    (value, maximum)
                } else {
                    (value, value)
                }
            }
            _ => (first.unwrap(), last_base.unwrap()),
        };

        if let Some(step) = step {
            let mut value = start;
            while value <= end {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        } else {
            let mut value = start;
            loop {
                selected[value as usize] = true;
                if value == end {
                    break;
                }
                value += 1;
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];

    for i in 0..5 {
        selected[i] = parse_field(pieces[i], fields[i])?;
    }

    Ok(Cron {
        selected,
        day_wildcards: [
            pieces[2].as_bytes() == b"*",
            pieces[4].as_bytes() == b"*",
        ],
    })
}
