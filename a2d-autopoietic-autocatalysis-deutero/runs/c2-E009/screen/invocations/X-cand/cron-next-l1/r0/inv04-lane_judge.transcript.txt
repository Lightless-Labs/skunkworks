fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = u32::from(byte - b'0');
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(digit))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

                    offset: item_start,

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let field_order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());

        let (values, exact_star) = parse_field(part, field)?;
        parsed[index] = values;
        exact_stars[index] = exact_star;
    }

    Ok(Cron {
        fields: parsed,
        day_of_month_is_exact_star: exact_stars[2],
        day_of_week_is_exact_star: exact_stars[4],
    })

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    kind: ErrorKind,

    }

    Ok(value)

fn parse_field(input: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(error(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let min = field_min(field) as u32;
    let max = field_max(field) as u32;
    let width = max - min + 1;
    let mut selected = vec![false; width as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(error(field, item_start, ErrorKind::EmptyItem));
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();
        if slash_positions.len() > 1 {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        }

        let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
        } else {
            (item, None, item_end)
        };

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })

            BaseShape::Single

            && base[hyphens[0] + 1..].iter().all(u8::is_ascii_digit)

            BaseShape::Range(hyphens[0])
        } else {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        };

        let (start, end, start_offset, end_offset) = match base_shape {
            BaseShape::Star => (min, max, item_start, item_start),
            BaseShape::Single => {
                let value = parse_number(base, field, item_start, ErrorKind::InvalidSyntax)?;
                if value < min || value > max {
                    return Err(error(field, item_start, ErrorKind::OutOfRange));

                    &base[hyphen + 1..],
                    field,
                    end_offset,
                    ErrorKind::InvalidSyntax,

                    return Err(error(field, item_start, ErrorKind::OutOfRange));
                }

                    return Err(error(field, end_offset, ErrorKind::OutOfRange));

                if start_value > end_value {
                    return Err(error(field, end_offset, ErrorKind::ReversedRange));
                }
                (start_value, end_value, item_start, end_offset)

            }
        }

        if item_end == bytes.len() {
            break;
        }

    }

    is_step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (_, max) = field_bounds(field);
        let width = (max as u32) + 1
            - field_bounds(field).0 as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (min, max) = field_bounds(field);
        if value < min as u32 || value > max as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, item_start));
    }

    let base_end_rel = slash_positions.first().copied().unwrap_or(item.len());
    let base = &item[..base_end_rel];
    let step_span = slash_positions
        .first()
        .map(|&slash| (item_start + slash + 1, item_end));

    let (kind, first_span, second_span) = if base == b"*" {
        (0u8, None, None)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1 {
            return Err(invalid_syntax(field, item_start));
        }
        if let Some(&hyphen) = hyphens.first() {
            if hyphen == 0 || hyphen + 1 == base.len() {
                return Err(invalid_syntax(field, item_start));
            }
            (
                2u8,
                Some((item_start, item_start + hyphen)),
                Some((item_start + hyphen + 1, item_start + base_end_rel)),
            )
        } else {
            if base.is_empty() || base.iter().any(|&b| !b.is_ascii_digit()) {
                return Err(invalid_syntax(field, item_start));
            }
            (1u8, Some((item_start, item_start + base.len())), None)
        }
    };

    // Reject malformed base structures before reading any numeric values.
    if kind == 0 && base != b"*" {
        return Err(invalid_syntax(field, item_start));
    }
    if let Some((start, end)) = step_span {
        if start == end {
            return Err(invalid_syntax(field, item_start));
        }
        if item[start - item_start..end - item_start]
            .iter()
            .any(|&b| !b.is_ascii_digit())
        {
            return Err(invalid_syntax(field, item_start));
        }
    }

    let mut start_value = None;
    let mut end_value = None;
    if let Some((start, end)) = first_span {
        start_value = Some(parse_number(bytes, start, end, field, false)?);
    }
    if let Some((start, end)) = second_span {
        end_value = Some(parse_number(bytes, start, end, field, false)?);
        if start_value.unwrap() > end_value.unwrap() {
            return Err(ParseError::ReversedRange {
                field,
                offset: start,
            });
        }
    }

    let step = if let Some((start, end)) = step_span {
        Some(parse_number(bytes, start, end, field, true)?)
    } else {
        None
    };

    let (domain_min, domain_max) = field_bounds(field);
    let (range_start, range_end) = match kind {
        0 => (domain_min, domain_max),
        1 => {
            let n = start_value.unwrap() as u8;
            (n, if step.is_some() { domain_max } else { n })
        }
        _ => (start_value.unwrap() as u8, end_value.unwrap() as u8),
    };

    let stride = step.unwrap_or(1) as u16;
    let mut value = range_start as u16;
    while value <= range_end as u16 {
        selected[value as usize] = true;
        value += stride;
    }
    Ok(())
}

fn parse_field(field_text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = field_text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && b != b'*' && b != b'-' && b != b'/' && b != b',' {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let day_star = matches!(field, Field::DayOfMonth | Field::DayOfWeek) && bytes == b"*";
    let mut selected = [false; 60];

    let mut item_start = 0usize;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(bytes, item_start, i, field, &mut selected)?;
            item_start = i + 1;
        }
    }

    let values = selected
        .iter()
        .enumerate()
        .filter_map(|(i, &is_selected)| is_selected.then_some(i as u8))
        .collect();

    Ok((values, day_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_stars = [false; 2];

    for (i, (&field_text, &field)) in fields.iter().zip(field_order.iter()).enumerate() {
        let (values, is_day_star) = parse_field(field_text, field)?;
        parsed[i] = values;
        if field == Field::DayOfMonth {
            day_stars[0] = is_day_star;
        } else if field == Field::DayOfWeek {
            day_stars[1] = is_day_star;
        }
    }

    Ok(Cron {
        selected: parsed,
        day_stars,
    })
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(u32::from(byte - b'0')))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (_, maximum) = field_info(field);
        let width = maximum - field_info(field).0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (minimum, maximum) = field_info(field);
        if value < minimum || value > maximum {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_info(field);
    let mut selected = [false; 60];
    let mut item_start = 0usize;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_range) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some((*slash + 1, item.len()))),
            _ => return Err(syntax_error(field, item_start)),
        };

        let base = &item[..base_end];
        let (kind, first_range, second_range) = if base == b"*" {
            (0u8, None, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();

            match hyphens.as_slice() {
                [] if !base.is_empty() => (1, Some((0, base.len())), None),
                [hyphen]
                    if *hyphen > 0 && *hyphen + 1 < base.len() =>
                {
                    (
                        2,
                        Some((0, *hyphen)),
                        Some((*hyphen + 1, base.len())),
                    )
                }
                _ => return Err(syntax_error(field, item_start)),
            }
        };

        if step_range.is_some() && base.is_empty() {
            return Err(syntax_error(field, item_start));
        }

        let (first, last_base) = match kind {
            0 => (None, None),
            1 => {
                let (start, end) = first_range.unwrap();
                let value = parse_number(item, start, end, field, false)?;
                (Some(value), Some(value))
            }
            _ => {
                let (start_start, start_end) = first_range.unwrap();
                let (end_start, end_end) = second_range.unwrap();
                let start = parse_number(item, start_start, start_end, field, false)?;
                let end = parse_number(item, end_start, end_end, field, false)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_start,
                    });
                }
                (Some(start), Some(end))
            }
        };

        let step = if let Some((start, end)) = step_range {
            if start == end {
                return Err(syntax_error(field, item_start));
            }
            Some(parse_number(item, start, end, field, true)?)
        } else {
            None
        };

        let (start, end) = match kind {
            0 => (minimum, maximum),
            1 => {
                let value = first.unwrap();
                if step.is_some() {
                    (value, maximum)
                } else {
                    (value, value)
                }
            }
            _ => (first.unwrap(), last_base.unwrap()),
        };

        if let Some(step) = step {
            let mut value = start;
            while value <= end {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        } else {
            let mut value = start;
            loop {
                selected[value as usize] = true;
                if value == end {
                    break;
                }
                value += 1;
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [

        Vec::new(),
        Vec::new(),
