#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_exact_star: bool,
    day_of_week_exact_star: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| {
                if is_selected {
                    Some(value as u8 + field_min(field))
                } else {
                    None
                }
            })
            .collect()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_min(field: Field) -> u8 {
    match field {
        Field::Minute | Field::Hour | Field::DayOfWeek => 0,
        Field::DayOfMonth | Field::Month => 1,
    }
}

fn field_max(field: Field) -> u8 {
    match field {
        Field::Minute => 59,
        Field::Hour => 23,
        Field::DayOfMonth => 31,
        Field::Month => 12,
        Field::DayOfWeek => 6,
    }
}

fn error(field: Field, offset: usize, kind: ErrorKind) -> ParseError {
    match kind {
        ErrorKind::InvalidCharacter => ParseError::InvalidCharacter { field, offset },
        ErrorKind::EmptyItem => ParseError::EmptyItem { field, offset },
        ErrorKind::InvalidSyntax => ParseError::InvalidSyntax { field, offset },
        ErrorKind::LeadingZero => ParseError::LeadingZero { field, offset },
        ErrorKind::NumberOverflow => ParseError::NumberOverflow { field, offset },
        ErrorKind::OutOfRange => ParseError::OutOfRange { field, offset },
        ErrorKind::ReversedRange => ParseError::ReversedRange { field, offset },
        ErrorKind::BadStep => ParseError::BadStep { field, offset },
    }
}

#[derive(Clone, Copy)]
enum ErrorKind {
    InvalidCharacter,
    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,
    OutOfRange,
    ReversedRange,
    BadStep,
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    kind: ErrorKind,
) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(error(field, offset, kind));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(error(field, offset, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| error(field, offset, ErrorKind::NumberOverflow))?;
    }
    Ok(value)
}

fn parse_field(input: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(error(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let min = field_min(field) as u32;
    let max = field_max(field) as u32;
    let width = max - min + 1;
    let mut selected = vec![false; width as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(error(field, item_start, ErrorKind::EmptyItem));
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();
        if slash_positions.len() > 1 {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        }

        let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
        } else {
            (item, None, item_end)
        };

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        let base_shape = if is_star && hyphens.is_empty() {
            BaseShape::Star
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            BaseShape::Single
        } else if hyphens.len() == 1
            && hyphens[0] > 0
            && hyphens[0] + 1 < base.len()
            && base[..hyphens[0]].iter().all(u8::is_ascii_digit)
            && base[hyphens[0] + 1..].iter().all(u8::is_ascii_digit)
        {
            BaseShape::Range(hyphens[0])
        } else {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        };

        let (start, end, start_offset, end_offset) = match base_shape {
            BaseShape::Star => (min, max, item_start, item_start),
            BaseShape::Single => {
                let value = parse_number(base, field, item_start, ErrorKind::InvalidSyntax)?;
                if value < min || value > max {
                    return Err(error(field, item_start, ErrorKind::OutOfRange));
                }
                (value, if step_bytes.is_some() { max } else { value }, item_start, item_start)
            }
            BaseShape::Range(hyphen) => {
                let start_value =
                    parse_number(&base[..hyphen], field, item_start, ErrorKind::InvalidSyntax)?;
                let end_offset = item_start + hyphen + 1;
                let end_value = parse_number(
                    &base[hyphen + 1..],
                    field,
                    end_offset,
                    ErrorKind::InvalidSyntax,
                )?;
                if start_value < min || start_value > max {
                    return Err(error(field, item_start, ErrorKind::OutOfRange));
                }
                if end_value < min || end_value > max {
                    return Err(error(field, end_offset, ErrorKind::OutOfRange));
                }
                if start_value > end_value {
                    return Err(error(field, end_offset, ErrorKind::ReversedRange));
                }
                (start_value, end_value, item_start, end_offset)
            }
        };

        let step = if let Some(step_bytes) = step_bytes {
            let step = parse_number(step_bytes, field, step_offset, ErrorKind::BadStep)?;
            if step < 1 || step > width {
                return Err(error(field, step_offset, ErrorKind::BadStep));
            }
            step
        } else {
            1
        };

        let _ = (start_offset, end_offset);
        let mut value = start;
        while value <= end {
            selected[(value - min) as usize] = true;
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

enum BaseShape {
    Star,
    Single,
    Range(usize),
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed = Vec::new();
    for (text, field) in fields.iter().zip(field_kinds) {
        parsed.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().expect("exactly five fields"),
        day_of_month_exact_star: fields[2].as_bytes() == b"*",
        day_of_week_exact_star: fields[4].as_bytes() == b"*",
    })
}
