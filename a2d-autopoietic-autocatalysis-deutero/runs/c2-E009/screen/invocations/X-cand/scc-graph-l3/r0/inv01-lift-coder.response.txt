```rust
use std::cmp::Reverse;
use std::collections::{BinaryHeap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse_adjacency = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse_adjacency[to].push(from);
    }

    // Iterative Kosaraju traversal avoids recursion depth limits.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::new();
    for start in 0..n {
        if visited[start] {
            continue;
        }
        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.pop() {
            if next_index < adjacency[vertex].len() {
                stack.push((vertex, next_index + 1));
                let next = adjacency[vertex][next_index];
                if !visited[next] {
                    visited[next] = true;
                    stack.push((next, 0));
                }
            } else {
                finish_order.push(vertex);
            }
        }
    }

    let mut raw_components = Vec::new();
    let mut assigned = vec![false; n];
    for &start in finish_order.iter().rev() {
        if assigned[start] {
            continue;
        }
        assigned[start] = true;
        let mut component = Vec::new();
        let mut stack = vec![start];
        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &next in &reverse_adjacency[vertex] {
                if !assigned[next] {
                    assigned[next] = true;
                    stack.push(next);
                }
            }
        }
        component.sort_unstable();
        raw_components.push(component);
    }

    raw_components.sort_by(|left, right| {
        right
            .len()
            .cmp(&left.len())
            .then_with(|| left[0].cmp(&right[0]))
    });

    let mut component_of = vec![0; n];
    for (component_id, component) in raw_components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
    }

    let component_count = raw_components.len();
    let mut cyclic = vec![false; component_count];
    let mut internal_edges = vec![0usize; component_count];
    let mut arc_data = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (component_id, component) in raw_components.iter().enumerate() {
        cyclic[component_id] = component.len() >= 2;
    }
    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let source = component_of[from];
        let target = component_of[to];
        if source == target {
            internal_edges[source] += 1;
            if from == to {
                cyclic[source] = true;
            }
        } else {
            let entry = arc_data.entry((source, target)).or_insert((0, edge_index));
            entry.0 += 1;
            entry.1 = entry.1.min(edge_index);
        }
    }

    let mut arcs = Vec::new();
    let mut outgoing = vec![Vec::new(); component_count];
    let mut indegree = vec![0usize; component_count];
    for (&(from, to), &(multiplicity, first_edge)) in &arc_data {
        arcs.push(Arc {
            from,
            to,
            multiplicity,
            first_edge,
        });
        outgoing[from].push(to);
        indegree[to] += 1;
    }
    arcs.sort_by(|left, right| {
        left.from
            .cmp(&right.from)
            .then_with(|| right.to.cmp(&left.to))
    });

    // Heap order: highest internal edge count, then highest minimum vertex.
    let mut ready = BinaryHeap::new();
    for component_id in 0..component_count {
        if indegree[component_id] == 0 {
            ready.push((
                internal_edges[component_id],
                raw_components[component_id][0],
                component_id,
            ));
        }
    }

    let mut topo = Vec::new();
    let mut depth = vec![0usize; component_count];
    while let Some((_, _, component_id)) = ready.pop() {
        topo.push(component_id);
        for &target in &outgoing[component_id] {
            depth[target] = depth[target].max(depth[component_id] + 1);
            indegree[target] -= 1;
            if indegree[target] == 0 {
                ready.push((
                    internal_edges[target],
                    raw_components[target][0],
                    target,
                ));
            }
        }
    }

    Ok(Graph {
        components: raw_components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut adjacency = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adjacency[arc.from].push(arc.to);
        }

        if let Some(route) = shortest_lex_route(&adjacency, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_route(&adjacency, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn shortest_lex_route(adjacency: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let count = adjacency.len();
    let mut distance = vec![None; count];
    let mut queue = VecDeque::new();
    distance[goal] = Some(0usize);
    queue.push_back(goal);

    // Reverse BFS computes distance to goal for every reachable component.
    let mut reverse = vec![Vec::new(); count];
    for (from, targets) in adjacency.iter().enumerate() {
        for &to in targets {
            reverse[to].push(from);
        }
    }
    while let Some(vertex) = queue.pop_front() {
        let next_distance = distance[vertex].unwrap() + 1;
        for &previous in &reverse[vertex] {
            if distance[previous].is_none() {
                distance[previous] = Some(next_distance);
                queue.push_back(previous);
            }
        }
    }

    distance[start]?;

    let mut route = vec![start];
    let mut current = start;
    while current != goal {
        let current_distance = distance[current]?;
        let next = adjacency[current]
            .iter()
            .copied()
            .filter(|&candidate| distance[candidate] == Some(current_distance - 1))
            .min()?;
        route.push(next);
        current = next;
    }
    Some(route)
}
```
