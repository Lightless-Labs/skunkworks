#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8], output: &mut String) {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        output.push('=');
    }

    if bytes.len() > 2 {
        output.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        output.push('=');
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut output = String::new();
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        encode_group(chunk, &mut output);
    }

    let remainder = chunks.remainder();
    if !remainder.is_empty() {
        encode_group(remainder, &mut output);
    }

    output
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                } else {
                    let group = [self.tail[0], self.tail[1], input[index]];
                    encode_group(&group, &mut output);
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                self.tail_len = 0;
            }
        }

        while input.len() - index >= 3 {
            encode_group(&input[index..index + 3], &mut output);
            index += 3;
        }

        while index < input.len() {
            self.tail[self.tail_len] = input[index];
            self.tail_len += 1;
            index += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        if self.tail_len > 0 {
            encode_group(&self.tail[..self.tail_len], &mut output);
            self.tail_len = 0;
        }
        self.finished = true;
        Ok(output)
    }
}

#[derive(Clone, Copy)]
enum Symbol {
    Digit(u8),
    Padding(usize),
}

pub struct Decoder {
    group: [Symbol; 4],
    group_len: usize,
    pending_padding: Option<usize>,
    terminal_padding: bool,
    offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [Symbol::Digit(0); 4],
            group_len: 0,
            pending_padding: None,
            terminal_padding: false,
            offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) {
        let (a, b, c) = match (self.group[0], self.group[1], self.group[2]) {
            (Symbol::Digit(a), Symbol::Digit(b), Symbol::Digit(c)) => (a, b, c),
            _ => unreachable!(),
        };

        output.push((a << 2) | (b >> 4));

        match self.group[3] {
            Symbol::Digit(d) => {
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
            }
            Symbol::Padding(_) => {
                if matches!(self.group[2], Symbol::Digit(_)) {
                    output.push((b << 4) | (c >> 2));
                }
            }
        }

        self.group_len = 0;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return Err(self.fail(Error::BadPadding {
                    offset: current_offset,
                }));
            }

            if let Some(padding_offset) = self.pending_padding {
                if byte != b'=' {
                    return Err(self.fail(Error::BadPadding {
                        offset: current_offset,
                    }));
                }

                self.group[3] = Symbol::Padding(current_offset);
                self.group_len = 4;
                self.pending_padding = None;
                self.terminal_padding = true;
                self.emit_group(&mut output);
                continue;
            }

            match self.group_len {
                0 | 1 => {
                    if byte == b'=' {
                        return Err(self.fail(Error::BadPadding {
                            offset: current_offset,
                        }));
                    }
                    match digit(byte) {
                        Some(value) => {
                            self.group[self.group_len] = Symbol::Digit(value);
                            self.group_len += 1;
                        }
                        None => {
                            return Err(self.fail(Error::InvalidByte {
                                offset: current_offset,
                                byte,
                            }));
                        }
                    }
                }
                2 => {
                    if byte == b'=' {
                        self.group[2] = Symbol::Padding(current_offset);
                        self.group_len = 3;
                        self.pending_padding = Some(current_offset);
                    } else {
                        match digit(byte) {
                            Some(value) => {
                                self.group[2] = Symbol::Digit(value);
                                self.group_len = 3;
                            }
                            None => {
                                return Err(self.fail(Error::InvalidByte {
                                    offset: current_offset,
                                    byte,
                                }));
                            }
                        }
                    }
                }
                3 => {
                    if byte == b'=' {
                        self.group[3] = Symbol::Padding(current_offset);
                        self.group_len = 4;
                        self.terminal_padding = true;
                        self.emit_group(&mut output);
                    } else {
                        match digit(byte) {
                            Some(value) => {
                                self.group[3] = Symbol::Digit(value);
                                self.group_len = 4;
                                self.emit_group(&mut output);
                            }
                            None => {
                                return Err(self.fail(Error::InvalidByte {
                                    offset: current_offset,
                                    byte,
                                }));
                            }
                        }
                    }
                }
                _ => unreachable!(),
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_padding {
            let error = Error::BadPadding { offset };
            return Err(self.fail(error));
        }

        if self.group_len != 0 {
            let first_offset = match self.group[0] {
                Symbol::Digit(_) => self.offset - self.group_len,
                Symbol::Padding(offset) => offset,
            };
            let error = Error::Truncated {
                offset: first_offset,
            };
            return Err(self.fail(error));
        }

        self.finished = true;
        Ok(Vec::new())
    }
}
