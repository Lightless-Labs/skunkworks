fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    let err = |kind| match kind {
        0 => ParseError::LeadingZero { field, offset: start },
        1 => ParseError::NumberOverflow { field, offset: start },
        2 => ParseError::BadStep { field, offset: start },
        _ => ParseError::OutOfRange { field, offset: start },
    };

    if end - start > 1 && bytes[start] == b'0' {
        return Err(err(0));
    }

    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| err(1))?;
    }

    if is_step {
        let (min, max) = bounds(field);
        let width = max - min + 1;
        if value == 0 || value > width {
            return Err(err(2));
        }
    } else {
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(err(3));
        }
    }

    Ok(value)
}

fn parse_field(input: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (domain_min, domain_max) = bounds(field);
    let mut selected = vec![false; (domain_max - domain_min + 1) as usize];
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map_or(bytes.len(), |n| item_start + n);

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_part) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some(slash + 1)),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let structure = if is_star && hyphens.is_empty() {
            Some((None, None))
        } else if hyphens.is_empty() && !base.is_empty() {
            if base.iter().all(u8::is_ascii_digit) {
                Some((Some((0, base.len())), None))
            } else {
                None
            }
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if !left.is_empty()
                && !right.is_empty()
                && left.iter().all(u8::is_ascii_digit)
                && right.iter().all(u8::is_ascii_digit)
            {
                Some((Some((0, hyphen)), Some((hyphen + 1, base.len()))))
            } else {
                None
            }
        } else {
            None
        };

        let (start_num, end_num) = match structure {
            Some(parts) => parts,
            None => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let step = match step_part {
            Some(start) => {
                if start == item.len() || !item[start..].iter().all(u8::is_ascii_digit) {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                Some(parse_number(item, start, item.len(), field, true)?)
            }
            None => None,
        };

        let (start, end) = if is_star {
            (domain_min, domain_max)
        } else if let Some((start_offset, start_end)) = start_num {
            let start = parse_number(item, start_offset, start_end, field, false)?;
            if let Some((end_offset, end_end)) = end_num {
                let end = parse_number(item, end_offset, end_end, field, false)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + end_offset,
                    });
                }
                (start, end)
            } else if step.is_some() {
                (start, domain_max)
            } else {
                (start, start)
            }
        } else {
            (domain_min, domain_max)
        };

        let step = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[(value - domain_min) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];
    for (i, field) in FIELDS.iter().copied().enumerate() {
        selected[i] = parse_field(pieces[i], field)?;
    }

    Ok(Cron {
        selected,
        day_of_month_star: pieces[2] == "*",
        day_of_week_star: pieces[4] == "*",
    })
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.contains(Field::DayOfMonth, at.day)
            && self.contains(Field::DayOfWeek, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut date = after;
        loop {
            if self.contains(Field::Month, date.month)
                && self.contains(Field::DayOfMonth, date.day)
                && self.contains(Field::DayOfWeek, weekday(date.year, date.month, date.day))
            {
                for hour in 0..=23u8 {
                    if !self.contains(Field::Hour, hour) {
                        continue;
                    }
                    for minute in 0..=59u8 {
                        if !self.contains(Field::Minute, minute) {
                            continue;
                        }
                        let candidate = CivilDateTime {
                            year: date.year,
                            month: date.month,
                            day: date.day,
                            hour,
                            minute,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }

            if !advance_day(&mut date) {
                return Ok(None);
            }
        }
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let (min, max) = bounds(field);
        value >= min as u8
            && value <= max as u8
            && self.selected[field_index(field)][(value - min as u8) as usize]
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn advance_day(date: &mut CivilDateTime) -> bool {
    if date.day < days_in_month(date.year, date.month) {
        date.day += 1;
    } else if date.month < 12 {
        date.month += 1;
        date.day = 1;
    } else if date.year < 2399 {
        date.year += 1;
        date.month = 1;
        date.day = 1;
    } else {
        return false;
    }
    true
}

fn parse_number(

    start: usize,
    field: Field,
    domain: Domain,
) -> Result<u32, ParseError> {
    let mut end = start;
    while end < bytes.len() && bytes[end].is_ascii_digit() {
        end += 1;
    }

    if end == start {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,

            field,
            offset: start,

    Ok(value)

    let (base_kind, first_start, second_start) = if is_star {
        (0u8, None, None)
    } else if hyphens.is_empty() {
        if base.iter().all(u8::is_ascii_digit) && !base.is_empty() {
            (1u8, Some(0usize), None)
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    } else if hyphens.len() == 1 {
        let h = hyphens[0];
        if h > 0
            && h + 1 < base.len()
            && base[..h].iter().all(u8::is_ascii_digit)
            && base[h + 1..].iter().all(u8::is_ascii_digit)
        {
            (2u8, Some(0usize), Some(h + 1))
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    } else {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    };

    if let Some(step_start) = step_start {
        if step_start >= item.len()
            || !item[step_start..].iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    }

    let (start, end) = match base_kind {
        0 => (domain.min, domain.max),
        1 => {
            let start = parse_number(
                base,
                first_start.expect("numeric base"),
                field,
                domain,
            )?;
            let end = if step_start.is_some() {
                domain.max
            } else {
                start
            };
            (start, end)
        }
        _ => {
            let start = parse_number(
                base,
                first_start.expect("range start"),
                field,
                domain,
            )?;
            let end = parse_number(
                base,
                second_start.expect("range end"),
                field,
                domain,
            )?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: second_start.expect("range end"),
                });
            }
            (start, end)
        }
    };

    let step = if let Some(step_start) = step_start {
        parse_step(item, step_start, field, domain)?
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected.insert(value as u8);
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let domain = Domain::for_field(field);
    let mut selected = BTreeSet::new();
    let mut item_start = 0usize;

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(
                &bytes[item_start..index],
                item_start,
                field,
                domain,
                &mut selected,
            )?;
            item_start = index + 1;
        }
    }

    parse_item(
        &bytes[item_start..],
        item_start,
        field,
        domain,
        &mut selected,
    )?;

    Ok(selected.into_iter().collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let minutes = parse_field(parts[0], Field::Minute)?;
    let hours = parse_field(parts[1], Field::Hour)?;
    let day_of_month_is_star = parts[2].as_bytes() == b"*";
    let days_of_month = parse_field(parts[2], Field::DayOfMonth)?;
    let months = parse_field(parts[3], Field::Month)?;
    let day_of_week_is_star = parts[4].as_bytes() == b"*";
    let days_of_week = parse_field(parts[4], Field::DayOfWeek)?;

    Ok(Cron {
        minutes,
        hours,
        days_of_month,
        months,
        days_of_week,
        day_of_month_is_star,
        day_of_week_is_star,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut day_index = 0u32;
    for year in 2000..at.year {
        day_index += if is_leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        day_index += days_in_month(at.year, month) as u32;
    }
    day_index += (at.day - 1) as u32;
    ((day_index + 6) % 7) as u8
}

fn contains(values: &[u8], value: u8) -> bool {
    values.binary_search(&value).is_ok()
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        match field {
            Field::Minute => self.minutes.clone(),
            Field::Hour => self.hours.clone(),
            Field::DayOfMonth => self.days_of_month.clone(),
            Field::Month => self.months.clone(),
            Field::DayOfWeek => self.days_of_week.clone(),
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at);
        Ok(contains(&self.minutes, at.minute)
            && contains(&self.hours, at.hour)
            && contains(&self.months, at.month)
            && contains(&self.days_of_month, at.day)
            && contains(&self.days_of_week, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut year = after.year;
        let mut month = after.month;

        while year <= 2399 {
            if !contains(&self.months, month) {
                if month == 12 {
                    year += 1;
                    month = 1;
                } else {
                    month += 1;
                }
                continue;
            }

            let month_days = days_in_month(year, month);
            for day in 1..=month_days {
                if !contains(&self.days_of_month, day) {
                    continue;
                }
                let date = CivilDateTime {
                    year,
                    month,
                    day,
                    hour: 0,
                    minute: 0,
                };
                if !contains(&self.days_of_week, weekday(date)) {
                    continue;
                }

                for &hour in &self.hours {
                    for &minute in &self.minutes {
                        let candidate = CivilDateTime {
                            year,
                            month,
                            day,
                            hour,
                            minute,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }

            if month == 12 {
                year += 1;
                month = 1;
            } else {
                month += 1;
            }
        }

        Ok(None)
    }
}
