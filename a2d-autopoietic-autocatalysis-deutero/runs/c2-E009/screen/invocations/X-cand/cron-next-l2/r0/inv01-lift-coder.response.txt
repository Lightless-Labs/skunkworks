#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    if step {
        let (_, max) = field_bounds(field);
        if value == 0 || value > max + 1 {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else {
        let (min, max) = field_bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset: start });
        }
    }

    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let item = &bytes[item_start..];
    let slash_count = item.iter().filter(|&&b| b == b'/').count();
    let base_end = item
        .iter()
        .position(|&b| b == b'/')
        .map_or(bytes.len(), |p| item_start + p);

    let base = &bytes[item_start..base_end];
    let hyphen_count = base.iter().filter(|&&b| b == b'-').count();

    let base_kind = if base == b"*" && hyphen_count == 0 {
        0
    } else if hyphen_count == 0 && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
        1
    } else if hyphen_count == 1 {
        let hyphen = item_start + base.iter().position(|&b| b == b'-').unwrap();
        let left = &bytes[item_start..hyphen];
        let right = &bytes[hyphen + 1..base_end];
        if left.is_empty()
            || right.is_empty()
            || !left.iter().all(u8::is_ascii_digit)
            || !right.iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        2
    } else {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    };

    if slash_count > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let mut step_range = None;
    if slash_count == 1 {
        let step_start = base_end + 1;
        if step_start >= bytes.len()
            || !bytes[step_start..].iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        step_range = Some((step_start, bytes.len()));
    }

    let (domain_min, domain_max) = field_bounds(field);
    let (start, end) = match base_kind {
        0 => (domain_min, domain_max),
        1 => {
            let value = parse_number(bytes, item_start, base_end, field, false)?;
            (value, if step_range.is_some() { domain_max } else { value })
        }
        _ => {
            let hyphen = item_start + base.iter().position(|&b| b == b'-').unwrap();
            let start = parse_number(bytes, item_start, hyphen, field, false)?;
            let end = parse_number(bytes, hyphen + 1, base_end, field, false)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: hyphen + 1,
                });
            }
            (start, end)
        }
    };

    let step = if let Some((step_start, step_end)) = step_range {
        parse_number(bytes, step_start, step_end, field, true)?
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - domain_min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = field_bounds(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut start = 0usize;

    for end in 0..=bytes.len() {
        if end == bytes.len() || bytes[end] == b',' {
            if end == start {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: start,
                });
            }
            parse_item(bytes, start, field, &mut selected)?;
            start = end + 1;
        }
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());

    for (index, field) in order.into_iter().enumerate() {
        parsed[index] = parse_field(fields[index], field)?;
    }

    Ok(Cron {
        allowed: parsed,
        day_of_month_is_star: fields[2] == "*",
        day_of_week_is_star: fields[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        let (min, _) = field_bounds(field);
        self.allowed[index]
            .iter()
            .enumerate()
            .filter_map(|(offset, &is_allowed)| {
                is_allowed.then_some(min as u8 + offset as u8)
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;

        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.allowed[0][at.minute as usize]
            && self.allowed[1][at.hour as usize]
            && self.allowed[2][(at.day - 1) as usize]
            && self.allowed[3][(at.month - 1) as usize]
            && self.allowed[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let start = datetime_to_minute_index(after);
        let final_index = datetime_to_minute_index(CivilDateTime {
            year: 2399,
            month: 12,
            day: 31,
            hour: 23,
            minute: 59,
        });

        let mut index = start + 1;
        while index <= final_index {
            let day_index = index / 1440;
            let minute_of_day = index % 1440;
            let (year, month, day) = date_from_day_index(day_index);
            let hour = (minute_of_day / 60) as u8;
            let minute = (minute_of_day % 60) as u8;

            if self.allowed[0][minute as usize] && self.allowed[1][hour as usize] {
                let at = CivilDateTime {
                    year,
                    month,
                    day,
                    hour,
                    minute,
                };
                if self.matches(at)? {
                    return Ok(Some(at));
                }
            }

            index += 1;
        }

        Ok(None)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn datetime_to_minute_index(at: CivilDateTime) -> u64 {
    let mut days = 0u64;
    for year in 2000..at.year {
        days += if is_leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u64;
    }
    days += (at.day - 1) as u64;
    days * 1440 + at.hour as u64 * 60 + at.minute as u64
}

fn date_from_day_index(mut index: u64) -> (u16, u8, u8) {
    let mut year = 2000u16;
    while year < 2400 {
        let year_days = if is_leap_year(year) { 366 } else { 365 };
        if index < year_days {
            break;
        }
        index -= year_days;
        year += 1;
    }

    let mut month = 1u8;
    while month <= 12 {
        let month_days = days_in_month(year, month) as u64;
        if index < month_days {
            break;
        }
        index -= month_days;
        month += 1;
    }

    (year, month, index as u8 + 1)
}
