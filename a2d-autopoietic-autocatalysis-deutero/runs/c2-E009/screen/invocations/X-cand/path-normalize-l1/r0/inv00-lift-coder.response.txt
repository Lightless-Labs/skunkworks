#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/');

    let mut components: Vec<(usize, &str)> = Vec::new();
    let mut last_nonempty: Option<&str> = None;
    let mut start = 0;

    for (index, byte) in path.bytes().enumerate() {
        if byte == b'/' {
            if start < index {
                let component = &path[start..index];
                last_nonempty = Some(component);
                process_component(component, start, absolute, &mut components)?;
            }
            start = index + 1;
        }
    }

    if start < path.len() {
        let component = &path[start..];
        last_nonempty = Some(component);
        process_component(component, start, absolute, &mut components)?;
    }

    let directory_intent =
        directory_intent || matches!(last_nonempty, Some(".") | Some(".."));

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    for (index, (_, component)) in components.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if components.is_empty() {
        if absolute {
            return Ok("/".to_owned());
        }
        return Ok(".".to_owned());
    }

    if directory_intent {
        result.push('/');
    }

    Ok(result)
}

fn process_component<'a>(
    component: &'a str,
    at: usize,
    absolute: bool,
    components: &mut Vec<(usize, &'a str)>,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if components
                .last()
                .is_some_and(|(_, previous)| *previous != "..")
            {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                components.push((at, component));
            }
        }
        _ => components.push((at, component)),
    }

    Ok(())
}
