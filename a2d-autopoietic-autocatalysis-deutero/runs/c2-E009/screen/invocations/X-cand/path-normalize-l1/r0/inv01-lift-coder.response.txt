#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let mut components: Vec<(usize, &str)> = Vec::new();
    let mut last_nonempty: Option<&str> = None;

    for (start, component) in path.split('/') {
        if component.is_empty() {
            continue;
        }

        last_nonempty = Some(component);

        match component {
            "." => {}
            ".." => {
                if components
                    .last()
                    .is_some_and(|(_, previous)| *previous != "..")
                {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push((start, component));
                }
            }
            _ => components.push((start, component)),
        }
    }

    let directory_intent = path.ends_with('/')
        || matches!(last_nonempty, Some(".") | Some(".."));

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    let mut first = true;
    for (_, component) in components {
        if !first {
            result.push('/');
        }
        result.push_str(component);
        first = false;
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}
