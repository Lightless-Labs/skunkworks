pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn validate_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

        let expected = radix.to_string();
        if &bytes[..hash_index] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash_index + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    let mut digits = Vec::new();
    for &byte in payload {
        if byte != b'_' {
            digits.push(digit_value(byte).expect("validated digit"));
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

    for limb in number.iter_mut() {
        let value = (*limb as u16) * radix as u16 + carry;
        *limb = (value % 256) as u8;
        carry = value / 256;
    }
    while carry != 0 {
        number.push((carry % 256) as u8);
        carry /= 256;
    }
}

fn add_numbers(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u16;
    let count = left.len().max(right.len());

    for index in 0..count {
        let a = left.get(index).copied().unwrap_or(0) as u16;
        let b = right.get(index).copied().unwrap_or(0) as u16;
        let sum = a + b + carry;
        result.push((sum % 256) as u8);
        carry = sum / 256;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn divide(number: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for limb in number.iter_mut().rev() {
        let value = remainder * 256 + *limb as u16;
        *limb = (value / divisor as u16) as u8;
        remainder = value % divisor as u16;
    }
    while number.last() == Some(&0) {
        number.pop();
    }
    remainder as u8
}

fn render(mut number: Vec<u8>, style: BaseStyle, radix: u8) -> String {
    let mut digits = Vec::new();
    if number.is_empty() {
        digits.push('0');
    } else {
        while !number.is_empty() {
            let value = divide(&mut number, radix);
            let character = match value {
                0..=9 => (b'0' + value) as char,
                _ if style.upper => (b'A' + value - 10) as char,
                _ => (b'a' + value - 10) as char,
            };
            digits.push(character);
        }
        digits.reverse();
    }

    while digits.len() < style.min_width {
        digits.insert(0, '0');
    }

    if style.group == 0 {
        return digits.into_iter().collect();
    }

    let mut output = String::new();
    let len = digits.len();
    for (index, ch) in digits.into_iter().enumerate() {
        if index != 0 && (len - index) % style.group == 0 {
            output.push('_');
        }
        output.push(ch);
    }
    output
}

    from: u8,
    to: u8,

        multiply_add(&mut right_number, base, digit);

    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn validate_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;
    if let Some(tag_end) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if bytes[..tag_end] != *expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = tag_end + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    // Invalid digits take precedence over separator and leading-zero errors.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte != b'_' && !matches!(digit_value(byte), Some(value) if value < radix) {
            return Err(BaseError::InvalidDigit {
                operand,
                index: payload_start + offset,
            });
        }
    }

    // Then ensure every separator is between two digits.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let digits: Vec<u8> = payload
        .iter()
        .filter_map(|&byte| if byte == b'_' { None } else { digit_value(byte) })
        .collect();

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(number: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for value in number.iter_mut() {
        let total = (*value as u16) * radix as u16 + carry;
        *value = (total % 256) as u8;
        carry = total / 256;
    }
    while carry != 0 {
        number.push((carry % 256) as u8);
        carry /= 256;
    }
}

fn parse_number(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut number = vec![0];
    for &digit in digits {
        multiply_add(&mut number, radix, digit);
    }
    while number.len() > 1 && number.last() == Some(&0) {
        number.pop();
    }
    number
}

fn add_numbers(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut sum = Vec::new();
    let mut carry = 0u16;
    let max_len = left.len().max(right.len());

    for index in 0..max_len {
        let lhs = left.get(index).copied().unwrap_or(0) as u16;
        let rhs = right.get(index).copied().unwrap_or(0) as u16;
        let total = lhs + rhs + carry;
        sum.push((total % 256) as u8);
        carry = total / 256;
    }
    if carry != 0 {
        sum.push(carry as u8);
    }

    while sum.len() > 1 && sum.last() == Some(&0) {
        sum.pop();
    }
    sum
}

fn divide_number(number: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for value in number.iter_mut().rev() {
        let current = remainder * 256 + *value as u16;
        *value = (current / divisor as u16) as u8;
        remainder = current % divisor as u16;
    }
    while number.len() > 1 && number.last() == Some(&0) {
        number.pop();
    }
    remainder as u8
}

fn render(number: &[u8], radix: u8, style: BaseStyle) -> String {
    let mut work = number.to_vec();
    let mut digits = Vec::new();

    while !(work.len() == 1 && work[0] == 0) {
        digits.push(divide_number(&mut work, radix));
    }
    if digits.is_empty() {
        digits.push(0);
    }

    while digits.len() < style.min_width {
        digits.push(0);
    }

    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };

    let mut output = String::new();
    for index in (0..digits.len()).rev() {
        output.push(alphabet[digits[index] as usize] as char);
        if style.group != 0 && index != 0 && index % style.group == 0 {
            output.push('_');
        }
    }
    output
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

    from: u8,
    to: u8,
    style: BaseStyle,

fn validate_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {

        let expected = radix.to_string();

            return Err(BaseError::InvalidTag {
                operand,
                index: 0,
            });
        }
        hash_index + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    // Check all digit bytes before reporting any separator error.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let first_digit = payload.iter().find(|&&byte| byte != b'_').copied();
    let digit_count = payload.iter().filter(|&&byte| byte != b'_').count();
    if digit_count >= 2 && first_digit == Some(b'0') {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

        .iter()
        .filter_map(|&byte| {
            if byte == b'_' {
                None
            } else {
                digit_value(byte)
            }
        })
        .collect())
}

fn multiply_add(number: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for digit in number.iter_mut() {
        let value = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (value % 256) as u8;
        carry = value / 256;
    }
    while carry > 0 {
        number.push((carry % 256) as u8);
        carry /= 256;
    }
}

fn convert_digits_to_base(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut value = Vec::<u8>::new(); // little-endian bytes, base 256
    for &digit in digits {
        multiply_add(&mut value, from, digit);
    }

    if value.is_empty() {
        return vec![0];
    }

    let mut output = Vec::new();
    while value.iter().any(|&byte| byte != 0) {
        let mut remainder = 0u32;
        for byte in value.iter_mut().rev() {
            let current = (remainder << 8) | (*byte as u32);
            *byte = (current / to as u32) as u8;
            remainder = current % to as u32;
        }
        output.push(remainder as u8);
        while value.last() == Some(&0) {
            value.pop();
        }
    }
    output.reverse();
    output
}

fn add_values(left: &[u8], right: &[u8], radix: u8) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u16;
    let count = left.len().max(right.len());
    for index in 0..count {
        let l = left
            .len()
            .checked_sub(index + 1)
            .map(|i| left[i])
            .unwrap_or(0) as u16;
        let r = right
            .len()
            .checked_sub(index + 1)
            .map(|i| right[i])
            .unwrap_or(0) as u16;
        let sum = l + r + carry;
        result.push((sum % radix as u16) as u8);
        carry = sum / radix as u16;
    }
    if carry > 0 {
        result.push(carry as u8);
    }
    if result.is_empty() {
        result.push(0);
    }
    result.reverse();
    result
}

fn format_digits(mut digits: Vec<u8>, style: BaseStyle) -> String {
    if digits.is_empty() {
        digits.push(0);
    }
    if digits.len() < style.min_width {
        let padding = style.min_width - digits.len();
        let mut padded = Vec::new();
        padded.extend(std::iter::repeat(0).take(padding));
        padded.extend(digits);
        digits = padded;
    }

    let mut output = String::new();
    for (index, &digit) in digits.iter().enumerate() {
        if style.group != 0 && index > 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        let byte = if digit < 10 {
            b'0' + digit
        } else if style.upper {
            b'A' + (digit - 10)
        } else {
            b'a' + (digit - 10)
        };
        output.push(byte as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }
    let input_digits = validate_operand(input, from, 0)?;
    let output_digits = convert_digits_to_base(&input_digits, from, to);
    Ok(format_digits(output_digits, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }
    let left_digits = validate_operand(left, base, 0)?;
    let right_digits = validate_operand(right, base, 1)?;
    let sum = add_values(&left_digits, &right_digits, base);
    Ok(format_digits(sum, style))
}
