#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn validate_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let payload_start = if let Some(hash_index) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash_index] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash_index + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    let mut digits = Vec::new();
    for &byte in payload {
        if byte != b'_' {
            digits.push(digit_value(byte).expect("validated digit"));
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(number: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in number.iter_mut() {
        let value = (*limb as u16) * radix as u16 + carry;
        *limb = (value % 256) as u8;
        carry = value / 256;
    }
    while carry != 0 {
        number.push((carry % 256) as u8);
        carry /= 256;
    }
}

fn add_numbers(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::new();
    let mut carry = 0u16;
    let count = left.len().max(right.len());

    for index in 0..count {
        let a = left.get(index).copied().unwrap_or(0) as u16;
        let b = right.get(index).copied().unwrap_or(0) as u16;
        let sum = a + b + carry;
        result.push((sum % 256) as u8);
        carry = sum / 256;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn divide(number: &mut Vec<u8>, divisor: u8) -> u8 {
    let mut remainder = 0u16;
    for limb in number.iter_mut().rev() {
        let value = remainder * 256 + *limb as u16;
        *limb = (value / divisor as u16) as u8;
        remainder = value % divisor as u16;
    }
    while number.last() == Some(&0) {
        number.pop();
    }
    remainder as u8
}

fn render(mut number: Vec<u8>, style: BaseStyle, radix: u8) -> String {
    let mut digits = Vec::new();
    if number.is_empty() {
        digits.push('0');
    } else {
        while !number.is_empty() {
            let value = divide(&mut number, radix);
            let character = match value {
                0..=9 => (b'0' + value) as char,
                _ if style.upper => (b'A' + value - 10) as char,
                _ => (b'a' + value - 10) as char,
            };
            digits.push(character);
        }
        digits.reverse();
    }

    while digits.len() < style.min_width {
        digits.insert(0, '0');
    }

    if style.group == 0 {
        return digits.into_iter().collect();
    }

    let mut output = String::new();
    let len = digits.len();
    for (index, ch) in digits.into_iter().enumerate() {
        if index != 0 && (len - index) % style.group == 0 {
            output.push('_');
        }
        output.push(ch);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = validate_operand(input, from, 0)?;
    let mut number = Vec::new();
    for digit in digits {
        multiply_add(&mut number, from, digit);
    }
    Ok(render(number, style, to))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = validate_operand(left, base, 0)?;
    let right_digits = validate_operand(right, base, 1)?;

    let mut left_number = Vec::new();
    for digit in left_digits {
        multiply_add(&mut left_number, base, digit);
    }

    let mut right_number = Vec::new();
    for digit in right_digits {
        multiply_add(&mut right_number, base, digit);
    }

    Ok(render(add_numbers(&left_number, &right_number), style, base))
}
