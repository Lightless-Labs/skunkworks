    },
    Delete {
        old_index: usize,
        line: String,
    },

        new_index: usize,
        line: String,
    },

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    TooLarge { old_len: usize, new_len: usize },
    EmbeddedNewline { side: Side, index: usize },
}

pub fn diff_lines(old: &[String], new: &[String]) -> Result<Vec<Edit>, DiffError> {
    const MAX_LINES: usize = 128;

    let old_len = old.len();
    let new_len = new.len();

    if old_len > MAX_LINES || new_len > MAX_LINES {
        return Err(DiffError::TooLarge { old_len, new_len });
    }

    for (index, line) in old.iter().enumerate() {
        if line.contains('\n') || line.contains('\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::Old,
                index,
            });
        }
    }

    for (index, line) in new.iter().enumerate() {
        if line.contains('\n') || line.contains('\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::New,
                index,
            });
        }
    }

    // Suffix LCS lengths. The table dimensions are bounded by the validated
    // input limit, and each entry is computed in constant time.
    let mut lcs = [[0usize; MAX_LINES + 1]; MAX_LINES + 1];
    for i in (0..old_len).rev() {
        for j in (0..new_len).rev() {
            lcs[i][j] = if old[i] == new[j] {
                lcs[i + 1][j + 1] + 1
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::new();
    let (mut i, mut j) = (0usize, 0usize);

    while i < old_len && j < new_len {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].clone(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];

            if delete_score > insert_score
                || (delete_score == insert_score && (i + j) % 2 == 0)
            {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].clone(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].clone(),
                });
                j += 1;
            }
        }
    }

    while i < old_len {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].clone(),
        });
        i += 1;
    }

    while j < new_len {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].clone(),
        });
        j += 1;
    }

    Ok(edits)
}
