        }

        let mut result = Vec::new();
        for row in 0..self.rows {
            let mut accumulator = 0.0;
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let column = self.col_idx[slot];
                let product = self.values[slot] * x[column];
                if !product.is_finite() {
                    return Err(Error::ProductOverflow { row, column });
                }
                accumulator = accumulator + product;
                if !accumulator.is_finite() {
                    return Err(Error::AccumulationOverflow { row, column });
                }
            }
            if accumulator == 0.0 {
                accumulator = 0.0;
            }
            result.push(accumulator);
        }
        Ok(result)
    }

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &column in &self.col_idx {
            counts[column] += 1;
        }

        let mut row_ptr = Vec::new();
        row_ptr.push(0);
        for &count in &counts {
            row_ptr.push(row_ptr.last().copied().unwrap_or(0) + count);
        }

        let mut next = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0f64; self.values.len()];

        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let transposed_row = self.col_idx[slot];
                let target = next[transposed_row];
                col_idx[target] = row;
                values[target] = self.values[slot];
                next[transposed_row] += 1;
            }
        }

        Csr {
            rows: self.cols,
            cols: self.rows,
            row_ptr,
            col_idx,
            values,
        }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch {
                left_cols: self.cols,
                right_rows: rhs.rows,
            });
        }

        let mut row_ptr = Vec::new();
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);

        for row in 0..self.rows {
            let mut accumulators: BTreeMap<usize, f64> = BTreeMap::new();
            for left_slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let inner = self.col_idx[left_slot];
                let left_value = self.values[left_slot];
                for right_slot in rhs.row_ptr[inner]..rhs.row_ptr[inner + 1] {
                    let column = rhs.col_idx[right_slot];
                    let product = left_value * rhs.values[right_slot];
                    if !product.is_finite() {
                        return Err(Error::MultiplyOverflow {
                            row,
                            inner,
                            column,
                        });
                    }
                    let accumulator = accumulators.entry(column).or_insert(0.0);
                    *accumulator = *accumulator + product;
                    if !accumulator.is_finite() {
                        return Err(Error::MultiplySumOverflow {
                            row,
                            inner,
                            column,
                        });
                    }
                }
            }

            for (column, value) in accumulators {
                if value != 0.0 {
                    col_idx.push(column);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }

        Ok(Csr {
            rows: self.rows,
            cols: rhs.cols,
            row_ptr,
            col_idx,
            values,
        })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 {
            return Err(Error::Tolerance);
        }

        let mut row_ptr = Vec::new();
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);

        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let value = self.values[slot];
                if value.abs() > tolerance {
                    col_idx.push(self.col_idx[slot]);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }

        Csr {
            rows: self.rows,
            cols: self.cols,
            row_ptr,
            col_idx,
            values,
        }
    }

    pub fn write_text(&self) -> String {
        let mut output = format!("SM1 {} {} {}\n", self.rows, self.cols, self.values.len());
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                output.push_str(&format!(
                    "{} {} {:016x}\n",
                    row,
                    self.col_idx[slot],
                    self.values[slot].to_bits()
                ));
            }
        }
        output
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        if !text.ends_with('\n') {
            let line = 1 + text.bytes().filter(|&byte| byte == b'\n').count();
            return Err(Error::Text {
                line,
                kind: TextError::Newline,
            });
        }

        let body = &text[..text.len() - 1];
        let lines: Vec<&str> = body.split('\n').collect();
        let header = match lines.first() {
            Some(line) => *line,
            None => {
                return Err(Error::Text {
                    line: 1,
                    kind: TextError::Header,
                });
            }
        };

        let header_fields: Vec<&str> = header.split(' ').collect();
        if header_fields.len() != 4
            || header_fields.iter().any(|field| field.is_empty())
            || header_fields[0] != "SM1"
        {
            return Err(Error::Text {
                line: 1,
                kind: TextError::Header,
            });
        }

        let rows = parse_decimal(header_fields[1]).ok_or(Error::Text {
            line: 1,
            kind: TextError::Decimal,
        })?;
        let cols = parse_decimal(header_fields[2]).ok_or(Error::Text {
            line: 1,
            kind: TextError::Decimal,
        })?;
        let nnz = parse_decimal(header_fields[3]).ok_or(Error::Text {
            line: 1,
            kind: TextError::Decimal,
        })?;

        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(Error::Text {
                line: 1,
                kind: TextError::Dimension,
            });
        }

        let record_lines = &lines[1..];
        if nnz != record_lines.len() {
            return Err(Error::Text {
                line: 1,
                kind: TextError::Count,
            });
        }

        let mut row_ptr = Vec::new();
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);

        let mut current_row = 0usize;
        let mut previous_coordinate: Option<(usize, usize)> = None;

        for (index, line) in record_lines.iter().enumerate() {
            let line_number = index + 2;
            let fields: Vec<&str> = line.split(' ').collect();
            if fields.len() != 3 || fields.iter().any(|field| field.is_empty()) {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::Record,
                });
            }

            let row = parse_decimal(fields[0]).ok_or(Error::Text {
                line: line_number,
                kind: TextError::Decimal,
            })?;
            let column = parse_decimal(fields[1]).ok_or(Error::Text {
                line: line_number,
                kind: TextError::Decimal,
            })?;

            let bits_token = fields[2].as_bytes();
            if bits_token.len() != 16
                || !bits_token
                    .iter()
                    .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(byte))
            {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::Hex,
                });
            }
            let bits = u64::from_str_radix(fields[2], 16).map_err(|_| Error::Text {
                line: line_number,
                kind: TextError::Hex,
            })?;
            let value = f64::from_bits(bits);

            if row >= rows {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::RowBounds,
                });
            }
            if column >= cols {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::ColumnBounds,
                });
            }
            if !value.is_finite() {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::NonFinite,
                });
            }
            if value == 0.0 {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::Zero,
                });
            }

            if let Some(previous) = previous_coordinate {
                if (row, column) <= previous {
                    return Err(Error::Text {
                        line: line_number,
                        kind: TextError::Order,
                    });
                }
            }
            previous_coordinate = Some((row, column));

            while current_row < row {
                row_ptr.push(col_idx.len());
                current_row += 1;
            }
            col_idx.push(column);
            values.push(value);
        }

        while current_row < rows {
            row_ptr.push(col_idx.len());
            current_row += 1;
        }

        Ok(Csr {
            rows,
            cols,
            row_ptr,
            col_idx,
            values,
        })
    }
}

fn parse_decimal(token: &str) -> Option<usize> {
    let bytes = token.as_bytes();
    if bytes.is_empty() {
        return None;
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return None;
    }
    if !bytes.iter().all(u8::is_ascii_digit) {
        return None;
    }

    let mut value = 0usize;
    for &byte in bytes {
        value = value.checked_mul(10)?;
        value = value.checked_add((byte - b'0') as usize)?;
    }
    Some(value)
}
