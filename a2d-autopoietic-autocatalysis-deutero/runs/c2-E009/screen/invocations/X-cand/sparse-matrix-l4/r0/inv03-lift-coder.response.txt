pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        check_shape(rows, cols)?;

        for (i, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Row,
                    index: entry.row,
                    limit: rows,
                });
            }
            if entry.col >= cols {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Column,
                    index: entry.col,
                    limit: cols,
                });
            }
            if !entry.value.is_finite() {
                return Err(Error::NonFinite { entry: i });
            }
        }

        let mut sums: Vec<((usize, usize), f64)> = Vec::new();
        for (i, entry) in entries.iter().enumerate() {
            let key = (entry.row, entry.col);
            let position = match sums.binary_search_by_key(&key, |(coord, _)| *coord) {
                Ok(pos) => pos,
                Err(pos) => {
                    sums.insert(pos, (key, 0.0));
                    pos
                }
            };
            let updated = sums[position].1 + entry.value;
            if !updated.is_finite() {
                return Err(Error::SumOverflow { entry: i });
            }
            sums[position].1 = updated;
        }

        let canonical = sums
            .into_iter()
            .filter_map(|((row, col), value)| {
                if value == 0.0 {
                    None
                } else {
                    Some(Entry { row, col, value })
                }
            })
            .collect();

        Ok(Self {
            rows,
            cols,
            entries: canonical,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}

#[derive(Clone, Debug, PartialEq)]
pub struct Csr {
    rows: usize,
    cols: usize,
    row_ptr: Vec<usize>,
    col_idx: Vec<usize>,
    values: Vec<f64>,
}

impl Sparse {
    pub fn to_csr(&self) -> Csr {
        let mut row_ptr = Vec::new();
        row_ptr.push(0);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        let mut entry_index = 0;

        for row in 0..self.rows {
            while entry_index < self.entries.len() && self.entries[entry_index].row == row {
                col_idx.push(self.entries[entry_index].col);
                values.push(self.entries[entry_index].value);
                entry_index += 1;
            }
            row_ptr.push(values.len());
        }

        Csr {
            rows: self.rows,
            cols: self.cols,
            row_ptr,
            col_idx,
            values,
        }
    }
}

impl Csr {
    pub fn from_parts(
        rows: usize,
        cols: usize,
        row_ptr: Vec<usize>,
        col_idx: Vec<usize>,
        values: Vec<f64>,
    ) -> Result<Self, Error> {
        check_shape(rows, cols)?;

        let expected_ptr_len = rows + 1;
        if row_ptr.len() != expected_ptr_len {
            return Err(Error::RowPtrLength {
                expected: expected_ptr_len,
                actual: row_ptr.len(),
            });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength {
                columns: col_idx.len(),
                values: values.len(),
            });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });
        }
        let expected_end = values.len();
        if row_ptr[rows] != expected_end {
            return Err(Error::RowPtrEnd {
                found: row_ptr[rows],
                expected: expected_end,
            });
        }

        for row in 0..rows {
            if row_ptr[row] > row_ptr[row + 1] {
                return Err(Error::RowPtrOrder { row });
            }
        }

        for row in 0..rows {
            let start = row_ptr[row];
            let end = row_ptr[row + 1];
            let mut previous: Option<usize> = None;
            for slot in start..end {
                let column = col_idx[slot];
                if column >= cols {
                    return Err(Error::ColumnBounds {
                        slot,
                        column,
                        limit: cols,
                    });
                }
                if let Some(prev) = previous {
                    if column <= prev {
                        return Err(Error::ColumnOrder { slot });
                    }
                }
                let value = values[slot];
                if !value.is_finite() {
                    return Err(Error::NonFiniteValue { slot });
                }
                if value == 0.0 {
                    return Err(Error::ExplicitZero { slot });
                }
                previous = Some(column);
            }
        }

        Ok(Self {
            rows,
            cols,
            row_ptr,
            col_idx,
            values,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn row_ptr(&self) -> &[usize] {
        &self.row_ptr
    }

    pub fn col_indices(&self) -> &[usize] {
        &self.col_idx
    }

    pub fn values(&self) -> &[f64] {
        &self.values
    }

    pub fn to_coo(&self) -> Sparse {
        let mut entries = Vec::new();
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                entries.push(Entry {
                    row,
                    col: self.col_idx[slot],
                    value: self.values[slot],
                });
            }
        }
        Sparse {
            rows: self.rows,
            cols: self.cols,
            entries,
        }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength {
                expected: self.cols,
                actual: x.len(),
            });
        }
        for (index, value) in x.iter().enumerate() {
            if !value.is_finite() {
                return Err(Error::NonFiniteVector { index });
            }
        }

        let mut result = Vec::new();
        for row in 0..self.rows {
            let mut sum = 0.0;
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let column = self.col_idx[slot];
                let product = self.values[slot] * x[column];
                if !product.is_finite() {
                    return Err(Error::ProductOverflow { row, column });
                }
                let updated = sum + product;
                if !updated.is_finite() {
                    return Err(Error::AccumulationOverflow { row, column });
                }
                sum = updated;
            }
            if sum == 0.0 {
                sum = 0.0;
            }
            result.push(sum);
        }
        Ok(result)
    }

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &column in &self.col_idx {
            counts[column] += 1;
        }

        let mut row_ptr = Vec::new();
        row_ptr.push(0);
        for count in &counts {
            row_ptr.push(row_ptr[row_ptr.len() - 1] + *count);
        }

        let mut next = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0f64; self.values.len()];

        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let transposed_row = self.col_idx[slot];
                let target = next[transposed_row];
                col_idx[target] = row;
                values[target] = self.values[slot];
                next[transposed_row] += 1;
            }
        }

        Csr {
            rows: self.cols,
            cols: self.rows,
            row_ptr,
            col_idx,
            values,
        }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch {
                left_cols: self.cols,
                right_rows: rhs.rows,
            });
        }

        let mut row_ptr = Vec::new();
        row_ptr.push(0);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();

        for row in 0..self.rows {
            let mut accumulators: Vec<(usize, f64)> = Vec::new();
            for left_slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let inner = self.col_idx[left_slot];
                let left_value = self.values[left_slot];
                for right_slot in rhs.row_ptr[inner]..rhs.row_ptr[inner + 1] {
                    let column = rhs.col_idx[right_slot];
                    let product = left_value * rhs.values[right_slot];
                    if !product.is_finite() {
                        return Err(Error::MultiplyOverflow {
                            row,
                            inner,
                            column,
                        });
                    }

                    let position = match accumulators
                        .binary_search_by_key(&column, |(col, _)| *col)
                    {
                        Ok(pos) => pos,
                        Err(pos) => {
                            accumulators.insert(pos, (column, 0.0));
                            pos
                        }
                    };
                    let updated = accumulators[position].1 + product;
                    if !updated.is_finite() {
                        return Err(Error::MultiplySumOverflow {
                            row,
                            inner,
                            column,
                        });
                    }
                    accumulators[position].1 = updated;
                }
            }

            for (column, value) in accumulators {
                if value != 0.0 {
                    col_idx.push(column);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }

        Ok(Csr {
            rows: self.rows,
            cols: rhs.cols,
            row_ptr,
            col_idx,
            values,
        })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 {
            return Err(Error::Tolerance);
        }

        let mut row_ptr = Vec::new();
        row_ptr.push(0);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();

        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let value = self.values[slot];
                if value.abs() > tolerance {
                    col_idx.push(self.col_idx[slot]);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }

        Ok(Csr {
            rows: self.rows,
            cols: self.cols,
            row_ptr,
            col_idx,
            values,
        })
    }

    pub fn write_text(&self) -> String {
        let mut output = format!("SM1 {} {} {}\n", self.rows, self.cols, self.values.len());
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                output.push_str(&format!(
                    "{} {} {:016x}\n",
                    row,
                    self.col_idx[slot],
                    self.values[slot].to_bits()
                ));
            }
        }
        output
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        if !text.ends_with('\n') {
            let line = 1 + text.bytes().filter(|&byte| byte == b'\n').count();
            return Err(Error::Text {
                line,
                kind: TextError::Newline,
            });
        }

        let without_final_lf = &text[..text.len() - 1];
        let lines: Vec<&str> = without_final_lf.split('\n').collect();
        let header = lines[0];
        let fields: Vec<&str> = header.split(' ').collect();
        if fields.len() != 4 || fields.iter().any(|field| field.is_empty()) || fields[0] != "SM1" {
            return Err(Error::Text {
                line: 1,
                kind: TextError::Header,
            });
        }

        let rows = parse_decimal(fields[1]).ok_or(Error::Text {
            line: 1,
            kind: TextError::Decimal,
        })?;
        let cols = parse_decimal(fields[2]).ok_or(Error::Text {
            line: 1,
            kind: TextError::Decimal,
        })?;
        let nnz = parse_decimal(fields[3]).ok_or(Error::Text {
            line: 1,
            kind: TextError::Decimal,
        })?;

        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(Error::Text {
                line: 1,
                kind: TextError::Dimension,
            });
        }

        let record_count = lines.len() - 1;
        if nnz != record_count {
            return Err(Error::Text {
                line: 1,
                kind: TextError::Count,
            });
        }

        let mut row_ptr = vec![0usize; rows + 1];
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        let mut previous: Option<(usize, usize)> = None;

        for (offset, line) in lines.iter().skip(1).enumerate() {
            let line_number = offset + 2;
            let fields: Vec<&str> = line.split(' ').collect();
            if fields.len() != 3 || fields.iter().any(|field| field.is_empty()) {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::Record,
                });
            }

            let row = parse_decimal(fields[0]).ok_or(Error::Text {
                line: line_number,
                kind: TextError::Decimal,
            })?;
            let column = parse_decimal(fields[1]).ok_or(Error::Text {
                line: line_number,
                kind: TextError::Decimal,
            })?;
            let bits = parse_hex(fields[2]).ok_or(Error::Text {
                line: line_number,
                kind: TextError::Hex,
            })?;
            let value = f64::from_bits(bits);

            if row >= rows {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::RowBounds,
                });
            }
            if column >= cols {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::ColumnBounds,
                });
            }
            if !value.is_finite() {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::NonFinite,
                });
            }
            if value == 0.0 {
                return Err(Error::Text {
                    line: line_number,
                    kind: TextError::Zero,
                });
            }
            let coordinate = (row, column);
            if let Some(prev) = previous {
                if coordinate <= prev {
                    return Err(Error::Text {
                        line: line_number,
                        kind: TextError::Order,
                    });
                }
            }
            previous = Some(coordinate);
            col_idx.push(column);
            values.push(value);
            row_ptr[row + 1] += 1;
        }

        for row in 0..rows {
            row_ptr[row + 1] += row_ptr[row];
        }

        Ok(Csr {
            rows,
            cols,
            row_ptr,
            col_idx,
            values,
        })
    }
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum TextError {
    Newline,
    Header,
    Decimal,
    Dimension,
    Count,
    Record,
    Hex,
    RowBounds,
    ColumnBounds,
    NonFinite,
    Zero,
    Order,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },

    RowPtrLength {
        expected: usize,
        actual: usize,
    },
    ColumnValueLength {
        columns: usize,
        values: usize,
    },
    RowPtrStart {
        found: usize,
    },
    RowPtrEnd {
        found: usize,
        expected: usize,
    },
    RowPtrOrder {
        row: usize,
    },
    ColumnBounds {
        slot: usize,
        column: usize,
        limit: usize,
    },
    ColumnOrder {
        slot: usize,
    },
    NonFiniteValue {
        slot: usize,
    },
    ExplicitZero {
        slot: usize,
    },
    VectorLength {
        expected: usize,
        actual: usize,
    },
    NonFiniteVector {
        index: usize,
    },
    ProductOverflow {
        row: usize,
        column: usize,
    },
    AccumulationOverflow {
        row: usize,
        column: usize,
    },

    DimensionMismatch {
        left_cols: usize,
        right_rows: usize,
    },
    MultiplyOverflow {
        row: usize,
        inner: usize,
        column: usize,
    },
    MultiplySumOverflow {
        row: usize,
        inner: usize,
        column: usize,
    },

    Tolerance,
    Text {
        line: usize,
        kind: TextError,
    },
}

fn check_shape(rows: usize, cols: usize) -> Result<(), Error> {
    if rows > MAX_DIM {
        return Err(Error::Shape {
            axis: Axis::Row,
            size: rows,
        });
    }
    if cols > MAX_DIM {
        return Err(Error::Shape {
            axis: Axis::Column,
            size: cols,
        });
    }
    Ok(())
}

fn parse_decimal(token: &str) -> Option<usize> {
    let bytes = token.as_bytes();
    if bytes.is_empty() || (bytes.len() > 1 && bytes[0] == b'0') {
        return None;
    }
    let mut value = 0usize;
    for &byte in bytes {
        if !byte.is_ascii_digit() {
            return None;
        }
        value = value.checked_mul(10)?.checked_add((byte - b'0') as usize)?;
    }
    Some(value)
}

fn parse_hex(token: &str) -> Option<u64> {
    let bytes = token.as_bytes();
    if bytes.len() != 16 {
        return None;
    }
    let mut value = 0u64;
    for &byte in bytes {
        let digit = match byte {
            b'0'..=b'9' => (byte - b'0') as u64,
            b'a'..=b'f' => (byte - b'a' + 10) as u64,
            _ => return None,
        };
        value = (value << 4) | digit;
    }
    Some(value)
}
