#[derive(Clone, Debug, PartialEq, Eq)]

    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

const BASE: u64 = 1_000_000_000;
const POW_LIMIT: u32 = 1024;

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
            b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
            _ => None,
        }
    }

    fn mul_small_abs(&mut self, multiplier: u32) {
        if multiplier == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = (*limb as u64) * (multiplier as u64) + carry;
            *limb = (value % BASE) as u32;
            carry = value / BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
    }

    fn add_small_abs(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[i] as u64 + carry;
            self.limbs[i] = (value % BASE) as u32;
            carry = value / BASE;
            i += 1;
        }
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        order => return order,
                    }
                }
                Ordering::Equal
            }
            order => order,
        }
    }

    // Requires |self| >= |rhs| and both values nonnegative.
    fn sub_abs_assign(&mut self, rhs: &Self) {
        let mut borrow = 0i64;
        for i in 0..self.limbs.len() {
            let right = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut value = self.limbs[i] as i64 - right - borrow;
            if value < 0 {
                value += BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            self.limbs[i] = value as u32;
        }
        self.normalize();
    }

    fn add_abs(lhs: &Self, rhs: &Self) -> Self {
        let mut result = Self {
            negative: false,
            limbs: Vec::new(),
        };
        let count = lhs.limbs.len().max(rhs.limbs.len());
        let mut carry = 0u64;

                + carry;
            result.limbs.push((value % BASE) as u32);
            carry = value / BASE;
        }
        if carry != 0 {
            result.limbs.push(carry as u32);
        }
        result
    }

            return Self::zero();
        }
        let mut result = Self {
            negative: false,
            limbs: vec![0; lhs.limbs.len() + rhs.limbs.len()],

                let value = result.limbs[k] as u64 + (a as u64) * (b as u64) + carry;
                result.limbs[k] = (value % BASE) as u32;
                carry = value / BASE;

                result.limbs[k] = (value % BASE) as u32;
                carry = value / BASE;
                k += 1;
            }
        }
        result.normalize();
        result
    }

    fn div_rem_abs(dividend: &Self, divisor: &Self) -> (Self, Self) {
        debug_assert!(!divisor.is_zero());
        if dividend.cmp_abs(divisor) == Ordering::Less {
            return (Self::zero(), dividend.abs());
        }

        let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
        let mut remainder = Self::zero();
        for &limb in dividend.limbs.iter().rev() {
            if !remainder.is_zero() {
                remainder.limbs.insert(0, limb);
            } else if limb != 0 {
                remainder.limbs.push(limb);
            }
            remainder.normalize();

            let mut low = 0u64;
            let mut high = BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = divisor.mul_small_abs_copy(mid as u32);
                if product.cmp_abs(&remainder) != Ordering::Greater {
                    best = mid;
                    low = mid + 1;
                } else {
                    if mid == 0 {
                        break;
                    }
                    high = mid - 1;
                }
            }
            quotient_be.push(best as u32);
            if best != 0 {
                let product = divisor.mul_small_abs_copy(best as u32);
                remainder.sub_abs_assign(&product);
            }
        }

        let mut quotient = Self {
            negative: false,
            limbs: quotient_be.into_iter().rev().collect(),
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn mul_small_abs_copy(&self, multiplier: u32) -> Self {
        let mut result = self.abs();
        result.mul_small_abs(multiplier);
        result
    }

    fn from_decimal_parts(negative: bool, digits: &[u8], radix: u32) -> Self {
        let mut result = Self::zero();
        for &digit in digits {
            result.mul_small_abs(radix);
            result.add_small_abs(digit as u32);
        }
        result.negative = negative && !result.is_zero();
        result
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && Self::digit_value(byte).is_none_or(|digit| digit >= radix) {
                invalid = Some(index);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_' {
                let before_valid = index > start
                    && bytes[index - 1] != b'_'
                    && Self::digit_value(bytes[index - 1]).is_some_and(|d| d < radix);
                let after_valid = index + 1 < bytes.len()
                    && bytes[index + 1] != b'_'
                    && Self::digit_value(bytes[index + 1]).is_some_and(|d| d < radix);
                if !before_valid || !after_valid {
                    separator = Some(index);
                    break;
                }
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&byte| byte != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&index| bytes[index] != b'_').unwrap();
        let first_digit = Self::digit_value(bytes[first_digit_index]).unwrap();
        if digit_count > 1 && first_digit == 0 {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }
        if digit_count > max_digits {
            let excess_number = max_digits;
            let mut seen = 0usize;
            let mut index = first_digit_index;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    if seen == excess_number {
                        index = i;
                        break;
                    }
                    seen += 1;
                }
            }
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let digits: Vec<u8> = bytes[start..]
            .iter()
            .filter_map(|&byte| {
                if byte == b'_' {
                    None
                } else {
                    Self::digit_value(byte)
                        .map(|digit| Some(digit as u8))
                        .unwrap_or(None)
                }
            })
            .collect();
        Ok(Self::from_decimal_parts(negative, &digits, radix))
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut value = self.abs();
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(0u8);
        } else {
            while !value.is_zero() {
                let (quotient, remainder) = value.div_rem_small(radix);
                digits.push(remainder as u8);
                value = quotient;
            }
            digits.reverse();
        }

        if digits.len() < min_digits {
            let padding = min_digits - digits.len();
            let mut padded = Vec::new();
            padded.resize(padding, 0);
            padded.extend(digits);
            digits = padded;
        }

        let mut output = String::new();
        if self.negative && !self.is_zero() {
            output.push('-');
        }
        for (index, digit) in digits.iter().enumerate() {
            if group > 0 && index > 0 && (digits.len() - index) % group == 0 {
                output.push('_');
            }
            output.push(match *digit {
                0..=9 => (b'0' + *digit) as char,
                _ => (b'a' + (*digit - 10)) as char,
            });
        }
        Ok(output)
    }

    fn div_rem_small(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = Self::zero();
        quotient.limbs.resize(self.limbs.len(), 0);
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let value = remainder * BASE + self.limbs[i] as u64;
            quotient.limbs[i] = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        quotient.normalize();
        (quotient, remainder as u32)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative;
            result.normalize();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {

                    result.sub_abs_assign(&rhs.abs());
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = rhs.abs();
                    result.sub_abs_assign(&self.abs());
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

        self.add(&rhs.neg())
    }

        if !result.is_zero() {

        result
    }

        } else if self.negative {

        } else {

            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = remainder.abs();
        }
        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs. Zero is represented by [0].
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const EXPONENT_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(&self, rhs: &Self) -> Self {
        let mut limbs = Vec::new();
        let n = self.limbs.len().max(rhs.limbs.len());
        let mut carry = 0u64;
        for i in 0..n {
            let a = self.limbs.get(i).copied().unwrap_or(0) as u64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = a + b + carry;
            limbs.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            limbs.push(carry as u32);
        }
        Self {
            negative: false,
            limbs,
        }
    }

    // Requires |self| >= |rhs|.
    fn sub_abs(&self, rhs: &Self) -> Self {
        let mut limbs = Vec::new();
        let mut borrow = 0i64;
        for i in 0..self.limbs.len() {
            let a = self.limbs[i] as i64 - borrow;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            if a < b {
                limbs.push((a + Self::BASE as i64 - b) as u32);
                borrow = 1;
            } else {
                limbs.push((a - b) as u32);
                borrow = 0;
            }
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    fn mul_small_abs(&self, factor: u32) -> Self {
        if factor == 0 || self.is_zero() {
            return Self::zero();
        }
        let mut limbs = Vec::new();
        let mut carry = 0u64;
        for &limb in &self.limbs {
            let product = limb as u64 * factor as u64 + carry;
            limbs.push((product % Self::BASE) as u32);
            carry = product / Self::BASE;
        }
        while carry != 0 {
            limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
        Self {
            negative: false,
            limbs,
        }
    }

    fn add_small_abs(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = vec![0; self.limbs.len()];
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let current = remainder * Self::BASE + self.limbs[i] as u64;
            quotient[i] = (current / divisor as u64) as u32;
            remainder = current % divisor as u64;
        }
        let mut q = Self {
            negative: false,
            limbs: quotient,
        };
        q.normalize();
        (q, remainder as u32)
    }

    fn mul_abs(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value =
                    limbs[index] as u64 + a as u64 * b as u64 + carry;
                limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == limbs.len() {
                    limbs.push(0);
                }
                let value = limbs[index] as u64 + carry;
                limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                index += 1;
            }
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    fn div_rem_abs(&self, divisor: &Self) -> (Self, Self) {
        if self.cmp_abs(divisor) == Ordering::Less {
            let mut remainder = self.clone();
            remainder.negative = false;
            return (Self::zero(), remainder);
        }

        let mut quotient_be = Vec::new();
        let mut remainder = Self::zero();

        for &limb in self.limbs.iter().rev() {
            if remainder.is_zero() {
                remainder.limbs[0] = limb;
            } else {
                remainder.limbs.insert(0, limb);
            }
            remainder.normalize();

            let mut low = 0u32;
            let mut high = (Self::BASE - 1) as u32;
            let mut best = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = divisor.mul_small_abs(mid);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        if mid == u32::MAX {
                            break;
                        }
                        low = mid + 1;
                    }
                }
            }

            if best != 0 {
                let product = divisor.mul_small_abs(best);
                remainder = remainder.sub_abs(&product);
            }
            quotient_be.push(best);
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
            b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && Self::digit_value(byte).is_none_or(|d| d >= radix) {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut first_bad_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > start
                    && Self::digit_value(bytes[i - 1]).is_some_and(|d| d < radix);
                let right_ok = i + 1 < bytes.len()
                    && Self::digit_value(bytes[i + 1]).is_some_and(|d| d < radix);
                if (!left_ok || !right_ok) && first_bad_separator.is_none() {
                    first_bad_separator = Some(i);
                }
            }
        }
        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        let first_digit_index = start;
        let first_digit = Self::digit_value(bytes[first_digit_index]).unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        if digit_count > 1 && first_digit == 0 {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }
        if digit_count > max_digits {
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    seen += 1;
                    if seen > max_digits {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            let digit = Self::digit_value(byte).unwrap();
            value = value.mul_small_abs(radix);
            value.add_small_abs(digit);
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.clone();
        value.negative = false;
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(0u32);
        } else {
            while !value.is_zero() {
                let (quotient, remainder) = value.div_rem_small_abs(radix);
                digits.push(remainder);
                value = quotient;
            }
            digits.reverse();
        }

        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut output = String::new();
        if self.negative && !self.is_zero() {
            output.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                output.push('_');
            }
            let byte = if *digit < 10 {
                b'0' + *digit as u8
            } else {
                b'a' + (*digit - 10) as u8
            };
            output.push(byte as char);
        }
        Ok(output)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = self.add_abs(rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = self.sub_abs(rhs);
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = rhs.sub_abs(self);
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut result = self.mul_abs(rhs);
        result.negative = (self.negative != rhs.negative) && !result.is_zero();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = self.div_rem_abs(rhs);
        quotient.negative =
            (self.negative != rhs.negative) && !quotient.is_zero();
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::EXPONENT_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::EXPONENT_LIMIT,
            });
        }
        let mut result = Self::one();
        let mut base = self.clone();
        let mut power = exponent;
        while power != 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, remainder) = a.div_rem_abs(&b);
            a = b;
            b = remainder;
        }
        a.negative = false;
        a
    }
}

        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add(a: &Self, b: &Self) -> Self {
        let mut result = Vec::new();
        let mut carry = 0u64;
        let count = a.limbs.len().max(b.limbs.len());
        for i in 0..count {
            let av = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = av + bv + carry;
            result.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            result.push(carry as u32);
        }
        let mut n = Self {
            negative: false,
            limbs: result,
        };
        n.normalize();
        n
    }

    // Requires |a| >= |b|.
    fn abs_sub(a: &Self, b: &Self) -> Self {
        let mut result = Vec::new();
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let av = a.limbs[i] as i64 - borrow;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as i64;
            if av < bv {
                result.push((av + Self::BASE as i64 - bv) as u32);
                borrow = 1;
            } else {
                result.push((av - bv) as u32);
                borrow = 0;
            }
        }
        let mut n = Self {
            negative: false,
            limbs: result,
        };
        n.normalize();
        n
    }

    fn abs_mul_small(&self, multiplier: u32) -> Self {
        if multiplier == 0 || self.is_zero() {
            return Self::zero();
        }
        let mut result = Vec::new();
        let mut carry = 0u64;
        for &limb in &self.limbs {
            let product = limb as u64 * multiplier as u64 + carry;
            result.push((product % Self::BASE) as u32);
            carry = product / Self::BASE;
        }
        while carry != 0 {
            result.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
        Self {
            negative: false,
            limbs: result,
        }
    }

    fn abs_add_small(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn abs_div_rem(&self, divisor: &Self) -> (Self, Self) {
        if self.abs_cmp(divisor) == Ordering::Less {
            let mut remainder = self.clone();
            remainder.negative = false;
            return (Self::zero(), remainder);
        }

        // Long division in base 1e9. Each quotient limb is found by binary search.
        let mut quotient_be = Vec::new();
        let mut remainder = Self::zero();
        for &limb in self.limbs.iter().rev() {
            if !remainder.is_zero() {
                remainder.limbs.insert(0, limb);
            } else if limb != 0 {
                remainder.limbs.push(limb);
            }
            remainder.normalize();

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            while low < high {
                let mid = (low + high + 1) / 2;
                let product = divisor.abs_mul_small(mid as u32);
                if product.abs_cmp(&remainder) != Ordering::Greater {
                    low = mid;
                } else {
                    high = mid - 1;
                }
            }
            quotient_be.push(low as u32);
            if low != 0 {
                let product = divisor.abs_mul_small(low as u32);
                remainder = Self::abs_sub(&remainder, &product);
            }
        }

        let mut quotient = Self {
            negative: false,
            limbs: quotient_be.into_iter().rev().collect(),
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {

            b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
            b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
            _ => None,
        }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            match Self::digit_value(bytes[i]) {
                Some(d) if d < radix => {
                    digit_count += 1;
                    if first_digit.is_none() {
                        first_digit = Some((i, d));
                    }
                    if digit_count == max_digits.saturating_add(1) {
                        first_excess = Some(i);
                    }
                }
                _ => {
                    if invalid.is_none() {
                        invalid = Some(i);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

                    && Self::digit_value(bytes[i + 1]).is_some_and(|d| d < radix);
                if !prev_ok || !next_ok {
                    return Err(Error::Separator { index: i });
                }

            return Err(Error::LeadingZero {
                index: first_digit.unwrap().0,
            });
        }

            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap_or(bytes.len()),
            });
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            let d = Self::digit_value(byte).unwrap();
            value = value.abs_mul_small(radix);
            value.abs_add_small(d);
        }
        value.negative = negative && !value.is_zero();

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });

        let mut value = self.clone();
        value.negative = false;
        if value.is_zero() {
            digits.push('0');
        } else {
            while !value.is_zero() {
                let (q, r) = value.div_rem_small(radix);
                digits.push(Self::digit_char(r));
                value = q;
            }
            digits.reverse();

            digits.insert(0, '0');
        }

        if self.negative && !self.is_zero() {
            result.push('-');

            result.push(*ch);

        Ok(result)
    }

    fn div_rem_small(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = Vec::new();
        let mut remainder = 0u64;
        for &limb in self.limbs.iter().rev() {
            let current = remainder * Self::BASE + limb as u64;
            quotient.push((current / divisor as u64) as u32);
            remainder = current % divisor as u64;
        }
        quotient.reverse();
        let mut q = Self {
            negative: false,
            limbs: quotient,
        };
        q.normalize();
        (q, remainder as u32)
    }

    fn digit_char(digit: u32) -> char {
        if digit < 10 {
            (b'0' + digit as u8) as char
        } else {
            (b'a' + (digit - 10) as u8) as char
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::abs_add(self, rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::abs_sub(self, rhs);
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = Self::abs_sub(rhs, self);
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let current = result[index] as u64 + a as u64 * b as u64 + carry;
                result[index] = (current % Self::BASE) as u32;
                carry = current / Self::BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == result.len() {
                    result.push(0);
                }
                let current = result[index] as u64 + carry;
                result[index] = (current % Self::BASE) as u32;
                carry = current / Self::BASE;
                index += 1;
            }
        }
        let mut n = Self {
            negative: self.negative != rhs.negative,
            limbs: result,
        };
        n.normalize();
        n
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = self.abs_div_rem(rhs);
        quotient.negative = !quotient.is_zero() && (self.negative != rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }
        let mut result = Self::one();
        let mut base = self.clone();
        let mut power = exponent;
        while power != 0 {
            if power & 1 != 0 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, mut r) = a.abs_div_rem(&b);
            r.negative = false;
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
}

    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {

        self.limbs.is_empty()

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add(a: &Self, b: &Self) -> Vec<u32> {
        let n = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::new();
        let mut carry = 0u64;
        for i in 0..n {
            let av = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = av + bv + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires |a| >= |b|.
    fn abs_sub(a: &Self, b: &Self) -> Vec<u32> {
        let mut out = Vec::new();
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let av = a.limbs[i] as i64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut d = av - bv - borrow;
            if d < 0 {
                d += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(d as u32);
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn from_parts(negative: bool, limbs: Vec<u32>) -> Self {
        let mut value = Self { negative, limbs };
        value.normalize();
        value
    }

    fn mul_small_add(&mut self, multiplier: u32, addend: u32) {
        let mut carry = addend as u64;
        for limb in &mut self.limbs {
            let value = (*limb as u64) * (multiplier as u64) + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let value = rem * Self::BASE + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            rem = value % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let digit_value = |b: u8| -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((b - b'A') as u32 + 10),
                _ => None,
            }
        };

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            if digit_value(byte).filter(|&d| d < radix).is_none() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut first_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let before_ok = i > start
                    && digit_value(bytes[i - 1]).is_some_and(|d| d < radix);
                let after_ok = i + 1 < bytes.len()
                    && digit_value(bytes[i + 1]).is_some_and(|d| d < radix);
                if (!before_ok || !after_ok) && first_separator.is_none() {
                    first_separator = Some(i);
                }
            }
        }
        if let Some(index) = first_separator {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_value = 0;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' {
                let d = digit_value(byte).expect("validated digit");
                if first_digit.is_none() {
                    first_digit = Some(index);
                    first_value = d;
                }
                digit_count += 1;
            }
        }

        if digit_count > 1 && first_value == 0 {
            return Err(Error::LeadingZero {
                index: first_digit.unwrap(),
            });
        }
        if digit_count > max_digits {
            let target = max_digits.saturating_add(1);
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    seen += 1;
                    if seen == target {
                        return Err(Error::TooManyDigits { limit: max_digits, index });
                    }
                }
            }
        }

        let mut result = Self::zero();
        for &byte in &bytes[start..] {
            if byte != b'_' {
                result.mul_small_add(radix, digit_value(byte).unwrap());
            }
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut magnitude = self.clone();
        magnitude.negative = false;
        let mut digits = Vec::new();
        if magnitude.is_zero() {
            digits.push(0u32);
        } else {
            while !magnitude.is_zero() {
                digits.push(magnitude.div_small(radix));
            }

        while digits.len() < min_digits {
            digits.push(0);
        }

            out.push('-');

        for (i, digit) in digits.iter().rev().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }

                (b'0' + *digit as u8) as char
            } else {
                (b'a' + (*digit as u8 - 10)) as char
            };
            out.push(ch);
        }
        Ok(out)
    }

            (false, true) => Ordering::Greater,

                Ordering::Greater => Self::from_parts(self.negative, Self::abs_sub(self, rhs)),
                Ordering::Less => Self::from_parts(rhs.negative, Self::abs_sub(rhs, self)),
                Ordering::Equal => Self::zero(),

        }

                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                k += 1;
            }

    }

    fn abs_mul_small(&self, factor: u32) -> Self {
        let mut result = Self::zero();
        for &limb in &self.limbs {
            let value = limb as u64 * factor as u64;
            result.limbs.push((value % Self::BASE) as u32);
        }
        let mut carry = 0u64;
        for limb in &mut result.limbs {
            let value = *limb as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            result.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
        result.normalize();
        result
    }

    fn abs_div_rem(a: &Self, b: &Self) -> (Self, Self) {
        if a.abs_cmp(b) == Ordering::Less {
            let mut rem = a.clone();
            rem.negative = false;
            return (Self::zero(), rem);
        }
        let mut quotient_be = Vec::new();
        let mut rem = Self::zero();
        for &limb in a.limbs.iter().rev() {
            rem.limbs.insert(0, limb);
            rem.normalize();

            let mut low = 0u32;
            let mut high = (Self::BASE - 1) as u32;
            let mut best = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = b.abs_mul_small(mid);
                match product.abs_cmp(&rem) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        if mid == u32::MAX { break; }
                        low = mid + 1;
                    }
                }
            }
            let product = b.abs_mul_small(best);
            rem = Self::from_parts(false, Self::abs_sub(&rem, &product));
            quotient_be.push(best);
        }
        quotient_be.reverse();
        let mut quotient = Self::from_parts(false, quotient_be);
        quotient.normalize();
        (quotient, rem)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::abs_div_rem(self, rhs);
        quotient.negative = !quotient.is_zero() && (self.negative ^ rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 != 0 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, r) = Self::abs_div_rem(&a, &b);
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
