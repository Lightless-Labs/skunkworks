```rust
use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs. Zero is an empty magnitude.
    limbs: Vec<u32>,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("limbs", &self.limbs)
            .finish()
    }
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add(a: &Self, b: &Self) -> Vec<u32> {
        let n = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::new();
        let mut carry = 0u64;
        for i in 0..n {
            let av = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = av + bv + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires |a| >= |b|.
    fn abs_sub(a: &Self, b: &Self) -> Vec<u32> {
        let mut out = Vec::new();
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let av = a.limbs[i] as i64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut d = av - bv - borrow;
            if d < 0 {
                d += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(d as u32);
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn from_parts(negative: bool, limbs: Vec<u32>) -> Self {
        let mut value = Self { negative, limbs };
        value.normalize();
        value
    }

    fn mul_small_add(&mut self, multiplier: u32, addend: u32) {
        let mut carry = addend as u64;
        for limb in &mut self.limbs {
            let value = (*limb as u64) * (multiplier as u64) + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let value = rem * Self::BASE + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            rem = value % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let digit_value = |b: u8| -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((b - b'A') as u32 + 10),
                _ => None,
            }
        };

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            if digit_value(byte).filter(|&d| d < radix).is_none() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut first_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let before_ok = i > start
                    && digit_value(bytes[i - 1]).is_some_and(|d| d < radix);
                let after_ok = i + 1 < bytes.len()
                    && digit_value(bytes[i + 1]).is_some_and(|d| d < radix);
                if (!before_ok || !after_ok) && first_separator.is_none() {
                    first_separator = Some(i);
                }
            }
        }
        if let Some(index) = first_separator {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_value = 0;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' {
                let d = digit_value(byte).expect("validated digit");
                if first_digit.is_none() {
                    first_digit = Some(index);
                    first_value = d;
                }
                digit_count += 1;
            }
        }

        if digit_count > 1 && first_value == 0 {
            return Err(Error::LeadingZero {
                index: first_digit.unwrap(),
            });
        }
        if digit_count > max_digits {
            let target = max_digits.saturating_add(1);
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    seen += 1;
                    if seen == target {
                        return Err(Error::TooManyDigits { limit: max_digits, index });
                    }
                }
            }
        }

        let mut result = Self::zero();
        for &byte in &bytes[start..] {
            if byte != b'_' {
                result.mul_small_add(radix, digit_value(byte).unwrap());
            }
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut magnitude = self.clone();
        magnitude.negative = false;
        let mut digits = Vec::new();
        if magnitude.is_zero() {
            digits.push(0u32);
        } else {
            while !magnitude.is_zero() {
                digits.push(magnitude.div_small(radix));
            }
        }
        while digits.len() < min_digits {
            digits.push(0);
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().rev().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            let ch = if *digit < 10 {
                (b'0' + *digit as u8) as char
            } else {
                (b'a' + (*digit as u8 - 10)) as char
            };
            out.push(ch);
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(self.negative, Self::abs_add(self, rhs))
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Greater => Self::from_parts(self.negative, Self::abs_sub(self, rhs)),
                Ordering::Less => Self::from_parts(rhs.negative, Self::abs_sub(rhs, self)),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let value = out[k] as u64 + a as u64 * b as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let value = out[k] as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                k += 1;
            }
        }
        Self::from_parts(self.negative ^ rhs.negative, out)
    }

    fn abs_mul_small(&self, factor: u32) -> Self {
        let mut result = Self::zero();
        for &limb in &self.limbs {
            let value = limb as u64 * factor as u64;
            result.limbs.push((value % Self::BASE) as u32);
        }
        let mut carry = 0u64;
        for limb in &mut result.limbs {
            let value = *limb as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            result.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
        result.normalize();
        result
    }

    fn abs_div_rem(a: &Self, b: &Self) -> (Self, Self) {
        if a.abs_cmp(b) == Ordering::Less {
            let mut rem = a.clone();
            rem.negative = false;
            return (Self::zero(), rem);
        }
        let mut quotient_be = Vec::new();
        let mut rem = Self::zero();
        for &limb in a.limbs.iter().rev() {
            rem.limbs.insert(0, limb);
            rem.normalize();

            let mut low = 0u32;
            let mut high = (Self::BASE - 1) as u32;
            let mut best = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = b.abs_mul_small(mid);
                match product.abs_cmp(&rem) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        if mid == u32::MAX { break; }
                        low = mid + 1;
                    }
                }
            }
            let product = b.abs_mul_small(best);
            rem = Self::from_parts(false, Self::abs_sub(&rem, &product));
            quotient_be.push(best);
        }
        quotient_be.reverse();
        let mut quotient = Self::from_parts(false, quotient_be);
        quotient.normalize();
        (quotient, rem)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::abs_div_rem(self, rhs);
        quotient.negative = !quotient.is_zero() && (self.negative ^ rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 != 0 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, r) = Self::abs_div_rem(&a, &b);
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
}
```
