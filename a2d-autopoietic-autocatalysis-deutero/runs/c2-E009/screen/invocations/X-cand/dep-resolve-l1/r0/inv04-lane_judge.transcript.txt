fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    fn rank(c: &Constraint) -> u8 {
        match c {
            Constraint::Exact(_) => 0,
            Constraint::AtLeast(_) => 1,
            Constraint::Before(_) => 2,
            Constraint::Caret(_) => 3,
        }
    }
    rank(a)
        .cmp(&rank(b))
        .then_with(|| match (a, b) {
            (Constraint::Exact(x), Constraint::Exact(y))
            | (Constraint::AtLeast(x), Constraint::AtLeast(y))
            | (Constraint::Before(x), Constraint::Before(y))
            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(y),
            _ => std::cmp::Ordering::Equal,
        })
}

            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

        if !valid_name(&package.name) {

                name: package.name.clone(),
            });

            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        seen_packages.push(&package.name);
    }

    for package in catalog {
        let mut seen_versions: Vec<Version> = Vec::new();
        for release in &package.releases {
            if seen_versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            seen_versions.push(release.version);
        }
    }

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

            .keys()
            .filter(|name| !catalog.iter().any(|package| package.name == ***name))
            .next()
            .cloned();

            return Err(ResolveError::Missing { package });

                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(constraints),
                });
            }
        }

        let next = active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(name) = next else {
            break;
        };

        let package = catalog
            .iter()
            .find(|package| package.name == name)
            .expect("missing packages checked above");

            .releases
            .iter()
            .filter(|release| {
                active[&name]
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            })
            .map(|release| release.version)
            .max()
            .expect("zero matching releases checked above");

            .releases
            .iter()

            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

    }

    for (name, _, release_index) in selected {
        if let Some(package) = find_package(catalog, name) {
            if let Some(release) = package.releases.get(*release_index) {
                for dependency in &release.deps {
                    add(
                        &mut active,
                        &dependency.package,
                        dependency.constraint,
                    );
                }
            }
        }
    }

    for (_, constraints) in &mut active {
        canonical_constraints(constraints);
    }
    active.sort_by(|a, b| a.0.cmp(&b.0));
    active
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        validate_name(format!("root[{index}]"), &root.package)?;
    }
    for (package_index, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{package_index}]"), &package.name)?;
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                validate_name(
                    format!(
                        "catalog[{package_index}].release[{release_index}].dep[{dependency_index}]"
                    ),
                    &dependency.package,
                )?;
            }
        }
    }

    for index in 0..catalog.len() {
        if catalog[..index]
            .iter()
            .any(|previous| previous.name == catalog[index].name)
        {
            return Err(ResolveError::DuplicatePackage {
                package: catalog[index].name.clone(),
            });
        }
    }

    for package in catalog {
        for (release_index, release) in package.releases.iter().enumerate() {
            if package.releases[..release_index]
                .iter()
                .any(|previous| previous.version == release.version)
            {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut selected: Vec<(String, Version, usize)> = Vec::new();

    loop {
        let active = active_constraints(roots, &selected, catalog);

        if let Some((name, _)) = active
            .iter()
            .find(|(name, _)| find_package(catalog, name).is_none())
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, constraints) in &active {
            if let Some((_, version, _)) = selected.iter().find(|(n, _, _)| n == name) {
                if !constraints
                    .iter()
                    .all(|constraint| matches_constraint(*version, *constraint))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.iter().any(|(selected_name, _, _)| selected_name == name) {
                continue;
            }
            let package = find_package(catalog, name).expect("missing packages checked above");
            let has_match = package.releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches_constraint(release.version, *constraint))
            });
            if !has_match {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active
            .iter()
            .find(|(name, _)| !selected.iter().any(|(selected_name, _, _)| selected_name == name));

        let Some((name, constraints)) = next else {
            break;
        };

        let package = find_package(catalog, name).expect("missing packages checked above");
        let (release_index, release) = package
            .releases
            .iter()
            .enumerate()
            .filter(|(_, release)| {
                constraints
                    .iter()
                    .all(|constraint| matches_constraint(release.version, *constraint))
            })
            .max_by_key(|(_, release)| release.version)
            .expect("zero candidates checked above");

        selected.push((name.clone(), release.version, release_index));
    }

    selected.sort_by(|a, b| a.0.cmp(&b.0));
    Ok(selected
        .into_iter()
        .map(|(package, version, _)| Selection { package, version })
        .collect())
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_indices: Vec<usize> = (0..catalog.len()).collect();
    package_indices.sort_unstable_by(|&a, &b| catalog[a].name.cmp(&catalog[b].name));
    for pair in package_indices.windows(2) {
        if catalog[pair[0]].name == catalog[pair[1]].name {
            return Err(ResolveError::DuplicatePackage {
                package: catalog[pair[1]].name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions: Vec<Version> = Vec::new();
        for release in &package.releases {
            if versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            versions.push(release.version);
        }
    }

    let mut constraints: Vec<(String, Vec<Constraint>)> = Vec::new();
    for root in roots {
        if let Some((_, list)) = constraints.iter_mut().find(|(name, _)| *name == root.package) {
            list.push(root.constraint);
        } else {
            constraints.push((root.package.clone(), vec![root.constraint]));
        }
    }

    let mut selected: Vec<(String, Version)> = Vec::new();

    loop {
        for (_, list) in &mut constraints {
            canonicalize(list);
        }

        let mut missing: Option<&str> = None;
        for (name, _) in &constraints {
            if catalog.binary_search_by(|p| p.name.as_str().cmp(name.as_str())).is_err()
                && missing.map_or(true, |current| name.as_str() < current)
            {
                missing = Some(name);
            }
        }
        if let Some(package) = missing {
            return Err(ResolveError::Missing {
                package: package.to_owned(),
            });
        }

        let mut active_indices: Vec<usize> = (0..constraints.len()).collect();
        active_indices.sort_unstable_by(|&a, &b| constraints[a].0.cmp(&constraints[b].0));

        for &index in &active_indices {
            let (name, requirements) = &constraints[index];
            if let Some((_, version)) = selected.iter().find(|(selected_name, _)| selected_name == name)
            {
                if !requirements
                    .iter()
                    .all(|&constraint| constraint_matches(constraint, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: requirements.clone(),
                    });
                }
            }
        }

        let mut next: Option<usize> = None;
        for &index in &active_indices {
            let (name, requirements) = &constraints[index];
            if selected.iter().any(|(selected_name, _)| selected_name == name) {
                continue;
            }
            let package_index = catalog
                .binary_search_by(|p| p.name.as_str().cmp(name.as_str()))
                .expect("missing packages were checked");
            let has_match = catalog[package_index].releases.iter().any(|release| {
                requirements.iter().all(|&constraint| {
                    constraint_matches(constraint, release.version)
                })
            });
            if !has_match {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: requirements.clone(),
                });
            }
            if next.is_none() {
                next = Some(index);
            }
        }

        let Some(index) = next else {
            break;
        };

        let name = constraints[index].0.clone();
        let requirements = &constraints[index].1;
        let package_index = catalog
            .binary_search_by(|p| p.name.as_str().cmp(name.as_str()))
            .expect("missing packages were checked");
        let release = catalog[package_index]
            .releases
            .iter()
            .filter(|release| {
                requirements.iter().all(|&constraint| {
                    constraint_matches(constraint, release.version)
                })
            })
            .max_by_key(|release| release.version)
            .expect("matching releases were checked");

        let version = release.version;
        let deps = release.deps.clone();
        selected.push((name.clone(), version));

        for dep in deps {
            if let Some((_, list)) = constraints
                .iter_mut()
                .find(|(active_name, _)| *active_name == dep.package)
            {
                list.push(dep.constraint);
            } else {
                constraints.push((dep.package, vec![dep.constraint]));
            }
        }
    }

    selected.sort_unstable_by(|a, b| a.0.cmp(&b.0));
    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

fn permits(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(required) => version == required,
        Constraint::AtLeast(required) => version >= required,
        Constraint::Before(bound) => version < bound,
        Constraint::Caret(required) => {
            if version < required {
                false
            } else if required.major > 0 {
                version.major == required.major
            } else if required.minor > 0 {
                version.major == 0 && version.minor == required.minor
            } else {
                version == required
            }
        }
    }

pub fn resolve(catalog: &[Package], roots: &[Requirement]) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if let Some(error) = invalid_name(format!("root[{i}]"), &root.package) {
            return Err(error);
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if let Some(error) = invalid_name(format!("catalog[{i}]"), &package.name) {
            return Err(error);
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                if let Some(error) = invalid_name(
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                    &dependency.package,
                ) {
                    return Err(error);
                }
            }
        }
    }

    let mut package_indices = BTreeMap::<String, usize>::new();
    for (i, package) in catalog.iter().enumerate() {
        if package_indices
            .insert(package.name.clone(), i)
            .is_some()
        {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut active = BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected = BTreeMap::<String, Version>::new();
    let mut missing = BTreeSet::<String>::new();
    let mut conflicts = BTreeSet::<String>::new();
    let mut no_matches = BTreeSet::<String>::new();

    fn refresh_status(
        name: &str,
        active: &BTreeMap<String, Vec<Constraint>>,
        package_indices: &BTreeMap<String, usize>,
        catalog: &[Package],
        selected: &BTreeMap<String, Version>,
        missing: &mut BTreeSet<String>,
        conflicts: &mut BTreeSet<String>,
        no_matches: &mut BTreeSet<String>,
    ) {
        missing.remove(name);
        conflicts.remove(name);
        no_matches.remove(name);

        let Some(constraints) = active.get(name) else {
            return;
        };

        let Some(&index) = package_indices.get(name) else {
            missing.insert(name.to_owned());
            return;
        };

        if let Some(&version) = selected.get(name) {
            if !satisfies(version, constraints) {
                conflicts.insert(name.to_owned());
            }
            return;
        }

        if !catalog[index]
            .releases
            .iter()
            .any(|release| satisfies(release.version, constraints))
        {
            no_matches.insert(name.to_owned());
        }
    }

    for name in active.keys() {
        refresh_status(
            name,
            &active,
            &package_indices,
            catalog,
            &selected,
            &mut missing,
            &mut conflicts,
            &mut no_matches,
        );
    }

    loop {
        if let Some(name) = missing.iter().next() {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        if let Some(name) = conflicts.iter().next() {
            return Err(ResolveError::Conflict {
                package: name.clone(),
                selected: selected[name],
                requirements: canonicalize(&active[name]),
            });
        }

        if let Some(name) = no_matches.iter().next() {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: canonicalize(&active[name]),
            });
        }

        let Some(name) = active.keys().find(|name| !selected.contains_key(*name)).cloned() else {
            break;
        };

        let index = package_indices[&name];
        let constraints = &active[&name];
        let best = catalog[index]
            .releases
            .iter()
            .filter(|release| satisfies(release.version, constraints))
            .map(|release| release.version)
            .max();

        let Some(version) = best else {
            return Err(ResolveError::NoMatch {
                package: name,
                requirements: canonicalize(constraints),
            });
        };

        selected.insert(name.clone(), version);
        refresh_status(
            &name,
            &active,
            &package_indices,
            catalog,
            &selected,
            &mut missing,
            &mut conflicts,
            &mut no_matches,
        );

        let release = catalog[index]
            .releases
            .iter()
            .find(|release| release.version == version)
            .expect("selected release must exist");

        for dependency in &release.deps {
            active
                .entry(dependency.package.clone())
                .or_default()
                .push(dependency.constraint);
            refresh_status(
                &dependency.package,
                &active,
                &package_indices,
                catalog,
                &selected,
                &mut missing,
                &mut conflicts,
                &mut no_matches,
            );
        }
    }

    Ok(selected
