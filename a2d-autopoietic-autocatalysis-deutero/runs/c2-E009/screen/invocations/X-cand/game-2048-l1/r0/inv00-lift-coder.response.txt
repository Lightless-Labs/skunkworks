```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut cells = vec![0; row.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut output_index = match side {
        Side::Left => 0,
        Side::Right => row.len().saturating_sub(1),
    };

    let mut indices: Vec<usize> = (0..row.len()).collect();
    if side == Side::Right {
        indices.reverse();
    }

    let mut tiles = indices.into_iter().filter(|&index| row[index] != 0).peekable();

    while let Some(first) = tiles.next() {
        let value = row[first];
        let next = tiles.peek().copied();

        if let Some(second) = next {
            if row[second] == value {
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first, second });
                }

                tiles.next();
                let merged_value = value * 2;
                cells[output_index] = merged_value;
                merges.push(Merge {
                    from: [first, second],
                    to: output_index,
                    value: merged_value,
                });
                score += u64::from(merged_value);
            } else {
                cells[output_index] = value;
            }
        } else {
            cells[output_index] = value;
        }

        match side {
            Side::Left => output_index += 1,
            Side::Right => output_index = output_index.saturating_sub(1),
        }
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
```
