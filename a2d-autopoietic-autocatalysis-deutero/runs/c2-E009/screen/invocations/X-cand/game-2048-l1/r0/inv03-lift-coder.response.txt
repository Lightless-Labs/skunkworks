#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (!(2..=2_147_483_648).contains(&value) || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut cells = vec![0; row.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut output_index = 0usize;

    match side {
        Side::Left => {
            let mut index = 0usize;
            while index < row.len() {
                if row[index] == 0 {
                    index += 1;
                    continue;
                }

                let first = index;
                let value = row[index];
                index += 1;

                if index < row.len() {
                    while index < row.len() && row[index] == 0 {
                        index += 1;
                    }
                    if index < row.len() && row[index] == value {
                        let second = index;
                        let merged_value = value.checked_mul(2).ok_or(Error::MergeOverflow {
                            first,
                            second,
                        })?;
                        cells[output_index] = merged_value;
                        merges.push(Merge {
                            from: [first, second],
                            to: output_index,
                            value: merged_value,
                        });
                        score += u64::from(merged_value);
                        output_index += 1;
                        index += 1;
                        continue;
                    }
                }

                cells[output_index] = value;
                output_index += 1;
            }
        }
        Side::Right => {
            let mut cursor = row.len();
            while cursor > 0 {
                cursor -= 1;
                if row[cursor] == 0 {
                    continue;
                }

                let first = cursor;
                let value = row[cursor];
                if cursor > 0 {
                    let mut second_cursor = cursor;
                    while second_cursor > 0 {
                        second_cursor -= 1;
                        if row[second_cursor] != 0 {
                            break;
                        }
                    }
                    if row[second_cursor] == value && row[second_cursor] != 0 {
                        let second = second_cursor;
                        let merged_value = value.checked_mul(2).ok_or(Error::MergeOverflow {
                            first,
                            second,
                        })?;
                        let to = row.len() - 1 - output_index;
                        cells[to] = merged_value;
                        merges.push(Merge {
                            from: [first, second],
                            to,
                            value: merged_value,
                        });
                        score += u64::from(merged_value);
                        output_index += 1;
                        cursor = second_cursor;
                        continue;
                    }
                }

                let to = row.len() - 1 - output_index;
                cells[to] = value;
                output_index += 1;
            }
        }
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
