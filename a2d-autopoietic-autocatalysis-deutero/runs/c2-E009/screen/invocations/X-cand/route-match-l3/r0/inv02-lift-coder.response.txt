#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        let (segments, trailing_slash) = parse_pattern(pattern)?;
        let mut rank = Vec::new();
        let mut has_wildcard = false;
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => {
                    rank.push(0);
                    has_wildcard = true;
                }
            }
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (path_segments, request_trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != request_trailing
            {
                continue;
            }
            if let Some(params) = match_route(route, &path_segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut allowed = Vec::new();
        for (route, _) in &candidates {
            if !allowed.iter().any(|m: &String| m == &route.method) {
                allowed.push(route.method.clone());
            }
        }
        if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
            allowed.push("HEAD".to_owned());
        }
        allowed.sort();

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut winner: Option<(&Route, Vec<(String, String)>)> = None;
        for (route, params) in candidates {
            if route.method != eligible_method {
                continue;
            }
            let replace = match &winner {
                None => true,
                Some((best, _)) => route.rank > best.rank,
            };
            if replace {
                winner = Some((route, params));
            }
        }

        let Some((route, params)) = winner else {
            return Err(Error::MethodNotAllowed { allowed });
        };

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing_slash != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing_slash {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if method.is_empty() || !method.bytes().all(|b| b.is_ascii_uppercase()) {
        return Err(Error::InvalidMethod);
    }
    Ok(())
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    let (raw_segments, trailing) = split_segments(pattern)
        .map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b'*' && index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_owned());
            if marker == b':' {
                segments.push(Segment::Param(name.to_owned()));
            } else {
                segments.push(Segment::Wildcard(name.to_owned()));
            }
        } else {
            let decoded = decode_static(raw, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, trailing))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    let (raw_segments, trailing) =
        split_segments(path).map_err(|at| Error::InvalidPath { at })?;
    let mut segments = Vec::new();
    for (raw, start) in raw_segments {
        segments.push(decode_static(raw, start).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((segments, trailing))
}

fn split_segments(input: &str) -> Result<(Vec<(&str, usize)>, bool), usize> {
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let mut slash = start;
        while slash < end && bytes[slash] != b'/' {
            slash += 1;
        }
        if slash == start {
            return Err(start);
        }
        segments.push((&input[start..slash], start));
        start = slash + 1;
    }

    if !trailing && end == 1 {
        return Err(1);
    }
    Ok((segments, trailing))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_static(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let Some(hi) = hex_upper(bytes[i + 1]) else {
                return Err(offset + i);
            };
            let Some(lo) = hex_upper(bytes[i + 2]) else {
                return Err(offset + i);
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value as char);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b as char);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }
    Ok(decoded)
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));

    let fixed_len = wildcard_index.unwrap_or(route.segments.len());
    if wildcard_index.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard_index.is_some() && path.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => return None,
        }
    }
    if let Some(index) = wildcard_index {
        let Segment::Wildcard(name) = &route.segments[index] else {
            return None;
        };
        let mut value = String::new();
        for (i, part) in path[index..].iter().enumerate() {
            if i > 0 {
                value.push('/');
            }
            value.push_str(part);
        }
        params.push((name.clone(), value));
    }
    Some(params)
}
