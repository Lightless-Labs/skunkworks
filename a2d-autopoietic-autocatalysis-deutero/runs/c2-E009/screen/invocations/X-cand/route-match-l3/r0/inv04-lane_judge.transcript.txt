            method

fn valid_method(method: &str) -> bool {

fn parse_pattern(pattern: &str) -> Result<ParsedPattern, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    let raw = pattern.as_bytes();
    if pattern == "/" {
        return Ok(ParsedPattern {
            segments: Vec::new(),
            trailing_slash: false,
            rank: vec![3],
            has_wildcard: false,
        });
    }

    let trailing_slash = raw.last() == Some(&b'/');
    let end = if trailing_slash {
        raw.len() - 1
    } else {
        raw.len()
    };

    let mut segments = Vec::new();
    let mut rank = Vec::new();
    let mut names: Vec<String> = Vec::new();
    let mut start = 1;
    let mut has_wildcard = false;

    while start < end {
        let next = raw[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let segment = &pattern[start..next];
        let first = raw[start];
        if first == b':' || first == b'*' {
            let name = &segment[1..];
            if !valid_capture_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());

            if first == b'*' {
                if next != end {
                    return Err(Error::InvalidPattern { at: start });
                }
                has_wildcard = true;
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            } else {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            }
        } else {
            let decoded = decode_segment(segment, start, true)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }

        if next == end {
            break;
        }
        start = next + 1;
        if start == end {
            return Err(Error::InvalidPattern { at: start });
        }
    }

    if !has_wildcard {
        rank.push(3);
    }

    Ok(ParsedPattern {
        segments,
        trailing_slash,
        rank,
        has_wildcard,
    })
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    let raw = path.as_bytes();
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let trailing_slash = raw.last() == Some(&b'/');
    let end = if trailing_slash {
        raw.len() - 1
    } else {
        raw.len()
    };

    let mut segments = Vec::new();
    let mut start = 1;
    while start < end {
        let next = raw[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(end);
        if next == start {
            return Err(Error::InvalidPath { at: start });
        }

        segments.push(decode_segment(&path[start..next], start, false)?);
        if next == end {
            break;
        }
        start = next + 1;
        if start == end {
            return Err(Error::InvalidPath { at: start });
        }
    }

    Ok(ParsedPath {
        segments,
        trailing_slash,
    })
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    let Some((&first, rest)) = bytes.split_first() else {
        return false;
    };
    first.is_ascii_lowercase()
        && rest
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(segment: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !bytes[i + 1].is_ascii_hexdigit()
                || !bytes[i + 2].is_ascii_hexdigit()
                || bytes[i + 1].is_ascii_lowercase()
                || bytes[i + 2].is_ascii_lowercase()
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                });
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                });
            }
            decoded.push(value as char);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b as char);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: base + i }
            } else {
                Error::InvalidPath { at: base + i }
            });
        }
    }
    Ok(decoded)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &ParsedPath) -> Option<Vec<(String, String)>> {
    if !route.has_wildcard {
        if route.segments.len() != path.segments.len() {
            return None;
        }
    } else if path.segments.len() + 1 < route.segments.len() {
        return None;
    }

    if route.has_wildcard && route.segments.len() > 0 {
        let wildcard_index = route.segments.len() - 1;
        if path.segments.len() < wildcard_index {
            return None;
        }
    }

    let mut params = Vec::new();
    for (index, segment) in route.segments.iter().enumerate() {
        match segment {
            Segment::Static(expected) => {
                if path.segments.get(index)? != expected {
                    return None;
                }
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.segments.get(index)?.clone()));
            }
            Segment::Wildcard(name) => {
                let mut value = String::new();
                for (i, part) in path.segments[index..].iter().enumerate() {
                    if i != 0 {
                        value.push('/');
                    }
                    value.push_str(part);
                }
                params.push((name.clone(), value));
            }
        }
    }

    Some(params)
}

                }

        Err(Error::InvalidMethod)
    } else {
        Ok(())
    }
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    parse_path_body(path).map_err(|at| Error::InvalidPath { at })
}

fn parse_path_body(path: &str) -> Result<ParsedPath, usize> {
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let trailing_slash = path.ends_with('/');
    let content_end = if trailing_slash {
        path.len() - 1
    } else {
        path.len()
    };

    let mut segments = Vec::new();
    let mut start = 1;
    while start < content_end {
        let end = path[start..content_end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(content_end);
        if end == start {
            return Err(start);
        }
        let decoded = decode_static_segment(&path[start..end], start)?;
        segments.push(decoded);
        start = end + 1;
    }

    if segments.is_empty() {
        return Err(1);
    }

    Ok(ParsedPath {
        segments,
        trailing_slash,
    })
}

fn parse_pattern(pattern: &str) -> Result<ParsedPattern, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(ParsedPattern {
            segments: Vec::new(),
            trailing_slash: false,
            has_wildcard: false,
            rank: vec![3],
        });
    }

    let trailing_slash = pattern.ends_with('/');
    let content_end = if trailing_slash {
        pattern.len() - 1
    } else {
        pattern.len()
    };

    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();
    let mut rank = Vec::new();
    let mut has_wildcard = false;

    let mut start = 1;
    while start < content_end {
        let end = pattern[start..content_end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(content_end);
        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..end];
        if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.starts_with('*');
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && end != content_end {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            if wildcard {
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
                has_wildcard = true;
            } else {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            }
        } else {
            let decoded = decode_static_segment(raw, start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
        start = end + 1;
    }

    if segments.is_empty() {
        return Err(Error::InvalidPattern { at: 1 });
    }
    if !has_wildcard {
        rank.push(3);
    }

    Ok(ParsedPattern {
        segments,
        trailing_slash,
        has_wildcard,
        rank,
    })
}

fn valid_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    match bytes.next() {
        Some(b) if b.is_ascii_lowercase() => {}
        _ => return false,
    }
    bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn decode_static_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(high) = uppercase_hex(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(low) = uppercase_hex(bytes[i + 2]) else {
                return Err(base + i);
            };
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value as char);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b as char);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(decoded)
}

fn uppercase_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(
    route: &Route,
    path: &ParsedPath,
    policy: SlashPolicy,
) -> Option<Vec<(String, String)>> {
    if policy == SlashPolicy::Strict && route.trailing_slash != path.trailing_slash {
        return None;
    }

    let wildcard_index = route
        .segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));

    let fixed_len = wildcard_index.unwrap_or(route.segments.len());
    if path.segments.len() < fixed_len {
        return None;
    }
    if wildcard_index.is_none() && path.segments.len() != route.segments.len() {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(value) if value == &path.segments[i] => {}
            Segment::Static(_) => return None,
            Segment::Param(name) => params.push((name.clone(), path.segments[i].clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if let Some(index) = wildcard_index {
        if let Segment::Wildcard(name) = &route.segments[index] {
            let mut value = String::new();
            for (offset, segment) in path.segments[index..].iter().enumerate() {
                if offset > 0 {
                    value.push('/');
                }
                value.push_str(segment);
            }
            params.push((name.clone(), value));
        }
    }

    Some(params)
}

pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (segments, trailing_slash) = parse_pattern(pattern)?;
        let mut rank = Vec::new();
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => rank.push(0),
            }
        }
        if !segments
            .last()
            .is_some_and(|segment| matches!(segment, Segment::Wildcard(_)))
        {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_string(),
            segments,
            trailing_slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, trailing_slash) = parse_path(head)?;

        let mut candidates = Vec::new();
        for (index, route) in self.routes.iter().enumerate() {
            if path_matches(
                &route.segments,
                route.trailing_slash,
                &request_segments,
                trailing_slash,
                self.policy,
            ) {
                candidates.push(index);
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_has_route = method == "HEAD"
            && candidates
                .iter()
                .any(|&index| self.routes[index].method == "HEAD");
        let effective_method = if method == "HEAD" && !head_has_route {
            "GET"
        } else {
            method
        };

        let mut eligible = Vec::new();
        let mut allowed = Vec::<String>::new();
        for &index in &candidates {
            let route_method = &self.routes[index].method;
            if !allowed.iter().any(|m| m == route_method) {
                allowed.push(route_method.clone());
            }
            if route_method == effective_method {
                eligible.push(index);
            }
        }

        if eligible.is_empty() {
            if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut winner = eligible[0];
        for &index in eligible.iter().skip(1) {
            if self.routes[index].rank > self.routes[winner].rank {
                winner = index;
            }
        }

        let route = &self.routes[winner];
        let params = capture_params(&route.segments, &request_segments);
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != trailing_slash
            && !route.segments.is_empty()
        {
            let mut target = head.to_string();
            if route.trailing_slash {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner,
            params,
            redirect,
        })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let (parts, trailing) = split_segments(pattern, true)?;
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();

    for (index, (part, start)) in parts.iter().enumerate() {
        let part_bytes = part.as_bytes();
        if part_bytes.first() == Some(&b':') || part_bytes.first() == Some(&b'*') {
            let marker = part_bytes[0];
            let name = &part[1..];
            if !valid_capture_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b'*' && index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_string());
            if marker == b':' {
                segments.push(Segment::Param(name.to_string()));
            } else {
                segments.push(Segment::Wildcard(name.to_string()));
            }
        } else {
            segments.push(Segment::Static(decode_segment(part, *start, true)?));
        }
    }

    Ok((segments, trailing))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let (parts, trailing) = split_segments(path, false)?;
    let mut segments = Vec::new();
    for (part, start) in parts {
        segments.push(decode_segment(part, start, false)?);
    }
    Ok((segments, trailing))
}

fn split_segments<'a>(
    input: &'a str,
    pattern: bool,
) -> Result<(Vec<(&'a str, usize)>, bool), Error> {
    let bytes = input.as_bytes();
    let mut end = bytes.len();
    let trailing = bytes.last() == Some(&b'/');
    if trailing {
        end -= 1;
    }

    let mut parts = Vec::new();
    let mut start = 1usize;
    let mut i = 1usize;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(if pattern {
                    Error::InvalidPattern { at: start }
                } else {
                    Error::InvalidPath { at: start }
                });
            }
            parts.push((&input[start..i], start));
            start = i + 1;
        }
        i += 1;
    }
    Ok((parts, trailing))
}

fn decode_segment(segment: &str, start: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = String::new();
    let mut i = 0usize;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(invalid_segment_error(pattern, start + i));
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid_segment_error(pattern, start + i));
            }
            decoded.push(value as char);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte as char);
            i += 1;
        } else {
            return Err(invalid_segment_error(pattern, start + i));
        }
    }
    Ok(decoded)
}

fn invalid_segment_error(pattern: bool, at: usize) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

fn path_matches(
    pattern: &[Segment],
    pattern_trailing: bool,
    request: &[String],
    request_trailing: bool,
    policy: SlashPolicy,
) -> bool {
    if policy == SlashPolicy::Strict && pattern_trailing != request_trailing {
        return false;
    }

    let has_wildcard = pattern
        .last()
        .is_some_and(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_count = if has_wildcard {
        pattern.len() - 1
    } else {
        pattern.len()
    };

    if request.len() < fixed_count || (!has_wildcard && request.len() != fixed_count) {
        return false;
    }

    for i in 0..fixed_count {
        if let Segment::Static(value) = &pattern[i] {
            if value != &request[i] {
                return false;
            }
        }
    }
    true
}

fn capture_params(pattern: &[Segment], request: &[String]) -> Vec<(String, String)> {
    let mut params = Vec::new();
    let mut request_index = 0usize;

    for segment in pattern {
        match segment {
            Segment::Static(_) => request_index += 1,
            Segment::Param(name) => {
                params.push((name.clone(), request[request_index].clone()));
                request_index += 1;
            }
            Segment::Wildcard(name) => {
                let mut value = String::new();
                for (i, part) in request[request_index..].iter().enumerate() {
                    if i > 0 {
                        value.push('/');
                    }
                    value.push_str(part);
                }
                params.push((name.clone(), value));
                break;
            }
        }
    }
    params
}

impl Router {

        let mut rank = Vec::new();
        let mut has_wildcard = false;
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => {
                    rank.push(0);
                    has_wildcard = true;
                }
            }
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (path_segments, request_trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != request_trailing
            {
                continue;
            }
            if let Some(params) = match_route(route, &path_segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut allowed = Vec::new();
        for (route, _) in &candidates {
            if !allowed.iter().any(|m: &String| m == &route.method) {
                allowed.push(route.method.clone());
            }
        }
        if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
            allowed.push("HEAD".to_owned());
        }
        allowed.sort();

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut winner: Option<(&Route, Vec<(String, String)>)> = None;
        for (route, params) in candidates {
            if route.method != eligible_method {
                continue;
            }
            let replace = match &winner {
                None => true,
                Some((best, _)) => route.rank > best.rank,
            };
            if replace {
                winner = Some((route, params));
            }
        }

        let Some((route, params)) = winner else {
            return Err(Error::MethodNotAllowed { allowed });
        };

        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && head != "/"
            && route.trailing_slash != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing_slash {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}

        return Err(Error::InvalidMethod);

}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    let (raw_segments, trailing) = split_segments(pattern)
        .map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::new();
    let mut names: Vec<String> = Vec::new();

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b'*' && index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_owned());
            if marker == b':' {
                segments.push(Segment::Param(name.to_owned()));
            } else {
                segments.push(Segment::Wildcard(name.to_owned()));
            }
        } else {
            let decoded = decode_static(raw, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, trailing))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    let (raw_segments, trailing) =
        split_segments(path).map_err(|at| Error::InvalidPath { at })?;
    let mut segments = Vec::new();
    for (raw, start) in raw_segments {
        segments.push(decode_static(raw, start).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((segments, trailing))
}

fn split_segments(input: &str) -> Result<(Vec<(&str, usize)>, bool), usize> {
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let mut slash = start;
        while slash < end && bytes[slash] != b'/' {
            slash += 1;
        }
        if slash == start {
            return Err(start);
        }
        segments.push((&input[start..slash], start));
        start = slash + 1;
    }

    if !trailing && end == 1 {
        return Err(1);
    }
    Ok((segments, trailing))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_static(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let Some(hi) = hex_upper(bytes[i + 1]) else {
                return Err(offset + i);
            };
            let Some(lo) = hex_upper(bytes[i + 2]) else {
                return Err(offset + i);
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value as char);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b as char);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }
    Ok(decoded)
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));

    let fixed_len = wildcard_index.unwrap_or(route.segments.len());
    if wildcard_index.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard_index.is_some() && path.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => return None,
        }
    }
    if let Some(index) = wildcard_index {
        let Segment::Wildcard(name) = &route.segments[index] else {
            return None;
        };
        let mut value = String::new();
        for (i, part) in path[index..].iter().enumerate() {
            if i > 0 {
                value.push('/');
            }
            value.push_str(part);
        }
        params.push((name.clone(), value));
    }
    Some(params)
}
