#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn alphabet_byte(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 if alphabet == Alphabet::Standard => b'+',
        63 if alphabet == Alphabet::Standard => b'/',
        62 => b'-',
        _ => b'_',
    }
}

fn encode_symbols(input: &[u8], alphabet: Alphabet, padding: Padding) -> Vec<u8> {
    let mut out = Vec::new();
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        out.push(alphabet_byte(a >> 2, alphabet));
        out.push(alphabet_byte(((a & 0x03) << 4) | (b >> 4), alphabet));
        out.push(alphabet_byte(((b & 0x0f) << 2) | (c >> 6), alphabet));
        out.push(alphabet_byte(c & 0x3f, alphabet));
    }
    let tail = chunks.remainder();
    if !tail.is_empty() {
        let a = tail[0];
        out.push(alphabet_byte(a >> 2, alphabet));
        if tail.len() == 1 {
            out.push(alphabet_byte((a & 0x03) << 4, alphabet));
            if padding != Padding::Forbidden {
                out.extend_from_slice(b"==");
            }
        } else {
            let b = tail[1];
            out.push(alphabet_byte(((a & 0x03) << 4) | (b >> 4), alphabet));
            out.push(alphabet_byte((b & 0x0f) << 2, alphabet));
            if padding != Padding::Forbidden {
                out.push(b'=');
            }
        }
    }
    out
}

fn wrap_symbols(symbols: &[u8], width: usize, ending: LineEnding) -> String {
    let mut out = String::new();
    for (i, chunk) in symbols.chunks(width).enumerate() {
        if i != 0 {
            match ending {
                LineEnding::Lf => out.push('\n'),
                LineEnding::CrLf => out.push_str("\r\n"),
            }
        }
        out.push_str(std::str::from_utf8(chunk).expect("encoded symbols are ASCII"));
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let symbols = encode_symbols(input, options.alphabet, options.padding);
    match options.whitespace {
        Whitespace::Wrapped { width, ending } => Ok(wrap_symbols(&symbols, width, ending)),
        _ => Ok(String::from_utf8(symbols).expect("encoded symbols are ASCII")),
    }
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    emitted_in_row: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            pending: Vec::new(),
            emitted_in_row: 0,
            finished: false,
        })
    }

    fn append_symbols(&mut self, symbols: &[u8], out: &mut String) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            for &byte in symbols {
                if self.emitted_in_row == width {
                    match ending {
                        LineEnding::Lf => out.push('\n'),
                        LineEnding::CrLf => out.push_str("\r\n"),
                    }
                    self.emitted_in_row = 0;
                }
                out.push(byte as char);
                self.emitted_in_row += 1;
            }
        } else {
            for &byte in symbols {
                out.push(byte as char);
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut combined = std::mem::take(&mut self.pending);
        combined.extend_from_slice(input);
        let complete_len = combined.len() / 3 * 3;
        let symbols = encode_symbols(
            &combined[..complete_len],
            self.options.alphabet,
            Padding::Forbidden,
        );
        self.pending.extend_from_slice(&combined[complete_len..]);
        let mut out = String::new();
        self.append_symbols(&symbols, &mut out);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let symbols = encode_symbols(
            &self.pending,
            self.options.alphabet,
            self.options.padding,
        );
        self.pending.clear();
        let mut out = String::new();
        self.append_symbols(&symbols, &mut out);
        Ok(out)
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

struct FrameState {
    row_symbols: usize,
    separator: u8,
    separator_offset: Option<usize>,
    after_separator: bool,
}

impl FrameState {
    fn new() -> Self {
        Self {
            row_symbols: 0,
            separator: 0,
            separator_offset: None,
            after_separator: false,
        }
    }
}

pub struct Decoder {
    options: Options,
    group: Vec<Symbol>,
    absolute_offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    frame: FrameState,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::new(),
            absolute_offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            frame: FrameState::new(),
            finished: false,
            sticky_error: None,
        })
    }

    fn scan_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if let Some(pending_offset) = self.pending_second_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_second_padding = None;
            self.group.push(Symbol { byte, offset });
            if self.group.len() == 4 {
                let first = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
                let second = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
                out.push((first << 2) | (second >> 4));
                self.group.clear();
                self.terminal_padding = true;
            }
            let _ = pending_offset;
            return Ok(());
        }

        let position = self.group.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_second_padding = Some(offset);
                    self.group.push(Symbol { byte, offset });
                    return Ok(());
                }
                3 => {
                    let first = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
                    let second = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
                    let third = alphabet_digit(self.group[2].byte, self.options.alphabet).unwrap();
                    out.push((first << 2) | (second >> 4));
                    out.push((second << 4) | (third >> 2));
                    self.group.clear();
                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = alphabet_digit(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset, byte })?;
        self.group.push(Symbol { byte, offset });
        if self.group.len() == 4 {
            let a = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
            let b = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
            let c = alphabet_digit(self.group[2].byte, self.options.alphabet).unwrap();
            let d = value;
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);
            self.group.clear();
        }
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.scan_symbol(byte, offset, out),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.scan_symbol(byte, offset, out)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                let frame = &mut self.frame;
                if frame.separator != 0 {
                    match (frame.separator, ending, byte) {
                        (1, LineEnding::Lf, b'\n') => {
                            frame.separator = 0;
                            frame.separator_offset = None;
                            frame.after_separator = true;
                            Ok(())
                        }
                        (1, LineEnding::CrLf, b'\n') => {
                            frame.separator = 0;
                            frame.separator_offset = None;
                            frame.after_separator = true;
                            Ok(())
                        }
                        (1, LineEnding::CrLf, _) => Err(Error::BadWrap { offset }),
                        (1, LineEnding::Lf, _) => Err(Error::BadWrap { offset }),
                        _ => unreachable!(),
                    }
                } else if frame.row_symbols == width {
                    if byte == b'\n' && ending == LineEnding::Lf {
                        frame.separator = 1;
                        frame.separator_offset = Some(offset);
                        self.finish_separator(out)
                    } else if byte == b'\r' && ending == LineEnding::CrLf {
                        frame.separator = 1;
                        frame.separator_offset = Some(offset);
                        Ok(())
                    } else {
                        Err(Error::BadWrap { offset })
                    }
                } else if byte == b'\n' || byte == b'\r' || matches!(byte, 9 | 11 | 12 | 32) {
                    Err(Error::BadWrap { offset })
                } else {
                    frame.row_symbols += 1;
                    frame.after_separator = false;
                    self.scan_symbol(byte, offset, out)
                }
            }
        }
    }

    fn finish_separator(&mut self, _out: &mut Vec<u8>) -> Result<(), Error> {
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut out = Vec::new();
        for &byte in input {
            let offset = self.absolute_offset;
            self.absolute_offset = self.absolute_offset.saturating_add(1);
            if let Err(error) = self.process_byte(byte, offset, &mut out) {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let error = match self.options.whitespace {
            Whitespace::Wrapped { .. } => {
                if let Some(offset) = self.frame.separator_offset {
                    Some(Error::BadWrap { offset })
                } else if self.frame.after_separator {
                    Some(Error::BadWrap {
                        offset: self.absolute_offset.saturating_sub(1),
                    })
                } else {
                    None
                }
            }
            _ => None,
        };
        if let Some(error) = error {
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let residual = self.group.len();
        let result = match residual {
            0 => Ok(Vec::new()),
            1 => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
            2 if self.options.padding != Padding::Required => {
                let a = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
                Ok(vec![(a << 2) | (b >> 4)])
            }
            3 if self.options.padding != Padding::Required => {
                let a = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
                let c = alphabet_digit(self.group[2].byte, self.options.alphabet).unwrap();
                Ok(vec![(a << 2) | (b >> 4), (b << 4) | (c >> 2)])
            }
            _ => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
        };
        match result {
            Ok(bytes) => {
                self.finished = true;
                Ok(bytes)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
