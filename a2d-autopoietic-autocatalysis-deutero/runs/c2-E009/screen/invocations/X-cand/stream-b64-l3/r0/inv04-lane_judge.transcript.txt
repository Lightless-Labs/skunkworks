fn validate_options(options: Options) -> Result<(), Error> {

fn alphabet_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn digit_byte(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 if alphabet == Alphabet::Standard => b'+',
        63 if alphabet == Alphabet::Standard => b'/',
        62 => b'-',
        _ => b'_',
    }
}

fn emit_symbol(out: &mut String, byte: u8, whitespace: Whitespace, column: &mut usize) {
    if let Whitespace::Wrapped { width, ending } = whitespace {
        if *column == width {
            match ending {
                LineEnding::Lf => out.push('\n'),
                LineEnding::CrLf => out.push_str("\r\n"),
            }
            *column = 0;
        }
        *column += 1;
    }
    out.push(byte as char);
}

fn encode_group(
    bytes: &[u8],
    options: Options,
    out: &mut String,
    column: &mut usize,
) {
    let b0 = bytes[0];
    let b1 = if bytes.len() > 1 { bytes[1] } else { 0 };
    let b2 = if bytes.len() > 2 { bytes[2] } else { 0 };
    let values = [
        b0 >> 2,
        ((b0 & 0x03) << 4) | (b1 >> 4),
        ((b1 & 0x0f) << 2) | (b2 >> 6),
        b2 & 0x3f,
    ];
    let count = match bytes.len() {
        1 => 2,
        2 => 3,
        _ => 4,
    };
    for value in values.iter().take(count) {
        emit_symbol(out, digit_byte(*value, options.alphabet), options.whitespace, column);
    }
    if bytes.len() < 3 && options.padding != Padding::Forbidden {
        for _ in count..4 {
            emit_symbol(out, b'=', options.whitespace, column);
        }
    }
}

impl Encoder {

        Ok(Self {
            options,
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        })

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {

            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        let mut column = 0;

                    self.tail_len = 0;

                i += 1;
            }

        while i + 3 <= input.len() {
            encode_group(&input[i..i + 3], self.options, &mut out, &mut column);
            i += 3;
        }
        while i < input.len() {
            self.tail[self.tail_len] = input[i];
            self.tail_len += 1;
            i += 1;
        }
        Ok(out)
    }

        if self.finished {
            return Err(Error::AlreadyFinished);

enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    pending_second_pad: Option<usize>,
    padded: bool,
    total_offset: usize,
    row_symbols: usize,
    frame_state: FrameState,
    sticky: Option<Error>,
    finished: bool,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            pending_second_pad: None,
            padded: false,
            total_offset: 0,
            row_symbols: 0,
            frame_state: FrameState::Normal,
            sticky: None,
            finished: false,
        })
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.padded {
            return Err(Error::BadPadding { offset });
        }
        if self.pending_second_pad.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }
        if self.group_len < 2 && byte == b'=' {
            return Err(Error::BadPadding { offset });
        }

        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if self.group_len == 2 {
                self.group[self.group_len] = byte;
                self.offsets[self.group_len] = offset;
                self.group_len += 1;
                self.pending_second_pad = Some(offset);
                return Ok(());
            }
            if self.group_len == 3 {
                self.group[self.group_len] = byte;
                self.offsets[self.group_len] = offset;
                self.group_len += 1;
                self.padded = true;
                self.emit_group(out)?;
                return Ok(());
            }
            return Err(Error::BadPadding { offset });
        }

        let value = alphabet_digit(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset, byte })?;
        self.group[self.group_len] = value;
        self.offsets[self.group_len] = offset;
        self.group_len += 1;

        if self.group_len == 4 {
            self.emit_group(out)?;
        }
        Ok(())
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.group_len != 4 {
            return Ok(());
        }
        if self.group[0] > 63 || self.group[1] > 63 {
            return Err(Error::BadPadding { offset: self.offsets[2] });
        }
        let a = self.group[0];
        let b = self.group[1];
        out.push((a << 2) | (b >> 4));

        if self.group[2] != b'=' {
            let c = self.group[2];
            out.push((b << 4) | (c >> 2));
            if self.group[3] != b'=' {
                let d = self.group[3];
                out.push((c << 6) | d);
            }
        }
        self.group_len = 0;
        self.pending_second_pad = None;
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut out = Vec::new();
        for &byte in input {
            let offset = self.total_offset;
            self.total_offset = self.total_offset.saturating_add(1);

            match self.options.whitespace {
                Whitespace::IgnoreAscii
                    if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) =>
                {
                    continue;
                }
                Whitespace::Wrapped { width, ending } => {
                    match self.frame_state {
                        FrameState::NeedLf { cr_offset } => {
                            if byte != b'\n' {
                                return self.fail(Error::BadWrap { offset });
                            }
                            self.frame_state = FrameState::NeedSymbol {
                                separator_offset: cr_offset,
                            };
                            continue;
                        }
                        FrameState::NeedSymbol { separator_offset } => {
                            self.frame_state = FrameState::Normal;
                            self.row_symbols = 0;
                            if matches!(byte, b'\r' | b'\n') {
                                return self.fail(Error::BadWrap { offset });
                            }
                            if let Err(error) = self.process_symbol(byte, offset, &mut out) {
                                return self.fail(error);
                            }
                            self.row_symbols += 1;
                            continue;
                        }
                        FrameState::Normal => {}
                    }

                    if self.row_symbols == width {
                        match ending {
                            LineEnding::Lf if byte == b'\n' => {
                                self.frame_state = FrameState::NeedSymbol {
                                    separator_offset: offset,
                                };
                                continue;
                            }
                            LineEnding::CrLf if byte == b'\r' => {
                                self.frame_state = FrameState::NeedLf { cr_offset: offset };
                                continue;
                            }
                            _ => return self.fail(Error::BadWrap { offset }),
                        }
                    } else if matches!(byte, b'\r' | b'\n') {
                        return self.fail(Error::BadWrap { offset });
                    }
                }
                _ => {}
            }

            if let Err(error) = self.process_symbol(byte, offset, &mut out) {
                return self.fail(error);
            }
            if matches!(self.options.whitespace, Whitespace::Wrapped { .. }) {
                self.row_symbols += 1;
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let FrameState::NeedLf { cr_offset } = self.frame_state {
            return self.fail(Error::BadWrap { offset: cr_offset });
        }
        if let FrameState::NeedSymbol { separator_offset } = self.frame_state {
            return self.fail(Error::BadWrap { offset: separator_offset });
        }

        if let Some(offset) = self.pending_second_pad {
            return self.fail(Error::BadPadding { offset });
        }

        let mut out = Vec::new();
        if self.group_len > 0 {
            match self.options.padding {
                Padding::Required => {
                    return self.fail(Error::Truncated {
                        offset: self.offsets[0],
                    });
                }
                Padding::Forbidden | Padding::Optional => {
                    if self.group_len == 1 {
                        return self.fail(Error::Truncated {
                            offset: self.offsets[0],
                        });
                    }
                    let a = self.group[0];
                    let b = self.group[1];
                    out.push((a << 2) | (b >> 4));
                    if self.group_len == 3 {
                        let c = self.group[2];
                        out.push((b << 4) | (c >> 2));
                    }
                }
            }
        }

        self.finished = true;
        Ok(out)
    }
}

    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    const STANDARD: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const URL_SAFE: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
    match alphabet {
        Alphabet::Standard => STANDARD,
        Alphabet::UrlSafe => URL_SAFE,
    }
}

fn digit_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let table = alphabet_bytes(alphabet);
    table.iter().position(|&b| b == byte).map(|n| n as u8)
}

fn append_symbol(
    out: &mut String,
    symbol: u8,
    options: Options,
    row_symbols: &mut usize,
) {
    if let Whitespace::Wrapped { width, ending } = options.whitespace {
        if *row_symbols == width {
            match ending {
                LineEnding::Lf => out.push('\n'),
                LineEnding::CrLf => out.push_str("\r\n"),
            }
            *row_symbols = 0;
        }
        *row_symbols += 1;
    }
    out.push(symbol as char);
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut out = encoder.push(input)?;
    out.push_str(&encoder.finish()?);
    Ok(out)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let table = alphabet_bytes(self.options.alphabet);
        let mut out = String::new();
        let mut row_symbols = 0usize;
        let mut i = 0usize;

        if self.tail_len != 0 {
            while self.tail_len < 2 && i < input.len() {
                self.tail[self.tail_len] = input[i];
                self.tail_len += 1;
                i += 1;
            }
            if self.tail_len == 2 && i < input.len() {
                let a = self.tail[0];
                let b = self.tail[1];
                let c = input[i];
                i += 1;
                append_symbol(&mut out, table[(a >> 2) as usize], self.options, &mut row_symbols);
                append_symbol(
                    &mut out,
                    table[(((a & 0x03) << 4) | (b >> 4)) as usize],
                    self.options,
                    &mut row_symbols,
                );
                append_symbol(
                    &mut out,
                    table[(((b & 0x0f) << 2) | (c >> 6)) as usize],
                    self.options,
                    &mut row_symbols,
                );
                append_symbol(&mut out, table[(c & 0x3f) as usize], self.options, &mut row_symbols);
                self.tail_len = 0;
            }
        }

        while i + 3 <= input.len() {
            let a = input[i];
            let b = input[i + 1];
            let c = input[i + 2];
            i += 3;
            append_symbol(&mut out, table[(a >> 2) as usize], self.options, &mut row_symbols);
            append_symbol(
                &mut out,
                table[(((a & 0x03) << 4) | (b >> 4)) as usize],
                self.options,
                &mut row_symbols,
            );
            append_symbol(
                &mut out,
                table[(((b & 0x0f) << 2) | (c >> 6)) as usize],
                self.options,
                &mut row_symbols,
            );
            append_symbol(&mut out, table[(c & 0x3f) as usize], self.options, &mut row_symbols);
        }

        while i < input.len() {
            self.tail[self.tail_len] = input[i];
            self.tail_len += 1;
            i += 1;
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut out = String::new();
        let mut row_symbols = 0usize;
        let table = alphabet_bytes(self.options.alphabet);

        match self.tail_len {
            0 => {}
            1 => {
                let a = self.tail[0];
                append_symbol(&mut out, table[(a >> 2) as usize], self.options, &mut row_symbols);
                append_symbol(
                    &mut out,
                    table[((a & 0x03) << 4) as usize],
                    self.options,
                    &mut row_symbols,
                );
                if self.options.padding != Padding::Forbidden {
                    append_symbol(&mut out, b'=', self.options, &mut row_symbols);
                    append_symbol(&mut out, b'=', self.options, &mut row_symbols);
                }
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                append_symbol(&mut out, table[(a >> 2) as usize], self.options, &mut row_symbols);
                append_symbol(
                    &mut out,
                    table[(((a & 0x03) << 4) | (b >> 4)) as usize],
                    self.options,
                    &mut row_symbols,
                );
                append_symbol(
                    &mut out,
                    table[((b & 0x0f) << 2) as usize],
                    self.options,
                    &mut row_symbols,
                );
                if self.options.padding != Padding::Forbidden {
                    append_symbol(&mut out, b'=', self.options, &mut row_symbols);
                }
            }
            _ => unreachable!(),
        }
        self.tail_len = 0;
        Ok(out)
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    value: u8,
    offset: usize,
    is_padding: bool,
}

#[derive(Clone, Copy)]
enum WrapState {
    Symbols { count: usize },
    NeedSeparator { first_offset: usize, seen_cr: bool },
    NeedSymbol { first_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: [Symbol; 4],
    group_len: usize,
    offset: usize,
    terminal_padding: bool,
    wrap_state: WrapState,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: [
                Symbol { value: 0, offset: 0, is_padding: false };
                4
            ],
            group_len: 0,
            offset: 0,
            terminal_padding: false,
            wrap_state: WrapState::Symbols { count: 0 },
            finished: false,
            sticky_error: None,
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn emit_group(&mut self, output: &mut Vec<u8>, current: Symbol) -> Result<(), Error> {
        let pos = self.group_len;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset: current.offset });
        }

        if pos == 2 && self.group[2].is_padding && !current.is_padding {
            return Err(Error::BadPadding { offset: current.offset });
        }

        if current.is_padding {
            if pos <= 1 || (pos == 2 && self.options.padding == Padding::Forbidden) {
                return Err(Error::BadPadding { offset: current.offset });
            }
            if pos == 2 {
                self.group[2] = current;
                self.group_len = 3;
                return Ok(());
            }
            if pos == 3 {
                let a = self.group[0].value;
                let b = self.group[1].value;
                if self.group[2].is_padding {
                    output.push((a << 2) | (b >> 4));
                } else {
                    let c = self.group[2].value;
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                }
                self.group_len = 0;
                self.terminal_padding = true;
                return Ok(());
            }
        }

        if self.group_len == 3 && self.group[2].is_padding {
            return Err(Error::BadPadding { offset: current.offset });
        }

        let value = match digit_value(current.value, self.options.alphabet) {
            Some(v) => v,
            None => return Err(Error::InvalidByte { offset: current.offset, byte: current.value }),
        };

        self.group[self.group_len] = Symbol { value, ..current };
        self.group_len += 1;

        if self.group_len == 4 {
            let a = self.group[0].value;
            let b = self.group[1].value;
            let c = self.group[2].value;
            let d = self.group[3].value;
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
            self.group_len = 0;
        }
        Ok(())
    }

    fn scan_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.scan_symbol(byte, offset, output),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.scan_symbol(byte, offset, output)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                let state = self.wrap_state;
                match state {
                    WrapState::NeedSeparator { first_offset, seen_cr } => {
                        match ending {
                            LineEnding::Lf => {
                                if byte != b'\n' {
                                    return Err(Error::BadWrap { offset });
                                }
                                self.wrap_state = WrapState::NeedSymbol { first_offset };
                                Ok(())
                            }
                            LineEnding::CrLf => {
                                if !seen_cr {
                                    if byte != b'\r' {
                                        return Err(Error::BadWrap { offset });
                                    }
                                    self.wrap_state = WrapState::NeedSeparator {
                                        first_offset: offset,
                                        seen_cr: true,
                                    };
                                    Ok(())
                                } else if byte != b'\n' {
                                    return Err(Error::BadWrap { offset });
                                } else {
                                    self.wrap_state = WrapState::NeedSymbol { first_offset };
                                    Ok(())
                                }
                            }
                        }
                    }
                    WrapState::NeedSymbol { .. } => {
                        if byte == b'\n' || byte == b'\r' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = WrapState::Symbols { count: 0 };
                        self.scan_symbol(byte, offset, output)
                    }
                    WrapState::Symbols { count } => {
                        if count == width {
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.wrap_state = WrapState::NeedSymbol { first_offset: offset };
                                    Ok(())
                                }
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.wrap_state = WrapState::NeedSeparator {
                                        first_offset: offset,
                                        seen_cr: false,
                                    };
                                    Ok(())
                                }
                                _ => Err(Error::BadWrap { offset }),
                            }
                        } else if byte == b'\n' || byte == b'\r' {
                            Err(Error::BadWrap { offset })
                        } else {
                            self.wrap_state = WrapState::Symbols { count: count + 1 };
                            self.scan_symbol(byte, offset, output)
                        }
                    }
                }
            }
        }
    }

    fn scan_symbol(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        self.emit_group(
            output,
            Symbol {
                value: byte,
                offset,
                is_padding: byte == b'=',
            },
        )
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset = self.offset.saturating_add(1);
            if let Err(error) = self.scan_byte(byte, offset, &mut output) {
                return Err(self.fail(error));
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { ending, .. } = self.options.whitespace {
            match self.wrap_state {
                WrapState::NeedSeparator { first_offset, seen_cr: true }
                    if ending == LineEnding::CrLf =>
                {
                    return Err(self.fail(Error::BadWrap { offset: first_offset }));
                }
                WrapState::NeedSeparator { first_offset, .. }
                | WrapState::NeedSymbol { first_offset } => {
                    return Err(self.fail(Error::BadWrap { offset: first_offset }));
                }
                WrapState::Symbols { .. } => {}
            }
        }

        if self.group_len == 3 && self.group[2].is_padding {
            return Err(self.fail(Error::BadPadding {
                offset: self.group[2].offset,
            }));
        }

        let mut output = Vec::new();
        match self.group_len {
            0 => {}
            1 => {
                return Err(self.fail(Error::Truncated {
                    offset: self.group[0].offset,
                }));
            }
            2 | 3 => match self.options.padding {
                Padding::Required => {
                    return Err(self.fail(Error::Truncated {
                        offset: self.group[0].offset,
                    }));
                }
                Padding::Forbidden if self.group_len == 3 => {
                    output.push(
                        (self.group[0].value << 2) | (self.group[1].value >> 4),
                    );
                    output.push(
                        (self.group[1].value << 4) | (self.group[2].value >> 2),
                    );
                }
                Padding::Forbidden | Padding::Optional => {
                    output.push(
                        (self.group[0].value << 2) | (self.group[1].value >> 4),
                    );
                    if self.group_len == 3 {
                        output.push(
                            (self.group[1].value << 4) | (self.group[2].value >> 2),
                        );
                    }
                }
            },
            _ => unreachable!(),
        }

        self.group_len = 0;
        self.finished = true;
        Ok(output)
    }
}
