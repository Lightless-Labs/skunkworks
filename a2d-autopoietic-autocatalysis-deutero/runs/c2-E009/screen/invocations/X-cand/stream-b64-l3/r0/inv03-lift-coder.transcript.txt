    fn scan_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if let Some(pending_offset) = self.pending_second_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_second_padding = None;
            self.group.push(Symbol { byte, offset });
            if self.group.len() == 4 {
                let first = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
                let second = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
                out.push((first << 2) | (second >> 4));
                self.group.clear();
                self.terminal_padding = true;
            }
            let _ = pending_offset;
            return Ok(());
        }

        let position = self.group.len();
        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_second_padding = Some(offset);
                    self.group.push(Symbol { byte, offset });
                    return Ok(());
                }

                    out.push((second << 4) | (third >> 2));
                    self.group.clear();
                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }

            .ok_or(Error::InvalidByte { offset, byte })?;

            let a = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
            let b = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();

            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);

        }

                            frame.separator = 0;
                            frame.separator_offset = None;
                            frame.after_separator = true;
                            Ok(())

                        (1, LineEnding::Lf, _) => Err(Error::BadWrap { offset }),

                        frame.separator_offset = Some(offset);

                        frame.separator_offset = Some(offset);

                    }

                    frame.after_separator = false;
                    self.scan_symbol(byte, offset, out)
                }
            }
        }

                    Some(Error::BadWrap { offset })
                } else if self.frame.after_separator {
                    Some(Error::BadWrap {
                        offset: self.absolute_offset.saturating_sub(1),
                    })
                } else {
                    None
                }
            }
            _ => None,
        };
        if let Some(error) = error {
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        let residual = self.group.len();
        let result = match residual {
            0 => Ok(Vec::new()),
            1 => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
            2 if self.options.padding != Padding::Required => {
                let a = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
                Ok(vec![(a << 2) | (b >> 4)])
            }
            3 if self.options.padding != Padding::Required => {
                let a = alphabet_digit(self.group[0].byte, self.options.alphabet).unwrap();
                let b = alphabet_digit(self.group[1].byte, self.options.alphabet).unwrap();
                let c = alphabet_digit(self.group[2].byte, self.options.alphabet).unwrap();
                Ok(vec![(a << 2) | (b >> 4), (b << 4) | (c >> 2)])
            }
            _ => Err(Error::Truncated {
                offset: self.group[0].offset,
            }),
        };
        match result {
            Ok(bytes) => {
                self.finished = true;
                Ok(bytes)
            }
            Err(error) => {
                self.sticky_error = Some(error.clone());
                Err(error)
            }
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
