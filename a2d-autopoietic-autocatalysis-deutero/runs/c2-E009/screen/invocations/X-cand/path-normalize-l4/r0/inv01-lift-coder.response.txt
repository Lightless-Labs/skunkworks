use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();
    if let Some(at) = bytes.iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = bytes.iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = bytes
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn render(stack: &[String], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        String::from("/")
    } else {
        String::new()
    };

    for (index, component) in stack.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if stack.is_empty() && !absolute {
        result.push('.');
    }
    if directory && !stack.is_empty() {
        result.push('/');
    }
    result
}

fn reduce(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    let mut stack = Vec::new();
    let mut offset = 0usize;

    for component in path.split('/') {
        let at = offset;
        offset += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|previous| previous != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push(String::from(".."));
            }
        } else {
            stack.push(component.to_string());
        }
    }

    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let directory = directory_intent(path);
    let stack = reduce(path, absolute)?;
    Ok(render(&stack, absolute, directory))
}

fn validate_context(path: &str, cwd: bool) -> Result<String, PathError> {
    let validated = if path.starts_with('/') {
        normalize(path)
    } else {
        Err(PathError::Empty)
    };

    match validated {
        Ok(path) => Ok(path),
        Err(_) if cwd => Err(PathError::InvalidCwd),
        Err(_) => Err(PathError::InvalidHome),
    }
}

fn prepare_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut by_path = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let canonical_path = link.path.starts_with('/')
            && link.path != "/"
            && !link.path.ends_with('/')
            && normalize(&link.path).is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();

        if !canonical_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = by_path.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        by_path.insert(link.path.clone(), index);
    }

    Ok(by_path)
}

#[derive(Clone)]
struct Frame {
    raw: String,
    position: usize,
    active_link: Option<usize>,
}

fn resolve_engine(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    by_path: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd = validate_context(cwd, true)?;
    let home = match home {
        Some(path) => Some(validate_context(path, false)?),
        None => None,
    };

    validate_raw(path)?;
    let directory = directory_intent(path);

    let (start, request, request_offset) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home.as_deref().ok_or(PathError::MissingHome)?;
        if bytes.len() > 1 {
            (home, &path[2..], 2)
        } else {
            (home, "", 1)
        }
    } else if path.starts_with('/') {
        ("/", path, 0)
    } else {
        (cwd.as_str(), path, 0)
    };

    let mut stack: Vec<String> = Vec::new();
    let mut frames = Vec::new();

    if start != "/" {
        frames.push(Frame {
            raw: start.to_string(),
            position: 0,
            active_link: None,
        });
    }
    frames.push(Frame {
        raw: request.to_string(),
        position: request_offset,
        active_link: None,
    });

    let mut active = Vec::new();
    let mut expansions = 0usize;

    while !frames.is_empty() {
        let frame_index = frames.len() - 1;

        if frames[frame_index].position >= frames[frame_index].raw.len() {
            let finished = frames.pop().unwrap();
            if let Some(link_index) = finished.active_link {
                if active.last() == Some(&link_index) {
                    active.pop();
                }
            }
            continue;
        }

        let raw = frames[frame_index].raw.clone();
        let position = frames[frame_index].position;
        let bytes = raw.as_bytes();
        let mut end = position;

        while end < bytes.len() && bytes[end] != b'/' {
            end += 1;
        }

        let component = &raw[position..end];
        frames[frame_index].position = if end == bytes.len() { end } else { end + 1 };

        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at: position });
            }
            continue;
        }

        stack.push(component.to_string());

        let mut key = String::from("/");
        for (index, part) in stack.iter().enumerate() {
            if index > 0 {
                key.push('/');
            }
            key.push_str(part);
        }

        if let Some(&link_index) = by_path.get(&key) {
            if active.contains(&link_index) {
                return Err(PathError::Loop { index: link_index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index: link_index });
            }

            expansions += 1;
            active.push(link_index);

            if links[link_index].target.starts_with('/') {
                stack.clear();
            } else {
                stack.pop();
            }

            frames.push(Frame {
                raw: links[link_index].target.clone(),
                position: 0,
                active_link: Some(link_index),
            });
        }
    }

    Ok(render(&stack, true, directory))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    resolve_engine(path, cwd, home, &[], &HashMap::new(), 0)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_context(cwd, true)?;
    if let Some(home) = home {
        validate_context(home, false)?;
    }
    let by_path = prepare_links(links)?;
    resolve_engine(path, cwd, home, links, &by_path, max_expansions)
}

fn absolute_components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_string)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let mut common = 0usize;
    while common < base.len()
        && common < target.len()
        && base[common] == target[common]
    {
        common += 1;
    }

    let mut parts = Vec::new();
    for _ in common..base.len() {
        parts.push(String::from(".."));
    }
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        String::from(".")
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = absolute_components(&resolved_base);
    let target_components = absolute_components(&resolved_target);
    let target_is_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probed = match resolve_links(&link.path, "/", None, links, max_expansions) {
            Ok(path) => path,
            Err(_) => continue,
        };
        let prefix = absolute_components(&probed);

        if prefix.len() <= target_components.len()
            && prefix
                .iter()
                .zip(target_components.iter())
                .all(|(left, right)| left == right)
        {
            let key_components = absolute_components(&link.path);
            let mut candidate = route(&base_components, &key_components);
            let suffix = &target_components[prefix.len()..];

            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();

    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_is_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if absolute_components(&resolved) == target_components
                && !valid.contains(&candidate)
            {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|left, right| {
        left.len()
            .cmp(&right.len())
            .then_with(|| left.as_bytes().cmp(right.as_bytes()))
    });

    valid
        .into_iter()
        .next()
        .ok_or(PathError::Escape { at: 0 })
}
