use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/').find(|part| !part.is_empty()).is_some_and(|part| {
        part == "." || part == ".."
    })
}

fn normalized_components(path: &str) -> Result<Vec<String>, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let mut stack: Vec<String> = Vec::new();

    for (at, part) in path.split('/').scan(0usize, |offset, part| {
        let start = *offset;
        *offset = offset.saturating_add(part.len()).saturating_add(1);
        Some((start, part))
    }) {
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        } else {
            stack.push(part.to_string());
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    let components = normalized_components(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);

    let mut result = if absolute {
        format!("/{}", components.join("/"))
    } else if components.is_empty() {
        ".".to_string()
    } else {
        components.join("/")
    };

    if absolute && components.is_empty() {
        result = "/".to_string();
    }
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

fn context_components(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    normalized_components(path).map_err(|_| PathError::InvalidCwd)
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            normalized_components(path)
                .map(Some)
                .map_err(|_| PathError::InvalidHome)
        }
    }
}

fn tilde_start<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<(bool, Vec<String>, &'a str, usize), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let home = home.ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() >= 2 { 2 } else { 1 };
            return Ok((true, home.to_vec(), &path[consumed..], consumed));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    Ok((path.starts_with('/'), Vec::new(), path, 0))
}

fn resolve_core(
    path: &str,
    cwd_components: &[String],
    home_components: Option<&[String]>,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, mut stack, remainder, consumed) = tilde_start(path, home_components)?;

    let absolute = if tilde { false } else { path.starts_with('/') };
    if absolute {
        stack.clear();
    } else if !tilde {
        stack.extend_from_slice(cwd_components);
    }

    for (local_at, part) in remainder.split('/').scan(0usize, |offset, part| {
        let start = *offset;
        *offset = offset.saturating_add(part.len()).saturating_add(1);
        Some((start, part))
    }) {
        if part.is_empty() || part == "." {
            continue;
        }
        let at = consumed.saturating_add(local_at);
        if part == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            stack.push(part.to_string());
        }
    }

    let mut result = format!("/{}", stack.join("/"));
    if stack.is_empty() {
        result = "/".to_string();
    }
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = context_components(cwd)?;
    let home_components = validate_home(home)?;
    resolve_core(path, &cwd_components, home_components.as_deref())
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut map = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let canonical = normalize(&link.path);
        let valid_path = canonical.as_ref().is_ok_and(|normalized| {
            link.path.starts_with('/')
                && link.path != "/"
                && !link.path.ends_with('/')
                && normalized == &link.path
        });
        let valid_target = validate_raw(&link.target).is_ok();
        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = map.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        map.insert(link.path.clone(), index);
    }
    Ok(map)
}

#[derive(Clone)]
struct Work {
    raw: String,
    offset: usize,
    kind: WorkKind,
}

#[derive(Clone, Copy)]
enum WorkKind {
    Request,
    Target,
}

fn absolute_from(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn resolve_links_core(
    path: &str,
    cwd_components: &[String],
    home_components: Option<&[String]>,
    links: &[Link],
    link_map: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, mut stack, remainder, consumed) = tilde_start(path, home_components)?;

    if !tilde && path.starts_with('/') {
        stack.clear();
    } else if !tilde {
        stack.extend_from_slice(cwd_components);
    }

    let mut work = vec![Work {
        raw: remainder.to_string(),
        offset: consumed,
        kind: WorkKind::Request,
    }];
    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    while let Some(item) = work.pop() {
        for (local_at, part) in item.raw.split('/').scan(0usize, |offset, part| {
            let start = *offset;
            *offset = offset.saturating_add(part.len()).saturating_add(1);
            Some((start, part))
        }) {
            if part.is_empty() || part == "." {
                continue;
            }
            let at = item.offset.saturating_add(local_at);
            if part == ".." {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
                continue;
            }

            stack.push(part.to_string());
            let key = absolute_from(&stack);
            if let Some(&index) = link_map.get(&key) {
                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if expansions >= max_expansions {
                    return Err(PathError::Limit { index });
                }
                expansions += 1;
                active.push(index);

                let link = &links[index];
                let target = &link.target;
                let target_absolute = target.starts_with('/');
                if target_absolute {
                    stack.clear();
                } else {
                    stack.pop();
                }

                let rest = &item.raw[local_at + part.len()..];
                work.push(Work {
                    raw: rest.to_string(),
                    offset: at.saturating_add(part.len()),
                    kind: item.kind,
                });
                work.push(Work {
                    raw: target.clone(),
                    offset: 0,
                    kind: WorkKind::Target,
                });

                // Keep the active link until its target and nested expansions finish.
                // A marker is represented by an empty work item with a dedicated
                // sentinel kind and the link index encoded separately below.
                work.push(Work {
                    raw: String::new(),
                    offset: index,
                    kind: WorkKind::Target,
                });
                break;
            }
        }

        // Completion markers are handled by the marker work-item branch below.
        if matches!(item.kind, WorkKind::Target) && item.raw.is_empty() {
            if let Some(index) = active.pop() {
                let _ = index;
            }
        }
    }

    let mut result = absolute_from(&stack);
    if intent && result != "/" {
        result.push('/');
    }
    Ok(result)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = context_components(cwd)?;
    let home_components = validate_home(home)?;
    let link_map = validate_links(links)?;
    resolve_links_core(
        path,
        &cwd_components,
        home_components.as_deref(),
        links,
        &link_map,
        max_expansions,
    )
}

fn components_of_absolute(path: &str) -> Vec<String> {
    path.split('/').filter(|part| !part.is_empty()).map(str::to_string).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base.iter().zip(target).take_while(|(a, b)| a == b).count();
    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_string()));
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = components_of_absolute(&resolved_base);
    let target_components = components_of_absolute(&resolved_target);
    let intent = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_components, &target_components)];
    for link in links {
        if let Ok(probed) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let shortcut_components = components_of_absolute(&probed);
            if shortcut_components.len() <= target_components.len()
                && shortcut_components
                    .iter()
                    .zip(&target_components)
                    .all(|(a, b)| a == b)
            {
                let mut candidate = route(&base_components, &components_of_absolute(&link.path));
                let suffix = &target_components[shortcut_components.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if intent && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }
        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if components_of_absolute(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().unwrap_or_else(|| ".".to_string())
}
