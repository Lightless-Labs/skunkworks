#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }

    let without_final_lf = &text[..text.len() - 1];
    without_final_lf.split('\n').collect()
}

fn validate(text: &str, side: Side) -> Result<(), DiffError> {
    if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn { side, byte });
    }
    if !text.is_empty() && !text.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side });
    }
    Ok(())
}

fn lcs_table(old: &[&str], new: &[&str]) -> Vec<Vec<usize>> {
    let rows = old.len() + 1;
    let cols = new.len() + 1;
    let mut table = vec![vec![0; cols]; rows];

    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            table[i][j] = if old[i] == new[j] {
                table[i + 1][j + 1] + 1
            } else {
                table[i + 1][j].max(table[i][j + 1])
            };
        }
    }

    table
}

fn canonical_script(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let table = lcs_table(old, new);
    let mut edits = Vec::new();
    let (mut i, mut j) = (0, 0);

    while i < old.len() && j < new.len() {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_score = table[i + 1][j];
            let insert_score = table[i][j + 1];
            if delete_score > insert_score
                || (delete_score == insert_score && (i + j) % 2 == 0)
            {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].to_owned(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].to_owned(),
                });
                j += 1;
            }
        }
    }

    while i < old.len() {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].to_owned(),
        });
        i += 1;
    }
    while j < new.len() {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].to_owned(),
        });
        j += 1;
    }

    edits
}

fn is_keep(edit: &Edit) -> bool {
    matches!(edit, Edit::Keep { .. })
}

fn append_hunk(output: &mut String, edits: &[Edit], start: usize, end: usize) {
    let old_count = edits[start..end]
        .iter()
        .filter(|edit| !matches!(edit, Edit::Insert { .. }))
        .count();
    let new_count = edits[start..end]
        .iter()
        .filter(|edit| !matches!(edit, Edit::Delete { .. }))
        .count();

    let (old_before, new_before) = if start < edits.len() {
        match &edits[start] {
            Edit::Keep {
                old_index,
                new_index,
                ..
            } => (*old_index, *new_index),
            Edit::Delete { old_index, .. } => {
                let new_index = edits[start..]
                    .iter()
                    .find_map(|e| match e {
                        Edit::Keep { new_index, .. } | Edit::Insert { new_index, .. } => {
                            Some(*new_index)
                        }
                        _ => None,
                    })
                    .unwrap_or_else(|| {
                        edits[..start]
                            .iter()
                            .filter(|e| !matches!(e, Edit::Delete { .. }))
                            .count()
                    });
                (*old_index, new_index)
            }
            Edit::Insert { new_index, .. } => {
                let old_index = edits[start..]
                    .iter()
                    .find_map(|e| match e {
                        Edit::Keep { old_index, .. } | Edit::Delete { old_index, .. } => {
                            Some(*old_index)
                        }
                        _ => None,
                    })
                    .unwrap_or_else(|| {
                        edits[..start]
                            .iter()
                            .filter(|e| !matches!(e, Edit::Insert { .. }))
                            .count()
                    });
                (old_index, *new_index)
            }
        }
    } else {
        (
            edits
                .iter()
                .filter(|e| !matches!(e, Edit::Insert { .. }))
                .count(),
            edits
                .iter()
                .filter(|e| !matches!(e, Edit::Delete { .. }))
                .count(),
        )
    };

    let old_start = if old_count > 0 {
        old_before + 1
    } else {
        old_before
    };
    let new_start = if new_count > 0 {
        new_before + 1
    } else {
        new_before
    };

    output.push_str(&format!(
        "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
    ));

    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { line, .. } => {
                output.push(' ');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Delete { line, .. } => {
                output.push('-');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Insert { line, .. } => {
                output.push('+');
                output.push_str(line);
                output.push('\n');
            }
        }
    }
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }
    validate(old, Side::Old)?;
    validate(new, Side::New)?;

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);

    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let edits = canonical_script(&old_lines, &new_lines);
    if edits.iter().all(is_keep) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut pos = 0;
    while pos < edits.len() {
        if is_keep(&edits[pos]) {
            pos += 1;
            continue;
        }
        let start = pos;
        while pos < edits.len() && !is_keep(&edits[pos]) {
            pos += 1;
        }
        groups.push((start, pos));
    }

    let gap_limit = 2 * context;
    let mut merged: Vec<(usize, usize)> = Vec::new();
    for group in groups {
        if let Some(last) = merged.last_mut() {
            let gap = group.0 - last.1;
            if gap <= gap_limit {
                last.1 = group.1;
                continue;
            }
        }
        merged.push(group);
    }

    let mut output = String::from("--- before\n+++ after\n");
    let mut previous_end = 0;
    for (group_start, group_end) in merged {
        let mut start = group_start;
        let mut preceding = 0;
        while start > previous_end && preceding < context && is_keep(&edits[start - 1]) {
            start -= 1;
            preceding += 1;
        }

        let mut end = group_end;
        let mut following = 0;
        while end < edits.len() && following < context && is_keep(&edits[end]) {
            end += 1;
            following += 1;
        }

        append_hunk(&mut output, &edits, start, end);
        previous_end = end;
    }

    Ok(output)
}
