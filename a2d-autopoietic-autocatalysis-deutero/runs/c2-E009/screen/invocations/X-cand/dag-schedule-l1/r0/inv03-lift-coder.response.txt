#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Plan {
    pub order: Vec<usize>,
    pub times: Vec<Timing>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let task_count = tasks.len();

    // Validate every dependency before doing any scheduling work.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = Vec::new();
    let mut children = Vec::new();
    for task in tasks {
        indegree.push(task.deps.len());
        children.push(Vec::new());
    }
    for (task_id, task) in tasks.iter().enumerate() {
        for &dependency in &task.deps {
            children[dependency].push(task_id);
        }
    }

    // Kahn traversal, always selecting the smallest currently available ID.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::new();
    let mut remaining = indegree;
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            remaining[child] -= 1;
            if remaining[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != task_count {
        let mut blocked = Vec::new();
        for (task_id, &degree) in remaining.iter().enumerate() {
            if degree != 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute earliest timings in Kahn order.
    let mut starts = Vec::new();
    let mut finishes = Vec::new();
    for _ in 0..task_count {
        starts.push(0_u64);
        finishes.push(0_u64);
    }

    let mut makespan = 0_u64;
    for &task_id in &order {
        let start = starts[task_id];
        let finish = match start.checked_add(tasks[task_id].duration) {
            Some(finish) => finish,
            None => return Err(Error::Overflow { task: task_id }),
        };

        finishes[task_id] = finish;
        if finish > makespan {
            makespan = finish;
        }

        for &child in &children[task_id] {
            if finish > starts[child] {
                starts[child] = finish;
            }
        }
    }

    let mut times = Vec::new();
    for task_id in 0..task_count {
        times.push(Timing {
            start: starts[task_id],
            finish: finishes[task_id],
        });
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
