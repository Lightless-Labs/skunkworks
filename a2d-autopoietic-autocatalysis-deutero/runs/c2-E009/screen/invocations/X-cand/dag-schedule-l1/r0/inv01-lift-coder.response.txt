```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Plan {
    pub order: Vec<usize>,
    pub times: Vec<Timing>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let task_count = tasks.len();

    // Validate every dependency before doing any scheduling work.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children: Vec<Vec<usize>> = (0..task_count).map(|_| Vec::new()).collect();

    for (task_id, task) in tasks.iter().enumerate() {
        for &dependency in &task.deps {
            children[dependency].push(task_id);
        }
    }

    // Keep available IDs sorted so each iteration removes the smallest one.
    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::new();
    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != task_count {
        let blocked = indegree
            .iter()
            .enumerate()
            .filter_map(|(task_id, &degree)| (degree != 0).then_some(task_id))
            .collect();
        return Err(Error::Cycle { blocked });
    }

    let mut starts = vec![0u64; task_count];
    let mut finishes = vec![0u64; task_count];
    let mut makespan = 0u64;

    for &task_id in &order {
        let start = starts[task_id];
        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;
        finishes[task_id] = finish;
        makespan = makespan.max(finish);

        for &child in &children[task_id] {
            starts[child] = starts[child].max(finish);
        }
    }

    let times = starts
        .into_iter()
        .zip(finishes)
        .map(|(start, finish)| Timing { start, finish })
        .collect();

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
```
