fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| constraint_matches(*constraint, version))
}

fn canonical_constraints(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut result = constraints.to_vec();
    result.sort();
    result.dedup();
    result
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints)
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut encoded = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    encoded
}

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    by_name: &std::collections::BTreeMap<String, usize>,
    assigned: &std::collections::BTreeMap<String, Version>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in assigned {
        if let Some(&package_index) = by_name.get(name) {
            if let Some(release) = catalog[package_index]
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for dependency in &release.deps {
                    active
                        .entry(dependency.package.clone())
                        .or_default()
                        .push(dependency.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical_constraints(constraints);
    }
    active
}

#[derive(Clone)]
struct SearchResult {
    best: Option<std::collections::BTreeMap<String, Version>>,
    best_cost: usize,
    first_failure: Option<ResolveError>,
}

struct SearchContext<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    by_name: std::collections::BTreeMap<String, usize>,
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
}

impl SearchContext<'_> {
    fn lock_cost(&self, assigned: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|lock| assigned.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn vector_better(
        &self,
        candidate: &std::collections::BTreeMap<String, Version>,
        current: &std::collections::BTreeMap<String, Version>,
    ) -> bool {
        let mut names: Vec<&String> = self.by_name.keys().collect();
        names.sort();
        for name in names {
            match (candidate.get(name), current.get(name)) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn record_failure(result: &mut SearchResult, error: ResolveError) {
        if result.first_failure.is_none() {
            result.first_failure = Some(error);
        }
    }

    fn visit(
        &mut self,
        assigned: &mut std::collections::BTreeMap<String, Version>,
        result: &mut SearchResult,
    ) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, self.catalog, &self.by_name, assigned);

        for name in active.keys() {
            if !self.by_name.contains_key(name) {
                Self::record_failure(
                    result,
                    ResolveError::Missing {
                        package: name.clone(),
                    },
                );
                return Ok(());
            }
        }

        for (name, version) in assigned.iter() {
            if let Some(constraints) = active.get(name) {
                if !matches_all(*version, constraints) {
                    Self::record_failure(
                        result,
                        ResolveError::Conflict {
                            package: name.clone(),
                            selected: *version,
                            requirements: constraints.clone(),
                        },
                    );
                    return Ok(());
                }
            }
        }

        for (name, constraints) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let package = &self.catalog[self.by_name[name]];
            if !package
                .releases
                .iter()
                .any(|release| matches_all(release.version, constraints))
            {
                Self::record_failure(
                    result,
                    ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    },
                );
                return Ok(());
            }
        }

        let next_name = active.keys().find(|name| !assigned.contains_key(*name)).cloned();
        let Some(name) = next_name else {
            let cost = self.lock_cost(assigned);
            let replace = match &result.best {
                None => true,
                Some(current) => {
                    cost < result.best_cost
                        || (cost == result.best_cost && self.vector_better(assigned, current))
                }
            };
            if replace {
                result.best = Some(assigned.clone());
                result.best_cost = cost;
            }
            return Ok(());
        };

        let constraints = active.get(&name).expect("active package has constraints");
        let package = &self.catalog[self.by_name[&name]];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| matches_all(*version, constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(index) = candidates.iter().position(|version| *version == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        let mut branch_failure = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), candidate);
            let child_result = self.visit(assigned, result);
            assigned.remove(&name);

            if let Err(error) = child_result {
                return Err(error);
            }
            if branch_failure.is_none() {
                branch_failure = result.first_failure.clone();
            }
        }

        if result.best.is_none() && result.first_failure.is_none() {
            if let Some(error) = branch_failure {
                Self::record_failure(result, error);
            }
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        validate_name(format!("root[{index}]"), &root.package)?;
    }
    for (package_index, package) in catalog.iter().enumerate() {
        validate_name(
            format!("catalog[{package_index}]"),
            &package.name,
        )?;
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                validate_name(
                    format!(
                        "catalog[{package_index}].release[{release_index}].dep[{dependency_index}]"
                    ),
                    &dependency.package,
                )?;
            }
        }
    }

    let mut seen_packages = std::collections::BTreeMap::<String, ()>::new();
    for package in catalog {
        if seen_packages.insert(package.name.clone(), ()).is_some() {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeMap::<Version, ()>::new();
        for release in &package.releases {
            if seen_versions.insert(release.version, ()).is_some() {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (index, lock) in locks.iter().enumerate() {
        validate_name(format!("lock[{index}]"), &lock.package)?;
    }
    let mut seen_locks = std::collections::BTreeMap::<String, ()>::new();
    for lock in locks {
        if seen_locks.insert(lock.package.clone(), ()).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut by_name = std::collections::BTreeMap::<String, usize>::new();
    for (index, package) in catalog.iter().enumerate() {
        by_name.insert(package.name.clone(), index);
    }

    let mut context = SearchContext {
        catalog,
        roots,
        by_name,
        locks,
        max_attempts,
        attempts: 0,
    };
    let mut result = SearchResult {
        best: None,
        best_cost: usize::MAX,
        first_failure: None,
    };
    let mut assigned = std::collections::BTreeMap::new();
    context.visit(&mut assigned, &mut result)?;

    let selected_map = match result.best {
        Some(best) => best,
        None => {
            return Err(result.first_failure.unwrap_or(ResolveError::NoMatch {
                package: String::new(),
                requirements: Vec::new(),
            }));
        }
    };

    let mut selected: Vec<Selection> = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selected.sort_by(|a, b| a.package.cmp(&b.package));

    let final_active = active_constraints(roots, catalog, &context.by_name, &selected_map);
    let mut before = std::collections::BTreeMap::<String, Version>::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut all_names = std::collections::BTreeMap::<String, ()>::new();
    for name in before.keys() {
        all_names.insert(name.clone(), ());
    }
    for name in selected_map.keys() {
        all_names.insert(name.clone(), ());
    }

    let mut changes = Vec::new();
    for name in all_names.keys() {
        let old = before.get(name).copied();
        let new = selected_map.get(name).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = context
                    .by_name
                    .get(name)
                    .map(|index| {
                        catalog[*index]
                            .releases
                            .iter()
                            .any(|release| release.version == old_version)
                    })
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(name)
                    .map(|constraints| !matches_all(old_version, constraints))
                    .unwrap_or(false)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name.clone(),
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn name_error(site: String, name: &str) -> ResolveError {
    ResolveError::InvalidName {
        site,
        name: name.to_string(),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn canonicalize(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_match(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonicalize(constraints.to_vec())
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(value: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {

            format!("invalid-name|site={}|name={}", encode_text(site), encode_text(name))

#[derive(Clone)]
struct SearchResult {
    selected: Vec<Selection>,
    constraints: Vec<(String, Vec<Constraint>)>,
}

struct SearchState<'a> {
    catalog: &'a [Package],
    package_indices: Vec<usize>,
    roots: &'a [Requirement],
    lock_versions: Vec<(String, Version)>,
    max_attempts: usize,
    attempts: usize,
    best: Option<SearchResult>,
    best_cost: usize,
    first_failure: Option<ResolveError>,
}

impl<'a> SearchState<'a> {
    fn package_index(&self, name: &str) -> Option<usize> {
        self.package_indices
            .binary_search_by(|&index| self.catalog[index].name.as_str().cmp(name))
            .ok()
            .map(|pos| self.package_indices[pos])
    }

    fn lock_version(&self, name: &str) -> Option<Version> {
        self.lock_versions
            .binary_search_by(|(package, _)| package.as_str().cmp(name))
            .ok()
            .map(|pos| self.lock_versions[pos].1)
    }

    fn active_constraints(
        &self,
        assignments: &[Option<usize>],
    ) -> Vec<(String, Vec<Constraint>)> {
        let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

        for root in self.roots {
            if let Some(pos) = active
                .binary_search_by(|(package, _)| package.as_str().cmp(&root.package))
                .ok()
            {
                active[pos].1.push(root.constraint);
            } else {
                let pos = active
                    .binary_search_by(|(package, _)| package.as_str().cmp(&root.package))
                    .unwrap_err();
                active.insert(pos, (root.package.clone(), vec![root.constraint]));
            }
        }

        for (package_pos, assignment) in assignments.iter().enumerate() {
            if let Some(release_index) = assignment {
                let package = &self.catalog[self.package_indices[package_pos]];
                let release = &package.releases[*release_index];
                for dep in &release.deps {
                    if let Some(pos) = active
                        .binary_search_by(|(name, _)| name.as_str().cmp(&dep.package))
                        .ok()
                    {
                        active[pos].1.push(dep.constraint);
                    } else {
                        let pos = active
                            .binary_search_by(|(name, _)| name.as_str().cmp(&dep.package))
                            .unwrap_err();
                        active.insert(pos, (dep.package.clone(), vec![dep.constraint]));
                    }
                }
            }
        }

        for (_, constraints) in &mut active {
            *constraints = canonicalize(std::mem::take(constraints));
        }
        active
    }

    fn fail(&mut self, error: ResolveError) {
        if self.first_failure.is_none() {
            self.first_failure = Some(error);
        }
    }

    fn solution_cost(&self, assignments: &[Option<usize>]) -> usize {
        self.lock_versions
            .iter()
            .filter(|(name, version)| {
                let Some(package_index) = self.package_index(name) else {
                    return true;
                };
                let position = self.package_indices
                    .binary_search_by(|&index| self.catalog[index].name.as_str().cmp(name))
                    .unwrap();
                let Some(release_index) = assignments[position] else {
                    return true;
                };
                self.catalog[package_index].releases[release_index].version != *version
            })
            .count()
    }

    fn record_solution(
        &mut self,
        assignments: &[Option<usize>],
        active: &[(String, Vec<Constraint>)],
    ) {
        let mut selected = Vec::new();
        for (position, assignment) in assignments.iter().enumerate() {
            if let Some(release_index) = assignment {
                let package = &self.catalog[self.package_indices[position]];
                selected.push(Selection {
                    package: package.name.clone(),
                    version: package.releases[*release_index].version,
                });
            }
        }
        let cost = self.solution_cost(assignments);
        let candidate = SearchResult {
            selected,
            constraints: active.to_vec(),
        };

        let replace = match &self.best {
            None => true,
            Some(best) if cost < self.best_cost => true,
            Some(best) if cost == self.best_cost => {
                compare_vectors(&candidate.selected, &best.selected) == std::cmp::Ordering::Greater
            }
            _ => false,
        };
        if replace {
            self.best_cost = cost;
            self.best = Some(candidate);
        }
    }

    fn visit(&mut self, assignments: &mut [Option<usize>]) -> bool {
        let active = self.active_constraints(assignments);

        for (name, _) in &active {
            if self.package_index(name).is_none() {
                self.fail(ResolveError::Missing {
                    package: name.clone(),
                });
                return true;
            }
        }

        for (name, constraints) in &active {
            if let Some(position) = self
                .package_indices
                .binary_search_by(|&index| self.catalog[index].name.as_str().cmp(name))
                .ok()
            {
                if let Some(release_index) = assignments[position] {
                    let package = &self.catalog[self.package_indices[position]];
                    let version = package.releases[release_index].version;
                    if !constraints_match(version, constraints) {
                        self.fail(ResolveError::Conflict {
                            package: name.clone(),
                            selected: version,
                            requirements: constraints.clone(),
                        });
                        return true;
                    }
                }
            }
        }

        for (name, constraints) in &active {
            let position = self
                .package_indices
                .binary_search_by(|&index| self.catalog[index].name.as_str().cmp(name))
                .unwrap();
            if assignments[position].is_none() {
                let package = &self.catalog[self.package_indices[position]];
                if !package
                    .releases
                    .iter()
                    .any(|release| constraints_match(release.version, constraints))
                {
                    self.fail(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                    return true;
                }
            }
        }

        let mut branch: Option<(usize, Vec<usize>)> = None;
        for (name, constraints) in &active {
            let position = self
                .package_indices
                .binary_search_by(|&index| self.catalog[index].name.as_str().cmp(name))
                .unwrap();
            if assignments[position].is_none() {
                let package = &self.catalog[self.package_indices[position]];
                let mut candidates: Vec<usize> = package
                    .releases
                    .iter()
                    .enumerate()
                    .filter(|(_, release)| {
                        constraints_match(release.version, constraints)
                    })
                    .map(|(index, _)| index)
                    .collect();
                candidates.sort_by(|&a, &b| {
                    package.releases[b]
                        .version
                        .cmp(&package.releases[a].version)
                });
                if let Some(lock) = self.lock_version(name) {
                    if let Some(lock_pos) = candidates
                        .iter()
                        .position(|&index| package.releases[index].version == lock)
                    {
                        let locked = candidates.remove(lock_pos);
                        candidates.insert(0, locked);
                    }
                }
                branch = Some((position, candidates));
                break;
            }
        }

        let Some((position, candidates)) = branch else {
            self.record_solution(assignments, &active);
            return false;
        };

        for release_index in candidates {
            if self.attempts == self.max_attempts {
                return true;
            }
            self.attempts += 1;
            assignments[position] = Some(release_index);
            let limit_reached = self.visit(assignments);
            assignments[position] = None;
            if limit_reached {
                return true;
            }
        }

        false
    }
}

fn compare_vectors(a: &[Selection], b: &[Selection]) -> std::cmp::Ordering {
    let mut ai = 0;
    let mut bi = 0;
    loop {
        match (a.get(ai), b.get(bi)) {
            (Some(left), Some(right)) if left.package == right.package => {
                match left.version.cmp(&right.version) {
                    std::cmp::Ordering::Equal => {
                        ai += 1;
                        bi += 1;
                    }
                    other => return other,
                }
            }
            (Some(left), Some(right)) => match left.package.cmp(&right.package) {
                std::cmp::Ordering::Less => return std::cmp::Ordering::Greater,
                std::cmp::Ordering::Greater => return std::cmp::Ordering::Less,
                std::cmp::Ordering::Equal => unreachable!(),
            },
            (Some(_), None) => return std::cmp::Ordering::Greater,
            (None, Some(_)) => return std::cmp::Ordering::Less,
            (None, None) => return std::cmp::Ordering::Equal,
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(name_error(format!("root[{index}]"), &root.package));
        }
    }
    for (package_index, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(name_error(
                format!("catalog[{package_index}]"),
                &package.name,
            ));
        }
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dep_index, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(name_error(
                        format!("catalog[{package_index}].release[{release_index}].dep[{dep_index}]"),
                        &dep.package,
                    ));
                }
            }
        }
    }

    let mut package_names = Vec::new();
    for package in catalog {
        if package_names.iter().any(|name: &String| name == &package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        package_names.push(package.name.clone());
    }

    for package in catalog {
        let mut versions = Vec::new();
        for release in &package.releases {
            if versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            versions.push(release.version);
        }
    }

    for (index, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(name_error(format!("lock[{index}]"), &lock.package));
        }
    }

    let mut lock_names = Vec::new();
    for lock in locks {
        if lock_names.iter().any(|name: &String| name == &lock.package) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
        lock_names.push(lock.package.clone());
    }

    let mut package_indices: Vec<usize> = (0..catalog.len()).collect();
    package_indices.sort_by(|&a, &b| catalog[a].name.cmp(&catalog[b].name));

    let mut lock_versions: Vec<(String, Version)> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    lock_versions.sort_by(|a, b| a.0.cmp(&b.0));

    let mut state = SearchState {
        catalog,
        package_indices,
        roots,
        lock_versions,
        max_attempts,
        attempts: 0,
        best: None,
        best_cost: 0,
        first_failure: None,
    };

    let mut assignments = vec![None; catalog.len()];
    let limited = state.visit(&mut assignments);
    if limited {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some(best) = state.best else {
        return Err(state.first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let mut before_after_names = Vec::new();
    for lock in locks {
        before_after_names.push(lock.package.clone());
    }
    for selection in &best.selected {
        if !before_after_names.iter().any(|name| name == &selection.package) {
            before_after_names.push(selection.package.clone());
        }
    }
    before_after_names.sort();
    before_after_names.dedup();

    let mut changes = Vec::new();
    for name in before_after_names {
        let before = locks
            .iter()
            .find(|lock| lock.package == name)
            .map(|lock| lock.version);
        let after = best
            .selected
            .iter()
            .find(|selection| selection.package == name)
            .map(|selection| selection.version);
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let package = catalog.iter().find(|package| package.name == name);
                let locked_release_exists = package
                    .map(|package| {
                        package
                            .releases
                            .iter()
                            .any(|release| release.version == locked)
                    })
                    .unwrap_or(false);
                if !locked_release_exists {
                    ChangeReason::Unavailable
                } else {
                    let final_constraints = best
                        .constraints
                        .iter()
                        .find(|(package, _)| package == &name)
                        .map(|(_, constraints)| constraints.as_slice())
                        .unwrap_or(&[]);
                    if !constraints_match(locked, final_constraints) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: best.selected,
        changes,
    })
}

    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x
                && if x.major > 0 {
                    version.major == x.major
                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x
                }
        }
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphabetic()
            || byte.is_ascii_digit()
            || *byte == b'.'
            || *byte == b'_'
            || *byte == b'-'
        {
            out.push(*byte as char);
        } else {
            out.push('%');
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 15) as usize] as char);
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

struct Search<'a> {
    catalog: &'a [Package],
    package_index: std::collections::BTreeMap<&'a str, usize>,
    locks: std::collections::BTreeMap<&'a str, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>, std::collections::BTreeMap<String, Vec<Constraint>>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn active(
        &self,
        assigned: &std::collections::BTreeMap<String, Version>,
    ) -> std::collections::BTreeMap<String, Vec<Constraint>> {
        let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
        for root in ROOTS.with(|roots| roots.borrow().iter().cloned().collect::<Vec<_>>()) {
            active
                .entry(root.package)
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in assigned {
            if let Some(&index) = self.package_index.get(name.as_str()) {
                if let Some(release) = self.catalog[index]
                    .releases
                    .iter()
                    .find(|r| r.version == *version)
                {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }
        active
    }

    fn inspect(
        &mut self,
        assigned: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let active = self.active(assigned);

        if let Some(name) = active
            .keys()
            .find(|name| !self.package_index.contains_key(name.as_str()))
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assigned.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints
                    .iter()
                    .all(|c| constraint_matches(*c, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let package = &self.catalog[self.package_index[name.as_str()]];
            if !package.releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|c| constraint_matches(*c, release.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !assigned.contains_key(*name));
        let Some(name) = next.cloned() else {
            let selected: Vec<Selection> = assigned
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self
                .locks
                .iter()
                .filter(|(package, version)| {
                    assigned.get(**package).copied() != Some(**version)
                })
                .count();
            let better = match &self.best {
                None => true,
                Some((best_cost, best_selected, _)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        self.vector_better(&selected, best_selected)
                    }
                }
            };
            if better {
                self.best = Some((cost, selected, active));
            }
            return Ok(());
        };

        let package = &self.catalog[self.package_index[name.as_str()]];
        let constraints = &active[&name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|release| {
                constraints
                    .iter()
                    .all(|c| constraint_matches(*c, release.version))
            })
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(name.as_str()) {
            if let Some(pos) = candidates.iter().position(|version| version == lock) {
                let version = candidates.remove(pos);
                candidates.insert(0, version);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), candidate);
            let result = self.inspect(assigned);
            assigned.remove(&name);
            match result {
                Ok(()) => {}
                Err(ResolveError::Limit { .. }) => {
                    return Err(ResolveError::Limit {
                        attempts: self.max_attempts,
                    });
                }
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
        }
        Ok(())
    }

    fn vector_better(&self, left: &[Selection], right: &[Selection]) -> bool {
        let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        names.sort();
        names.dedup();
        let l: std::collections::BTreeMap<&str, Version> = left
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect();
        let r: std::collections::BTreeMap<&str, Version> = right
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect();
        for name in names {
            match (l.get(name), r.get(name)) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }
}

thread_local! {
    static ROOTS: std::cell::RefCell<Vec<Requirement>> = const { std::cell::RefCell::new(Vec::new()) };
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeMap::<&str, ()>::new();
    for package in catalog {
        if seen.insert(package.name.as_str(), ()).is_some() {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = std::collections::BTreeMap::new();
        for release in &package.releases {
            if versions.insert(release.version, ()).is_some() {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeMap::<&str, ()>::new();
    for lock in locks {
        if lock_names.insert(lock.package.as_str(), ()).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut package_index = std::collections::BTreeMap::new();
    for (i, package) in catalog.iter().enumerate() {
        package_index.insert(package.name.as_str(), i);
    }
    let mut lock_map = std::collections::BTreeMap::new();
    for lock in locks {
        lock_map.insert(lock.package.as_str(), lock.version);
    }

    ROOTS.with(|slot| {
        *slot.borrow_mut() = roots.to_vec();
    });

    let mut search = Search {
        catalog,
        package_index,
        locks: lock_map,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut assigned = std::collections::BTreeMap::new();
    let outcome = search.inspect(&mut assigned);
    ROOTS.with(|slot| slot.borrow_mut().clear());

    if let Err(ResolveError::Limit { .. }) = outcome {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }
    if let Some((_, selected, active)) = search.best {
        let chosen: std::collections::BTreeMap<&str, Version> = selected
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect();
        let mut all_names = std::collections::BTreeMap::<&str, Option<Version>>::new();
        for lock in locks {
            all_names.insert(lock.package.as_str(), Some(lock.version));
        }
        for item in &selected {
            all_names
                .entry(item.package.as_str())
                .or_insert(None);
        }
        for item in &selected {
            all_names.insert(item.package.as_str(), Some(item.version));
        }

        let mut changes = Vec::new();
        for (name, before) in all_names {
            let after = chosen.get(name).copied();
            if before == after {
                continue;
            }
            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let package = &catalog[search.package_index[name]];
                    if !package.releases.iter().any(|r| r.version == old) {
                        ChangeReason::Unavailable
                    } else if active
                        .get(name)
                        .map(|cs| cs.iter().any(|c| !constraint_matches(*c, old)))
                        .unwrap_or(false)
                    {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
                (None, None) => continue,
            };
            changes.push(Change {
                package: name.to_string(),
                before,
                after,
                reason,
            });
        }
        Ok(Resolution { selected, changes })
    } else {
        Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }))
    }
}
