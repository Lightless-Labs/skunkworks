```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x
                && if x.major > 0 {
                    version.major == x.major
                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x
                }
        }
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn encode_text(text: &str) -> String {
    let mut result = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            result.push(*byte as char);
        } else {
            result.push('%');
            result.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            result.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    result
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let canonical = canonical_constraints(constraints.to_vec());
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, requirement) in roots.iter().enumerate() {
        if !valid_name(&requirement.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: requirement.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, requirement) in release.deps.iter().enumerate() {
                if !valid_name(&requirement.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: requirement.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut names: Vec<&str> = Vec::new();
    for package in catalog {
        if names.iter().any(|name| *name == package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        names.push(&package.name);
    }

    for package in catalog {
        let mut versions: Vec<Version> = Vec::new();
        for release in &package.releases {
            if versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            versions.push(release.version);
        }
    }
    Ok(())
}

fn active_constraints(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &[(String, Version)],
) -> Vec<(String, Vec<Constraint>)> {
    let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

    fn add(active: &mut Vec<(String, Vec<Constraint>)>, requirement: &Requirement) {
        if let Some((_, constraints)) = active
            .iter_mut()
            .find(|(name, _)| name == &requirement.package)
        {
            constraints.push(requirement.constraint);
        } else {
            active.push((
                requirement.package.clone(),
                vec![requirement.constraint],
            ));
        }
    }

    for requirement in roots {
        add(&mut active, requirement);
    }

    for (name, version) in selected {
        if let Some(package) = catalog.iter().find(|package| package.name == *name) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for requirement in &release.deps {
                    add(&mut active, requirement);
                }
            }
        }
    }

    for (_, constraints) in &mut active {
        constraints.sort();
        constraints.dedup();
    }
    active.sort_by(|a, b| a.0.cmp(&b.0));
    active
}

fn selected_version(selected: &[(String, Version)], package: &str) -> Option<Version> {
    selected
        .iter()
        .find(|(name, _)| name == package)
        .map(|(_, version)| *version)
}

fn inspect_node(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &[(String, Version)],
) -> Result<(Vec<(String, Vec<Constraint>)>, Option<String>), ResolveError> {
    let active = active_constraints(catalog, roots, selected);

    for (package, _) in &active {
        if !catalog.iter().any(|entry| entry.name == *package) {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }
    }

    for (package, constraints) in &active {
        if let Some(version) = selected_version(selected, package) {
            if !constraints.iter().all(|c| matches(version, *c)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (package, constraints) in &active {
        if selected_version(selected, package).is_none() {
            let entry = catalog.iter().find(|entry| entry.name == *package).unwrap();
            if !entry.releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            }) {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let next = active
        .iter()
        .find(|(package, _)| selected_version(selected, package).is_none())
        .map(|(package, _)| package.clone());
    Ok((active, next))
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut Vec<(String, Version)>,
    attempts: &mut usize,
    max_attempts: usize,
    first_failure: &mut Option<ResolveError>,
) -> Result<Option<Vec<Selection>>, ResolveError> {
    let (_, next) = match inspect_node(catalog, roots, selected) {
        Ok(result) => result,
        Err(error) => {
            if first_failure.is_none() {
                *first_failure = Some(error);
            }
            return Ok(None);
        }
    };

    let package_name = match next {
        Some(name) => name,
        None => {
            let mut result: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            result.sort_by(|a, b| a.package.cmp(&b.package));
            return Ok(Some(result));
        }
    };

    let entry = catalog
        .iter()
        .find(|package| package.name == package_name)
        .unwrap();
    let mut candidates: Vec<Version> = entry
        .releases
        .iter()
        .map(|release| release.version)
        .filter(|version| {
            let constraints = inspect_node(catalog, roots, selected)
                .ok()
                .and_then(|(active, _)| {
                    active
                        .into_iter()
                        .find(|(name, _)| name == &package_name)
                        .map(|(_, constraints)| constraints)
                })
                .unwrap_or_default();
            constraints
                .iter()
                .all(|constraint| matches(*version, *constraint))
        })
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.push((package_name.clone(), version));
        match search(
            catalog,
            roots,
            selected,
            attempts,
            max_attempts,
            first_failure,
        ) {
            Ok(Some(solution)) => return Ok(Some(solution)),
            Ok(None) => {
                selected.pop();
            }
            Err(error) => {
                selected.pop();
                return Err(error);
            }
        }
    }

    Ok(None)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;
    validate_duplicates(catalog)?;

    let mut selected = Vec::new();
    let mut attempts = 0;
    let mut first_failure = None;
    match search(
        catalog,
        roots,
        &mut selected,
        &mut attempts,
        max_attempts,
        &mut first_failure,
    )? {
        Some(solution) => Ok(solution),
        None => Err(first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    }
}
```
