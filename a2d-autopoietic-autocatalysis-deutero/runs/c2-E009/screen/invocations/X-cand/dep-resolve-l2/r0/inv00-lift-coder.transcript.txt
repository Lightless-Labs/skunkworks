fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

impl Search<'_> {
    fn inspect(
        &mut self,
        selected: &mut Vec<(String, Version)>,
        roots: &[Requirement],
    ) -> Result<Vec<Selection>, ResolveError> {
        let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

        fn add_constraint(
            active: &mut Vec<(String, Vec<Constraint>)>,
            package: &str,
            constraint: Constraint,
        ) {
            if let Some((_, constraints)) = active.iter_mut().find(|(name, _)| name == package) {
                constraints.push(constraint);
            } else {
                active.push((package.to_owned(), vec![constraint]));
            }
        }

        for root in roots {
            add_constraint(&mut active, &root.package, root.constraint);
        }

        for (selected_name, version) in selected.iter() {
            if let Some(package) = self.catalog.iter().find(|p| p.name == *selected_name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        add_constraint(&mut active, &dep.package, dep.constraint);
                    }
                }
            }
        }

        active.sort_by(|a, b| a.0.cmp(&b.0));
        for (_, constraints) in &mut active {
            *constraints = canonical_constraints(std::mem::take(constraints));
        }

        for (package, _) in &active {
            if !self.catalog.iter().any(|p| p.name == *package) {
                return Err(ResolveError::Missing {
                    package: package.clone(),
                });
            }
        }

        for (package, constraints) in &active {
            if let Some((_, version)) = selected.iter().find(|(name, _)| name == package) {
                if constraints.iter().any(|c| !matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (package, constraints) in &active {
            if selected.iter().any(|(name, _)| name == package) {
                continue;
            }
            let catalog_package = self.catalog.iter().find(|p| p.name == *package).unwrap();
            if !catalog_package
                .releases
                .iter()
                .any(|release| constraints.iter().all(|c| matches(release.version, *c)))
            {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active
            .iter()
            .find(|(package, _)| !selected.iter().any(|(name, _)| name == package))
            .map(|(package, constraints)| (package.clone(), constraints.clone()));

        let Some((package_name, constraints)) = next else {
            let mut result = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect::<Vec<_>>();
            result.sort_by(|a, b| a.package.cmp(&b.package));
            return Ok(result);
        };

        let catalog_package = self.catalog.iter().find(|p| p.name == package_name).unwrap();
        let mut candidates = catalog_package
            .releases
            .iter()
            .filter(|release| constraints.iter().all(|c| matches(release.version, *c)))
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.version.cmp(&a.version));

        let mut first_error: Option<ResolveError> = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.push((package_name.clone(), candidate.version));
            let result = self.inspect(selected, roots);
            selected.pop();

            match result {
                Ok(solution) => return Ok(solution),
                Err(ResolveError::Limit { attempts }) => {
                    return Err(ResolveError::Limit { attempts })
                }
                Err(error) => {
                    if first_error.is_none() {
                        first_error = Some(error);
                    }
                }
            }
        }

        Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: package_name,
            requirements: constraints,
        }))
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = Vec::<String>::new();
    for package in catalog {
        if seen_packages.iter().any(|name| name == &package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        seen_packages.push(package.name.clone());
    }

    for package in catalog {
        let mut seen_versions = Vec::<Version>::new();
        for release in &package.releases {
            if seen_versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            seen_versions.push(release.version);
        }
    }

    if roots.is_empty() {
        return Ok(Vec::new());
    }

    let mut search = Search {
        catalog,
        attempts: 0,
        max_attempts,
    };
    search.inspect(&mut Vec::new(), roots)
}
