fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, requirement) in roots.iter().enumerate() {
        if !valid_name(&requirement.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: requirement.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, requirement) in release.deps.iter().enumerate() {
                if !valid_name(&requirement.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: requirement.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut names: Vec<&str> = Vec::new();
    for package in catalog {
        if names.iter().any(|name| *name == package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        names.push(&package.name);
    }

    for package in catalog {
        let mut versions: Vec<Version> = Vec::new();
        for release in &package.releases {
            if versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            versions.push(release.version);
        }
    }
    Ok(())
}

fn active_constraints(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &[(String, Version)],
) -> Vec<(String, Vec<Constraint>)> {
    let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

    fn add(active: &mut Vec<(String, Vec<Constraint>)>, requirement: &Requirement) {
        if let Some((_, constraints)) = active
            .iter_mut()
            .find(|(name, _)| name == &requirement.package)
        {
            constraints.push(requirement.constraint);
        } else {
            active.push((
                requirement.package.clone(),
                vec![requirement.constraint],
            ));
        }
    }

    for requirement in roots {
        add(&mut active, requirement);
    }

    for (name, version) in selected {
        if let Some(package) = catalog.iter().find(|package| package.name == *name) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for requirement in &release.deps {
                    add(&mut active, requirement);
                }
            }
        }
    }

    for (_, constraints) in &mut active {
        constraints.sort();
        constraints.dedup();
    }
    active.sort_by(|a, b| a.0.cmp(&b.0));
    active
}

fn selected_version(selected: &[(String, Version)], package: &str) -> Option<Version> {
    selected
        .iter()
        .find(|(name, _)| name == package)
        .map(|(_, version)| *version)
}

fn inspect_node(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &[(String, Version)],
) -> Result<(Vec<(String, Vec<Constraint>)>, Option<String>), ResolveError> {
    let active = active_constraints(catalog, roots, selected);

    for (package, _) in &active {
        if !catalog.iter().any(|entry| entry.name == *package) {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }
    }

    for (package, constraints) in &active {
        if let Some(version) = selected_version(selected, package) {
            if !constraints.iter().all(|c| matches(version, *c)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (package, constraints) in &active {
        if selected_version(selected, package).is_none() {
            let entry = catalog.iter().find(|entry| entry.name == *package).unwrap();
            if !entry.releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            }) {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let next = active
        .iter()
        .find(|(package, _)| selected_version(selected, package).is_none())
        .map(|(package, _)| package.clone());
    Ok((active, next))
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut Vec<(String, Version)>,
    attempts: &mut usize,
    max_attempts: usize,
    first_failure: &mut Option<ResolveError>,
) -> Result<Option<Vec<Selection>>, ResolveError> {
    let (_, next) = match inspect_node(catalog, roots, selected) {
        Ok(result) => result,
        Err(error) => {
            if first_failure.is_none() {
                *first_failure = Some(error);
            }
            return Ok(None);
        }
    };

    let package_name = match next {
        Some(name) => name,
        None => {
            let mut result: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            result.sort_by(|a, b| a.package.cmp(&b.package));
            return Ok(Some(result));
        }
    };

    let entry = catalog
        .iter()
        .find(|package| package.name == package_name)
        .unwrap();
    let mut candidates: Vec<Version> = entry
        .releases
        .iter()
        .map(|release| release.version)
        .filter(|version| {
            let constraints = inspect_node(catalog, roots, selected)
                .ok()
                .and_then(|(active, _)| {
                    active
                        .into_iter()
                        .find(|(name, _)| name == &package_name)
                        .map(|(_, constraints)| constraints)
                })
                .unwrap_or_default();
            constraints
                .iter()
                .all(|constraint| matches(*version, *constraint))
        })
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.push((package_name.clone(), version));
        match search(
            catalog,
            roots,
            selected,
            attempts,
            max_attempts,
            first_failure,
        ) {
            Ok(Some(solution)) => return Ok(Some(solution)),
            Ok(None) => {
                selected.pop();
            }
            Err(error) => {
                selected.pop();
                return Err(error);
            }
        }
    }

    Ok(None)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;
    validate_duplicates(catalog)?;

    let mut selected = Vec::new();
    let mut attempts = 0;
    let mut first_failure = None;
    match search(
        catalog,
        roots,
        &mut selected,
        &mut attempts,
        max_attempts,
        &mut first_failure,
    )? {
        Some(solution) => Ok(solution),
        None => Err(first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    }
}

            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => {
            let constraints =
                canonical_constraints(requirements.clone())
                    .iter()
                    .map(|c| constraint_text(*c))
                    .collect::<Vec<_>>()
                    .join(",");
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => {
            let constraints =
                canonical_constraints(requirements.clone())
                    .iter()
                    .map(|c| constraint_text(*c))
                    .collect::<Vec<_>>()
                    .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints
            )
        }
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

struct Search<'a> {
    catalog: &'a [Package],
    attempts: usize,
    max_attempts: usize,
}

impl Search<'_> {
    fn inspect(
        &mut self,
        selected: &mut Vec<(String, Version)>,
        roots: &[Requirement],
    ) -> Result<Vec<Selection>, ResolveError> {
        let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

        fn add_constraint(
            active: &mut Vec<(String, Vec<Constraint>)>,
            package: &str,
            constraint: Constraint,
        ) {
            if let Some((_, constraints)) = active.iter_mut().find(|(name, _)| name == package) {
                constraints.push(constraint);
            } else {
                active.push((package.to_owned(), vec![constraint]));
            }
        }

        for root in roots {
            add_constraint(&mut active, &root.package, root.constraint);
        }

        for (selected_name, version) in selected.iter() {
            if let Some(package) = self.catalog.iter().find(|p| p.name == *selected_name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        add_constraint(&mut active, &dep.package, dep.constraint);
                    }
                }
            }
        }

        active.sort_by(|a, b| a.0.cmp(&b.0));
        for (_, constraints) in &mut active {
            *constraints = canonical_constraints(std::mem::take(constraints));
        }

        for (package, _) in &active {
            if !self.catalog.iter().any(|p| p.name == *package) {
                return Err(ResolveError::Missing {
                    package: package.clone(),
                });
            }
        }

        for (package, constraints) in &active {
            if let Some((_, version)) = selected.iter().find(|(name, _)| name == package) {
                if constraints.iter().any(|c| !matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (package, constraints) in &active {
            if selected.iter().any(|(name, _)| name == package) {
                continue;
            }
            let catalog_package = self.catalog.iter().find(|p| p.name == *package).unwrap();
            if !catalog_package
                .releases
                .iter()
                .any(|release| constraints.iter().all(|c| matches(release.version, *c)))
            {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active
            .iter()
            .find(|(package, _)| !selected.iter().any(|(name, _)| name == package))
            .map(|(package, constraints)| (package.clone(), constraints.clone()));

        let Some((package_name, constraints)) = next else {
            let mut result = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect::<Vec<_>>();
            result.sort_by(|a, b| a.package.cmp(&b.package));
            return Ok(result);
        };

        let catalog_package = self.catalog.iter().find(|p| p.name == package_name).unwrap();
        let mut candidates = catalog_package
            .releases
            .iter()
            .filter(|release| constraints.iter().all(|c| matches(release.version, *c)))
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.version.cmp(&a.version));

        let mut first_error: Option<ResolveError> = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.push((package_name.clone(), candidate.version));
            let result = self.inspect(selected, roots);
            selected.pop();

            match result {
                Ok(solution) => return Ok(solution),
                Err(ResolveError::Limit { attempts }) => {
                    return Err(ResolveError::Limit { attempts })
                }
                Err(error) => {
                    if first_error.is_none() {
                        first_error = Some(error);
                    }
                }
            }
        }

        Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: package_name,
            requirements: constraints,
        }))
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = Vec::<String>::new();
    for package in catalog {
        if seen_packages.iter().any(|name| name == &package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        seen_packages.push(package.name.clone());
    }

    for package in catalog {
        let mut seen_versions = Vec::<Version>::new();
        for release in &package.releases {
            if seen_versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            seen_versions.push(release.version);
        }
    }

    if roots.is_empty() {
        return Ok(Vec::new());
    }

    let mut search = Search {
        catalog,
        attempts: 0,
        max_attempts,
    };
    search.inspect(&mut Vec::new(), roots)
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    let mut out = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for byte in text.bytes() {
        if byte.is_ascii_alphabetic()
            || byte.is_ascii_digit()
            || matches!(byte, b'.' | b'_' | b'-')
        {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {

            format!(
                "invalid-name|site={}|name={}",
                encode_text(site),
                encode_text(name)

        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {

            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, requirement) in roots.iter().enumerate() {
        if !valid_name(&requirement.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: requirement.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn collect_active(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for requirement in roots {
        active
            .entry(requirement.package.clone())
            .or_default()
            .push(requirement.constraint);
    }

    for (name, version) in selected {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dependency in &release.deps {
                    active
                        .entry(dependency.package.clone())
                        .or_default()
                        .push(dependency.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        canonicalize(constraints);
    }
    active
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut Vec<(String, Version)>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let active = collect_active(roots, catalog, selected);

    for name in active.keys() {
        if !catalog.iter().any(|package| package.name == *name) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }
    }

    let mut selected_sorted: Vec<_> = selected.iter().collect();
    selected_sorted.sort_by(|a, b| a.0.cmp(&b.0));
    for (name, version) in selected_sorted {
        if let Some(constraints) = active.get(name) {
            if !constraints
                .iter()
                .all(|constraint| constraint_matches(*constraint, *version))
            {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let selected_names: std::collections::BTreeSet<&str> =
        selected.iter().map(|(name, _)| name.as_str()).collect();

    for (name, constraints) in &active {
        if selected_names.contains(name.as_str()) {
            continue;
        }
        let package = catalog
            .iter()
            .find(|package| package.name == *name)
            .expect("missing packages handled above");
        let has_match = package.releases.iter().any(|release| {
            constraints
                .iter()
                .all(|constraint| constraint_matches(*constraint, release.version))
        });
        if !has_match {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let next = active
        .keys()
        .find(|name| !selected_names.contains(name.as_str()))
        .cloned();

    let Some(name) = next else {
        let mut result: Vec<Selection> = selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();
        result.sort_by(|a, b| a.package.cmp(&b.package));
        return Ok(result);
    };

    let constraints = active.get(&name).expect("active package");
    let package = catalog
        .iter()
        .find(|package| package.name == name)
        .expect("missing packages handled above");

    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|release| {
            constraints
                .iter()
                .all(|constraint| constraint_matches(*constraint, release.version))
        })
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    let mut first_error: Option<ResolveError> = None;
    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.push((name.clone(), version));
        let outcome = search(catalog, roots, selected, attempts, max_attempts);
        selected.pop();

        match outcome {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => return Err(error),
            Err(error) => {
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
        package: name,
        requirements: constraints.clone(),
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    if roots.is_empty() {
        return Ok(Vec::new());
    }

    let mut selected = Vec::new();
    let mut attempts = 0usize;
    search(catalog, roots, &mut selected, &mut attempts, max_attempts)
}
