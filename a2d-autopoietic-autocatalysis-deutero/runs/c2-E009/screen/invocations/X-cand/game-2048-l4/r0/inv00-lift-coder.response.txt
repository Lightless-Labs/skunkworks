#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
    ScoreOverflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate_board(board)?;

    let mut cells = board.cells.clone();
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };
    let line_len = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };

    for line in 0..line_count {
        let mut tiles: Vec<(usize, u32)> = Vec::new();
        for offset in 0..line_len {
            let index = line_index(board, direction, line, offset);
            let value = board.cells[index];
            if value != 0 {
                tiles.push((index, value));
            }
        }

        let mut outputs: Vec<(u32, Option<[usize; 2]>)> = Vec::new();
        let mut i = 0;
        while i < tiles.len() {
            if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
                let doubled = tiles[i].1.checked_mul(2).ok_or(Error::MergeOverflow {
                    first: tiles[i].0,
                    second: tiles[i + 1].0,
                })?;
                outputs.push((doubled, Some([tiles[i].0, tiles[i + 1].0])));
                score = score
                    .checked_add(doubled as u64)
                    .ok_or(Error::ScoreOverflow)?;
                i += 2;
            } else {
                outputs.push((tiles[i].1, None));
                i += 1;
            }
        }

        for offset in 0..line_len {
            let index = line_index(board, direction, line, offset);
            cells[index] = outputs.get(offset).map_or(0, |output| output.0);
        }

        for (offset, (value, from)) in outputs.into_iter().enumerate() {
            if let Some(from) = from {
                merges.push(Merge {
                    from,
                    to: line_index(board, direction, line, offset),
                    value,
                });
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate_board(board)?;
    let mut directions = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        if let Ok(result) = move_board(board, direction) {
            if result.changed {
                directions.push(direction);
            }
        }
    }
    Ok(directions)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Spawn {
    pub index: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate_board(board)?;

    let mut empty_indices = Vec::new();
    for (index, &value) in board.cells.iter().enumerate() {
        if value == 0 {
            empty_indices.push(index);
        }
    }

    if empty_indices.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let a = next_rng(rng);
    let index = empty_indices[(a % empty_indices.len() as u64) as usize];
    let b = next_rng(a);
    let value = if b % 10 == 0 { 4 } else { 2 };

    let mut cells = board.cells.clone();
    cells[index] = value;
    Ok(SpawnResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: moved.changed,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {
    pub board: Board,
    pub rng: u64,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
    pub spawn: Option<Spawn>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub board: Board,
    pub score: u64,
    pub rng: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Game {
    current: State,
    undo: Option<State>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Command {
    Move(Direction),
    Undo,
    Check,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Event {
    Moved(Turn),
    Undone(bool),
    Checked(bool),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Replay {
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReplayError {
    pub index: Option<usize>,
    pub error: Error,
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

pub fn new_game(initial: State) -> Result<Game, Error> {
    validate_board(&initial.board)?;
    Ok(Game {
        current: initial,
        undo: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(&game.current.board, direction)?;
    let new_score = game
        .current
        .score
        .checked_add(moved.score)
        .ok_or(Error::ScoreOverflow)?;

    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, game.current.rng)?;
    let prior = game.current.clone();
    game.current = State {
        board: spawned.board.clone(),
        score: new_score,
        rng: spawned.rng,
    };
    game.undo = Some(prior);

    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

pub fn undo(game: &mut Game) -> bool {
    if let Some(previous) = game.undo.take() {
        game.current = previous;
        true
    } else {
        false
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.undo.is_some()
}

pub fn game_over(game: &Game) -> bool {
    legal_moves(&game.current.board)
        .map(|moves| moves.is_empty())
        .unwrap_or(true)
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };

    let mut events = Vec::new();
    for (index, command) in commands.iter().enumerate() {
        match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => events.push(Event::Moved(turn)),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => events.push(Event::Undone(undo(&mut game))),
            Command::Check => events.push(Event::Checked(game_over(&game))),
        }
    }

    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}

fn validate_board(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    if board.cells.len() != board.width * board.height {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn line_index(board: &Board, direction: Direction, line: usize, offset: usize) -> usize {
    match direction {
        Direction::Left => line * board.width + offset,
        Direction::Right => line * board.width + (board.width - 1 - offset),
        Direction::Up => offset * board.width + line,
        Direction::Down => (board.height - 1 - offset) * board.width + line,
    }
}

fn next_rng(state: u64) -> u64 {
    state
        .wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}
