#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

const TOKENS: [(u32, &str); 13] = [
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
];

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }

    if value == 0 {
        return Ok(match case {
            RomanCase::Upper => "N".to_owned(),
            RomanCase::Lower => "n".to_owned(),
        });
    }

    let mut remaining = value;
    let mut output = String::new();

    for &(amount, token) in &TOKENS {
        while remaining >= amount {
            match case {
                RomanCase::Upper => output.push_str(token),
                RomanCase::Lower => output.push_str(&token.to_ascii_lowercase()),
            }
            remaining -= amount;
        }
    }

    Ok(output)
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !matches!(byte, b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
            | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n')
        {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let first_case = input.bytes().find_map(|byte| match byte {
        b'A'..=b'Z' => Some(true),
        b'a'..=b'z' => Some(false),
        _ => None,
    });

    if let Some(uppercase) = first_case {
        for (index, byte) in input.bytes().enumerate() {
            let is_uppercase = byte.is_ascii_uppercase();
            if is_uppercase != uppercase {
                return Err(RomanError::MixedCase { index });
            }
        }
    }

    if input == "N" || input == "n" {
        return Ok(0);
    }

    if let Some(index) = input.bytes().position(|byte| byte == b'N' || byte == b'n') {
        let _ = index;
        return Err(RomanError::NonCanonical { index: 0 });
    }

    let mut value = 0u32;
    let bytes = input.as_bytes();
    let mut index = 0;

    while index < bytes.len() {
        let current = symbol_value(bytes[index]);
        if index + 1 < bytes.len() {
            let next = symbol_value(bytes[index + 1]);
            if next > current {
                value = value.saturating_add(next - current).min(4000);
                index += 2;
                continue;
            }
        }

        value = value.saturating_add(current).min(4000);
        index += 1;
    }

    if value > 3999 {
        return Err(RomanError::Range);
    }

    let decoded_case = if first_case.unwrap_or(true) {
        RomanCase::Upper
    } else {
        RomanCase::Lower
    };
    let canonical = roman_encode(value, decoded_case)?;

    if input != canonical {
        let differing_index = input
            .bytes()
            .zip(canonical.bytes())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| input.len().min(canonical.len()));
        return Err(RomanError::NonCanonical {
            index: differing_index,
        });
    }

    Ok(value)
}

fn symbol_value(byte: u8) -> u32 {
    match byte.to_ascii_uppercase() {
        b'I' => 1,
        b'V' => 5,
        b'X' => 10,
        b'L' => 50,
        b'C' => 100,
        b'D' => 500,
        b'M' => 1000,
        _ => 0,
    }
}
