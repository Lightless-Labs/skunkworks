#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }

    if value == 0 {
        return Ok(match case {
            RomanCase::Upper => "N".to_string(),
            RomanCase::Lower => "n".to_string(),
        });
    }

    const TOKENS: &[(u32, &str)] = &[
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    let mut remainder = value;
    let mut result = String::new();

    for &(token_value, token) in TOKENS {
        while remainder >= token_value {
            for byte in token.bytes() {
                result.push(match case {
                    RomanCase::Upper => byte as char,
                    RomanCase::Lower => (byte + b'a' - b'A') as char,
                });
            }
            remainder -= token_value;
        }
    }

    Ok(result)
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !matches!(
            byte,
            b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
                | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n'
        ) {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let mut letter_case: Option<bool> = None;
    for (index, byte) in input.bytes().enumerate() {
        let is_lower = byte.is_ascii_lowercase();
        if let Some(first_is_lower) = letter_case {
            if first_is_lower != is_lower {
                return Err(RomanError::MixedCase { index });
            }
        } else {
            letter_case = Some(is_lower);
        }
    }

    if input == "N" || input == "n" {
        return Ok(0);
    }

    if input.bytes().any(|byte| byte == b'N' || byte == b'n') {
        return Err(RomanError::NonCanonical { index: 0 });
    }

    fn symbol_value(byte: u8) -> u32 {
        match byte.to_ascii_uppercase() {
            b'I' => 1,
            b'V' => 5,
            b'X' => 10,
            b'L' => 50,
            b'C' => 100,
            b'D' => 500,
            b'M' => 1000,
            _ => 0,
        }
    }

    let bytes = input.as_bytes();
    let mut index = 0;
    let mut value = 0u32;

    while index < bytes.len() {
        let current = symbol_value(bytes[index]);
        if index + 1 < bytes.len() {
            let next = symbol_value(bytes[index + 1]);
            if next > current {
                value = value.saturating_add(next - current).min(4000);
                index += 2;
                continue;
            }
        }
        value = value.saturating_add(current).min(4000);
        index += 1;
    }

    if value > 3999 {
        return Err(RomanError::Range);
    }

    let case = if letter_case.unwrap_or(false) {
        RomanCase::Lower
    } else {
        RomanCase::Upper
    };
    let canonical = roman_encode(value, case)?;

    let differing_index = canonical
        .bytes()
        .zip(input.bytes())
        .position(|(expected, actual)| expected != actual)
        .unwrap_or_else(|| canonical.len().min(input.len()));

    if canonical != input {
        return Err(RomanError::NonCanonical {
            index: differing_index,
        });
    }

    Ok(value)
}
