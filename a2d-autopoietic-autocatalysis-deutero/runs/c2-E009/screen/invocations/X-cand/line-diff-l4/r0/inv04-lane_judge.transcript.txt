fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let rows = old.len() + 1;
    let cols = new.len() + 1;
    let mut lcs = vec![vec![0usize; cols]; rows];

    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0usize, 0usize, 0usize);
    let mut changes = Vec::new();
    let mut active_start: Option<usize> = None;
    let mut deletes = 0usize;
    let mut inserted = Vec::new();

    while i < old.len() || j < new.len() {
        if i < old.len() && j < new.len() && old[i] == new[j] {
            if let Some(start) = active_start.take() {
                changes.push(Change {
                    start,
                    end: start + deletes,
                    replacement: std::mem::take(&mut inserted),
                });
                deletes = 0;
            }
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        let do_delete = if i == old.len() {
            false
        } else if j == new.len() {
            true
        } else {
            let del_score = lcs[i + 1][j];
            let ins_score = lcs[i][j + 1];
            if del_score > ins_score {
                true
            } else if ins_score > del_score {
                false
            } else {
                (i + j) % 2 == 0
            }
        };

        if active_start.is_none() {
            active_start = Some(base_pos);
        }

        if do_delete {
            i += 1;
            base_pos += 1;
            deletes += 1;
        } else {
            inserted.push(new[j].clone());
            j += 1;
        }
    }

    if let Some(start) = active_start {
        changes.push(Change {
            start,
            end: start + deletes,
            replacement: inserted,
        });
    }

    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;
    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn project(
    base: &[String],
    region_start: usize,
    region_end: usize,
    changes: &[Change],
) -> Vec<String> {
    let mut output = Vec::new();
    let mut cursor = region_start;

    for change in changes {
        while cursor < change.start {
            output.push(base[cursor].clone());
            cursor += 1;
        }
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    while cursor < region_end {
        output.push(base[cursor].clone());
        cursor += 1;
    }
    output
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {

    order.sort_by_key(|&i| {

fn changes(base: &[String], other: &[String], side: MergeSide) -> Vec<Change> {
    let n = base.len();

        (None, 0usize, Vec::<String>::new());

        if i == n && j == m {
            if let Some(start) = run_start.take() {
                result.push(Change {
                    start,
                    end: start + run_deletes,
                    replacement: std::mem::take(&mut run_replacements),

                    start,
                    end: start + run_deletes,
                    replacement: std::mem::take(&mut run_replacements),
                    side: side.clone(),

                run_deletes = 0;
            }
            i += 1;
            j += 1;
            continue;

            i < n
        } else if i == n {
            false
        } else {

            let insert_len = lcs[i][j + 1];

    for change in changes {
        if change.start < cursor || change.start > end {
            continue;
        }
        output.extend(base[cursor..change.start].iter().cloned());
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    output.extend(base[cursor..end].iter().cloned());
    output
}

fn emit_line(text: &mut String, line: &str) {
    text.push_str(line);
    text.push('\n');
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_count = line_count(base);
    let left_count = line_count(left);
    let right_count = line_count(right);
    if base_count > 128 || left_count > 128 || right_count > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_count,
            left_len: left_count,
            right_len: right_count,
        });
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    let mut all_changes = changes(&base_lines, &left_lines, MergeSide::Left);
    all_changes.extend(changes(&base_lines, &right_lines, MergeSide::Right));

    let mut visited = vec![false; all_changes.len()];
    let mut components = Vec::new();

    for seed in 0..all_changes.len() {
        if visited[seed] {
            continue;
        }
        visited[seed] = true;
        let mut indices = vec![seed];
        let mut cursor = 0usize;
        while cursor < indices.len() {
            let current = indices[cursor];
            cursor += 1;
            for candidate in 0..all_changes.len() {
                if !visited[candidate] && interacts(&all_changes[current], &all_changes[candidate]) {
                    visited[candidate] = true;
                    indices.push(candidate);
                }
            }
        }

        let start = indices
            .iter()
            .map(|&i| all_changes[i].start)
            .min()
            .unwrap_or(0);
        let end = indices
            .iter()
            .map(|&i| all_changes[i].end)
            .max()
            .unwrap_or(start);
        let changes = indices
            .into_iter()
            .map(|i| all_changes[i].clone())
            .collect();
        components.push(Component { start, end, changes });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start < a.end).cmp(&(b.start < b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output_lines = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        output_lines.extend(base_lines[base_cursor..component.start].iter().cloned());

        let base_projection = base_lines[component.start..component.end].to_vec();
        let mut left_changes: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == MergeSide::Left)
            .cloned()
            .collect();
        let mut right_changes: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == MergeSide::Right)
            .cloned()
            .collect();
        left_changes.sort_by_key(|c| c.start);
        right_changes.sort_by_key(|c| c.start);

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
        );

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines.len();
                    output_lines.push(format!("<<<<<<< LEFT {}", ordinal));
                    output_lines.extend(left_projection);
                    let base_start_marker = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output_lines.push(format!(
                        "||||||| BASE {},{}",
                        base_start_marker,
                        component.end - component.start
                    ));
                    output_lines.extend(base_projection);
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection);
                    output_lines.push(format!(">>>>>>> RIGHT {}", ordinal));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: output_lines.len() - output_start,
                    });
                }
            }
        }
        base_cursor = component.end;
    }
    output_lines.extend(base_lines[base_cursor..].iter().cloned());

    let mut text = String::new();
    for line in &output_lines {
        emit_line(&mut text, line);
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn canonical_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                lcs[i + 1][j + 1] + 1
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut changes = Vec::new();
    let mut active_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            if let Some(start) = active_start.take() {
                changes.push(Change {
                    start,
                    end: start + deletes,
                    replacement: std::mem::take(&mut inserts),
                });
                deletes = 0;
            }
            i += 1;
            j += 1;
            continue;
        }

        if active_start.is_none() {
            active_start = Some(i);
        }

        if i == n {
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            deletes += 1;
            i += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let choose_delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            };
            if choose_delete {
                deletes += 1;
                i += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    if let Some(start) = active_start {
        changes.push(Change {
            start,
            end: start + deletes,
            replacement: inserts,
        });
    }

    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn build_components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let total = left.len() + right.len();
    let mut parent: Vec<usize> = (0..total).collect();

    fn root(parent: &mut [usize], x: usize) -> usize {
        let mut r = x;
        while parent[r] != r {
            r = parent[r];
        }
        let mut cur = x;
        while parent[cur] != cur {
            let next = parent[cur];
            parent[cur] = r;
            cur = next;
        }
        r
    }

    fn union(parent: &mut [usize], a: usize, b: usize) {
        let ra = root(parent, a);
        let rb = root(parent, b);
        if ra != rb {
            parent[rb] = ra;
        }
    }

    for i in 0..total {
        let (ci, si) = if i < left.len() {
            (&left[i], 0usize)
        } else {
            (&right[i - left.len()], 1usize)
        };
        for j in i + 1..total {
            let (cj, sj) = if j < left.len() {
                (&left[j], 0usize)
            } else {
                (&right[j - left.len()], 1usize)
            };
            if si != sj && interacts(ci, cj) {
                union(&mut parent, i, j);
            }
        }
    }

    let mut groups: Vec<(usize, usize, Vec<usize>, Vec<usize>)> = Vec::new();
    for idx in 0..total {
        let r = root(&mut parent, idx);
        let change = if idx < left.len() {
            &left[idx]
        } else {
            &right[idx - left.len()]
        };
        let pos = groups.iter().position(|g| g.0 == r);
        let gi = match pos {
            Some(p) => p,
            None => {
                groups.push((r, change.start, Vec::new(), Vec::new()));
                groups.len() - 1
            }
        };
        let g = &mut groups[gi];
        g.1 = g.1.min(change.start);
        if idx < left.len() {
            g.2.push(idx);
        } else {
            g.3.push(idx - left.len());
        }
    }

    let mut components = Vec::new();
    for (_, _, left_ids, right_ids) in groups {
        let mut start = usize::MAX;
        let mut end = 0usize;
        for &i in &left_ids {
            start = start.min(left[i].start);
            end = end.max(left[i].end);
        }
        for &i in &right_ids {
            start = start.min(right[i].start);
            end = end.max(right[i].end);
        }
        components.push(Component {
            start,
            end,
            left: left_ids,
            right: right_ids,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.end > a.start).cmp(&(b.end > b.start)))
            .then_with(|| a.end.cmp(&b.end))
    });
    components
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change], ids: &[usize]) -> Vec<String> {
    let mut ordered = ids.to_vec();
    ordered.sort_by_key(|&i| (changes[i].start, changes[i].end));

    let mut out = Vec::new();
    let mut cursor = start;
    for idx in ordered {
        let change = &changes[idx];
        while cursor < change.start {
            out.push(base[cursor].clone());
            cursor += 1;
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    while cursor < end {
        out.push(base[cursor].clone());
        cursor += 1;
    }
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate(base, MergeSide::Base)?;
    validate(left, MergeSide::Left)?;
    validate(right, MergeSide::Right)?;

    let base_len = line_count(base);
    let left_len = line_count(left);
    let right_len = line_count(right);
    if base_len > 128 || left_len > 128 || right_len > 128 {
        return Err(MergeError::TooLarge {
            base_len,
            left_len,
            right_len,
        });
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    let left_changes = canonical_changes(&base_lines, &left_lines);
    let right_changes = canonical_changes(&base_lines, &right_lines);
    let components = build_components(&left_changes, &right_changes);

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0usize;
    let mut ordinal = 0usize;

    for component in components {
        while cursor < component.start {
            output.push(base_lines[cursor].clone());
            cursor += 1;
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
            &component.left,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
            &component.right,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    ordinal += 1;
                    let output_start = output.len();
                    let base_count = component.end - component.start;
                    let base_position = if base_count > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());
                    output.push(format!("||||||| BASE {base_position},{base_count}"));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: base_count,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }
        cursor = component.end;
    }

    while cursor < base_lines.len() {
        output.push(base_lines[cursor].clone());
        cursor += 1;
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn canonical_changes(old: &[String], new: &[String]) -> Vec<Change> {
    let rows = old.len() + 1;
    let cols = new.len() + 1;
    let mut lcs = vec![vec![0usize; cols]; rows];

    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0usize, 0usize, 0usize);
    let mut changes = Vec::new();
    let mut run_start: Option<usize> = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    while i < old.len() || j < new.len() {
        if i < old.len() && j < new.len() && old[i] == new[j] {
            if let Some(start) = run_start.take() {
                changes.push(Change {
                    start,
                    end: start + deletes,
                    replacement: std::mem::take(&mut inserts),
                });
                deletes = 0;
            }
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        let delete = if i == old.len() {
            false
        } else if j == new.len() {
            true
        } else {
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            if delete_len > insert_len {
                true
            } else if insert_len > delete_len {
                false
            } else {
                (i + j) % 2 == 0
            }
        };

        if run_start.is_none() {
            run_start = Some(base_pos);
        }

        if delete {
            i += 1;
            base_pos += 1;
            deletes += 1;
        } else {
            inserts.push(new[j].clone());
            j += 1;
        }
    }

    if let Some(start) = run_start {
        changes.push(Change {
            start,
            end: start + deletes,
            replacement: inserts,
        });
    }

    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;

    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn build_components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let mut all = Vec::new();
    all.extend(left.iter().cloned().map(|change| (change, true)));
    all.extend(right.iter().cloned().map(|change| (change, false)));

    let n = all.len();
    let mut parent: Vec<usize> = (0..n).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            let next = parent[x];
            parent[x] = parent[next];
            x = parent[x];
        }
        x
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if interact(&all[i].0, &all[j].0) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups: Vec<(usize, Component)> = Vec::new();
    for (idx, (change, _)) in all.iter().enumerate() {
        let r = root(&mut parent, idx);
        let group = if let Some(pos) = groups.iter().position(|(key, _)| *key == r) {
            &mut groups[pos].1
        } else {
            groups.push((
                r,
                Component {
                    start: change.start,
                    end: change.end,
                    changes: Vec::new(),
                },
            ));
            &mut groups.last_mut().unwrap().1
        };
        group.start = group.start.min(change.start);
        group.end = group.end.max(change.end);
        group.changes.push(idx);
    }

    let mut components: Vec<Component> = groups.into_iter().map(|(_, c)| c).collect();
    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then_with(|| a.end.cmp(&b.end))
    });
    components
}

fn project(
    base: &[String],
    component: &Component,
    changes: &[Change],
    side_is_left: bool,
    all_left: &[Change],
    all_right: &[Change],
) -> Vec<String> {
    let selected: Vec<&Change> = component
        .changes
        .iter()
        .filter_map(|&idx| {
            if side_is_left {
                all_left.get(idx).or_else(|| {
                    if idx >= all_left.len() {
                        None
                    } else {
                        None
                    }
                })
            } else {
                all_right.get(idx)
            }
        })
        .collect();

    let _ = (changes, selected);
    Vec::new()
}

fn project_side(
    base: &[String],
    component: &Component,
    side_changes: &[Change],
) -> Vec<String> {
    let mut output = Vec::new();
    let mut cursor = component.start;

    for change in side_changes {
        if change.start < component.start || change.end > component.end {
            continue;
        }
        if change.start < cursor {
            continue;
        }
        output.extend(base[cursor..change.start].iter().cloned());
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    output.extend(base[cursor..component.end].iter().cloned());
    output
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let (base_lines, left_lines, right_lines) = validate(base, left, right)?;
    let left_changes = canonical_changes(&base_lines, &left_lines);
    let right_changes = canonical_changes(&base_lines, &right_lines);

    let mut tagged = Vec::new();
    tagged.extend(left_changes.iter().map(|change| (change.clone(), true)));
    tagged.extend(right_changes.iter().map(|change| (change.clone(), false)));

    let n = tagged.len();
    let mut parent: Vec<usize> = (0..n).collect();

    fn find(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            let p = parent[x];
            parent[x] = parent[p];
            x = parent[x];
        }
        x
    }

    for i in 0..n {
        for j in i + 1..n {
            if interact(&tagged[i].0, &tagged[j].0) {
                let a = find(&mut parent, i);
                let b = find(&mut parent, j);
                if a != b {
                    parent[b] = a;
                }
            }
        }
    }

    let mut groups: Vec<(usize, Component)> = Vec::new();
    for (index, (change, _)) in tagged.iter().enumerate() {
        let key = find(&mut parent, index);
        let pos = groups.iter().position(|(group_key, _)| *group_key == key);
        let component = if let Some(pos) = pos {
            &mut groups[pos].1
        } else {
            groups.push((
                key,
                Component {
                    start: change.start,
                    end: change.end,
                    changes: Vec::new(),
                },
            ));
            &mut groups.last_mut().unwrap().1
        };
        component.start = component.start.min(change.start);
        component.end = component.end.max(change.end);
        component.changes.push(index);
    }

    let mut components: Vec<Component> = groups.into_iter().map(|(_, c)| c).collect();
    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then_with(|| a.end.cmp(&b.end))
    });

    let mut output_lines: Vec<String> = Vec::new();
    let mut cursor = 0usize;
    let mut regions = Vec::new();

    for component in components {
        output_lines.extend(base_lines[cursor..component.start].iter().cloned());

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project_side(&base_lines, &component, &left_changes);
        let right_projection = project_side(&base_lines, &component, &right_changes);

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = regions.len() + 1;
                    let output_start = output_lines.len();
                    let base_len = component.end - component.start;
                    let base_display_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output_lines.push(format!("<<<<<<< LEFT {ordinal}"));
                    output_lines.extend(left_projection);
                    output_lines.push(format!(
                        "||||||| BASE {base_display_start},{base_len}"
                    ));
                    output_lines.extend(base_projection);
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection);
                    output_lines.push(format!(">>>>>>> RIGHT {ordinal}"));

                    let output_len = 4
                        + left_projection_len(&left_changes, &right_changes, &component, true, &base_lines)
                        + (component.end - component.start)
                        + left_projection_len(
                            &left_changes,
                            &right_changes,
                            &component,
                            false,
                            &base_lines,
                        );

                    regions.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        cursor = component.end;
    }

    output_lines.extend(base_lines[cursor..].iter().cloned());

    let mut text = String::new();
    for line in &output_lines {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}

fn left_projection_len(
    left: &[Change],
    right: &[Change],
    component: &Component,
    is_left: bool,
    base: &[String],
) -> usize {
    project_side(
        base,
        component,
        if is_left { left } else { right },
    )
    .len()
}
